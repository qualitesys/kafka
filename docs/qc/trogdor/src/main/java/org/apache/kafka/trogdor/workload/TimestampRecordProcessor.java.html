34   /**
35    * This class will process records containing timestamps and generate a histogram based on the data.  It will then be
36    * present in the status from the `ConsumeBenchWorker` class.  This must be used with a timestamped PayloadGenerator
37    * implementation.
38    *
39    * Example spec:
40    * {
41    *    "type": "timestamp",
42    *    "histogramMaxMs": 10000,
43    *    "histogramMinMs": 0,
44    *    "histogramStepMs": 1
45    * }
46    *
47    * This will track total E2E latency up to 10 seconds, using 1ms resolution and a timestamp size of 8 bytes.
48    */
49   
50   public class TimestampRecordProcessor implements RecordProcessor {
51       private final int histogramMaxMs;
52       private final int histogramMinMs;
53       private final int histogramStepMs;
54       private final ByteBuffer buffer;
55       private final Histogram histogram;
56   
57       private final Logger log = LoggerFactory.getLogger(TimestampRecordProcessor.class);
58   
59       final static float[] PERCENTILES = {0.5f, 0.95f, 0.99f};
60   
61       @JsonCreator
62       public TimestampRecordProcessor(@JsonProperty("histogramMaxMs") int histogramMaxMs,
63                                       @JsonProperty("histogramMinMs") int histogramMinMs,
64                                       @JsonProperty("histogramStepMs") int histogramStepMs) {
65           this.histogramMaxMs = histogramMaxMs;
66           this.histogramMinMs = histogramMinMs;
67           this.histogramStepMs = histogramStepMs;
68           this.histogram = new Histogram((histogramMaxMs - histogramMinMs) / histogramStepMs);
69           buffer = ByteBuffer.allocate(Long.BYTES);
70           buffer.order(ByteOrder.LITTLE_ENDIAN);
71       }
72   
73       @JsonProperty
74       public int histogramMaxMs() {
75           return histogramMaxMs;
76       }
77   
78       @JsonProperty
79       public int histogramMinMs() {
80           return histogramMinMs;
81       }
82   
83       @JsonProperty
84       public int histogramStepMs() {
85           return histogramStepMs;
86       }
87   
88       private void putHistogram(long latency) {
89           histogram.add(Long.max(0L, (latency - histogramMinMs) / histogramStepMs));
90       }
91   
92       @Override
93       public synchronized void processRecords(ConsumerRecords<byte[], byte[]> consumerRecords) {
94           // Save the current time to prevent skew by processing time.
95           long curTime = Time.SYSTEM.milliseconds();
96           for (ConsumerRecord<byte[], byte[]> record : consumerRecords) {
97               try {
98                   buffer.clear();
99                   buffer.put(record.value(), 0, Long.BYTES);
100                   buffer.rewind();
101                   putHistogram(curTime - buffer.getLong());
102               } catch (RuntimeException e) {
103                   log.error("Error in processRecords:", e);
104               }
105           }
106       }
107   
108       @Override
109       public JsonNode processorStatus() {
110           Histogram.Summary summary = histogram.summarize(PERCENTILES);
111           StatusData statusData = new StatusData(
112                   summary.average() * histogramStepMs + histogramMinMs,
113                   summary.percentiles().get(0).value() * histogramStepMs + histogramMinMs,
114                   summary.percentiles().get(1).value() * histogramStepMs + histogramMinMs,
115                   summary.percentiles().get(2).value() * histogramStepMs + histogramMinMs);
116           return JsonUtil.JSON_SERDE.valueToTree(statusData);
117       }
118   
119       private static class StatusData {
120           private final float averageLatencyMs;
121           private final int p50LatencyMs;
122           private final int p95LatencyMs;
123           private final int p99LatencyMs;
124   
125           /**
126            * The percentiles to use when calculating the histogram data.
127            * These should match up with the p50LatencyMs, p95LatencyMs, etc. fields.
128            */
129           final static float[] PERCENTILES = {0.5f, 0.95f, 0.99f};
130   
131           @JsonCreator
132           StatusData(@JsonProperty("averageLatencyMs") float averageLatencyMs,
133                      @JsonProperty("p50LatencyMs") int p50latencyMs,
134                      @JsonProperty("p95LatencyMs") int p95latencyMs,
135                      @JsonProperty("p99LatencyMs") int p99latencyMs) {
136               this.averageLatencyMs = averageLatencyMs;
137               this.p50LatencyMs = p50latencyMs;
138               this.p95LatencyMs = p95latencyMs;
139               this.p99LatencyMs = p99latencyMs;
140           }
141   
142           @JsonProperty
143           public float averageLatencyMs() {
144               return averageLatencyMs;
145           }
146   
147           @JsonProperty
148           public int p50LatencyMs() {
149               return p50LatencyMs;
150           }
151   
152           @JsonProperty
153           public int p95LatencyMs() {
154               return p95LatencyMs;
155           }
156   
157           @JsonProperty
158           public int p99LatencyMs() {
159               return p99LatencyMs;
160           }
161       }
162   }
