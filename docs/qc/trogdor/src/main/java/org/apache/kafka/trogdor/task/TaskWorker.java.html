18   package org.apache.kafka.trogdor.task;
19   
20   import org.apache.kafka.common.internals.KafkaFutureImpl;
21   import org.apache.kafka.trogdor.common.Platform;
22   
23   /**
24    * The agent-side interface for implementing tasks.
25    */
26   public interface TaskWorker {
27       /**
28        * Starts the TaskWorker.
29        *
30        * We do not hold any locks or block the WorkerManager state machine on this call.
31        * However, createTask requests to the agent call this function directly.
32        * Therefore, your start() implementation may take a little while, but not too long.
33        * While you can perform short blocking tasks in this function, it is better to
34        * start a background thread to do something time-consuming.
35        *
36        * If the start() function throws an exception, the Agent will assume that the TaskWorker
37        * never started.  Therefore, stop() will never be invoked.  On the other hand, if the
38        * errorFuture is completed, either by a background task or by the start function itself,
39        * the Agent will invoke the stop() method to clean up the worker.
40        *
41        *
42        * @param platform          The platform to use.
43        * @param status            The current status.  The TaskWorker can update
44        *                          this at any time to provide an updated status.
45        * @param haltFuture        A future which the worker should complete if it halts.
46        *                          If it is completed with an empty string, that means the task
47        *                          halted with no error.  Otherwise, the string is treated as the error.
48        *                          If you start a background thread, you may pass haltFuture
49        *                          to that thread.  Then, the thread can use this future to indicate
50        *                          that the worker should be stopped.
51        *
52        * @throws Exception        If the TaskWorker failed to start.  stop() will not be invoked.
53        */
54       void start(Platform platform, WorkerStatusTracker status, KafkaFutureImpl<String> haltFuture)
55           throws Exception;
56   
57       /**
58        * Stops the TaskWorker.
59        *
60        * A TaskWorker may be stopped because it has run for its assigned duration, or because a
61        * request arrived instructing the Agent to stop the worker.  The TaskWorker will
62        * also be stopped if errorFuture was completed to indicate that there was an error.
63        *
64        * Regardless of why the TaskWorker was stopped, the stop() function should release all
65        * resources and stop all threads before returning.  The stop() function can block for
66        * as long as it wants.  It is run in a background thread which will not block other
67        * agent operations.  All tasks will be stopped when the Agent cleanly shuts down.
68        *
69        * @param platform          The platform to use.
70        *
71        * @throws Exception        If there was an error cleaning up the TaskWorker.
72        *                          If there is no existing TaskWorker error, the worker will be
73        *                          treated as having failed with the given error.
74        */
75       void stop(Platform platform) throws Exception;
76   }
