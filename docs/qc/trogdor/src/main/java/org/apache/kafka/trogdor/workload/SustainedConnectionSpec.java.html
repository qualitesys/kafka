29   /**
30    * The specification for a benchmark that creates sustained connections.
31    *
32    * An example JSON representation which will result in a test that creates 27 connections (9 of each), refreshes them
33    * every 10 seconds using 2 threads, running against topic `topic1`, for a duration of 1 hour, and with various other
34    * options set:
35    *
36    * #{@code
37    *   {
38    *    "class": "org.apache.kafka.trogdor.workload.SustainedConnectionSpec",
39    *    "durationMs": 3600000,
40    *    "clientNode": "node0",
41    *    "bootstrapServers": "localhost:9092",
42    *    "commonClientConf": {
43    *      "compression.type": "lz4",
44    *      "auto.offset.reset": "earliest",
45    *      "linger.ms": "100"
46    *    },
47    *    "keyGenerator": {
48    *      "type": "sequential",
49    *      "size": 4,
50    *      "startOffset": 0
51    *    },
52    *    "valueGenerator": {
53    *      "type": "uniformRandom",
54    *      "size": 512,
55    *      "seed": 0,
56    *      "padding": 0
57    *    },
58    *    "producerConnectionCount": 9,
59    *    "consumerConnectionCount": 9,
60    *    "metadataConnectionCount": 9,
61    *    "topicName": "test-topic1-1",
62    *    "numThreads": 2,
63    *    "refreshRateMs": 10000
64    *   }
65    *  }
66    */
67   public class SustainedConnectionSpec extends TaskSpec {
68       private final String clientNode;
69       private final String bootstrapServers;
70       private final Map<String, String> producerConf;
71       private final Map<String, String> consumerConf;
72       private final Map<String, String> adminClientConf;
73       private final Map<String, String> commonClientConf;
74       private final PayloadGenerator keyGenerator;
75       private final PayloadGenerator valueGenerator;
76       private final int producerConnectionCount;
77       private final int consumerConnectionCount;
78       private final int metadataConnectionCount;
79       private final String topicName;
80       private final int numThreads;
81       private final int refreshRateMs;
82   
83       @JsonCreator
84       public SustainedConnectionSpec(
85               @JsonProperty("startMs") long startMs,
86               @JsonProperty("durationMs") long durationMs,
87               @JsonProperty("clientNode") String clientNode,
88               @JsonProperty("bootstrapServers") String bootstrapServers,
89               @JsonProperty("producerConf") Map<String, String> producerConf,
90               @JsonProperty("consumerConf") Map<String, String> consumerConf,
91               @JsonProperty("adminClientConf") Map<String, String> adminClientConf,
92               @JsonProperty("commonClientConf") Map<String, String> commonClientConf,
93               @JsonProperty("keyGenerator") PayloadGenerator keyGenerator,
94               @JsonProperty("valueGenerator") PayloadGenerator valueGenerator,
95               @JsonProperty("producerConnectionCount") int producerConnectionCount,
96               @JsonProperty("consumerConnectionCount") int consumerConnectionCount,
97               @JsonProperty("metadataConnectionCount") int metadataConnectionCount,
98               @JsonProperty("topicName") String topicName,
99               @JsonProperty("numThreads") int numThreads,
100               @JsonProperty("refreshRateMs") int refreshRateMs) {
101           super(startMs, durationMs);
102           this.clientNode = clientNode == null ? "" : clientNode;
103           this.bootstrapServers = (bootstrapServers == null) ? "" : bootstrapServers;
104           this.producerConf = configOrEmptyMap(producerConf);
105           this.consumerConf = configOrEmptyMap(consumerConf);
106           this.adminClientConf = configOrEmptyMap(adminClientConf);
107           this.commonClientConf = configOrEmptyMap(commonClientConf);
108           this.keyGenerator = keyGenerator;
109           this.valueGenerator = valueGenerator;
110           this.producerConnectionCount = producerConnectionCount;
111           this.consumerConnectionCount = consumerConnectionCount;
112           this.metadataConnectionCount = metadataConnectionCount;
113           this.topicName = topicName;
114           this.numThreads = numThreads < 1 ? 1 : numThreads;
115           this.refreshRateMs = refreshRateMs < 1 ? 1 : refreshRateMs;
116       }
