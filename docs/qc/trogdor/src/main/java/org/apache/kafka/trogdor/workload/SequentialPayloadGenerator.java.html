17   package org.apache.kafka.trogdor.workload;
18   
19   import com.fasterxml.jackson.annotation.JsonCreator;
20   import com.fasterxml.jackson.annotation.JsonProperty;
21   
22   import java.nio.ByteBuffer;
23   import java.nio.ByteOrder;
24   
25   /**
26    * A PayloadGenerator which generates a sequentially increasing payload.
27    *
28    * The generated number will wrap around to 0 after the maximum value is reached.
29    * Payloads bigger than 8 bytes will always just be padded with zeros after byte 8.
30    */
31   public class SequentialPayloadGenerator implements PayloadGenerator {
32       private final int size;
33       private final long startOffset;
34       private final ByteBuffer buf;
35   
36       @JsonCreator
37       public SequentialPayloadGenerator(@JsonProperty("size") int size,
38                                         @JsonProperty("offset") long startOffset) {
39           this.size = size;
40           this.startOffset = startOffset;
41           this.buf = ByteBuffer.allocate(8);
42           // Little-endian byte order allows us to support arbitrary lengths more easily,
43           // since the first byte is always the lowest-order byte.
44           this.buf.order(ByteOrder.LITTLE_ENDIAN);
45       }
46   
47       @JsonProperty
48       public int size() {
49           return size;
50       }
51   
52       @JsonProperty
53       public long startOffset() {
54           return startOffset;
55       }
56   
57       @Override
58       public synchronized byte[] generate(long position) {
59           buf.clear();
60           buf.putLong(position + startOffset);
61           byte[] result = new byte[size];
62           System.arraycopy(buf.array(), 0, result, 0, Math.min(buf.array().length, result.length));
63           return result;
64       }
65   }
