56   /**
57    * A client for the Trogdor agent.
58    */
59   public class AgentClient {
60       private final Logger log;
61   
62       /**
63        * The maximum number of tries to make.
64        */
65       private final int maxTries;
66   
67       /**
68        * The URL target.
69        */
70       private final String target;
71   
72       public static class Builder {
73           private Logger log = LoggerFactory.getLogger(AgentClient.class);
74           private int maxTries = 1;
75           private String target = null;
76   
77           public Builder() {
78           }
79   
80           public Builder log(Logger log) {
81               this.log = log;
82               return this;
83           }
84   
85           public Builder maxTries(int maxTries) {
86               this.maxTries = maxTries;
87               return this;
88           }
89   
90           public Builder target(String target) {
91               this.target = target;
92               return this;
93           }
94   
95           public Builder target(String host, int port) {
96               this.target = String.format("%s:%d", host, port);
97               return this;
98           }
99   
100           public AgentClient build() {
101               if (target == null) {
102                   throw new RuntimeException("You must specify a target.");
103               }
104               return new AgentClient(log, maxTries, target);
105           }
106       }
107   
108       private AgentClient(Logger log, int maxTries, String target) {
109           this.log = log;
110           this.maxTries = maxTries;
111           this.target = target;
112       }
113   
114       public String target() {
115           return target;
116       }
117   
118       public int maxTries() {
119           return maxTries;
120       }
121   
122       private String url(String suffix) {
123           return String.format("http://%s%s", target, suffix);
124       }
125   
126       public AgentStatusResponse status() throws Exception {
127           HttpResponse<AgentStatusResponse> resp =
128               JsonRestServer.<AgentStatusResponse>httpRequest(url("/agent/status"), "GET",
129                   null, new TypeReference<AgentStatusResponse>() { }, maxTries);
130           return resp.body();
131       }
132   
133       public UptimeResponse uptime() throws Exception {
134           HttpResponse<UptimeResponse> resp =
135               JsonRestServer.httpRequest(url("/agent/uptime"), "GET",
136                   null, new TypeReference<UptimeResponse>() { }, maxTries);
137           return resp.body();
138       }
139   
140       public void createWorker(CreateWorkerRequest request) throws Exception {
141           HttpResponse<Empty> resp =
142               JsonRestServer.<Empty>httpRequest(
143                   url("/agent/worker/create"), "POST",
144                   request, new TypeReference<Empty>() { }, maxTries);
145           resp.body();
146       }
147   
148       public void stopWorker(StopWorkerRequest request) throws Exception {
149           HttpResponse<Empty> resp =
150               JsonRestServer.<Empty>httpRequest(url(
151                   "/agent/worker/stop"), "PUT",
152                   request, new TypeReference<Empty>() { }, maxTries);
153           resp.body();
154       }
155   
156       public void destroyWorker(DestroyWorkerRequest request) throws Exception {
157           UriBuilder uriBuilder = UriBuilder.fromPath(url("/agent/worker"));
158           uriBuilder.queryParam("workerId", request.workerId());
159           HttpResponse<Empty> resp =
160               JsonRestServer.<Empty>httpRequest(uriBuilder.build().toString(), "DELETE",
161                   null, new TypeReference<Empty>() { }, maxTries);
162           resp.body();
163       }
164   
165       public void invokeShutdown() throws Exception {
166           HttpResponse<Empty> resp =
167               JsonRestServer.<Empty>httpRequest(url(
168                   "/agent/shutdown"), "PUT",
169                   null, new TypeReference<Empty>() { }, maxTries);
170           resp.body();
171       }
172   
173       private static void addTargetArgument(ArgumentParser parser) {
174           parser.addArgument("--target", "-t")
175               .action(store())
176               .required(true)
177               .type(String.class)
178               .dest("target")
179               .metavar("TARGET")
180               .help("A colon-separated host and port pair.  For example, example.com:8888");
181       }
182   
183       private static void addJsonArgument(ArgumentParser parser) {
184           parser.addArgument("--json")
185               .action(storeTrue())
186               .dest("json")
187               .metavar("JSON")
188               .help("Show the full response as JSON.");
189       }
190   
191       private static void addWorkerIdArgument(ArgumentParser parser, String help) {
192           parser.addArgument("--workerId")
193               .action(storeTrue())
194               .type(Long.class)
195               .dest("workerId")
196               .metavar("WORKER_ID")
197               .help(help);
198       }
199   
200       public static void main(String[] args) throws Exception {
201           ArgumentParser rootParser = ArgumentParsers
202               .newArgumentParser("trogdor-agent-client")
203               .defaultHelp(true)
204               .description("The Trogdor agent client.");
205           Subparsers subParsers = rootParser.addSubparsers().
206               dest("command");
207           Subparser uptimeParser = subParsers.addParser("uptime")
208               .help("Get the agent uptime.");
209           addTargetArgument(uptimeParser);
210           addJsonArgument(uptimeParser);
211           Subparser statusParser = subParsers.addParser("status")
212               .help("Get the agent status.");
213           addTargetArgument(statusParser);
214           addJsonArgument(statusParser);
215           Subparser createWorkerParser = subParsers.addParser("createWorker")
216               .help("Create a new worker.");
217           addTargetArgument(createWorkerParser);
218           addWorkerIdArgument(createWorkerParser, "The worker ID to create.");
219           createWorkerParser.addArgument("--taskId")
220               .action(store())
221               .required(true)
222               .type(String.class)
223               .dest("taskId")
224               .metavar("TASK_ID")
225               .help("The task ID to create.");
226           createWorkerParser.addArgument("--spec", "-s")
227               .action(store())
228               .required(true)
229               .type(String.class)
230               .dest("taskSpec")
231               .metavar("TASK_SPEC")
232               .help("The task spec to create, or a path to a file containing the task spec.");
233           Subparser stopWorkerParser = subParsers.addParser("stopWorker")
234               .help("Stop a worker.");
235           addTargetArgument(stopWorkerParser);
236           addWorkerIdArgument(stopWorkerParser, "The worker ID to stop.");
237           Subparser destroyWorkerParser = subParsers.addParser("destroyWorker")
238               .help("Destroy a worker.");
239           addTargetArgument(destroyWorkerParser);
240           addWorkerIdArgument(destroyWorkerParser, "The worker ID to destroy.");
241           Subparser shutdownParser = subParsers.addParser("shutdown")
242               .help("Shut down the agent.");
243           addTargetArgument(shutdownParser);
244   
245           Namespace res = rootParser.parseArgsOrFail(args);
246           String target = res.getString("target");
247           AgentClient client = new Builder().
248               maxTries(3).
249               target(target).
250               build();
251           ZoneOffset localOffset = OffsetDateTime.now().getOffset();
252           switch (res.getString("command")) {
253               case "uptime": {
254                   UptimeResponse uptime = client.uptime();
255                   if (res.getBoolean("json")) {
256                       System.out.println(JsonUtil.toJsonString(uptime));
257                   } else {
258                       System.out.printf("Agent is running at %s.%n", target);
259                       System.out.printf("\tStart time: %s%n",
260                           dateString(uptime.serverStartMs(), localOffset));
261                       System.out.printf("\tCurrent server time: %s%n",
262                           dateString(uptime.nowMs(), localOffset));
263                       System.out.printf("\tUptime: %s%n",
264                           durationString(uptime.nowMs() - uptime.serverStartMs()));
265                   }
266                   break;
267               }
268               case "status": {
269                   AgentStatusResponse status = client.status();
270                   if (res.getBoolean("json")) {
271                       System.out.println(JsonUtil.toJsonString(status));
272                   } else {
273                       System.out.printf("Agent is running at %s.%n", target);
274                       System.out.printf("\tStart time: %s%n",
275                           dateString(status.serverStartMs(), localOffset));
276                       List<List<String>> lines = new ArrayList<>();
277                       List<String> header = new ArrayList<>(
278                           Arrays.asList("WORKER_ID", "TASK_ID", "STATE", "TASK_TYPE"));
279                       lines.add(header);
280                       for (Map.Entry<Long, WorkerState> entry : status.workers().entrySet()) {
281                           List<String> cols = new ArrayList<>();
282                           cols.add(Long.toString(entry.getKey()));
283                           cols.add(entry.getValue().taskId());
284                           cols.add(entry.getValue().getClass().getSimpleName());
285                           cols.add(entry.getValue().spec().getClass().getCanonicalName());
286                           lines.add(cols);
287                       }
288                       System.out.print(StringFormatter.prettyPrintGrid(lines));
289                   }
290                   break;
291               }
292               case "createWorker": {
293                   long workerId = res.getLong("workerId");
294                   String taskId = res.getString("taskId");
295                   TaskSpec taskSpec = JsonUtil.
296                       objectFromCommandLineArgument(res.getString("taskSpec"), TaskSpec.class);
297                   CreateWorkerRequest req =
298                       new CreateWorkerRequest(workerId, taskId, taskSpec);
299                   client.createWorker(req);
300                   System.out.printf("Sent CreateWorkerRequest for worker %d%n.", req.workerId());
301                   break;
302               }
303               case "stopWorker": {
304                   long workerId = res.getLong("workerId");
305                   client.stopWorker(new StopWorkerRequest(workerId));
306                   System.out.printf("Sent StopWorkerRequest for worker %d%n.", workerId);
307                   break;
308               }
309               case "destroyWorker": {
310                   long workerId = res.getLong("workerId");
311                   client.destroyWorker(new DestroyWorkerRequest(workerId));
312                   System.out.printf("Sent DestroyWorkerRequest for worker %d%n.", workerId);
313                   break;
314               }
315               case "shutdown": {
316                   client.invokeShutdown();
317                   System.out.println("Sent ShutdownRequest.");
318                   break;
319               }
320               default: {
321                   System.out.println("You must choose an action. Type --help for help.");
322                   Exit.exit(1);
323               }
324           }
325       }
326   }
