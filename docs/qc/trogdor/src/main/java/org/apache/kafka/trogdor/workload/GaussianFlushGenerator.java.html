18   package org.apache.kafka.trogdor.workload;
19   
20   import com.fasterxml.jackson.annotation.JsonCreator;
21   import com.fasterxml.jackson.annotation.JsonProperty;
22   import org.apache.kafka.clients.producer.KafkaProducer;
23   import org.apache.kafka.common.errors.InterruptException;
24   import java.util.Random;
25   
26   /**
27    * This generator will flush the producer after a specific number of messages, determined by a gaussian distribution.
28    * This is useful to simulate a specific number of messages in a batch regardless of the message size, since batch
29    * flushing is not exposed in the KafkaProducer.
30    *
31    * WARNING: This does not directly control when KafkaProducer will batch, this only makes best effort.  This also
32    * cannot tell when a KafkaProducer batch is closed.  If the KafkaProducer sends a batch before this executes, this
33    * will continue to execute on its own cadence.  To alleviate this, make sure to set `linger.ms` to allow for messages
34    * to be generated up to your upper limit threshold, and make sure to set `batch.size` to allow for all these messages.
35    *
36    * Here is an example spec:
37    *
38    * {
39    *    "type": "gaussian",
40    *    "messagesPerFlushAverage": 16,
41    *    "messagesPerFlushDeviation": 4
42    * }
43    *
44    * This example will flush the producer on average every 16 messages, assuming `linger.ms` and `batch.size` allow for
45    * it.  That average changes based on a normal distribution after each flush:
46    *
47    *    An average of the flushes will be at 16 messages.
48    *    ~68% of the flushes are at between 12 and 20 messages.
49    *    ~95% of the flushes are at between 8 and 24 messages.
50    *    ~99% of the flushes are at between 4 and 28 messages.
51    */
52   
53   public class GaussianFlushGenerator implements FlushGenerator {
54       private final int messagesPerFlushAverage;
55       private final double messagesPerFlushDeviation;
56   
57       private final Random random = new Random();
58   
59       private int messageTracker = 0;
60       private int flushSize = 0;
61   
62       @JsonCreator
63       public GaussianFlushGenerator(@JsonProperty("messagesPerFlushAverage") int messagesPerFlushAverage,
64                                     @JsonProperty("messagesPerFlushDeviation") double messagesPerFlushDeviation) {
65           this.messagesPerFlushAverage = messagesPerFlushAverage;
66           this.messagesPerFlushDeviation = messagesPerFlushDeviation;
67           calculateFlushSize();
68       }
69   
70       @JsonProperty
71       public int messagesPerFlushAverage() {
72           return messagesPerFlushAverage;
73       }
74   
75       @JsonProperty
76       public double messagesPerFlushDeviation() {
77           return messagesPerFlushDeviation;
78       }
79   
80       private synchronized void calculateFlushSize() {
81           flushSize = Math.max((int) (random.nextGaussian() * messagesPerFlushDeviation) + messagesPerFlushAverage, 1);
82           messageTracker = 0;
83       }
84   
85       @Override
86       public synchronized <K, V> void increment(KafkaProducer<K, V> producer) {
87           // Increment the message tracker.
88           messageTracker += 1;
89   
90           // Compare the tracked message count with the throttle limits.
91           if (messageTracker >= flushSize) {
92               try {
93                   producer.flush();
94               } catch (InterruptException e) {
95                   // Ignore flush interruption exceptions.
96               }
97               calculateFlushSize();
98           }
99       }
100   }
