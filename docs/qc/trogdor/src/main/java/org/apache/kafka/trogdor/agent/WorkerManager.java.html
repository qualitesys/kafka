101       /**
102        * The shutdown manager handles shutting down gracefully.
103        *
104        * We can shut down gracefully only when all the references handed out
105        * by the ShutdownManager has been closed, and the shutdown bit has
106        * been set.  RPC operations hold a reference for the duration of their
107        * execution, and so do Workers which have not been shut down.
108        * This prevents us from shutting down in the middle of an RPC, or with
109        * workers which are still running.
110        */
111       static class ShutdownManager {
112           private boolean shutdown = false;
113           private long refCount = 0;
114   
115           class Reference implements AutoCloseable {
116               AtomicBoolean closed = new AtomicBoolean(false);
117   
118               @Override
119               public void close() {
120                   if (closed.compareAndSet(false, true)) {
121                       synchronized (ShutdownManager.this) {
122                           refCount--;
123                           if (shutdown && (refCount == 0)) {
124                               ShutdownManager.this.notifyAll();
125                           }
126                       }
127                   }
128               }
129           }
130   
131           synchronized Reference takeReference() {
132               if (shutdown) {
133                   throw new KafkaException("WorkerManager is shut down.");
134               }
135               refCount++;
136               return new Reference();
137           }
138   
139           synchronized boolean shutdown() {
140               if (shutdown) {
141                   return false;
142               }
143               shutdown = true;
144               if (refCount == 0) {
145                   this.notifyAll();
146               }
147               return true;
148           }
149   
150           synchronized void waitForQuiescence() throws InterruptedException {
151               while ((!shutdown) || (refCount > 0)) {
152                   this.wait();
153               }
154           }
155       }

167           this.shutdownExecutor = Executors.newScheduledThreadPool(0,
168               ThreadUtils.createThreadFactory("WorkerManagerShutdownThread%d", false));
169       }

318       public KafkaFuture<String> createWorker(long workerId, String taskId, TaskSpec spec) throws Throwable {
319           try (ShutdownManager.Reference ref = shutdownManager.takeReference()) {
320               final Worker worker = stateChangeExecutor.
321                   submit(new CreateWorker(workerId, taskId, spec, time.milliseconds())).get();
322               if (worker.doneFuture != null) {
323                   log.info("{}: Ignoring request to create worker {}, because there is already " +
324                       "a worker with that id.", nodeName, workerId);
325                   return worker.doneFuture;
326               }
327               worker.doneFuture = new KafkaFutureImpl<>();
328               if (worker.spec.endMs() <= time.milliseconds()) {
329                   log.info("{}: Will not run worker {} as it has expired.", nodeName, worker);
330                   stateChangeExecutor.submit(new HandleWorkerHalting(worker,
331                       "worker expired", true));
332                   return worker.doneFuture;
333               }
334               KafkaFutureImpl<String> haltFuture = new KafkaFutureImpl<>();
335               haltFuture.thenApply((KafkaFuture.BaseFunction<String, Void>) errorString -> {
336                   if (errorString == null)
337                       errorString = "";
338                   if (errorString.isEmpty()) {
339                       log.info("{}: Worker {} is halting.", nodeName, worker);
340                   } else {
341                       log.info("{}: Worker {} is halting with error {}",
342                           nodeName, worker, errorString);
343                   }
344                   stateChangeExecutor.submit(
345                       new HandleWorkerHalting(worker, errorString, false));
346                   return null;
347               });
348               try {
349                   worker.taskWorker.start(platform, worker.status, haltFuture);
350               } catch (Exception e) {
351                   log.info("{}: Worker {} start() exception", nodeName, worker, e);
352                   stateChangeExecutor.submit(new HandleWorkerHalting(worker,
353                       "worker.start() exception: " + Utils.stackTrace(e), true));
354               }
355               stateChangeExecutor.submit(new FinishCreatingWorker(worker));
356               return worker.doneFuture;
357           } catch (ExecutionException e) {
358               if (e.getCause() instanceof RequestConflictException) {
359                   log.info("{}: request conflict while creating worker {} for task {} with spec {}.",
360                       nodeName, workerId, taskId, spec);
361               } else {
362                   log.info("{}: Error creating worker {} for task {} with spec {}",
363                       nodeName, workerId, taskId, spec, e);
364               }
365               throw e.getCause();
366           }
367       }

369       /**
370        * Handles a request to create a new worker.  Processed by the state change thread.
371        */
372       class CreateWorker implements Callable<Worker> {
373           private final long workerId;
374           private final String taskId;
375           private final TaskSpec spec;
376           private final long now;
377   
378           CreateWorker(long workerId, String taskId, TaskSpec spec, long now) {
379               this.workerId = workerId;
380               this.taskId = taskId;
381               this.spec = spec;
382               this.now = now;
383           }
384   
385           @Override
386           public Worker call() throws Exception {
387               try {
388                   Worker worker = workers.get(workerId);
389                   if (worker != null) {
390                       if (!worker.taskId().equals(taskId)) {
391                           throw new RequestConflictException("There is already a worker ID " + workerId +
392                               " with a different task ID.");
393                       } else if (!worker.spec().equals(spec)) {
394                           throw new RequestConflictException("There is already a worker ID " + workerId +
395                               " with a different task spec.");
396                       } else {
397                           return worker;
398                       }
399                   }
400                   worker = new Worker(workerId, taskId, spec, now);
401                   workers.put(workerId, worker);
402                   log.info("{}: Created worker {} with spec {}", nodeName, worker, spec);
403                   return worker;
404               } catch (Exception e) {
405                   log.info("{}: unable to create worker {} for task {}, with spec {}",
406                       nodeName, workerId, taskId, spec, e);
407                   throw e;
408               }
409           }
410       }

427                           "Transitioning to STOPPING.", nodeName, worker);

437               }

439           }

442       /**
443        * Handles a worker halting.  Processed by the state change thread.
444        */
445       class HandleWorkerHalting implements Callable<Void> {
446           private final Worker worker;
447           private final String failure;
448           private final boolean startupHalt;
449   
450           HandleWorkerHalting(Worker worker, String failure, boolean startupHalt) {
451               this.worker = worker;
452               this.failure = failure;
453               this.startupHalt = startupHalt;
454           }
455   
456           @Override
457           public Void call() throws Exception {
458               if (worker.error.isEmpty()) {
459                   worker.error = failure;
460               }
461               String verb = (worker.error.isEmpty()) ? "halting" :
462                   "halting with error [" + worker.error + "]";
463               switch (worker.state) {
464                   case STARTING:
465                       if (startupHalt) {
466                           log.info("{}: Worker {} {} during startup.  Transitioning to DONE.",
467                               nodeName, worker, verb);
468                           worker.transitionToDone();
469                       } else {
470                           log.info("{}: Worker {} {} during startup.  Transitioning to CANCELLING.",
471                               nodeName, worker, verb);
472                           worker.state = State.CANCELLING;
473                       }
474                       break;
475                   case CANCELLING:
476                       log.info("{}: Cancelling worker {} {}.  ",
477                               nodeName, worker, verb);
478                       break;
479                   case RUNNING:
480                       log.info("{}: Running worker {} {}.  Transitioning to STOPPING.",
481                           nodeName, worker, verb);
482                       worker.transitionToStopping();
483                       break;
484                   case STOPPING:
485                       log.info("{}: Stopping worker {} {}.", nodeName, worker, verb);
486                       break;
487                   case DONE:
488                       log.info("{}: Can't halt worker {} because it is already DONE.",
489                           nodeName, worker);
490                       break;
491               }
492               return null;
493           }
494       }

497        * Transitions a worker to WorkerDone.  Processed by the state change thread.
498        */
499       class CompleteWorker implements Callable<Void> {
500           private final Worker worker;
501   
502           private final String failure;
503   
504           CompleteWorker(Worker worker, String failure) {
505               this.worker = worker;
506               this.failure = failure;
507           }
508   
509           @Override
510           public Void call() throws Exception {
511               if (worker.error.isEmpty() && !failure.isEmpty()) {
512                   worker.error = failure;
513               }
514               worker.transitionToDone();
515               if (worker.mustDestroy) {
516                   log.info("{}: destroying worker {} with error {}",
517                       nodeName, worker, worker.error);
518                   workers.remove(worker.workerId);
519               } else {
520                   log.info("{}: completed worker {} with error {}",
521                       nodeName, worker, worker.error);
522               }
523               return null;
524           }
525       }

544               this.mustDestroy = mustDestroy;
545           }
546   
547           @Override
548           public Void call() throws Exception {
549               Worker worker = workers.get(workerId);
550               if (worker == null) {
551                   log.info("{}: Can't stop worker {} because there is no worker with that ID.",
552                       nodeName, workerId);
553                   return null;
554               }
555               if (mustDestroy) {
556                   worker.mustDestroy = true;
557               }
558               switch (worker.state) {
559                   case STARTING:
560                       log.info("{}: Cancelling worker {} during its startup process.",
561                           nodeName, worker);
562                       worker.state = State.CANCELLING;
563                       break;
564                   case CANCELLING:
565                       log.info("{}: Can't stop worker {}, because it is already being " +
566                           "cancelled.", nodeName, worker);
567                       break;
568                   case RUNNING:
569                       log.info("{}: Stopping running worker {}.", nodeName, worker);
570                       worker.transitionToStopping();
571                       break;
572                   case STOPPING:
573                       log.info("{}: Can't stop worker {}, because it is already " +
574                               "stopping.", nodeName, worker);
575                       break;
576                   case DONE:
577                       if (worker.mustDestroy) {
578                           log.info("{}: destroying worker {} with error {}",
579                               nodeName, worker, worker.error);
580                           workers.remove(worker.workerId);
581                       } else {
582                           log.debug("{}: Can't stop worker {}, because it is already done.",
583                               nodeName, worker);
584                       }
585                       break;
586               }
587               return null;
588           }
589       }

645       class Shutdown implements Callable<Void> {
646           @Override
647           public Void call() throws Exception {
648               log.info("{}: Shutting down WorkerManager.", nodeName);
649               try {
650                   stateChangeExecutor.submit(new DestroyAllWorkers()).get();
651                   log.info("{}: Waiting for shutdownManager quiescence...", nodeName);
652                   shutdownManager.waitForQuiescence();
653                   workerCleanupExecutor.shutdownNow();
654                   stateChangeExecutor.shutdownNow();
655                   log.info("{}: Waiting for workerCleanupExecutor to terminate...", nodeName);
656                   workerCleanupExecutor.awaitTermination(1, TimeUnit.DAYS);
657                   log.info("{}: Waiting for stateChangeExecutor to terminate...", nodeName);
658                   stateChangeExecutor.awaitTermination(1, TimeUnit.DAYS);
659                   log.info("{}: Shutting down shutdownExecutor.", nodeName);
660                   shutdownExecutor.shutdown();
661               } catch (Exception e) {
662                   log.info("{}: Caught exception while shutting down WorkerManager", nodeName, e);
663                   throw e;
664               }
665               return null;
666           }
667       }

669       /**
670        * Begins the process of destroying all workers.  Processed by the state change thread.
671        */
672       class DestroyAllWorkers implements Callable<Void> {
673           @Override
674           public Void call() throws Exception {
675               log.info("{}: Destroying all workers.", nodeName);
676   
677               // StopWorker may remove elements from the set of worker IDs.  That might generate
678               // a ConcurrentModificationException if we were iterating over the worker ID
679               // set directly.  Therefore, we make a copy of the worker IDs here and iterate
680               // over that instead.
681               //
682               // Note that there is no possible way that more worker IDs can be added while this
683               // callable is running, because the state change executor is single-threaded.
684               ArrayList<Long> workerIds = new ArrayList<>(workers.keySet());
685   
686               for (long workerId : workerIds) {
687                   try {
688                       new StopWorker(workerId, true).call();
689                   } catch (Exception e) {
690                       log.error("Failed to stop worker {}", workerId, e);
691                   }
692               }
693               return null;
694           }
695       }
