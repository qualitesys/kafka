63       private final String id;
64

80       @Override
81       public void start(Platform platform, WorkerStatusTracker status,
82                         KafkaFutureImpl<String> doneFuture) {
83           if (!running.compareAndSet(false, true)) {
84               throw new IllegalStateException("ProducerBenchWorker is already running.");
85           }
86           log.info("{}: Activating ProduceBenchWorker with {}", id, spec);
87           // Create an executor with 2 threads.  We need the second thread so
88           // that the StatusUpdater can run in parallel with SendRecords.
89           this.executor = Executors.newScheduledThreadPool(2,
90               ThreadUtils.createThreadFactory("ProduceBenchWorkerThread%d", false));
91           this.status = status;
92           this.doneFuture = doneFuture;
93           executor.submit(new Prepare());
94       }

96       public class Prepare implements Runnable {
97           @Override
98           public void run() {
99               try {
100                   Map<String, NewTopic> newTopics = new HashMap<>();
101                   HashSet<TopicPartition> active = new HashSet<>();
102                   for (Map.Entry<String, PartitionsSpec> entry :
103                           spec.activeTopics().materialize().entrySet()) {
104                       String topicName = entry.getKey();
105                       PartitionsSpec partSpec = entry.getValue();
106                       newTopics.put(topicName, partSpec.newTopic(topicName));
107                       for (Integer partitionNumber : partSpec.partitionNumbers()) {
108                           active.add(new TopicPartition(topicName, partitionNumber));
109                       }
110                   }
111                   if (active.isEmpty()) {
112                       throw new RuntimeException("You must specify at least one active topic.");
113                   }
114                   for (Map.Entry<String, PartitionsSpec> entry :
115                           spec.inactiveTopics().materialize().entrySet()) {
116                       String topicName = entry.getKey();
117                       PartitionsSpec partSpec = entry.getValue();
118                       newTopics.put(topicName, partSpec.newTopic(topicName));
119                   }
120                   status.update(new TextNode("Creating " + newTopics.keySet().size() + " topic(s)"));
121                   WorkerUtils.createTopics(log, spec.bootstrapServers(), spec.commonClientConf(),
122                                            spec.adminClientConf(), newTopics, false);
123                   status.update(new TextNode("Created " + newTopics.keySet().size() + " topic(s)"));
124                   executor.submit(new SendRecords(active));
125               } catch (Throwable e) {
126                   WorkerUtils.abort(log, "Prepare", e, doneFuture);
127               }
128           }
129       }

131       private static class SendRecordsCallback implements Callback {
132           private final SendRecords sendRecords;
133           private final long startMs;
134   
135           SendRecordsCallback(SendRecords sendRecords, long startMs) {
136               this.sendRecords = sendRecords;
137               this.startMs = startMs;
138           }
139   
140           @Override
141           public void onCompletion(RecordMetadata metadata, Exception exception) {
142               long now = Time.SYSTEM.milliseconds();
143               long durationMs = now - startMs;
144               sendRecords.recordDuration(durationMs);
145               if (exception != null) {
146                   log.error("SendRecordsCallback: error", exception);
147               }
148           }
149       }
150   
151       /**
152        * A subclass of Throttle which flushes the Producer right before the throttle injects a delay.
153        * This avoids including throttling latency in latency measurements.
154        */
155       private static class SendRecordsThrottle extends Throttle {
156           private final KafkaProducer<?, ?> producer;
157   
158           SendRecordsThrottle(int maxPerPeriod, KafkaProducer<?, ?> producer) {
159               super(maxPerPeriod, THROTTLE_PERIOD_MS);
160               this.producer = producer;
161           }
162   
163           @Override
164           protected synchronized void delay(long amount) throws InterruptedException {
165               long startMs = time().milliseconds();
166               producer.flush();
167               long endMs = time().milliseconds();
168               long delta = endMs - startMs;
169               super.delay(amount - delta);
170           }
171       }

173       public class SendRecords implements Callable<Void> {
174           private final HashSet<TopicPartition> activePartitions;
175   
176           private final Histogram histogram;
177   
178           private final Future<?> statusUpdaterFuture;
179   
180           private final KafkaProducer<byte[], byte[]> producer;
181   
182           private final PayloadIterator keys;
183   
184           private final PayloadIterator values;
185   
186           private final Optional<TransactionGenerator> transactionGenerator;
187   
188           private final Throttle throttle;
189   
190           private Iterator<TopicPartition> partitionsIterator;
191           private Future<RecordMetadata> sendFuture;
192           private AtomicLong transactionsCommitted;
193           private boolean enableTransactions;
194   
195           SendRecords(HashSet<TopicPartition> activePartitions) {
196               this.activePartitions = activePartitions;
197               this.partitionsIterator = activePartitions.iterator();
198               this.histogram = new Histogram(10000);
199   
200               this.transactionGenerator = spec.transactionGenerator();
201               this.enableTransactions = this.transactionGenerator.isPresent();
202               this.transactionsCommitted = new AtomicLong();
203   
204               int perPeriod = WorkerUtils.perSecToPerPeriod(spec.targetMessagesPerSec(), THROTTLE_PERIOD_MS);
205               this.statusUpdaterFuture = executor.scheduleWithFixedDelay(
206                   new StatusUpdater(histogram, transactionsCommitted), 30, 30, TimeUnit.SECONDS);
207   
208               Properties props = new Properties();
209               props.put(ProducerConfig.BOOTSTRAP_SERVERS_CONFIG, spec.bootstrapServers());
210               if (enableTransactions)
211                   props.put(ProducerConfig.TRANSACTIONAL_ID_CONFIG, "produce-bench-transaction-id-" + UUID.randomUUID());
212               // add common client configs to producer properties, and then user-specified producer configs
213               WorkerUtils.addConfigsToProperties(props, spec.commonClientConf(), spec.producerConf());
214               this.producer = new KafkaProducer<>(props, new ByteArraySerializer(), new ByteArraySerializer());
215               this.keys = new PayloadIterator(spec.keyGenerator());
216               this.values = new PayloadIterator(spec.valueGenerator());
217               if (spec.skipFlush()) {
218                   this.throttle = new Throttle(perPeriod, THROTTLE_PERIOD_MS);
219               } else {
220                   this.throttle = new SendRecordsThrottle(perPeriod, producer);
221               }
222           }
223   
224           @Override
225           public Void call() throws Exception {
226               long startTimeMs = Time.SYSTEM.milliseconds();
227               try {
228                   try {
229                       if (enableTransactions)
230                           producer.initTransactions();
231   
232                       long sentMessages = 0;
233                       while (sentMessages < spec.maxMessages()) {
234                           if (enableTransactions) {
235                               boolean tookAction = takeTransactionAction();
236                               if (tookAction)
237                                   continue;
238                           }
239                           sendMessage();
240                           sentMessages++;
241                       }
242                       if (enableTransactions)
243                           takeTransactionAction(); // give the transactionGenerator a chance to commit if configured evenly
244                   } catch (Exception e) {
245                       if (enableTransactions)
246                           producer.abortTransaction();
247                       throw e;
248                   } finally {
249                       if (sendFuture != null) {
250                           try {
251                               sendFuture.get();
252                           } catch (Exception e) {
253                               log.error("Exception on final future", e);
254                           }
255                       }
256                       producer.close();
257                   }
258               } catch (Exception e) {
259                   WorkerUtils.abort(log, "SendRecords", e, doneFuture);
260               } finally {
261                   statusUpdaterFuture.cancel(false);
262                   StatusData statusData = new StatusUpdater(histogram, transactionsCommitted).update();
263                   long curTimeMs = Time.SYSTEM.milliseconds();
264                   log.info("Sent {} total record(s) in {} ms.  status: {}",
265                       histogram.summarize().numSamples(), curTimeMs - startTimeMs, statusData);
266               }
267               doneFuture.complete("");
268               return null;
269           }
270   
271           private boolean takeTransactionAction() {
272               boolean tookAction = true;
273               TransactionAction nextAction = transactionGenerator.get().nextAction();
274               switch (nextAction) {
275                   case BEGIN_TRANSACTION:
276                       log.debug("Beginning transaction.");
277                       producer.beginTransaction();
278                       break;
279                   case COMMIT_TRANSACTION:
280                       log.debug("Committing transaction.");
281                       producer.commitTransaction();
282                       transactionsCommitted.getAndIncrement();
283                       break;
284                   case ABORT_TRANSACTION:
285                       log.debug("Aborting transaction.");
286                       producer.abortTransaction();
287                       break;
288                   case NO_OP:
289                       tookAction = false;
290                       break;
291               }
292               return tookAction;
293           }
294   
295           private void sendMessage() throws InterruptedException {
296               if (!partitionsIterator.hasNext())
297                   partitionsIterator = activePartitions.iterator();
298   
299               TopicPartition partition = partitionsIterator.next();
300               ProducerRecord<byte[], byte[]> record;
301               if (spec.useConfiguredPartitioner()) {
302                   record = new ProducerRecord<>(
303                       partition.topic(), keys.next(), values.next());
304               } else {
305                   record = new ProducerRecord<>(
306                       partition.topic(), partition.partition(), keys.next(), values.next());
307               }
308               sendFuture = producer.send(record,
309                   new SendRecordsCallback(this, Time.SYSTEM.milliseconds()));
310               throttle.increment();
311           }
312   
313           void recordDuration(long durationMs) {
314               histogram.add(durationMs);
315           }
316       }

318       public class StatusUpdater implements Runnable {
319           private final Histogram histogram;
320           private final AtomicLong transactionsCommitted;
321   
322           StatusUpdater(Histogram histogram, AtomicLong transactionsCommitted) {
323               this.histogram = histogram;
324               this.transactionsCommitted = transactionsCommitted;
325           }
326   
327           @Override
328           public void run() {
329               try {
330                   update();
331               } catch (Exception e) {
332                   WorkerUtils.abort(log, "StatusUpdater", e, doneFuture);
333               }
334           }
335   
336           StatusData update() {
337               Histogram.Summary summary = histogram.summarize(StatusData.PERCENTILES);
338               StatusData statusData = new StatusData(summary.numSamples(), summary.average(),
339                   summary.percentiles().get(0).value(),
340                   summary.percentiles().get(1).value(),
341                   summary.percentiles().get(2).value(),
342                   transactionsCommitted.get());
343               status.update(JsonUtil.JSON_SERDE.valueToTree(statusData));
344               return statusData;
345           }
346       }
