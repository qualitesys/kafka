253           showTaskParser.addArgument("--show-status", "-S")
254               .action(storeTrue())
255               .dest("showStatus")

310           addTargetArgument(destroyTaskParser);
311           destroyTaskParser.addArgument("--id", "-i")

317               .help("The task ID to destroy.");

322           Namespace res = rootParser.parseArgsOrFail(args);
323           String target = res.getString("target");
324           CoordinatorClient client = new Builder().
325               maxTries(3).
326               target(target).
327               build();
328           ZoneOffset localOffset = OffsetDateTime.now().getOffset();
329           switch (res.getString("command")) {
330               case "uptime": {
331                   UptimeResponse uptime = client.uptime();
332                   if (res.getBoolean("json")) {
333                       System.out.println(JsonUtil.toJsonString(uptime));
334                   } else {
335                       System.out.printf("Coordinator is running at %s.%n", target);
336                       System.out.printf("\tStart time: %s%n",
337                           dateString(uptime.serverStartMs(), localOffset));
338                       System.out.printf("\tCurrent server time: %s%n",
339                           dateString(uptime.nowMs(), localOffset));
340                       System.out.printf("\tUptime: %s%n",
341                           durationString(uptime.nowMs() - uptime.serverStartMs()));
342                   }
343                   break;
344               }
345               case "status": {
346                   CoordinatorStatusResponse response = client.status();
347                   if (res.getBoolean("json")) {
348                       System.out.println(JsonUtil.toJsonString(response));
349                   } else {
350                       System.out.printf("Coordinator is running at %s.%n", target);
351                       System.out.printf("\tStart time: %s%n", dateString(response.serverStartMs(), localOffset));
352                   }
353                   break;
354               }
355               case "showTask": {
356                   String taskId = res.getString("taskId");
357                   TaskRequest req = new TaskRequest(taskId);
358                   TaskState taskState = null;
359                   try {
360                       taskState = client.task(req);
361                   } catch (NotFoundException e) {
362                       System.out.printf("Task %s was not found.%n", taskId);
363                       Exit.exit(1);
364                   }
365                   if (res.getBoolean("json")) {
366                       System.out.println(JsonUtil.toJsonString(taskState));
367                   } else {
368                       System.out.printf("Task %s of type %s is %s. %s%n", taskId,
369                           taskState.spec().getClass().getCanonicalName(),
370                           taskState.stateType(), prettyPrintTaskInfo(taskState, localOffset));
371                       if (taskState instanceof TaskDone) {
372                           TaskDone taskDone = (TaskDone) taskState;
373                           if ((taskDone.error() != null) && (!taskDone.error().isEmpty())) {
374                               System.out.printf("Error: %s%n", taskDone.error());
375                           }
376                       }
377                       if (res.getBoolean("verbose")) {
378                           System.out.printf("Spec: %s%n%n", JsonUtil.toPrettyJsonString(taskState.spec()));
379                       }
380                       if (res.getBoolean("verbose") || res.getBoolean("showStatus")) {
381                           System.out.printf("Status: %s%n%n", JsonUtil.toPrettyJsonString(taskState.status()));
382                       }
383                   }
384                   break;
385               }
386               case "showTasks": {
387                   TaskStateType taskStateType = res.<TaskStateType>get("taskStateType");
388                   List<String> taskIds = new ArrayList<>();
389                   Pattern taskIdPattern = null;
390                   if (res.getList("taskIds") != null) {
391                       for (Object taskId : res.getList("taskIds")) {
392                           taskIds.add((String) taskId);
393                       }
394                   } else if (res.getString("taskIdPattern") != null) {
395                       try {
396                           taskIdPattern = Pattern.compile(res.getString("taskIdPattern"));
397                       } catch (PatternSyntaxException e) {
398                           System.out.println("Invalid task ID regular expression " + res.getString("taskIdPattern"));
399                           e.printStackTrace();
400                           Exit.exit(1);
401                       }
402                   }
403                   TasksRequest req = new TasksRequest(taskIds, 0, 0, 0, 0,
404                       Optional.ofNullable(taskStateType));
405                   TasksResponse response = client.tasks(req);
406                   if (taskIdPattern != null) {
407                       TreeMap<String, TaskState> filteredTasks = new TreeMap<>();
408                       for (Map.Entry<String, TaskState> entry : response.tasks().entrySet()) {
409                           if (taskIdPattern.matcher(entry.getKey()).matches()) {
410                               filteredTasks.put(entry.getKey(), entry.getValue());
411                           }
412                       }
413                       response = new TasksResponse(filteredTasks);
414                   }
415                   if (res.getBoolean("json")) {
416                       System.out.println(JsonUtil.toJsonString(response));
417                   } else {
418                       System.out.println(prettyPrintTasksResponse(response, localOffset));
419                   }
420                   if (response.tasks().isEmpty()) {
421                       Exit.exit(1);
422                   }
423                   break;
424               }
425               case "createTask": {
426                   String taskId = res.getString("taskId");
427                   TaskSpec taskSpec = JsonUtil.
428                       objectFromCommandLineArgument(res.getString("taskSpec"), TaskSpec.class);
429                   CreateTaskRequest req = new CreateTaskRequest(taskId, taskSpec);
430                   try {
431                       client.createTask(req);
432                       System.out.printf("Sent CreateTaskRequest for task %s.%n", req.id());
433                   } catch (RequestConflictException rce) {
434                       System.out.printf("CreateTaskRequest for task %s got a 409 status code - " +
435                           "a task with the same ID but a different specification already exists.%nException: %s%n",
436                           req.id(), rce.getMessage());
437                       Exit.exit(1);
438                   }
439                   break;
440               }
441               case "stopTask": {
442                   String taskId = res.getString("taskId");
443                   StopTaskRequest req = new StopTaskRequest(taskId);
444                   client.stopTask(req);
445                   System.out.printf("Sent StopTaskRequest for task %s.%n", taskId);
446                   break;
447               }
448               case "destroyTask": {
449                   String taskId = res.getString("taskId");
450                   DestroyTaskRequest req = new DestroyTaskRequest(taskId);
451                   client.destroyTask(req);
452                   System.out.printf("Sent DestroyTaskRequest for task %s.%n", taskId);
453                   break;
454               }
455               case "shutdown": {
456                   client.shutdown();
457                   System.out.println("Sent ShutdownRequest.");
458                   break;
459               }
460               default: {
461                   System.out.println("You must choose an action. Type --help for help.");
462                   Exit.exit(1);
463               }
464           }
465       }

467       static String prettyPrintTasksResponse(TasksResponse response, ZoneOffset zoneOffset) {
468           if (response.tasks().isEmpty()) {
469               return "No matching tasks found.";
470           }
471           List<List<String>> lines = new ArrayList<>();
472           List<String> header = new ArrayList<>(
473               Arrays.asList("ID", "TYPE", "STATE", "INFO"));
474           lines.add(header);
475           for (Map.Entry<String, TaskState> entry : response.tasks().entrySet()) {
476               String taskId = entry.getKey();
477               TaskState taskState = entry.getValue();
478               List<String> cols = new ArrayList<>();
479               cols.add(taskId);
480               cols.add(taskState.spec().getClass().getCanonicalName());
481               cols.add(taskState.stateType().toString());
482               cols.add(prettyPrintTaskInfo(taskState, zoneOffset));
483               lines.add(cols);
484           }
485           return StringFormatter.prettyPrintGrid(lines);
486       }

488       static String prettyPrintTaskInfo(TaskState taskState, ZoneOffset zoneOffset) {
489           if (taskState instanceof TaskPending) {
490               return "Will start at " + dateString(taskState.spec().startMs(), zoneOffset);
491           } else if (taskState instanceof TaskRunning) {
492               TaskRunning runState = (TaskRunning) taskState;
493               return "Started " + dateString(runState.startedMs(), zoneOffset) +
494                   "; will stop after " + durationString(taskState.spec().durationMs());
495           } else if (taskState instanceof TaskStopping) {
496               TaskStopping stoppingState = (TaskStopping) taskState;
497               return "Started " + dateString(stoppingState.startedMs(), zoneOffset);
498           } else if (taskState instanceof TaskDone) {
499               TaskDone doneState = (TaskDone) taskState;
500               String status = null;
501               if (doneState.error() == null || doneState.error().isEmpty()) {
502                   if (doneState.cancelled()) {
503                       status = "CANCELLED";
504                   } else {
505                       status = "FINISHED";
506                   }
507               } else {
508                   status = "FAILED";
509               }
510               return String.format("%s at %s after %s", status,
511                   dateString(doneState.doneMs(), zoneOffset),
512                   durationString(doneState.doneMs() - doneState.startedMs()));
513           } else {
514               throw new RuntimeException("Unknown task state type " + taskState.stateType());
515           }
516       }
517   }
