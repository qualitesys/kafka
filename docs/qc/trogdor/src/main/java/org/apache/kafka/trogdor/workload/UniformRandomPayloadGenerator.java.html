18   package org.apache.kafka.trogdor.workload;
19   
20   import com.fasterxml.jackson.annotation.JsonCreator;
21   import com.fasterxml.jackson.annotation.JsonProperty;
22   
23   import java.util.Random;
24   
25   /**
26    * A PayloadGenerator which generates a uniform random payload.
27    *
28    * This generator generates pseudo-random payloads that can be reproduced from run to run.
29    * The guarantees are the same as those of java.util.Random.
30    *
31    * This payload generator also has the option to append padding bytes at the end of the payload.
32    * The padding bytes are always the same, no matter what the position is.  This is useful when
33    * simulating a partly-compressible stream of user data.
34    */
35   public class UniformRandomPayloadGenerator implements PayloadGenerator {
36       private final int size;
37       private final long seed;
38       private final int padding;
39       private final Random random = new Random();
40       private final byte[] padBytes;
41       private final byte[] randomBytes;
42   
43       @JsonCreator
44       public UniformRandomPayloadGenerator(@JsonProperty("size") int size,
45                                            @JsonProperty("seed") long seed,
46                                            @JsonProperty("padding") int padding) {
47           this.size = size;
48           this.seed = seed;
49           this.padding = padding;
50           if (padding < 0 || padding > size) {
51               throw new RuntimeException("Invalid value " + padding + " for " +
52                   "padding: the number of padding bytes must not be smaller than " +
53                   "0 or greater than the total payload size.");
54           }
55           this.padBytes = new byte[padding];
56           random.setSeed(seed);
57           random.nextBytes(padBytes);
58           this.randomBytes = new byte[size - padding];
59       }
60   
61       @JsonProperty
62       public int size() {
63           return size;
64       }
65   
66       @JsonProperty
67       public long seed() {
68           return seed;
69       }
70   
71       @JsonProperty
72       public int padding() {
73           return padding;
74       }
75   
76       @Override
77       public synchronized byte[] generate(long position) {
78           byte[] result = new byte[size];
79           if (randomBytes.length > 0) {
80               random.setSeed(seed + position);
81               random.nextBytes(randomBytes);
82               System.arraycopy(randomBytes, 0, result, 0, Math.min(randomBytes.length, result.length));
83           }
84           if (padBytes.length > 0) {
85               System.arraycopy(padBytes, 0, result, randomBytes.length, result.length - randomBytes.length);
86           }
87           return result;
88       }
89   }
