50   /**
51    * The REST resource for the Coordinator. This describes the RPCs which the coordinator
52    * can accept.
53    *
54    * RPCs should be idempotent.  This is important because if the server's response is
55    * lost, the client will simply retransmit the same request. The server's response must
56    * be the same the second time around.
57    *
58    * We return the empty JSON object {} rather than void for RPCs that have no results.
59    * This ensures that if we want to add more return results later, we can do so in a
60    * compatible way.
61    */
62   @Path("/coordinator")
63   @Produces(MediaType.APPLICATION_JSON)
64   @Consumes(MediaType.APPLICATION_JSON)
65   public class CoordinatorRestResource {
66       private final AtomicReference<Coordinator> coordinator = new AtomicReference<Coordinator>();
67   
68       @javax.ws.rs.core.Context
69       private ServletContext context;
70   
71       public void setCoordinator(Coordinator myCoordinator) {
72           coordinator.set(myCoordinator);
73       }
74   
75       @GET
76       @Path("/status")
77       public CoordinatorStatusResponse status() throws Throwable {
78           return coordinator().status();
79       }
80   
81       @GET
82       @Path("/uptime")
83       public UptimeResponse uptime() {
84           return coordinator().uptime();
85       }
86   
87       @POST
88       @Path("/task/create")
89       public Empty createTask(CreateTaskRequest request) throws Throwable {
90           coordinator().createTask(request);
91           return Empty.INSTANCE;
92       }
93   
94       @PUT
95       @Path("/task/stop")
96       public Empty stopTask(StopTaskRequest request) throws Throwable {
97           coordinator().stopTask(request);
98           return Empty.INSTANCE;
99       }
100   
101       @DELETE
102       @Path("/tasks")
103       public Empty destroyTask(@DefaultValue("") @QueryParam("taskId") String taskId) throws Throwable {
104           coordinator().destroyTask(new DestroyTaskRequest(taskId));
105           return Empty.INSTANCE;
106       }
107   
108       @GET
109       @Path("/tasks/")
110       public Response tasks(@QueryParam("taskId") List<String> taskId,
111               @DefaultValue("0") @QueryParam("firstStartMs") long firstStartMs,
112               @DefaultValue("0") @QueryParam("lastStartMs") long lastStartMs,
113               @DefaultValue("0") @QueryParam("firstEndMs") long firstEndMs,
114               @DefaultValue("0") @QueryParam("lastEndMs") long lastEndMs,
115               @DefaultValue("") @QueryParam("state") String state) throws Throwable {
116           boolean isEmptyState = state.equals("");
117           if (!isEmptyState && !TaskStateType.Constants.VALUES.contains(state)) {
118               return Response.status(400).entity(
119                   String.format("State %s is invalid. Must be one of %s",
120                       state, TaskStateType.Constants.VALUES)
121               ).build();
122           }
123   
124           Optional<TaskStateType> givenState = Optional.ofNullable(isEmptyState ? null : TaskStateType.valueOf(state));
125           TasksResponse resp = coordinator().tasks(new TasksRequest(taskId, firstStartMs, lastStartMs, firstEndMs, lastEndMs, givenState));
126   
127           return Response.status(200).entity(resp).build();
128       }
129   
130       @GET
131       @Path("/tasks/{taskId}")
132       public TaskState tasks(@PathParam("taskId") String taskId) throws Throwable {
133           TaskState response = coordinator().task(new TaskRequest(taskId));
134           if (response == null)
135               throw new NotFoundException(String.format("No task with ID \"%s\" exists.", taskId));
136   
137           return response;
138       }
139   
140       @PUT
141       @Path("/shutdown")
142       public Empty beginShutdown(CoordinatorShutdownRequest request) throws Throwable {
143           coordinator().beginShutdown(request.stopAgents());
144           return Empty.INSTANCE;
145       }
146   
147       private Coordinator coordinator() {
148           Coordinator myCoordinator = coordinator.get();
149           if (myCoordinator == null) {
150               throw new RuntimeException("CoordinatorRestResource has not been initialized yet.");
151           }
152           return myCoordinator;
153       }
154   }
