20   import java.time.Duration;
21   import java.time.ZoneOffset;
22   import java.time.format.DateTimeFormatter;
23   import java.util.ArrayList;
24   import java.util.Date;
25   import java.util.List;
26   
27   /**
28    * Utilities for formatting strings.
29    */
30   public class StringFormatter {
31       /**
32        * Pretty-print a date string.
33        *
34        * @param timeMs        The time since the epoch in milliseconds.
35        * @param zoneOffset    The time zone offset.
36        * @return              The date string in ISO format.
37        */
38       public static String dateString(long timeMs, ZoneOffset zoneOffset) {
39           return new Date(timeMs).toInstant().
40               atOffset(zoneOffset).
41               format(DateTimeFormatter.ISO_OFFSET_DATE_TIME);
42       }
43   
44       /**
45        * Pretty-print a duration.
46        *
47        * @param periodMs      The duration in milliseconds.
48        * @return              A human-readable duration string.
49        */
50       public static String durationString(long periodMs) {
51           StringBuilder bld = new StringBuilder();
52           Duration duration = Duration.ofMillis(periodMs);
53           long hours = duration.toHours();
54           if (hours > 0) {
55               bld.append(hours).append("h");
56               duration = duration.minusHours(hours);
57           }
58           long minutes = duration.toMinutes();
59           if (minutes > 0) {
60               bld.append(minutes).append("m");
61               duration = duration.minusMinutes(minutes);
62           }
63           long seconds = duration.getSeconds();
64           if ((seconds != 0) || bld.toString().isEmpty()) {
65               bld.append(seconds).append("s");
66           }
67           return bld.toString();
68       }
69   
70       /**
71        * Formats strings in a grid pattern.
72        *
73        * All entries in the same column will have the same width.
74        *
75        * @param lines     A list of lines.  Each line contains a list of columns.
76        *                  Each line must contain the same number of columns.
77        * @return          The string.
78        */
79       public static String prettyPrintGrid(List<List<String>> lines) {
80           int numColumns = -1;
81           int rowIndex = 0;
82           for (List<String> col : lines) {
83               if (numColumns == -1) {
84                   numColumns = col.size();
85               } else if (numColumns != col.size()) {
86                   throw new RuntimeException("Expected " + numColumns + " columns in row " +
87                       rowIndex + ", but got " + col.size());
88               }
89               rowIndex++;
90           }
91           List<Integer> widths = new ArrayList<>(numColumns);
92           for (int x = 0; x < numColumns; x++) {
93               int w = 0;
94               for (List<String> cols : lines) {
95                   w = Math.max(w, cols.get(x).length() + 1);
96               }
97               widths.add(w);
98           }
99           StringBuilder bld = new StringBuilder();
100           for (int y = 0; y < lines.size(); y++) {
101               List<String> cols = lines.get(y);
102               for (int x = 0; x < cols.size(); x++) {
103                   String val = cols.get(x);
104                   int minWidth = widths.get(x);
105                   bld.append(val);
106                   for (int i = 0; i < minWidth - val.length(); i++) {
107                       bld.append(" ");
108                   }
109               }
110               bld.append(String.format("%n"));
111           }
112           return bld.toString();
113       }
114   }
