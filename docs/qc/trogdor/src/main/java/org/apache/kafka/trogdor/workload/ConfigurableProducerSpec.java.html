29   /**
30    * This is the spec to pass in to be able to run the `ConfigurableProducerWorker` workload.  This allows for customized
31    * and even variable configurations in terms of messages per second, message size, batch size, key size, and even the
32    * ability to target a specific partition out of a topic.
33    *
34    * This has several notable differences from the ProduceBench classes, namely the ability to dynamically control
35    * flushing and throughput through configurable classes, but also the ability to run against specific partitions within
36    * a topic directly.  This workload can only run against one topic at a time, unlike the ProduceBench workload.
37    *
38    * The parameters that differ from ProduceBenchSpec:
39    *
40    *     `flushGenerator` -      Used to instruct the KafkaProducer when to issue flushes.  This allows us to simulate
41    *                             variable batching since batch flushing is not currently exposed within the KafkaProducer
42    *                             class.  See the `FlushGenerator` interface for more information.
43    *
44    *     `throughputGenerator` - Used to throttle the ConfigurableProducerWorker based on a calculated number of messages
45    *                             within a window.  See the `ThroughputGenerator` interface for more information.
46    *
47    *     `activeTopic`         - This class only supports execution against a single topic at a time.  If more than one
48    *                             topic is specified, the ConfigurableProducerWorker will throw an error.
49    *
50    *     `activePartition`     - Specify a specific partition number within the activeTopic to run load against, or
51    *                             specify `-1` to allow use of all partitions.
52    *
53    * Here is an example spec:
54    *
55    * {
56    *     "startMs": 1606949497662,
57    *     "durationMs": 3600000,
58    *     "producerNode": "trogdor-agent-0",
59    *     "bootstrapServers": "some.example.kafka.server:9091",
60    *     "flushGenerator": {
61    *         "type": "gaussian",
62    *         "messagesPerFlushAverage": 16,
63    *         "messagesPerFlushDeviation": 4
64    *     },
65    *     "throughputGenerator": {
66    *         "type": "gaussian",
67    *         "messagesPerSecondAverage": 500,
68    *         "messagesPerSecondDeviation": 50,
69    *         "windowsUntilRateChange": 100,
70    *         "windowSizeMs": 100
71    *     },
72    *     "keyGenerator": {
73    *         "type": "constant",
74    *         "size": 8
75    *     },
76    *     "valueGenerator": {
77    *         "type": "gaussianTimestampRandom",
78    *         "messageSizeAverage": 512,
79    *         "messageSizeDeviation": 100,
80    *         "messagesUntilSizeChange": 100
81    *     },
82    *     "producerConf": {
83    *         "acks": "all"
84    *     },
85    *     "commonClientConf": {},
86    *     "adminClientConf": {},
87    *     "activeTopic": {
88    *         "topic0": {
89    *             "numPartitions": 100,
90    *             "replicationFactor": 3,
91    *             "configs": {
92    *                 "retention.ms": "1800000"
93    *             }
94    *         }
95    *     },
96    *     "activePartition": 5
97    * }
98    *
99    * This example spec performed the following:
100    *
101    *   * Ran on `trogdor-agent-0` for 1 hour starting at 2020-12-02 22:51:37.662 GMT
102    *   * Produced with acks=all to Partition 5 of `topic0` on kafka server `some.example.kafka.server:9091`.
103    *   * The average batch had 16 messages, with a standard deviation of 4 messages.
104    *   * The messages had 8-byte constant keys with an average size of 512 bytes and a standard deviation of 100 bytes.
105    *   * The messages had millisecond timestamps embedded in the first several bytes of the value.
106    *   * The average throughput was 500 messages/second, with a window of 100ms and a deviation of 50 messages/second.
107    */
108   
109   public class ConfigurableProducerSpec extends TaskSpec {
110       private final String producerNode;
111       private final String bootstrapServers;
112       private final Optional<FlushGenerator> flushGenerator;
113       private final ThroughputGenerator throughputGenerator;
114       private final PayloadGenerator keyGenerator;
115       private final PayloadGenerator valueGenerator;
116       private final Map<String, String> producerConf;
117       private final Map<String, String> adminClientConf;
118       private final Map<String, String> commonClientConf;
119       private final TopicsSpec activeTopic;
120       private final int activePartition;
121   
122       @JsonCreator
123       public ConfigurableProducerSpec(@JsonProperty("startMs") long startMs,
124                                       @JsonProperty("durationMs") long durationMs,
125                                       @JsonProperty("producerNode") String producerNode,
126                                       @JsonProperty("bootstrapServers") String bootstrapServers,
127                                       @JsonProperty("flushGenerator") Optional<FlushGenerator> flushGenerator,
128                                       @JsonProperty("throughputGenerator") ThroughputGenerator throughputGenerator,
129                                       @JsonProperty("keyGenerator") PayloadGenerator keyGenerator,
130                                       @JsonProperty("valueGenerator") PayloadGenerator valueGenerator,
131                                       @JsonProperty("producerConf") Map<String, String> producerConf,
132                                       @JsonProperty("commonClientConf") Map<String, String> commonClientConf,
133                                       @JsonProperty("adminClientConf") Map<String, String> adminClientConf,
134                                       @JsonProperty("activeTopic") TopicsSpec activeTopic,
135                                       @JsonProperty("activePartition") int activePartition) {
136           super(startMs, durationMs);
137           this.producerNode = (producerNode == null) ? "" : producerNode;
138           this.bootstrapServers = (bootstrapServers == null) ? "" : bootstrapServers;
139           this.flushGenerator = flushGenerator;
140           this.keyGenerator = keyGenerator;
141           this.valueGenerator = valueGenerator;
142           this.throughputGenerator = throughputGenerator;
143           this.producerConf = configOrEmptyMap(producerConf);
144           this.commonClientConf = configOrEmptyMap(commonClientConf);
145           this.adminClientConf = configOrEmptyMap(adminClientConf);
146           this.activeTopic = activeTopic.immutableCopy();
147           this.activePartition = activePartition;
148       }
