18   package org.apache.kafka.trogdor.basic;
19   
20   import com.fasterxml.jackson.databind.JsonNode;
21   import org.apache.kafka.common.utils.Scheduler;
22   import org.apache.kafka.common.utils.Shell;
23   import org.apache.kafka.common.utils.Utils;
24   import org.apache.kafka.trogdor.common.Node;
25   import org.apache.kafka.trogdor.common.Platform;
26   import org.apache.kafka.trogdor.common.Topology;
27   import org.slf4j.Logger;
28   import org.slf4j.LoggerFactory;
29   
30   import java.io.IOException;
31   
32   /**
33    * Defines a cluster topology
34    */
35   public class BasicPlatform implements Platform {
36       private static final Logger log = LoggerFactory.getLogger(BasicPlatform.class);
37   
38       private final Node curNode;
39       private final BasicTopology topology;
40       private final Scheduler scheduler;
41       private final CommandRunner commandRunner;
42   
43       public interface CommandRunner {
44           String run(Node curNode, String[] command) throws IOException;
45       }
46   
47       public static class ShellCommandRunner implements CommandRunner {
48           @Override
49           public String run(Node curNode, String[] command) throws IOException {
50               try {
51                   String result = Shell.execCommand(command);
52                   log.info("RUN: {}. RESULT: [{}]", Utils.join(command, " "), result);
53                   return result;
54               } catch (RuntimeException | IOException e) {
55                   log.info("RUN: {}. ERROR: [{}]", Utils.join(command, " "), e.getMessage());
56                   throw e;
57               }
58           }
59       }
60   
61       public BasicPlatform(String curNodeName, BasicTopology topology,
62                            Scheduler scheduler, CommandRunner commandRunner) {
63           this.curNode = topology.node(curNodeName);
64           if (this.curNode == null) {
65               throw new RuntimeException(String.format("No node named %s found " +
66                       "in the cluster!  Cluster nodes are: %s", curNodeName,
67                   Utils.join(topology.nodes().keySet(), ",")));
68           }
69           this.topology = topology;
70           this.scheduler = scheduler;
71           this.commandRunner = commandRunner;
72       }
73   
74       public BasicPlatform(String curNodeName, Scheduler scheduler, JsonNode configRoot) {
75           JsonNode nodes = configRoot.get("nodes");
76           if (nodes == null) {
77               throw new RuntimeException("Expected to find a 'nodes' field " +
78                   "in the root JSON configuration object");
79           }
80           this.topology = new BasicTopology(nodes);
81           this.scheduler = scheduler;
82           this.curNode = topology.node(curNodeName);
83           if (this.curNode == null) {
84               throw new RuntimeException(String.format("No node named %s found " +
85                   "in the cluster!  Cluster nodes are: %s", curNodeName,
86                   Utils.join(topology.nodes().keySet(), ",")));
87           }
88           this.commandRunner = new ShellCommandRunner();
89       }
90   
91       @Override
92       public String name() {
93           return "BasicPlatform";
94       }
95   
96       @Override
97       public Node curNode() {
98           return curNode;
99       }
100   
101       @Override
102       public Topology topology() {
103           return topology;
104       }
105   
106       @Override
107       public Scheduler scheduler() {
108           return scheduler;
109       }
110   
111       @Override
112       public String runCommand(String[] command) throws IOException {
113           return commandRunner.run(curNode, command);
114       }
115   }
