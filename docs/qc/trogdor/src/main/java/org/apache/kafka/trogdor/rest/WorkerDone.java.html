18   package org.apache.kafka.trogdor.rest;
19   
20   import com.fasterxml.jackson.annotation.JsonCreator;
21   import com.fasterxml.jackson.annotation.JsonProperty;
22   import com.fasterxml.jackson.databind.JsonNode;
23   import com.fasterxml.jackson.databind.node.NullNode;
24   import org.apache.kafka.trogdor.task.TaskSpec;
25   
26   /**
27    * The state a worker is in once it's done.
28    */
29   public class WorkerDone extends WorkerState {
30       /**
31        * The time on the agent when the task was started.
32        */
33       private final long startedMs;
34   
35       /**
36        * The time on the agent when the task was completed.
37        */
38       private final long doneMs;
39   
40       /**
41        * The task status.  The format will depend on the type of task that is
42        * being run.
43        */
44       private final JsonNode status;
45   
46       /**
47        * Empty if the task completed without error; the error message otherwise.
48        */
49       private final String error;
50   
51       @JsonCreator
52       public WorkerDone(@JsonProperty("taskId") String taskId,
53               @JsonProperty("spec") TaskSpec spec,
54               @JsonProperty("startedMs") long startedMs,
55               @JsonProperty("doneMs") long doneMs,
56               @JsonProperty("status") JsonNode status,
57               @JsonProperty("error") String error) {
58           super(taskId, spec);
59           this.startedMs = startedMs;
60           this.doneMs = doneMs;
61           this.status = status == null ? NullNode.instance : status;
62           this.error = error == null ? "" : error;
63       }
64   
65       @JsonProperty
66       @Override
67       public long startedMs() {
68           return startedMs;
69       }
70   
71       @JsonProperty
72       public long doneMs() {
73           return doneMs;
74       }
75   
76       @JsonProperty
77       @Override
78       public JsonNode status() {
79           return status;
80       }
81   
82       @JsonProperty
83       public String error() {
84           return error;
85       }
86   
87       @Override
88       public boolean done() {
89           return true;
90       }
91   }
