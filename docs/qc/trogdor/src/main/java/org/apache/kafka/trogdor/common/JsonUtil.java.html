18   package org.apache.kafka.trogdor.common;
19   
20   import com.fasterxml.jackson.annotation.JsonInclude;
21   import com.fasterxml.jackson.core.JsonParser;
22   import com.fasterxml.jackson.core.JsonProcessingException;
23   import com.fasterxml.jackson.databind.DeserializationFeature;
24   import com.fasterxml.jackson.databind.ObjectMapper;
25   import com.fasterxml.jackson.databind.SerializationFeature;
26   import com.fasterxml.jackson.datatype.jdk8.Jdk8Module;
27   
28   import java.io.File;
29   
30   /**
31    * Utilities for working with JSON.
32    */
33   public class JsonUtil {
34       public static final ObjectMapper JSON_SERDE;
35   
36       static {
37           JSON_SERDE = new ObjectMapper();
38           JSON_SERDE.configure(SerializationFeature.FAIL_ON_EMPTY_BEANS, false);
39           JSON_SERDE.configure(DeserializationFeature.ACCEPT_SINGLE_VALUE_AS_ARRAY, true);
40           JSON_SERDE.configure(JsonParser.Feature.ALLOW_COMMENTS, true);
41           JSON_SERDE.registerModule(new Jdk8Module());
42           JSON_SERDE.setSerializationInclusion(JsonInclude.Include.NON_EMPTY);
43       }
44   
45       public static String toJsonString(Object object) {
46           try {
47               return JSON_SERDE.writeValueAsString(object);
48           } catch (JsonProcessingException e) {
49               throw new RuntimeException(e);
50           }
51       }
52   
53       public static String toPrettyJsonString(Object object) {
54           try {
55               return JSON_SERDE.writerWithDefaultPrettyPrinter().writeValueAsString(object);
56           } catch (JsonProcessingException e) {
57               throw new RuntimeException(e);
58           }
59       }
60   
61       /**
62        * Determine if a string is a JSON object literal.
63        * Object literals must begin with an open brace.
64        *
65        * @param input         The input string.
66        * @return              True if the string is a JSON literal.
67        */
68       static boolean openBraceComesFirst(String input) {
69           for (int i = 0; i < input.length(); i++) {
70               char c = input.charAt(i);
71               if (!Character.isWhitespace(c)) {
72                   return c == '{';
73               }
74           }
75           return false;
76       }
77   
78       /**
79        * Read a JSON object from a command-line argument.  This can take the form of a path to
80        * a file containing the JSON object, or simply the raw JSON object itself.  We will assume
81        * that if the string is a valid JSON object, the latter is true.  If you want to specify a
82        * file name containing an open brace, you can force it to be interpreted as a file name be
83        * prefixing a ./ or full path.
84        *
85        * @param argument      The command-line argument.
86        * @param clazz         The class of the object to be read.
87        * @param <T>           The object type.
88        * @return              The object which we read.
89        */
90       public static <T> T objectFromCommandLineArgument(String argument, Class<T> clazz) throws Exception {
91           if (openBraceComesFirst(argument)) {
92               return JSON_SERDE.readValue(argument, clazz);
93           } else {
94               return JSON_SERDE.readValue(new File(argument), clazz);
95           }
96       }
97   }
