33   /**
34    * Extension of {@link HttpEntity} that adds a {@link HttpStatus} status code.
35    * Used in {@code RestTemplate} as well {@code @Controller} methods.
36    *
37    * <p>In {@code RestTemplate}, this class is returned by
38    * {@link org.springframework.web.client.RestTemplate#getForEntity getForEntity()} and
39    * {@link org.springframework.web.client.RestTemplate#exchange exchange()}:
40    * <pre class="code">
41    * ResponseEntity&lt;String&gt; entity = template.getForEntity("https://example.com", String.class);
42    * String body = entity.getBody();
43    * MediaType contentType = entity.getHeaders().getContentType();
44    * HttpStatus statusCode = entity.getStatusCode();
45    * </pre>
46    *
47    * <p>Can also be used in Spring MVC, as the return value from a @Controller method:
48    * <pre class="code">
49    * &#64;RequestMapping("/handle")
50    * public ResponseEntity&lt;String&gt; handle() {
51    *   URI location = ...;
52    *   HttpHeaders responseHeaders = new HttpHeaders();
53    *   responseHeaders.setLocation(location);
54    *   responseHeaders.set("MyResponseHeader", "MyValue");
55    *   return new ResponseEntity&lt;String&gt;("Hello World", responseHeaders, HttpStatus.CREATED);
56    * }
57    * </pre>
58    *
59    * Or, by using a builder accessible via static methods:
60    * <pre class="code">
61    * &#64;RequestMapping("/handle")
62    * public ResponseEntity&lt;String&gt; handle() {
63    *   URI location = ...;
64    *   return ResponseEntity.created(location).header("MyResponseHeader", "MyValue").body("Hello World");
65    * }
66    * </pre>
67    *
68    * @author Arjen Poutsma
69    * @author Brian Clozel
70    * @since 3.0.2
71    * @param <T> the body type
72    * @see #getStatusCode()
73    * @see org.springframework.web.client.RestOperations#getForEntity(String, Class, Object...)
74    * @see org.springframework.web.client.RestOperations#getForEntity(String, Class, java.util.Map)
75    * @see org.springframework.web.client.RestOperations#getForEntity(URI, Class)
76    * @see RequestEntity
77    */

180       @Override
181       public String toString() {
182           StringBuilder builder = new StringBuilder("<");
183           builder.append(this.status.toString());
184           if (this.status instanceof HttpStatus) {
185               builder.append(' ');
186               builder.append(((HttpStatus) this.status).getReasonPhrase());
187           }
188           builder.append(',');
189           T body = getBody();
190           HttpHeaders headers = getHeaders();
191           if (body != null) {
192               builder.append(body);
193               builder.append(',');
194           }
195           builder.append(headers);
196           builder.append('>');
197           return builder.toString();
198       }
