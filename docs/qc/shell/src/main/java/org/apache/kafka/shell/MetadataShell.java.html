43   /**
44    * The Kafka metadata shell.
45    */
46   public final class MetadataShell {
47       private static final Logger log = LoggerFactory.getLogger(MetadataShell.class);
48   
49       public static class Builder {
50           private String snapshotPath;
51   
52           public Builder setSnapshotPath(String snapshotPath) {
53               this.snapshotPath = snapshotPath;
54               return this;
55           }
56   
57           public MetadataShell build() throws Exception {
58               if (snapshotPath == null) {
59                   throw new RuntimeException("You must supply the log path via --snapshot");
60               }
61               MetadataNodeManager nodeManager = null;
62               SnapshotFileReader reader = null;
63               try {
64                   nodeManager = new MetadataNodeManager();
65                   reader = new SnapshotFileReader(snapshotPath, nodeManager.logListener());
66                   return new MetadataShell(null, reader, nodeManager);
67               } catch (Throwable e) {
68                   log.error("Initialization error", e);
69                   if (reader != null) {
70                       reader.close();
71                   }
72                   if (nodeManager != null) {
73                       nodeManager.close();
74                   }
75                   throw e;
76               }
77           }
78       }
79   
80       private final KafkaRaftManager<ApiMessageAndVersion> raftManager;
81   
82       private final SnapshotFileReader snapshotFileReader;
83   
84       private final MetadataNodeManager nodeManager;
85   
86       public MetadataShell(KafkaRaftManager<ApiMessageAndVersion> raftManager,
87                           SnapshotFileReader snapshotFileReader,
88                           MetadataNodeManager nodeManager) {
89           this.raftManager = raftManager;
90           this.snapshotFileReader = snapshotFileReader;
91           this.nodeManager = nodeManager;
92       }
93   
94       public void run(List<String> args) throws Exception {
95           nodeManager.setup();
96           if (raftManager != null) {
97               raftManager.startup();
98               raftManager.register(nodeManager.logListener());
99           } else if (snapshotFileReader != null) {
100               snapshotFileReader.startup();
101           } else {
102               throw new RuntimeException("Expected either a raft manager or snapshot reader");
103           }
104           if (args == null || args.isEmpty()) {
105               // Interactive mode.
106               System.out.println("Loading...");
107               waitUntilCaughtUp();
108               System.out.println("Starting...");
109               try (InteractiveShell shell = new InteractiveShell(nodeManager)) {
110                   shell.runMainLoop();
111               }
112           } else {
113               // Non-interactive mode.
114               waitUntilCaughtUp();
115               Commands commands = new Commands(false);
116               try (PrintWriter writer = new PrintWriter(new BufferedWriter(
117                       new OutputStreamWriter(System.out, StandardCharsets.UTF_8)))) {
118                   Commands.Handler handler = commands.parseCommand(args);
119                   handler.run(Optional.empty(), writer, nodeManager);
120                   writer.flush();
121               }
122           }
123       }
124   
125       public void close() throws Exception {
126           if (raftManager != null) {
127               raftManager.shutdown();
128           }
129           if (snapshotFileReader != null) {
130               snapshotFileReader.close();
131           }
132           nodeManager.close();
133       }
134   
135       public static void main(String[] args) throws Exception {
136           ArgumentParser parser = ArgumentParsers
137               .newArgumentParser("metadata-tool")
138               .defaultHelp(true)
139               .description("The Apache Kafka metadata tool");
140           parser.addArgument("--snapshot", "-s")
141               .type(String.class)
142               .help("The snapshot file to read.");
143           parser.addArgument("command")
144               .nargs("*")
145               .help("The command to run.");
146           Namespace res = parser.parseArgsOrFail(args);
147           try {
148               Builder builder = new Builder();
149               builder.setSnapshotPath(res.getString("snapshot"));
150               Path tempDir = Files.createTempDirectory("MetadataShell");
151               Exit.addShutdownHook("agent-shutdown-hook", () -> {
152                   log.debug("Removing temporary directory " + tempDir.toAbsolutePath().toString());
153                   try {
154                       Utils.delete(tempDir.toFile());
155                   } catch (Exception e) {
156                       log.error("Got exception while removing temporary directory " +
157                           tempDir.toAbsolutePath().toString());
158                   }
159               });
160               MetadataShell shell = builder.build();
161               try {
162                   shell.run(res.getList("command"));
163               } finally {
164                   shell.close();
165               }
166               Exit.exit(0);
167           } catch (TerseFailure e) {
168               System.err.println("Error: " + e.getMessage());
169               Exit.exit(1);
170           } catch (Throwable e) {
171               System.err.println("Unexpected error: " +
172                   (e.getMessage() == null ? "" : e.getMessage()));
173               e.printStackTrace(System.err);
174               Exit.exit(1);
175           }
176       }
177   
178       void waitUntilCaughtUp() throws ExecutionException, InterruptedException {
179           snapshotFileReader.caughtUpFuture().get();
180       }
181   }
