90       static class TargetDirectory {
91           private final String name;
92           private final List<String> children;
93   
94           TargetDirectory(String name, List<String> children) {
95               this.name = name;
96               this.children = children;
97           }

100       @Override
101       public void run(Optional<InteractiveShell> shell,
102                       PrintWriter writer,
103                       MetadataNodeManager manager) throws Exception {
104           List<String> targetFiles = new ArrayList<>();
105           List<TargetDirectory> targetDirectories = new ArrayList<>();
106           for (String target : CommandUtils.getEffectivePaths(targets)) {
107               manager.visit(new GlobVisitor(target, entryOption -> {
108                   if (entryOption.isPresent()) {
109                       MetadataNodeInfo info = entryOption.get();
110                       MetadataNode node = info.node();
111                       if (node instanceof DirectoryNode) {
112                           DirectoryNode directory = (DirectoryNode) node;
113                           List<String> children = new ArrayList<>();
114                           children.addAll(directory.children().keySet());
115                           targetDirectories.add(
116                               new TargetDirectory(info.lastPathComponent(), children));
117                       } else if (node instanceof FileNode) {
118                           targetFiles.add(info.lastPathComponent());
119                       }
120                   } else {
121                       writer.println("ls: " + target + ": no such file or directory.");
122                   }
123               }));
124           }
125           OptionalInt screenWidth = shell.isPresent() ?
126               OptionalInt.of(shell.get().screenWidth()) : OptionalInt.empty();
127           log.trace("LS : targetFiles = {}, targetDirectories = {}, screenWidth = {}",
128               targetFiles, targetDirectories, screenWidth);
129           printTargets(writer, screenWidth, targetFiles, targetDirectories);
130       }
131   
132       static void printTargets(PrintWriter writer,
133                                OptionalInt screenWidth,
134                                List<String> targetFiles,
135                                List<TargetDirectory> targetDirectories) {
136           printEntries(writer, "", screenWidth, targetFiles);
137           boolean needIntro = targetFiles.size() > 0 || targetDirectories.size() > 1;
138           boolean firstIntro = targetFiles.isEmpty();
139           for (TargetDirectory targetDirectory : targetDirectories) {
140               String intro = "";
141               if (needIntro) {
142                   if (!firstIntro) {
143                       intro = intro + String.format("%n");
144                   }
145                   intro = intro + targetDirectory.name + ":";
146                   firstIntro = false;
147               }
148               log.trace("LS : targetDirectory name = {}, children = {}",
149                   targetDirectory.name, targetDirectory.children);
150               printEntries(writer, intro, screenWidth, targetDirectory.children);
151           }
152       }
153   
154       static void printEntries(PrintWriter writer,
155                                String intro,
156                                OptionalInt screenWidth,
157                                List<String> entries) {
158           if (entries.isEmpty()) {
159               return;
160           }
161           if (!intro.isEmpty()) {
162               writer.println(intro);
163           }
164           ColumnSchema columnSchema = calculateColumnSchema(screenWidth, entries);
165           int numColumns = columnSchema.numColumns();
166           int numLines = (entries.size() + numColumns - 1) / numColumns;
167           for (int line = 0; line < numLines; line++) {
168               StringBuilder output = new StringBuilder();
169               for (int column = 0; column < numColumns; column++) {
170                   int entryIndex = line + (column * columnSchema.entriesPerColumn());
171                   if (entryIndex < entries.size()) {
172                       String entry = entries.get(entryIndex);
173                       output.append(entry);
174                       if (column < numColumns - 1) {
175                           int width = columnSchema.columnWidth(column);
176                           for (int i = 0; i < width - entry.length(); i++) {
177                               output.append(" ");
178                           }
179                       }
180                   }
181               }
182               writer.println(output.toString());
183           }
184       }

186       static ColumnSchema calculateColumnSchema(OptionalInt screenWidth,
187                                                 List<String> entries) {
188           if (!screenWidth.isPresent()) {
189               return new ColumnSchema(1, entries.size());
190           }
191           int maxColumns = screenWidth.getAsInt() / 4;
192           if (maxColumns <= 1) {
193               return new ColumnSchema(1, entries.size());
194           }
195           ColumnSchema[] schemas = new ColumnSchema[maxColumns];
196           for (int numColumns = 1; numColumns <= maxColumns; numColumns++) {
197               schemas[numColumns - 1] = new ColumnSchema(numColumns,
198                   (entries.size() + numColumns - 1) / numColumns);
199           }
200           for (int i = 0; i < entries.size(); i++) {
201               String entry = entries.get(i);
202               for (int s = 0; s < schemas.length; s++) {
203                   ColumnSchema schema = schemas[s];
204                   schema.process(i, entry);
205               }
206           }
207           for (int s = schemas.length - 1; s > 0; s--) {
208               ColumnSchema schema = schemas[s];
209               if (schema.columnWidths[schema.columnWidths.length - 1] != 0 &&
210                       schema.totalWidth() <= screenWidth.getAsInt()) {
211                   return schema;
212               }
213           }
214           return schemas[0];
215       }

217       static class ColumnSchema {
218           private final int[] columnWidths;
219           private final int entriesPerColumn;
220   
221           ColumnSchema(int numColumns, int entriesPerColumn) {
222               this.columnWidths = new int[numColumns];
223               this.entriesPerColumn = entriesPerColumn;
224           }
225   
226           ColumnSchema setColumnWidths(Integer... widths) {
227               for (int i = 0; i < widths.length; i++) {
228                   columnWidths[i] = widths[i];
229               }
230               return this;
231           }
232   
233           void process(int entryIndex, String output) {
234               int columnIndex = entryIndex / entriesPerColumn;
235               columnWidths[columnIndex] = Math.max(
236                   columnWidths[columnIndex], output.length() + 2);
237           }
238   
239           int totalWidth() {
240               int total = 0;
241               for (int i = 0; i < columnWidths.length; i++) {
242                   total += columnWidths[i];
243               }
244               return total;
245           }
246   
247           int numColumns() {
248               return columnWidths.length;
249           }
250   
251           int columnWidth(int columnIndex) {
252               return columnWidths[columnIndex];
253           }
254   
255           int entriesPerColumn() {
256               return entriesPerColumn;
257           }
258   
259           @Override
260           public int hashCode() {
261               return Objects.hash(columnWidths, entriesPerColumn);
262           }
263   
264           @Override
265           public boolean equals(Object o) {
266               if (!(o instanceof ColumnSchema)) return false;
267               ColumnSchema other = (ColumnSchema) o;
268               if (entriesPerColumn != other.entriesPerColumn) return false;
269               if (!Arrays.equals(columnWidths, other.columnWidths)) return false;
270               return true;
271           }
272   
273           @Override
274           public String toString() {
275               StringBuilder bld = new StringBuilder("ColumnSchema(columnWidths=[");
276               String prefix = "";
277               for (int i = 0; i < columnWidths.length; i++) {
278                   bld.append(prefix);
279                   bld.append(columnWidths[i]);
280                   prefix = ", ";
281               }
282               bld.append("], entriesPerColumn=").append(entriesPerColumn).append(")");
283               return bld.toString();
284           }
285       }
