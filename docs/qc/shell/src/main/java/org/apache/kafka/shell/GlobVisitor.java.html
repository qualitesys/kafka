20   import java.util.Arrays;
21   import java.util.List;
22   import java.util.Map.Entry;
23   import java.util.Objects;
24   import java.util.Optional;
25   import java.util.function.Consumer;
26   
27   /**
28    * Visits metadata paths based on a glob string.
29    */
30   public final class GlobVisitor implements Consumer<MetadataNodeManager.Data> {
31       private final String glob;
32       private final Consumer<Optional<MetadataNodeInfo>> handler;
33   
34       public GlobVisitor(String glob,
35                          Consumer<Optional<MetadataNodeInfo>> handler) {
36           this.glob = glob;
37           this.handler = handler;
38       }
39   
40       public static class MetadataNodeInfo {
41           private final String[] path;
42           private final MetadataNode node;
43   
44           MetadataNodeInfo(String[] path, MetadataNode node) {
45               this.path = path;
46               this.node = node;
47           }
48   
49           public String[] path() {
50               return path;
51           }
52   
53           public MetadataNode node() {
54               return node;
55           }
56   
57           public String lastPathComponent() {
58               if (path.length == 0) {
59                   return "/";
60               } else {
61                   return path[path.length - 1];
62               }
63           }
64   
65           public String absolutePath() {
66               return "/" + String.join("/", path);
67           }
68   
69           @Override
70           public int hashCode() {
71               return Objects.hash(path, node);
72           }
73   
74           @Override
75           public boolean equals(Object o) {
76               if (!(o instanceof MetadataNodeInfo)) return false;
77               MetadataNodeInfo other = (MetadataNodeInfo) o;
78               if (!Arrays.equals(path, other.path)) return false;
79               if (!node.equals(other.node)) return false;
80               return true;
81           }
82   
83           @Override
84           public String toString() {
85               StringBuilder bld = new StringBuilder("MetadataNodeInfo(path=");
86               for (int i = 0; i < path.length; i++) {
87                   bld.append("/");
88                   bld.append(path[i]);
89               }
90               bld.append(", node=").append(node).append(")");
91               return bld.toString();
92           }
93       }
94   
95       @Override
96       public void accept(MetadataNodeManager.Data data) {
97           String fullGlob = glob.startsWith("/") ? glob :
98               data.workingDirectory() + "/" + glob;
99           List<String> globComponents =
100               CommandUtils.stripDotPathComponents(CommandUtils.splitPath(fullGlob));
101           if (!accept(globComponents, 0, data.root(), new String[0])) {
102               handler.accept(Optional.empty());
103           }
104       }
105   
106       private boolean accept(List<String> globComponents,
107                              int componentIndex,
108                              MetadataNode node,
109                              String[] path) {
110           if (componentIndex >= globComponents.size()) {
111               handler.accept(Optional.of(new MetadataNodeInfo(path, node)));
112               return true;
113           }
114           String globComponentString = globComponents.get(componentIndex);
115           GlobComponent globComponent = new GlobComponent(globComponentString);
116           if (globComponent.literal()) {
117               if (!(node instanceof MetadataNode.DirectoryNode)) {
118                   return false;
119               }
120               MetadataNode.DirectoryNode directory = (MetadataNode.DirectoryNode) node;
121               MetadataNode child = directory.child(globComponent.component());
122               if (child == null) {
123                   return false;
124               }
125               String[] newPath = new String[path.length + 1];
126               System.arraycopy(path, 0, newPath, 0, path.length);
127               newPath[path.length] = globComponent.component();
128               return accept(globComponents, componentIndex + 1, child, newPath);
129           }
130           if (!(node instanceof MetadataNode.DirectoryNode)) {
131               return false;
132           }
133           MetadataNode.DirectoryNode directory = (MetadataNode.DirectoryNode) node;
134           boolean matchedAny = false;
135           for (Entry<String, MetadataNode> entry : directory.children().entrySet()) {
136               String nodeName = entry.getKey();
137               if (globComponent.matches(nodeName)) {
138                   String[] newPath = new String[path.length + 1];
139                   System.arraycopy(path, 0, newPath, 0, path.length);
140                   newPath[path.length] = nodeName;
141                   if (accept(globComponents, componentIndex + 1, entry.getValue(), newPath)) {
142                       matchedAny = true;
143                   }
144               }
145           }
146           return matchedAny;
147       }
148   }
