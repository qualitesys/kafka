19   import org.apache.kafka.common.MetricName;

21   import org.apache.kafka.common.metrics.Metrics;

26   import org.apache.kafka.common.metrics.stats.WindowedSum;

30   import java.util.Arrays;
31   import java.util.OptionalLong;
32   import java.util.concurrent.TimeUnit;

34   public class KafkaRaftMetrics implements AutoCloseable {
35   
36       private final Metrics metrics;
37   
38       private volatile OffsetAndEpoch logEndOffset;
39       private volatile int numUnknownVoterConnections;
40       private volatile OptionalLong electionStartMs;
41       private volatile OptionalLong pollStartMs;
42       private volatile OptionalLong pollEndMs;
43   
44       private final MetricName currentLeaderIdMetricName;
45       private final MetricName currentVotedIdMetricName;
46       private final MetricName currentEpochMetricName;
47       private final MetricName currentStateMetricName;
48       private final MetricName highWatermarkMetricName;
49       private final MetricName logEndOffsetMetricName;
50       private final MetricName logEndEpochMetricName;
51       private final MetricName numUnknownVoterConnectionsMetricName;
52       private final Sensor commitTimeSensor;
53       private final Sensor electionTimeSensor;
54       private final Sensor fetchRecordsSensor;
55       private final Sensor appendRecordsSensor;
56       private final Sensor pollIdleSensor;

58       public KafkaRaftMetrics(Metrics metrics, String metricGrpPrefix, QuorumState state) {
59           this.metrics = metrics;
60           String metricGroupName = metricGrpPrefix + "-metrics";
61   
62           this.pollStartMs = OptionalLong.empty();
63           this.pollEndMs = OptionalLong.empty();
64           this.electionStartMs = OptionalLong.empty();
65           this.numUnknownVoterConnections = 0;
66           this.logEndOffset = new OffsetAndEpoch(0L, 0);
67   
68           this.currentStateMetricName = metrics.metricName("current-state", metricGroupName, "The current state of this member; possible values are leader, candidate, voted, follower, unattached");
69           Gauge<String> stateProvider = (mConfig, currentTimeMs) -> {
70               if (state.isLeader()) {
71                   return "leader";
72               } else if (state.isCandidate()) {
73                   return "candidate";
74               } else if (state.isVoted()) {
75                   return "voted";
76               } else if (state.isFollower()) {
77                   return "follower";
78               } else {
79                   return "unattached";
80               }
81           };
82           metrics.addMetric(this.currentStateMetricName, null, stateProvider);
83   
84           this.currentLeaderIdMetricName = metrics.metricName("current-leader", metricGroupName, "The current quorum leader's id; -1 indicates unknown");
85           metrics.addMetric(this.currentLeaderIdMetricName, (mConfig, currentTimeMs) -> state.leaderId().orElse(-1));
86   
87           this.currentVotedIdMetricName = metrics.metricName("current-vote", metricGroupName, "The current voted leader's id; -1 indicates not voted for anyone");
88           metrics.addMetric(this.currentVotedIdMetricName, (mConfig, currentTimeMs) -> {
89               if (state.isLeader() || state.isCandidate()) {
90                   return state.localIdOrThrow();
91               } else if (state.isVoted()) {
92                   return state.votedStateOrThrow().votedId();
93               } else {
94                   return -1;
95               }
96           });
97   
98           this.currentEpochMetricName = metrics.metricName("current-epoch", metricGroupName, "The current quorum epoch.");
99           metrics.addMetric(this.currentEpochMetricName, (mConfig, currentTimeMs) -> state.epoch());
100   
101           this.highWatermarkMetricName = metrics.metricName("high-watermark", metricGroupName, "The high watermark maintained on this member; -1 if it is unknown");
102           metrics.addMetric(this.highWatermarkMetricName, (mConfig, currentTimeMs) -> state.highWatermark().map(hw -> hw.offset).orElse(-1L));
103   
104           this.logEndOffsetMetricName = metrics.metricName("log-end-offset", metricGroupName, "The current raft log end offset.");
105           metrics.addMetric(this.logEndOffsetMetricName, (mConfig, currentTimeMs) -> logEndOffset.offset);
106   
107           this.logEndEpochMetricName = metrics.metricName("log-end-epoch", metricGroupName, "The current raft log end epoch.");
108           metrics.addMetric(this.logEndEpochMetricName, (mConfig, currentTimeMs) -> logEndOffset.epoch);
109   
110           this.numUnknownVoterConnectionsMetricName = metrics.metricName("number-unknown-voter-connections", metricGroupName, "The number of voter connections recognized at this member.");
111           metrics.addMetric(this.numUnknownVoterConnectionsMetricName, (mConfig, currentTimeMs) -> numUnknownVoterConnections);
112   
113           this.commitTimeSensor = metrics.sensor("commit-latency");
114           this.commitTimeSensor.add(metrics.metricName("commit-latency-avg", metricGroupName,
115                   "The average time in milliseconds to commit an entry in the raft log."), new Avg());
116           this.commitTimeSensor.add(metrics.metricName("commit-latency-max", metricGroupName,
117                   "The maximum time in milliseconds to commit an entry in the raft log."), new Max());
118   
119           this.electionTimeSensor = metrics.sensor("election-latency");
120           this.electionTimeSensor.add(metrics.metricName("election-latency-avg", metricGroupName,
121                   "The average time in milliseconds to elect a new leader."), new Avg());
122           this.electionTimeSensor.add(metrics.metricName("election-latency-max", metricGroupName,
123                   "The maximum time in milliseconds to elect a new leader."), new Max());
124   
125           this.fetchRecordsSensor = metrics.sensor("fetch-records");
126           this.fetchRecordsSensor.add(metrics.metricName("fetch-records-rate", metricGroupName,
127                   "The average number of records fetched from the leader of the raft quorum."),
128                   new Rate(TimeUnit.SECONDS, new WindowedSum()));
129   
130           this.appendRecordsSensor = metrics.sensor("append-records");
131           this.appendRecordsSensor.add(metrics.metricName("append-records-rate", metricGroupName,
132                   "The average number of records appended per sec as the leader of the raft quorum."),
133                   new Rate(TimeUnit.SECONDS, new WindowedSum()));
134   
135           this.pollIdleSensor = metrics.sensor("poll-idle-ratio");
136           this.pollIdleSensor.add(metrics.metricName("poll-idle-ratio-avg",
137                   metricGroupName,
138                   "The average fraction of time the client's poll() is idle as opposed to waiting for the user code to process records."),
139                   new Avg());
140       }

142       public void updatePollStart(long currentTimeMs) {
143           if (pollEndMs.isPresent() && pollStartMs.isPresent()) {
144               long pollTimeMs = Math.max(pollEndMs.getAsLong() - pollStartMs.getAsLong(), 0L);
145               long totalTimeMs = Math.max(currentTimeMs - pollStartMs.getAsLong(), 1L);
146               this.pollIdleSensor.record(pollTimeMs / (double) totalTimeMs, currentTimeMs);
147           }
148   
149           this.pollStartMs = OptionalLong.of(currentTimeMs);
150           this.pollEndMs = OptionalLong.empty();
151       }
152   
153       public void updatePollEnd(long currentTimeMs) {
154           this.pollEndMs = OptionalLong.of(currentTimeMs);
155       }
156   
157       public void updateLogEnd(OffsetAndEpoch logEndOffset) {
158           this.logEndOffset = logEndOffset;
159       }
160   
161       public void updateNumUnknownVoterConnections(int numUnknownVoterConnections) {
162           this.numUnknownVoterConnections = numUnknownVoterConnections;
163       }
164   
165       public void updateAppendRecords(long numRecords) {
166           appendRecordsSensor.record(numRecords);
167       }
168   
169       public void updateFetchedRecords(long numRecords) {
170           fetchRecordsSensor.record(numRecords);
171       }
172   
173       public void updateCommitLatency(double latencyMs, long currentTimeMs) {
174           commitTimeSensor.record(latencyMs, currentTimeMs);
175       }
176   
177       public void updateElectionStartMs(long currentTimeMs) {
178           electionStartMs = OptionalLong.of(currentTimeMs);
179       }
180   
181       public void maybeUpdateElectionLatency(long currentTimeMs) {
182           if (electionStartMs.isPresent()) {
183               electionTimeSensor.record(currentTimeMs - electionStartMs.getAsLong(), currentTimeMs);
184               electionStartMs = OptionalLong.empty();
185           }
186       }
187   
188       @Override
189       public void close() {
190           Arrays.asList(
191               currentLeaderIdMetricName,
192               currentVotedIdMetricName,
193               currentEpochMetricName,
194               currentStateMetricName,
195               highWatermarkMetricName,
196               logEndOffsetMetricName,
197               logEndEpochMetricName,
198               numUnknownVoterConnectionsMetricName
199           ).forEach(metrics::removeMetric);
200   
201           Arrays.asList(
202               commitTimeSensor.name(),
203               electionTimeSensor.name(),
204               fetchRecordsSensor.name(),
205               appendRecordsSensor.name(),
206               pollIdleSensor.name()
207           ).forEach(metrics::removeSensor);
208       }
209   }
