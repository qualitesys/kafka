19   import org.apache.kafka.common.utils.LogContext;
20   import org.apache.kafka.common.utils.Time;
21   import org.apache.kafka.common.utils.Timer;
22   import org.apache.kafka.snapshot.RawSnapshotWriter;
23   import org.slf4j.Logger;
24   
25   import java.util.Optional;
26   import java.util.OptionalInt;
27   import java.util.OptionalLong;
28   import java.util.Set;

30   public class FollowerState implements EpochState {
31       private final int fetchTimeoutMs;
32       private final int epoch;
33       private final int leaderId;
34       private final Set<Integer> voters;
35       // Used for tracking the expiration of both the Fetch and FetchSnapshot requests
36       private final Timer fetchTimer;
37       private Optional<LogOffsetMetadata> highWatermark;
38       /* Used to track the currently fetching snapshot. When fetching snapshot regular
39        * Fetch request are paused
40        */
41       private Optional<RawSnapshotWriter> fetchingSnapshot;
42   
43       private final Logger log;
44   
45       public FollowerState(
46           Time time,
47           int epoch,
48           int leaderId,
49           Set<Integer> voters,
50           Optional<LogOffsetMetadata> highWatermark,
51           int fetchTimeoutMs,
52           LogContext logContext
53       ) {
54           this.fetchTimeoutMs = fetchTimeoutMs;
55           this.epoch = epoch;
56           this.leaderId = leaderId;
57           this.voters = voters;
58           this.fetchTimer = time.timer(fetchTimeoutMs);
59           this.highWatermark = highWatermark;
60           this.fetchingSnapshot = Optional.empty();
61           this.log = logContext.logger(FollowerState.class);
62       }
63   
64       @Override
65       public ElectionState election() {
66           return new ElectionState(
67               epoch,
68               OptionalInt.of(leaderId),
69               OptionalInt.empty(),
70               voters
71           );
72       }
73   
74       @Override
75       public int epoch() {
76           return epoch;
77       }
78   
79       @Override
80       public String name() {
81           return "Follower";
82       }
83   
84       public long remainingFetchTimeMs(long currentTimeMs) {
85           fetchTimer.update(currentTimeMs);
86           return fetchTimer.remainingMs();
87       }
88   
89       public int leaderId() {
90           return leaderId;
91       }
92   
93       public boolean hasFetchTimeoutExpired(long currentTimeMs) {
94           fetchTimer.update(currentTimeMs);
95           return fetchTimer.isExpired();
96       }
97   
98       public void resetFetchTimeout(long currentTimeMs) {
99           fetchTimer.update(currentTimeMs);
100           fetchTimer.reset(fetchTimeoutMs);
101       }
102   
103       public void overrideFetchTimeout(long currentTimeMs, long timeoutMs) {
104           fetchTimer.update(currentTimeMs);
105           fetchTimer.reset(timeoutMs);
106       }
107   
108       public boolean updateHighWatermark(OptionalLong highWatermark) {
109           if (!highWatermark.isPresent() && this.highWatermark.isPresent())
110               throw new IllegalArgumentException("Attempt to overwrite current high watermark " + this.highWatermark +
111                   " with unknown value");
112   
113           if (this.highWatermark.isPresent()) {
114               long previousHighWatermark = this.highWatermark.get().offset;
115               long updatedHighWatermark = highWatermark.getAsLong();
116   
117               if (updatedHighWatermark < 0)
118                   throw new IllegalArgumentException("Illegal negative high watermark update");
119               if (previousHighWatermark > updatedHighWatermark)
120                   throw new IllegalArgumentException("Non-monotonic update of high watermark attempted");
121               if (previousHighWatermark == updatedHighWatermark)
122                   return false;
123           }
124   
125           this.highWatermark = highWatermark.isPresent() ?
126               Optional.of(new LogOffsetMetadata(highWatermark.getAsLong())) :
127               Optional.empty();
128           return true;
129       }
130   
131       @Override
132       public Optional<LogOffsetMetadata> highWatermark() {
133           return highWatermark;
134       }
135   
136       public Optional<RawSnapshotWriter> fetchingSnapshot() {
137           return fetchingSnapshot;
138       }
139   
140       public void setFetchingSnapshot(Optional<RawSnapshotWriter> newSnapshot) {
141           if (fetchingSnapshot.isPresent()) {
142               fetchingSnapshot.get().close();
143           }
144           fetchingSnapshot = newSnapshot;
145       }
146   
147       @Override
148       public boolean canGrantVote(int candidateId, boolean isLogUpToDate) {
149           log.debug("Rejecting vote request from candidate {} since we already have a leader {} in epoch {}",
150                   candidateId, leaderId(), epoch);
151           return false;
152       }
153   
154       @Override
155       public String toString() {
156           return "FollowerState(" +
157               "fetchTimeoutMs=" + fetchTimeoutMs +
158               ", epoch=" + epoch +
159               ", leaderId=" + leaderId +
160               ", voters=" + voters +
161               ", highWatermark=" + highWatermark +
162               ", fetchingSnapshot=" + fetchingSnapshot +
163               ')';
164       }
165   
166       @Override
167       public void close() {
168           if (fetchingSnapshot.isPresent()) {
169               fetchingSnapshot.get().close();
170           }
171       }
172   }
