18   
19   import org.apache.kafka.raft.OffsetAndEpoch;
20   import org.apache.kafka.common.utils.Utils;
21   import org.slf4j.Logger;
22   import org.slf4j.LoggerFactory;
23   
24   import java.io.IOException;
25   import java.io.UncheckedIOException;
26   import java.nio.file.Files;
27   import java.nio.file.Path;
28   import java.text.NumberFormat;
29   import java.util.Optional;

31   public final class Snapshots {
32       private static final Logger log = LoggerFactory.getLogger(Snapshots.class);
33       public static final String SUFFIX = ".checkpoint";
34       private static final String PARTIAL_SUFFIX = String.format("%s.part", SUFFIX);
35       private static final String DELETE_SUFFIX = String.format("%s.deleted", SUFFIX);
36   
37       private static final NumberFormat OFFSET_FORMATTER = NumberFormat.getInstance();
38       private static final NumberFormat EPOCH_FORMATTER = NumberFormat.getInstance();
39   
40       private static final int OFFSET_WIDTH = 20;
41       private static final int EPOCH_WIDTH = 10;
42   
43       static {
44           OFFSET_FORMATTER.setMinimumIntegerDigits(OFFSET_WIDTH);
45           OFFSET_FORMATTER.setGroupingUsed(false);
46   
47           EPOCH_FORMATTER.setMinimumIntegerDigits(EPOCH_WIDTH);
48           EPOCH_FORMATTER.setGroupingUsed(false);
49       }
50   
51       static Path snapshotDir(Path logDir) {
52           return logDir;
53       }
54   
55       static String filenameFromSnapshotId(OffsetAndEpoch snapshotId) {
56           return String.format("%s-%s", OFFSET_FORMATTER.format(snapshotId.offset), EPOCH_FORMATTER.format(snapshotId.epoch));
57       }
58   
59       static Path moveRename(Path source, OffsetAndEpoch snapshotId) {
60           return source.resolveSibling(filenameFromSnapshotId(snapshotId) + SUFFIX);
61       }
62   
63       static Path deleteRename(Path source, OffsetAndEpoch snapshotId) {
64           return source.resolveSibling(filenameFromSnapshotId(snapshotId) + DELETE_SUFFIX);
65       }
66   
67       public static Path snapshotPath(Path logDir, OffsetAndEpoch snapshotId) {
68           return snapshotDir(logDir).resolve(filenameFromSnapshotId(snapshotId) + SUFFIX);
69       }
70   
71       public static Path createTempFile(Path logDir, OffsetAndEpoch snapshotId) {
72           Path dir = snapshotDir(logDir);
73   
74           try {
75               // Create the snapshot directory if it doesn't exists
76               Files.createDirectories(dir);
77               String prefix = String.format("%s-", filenameFromSnapshotId(snapshotId));
78               return Files.createTempFile(dir, prefix, PARTIAL_SUFFIX);
79           } catch (IOException e) {
80               throw new UncheckedIOException(
81                   String.format("Error creating temporary file, logDir = %s, snapshotId = %s.",
82                        dir.toAbsolutePath(), snapshotId), e);
83           }
84       }
85   
86       public static Optional<SnapshotPath> parse(Path path) {
87           Path filename = path.getFileName();
88           if (filename == null) {
89               return Optional.empty();
90           }
91   
92           String name = filename.toString();
93   
94           boolean partial = false;
95           boolean deleted = false;
96           if (name.endsWith(PARTIAL_SUFFIX)) {
97               partial = true;
98           } else if (name.endsWith(DELETE_SUFFIX)) {
99               deleted = true;
100           } else if (!name.endsWith(SUFFIX)) {
101               return Optional.empty();
102           }
103   
104           long endOffset = Long.parseLong(name.substring(0, OFFSET_WIDTH));
105           int epoch = Integer.parseInt(
106               name.substring(OFFSET_WIDTH + 1, OFFSET_WIDTH + EPOCH_WIDTH + 1)
107           );
108   
109           return Optional.of(new SnapshotPath(path, new OffsetAndEpoch(endOffset, epoch), partial, deleted));
110       }
111   
112       /**
113        * Delete the snapshot from the filesystem.
114        */
115       public static boolean deleteIfExists(Path logDir, OffsetAndEpoch snapshotId) {
116           Path immutablePath = snapshotPath(logDir, snapshotId);
117           Path deletedPath = deleteRename(immutablePath, snapshotId);
118           try {
119               boolean deleted = Files.deleteIfExists(immutablePath) | Files.deleteIfExists(deletedPath);
120               if (deleted) {
121                   log.info("Deleted snapshot files for snapshot {}.", snapshotId);
122               } else {
123                   log.info("Did not delete snapshot files for snapshot {} since they did not exist.", snapshotId);
124               }
125               return deleted;
126           } catch (IOException e) {
127               log.error("Error deleting snapshot files {} and {}", immutablePath, deletedPath, e);
128               return false;
129           }
130       }

132       /**
133        * Mark a snapshot for deletion by renaming with the deleted suffix
134        */
135       public static void markForDelete(Path logDir, OffsetAndEpoch snapshotId) {
136           Path immutablePath = snapshotPath(logDir, snapshotId);
137           Path deletedPath = deleteRename(immutablePath, snapshotId);
138           try {
139               Utils.atomicMoveWithFallback(immutablePath, deletedPath, false);
140           } catch (IOException e) {
141               throw new UncheckedIOException(
142                   String.format(
143                       "Error renaming snapshot file from %s to %s.",
144                       immutablePath,
145                       deletedPath
146                   ),
147                   e
148               );
149           }
150       }
