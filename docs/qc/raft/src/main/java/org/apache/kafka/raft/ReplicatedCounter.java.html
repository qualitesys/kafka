17   package org.apache.kafka.raft;
18   
19   import org.apache.kafka.common.KafkaException;
20   import org.apache.kafka.common.utils.LogContext;
21   import org.apache.kafka.raft.errors.NotLeaderException;
22   import org.apache.kafka.snapshot.SnapshotReader;
23   import org.apache.kafka.snapshot.SnapshotWriter;
24   import org.slf4j.Logger;
25   
26   import java.util.Optional;
27   import java.util.OptionalInt;
28   
29   import static java.util.Collections.singletonList;
30   
31   public class ReplicatedCounter implements RaftClient.Listener<Integer> {
32       private final int nodeId;
33       private final Logger log;
34       private final RaftClient<Integer> client;
35       private final int snapshotDelayInRecords = 10;
36   
37       private int committed = 0;
38       private int uncommitted = 0;
39       private OptionalInt claimedEpoch = OptionalInt.empty();
40       private long lastOffsetSnapshotted = -1;
41   
42       private int handleSnapshotCalls = 0;
43   
44       public ReplicatedCounter(
45           int nodeId,
46           RaftClient<Integer> client,
47           LogContext logContext
48       ) {
49           this.nodeId = nodeId;
50           this.client = client;
51           log = logContext.logger(ReplicatedCounter.class);
52       }
53   
54       public synchronized boolean isWritable() {
55           return claimedEpoch.isPresent();
56       }
57   
58       public synchronized void increment() {
59           if (!claimedEpoch.isPresent()) {
60               throw new KafkaException("Counter is not currently writable");
61           }
62   
63           int epoch = claimedEpoch.getAsInt();
64           uncommitted += 1;
65           try {
66               long offset = client.scheduleAppend(epoch, singletonList(uncommitted));
67               log.debug("Scheduled append of record {} with epoch {} at offset {}",
68                   uncommitted, epoch, offset);
69           } catch (NotLeaderException e) {
70               log.info("Appending failed, transition to resigned", e);
71               client.resign(epoch);
72           }
73       }
74   
75       @Override
76       public synchronized void handleCommit(BatchReader<Integer> reader) {
77           try {
78               int initialCommitted = committed;
79               long lastCommittedOffset = -1;
80               int lastCommittedEpoch = 0;
81               long lastCommittedTimestamp = -1;
82   
83               while (reader.hasNext()) {
84                   Batch<Integer> batch = reader.next();
85                   log.debug(
86                       "Handle commit of batch with records {} at base offset {}",
87                       batch.records(),
88                       batch.baseOffset()
89                   );
90                   for (Integer nextCommitted: batch.records()) {
91                       if (nextCommitted != committed + 1) {
92                           throw new AssertionError(
93                               String.format(
94                                   "Expected next committed value to be %d, but instead found %d on node %d",
95                                   committed + 1,
96                                   nextCommitted,
97                                   nodeId
98                               )
99                           );
100                       }
101                       committed = nextCommitted;
102                   }
103   
104                   lastCommittedOffset = batch.lastOffset();
105                   lastCommittedEpoch = batch.epoch();
106                   lastCommittedTimestamp = batch.appendTimestamp();
107               }
108               log.debug("Counter incremented from {} to {}", initialCommitted, committed);
109   
110               if (lastOffsetSnapshotted + snapshotDelayInRecords < lastCommittedOffset) {
111                   log.debug(
112                       "Generating new snapshot with committed offset {} and epoch {} since the previoud snapshot includes {}",
113                       lastCommittedOffset,
114                       lastCommittedEpoch,
115                       lastOffsetSnapshotted
116                   );
117                   Optional<SnapshotWriter<Integer>> snapshot = client.createSnapshot(
118                       lastCommittedOffset,
119                       lastCommittedEpoch,
120                       lastCommittedTimestamp);
121                   if (snapshot.isPresent()) {
122                       try {
123                           snapshot.get().append(singletonList(committed));
124                           snapshot.get().freeze();
125                           lastOffsetSnapshotted = lastCommittedOffset;
126                       } finally {
127                           snapshot.get().close();
128                       }
129                   } else {
130                       lastOffsetSnapshotted = lastCommittedOffset;
131                   }
132               }
133           } finally {
134               reader.close();
135           }
136       }
137   
138       @Override
139       public synchronized void handleSnapshot(SnapshotReader<Integer> reader) {
140           try {
141               log.debug("Loading snapshot {}", reader.snapshotId());
142               while (reader.hasNext()) {
143                   Batch<Integer> batch = reader.next();
144                   if (batch.records().size() != 1) {
145                       throw new AssertionError(
146                           String.format(
147                               "Expected the snapshot at %s to only contain one record %s",
148                               reader.snapshotId(),
149                               batch.records()
150                           )
151                       );
152                   }
153   
154                   for (Integer value : batch) {
155                       log.debug("Setting value: {}", value);
156                       committed = value;
157                       uncommitted = value;
158                   }
159               }
160               lastOffsetSnapshotted = reader.lastContainedLogOffset();
161               handleSnapshotCalls += 1;
162               log.debug("Finished loading snapshot. Set value: {}", committed);
163           } finally {
164               reader.close();
165           }
166       }
167   
168       @Override
169       public synchronized void handleLeaderChange(LeaderAndEpoch newLeader) {
170           if (newLeader.isLeader(nodeId)) {
171               log.debug("Counter uncommitted value initialized to {} after claiming leadership in epoch {}",
172                   committed, newLeader);
173               uncommitted = committed;
174               claimedEpoch = OptionalInt.of(newLeader.epoch());
175           } else {
176               log.debug("Counter uncommitted value reset after resigning leadership");
177               uncommitted = -1;
178               claimedEpoch = OptionalInt.empty();
179           }
180           handleSnapshotCalls = 0;
181       }
182   
183       /** Use handleSnapshotCalls to verify leader is never asked to load snapshot */
184       public int handleSnapshotCalls() {
185           return handleSnapshotCalls;
186       }
187   }
