92       /**
93        * Append a list of records into as many batches as necessary.
94        *
95        * The order of the elements in the records argument will match the order in the batches.
96        * This method will use as many batches as necessary to serialize all of the records. Since
97        * this method can split the records into multiple batches it is possible that some of the
98        * records will get committed while other will not when the leader fails.
99        *
100        * @param epoch the expected leader epoch. If this does not match, then {@link NotLeaderException}
101        *              will be thrown
102        * @param records the list of records to include in the batches
103        * @return the expected offset of the last record
104        * @throws RecordBatchTooLargeException if the size of one record T is greater than the maximum
105        *         batch size; if this exception is throw some of the elements in records may have
106        *         been committed
107        * @throws NotLeaderException if the epoch is less than the leader epoch
108        * @throws IllegalArgumentException if the epoch is invalid (greater than the leader epoch)
109        * @throws BufferAllocationException if we failed to allocate memory for the records
110        */
111       public long append(int epoch, List<T> records) {
112           return append(epoch, records, false);
113       }

115       /**
116        * Append a list of records into an atomic batch. We guarantee all records are included in the
117        * same underlying record batch so that either all of the records become committed or none of
118        * them do.
119        *
120        * @param epoch the expected leader epoch. If this does not match, then {@link NotLeaderException}
121        *              will be thrown
122        * @param records the list of records to include in a batch
123        * @return the expected offset of the last record
124        * @throws RecordBatchTooLargeException if the size of the records is greater than the maximum
125        *         batch size; if this exception is throw none of the elements in records were
126        *         committed
127        * @throws NotLeaderException if the epoch is less than the leader epoch
128        * @throws IllegalArgumentException if the epoch is invalid (greater than the leader epoch)
129        * @throws BufferAllocationException if we failed to allocate memory for the records
130        */
131       public long appendAtomic(int epoch, List<T> records) {
132           return append(epoch, records, true);
133       }

135       private long append(int epoch, List<T> records, boolean isAtomic) {
136           if (epoch < this.epoch) {
137               throw new NotLeaderException("Append failed because the given epoch " + epoch + " is stale. " +
138                       "Current leader epoch = " + this.epoch());
139           } else if (epoch > this.epoch) {
140               throw new IllegalArgumentException("Attempt to append from epoch " + epoch +
141                   " which is larger than the current epoch " + this.epoch);
142           }
143   
144           ObjectSerializationCache serializationCache = new ObjectSerializationCache();
145   
146           appendLock.lock();
147           try {
148               maybeCompleteDrain();
149   
150               BatchBuilder<T> batch = null;
151               if (isAtomic) {
152                   batch = maybeAllocateBatch(records, serializationCache);
153               }
154   
155               for (T record : records) {
156                   if (!isAtomic) {
157                       batch = maybeAllocateBatch(Collections.singleton(record), serializationCache);
158                   }
159   
160                   if (batch == null) {
161                       throw new BufferAllocationException("Append failed because we failed to allocate memory to write the batch");
162                   }
163   
164                   batch.appendRecord(record, serializationCache);
165                   nextOffset += 1;
166               }
167   
168               maybeResetLinger();
169   
170               return nextOffset - 1;
171           } finally {
172               appendLock.unlock();
173           }
174       }

176       private void maybeResetLinger() {
177           if (!lingerTimer.isRunning()) {
178               lingerTimer.reset(time.milliseconds() + lingerMs);
179           }
180       }

182       private BatchBuilder<T> maybeAllocateBatch(
183           Collection<T> records,
184           ObjectSerializationCache serializationCache
185       ) {
186           if (currentBatch == null) {
187               startNewBatch();
188           }
189   
190           if (currentBatch != null) {
191               OptionalInt bytesNeeded = currentBatch.bytesNeeded(records, serializationCache);
192               if (bytesNeeded.isPresent() && bytesNeeded.getAsInt() > maxBatchSize) {
193                   throw new RecordBatchTooLargeException(
194                       String.format(
195                           "The total record(s) size of %d exceeds the maximum allowed batch size of %d",
196                           bytesNeeded.getAsInt(),
197                           maxBatchSize
198                       )
199                   );
200               } else if (bytesNeeded.isPresent()) {
201                   completeCurrentBatch();
202                   startNewBatch();
203               }
204           }
205   
206           return currentBatch;
207       }

209       private void completeCurrentBatch() {
210           MemoryRecords data = currentBatch.build();
211           completed.add(new CompletedBatch<>(
212               currentBatch.baseOffset(),
213               currentBatch.records(),
214               data,
215               memoryPool,
216               currentBatch.initialBuffer()
217           ));
218           currentBatch = null;
219       }

221       /**
222        * Append a control batch from a supplied memory record.
223        *
224        * See the {@code valueCreator} parameter description for requirements on this function.
225        *
226        * @param valueCreator a function that uses the passed buffer to create the control
227        *        batch that will be appended. The memory records returned must contain one
228        *        control batch and that control batch have one record.
229        */
230       private void appendControlMessage(Function<ByteBuffer, MemoryRecords> valueCreator) {
231           appendLock.lock();
232           try {
233               ByteBuffer buffer = memoryPool.tryAllocate(256);
234               if (buffer != null) {
235                   try {
236                       forceDrain();
237                       completed.add(
238                           new CompletedBatch<>(
239                               nextOffset,
240                               1,
241                               valueCreator.apply(buffer),
242                               memoryPool,
243                               buffer
244                           )
245                       );
246                       nextOffset += 1;
247                   } catch (Exception e) {
248                       // Release the buffer now since the buffer was not stored in completed for a delayed release
249                       memoryPool.release(buffer);
250                       throw e;
251                   }
252               } else {
253                   throw new IllegalStateException("Could not allocate buffer for the control record");
254               }
255           } finally {
256               appendLock.unlock();
257           }
258       }

326       public void forceDrain() {
327           appendLock.lock();
328           try {
329               drainStatus = DrainStatus.STARTED;
330               maybeCompleteDrain();
331           } finally {
332               appendLock.unlock();
333           }
334       }

336       private void maybeCompleteDrain() {
337           if (drainStatus == DrainStatus.STARTED) {
338               if (currentBatch != null && currentBatch.nonEmpty()) {
339                   completeCurrentBatch();
340               }
341               // Reset the timer to a large value. The linger clock will begin
342               // ticking after the next append.
343               lingerTimer.reset(Long.MAX_VALUE);
344               drainStatus = DrainStatus.FINISHED;
345           }
346       }

348       private void startNewBatch() {
349           ByteBuffer buffer = memoryPool.tryAllocate(maxBatchSize);
350           if (buffer != null) {
351               currentBatch = new BatchBuilder<>(
352                   buffer,
353                   serde,
354                   compressionType,
355                   nextOffset,
356                   time.milliseconds(),
357                   false,
358                   epoch,
359                   maxBatchSize
360               );
361           }
362       }

374       /**
375        * Check the time remaining until the next needed drain. If the accumulator
376        * is empty, then {@link Long#MAX_VALUE} will be returned.
377        *

382           if (drainStatus == DrainStatus.FINISHED) {

398       /**
399        * Drain completed batches. The caller is expected to first check whether
400        * {@link #needsDrain(long)} returns true in order to avoid unnecessary draining.
401        *
402        * Note on thread-safety: this method is safe in the presence of concurrent
403        * appends, but it assumes a single thread is responsible for draining.
404        *
405        * This call will not block, but the drain may require multiple attempts before
406        * it can be completed if the thread responsible for appending is holding the
407        * append lock. In the worst case, the append will be completed on the next
408        * call to {@link #append(int, List)} following the initial call to this method.
409        * The caller should respect the time to the next flush as indicated by
410        * {@link #timeUntilDrain(long)}.
411        *
412        * @return the list of completed batches
413        */
414       public List<CompletedBatch<T>> drain() {
415           // Start the drain if it has not been started already
416           if (drainStatus == DrainStatus.NONE) {
417               drainStatus = DrainStatus.STARTED;
418           }
419   
420           // Complete the drain ourselves if we can acquire the lock
421           if (appendLock.tryLock()) {
422               try {
423                   maybeCompleteDrain();
424               } finally {
425                   appendLock.unlock();
426               }
427           }
428   
429           // If the drain has finished, then all of the batches will be completed
430           if (drainStatus == DrainStatus.FINISHED) {
431               drainStatus = DrainStatus.NONE;
432               return drainCompleted();
433           } else {
434               return Collections.emptyList();
435           }
436       }
437   
438       private List<CompletedBatch<T>> drainCompleted() {
439           List<CompletedBatch<T>> res = new ArrayList<>(completed.size());
440           while (true) {
441               CompletedBatch<T> batch = completed.poll();
442               if (batch == null) {
443                   return res;
444               } else {
445                   res.add(batch);
446               }
447           }
448       }
