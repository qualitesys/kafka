17   package org.apache.kafka.snapshot;
18   
19   import org.apache.kafka.raft.Batch;
20   import org.apache.kafka.raft.OffsetAndEpoch;
21   
22   import java.util.Iterator;
23   
24   /**
25    * A type for reading an immutable snapshot.
26    *
27    * A snapshot reader can be used to scan through all of the objects T in a snapshot. It
28    * is assumed that the content of the snapshot represents all of the objects T for the topic
29    * partition from offset 0 up to but not including the end offset in the snapshot id.
30    *
31    * The offsets ({@code baseOffset()} and {@code lastOffset()} stored in {@code Batch<T>}
32    * objects returned by this iterator are independent of the offset of the records in the
33    * log used to generate this batch.
34    *
35    * Use {@code lastContainedLogOffset()} and {@code lastContainedLogEpoch()} to query which
36    * offsets and epoch from the log are included in this snapshot. Both of these values are
37    * inclusive.
38    */
39   public interface SnapshotReader<T> extends AutoCloseable, Iterator<Batch<T>> {
40       /**
41        * Returns the end offset and epoch for the snapshot.
42        */
43       OffsetAndEpoch snapshotId();
44   
45       /**
46        * Returns the last log offset which is represented in the snapshot.
47        */
48       long lastContainedLogOffset();
49   
50       /**
51        * Returns the epoch of the last log offset which is represented in the snapshot.
52        */
53       int lastContainedLogEpoch();
54   
55       /**
56        * Returns the timestamp of the last log offset which is represented in the snapshot.
57        */
58       long lastContainedLogTimestamp();
59   
60       /**
61        * Closes the snapshot reader.
62        */
63       void close();
64   }
