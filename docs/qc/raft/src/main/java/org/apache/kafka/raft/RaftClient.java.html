17   package org.apache.kafka.raft;
18   
19   import org.apache.kafka.raft.errors.BufferAllocationException;
20   import org.apache.kafka.raft.errors.NotLeaderException;
21   import org.apache.kafka.snapshot.SnapshotReader;
22   import org.apache.kafka.snapshot.SnapshotWriter;
23   
24   import java.util.List;
25   import java.util.Optional;
26   import java.util.OptionalInt;
27   import java.util.OptionalLong;
28   import java.util.concurrent.CompletableFuture;
29   
30   public interface RaftClient<T> extends AutoCloseable {
31   
32       interface Listener<T> {
33           /**
34            * Callback which is invoked for all records committed to the log.
35            * It is the responsibility of this implementation to invoke {@link BatchReader#close()}
36            * after consuming the reader.
37            *
38            * Note that there is not a one-to-one correspondence between writes through
39            * {@link #scheduleAppend(int, List)} or {@link #scheduleAtomicAppend(int, List)}
40            * and this callback. The Raft implementation is free to batch together the records
41            * from multiple append calls provided that batch boundaries are respected. Records
42            * specified through {@link #scheduleAtomicAppend(int, List)} are guaranteed to be a
43            * subset of a batch provided by the {@link BatchReader}. Records specified through
44            * {@link #scheduleAppend(int, List)} are guaranteed to be in the same order but
45            * they can map to any number of batches provided by the {@link BatchReader}.
46            *
47            * @param reader reader instance which must be iterated and closed
48            */
49           void handleCommit(BatchReader<T> reader);
50   
51           /**
52            * Callback which is invoked when the Listener needs to load a snapshot.
53            * It is the responsibility of this implementation to invoke {@link SnapshotReader#close()}
54            * after consuming the reader.
55            *
56            * When handling this call, the implementation must assume that all previous calls
57            * to {@link #handleCommit} contain invalid data.
58            *
59            * @param reader snapshot reader instance which must be iterated and closed
60            */
61           void handleSnapshot(SnapshotReader<T> reader);
62   
63           /**
64            * Called on any change to leadership. This includes both when a leader is elected and
65            * when a leader steps down or fails.
66            *
67            * If this node is the leader, then the notification of leadership will be delayed until
68            * the implementation of this interface has caught up to the high-watermark through calls to
69            * {@link #handleSnapshot(SnapshotReader)} and {@link #handleCommit(BatchReader)}.
70            *
71            * If this node is not the leader, then this method will be called as soon as possible. In
72            * this case the leader may or may not be known for the current epoch.
73            *
74            * Subsequent calls to this method will expose a monotonically increasing epoch. For a
75            * given epoch the leader may be unknown, {@code leader.leaderId} is {@code OptionalInt#empty},
76            * or known {@code leader.leaderId} is {@code OptionalInt#of}. Once a leader is known for
77            * a given epoch it will remain the leader for that epoch. In other words, the implementation of
78            * method should expect this method will be called at most twice for each epoch. Once if the
79            * epoch changed but the leader is not known and once when the leader is known for the current
80            * epoch.
81            *
82            * @param leader the current leader and epoch
83            */
84           default void handleLeaderChange(LeaderAndEpoch leader) {}
85   
86           default void beginShutdown() {}
87       }
88   
89       /**
90        * Initialize the client. This should only be called once on startup.
91        */
92       void initialize();
93   
94       /**
95        * Register a listener to get commit, snapshot and leader notifications.
96        *
97        * The implementation of this interface assumes that each call to {@code register} uses
98        * a different {@code Listener} instance. If the same instance is used for multiple calls
99        * to this method, then only one {@code Listener} will be registered.
100        *
101        * @param listener the listener to register
102        */
103       void register(Listener<T> listener);
104   
105       /**
106        * Unregisters a listener.
107        *
108        * To distinguish from events that happend before the call to {@code unregister} and a future
109        * call to {@code register}, different {@code Listener} instances must be used.
110        *
111        * If the {@code Listener} provided was never registered then the unregistration is ignored. 
112        *
113        * @param listener the listener to unregister
114        */
115       void unregister(Listener<T> listener);
116   
117       /**
118        * Returns the current high water mark, or OptionalLong.empty if it is not known.
119        */
120       OptionalLong highWatermark();
121   
122       /**
123        * Return the current {@link LeaderAndEpoch}.
124        *
125        * @return the current leader and epoch
126        */
127       LeaderAndEpoch leaderAndEpoch();
128   
129       /**
130        * Get local nodeId if one is defined. This may be absent when the client is used
131        * as an anonymous observer, as in the case of the metadata shell.
132        *
133        * @return optional node id
134        */
135       OptionalInt nodeId();
136   
137       /**
138        * Append a list of records to the log. The write will be scheduled for some time
139        * in the future. There is no guarantee that appended records will be written to
140        * the log and eventually committed. While the order of the records is preserve, they can
141        * be appended to the log using one or more batches. Each record may be committed independently.
142        * If a record is committed, then all records scheduled for append during this epoch
143        * and prior to this record are also committed.
144        *
145        * If the provided current leader epoch does not match the current epoch, which
146        * is possible when the state machine has yet to observe the epoch change, then
147        * this method will throw an {@link NotLeaderException} to indicate the leader
148        * to resign its leadership. The state machine is expected to discard all
149        * uncommitted entries after observing an epoch change.
150        *
151        * @param epoch the current leader epoch
152        * @param records the list of records to append
153        * @return the expected offset of the last record if append succeed
154        * @throws org.apache.kafka.common.errors.RecordBatchTooLargeException if the size of the records is greater than the maximum
155        *         batch size; if this exception is throw none of the elements in records were
156        *         committed
157        * @throws NotLeaderException if we are not the current leader or the epoch doesn't match the leader epoch
158        * @throws BufferAllocationException if we failed to allocate memory for the records
159        */
160       long scheduleAppend(int epoch, List<T> records);
161   
162       /**
163        * Append a list of records to the log. The write will be scheduled for some time
164        * in the future. There is no guarantee that appended records will be written to
165        * the log and eventually committed. However, it is guaranteed that if any of the
166        * records become committed, then all of them will be.
167        *
168        * If the provided current leader epoch does not match the current epoch, which
169        * is possible when the state machine has yet to observe the epoch change, then
170        * this method will throw an {@link NotLeaderException} to indicate the leader
171        * to resign its leadership. The state machine is expected to discard all
172        * uncommitted entries after observing an epoch change.
173        *
174        * @param epoch the current leader epoch
175        * @param records the list of records to append
176        * @return the expected offset of the last record if append succeed
177        * @throws org.apache.kafka.common.errors.RecordBatchTooLargeException if the size of the records is greater than the maximum
178        *         batch size; if this exception is throw none of the elements in records were
179        *         committed
180        * @throws NotLeaderException if we are not the current leader or the epoch doesn't match the leader epoch
181        * @throws BufferAllocationException we failed to allocate memory for the records
182        */
183       long scheduleAtomicAppend(int epoch, List<T> records);
184   
185       /**
186        * Attempt a graceful shutdown of the client. This allows the leader to proactively
187        * resign and help a new leader to get elected rather than forcing the remaining
188        * voters to wait for the fetch timeout.
189        *
190        * Note that if the client has hit an unexpected exception which has left it in an
191        * indeterminate state, then the call to shutdown should be skipped. However, it
192        * is still expected that {@link #close()} will be used to clean up any resources
193        * in use.
194        *
195        * @param timeoutMs How long to wait for graceful completion of pending operations.
196        * @return A future which is completed when shutdown completes successfully or the timeout expires.
197        */
198       CompletableFuture<Void> shutdown(int timeoutMs);
199   
200       /**
201        * Resign the leadership. The leader will give up its leadership in the passed epoch
202        * (if it matches the current epoch), and a new election will be held. Note that nothing
203        * prevents this node from being reelected as the leader.
204        *
205        * Notification of successful resignation can be observed through
206        * {@link Listener#handleLeaderChange(LeaderAndEpoch)}.
207        *
208        * @param epoch the epoch to resign from. If this does not match the current epoch, this
209        *              call will be ignored.
210        */
211       void resign(int epoch);
212   
213       /**
214        * Create a writable snapshot file for a committed offset and epoch.
215        *
216        * The RaftClient assumes that the snapshot returned will contain the records up to and
217        * including the committed offset and epoch. See {@link SnapshotWriter} for details on
218        * how to use this object. If a snapshot already exists then returns an
219        * {@link Optional#empty()}.
220        *
221        * @param committedEpoch the epoch of the committed offset
222        * @param committedOffset the last committed offset that will be included in the snapshot
223        * @param lastContainedLogTime The append time of the highest record contained in this snapshot
224        * @return a writable snapshot if it doesn't already exists
225        * @throws IllegalArgumentException if the committed offset is greater than the high-watermark
226        *         or less than the log start offset.
227        */
228       Optional<SnapshotWriter<T>> createSnapshot(long committedOffset, int committedEpoch, long lastContainedLogTime);
229   }
