17   package org.apache.kafka.raft;
18   
19   import org.apache.kafka.common.utils.LogContext;
20   import org.apache.kafka.common.utils.Time;
21   import org.apache.kafka.common.utils.Timer;
22   import org.slf4j.Logger;
23   
24   import java.util.HashSet;
25   import java.util.List;
26   import java.util.Set;
27   
28   /**
29    * This state represents a leader which has fenced itself either because it
30    * is shutting down or because it has encountered a soft failure of some sort.
31    * No writes are accepted in this state and we are not permitted to vote for
32    * any other candidate in this epoch.
33    *
34    * A resigned leader may initiate a new election by sending `EndQuorumEpoch`
35    * requests to all of the voters. This state tracks delivery of this request
36    * in order to prevent unnecessary retries.
37    *
38    * A voter will remain in the `Resigned` state until we either learn about
39    * another election, or our own election timeout expires and we become a
40    * Candidate.
41    */
42   public class ResignedState implements EpochState {
43       private final int localId;
44       private final int epoch;
45       private final Set<Integer> voters;
46       private final long electionTimeoutMs;
47       private final Set<Integer> unackedVoters;
48       private final Timer electionTimer;
49       private final List<Integer> preferredSuccessors;
50       private final Logger log;
51   
52       public ResignedState(
53           Time time,
54           int localId,
55           int epoch,
56           Set<Integer> voters,
57           long electionTimeoutMs,
58           List<Integer> preferredSuccessors,
59           LogContext logContext
60       ) {
61           this.localId = localId;
62           this.epoch = epoch;
63           this.voters = voters;
64           this.unackedVoters = new HashSet<>(voters);
65           this.unackedVoters.remove(localId);
66           this.electionTimeoutMs = electionTimeoutMs;
67           this.electionTimer = time.timer(electionTimeoutMs);
68           this.preferredSuccessors = preferredSuccessors;
69           this.log = logContext.logger(ResignedState.class);
70       }
71   
72       @Override
73       public ElectionState election() {
74           return ElectionState.withElectedLeader(epoch, localId, voters);
75       }
76   
77       @Override
78       public int epoch() {
79           return epoch;
80       }
81   
82       /**
83        * Get the set of voters which have yet to acknowledge the resignation.
84        * This node will send `EndQuorumEpoch` requests to this set until these
85        * voters acknowledge the request or we transition to another state.
86        *
87        * @return the set of unacknowledged voters
88        */
89       public Set<Integer> unackedVoters() {
90           return unackedVoters;
91       }
92   
93       /**
94        * Invoked after receiving a successful `EndQuorumEpoch` response. This
95        * is in order to prevent unnecessary retries.
96        *
97        * @param voterId the ID of the voter that send the successful response
98        */
99       public void acknowledgeResignation(int voterId) {
100           if (!voters.contains(voterId)) {
101               throw new IllegalArgumentException("Attempt to acknowledge delivery of `EndQuorumEpoch` " +
102                   "by a non-voter " + voterId);
103           }
104           unackedVoters.remove(voterId);
105       }
106   
107       /**
108        * Check whether the timeout has expired.
109        *
110        * @param currentTimeMs current time in milliseconds
111        * @return true if the timeout has expired, false otherwise
112        */
113       public boolean hasElectionTimeoutExpired(long currentTimeMs) {
114           electionTimer.update(currentTimeMs);
115           return electionTimer.isExpired();
116       }
117   
118       /**
119        * Check the time remaining until the timeout expires.
120        *
121        * @param currentTimeMs current time in milliseconds
122        * @return the duration in milliseconds from the current time before the timeout expires
123        */
124       public long remainingElectionTimeMs(long currentTimeMs) {
125           electionTimer.update(currentTimeMs);
126           return electionTimer.remainingMs();
127       }
128   
129       public List<Integer> preferredSuccessors() {
130           return preferredSuccessors;
131       }
132   
133       @Override
134       public boolean canGrantVote(int candidateId, boolean isLogUpToDate) {
135           log.debug("Rejecting vote request from candidate {} since we have resigned as candidate/leader in epoch {}",
136               candidateId, epoch);
137           return false;
138       }
139   
140       @Override
141       public String name() {
142           return "Resigned";
143       }
144   
145       @Override
146       public String toString() {
147           return "ResignedState(" +
148               "localId=" + localId +
149               ", epoch=" + epoch +
150               ", voters=" + voters +
151               ", electionTimeoutMs=" + electionTimeoutMs +
152               ", unackedVoters=" + unackedVoters +
153               ", preferredSuccessors=" + preferredSuccessors +
154               ')';
155       }
156   
157       @Override
158       public void close() {}
159   }
