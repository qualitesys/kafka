40   /**
41    * Collect a set of records into a single batch. New records are added
42    * through {@link #appendRecord(Object, ObjectSerializationCache)}, but the caller must first
43    * check whether there is room using {@link #bytesNeeded(Collection, ObjectSerializationCache)}. Once the
44    * batch is ready, then {@link #build()} should be used to get the resulting
45    * {@link MemoryRecords} instance.
46    *
47    * @param <T> record type indicated by {@link RecordSerde} passed in constructor
48    */
49   public class BatchBuilder<T> {
50       private final ByteBuffer initialBuffer;
51       private final CompressionType compressionType;
52       private final ByteBufferOutputStream batchOutput;
53       private final DataOutputStreamWritable recordOutput;
54       private final long baseOffset;
55       private final long appendTime;
56       private final boolean isControlBatch;
57       private final int leaderEpoch;
58       private final int initialPosition;
59       private final int maxBytes;
60       private final RecordSerde<T> serde;
61       private final List<T> records;
62   
63       private long nextOffset;
64       private int unflushedBytes;
65       private boolean isOpenForAppends = true;
66   
67       public BatchBuilder(
68           ByteBuffer buffer,
69           RecordSerde<T> serde,
70           CompressionType compressionType,
71           long baseOffset,
72           long appendTime,
73           boolean isControlBatch,
74           int leaderEpoch,
75           int maxBytes
76       ) {
77           this.initialBuffer = buffer;
78           this.batchOutput = new ByteBufferOutputStream(buffer);
79           this.serde = serde;
80           this.compressionType = compressionType;
81           this.baseOffset = baseOffset;
82           this.nextOffset = baseOffset;
83           this.appendTime = appendTime;
84           this.isControlBatch = isControlBatch;
85           this.initialPosition = batchOutput.position();
86           this.leaderEpoch = leaderEpoch;
87           this.maxBytes = maxBytes;
88           this.records = new ArrayList<>();
89   
90           // field compressionType must be set before calculating the batch header size
91           int batchHeaderSizeInBytes = batchHeaderSizeInBytes();
92           batchOutput.position(initialPosition + batchHeaderSizeInBytes);
93   
94           this.recordOutput = new DataOutputStreamWritable(new DataOutputStream(
95               compressionType.wrapForOutput(this.batchOutput, RecordBatch.MAGIC_VALUE_V2)));
96       }
97   
98       /**
99        * Append a record to this batch. The caller must first verify there is room for the batch
100        * using {@link #bytesNeeded(Collection, ObjectSerializationCache)}.
101        *
102        * @param record the record to append
103        * @param serializationCache serialization cache for use in {@link RecordSerde#write(Object, ObjectSerializationCache, Writable)}
104        * @return the offset of the appended batch
105        */
106       public long appendRecord(T record, ObjectSerializationCache serializationCache) {
107           if (!isOpenForAppends) {
108               throw new IllegalStateException("Cannot append new records after the batch has been built");
109           }
110   
111           if (nextOffset - baseOffset > Integer.MAX_VALUE) {
112               throw new IllegalArgumentException("Cannot include more than " + Integer.MAX_VALUE +
113                   " records in a single batch");
114           }
115   
116           long offset = nextOffset++;
117           int recordSizeInBytes = writeRecord(
118               offset,
119               record,
120               serializationCache
121           );
122           unflushedBytes += recordSizeInBytes;
123           records.add(record);
124           return offset;
125       }
126   
127       /**
128        * Check whether the batch has enough room for all the record values.
129        *
130        * Returns an empty {@link OptionalInt} if the batch builder has room for this list of records.
131        * Otherwise it returns the expected number of bytes needed for a batch to contain these records.
132        *
133        * @param records the records to use when checking for room
134        * @param serializationCache serialization cache for computing sizes
135        * @return empty {@link OptionalInt} if there is room for the records to be appended, otherwise
136        *         returns the number of bytes needed
137        */
138       public OptionalInt bytesNeeded(Collection<T> records, ObjectSerializationCache serializationCache) {
139           int bytesNeeded = bytesNeededForRecords(
140               records,
141               serializationCache
142           );
143   
144           if (!isOpenForAppends) {
145               return OptionalInt.of(batchHeaderSizeInBytes() + bytesNeeded);
146           }
147   
148           int approxUnusedSizeInBytes = maxBytes - approximateSizeInBytes();
149           if (approxUnusedSizeInBytes >= bytesNeeded) {
150               return OptionalInt.empty();
151           } else if (unflushedBytes > 0) {
152               recordOutput.flush();
153               unflushedBytes = 0;
154               int unusedSizeInBytes = maxBytes - flushedSizeInBytes();
155               if (unusedSizeInBytes >= bytesNeeded) {
156                   return OptionalInt.empty();
157               }
158           }
159   
160           return OptionalInt.of(batchHeaderSizeInBytes() + bytesNeeded);
161       }
162   
163       private int flushedSizeInBytes() {
164           return batchOutput.position() - initialPosition;
165       }
166   
167       /**
168        * Get an estimate of the current size of the appended data. This estimate
169        * is precise if no compression is in use.
170        *
171        * @return estimated size in bytes of the appended records
172        */
173       public int approximateSizeInBytes() {
174           return flushedSizeInBytes() + unflushedBytes;
175       }
176   
177       /**
178        * Get the base offset of this batch. This is constant upon constructing
179        * the builder instance.
180        *
181        * @return the base offset
182        */
183       public long baseOffset() {
184           return baseOffset;
185       }
186   
187       /**
188        * Return the offset of the last appended record. This is updated after
189        * every append and can be used after the batch has been built to obtain
190        * the last offset.
191        *
192        * @return the offset of the last appended record
193        */
194       public long lastOffset() {
195           return nextOffset - 1;
196       }
197   
198       /**
199        * Get the number of records appended to the batch. This is updated after
200        * each append.
201        *
202        * @return the number of appended records
203        */
204       public int numRecords() {
205           return (int) (nextOffset - baseOffset);
206       }
207   
208       /**
209        * Check whether there has been at least one record appended to the batch.
210        *
211        * @return true if one or more records have been appended
212        */
213       public boolean nonEmpty() {
214           return numRecords() > 0;
215       }
216   
217       /**
218        * Return the reference to the initial buffer passed through the constructor.
219        * This is used in case the buffer needs to be returned to a pool (e.g.
220        * in {@link org.apache.kafka.common.memory.MemoryPool#release(ByteBuffer)}.
221        *
222        * @return the initial buffer passed to the constructor
223        */
224       public ByteBuffer initialBuffer() {
225           return initialBuffer;
226       }
227   
228       /**
229        * Get a list of the records appended to the batch.
230        * @return a list of records
231        */
232       public List<T> records() {
233           return records;
234       }

236       private void writeDefaultBatchHeader() {
237           ByteBuffer buffer = batchOutput.buffer();
238           int lastPosition = buffer.position();
239   
240           buffer.position(initialPosition);
241           int size = lastPosition - initialPosition;
242           int lastOffsetDelta = (int) (lastOffset() - baseOffset);
243   
244           DefaultRecordBatch.writeHeader(
245               buffer,
246               baseOffset,
247               lastOffsetDelta,
248               size,
249               RecordBatch.MAGIC_VALUE_V2,
250               compressionType,
251               TimestampType.CREATE_TIME,
252               appendTime,
253               appendTime,
254               RecordBatch.NO_PRODUCER_ID,
255               RecordBatch.NO_PRODUCER_EPOCH,
256               RecordBatch.NO_SEQUENCE,
257               false,
258               isControlBatch,
259               false,
260               leaderEpoch,
261               numRecords()
262           );
263   
264           buffer.position(lastPosition);
265       }
266   
267       public MemoryRecords build() {
268           recordOutput.close();
269           writeDefaultBatchHeader();
270           ByteBuffer buffer = batchOutput.buffer().duplicate();
271           buffer.flip();
272           buffer.position(initialPosition);
273           isOpenForAppends = false;
274           return MemoryRecords.readableRecords(buffer.slice());
275       }

277       public int writeRecord(
278           long offset,
279           T payload,
280           ObjectSerializationCache serializationCache
281       ) {
282           int offsetDelta = (int) (offset - baseOffset);
283           long timestampDelta = 0;
284   
285           int payloadSize = serde.recordSize(payload, serializationCache);
286           int sizeInBytes = DefaultRecord.sizeOfBodyInBytes(
287               offsetDelta,
288               timestampDelta,
289               -1,
290               payloadSize,
291               DefaultRecord.EMPTY_HEADERS
292           );
293           recordOutput.writeVarint(sizeInBytes);
294   
295           // Write attributes (currently unused)
296           recordOutput.writeByte((byte) 0);
297   
298           // Write timestamp and offset
299           recordOutput.writeVarlong(timestampDelta);
300           recordOutput.writeVarint(offsetDelta);
301   
302           // Write key, which is always null for controller messages
303           recordOutput.writeVarint(-1);
304   
305           // Write value
306           recordOutput.writeVarint(payloadSize);
307           serde.write(payload, serializationCache, recordOutput);
308   
309           // Write headers (currently unused)
310           recordOutput.writeVarint(0);
311           return ByteUtils.sizeOfVarint(sizeInBytes) + sizeInBytes;
312       }
313   
314       private int batchHeaderSizeInBytes() {
315           return AbstractRecords.recordBatchHeaderSizeInBytes(
316               RecordBatch.MAGIC_VALUE_V2,
317               compressionType
318           );
319       }

321       private int bytesNeededForRecords(
322           Collection<T> records,
323           ObjectSerializationCache serializationCache
324       ) {
325           long expectedNextOffset = nextOffset;
326           int bytesNeeded = 0;
327           for (T record : records) {
328               if (expectedNextOffset - baseOffset >= Integer.MAX_VALUE) {
329                   throw new IllegalArgumentException(
330                       String.format(
331                           "Adding %s records to a batch with base offset of %s and next offset of %s",
332                           records.size(),
333                           baseOffset,
334                           expectedNextOffset
335                       )
336                   );
337               }
338   
339               int recordSizeInBytes = DefaultRecord.sizeOfBodyInBytes(
340                   (int) (expectedNextOffset  - baseOffset),
341                   0,
342                   -1,
343                   serde.recordSize(record, serializationCache),
344                   DefaultRecord.EMPTY_HEADERS
345               );
346   
347               bytesNeeded = Math.addExact(bytesNeeded, ByteUtils.sizeOfVarint(recordSizeInBytes));
348               bytesNeeded = Math.addExact(bytesNeeded, recordSizeInBytes);
349   
350               expectedNextOffset += 1;
351           }
352   
353           return bytesNeeded;
354       }
