35   /**
36    * This class is responsible for managing the current state of this node and ensuring
37    * only valid state transitions. Below we define the possible state transitions and
38    * how they are triggered:
39    *
40    * Unattached|Resigned transitions to:
41    *    Unattached: After learning of a new election with a higher epoch
42    *    Voted: After granting a vote to a candidate
43    *    Candidate: After expiration of the election timeout
44    *    Follower: After discovering a leader with an equal or larger epoch
45    *
46    * Voted transitions to:
47    *    Unattached: After learning of a new election with a higher epoch
48    *    Candidate: After expiration of the election timeout
49    *
50    * Candidate transitions to:
51    *    Unattached: After learning of a new election with a higher epoch
52    *    Candidate: After expiration of the election timeout
53    *    Leader: After receiving a majority of votes
54    *
55    * Leader transitions to:
56    *    Unattached: After learning of a new election with a higher epoch
57    *    Resigned: When shutting down gracefully
58    *
59    * Follower transitions to:
60    *    Unattached: After learning of a new election with a higher epoch
61    *    Candidate: After expiration of the fetch timeout
62    *    Follower: After discovering a leader with a larger epoch
63    *
64    * Observers follow a simpler state machine. The Voted/Candidate/Leader/Resigned
65    * states are not possible for observers, so the only transitions that are possible
66    * are between Unattached and Follower.
67    *
68    * Unattached transitions to:
69    *    Unattached: After learning of a new election with a higher epoch
70    *    Follower: After discovering a leader with an equal or larger epoch
71    *
72    * Follower transitions to:
73    *    Unattached: After learning of a new election with a higher epoch
74    *    Follower: After discovering a leader with a larger epoch
75    *
76    */

109       public void initialize(OffsetAndEpoch logEndOffsetAndEpoch) throws IllegalStateException {
110           // We initialize in whatever state we were in on shutdown. If we were a leader
111           // or candidate, probably an election was held, but we will find out about it
112           // when we send Vote or BeginEpoch requests.
113   
114           ElectionState election;
115           try {
116               election = store.readElectionState();
117               if (election == null) {
118                   election = ElectionState.withUnknownLeader(0, voters);
119               }
120           } catch (final UncheckedIOException e) {
121               // For exceptions during state file loading (missing or not readable),
122               // we could assume the file is corrupted already and should be cleaned up.
123               log.warn("Clearing local quorum state store after error loading state {}",
124                   store.toString(), e);
125               store.clear();
126               election = ElectionState.withUnknownLeader(0, voters);
127           }
128   
129           final EpochState initialState;
130           if (!election.voters().isEmpty() && !voters.equals(election.voters())) {
131               throw new IllegalStateException("Configured voter set: " + voters
132                   + " is different from the voter set read from the state file: " + election.voters()
133                   + ". Check if the quorum configuration is up to date, "
134                   + "or wipe out the local state file if necessary");
135           } else if (election.hasVoted() && !isVoter()) {
136               String localIdDescription = localId.isPresent() ?
137                   localId.getAsInt() + " is not a voter" :
138                   "is undefined";
139               throw new IllegalStateException("Initialized quorum state " + election
140                   + " with a voted candidate, which indicates this node was previously "
141                   + " a voter, but the local id " + localIdDescription);
142           } else if (election.epoch < logEndOffsetAndEpoch.epoch) {
143               log.warn("Epoch from quorum-state file is {}, which is " +
144                   "smaller than last written epoch {} in the log",
145                   election.epoch, logEndOffsetAndEpoch.epoch);
146               initialState = new UnattachedState(
147                   time,
148                   logEndOffsetAndEpoch.epoch,
149                   voters,
150                   Optional.empty(),
151                   randomElectionTimeoutMs(),
152                   logContext
153               );
154           } else if (localId.isPresent() && election.isLeader(localId.getAsInt())) {
155               // If we were previously a leader, then we will start out as resigned
156               // in the same epoch. This serves two purposes:
157               // 1. It ensures that we cannot vote for another leader in the same epoch.
158               // 2. It protects the invariant that each record is uniquely identified by
159               //    offset and epoch, which might otherwise be violated if unflushed data
160               //    is lost after restarting.
161               initialState = new ResignedState(
162                   time,
163                   localId.getAsInt(),
164                   election.epoch,
165                   voters,
166                   randomElectionTimeoutMs(),
167                   Collections.emptyList(),
168                   logContext
169               );
170           } else if (localId.isPresent() && election.isVotedCandidate(localId.getAsInt())) {
171               initialState = new CandidateState(
172                   time,
173                   localId.getAsInt(),
174                   election.epoch,
175                   voters,
176                   Optional.empty(),
177                   1,
178                   randomElectionTimeoutMs(),
179                   logContext
180               );
181           } else if (election.hasVoted()) {
182               initialState = new VotedState(
183                   time,
184                   election.epoch,
185                   election.votedId(),
186                   voters,
187                   Optional.empty(),
188                   randomElectionTimeoutMs(),
189                   logContext
190               );
191           } else if (election.hasLeader()) {
192               initialState = new FollowerState(
193                   time,
194                   election.epoch,
195                   election.leaderId(),
196                   voters,
197                   Optional.empty(),
198                   fetchTimeoutMs,
199                   logContext
200               );
201           } else {
202               initialState = new UnattachedState(
203                   time,
204                   election.epoch,
205                   voters,
206                   Optional.empty(),
207                   randomElectionTimeoutMs(),
208                   logContext
209               );
210           }
211   
212           transitionTo(initialState);
213       }

272       public void transitionToResigned(List<Integer> preferredSuccessors) {
273           if (!isLeader()) {
274               throw new IllegalStateException("Invalid transition to Resigned state from " + state);
275           }
276   
277           // The Resigned state is a soft state which does not need to be persisted.
278           // A leader will always be re-initialized in this state.
279           int epoch = state.epoch();
280           this.state = new ResignedState(
281               time,
282               localIdOrThrow(),
283               epoch,
284               voters,
285               randomElectionTimeoutMs(),
286               preferredSuccessors,
287               logContext
288           );
289           log.info("Completed transition to {}", state);
290       }
291   
292       /**
293        * Transition to the "unattached" state. This means we have found an epoch greater than
294        * or equal to the current epoch, but wo do not yet know of the elected leader.
295        */
296       public void transitionToUnattached(int epoch) {
297           int currentEpoch = state.epoch();
298           if (epoch <= currentEpoch) {
299               throw new IllegalStateException("Cannot transition to Unattached with epoch= " + epoch +
300                   " from current state " + state);
301           }
302   
303           final long electionTimeoutMs;
304           if (isObserver()) {
305               electionTimeoutMs = Long.MAX_VALUE;
306           } else if (isCandidate()) {
307               electionTimeoutMs = candidateStateOrThrow().remainingElectionTimeMs(time.milliseconds());
308           } else if (isVoted()) {
309               electionTimeoutMs = votedStateOrThrow().remainingElectionTimeMs(time.milliseconds());
310           } else if (isUnattached()) {
311               electionTimeoutMs = unattachedStateOrThrow().remainingElectionTimeMs(time.milliseconds());
312           } else {
313               electionTimeoutMs = randomElectionTimeoutMs();
314           }
315   
316           transitionTo(new UnattachedState(
317               time,
318               epoch,
319               voters,
320               state.highWatermark(),
321               electionTimeoutMs,
322               logContext
323           ));
324       }
325   
326       /**
327        * Grant a vote to a candidate and become a follower for this epoch. We will remain in this
328        * state until either the election timeout expires or a leader is elected. In particular,
329        * we do not begin fetching until the election has concluded and {@link #transitionToFollower(int, int)}
330        * is invoked.
331        */
332       public void transitionToVoted(
333           int epoch,
334           int candidateId
335       ) {
336           if (localId.isPresent() && candidateId == localId.getAsInt()) {
337               throw new IllegalStateException("Cannot transition to Voted with votedId=" + candidateId +
338                   " and epoch=" + epoch + " since it matches the local broker.id");
339           } else if (isObserver()) {
340               throw new IllegalStateException("Cannot transition to Voted with votedId=" + candidateId +
341                   " and epoch=" + epoch + " since the local broker.id=" + localId + " is not a voter");
342           } else if (!isVoter(candidateId)) {
343               throw new IllegalStateException("Cannot transition to Voted with voterId=" + candidateId +
344                   " and epoch=" + epoch + " since it is not one of the voters " + voters);
345           }
346   
347           int currentEpoch = state.epoch();
348           if (epoch < currentEpoch) {
349               throw new IllegalStateException("Cannot transition to Voted with votedId=" + candidateId +
350                   " and epoch=" + epoch + " since the current epoch " + currentEpoch + " is larger");
351           } else if (epoch == currentEpoch && !isUnattached()) {
352               throw new IllegalStateException("Cannot transition to Voted with votedId=" + candidateId +
353                   " and epoch=" + epoch + " from the current state " + state);
354           }
355   
356           // Note that we reset the election timeout after voting for a candidate because we
357           // know that the candidate has at least as good of a chance of getting elected as us
358   
359           transitionTo(new VotedState(
360               time,
361               epoch,
362               candidateId,
363               voters,
364               state.highWatermark(),
365               randomElectionTimeoutMs(),
366               logContext
367           ));
368       }

370       /**
371        * Become a follower of an elected leader so that we can begin fetching.
372        */
373       public void transitionToFollower(
374           int epoch,
375           int leaderId
376       ) {
377           if (localId.isPresent() && leaderId == localId.getAsInt()) {
378               throw new IllegalStateException("Cannot transition to Follower with leaderId=" + leaderId +
379                   " and epoch=" + epoch + " since it matches the local broker.id=" + localId);
380           } else if (!isVoter(leaderId)) {
381               throw new IllegalStateException("Cannot transition to Follower with leaderId=" + leaderId +
382                   " and epoch=" + epoch + " since it is not one of the voters " + voters);
383           }
384   
385           int currentEpoch = state.epoch();
386           if (epoch < currentEpoch) {
387               throw new IllegalStateException("Cannot transition to Follower with leaderId=" + leaderId +
388                   " and epoch=" + epoch + " since the current epoch " + currentEpoch + " is larger");
389           } else if (epoch == currentEpoch
390               && (isFollower() || isLeader())) {
391               throw new IllegalStateException("Cannot transition to Follower with leaderId=" + leaderId +
392                   " and epoch=" + epoch + " from state " + state);
393           }
394   
395           transitionTo(new FollowerState(
396               time,
397               epoch,
398               leaderId,
399               voters,
400               state.highWatermark(),
401               fetchTimeoutMs,
402               logContext
403           ));
404       }

406       public void transitionToCandidate() {
407           if (isObserver()) {
408               throw new IllegalStateException("Cannot transition to Candidate since the local broker.id=" + localId +
409                   " is not one of the voters " + voters);
410           } else if (isLeader()) {
411               throw new IllegalStateException("Cannot transition to Candidate since the local broker.id=" + localId +
412                   " since this node is already a Leader with state " + state);
413           }
414   
415           int retries = isCandidate() ? candidateStateOrThrow().retries() + 1 : 1;
416           int newEpoch = epoch() + 1;
417           int electionTimeoutMs = randomElectionTimeoutMs();
418   
419           transitionTo(new CandidateState(
420               time,
421               localIdOrThrow(),
422               newEpoch,
423               voters,
424               state.highWatermark(),
425               retries,
426               electionTimeoutMs,
427               logContext
428           ));
429       }

431       public <T> LeaderState<T> transitionToLeader(long epochStartOffset, BatchAccumulator<T> accumulator) {
432           if (isObserver()) {
433               throw new IllegalStateException("Cannot transition to Leader since the local broker.id="  + localId +
434                   " is not one of the voters " + voters);
435           } else if (!isCandidate()) {
436               throw new IllegalStateException("Cannot transition to Leader from current state " + state);
437           }
438   
439           CandidateState candidateState = candidateStateOrThrow();
440           if (!candidateState.isVoteGranted())
441               throw new IllegalStateException("Cannot become leader without majority votes granted");
442   
443           // Note that the leader does not retain the high watermark that was known
444           // in the previous state. The reason for this is to protect the monotonicity
445           // of the global high watermark, which is exposed through the leader. The
446           // only way a new leader can be sure that the high watermark is increasing
447           // monotonically is to wait until a majority of the voters have reached the
448           // starting offset of the new epoch. The downside of this is that the local
449           // state machine is temporarily stalled by the advancement of the global
450           // high watermark even though it only depends on local monotonicity. We
451           // could address this problem by decoupling the local high watermark, but
452           // we typically expect the state machine to be caught up anyway.
453   
454           LeaderState<T> state = new LeaderState<>(
455               localIdOrThrow(),
456               epoch(),
457               epochStartOffset,
458               voters,

475           }
