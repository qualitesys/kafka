103   /**
104    * This class implements a Kafkaesque version of the Raft protocol. Leader election
105    * is more or less pure Raft, but replication is driven by replica fetching and we use Kafka's
106    * log reconciliation protocol to truncate the log to a common point following each leader
107    * election.
108    *
109    * Like Zookeeper, this protocol distinguishes between voters and observers. Voters are
110    * the only ones who are eligible to handle protocol requests and they are the only ones
111    * who take part in elections. The protocol does not yet support dynamic quorum changes.
112    *
113    * These are the APIs in this protocol:
114    *
115    * 1) {@link VoteRequestData}: Sent by valid voters when their election timeout expires and they
116    *    become a candidate. This request includes the last offset in the log which electors use
117    *    to tell whether or not to grant the vote.
118    *
119    * 2) {@link BeginQuorumEpochRequestData}: Sent by the leader of an epoch only to valid voters to
120    *    assert its leadership of the new epoch. This request will be retried indefinitely for
121    *    each voter until it acknowledges the request or a new election occurs.
122    *
123    *    This is not needed in usual Raft because the leader can use an empty data push
124    *    to achieve the same purpose. The Kafka Raft implementation, however, is driven by
125    *    fetch requests from followers, so there must be a way to find the new leader after
126    *    an election has completed.
127    *
128    * 3) {@link EndQuorumEpochRequestData}: Sent by the leader of an epoch to valid voters in order to
129    *    gracefully resign from the current epoch. This causes remaining voters to immediately
130    *    begin a new election.
131    *
132    * 4) {@link FetchRequestData}: This is the same as the usual Fetch API in Kafka, but we add snapshot
133    *    check before responding, and we also piggyback some additional metadata on responses (i.e. current
134    *    leader and epoch). Unlike partition replication, we also piggyback truncation detection on this API
135    *    rather than through a separate truncation state.
136    *
137    * 5) {@link FetchSnapshotRequestData}: Sent by the follower to the epoch leader in order to fetch a snapshot.
138    *    This happens when a FetchResponse includes a snapshot ID due to the follower's log end offset being less
139    *    than the leader's log start offset. This API is similar to the Fetch API since the snapshot is stored
140    *    as FileRecords, but we use {@link UnalignedRecords} in FetchSnapshotResponse because the records
141    *    are not necessarily offset-aligned.
142    */

307       private void updateListenersProgress(long highWatermark) {
308           for (ListenerContext listenerContext : listenerContexts.values()) {
309               listenerContext.nextExpectedOffset().ifPresent(nextExpectedOffset -> {
310                   if (nextExpectedOffset < log.startOffset() && nextExpectedOffset < highWatermark) {
311                       SnapshotReader<T> snapshot = latestSnapshot().orElseThrow(() -> new IllegalStateException(
312                           String.format(
313                               "Snapshot expected since next offset of %s is %d, log start offset is %d and high-watermark is %d",
314                               listenerContext.listenerName(),
315                               nextExpectedOffset,
316                               log.startOffset(),
317                               highWatermark
318                           )
319                       ));
320                       listenerContext.fireHandleSnapshot(snapshot);
321                   }
322               });
323   
324               // Re-read the expected offset in case the snapshot had to be reloaded
325               listenerContext.nextExpectedOffset().ifPresent(nextExpectedOffset -> {
326                   if (nextExpectedOffset < highWatermark) {
327                       LogFetchInfo readInfo = log.read(nextExpectedOffset, Isolation.COMMITTED);
328                       listenerContext.fireHandleCommit(nextExpectedOffset, readInfo.records);
329                   }
330               });
331           }
332       }
