17   package org.apache.kafka.snapshot;
18   
19   import org.apache.kafka.common.record.MemoryRecords;
20   import org.apache.kafka.common.record.UnalignedMemoryRecords;
21   import org.apache.kafka.common.utils.Utils;
22   import org.apache.kafka.raft.OffsetAndEpoch;
23   import org.apache.kafka.raft.ReplicatedLog;
24   
25   import java.io.IOException;
26   import java.io.UncheckedIOException;
27   import java.nio.channels.FileChannel;
28   import java.nio.file.Files;
29   import java.nio.file.Path;
30   import java.nio.file.StandardOpenOption;
31   import java.util.Optional;
32   
33   public final class FileRawSnapshotWriter implements RawSnapshotWriter {
34       private final Path tempSnapshotPath;
35       private final FileChannel channel;
36       private final OffsetAndEpoch snapshotId;
37       private final Optional<ReplicatedLog> replicatedLog;
38       private boolean frozen = false;
39   
40       private FileRawSnapshotWriter(
41           Path tempSnapshotPath,
42           FileChannel channel,
43           OffsetAndEpoch snapshotId,
44           Optional<ReplicatedLog> replicatedLog
45       ) {
46           this.tempSnapshotPath = tempSnapshotPath;
47           this.channel = channel;
48           this.snapshotId = snapshotId;
49           this.replicatedLog = replicatedLog;
50       }
51   
52       @Override
53       public OffsetAndEpoch snapshotId() {
54           return snapshotId;
55       }
56   
57       @Override
58       public long sizeInBytes() {
59           try {
60               return channel.size();
61           } catch (IOException e) {
62               throw new UncheckedIOException(
63                   String.format(
64                       "Error calculating snapshot size. temp path = %s, snapshotId = %s.",
65                       tempSnapshotPath,
66                       snapshotId),
67                   e
68               );
69           }
70       }
71   
72       @Override
73       public void append(UnalignedMemoryRecords records) {
74           try {
75               checkIfFrozen("Append");
76               Utils.writeFully(channel, records.buffer());
77           } catch (IOException e) {
78               throw new UncheckedIOException(
79                   String.format("Error writing file snapshot, " +
80                       "temp path = %s, snapshotId = %s.", this.tempSnapshotPath, this.snapshotId),
81                   e
82               );
83           }
84       }
85   
86       @Override
87       public void append(MemoryRecords records) {
88           try {
89               checkIfFrozen("Append");
90               Utils.writeFully(channel, records.buffer());
91           } catch (IOException e) {
92               throw new UncheckedIOException(
93                   String.format("Error writing file snapshot, " +
94                       "temp path = %s, snapshotId = %s.", this.tempSnapshotPath, this.snapshotId),
95                   e
96               );
97           }
98       }
99   
100       @Override
101       public boolean isFrozen() {
102           return frozen;
103       }
104   
105       @Override
106       public void freeze() {
107           try {
108               checkIfFrozen("Freeze");
109   
110               channel.close();
111               frozen = true;
112   
113               if (!tempSnapshotPath.toFile().setReadOnly()) {
114                   throw new IllegalStateException(String.format("Unable to set file (%s) as read-only", tempSnapshotPath));
115               }
116   
117               Path destination = Snapshots.moveRename(tempSnapshotPath, snapshotId);
118               Utils.atomicMoveWithFallback(tempSnapshotPath, destination);
119   
120               replicatedLog.ifPresent(log -> log.onSnapshotFrozen(snapshotId));
121           } catch (IOException e) {
122               throw new UncheckedIOException(
123                   String.format("Error freezing file snapshot, " +
124                       "temp path = %s, snapshotId = %s.", this.tempSnapshotPath, this.snapshotId),
125                   e
126               );
127           }
128       }
129   
130       @Override
131       public void close() {
132           try {
133               channel.close();
134               // This is a noop if freeze was called before calling close
135               Files.deleteIfExists(tempSnapshotPath);
136           } catch (IOException e) {
137               throw new UncheckedIOException(
138                   String.format("Error closing snapshot writer, " +
139                       "temp path = %s, snapshotId %s.", this.tempSnapshotPath, this.snapshotId),
140                   e
141               );
142           }
143       }
144   
145       @Override
146       public String toString() {
147           return String.format(
148               "FileRawSnapshotWriter(path=%s, snapshotId=%s, frozen=%s)",
149               tempSnapshotPath,
150               snapshotId,
151               frozen
152           );
153       }
154   
155       void checkIfFrozen(String operation) {
156           if (frozen) {
157               throw new IllegalStateException(
158                   String.format(
159                       "%s is not supported. Snapshot is already frozen: id = %s; temp path = %s",
160                       operation,
161                       snapshotId,
162                       tempSnapshotPath
163                   )
164               );
165           }
166       }
167   
168       /**
169        * Create a snapshot writer for topic partition log dir and snapshot id.
170        *
171        * @param logDir the directory for the topic partition
172        * @param snapshotId the end offset and epoch for the snapshotId
173        */
174       public static FileRawSnapshotWriter create(
175           Path logDir,
176           OffsetAndEpoch snapshotId,
177           Optional<ReplicatedLog> replicatedLog
178       ) {
179           Path path = Snapshots.createTempFile(logDir, snapshotId);
180   
181           try {
182               return new FileRawSnapshotWriter(
183                   path,
184                   FileChannel.open(path, StandardOpenOption.WRITE, StandardOpenOption.APPEND),
185                   snapshotId,
186                   replicatedLog
187               );
188           } catch (IOException e) {
189               throw new UncheckedIOException(
190                   String.format(
191                       "Error creating snapshot writer. path = %s, snapshotId %s.",
192                       path,
193                       snapshotId
194                   ),
195                   e
196               );
197           }
198       }
199   }
