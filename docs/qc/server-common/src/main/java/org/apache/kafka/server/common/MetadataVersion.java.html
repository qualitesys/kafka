26   /**
27    * This class contains the different Kafka versions.
28    * Right now, we use them for upgrades - users can configure the version of the API brokers will use to communicate between themselves.
29    * This is only for inter-broker communications - when communicating with clients, the client decides on the API version.
30    *
31    * Note that the ID we initialize for each version is important.
32    * We consider a version newer than another if it is lower in the enum list (to avoid depending on lexicographic order)
33    *
34    * Since the api protocol may change more than once within the same release and to facilitate people deploying code from
35    * trunk, we have the concept of internal versions (first introduced during the 0.10.0 development cycle). For example,
36    * the first time we introduce a version change in a release, say 0.10.0, we will add a config value "0.10.0-IV0" and a
37    * corresponding enum constant IBP_0_10_0-IV0. We will also add a config value "0.10.0" that will be mapped to the
38    * latest internal version object, which is IBP_0_10_0-IV0. When we change the protocol a second time while developing
39    * 0.10.0, we will add a new config value "0.10.0-IV1" and a corresponding enum constant IBP_0_10_0-IV1. We will change
40    * the config value "0.10.0" to map to the latest internal version IBP_0_10_0-IV1. The config value of
41    * "0.10.0-IV0" is still mapped to IBP_0_10_0-IV0. This way, if people are deploying from trunk, they can use
42    * "0.10.0-IV0" and "0.10.0-IV1" to upgrade one internal version at a time. For most people who just want to use
43    * released version, they can use "0.10.0" when upgrading to the 0.10.0 release.
44    */

363       /**
364        * Return an `MetadataVersion` instance for `versionString`, which can be in a variety of formats (e.g. "0.8.0", "0.8.0.x",
365        * "0.10.0", "0.10.0-IV1"). `IllegalArgumentException` is thrown if `versionString` cannot be mapped to an `MetadataVersion`.
366        * Note that 'misconfigured' values such as "1.0.1" will be parsed to `IBP_1_0_IV0` as we ignore anything after the first
367        * two digits for versions that don't start with "0."
368        */
369       public static MetadataVersion fromVersionString(String versionString) {
370           String[] versionSegments = versionString.split(Pattern.quote("."));
371           int numSegments = (versionString.startsWith("0.")) ? 3 : 2;
372           String key;
373           if (numSegments >= versionSegments.length) {
374               key = versionString;
375           } else {
376               key = String.join(".", Arrays.copyOfRange(versionSegments, 0, numSegments));
377           }
378           return Optional.ofNullable(IBP_VERSIONS.get(key)).orElseThrow(() ->
379               new IllegalArgumentException("Version " + versionString + " is not a valid version")
380           );
381       }

388           }
389           throw new IllegalArgumentException("No MetadataVersion with metadata version " + version);
390       }

412       public static boolean checkIfMetadataChanged(MetadataVersion sourceVersion, MetadataVersion targetVersion) {
413           if (sourceVersion == targetVersion) {
414               return false;
415           }
416   
417           final MetadataVersion highVersion, lowVersion;
418           if (sourceVersion.compareTo(targetVersion) < 0) {
419               highVersion = targetVersion;
420               lowVersion = sourceVersion;
421           } else {
422               highVersion = sourceVersion;
423               lowVersion = targetVersion;
424           }
425           return checkIfMetadataChangedOrdered(highVersion, lowVersion);
426       }

428       private static boolean checkIfMetadataChangedOrdered(MetadataVersion highVersion, MetadataVersion lowVersion) {
429           MetadataVersion version = highVersion;
430           while (!version.didMetadataChange() && version != lowVersion) {
431               Optional<MetadataVersion> prev = version.previous();
432               if (prev.isPresent()) {
433                   version = prev.get();
434               } else {
435                   break;
436               }
437           }
438           return version != lowVersion;
439       }
