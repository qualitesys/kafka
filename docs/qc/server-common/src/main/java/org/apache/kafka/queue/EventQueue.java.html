20   import org.slf4j.Logger;
21   
22   import java.util.OptionalLong;
23   import java.util.concurrent.RejectedExecutionException;
24   import java.util.concurrent.TimeUnit;
25   import java.util.function.Function;

28   public interface EventQueue extends AutoCloseable {
29       interface Event {
30           /**
31            * Run the event.
32            */
33           void run() throws Exception;
34   
35           /**
36            * Handle an exception that was either generated by running the event, or by the
37            * event queue's inability to run the event.
38            *
39            * @param e     The exception.  This will be a TimeoutException if the event hit
40            *              its deadline before it could be scheduled.
41            *              It will be a RejectedExecutionException if the event could not be
42            *              scheduled because the event queue has already been closed.
43            *              Otherweise, it will be whatever exception was thrown by run().
44            */
45           default void handleException(Throwable e) {}
46       }
47   
48       abstract class FailureLoggingEvent implements Event {
49           private final Logger log;
50   
51           public FailureLoggingEvent(Logger log) {
52               this.log = log;
53           }
54   
55           @Override
56           public void handleException(Throwable e) {
57               if (e instanceof RejectedExecutionException) {
58                   log.info("Not processing {} because the event queue is closed.", this);
59               } else {
60                   log.error("Unexpected error handling {}", this, e);
61               }
62           }
63   
64           @Override
65           public String toString() {
66               return this.getClass().getSimpleName();
67           }
68       }
69   
70       class NoDeadlineFunction implements Function<OptionalLong, OptionalLong> {
71           public static final NoDeadlineFunction INSTANCE = new NoDeadlineFunction();
72   
73           @Override
74           public OptionalLong apply(OptionalLong ignored) {
75               return OptionalLong.empty();
76           }
77       }
78   
79       class DeadlineFunction implements Function<OptionalLong, OptionalLong> {
80           private final long deadlineNs;
81   
82           public DeadlineFunction(long deadlineNs) {
83               this.deadlineNs = deadlineNs;
84           }
85   
86           @Override
87           public OptionalLong apply(OptionalLong ignored) {
88               return OptionalLong.of(deadlineNs);
89           }
90       }
91   
92       class EarliestDeadlineFunction implements Function<OptionalLong, OptionalLong> {
93           private final long newDeadlineNs;
94   
95           public EarliestDeadlineFunction(long newDeadlineNs) {
96               this.newDeadlineNs = newDeadlineNs;
97           }
98   
99           @Override
100           public OptionalLong apply(OptionalLong prevDeadlineNs) {
101               if (!prevDeadlineNs.isPresent()) {
102                   return OptionalLong.of(newDeadlineNs);
103               } else if (prevDeadlineNs.getAsLong() < newDeadlineNs) {
104                   return prevDeadlineNs;
105               } else {
106                   return OptionalLong.of(newDeadlineNs);
107               }
108           }
109       }

111       class VoidEvent implements Event {
112           public final static VoidEvent INSTANCE = new VoidEvent();
113   
114           @Override
115           public void run() throws Exception {
116           }
117       }
118   
119       /**
120        * Add an element to the front of the queue.
121        *
122        * @param event             The mandatory event to prepend.
123        */
124       default void prepend(Event event) {
125           enqueue(EventInsertionType.PREPEND, null, NoDeadlineFunction.INSTANCE, event);
126       }
127   
128       /**
129        * Add an element to the end of the queue.
130        *
131        * @param event             The event to append.
132        */
133       default void append(Event event) {
134           enqueue(EventInsertionType.APPEND, null, NoDeadlineFunction.INSTANCE, event);
135       }
136   
137       /**
138        * Add an event to the end of the queue.
139        *
140        * @param deadlineNs        The deadline for starting the event, in monotonic
141        *                          nanoseconds.  If the event has not started by this
142        *                          deadline, handleException is called with a
143        *                          {@link org.apache.kafka.common.errors.TimeoutException},
144        *                          and the event is cancelled.
145        * @param event             The event to append.
146        */
147       default void appendWithDeadline(long deadlineNs, Event event) {
148           enqueue(EventInsertionType.APPEND, null, new DeadlineFunction(deadlineNs), event);
149       }
150   
151       /**
152        * Schedule an event to be run at a specific time.
153        *
154        * @param tag                   If this is non-null, the unique tag to use for this
155        *                              event.  If an event with this tag already exists, it
156        *                              will be cancelled.
157        * @param deadlineNsCalculator  A function which takes as an argument the existing
158        *                              deadline for the event with this tag (or empty if the
159        *                              event has no tag, or if there is none such), and
160        *                              produces the deadline to use for this event.
161        *                              Once the deadline has arrived, the event will be
162        *                              run.  Events whose deadlines are only a few nanoseconds
163        *                              apart may be executed in any order.
164        * @param event                 The event to schedule.
165        */
166       default void scheduleDeferred(String tag,
167                                     Function<OptionalLong, OptionalLong> deadlineNsCalculator,
168                                     Event event) {
169           enqueue(EventInsertionType.DEFERRED, tag, deadlineNsCalculator, event);
170       }
171   
172       /**
173        * Cancel a deferred event.
174        *
175        * @param tag                   The unique tag for the event to be cancelled.  Must be
176        *                              non-null.  If the event with the tag has not been
177        *                              scheduled, this call will be ignored.
178        */
179       void cancelDeferred(String tag);
180   
181       enum EventInsertionType {
182           PREPEND,
183           APPEND,
184           DEFERRED
185       }
186   
187       /**
188        * Add an event to the queue.
189        *
190        * @param insertionType         How to insert the event.
191        *                              PREPEND means insert the event as the first thing
192        *                              to run.  APPEND means insert the event as the last
193        *                              thing to run.  DEFERRED means insert the event to
194        *                              run after a delay.
195        * @param tag                   If this is non-null, the unique tag to use for
196        *                              this event.  If an event with this tag already
197        *                              exists, it will be cancelled.
198        * @param deadlineNsCalculator  If this is non-null, it is a function which takes
199        *                              as an argument the existing deadline for the
200        *                              event with this tag (or null if the event has no
201        *                              tag, or if there is none such), and produces the
202        *                              deadline to use for this event (or empty to use
203        *                              none.)  Events whose deadlines are only a few
204        *                              nanoseconds apart may be executed in any order.
205        * @param event                 The event to enqueue.
206        */
207       void enqueue(EventInsertionType insertionType,
208                    String tag,
209                    Function<OptionalLong, OptionalLong> deadlineNsCalculator,
210                    Event event);
211   
212       /**
213        * Asynchronously shut down the event queue with no unnecessary delay.
214        * @see #beginShutdown(String, Event, long, TimeUnit)
215        *
216        * @param source                The source of the shutdown.
217        */
218       default void beginShutdown(String source) {
219           beginShutdown(source, new VoidEvent());
220       }
221   
222       /**
223        * Asynchronously shut down the event queue with no unnecessary delay.
224        *
225        * @param source        The source of the shutdown.
226        * @param cleanupEvent  The mandatory event to invoke after all other events have
227        *                      been processed.
228        * @see #beginShutdown(String, Event, long, TimeUnit)
229        */
230       default void beginShutdown(String source, Event cleanupEvent) {
231           beginShutdown(source, cleanupEvent, 0, TimeUnit.SECONDS);
232       }
233   
234       /**
235        * Asynchronously shut down the event queue.
236        *
237        * No new events will be accepted, and the timeout will be initiated
238        * for all existing events.
239        *
240        * @param source        The source of the shutdown.
241        * @param cleanupEvent  The mandatory event to invoke after all other events have
242        *                      been processed.
243        * @param timeSpan      The amount of time to use for the timeout.
244        *                      Once the timeout elapses, any remaining queued
245        *                      events will get a
246        *                      {@link org.apache.kafka.common.errors.TimeoutException}.
247        * @param timeUnit      The time unit to use for the timeout.
248        */
249       void beginShutdown(String source, Event cleanupEvent, long timeSpan, TimeUnit timeUnit);
250   
251       /**
252        * This method is used during unit tests where MockTime is in use.
253        * It is used to alert the queue that the mock time has changed.
254        */
255       default void wakeup() { }
256   
257       /**
258        * Synchronously close the event queue and wait for any threads to be joined.
259        */
260       void close() throws InterruptedException;
261   }
