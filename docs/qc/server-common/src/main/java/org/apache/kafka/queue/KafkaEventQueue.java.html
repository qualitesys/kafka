37   public final class KafkaEventQueue implements EventQueue {
38       /**
39        * A context object that wraps events.
40        */
41       private static class EventContext {
42           /**
43            * The caller-supplied event.
44            */
45           private final Event event;
46   
47           /**
48            * How this event was inserted.
49            */
50           private final EventInsertionType insertionType;
51   
52           /**
53            * The previous pointer of our circular doubly-linked list.
54            */
55           private EventContext prev = this;
56   
57           /**
58            * The next pointer in our circular doubly-linked list.
59            */
60           private EventContext next = this;
61   
62           /**
63            * If this event is in the delay map, this is the key it is there under.
64            * If it is not in the map, this is null.
65            */
66           private OptionalLong deadlineNs = OptionalLong.empty();
67   
68           /**
69            * The tag associated with this event.
70            */
71           private String tag;
72   
73           EventContext(Event event, EventInsertionType insertionType, String tag) {
74               this.event = event;
75               this.insertionType = insertionType;
76               this.tag = tag;
77           }
78   
79           /**
80            * Insert the event context in the circularly linked list after this node.
81            */
82           void insertAfter(EventContext other) {
83               this.next.prev = other;
84               other.next = this.next;
85               other.prev = this;
86               this.next = other;
87           }
88   
89           /**
90            * Insert a new node in the circularly linked list before this node.
91            */
92           void insertBefore(EventContext other) {
93               this.prev.next = other;
94               other.prev = this.prev;
95               other.next = this;
96               this.prev = other;
97           }
98   
99           /**
100            * Remove this node from the circularly linked list.
101            */
102           void remove() {
103               this.prev.next = this.next;
104               this.next.prev = this.prev;
105               this.prev = this;
106               this.next = this;
107           }
108   
109           /**
110            * Returns true if this node is the only element in its list.
111            */
112           boolean isSingleton() {
113               return prev == this && next == this;
114           }
115   
116           /**
117            * Run the event associated with this EventContext.
118            */
119           void run(Logger log) throws InterruptedException {
120               try {
121                   event.run();
122               } catch (InterruptedException e) {
123                   throw e;
124               } catch (Exception e) {
125                   try {
126                       event.handleException(e);
127                   } catch (Throwable t) {
128                       log.error("Unexpected exception in handleException", t);
129                   }
130               }
131           }
132   
133           /**
134            * Complete the event associated with this EventContext with a timeout exception.
135            */
136           void completeWithTimeout() {
137               completeWithException(new TimeoutException());
138           }
139   
140           /**
141            * Complete the event associated with this EventContext with the specified
142            * exception.
143            */
144           void completeWithException(Throwable t) {
145               event.handleException(t);
146           }
147       }
148   
149       private class EventHandler implements Runnable {
150           /**
151            * Event contexts indexed by tag.  Events without a tag are not included here.
152            */
153           private final Map<String, EventContext> tagToEventContext = new HashMap<>();
154   
155           /**
156            * The head of the event queue.
157            */
158           private final EventContext head = new EventContext(null, null, null);
159   
160           /**
161            * An ordered map of times in monotonic nanoseconds to events to time out.
162            */
163           private final TreeMap<Long, EventContext> deadlineMap = new TreeMap<>();
164   
165           /**
166            * A condition variable for waking up the event handler thread.
167            */
168           private final Condition cond = lock.newCondition();
169   
170           @Override
171           public void run() {
172               try {
173                   handleEvents();
174                   cleanupEvent.run();
175               } catch (Throwable e) {
176                   log.warn("event handler thread exiting with exception", e);
177               }
178           }
179   
180           private void remove(EventContext eventContext) {
181               eventContext.remove();
182               if (eventContext.deadlineNs.isPresent()) {
183                   deadlineMap.remove(eventContext.deadlineNs.getAsLong());
184                   eventContext.deadlineNs = OptionalLong.empty();
185               }
186               if (eventContext.tag != null) {
187                   tagToEventContext.remove(eventContext.tag, eventContext);
188                   eventContext.tag = null;
189               }
190           }
191   
192           private void handleEvents() throws InterruptedException {
193               EventContext toTimeout = null;
194               EventContext toRun = null;
195               while (true) {
196                   if (toTimeout != null) {
197                       toTimeout.completeWithTimeout();
198                       toTimeout = null;
199                   } else if (toRun != null) {
200                       toRun.run(log);
201                       toRun = null;
202                   }
203                   lock.lock();
204                   try {
205                       long awaitNs = Long.MAX_VALUE;
206                       Map.Entry<Long, EventContext> entry = deadlineMap.firstEntry();
207                       if (entry != null) {
208                           // Search for timed-out events or deferred events that are ready
209                           // to run.
210                           long now = time.nanoseconds();
211                           long timeoutNs = entry.getKey();
212                           EventContext eventContext = entry.getValue();
213                           if (timeoutNs <= now) {
214                               if (eventContext.insertionType == EventInsertionType.DEFERRED) {
215                                   // The deferred event is ready to run.  Prepend it to the
216                                   // queue.  (The value for deferred events is a schedule time
217                                   // rather than a timeout.)
218                                   remove(eventContext);
219                                   toRun = eventContext;
220                               } else {
221                                   // not a deferred event, so it is a deadline, and it is timed out.
222                                   remove(eventContext);
223                                   toTimeout = eventContext;
224                               }
225                               continue;
226                           } else if (closingTimeNs <= now) {
227                               remove(eventContext);
228                               toTimeout = eventContext;
229                               continue;
230                           }
231                           awaitNs = timeoutNs - now;
232                       }
233                       if (head.next == head) {
234                           if ((closingTimeNs != Long.MAX_VALUE) && deadlineMap.isEmpty()) {
235                               // If there are no more entries to process, and the queue is
236                               // closing, exit the thread.
237                               return;
238                           }
239                       } else {
240                           toRun = head.next;
241                           remove(toRun);
242                           continue;
243                       }
244                       if (closingTimeNs != Long.MAX_VALUE) {
245                           long now = time.nanoseconds();
246                           if (awaitNs > closingTimeNs - now) {
247                               awaitNs = closingTimeNs - now;
248                           }
249                       }
250                       if (awaitNs == Long.MAX_VALUE) {
251                           cond.await();
252                       } else {
253                           cond.awaitNanos(awaitNs);
254                       }
255                   } finally {
256                       lock.unlock();
257                   }
258               }
259           }
260   
261           Exception enqueue(EventContext eventContext,
262                             Function<OptionalLong, OptionalLong> deadlineNsCalculator) {
263               lock.lock();
264               try {
265                   if (closingTimeNs != Long.MAX_VALUE) {
266                       return new RejectedExecutionException();
267                   }
268                   OptionalLong existingDeadlineNs = OptionalLong.empty();
269                   if (eventContext.tag != null) {
270                       EventContext toRemove =
271                           tagToEventContext.put(eventContext.tag, eventContext);
272                       if (toRemove != null) {
273                           existingDeadlineNs = toRemove.deadlineNs;
274                           remove(toRemove);
275                       }
276                   }
277                   OptionalLong deadlineNs = deadlineNsCalculator.apply(existingDeadlineNs);
278                   boolean queueWasEmpty = head.isSingleton();
279                   boolean shouldSignal = false;
280                   switch (eventContext.insertionType) {
281                       case APPEND:
282                           head.insertBefore(eventContext);
283                           if (queueWasEmpty) {
284                               shouldSignal = true;
285                           }
286                           break;
287                       case PREPEND:
288                           head.insertAfter(eventContext);
289                           if (queueWasEmpty) {
290                               shouldSignal = true;
291                           }
292                           break;
293                       case DEFERRED:
294                           if (!deadlineNs.isPresent()) {
295                               return new RuntimeException(
296                                   "You must specify a deadline for deferred events.");
297                           }
298                           break;
299                   }
300                   if (deadlineNs.isPresent()) {
301                       long insertNs = deadlineNs.getAsLong();
302                       long prevStartNs = deadlineMap.isEmpty() ? Long.MAX_VALUE : deadlineMap.firstKey();
303                       // If the time in nanoseconds is already taken, take the next one.
304                       while (deadlineMap.putIfAbsent(insertNs, eventContext) != null) {
305                           insertNs++;
306                       }
307                       eventContext.deadlineNs = OptionalLong.of(insertNs);
308                       // If the new timeout is before all the existing ones, wake up the
309                       // timeout thread.
310                       if (insertNs <= prevStartNs) {
311                           shouldSignal = true;
312                       }
313                   }
314                   if (shouldSignal) {
315                       cond.signal();
316                   }
317               } finally {
318                   lock.unlock();
319               }
320               return null;
321           }
322   
323           void cancelDeferred(String tag) {
324               lock.lock();
325               try {
326                   EventContext eventContext = tagToEventContext.get(tag);
327                   if (eventContext != null) {
328                       remove(eventContext);
329                   }
330               } finally {
331                   lock.unlock();
332               }
333           }
334   
335           void wakeUp() {
336               lock.lock();
337               try {
338                   eventHandler.cond.signal();
339               } finally {
340                   lock.unlock();
341               }
342           }
343       }

345       private final Time time;
346       private final ReentrantLock lock;
347       private final Logger log;
348       private final EventHandler eventHandler;
349       private final Thread eventHandlerThread;
350   
351       /**
352        * The time in monotonic nanoseconds when the queue is closing, or Long.MAX_VALUE if
353        * the queue is not currently closing.
354        */
355       private long closingTimeNs;
356   
357       private Event cleanupEvent;
358   
359       public KafkaEventQueue(Time time,
360                              LogContext logContext,
361                              String threadNamePrefix) {
362           this.time = time;
363           this.lock = new ReentrantLock();
364           this.log = logContext.logger(KafkaEventQueue.class);
365           this.eventHandler = new EventHandler();
366           this.eventHandlerThread = new KafkaThread(threadNamePrefix + "EventHandler",
367               this.eventHandler, false);
368           this.closingTimeNs = Long.MAX_VALUE;
369           this.cleanupEvent = null;
370           this.eventHandlerThread.start();
371       }
372   
373       @Override
374       public void enqueue(EventInsertionType insertionType,
375                           String tag,
376                           Function<OptionalLong, OptionalLong> deadlineNsCalculator,
377                           Event event) {
378           EventContext eventContext = new EventContext(event, insertionType, tag);
379           Exception e = eventHandler.enqueue(eventContext, deadlineNsCalculator);
380           if (e != null) {
381               eventContext.completeWithException(e);
382           }
383       }
384   
385       @Override
386       public void cancelDeferred(String tag) {
387           eventHandler.cancelDeferred(tag);
388       }
389   
390       @Override
391       public void beginShutdown(String source, Event newCleanupEvent,
392                                 long timeSpan, TimeUnit timeUnit) {
393           if (timeSpan < 0) {
394               throw new IllegalArgumentException("beginShutdown must be called with a " +
395                   "non-negative timeout.");
396           }
397           Objects.requireNonNull(newCleanupEvent);
398           lock.lock();
399           try {
400               if (cleanupEvent != null) {
401                   log.debug("{}: Event queue is already shutting down.", source);
402                   return;
403               }
404               log.info("{}: shutting down event queue.", source);
405               cleanupEvent = newCleanupEvent;
406               long newClosingTimeNs = time.nanoseconds() + timeUnit.toNanos(timeSpan);
407               if (closingTimeNs >= newClosingTimeNs)
408                   closingTimeNs = newClosingTimeNs;
409               eventHandler.cond.signal();
410           } finally {
411               lock.unlock();
412           }
413       }
414   
415       @Override
416       public void wakeup() {
417           eventHandler.wakeUp();
418       }
419   
420       @Override
421       public void close() throws InterruptedException {
422           beginShutdown("KafkaEventQueue#close");
423           eventHandlerThread.join();
424           log.info("closed event queue.");
425       }
426   }
