17   package org.apache.kafka.server.common.serialization;
18   
19   import org.apache.kafka.common.protocol.ObjectSerializationCache;
20   import org.apache.kafka.common.protocol.Readable;
21   import org.apache.kafka.common.protocol.Writable;
22   
23   /**
24    * Serde interface for records written to a metadata log. This class assumes
25    * a two-pass serialization, with the first pass used to compute the size of the
26    * serialized record, and the second pass to write the object.
27    */
28   public interface RecordSerde<T> {
29       /**
30        * Get the size of a record. This must be called first before writing
31        * the data through {@link #write(Object, ObjectSerializationCache, Writable)}.
32        *
33        * @param data the record that will be serialized
34        * @param serializationCache serialization cache
35        * @return the size in bytes of the serialized record
36        */
37       int recordSize(T data, ObjectSerializationCache serializationCache);
38   
39       /**
40        * Write the record to the output stream. This must be called after
41        * computing the size with {@link #recordSize(Object, ObjectSerializationCache)}.
42        * The same {@link ObjectSerializationCache} instance must be used in both calls.
43        *
44        * @param data the record to serialize and write
45        * @param serializationCache serialization cache
46        * @param out the output stream to write the record to
47        */
48       void write(T data, ObjectSerializationCache serializationCache, Writable out);
49   
50       /**
51        * Read a record from a {@link Readable} input.
52        *
53        * @param input the input stream to deserialize
54        * @param size the size of the record in bytes
55        * @return the deserialized record
56        */
57       T read(Readable input, int size);
58   }
