38   /**
39    * This class represents a utility to capture a checkpoint in a file. It writes down to the file in the below format.
40    *
41    * ========= File beginning =========
42    * version: int
43    * entries-count: int
44    * entry-as-string-on-each-line
45    * ========= File end ===============
46    *
47    * Each entry is represented as a string on each line in the checkpoint file. {@link EntryFormatter} is used
48    * to convert the entry into a string and vice versa.
49    *
50    * @param <T> entry type.
51    */
52   public class CheckpointFile<T> {
53   
54       private final int version;
55       private final EntryFormatter<T> formatter;
56       private final Object lock = new Object();
57       private final Path absolutePath;
58       private final Path tempPath;
59   
60       public CheckpointFile(File file,
61                             int version,
62                             EntryFormatter<T> formatter) throws IOException {
63           this.version = version;
64           this.formatter = formatter;
65           try {
66               // Create the file if it does not exist.
67               Files.createFile(file.toPath());
68           } catch (FileAlreadyExistsException ex) {
69               // Ignore if file already exists.
70           }
71           absolutePath = file.toPath().toAbsolutePath();
72           tempPath = Paths.get(absolutePath.toString() + ".tmp");
73       }
74   
75       public void write(Collection<T> entries) throws IOException {
76           synchronized (lock) {
77               // write to temp file and then swap with the existing file
78               try (FileOutputStream fileOutputStream = new FileOutputStream(tempPath.toFile());
79                    BufferedWriter writer = new BufferedWriter(new OutputStreamWriter(fileOutputStream, StandardCharsets.UTF_8))) {
80                   // Write the version
81                   writer.write(Integer.toString(version));
82                   writer.newLine();
83   
84                   // Write the entries count
85                   writer.write(Integer.toString(entries.size()));
86                   writer.newLine();
87   
88                   // Write each entry on a new line.
89                   for (T entry : entries) {
90                       writer.write(formatter.toString(entry));
91                       writer.newLine();
92                   }
93   
94                   writer.flush();
95                   fileOutputStream.getFD().sync();
96               }
97   
98               Utils.atomicMoveWithFallback(tempPath, absolutePath);
99           }
100       }
101   
102       public List<T> read() throws IOException {
103           synchronized (lock) {
104               try (BufferedReader reader = Files.newBufferedReader(absolutePath)) {
105                   CheckpointReadBuffer<T> checkpointBuffer = new CheckpointReadBuffer<>(absolutePath.toString(), reader, version, formatter);
106                   return checkpointBuffer.read();
107               }
108           }
109       }
110   
111       private static class CheckpointReadBuffer<T> {
112   
113           private final String location;
114           private final BufferedReader reader;
115           private final int version;
116           private final EntryFormatter<T> formatter;
117   
118           CheckpointReadBuffer(String location,
119                                BufferedReader reader,
120                                int version,
121                                EntryFormatter<T> formatter) {
122               this.location = location;
123               this.reader = reader;
124               this.version = version;
125               this.formatter = formatter;
126           }
127   
128           List<T> read() throws IOException {
129               String line = reader.readLine();
130               if (line == null)
131                   return Collections.emptyList();
132   
133               int readVersion = toInt(line);
134               if (readVersion != version) {
135                   throw new IOException("Unrecognised version:" + readVersion + ", expected version: " + version
136                                                 + " in checkpoint file at: " + location);
137               }
138   
139               line = reader.readLine();
140               if (line == null) {
141                   return Collections.emptyList();
142               }
143               int expectedSize = toInt(line);
144               List<T> entries = new ArrayList<>(expectedSize);
145               line = reader.readLine();
146               while (line != null) {
147                   Optional<T> maybeEntry = formatter.fromString(line);
148                   if (!maybeEntry.isPresent()) {
149                       throw buildMalformedLineException(line);
150                   }
151                   entries.add(maybeEntry.get());
152                   line = reader.readLine();
153               }
154   
155               if (entries.size() != expectedSize) {
156                   throw new IOException("Expected [" + expectedSize + "] entries in checkpoint file ["
157                                                 + location + "], but found only [" + entries.size() + "]");
158               }
159   
160               return entries;
161           }
162   
163           private int toInt(String line) throws IOException {
164               try {
165                   return Integer.parseInt(line);
166               } catch (NumberFormatException e) {
167                   throw buildMalformedLineException(line);
168               }
169           }
170   
171           private IOException buildMalformedLineException(String line) {
172               return new IOException(String.format("Malformed line in checkpoint file [%s]: %s", location, line));
173           }
174       }
175   
176       /**
177        * This is used to convert the given entry of type {@code T} into a string and vice versa.
178        *
179        * @param <T> entry type
180        */
181       public interface EntryFormatter<T> {
182   
183           /**
184            * @param entry entry to be converted into string.
185            * @return String representation of the given entry.
186            */
187           String toString(T entry);
188   
189           /**
190            * @param value string representation of an entry.
191            * @return entry converted from the given string representation if possible. {@link Optional#empty()} represents
192            * that the given string representation could not be converted into an entry.
193            */
194           Optional<T> fromString(String value);
195       }
196   }
