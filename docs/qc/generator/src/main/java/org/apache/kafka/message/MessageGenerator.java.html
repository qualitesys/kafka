177       private static List<TypeClassGenerator> createTypeClassGenerators(String packageName,
178                                                                         List<String> types) {
179           if (types == null) return Collections.emptyList();
180           List<TypeClassGenerator> generators = new ArrayList<>();
181           for (String type : types) {
182               switch (type) {
183                   case "ApiMessageTypeGenerator":
184                       generators.add(new ApiMessageTypeGenerator(packageName));
185                       break;
186                   case "MetadataRecordTypeGenerator":
187                       generators.add(new MetadataRecordTypeGenerator(packageName));
188                       break;
189                   case "MetadataJsonConvertersGenerator":
190                       generators.add(new MetadataJsonConvertersGenerator(packageName));
191                       break;
192                   default:
193                       throw new RuntimeException("Unknown type class generator type '" + type + "'");
194               }
195           }
196           return generators;
197       }

199       private static List<MessageClassGenerator> createMessageClassGenerators(String packageName,
200                                                                               List<String> types) {
201           if (types == null) return Collections.emptyList();
202           List<MessageClassGenerator> generators = new ArrayList<>();
203           for (String type : types) {
204               switch (type) {
205                   case "MessageDataGenerator":
206                       generators.add(new MessageDataGenerator(packageName));
207                       break;
208                   case "JsonConverterGenerator":
209                       generators.add(new JsonConverterGenerator(packageName));
210                       break;
211                   default:
212                       throw new RuntimeException("Unknown message class generator type '" + type + "'");
213               }
214           }
215           return generators;
216       }

218       public static void processDirectories(String packageName,
219                                             String outputDir,
220                                             String inputDir,
221                                             List<String> typeClassGeneratorTypes,
222                                             List<String> messageClassGeneratorTypes) throws Exception {
223           Files.createDirectories(Paths.get(outputDir));
224           int numProcessed = 0;
225   
226           List<TypeClassGenerator> typeClassGenerators =
227                   createTypeClassGenerators(packageName, typeClassGeneratorTypes);
228           HashSet<String> outputFileNames = new HashSet<>();
229           try (DirectoryStream<Path> directoryStream = Files
230                   .newDirectoryStream(Paths.get(inputDir), JSON_GLOB)) {
231               for (Path inputPath : directoryStream) {
232                   try {
233                       MessageSpec spec = JSON_SERDE.
234                           readValue(inputPath.toFile(), MessageSpec.class);
235                       List<MessageClassGenerator> generators =
236                           createMessageClassGenerators(packageName, messageClassGeneratorTypes);
237                       for (MessageClassGenerator generator : generators) {
238                           String name = generator.outputName(spec) + JAVA_SUFFIX;
239                           outputFileNames.add(name);
240                           Path outputPath = Paths.get(outputDir, name);
241                           try (BufferedWriter writer = Files.newBufferedWriter(outputPath)) {
242                               generator.generateAndWrite(spec, writer);
243                           }
244                       }
245                       numProcessed++;
246                       typeClassGenerators.forEach(generator -> generator.registerMessageType(spec));
247                   } catch (Exception e) {
248                       throw new RuntimeException("Exception while processing " + inputPath.toString(), e);
249                   }
250               }
251           }
252           for (TypeClassGenerator typeClassGenerator : typeClassGenerators) {
253               outputFileNames.add(typeClassGenerator.outputName());
254               Path factoryOutputPath = Paths.get(outputDir, typeClassGenerator.outputName());
255               try (BufferedWriter writer = Files.newBufferedWriter(factoryOutputPath)) {
256                   typeClassGenerator.generateAndWrite(writer);
257               }
258           }
259           try (DirectoryStream<Path> directoryStream = Files.
260                   newDirectoryStream(Paths.get(outputDir))) {
261               for (Path outputPath : directoryStream) {
262                   Path fileName = outputPath.getFileName();
263                   if (fileName != null) {
264                       if (!outputFileNames.contains(fileName.toString())) {
265                           Files.delete(outputPath);
266                       }
267                   }
268               }
269           }
270           System.out.printf("MessageGenerator: processed %d Kafka message JSON files(s).%n", numProcessed);
271       }

296       static String toSnakeCase(String string) {
297           StringBuilder bld = new StringBuilder();
298           boolean prevWasCapitalized = true;
299           for (int i = 0; i < string.length(); i++) {
300               char c = string.charAt(i);
301               if (Character.isUpperCase(c)) {
302                   if (!prevWasCapitalized) {
303                       bld.append('_');
304                   }
305                   bld.append(Character.toLowerCase(c));
306                   prevWasCapitalized = true;
307               } else {
308                   bld.append(c);
309                   prevWasCapitalized = false;
310               }
311           }
312           return bld.toString();
313       }

315       static String stripSuffix(String str, String suffix) {
316           if (str.endsWith(suffix)) {
317               return str.substring(0, str.length() - suffix.length());
318           } else {
319               throw new RuntimeException("String " + str + " does not end with the " +
320                   "expected suffix " + suffix);
321           }
322       }

324       /**
325        * Return the number of bytes needed to encode an integer in unsigned variable-length format.
326        */
327       static int sizeOfUnsignedVarint(int value) {
328           int bytes = 1;
329           while ((value & 0xffffff80) != 0L) {
330               bytes += 1;
331               value >>>= 7;
332           }
333           return bytes;
334       }

336       public static void main(String[] args) throws Exception {
337           ArgumentParser parser = ArgumentParsers
338               .newArgumentParser("message-generator")
339               .defaultHelp(true)
340               .description("The Kafka message generator");
341           parser.addArgument("--package", "-p")
342               .action(store())
343               .required(true)
344               .metavar("PACKAGE")
345               .help("The java package to use in generated files.");
346           parser.addArgument("--output", "-o")
347               .action(store())
348               .required(true)
349               .metavar("OUTPUT")
350               .help("The output directory to create.");
351           parser.addArgument("--input", "-i")
352               .action(store())
353               .required(true)
354               .metavar("INPUT")
355               .help("The input directory to use.");
356           parser.addArgument("--typeclass-generators", "-t")
357               .nargs("+")
358               .action(store())
359               .metavar("TYPECLASS_GENERATORS")
360               .help("The type class generators to use, if any.");
361           parser.addArgument("--message-class-generators", "-m")
362               .nargs("+")
363               .action(store())
364               .metavar("MESSAGE_CLASS_GENERATORS")
365               .help("The message class generators to use.");
366           Namespace res = parser.parseArgsOrFail(args);
367           processDirectories(res.getString("package"), res.getString("output"),
368               res.getString("input"), res.getList("typeclass_generators"),
369               res.getList("message_class_generators"));
370       }
