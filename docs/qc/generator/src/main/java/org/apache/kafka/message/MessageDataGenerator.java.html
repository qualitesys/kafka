57       void generate(MessageSpec message) throws Exception {
58           if (message.struct().versions().contains(Short.MAX_VALUE)) {
59               throw new RuntimeException("Message " + message.name() + " does " +
60                   "not specify a maximum version.");
61           }
62           structRegistry.register(message);
63           schemaGenerator.generateSchemas(message);
64           messageFlexibleVersions = message.flexibleVersions();
65           generateClass(Optional.of(message),
66               message.dataClassName(),
67               message.struct(),
68               message.struct().versions());
69           headerGenerator.generate();
70       }

73           headerGenerator.buffer().write(writer);

77       private void generateClass(Optional<MessageSpec> topLevelMessageSpec,
78                                  String className,
79                                  StructSpec struct,
80                                  Versions parentVersions) throws Exception {
81           buffer.printf("%n");
82           boolean isTopLevel = topLevelMessageSpec.isPresent();
83           boolean isSetElement = struct.hasKeys(); // Check if the class is inside a set.
84           if (isTopLevel && isSetElement) {
85               throw new RuntimeException("Cannot set mapKey on top level fields.");
86           }
87           generateClassHeader(className, isTopLevel, isSetElement);
88           buffer.incrementIndent();
89           generateFieldDeclarations(struct, isSetElement);
90           buffer.printf("%n");
91           schemaGenerator.writeSchema(className, buffer);
92           generateClassConstructors(className, struct, isSetElement);
93           buffer.printf("%n");
94           if (isTopLevel) {
95               generateShortAccessor("apiKey", topLevelMessageSpec.get().apiKey().orElse((short) -1));
96           }
97           buffer.printf("%n");
98           generateShortAccessor("lowestSupportedVersion", parentVersions.lowest());
99           buffer.printf("%n");
100           generateShortAccessor("highestSupportedVersion", parentVersions.highest());
101           buffer.printf("%n");
102           generateClassReader(className, struct, parentVersions);
103           buffer.printf("%n");
104           generateClassWriter(className, struct, parentVersions);
105           buffer.printf("%n");
106           generateClassMessageSize(className, struct, parentVersions);
107           if (isSetElement) {
108               buffer.printf("%n");
109               generateClassEquals(className, struct, true);
110           }
111           buffer.printf("%n");
112           generateClassEquals(className, struct, false);
113           buffer.printf("%n");
114           generateClassHashCode(struct, isSetElement);
115           buffer.printf("%n");
116           generateClassDuplicate(className, struct);
117           buffer.printf("%n");
118           generateClassToString(className, struct);
119           generateFieldAccessors(struct, isSetElement);
120           buffer.printf("%n");
121           generateUnknownTaggedFieldsAccessor();
122           generateFieldMutators(struct, className, isSetElement);
123   
124           if (!isTopLevel) {
125               buffer.decrementIndent();
126               buffer.printf("}%n");
127           }
128           generateSubclasses(className, struct, parentVersions, isSetElement);
129           if (isTopLevel) {
130               for (Iterator<StructSpec> iter = structRegistry.commonStructs(); iter.hasNext(); ) {
131                   StructSpec commonStruct = iter.next();
132                   generateClass(Optional.empty(),
133                           commonStruct.name(),
134                           commonStruct,
135                           commonStruct.versions());
136               }
137               buffer.decrementIndent();
138               buffer.printf("}%n");
139           }
140       }

142       private void generateClassHeader(String className, boolean isTopLevel,
143                                        boolean isSetElement) {
144           Set<String> implementedInterfaces = new HashSet<>();
145           if (isTopLevel) {
146               implementedInterfaces.add("ApiMessage");
147               headerGenerator.addImport(MessageGenerator.API_MESSAGE_CLASS);
148           } else {
149               implementedInterfaces.add("Message");
150               headerGenerator.addImport(MessageGenerator.MESSAGE_CLASS);
151           }
152           if (isSetElement) {
153               headerGenerator.addImport(MessageGenerator.IMPLICIT_LINKED_HASH_MULTI_COLLECTION_CLASS);
154               implementedInterfaces.add("ImplicitLinkedHashMultiCollection.Element");
155           }
156           Set<String> classModifiers = new LinkedHashSet<>();
157           classModifiers.add("public");
158           if (!isTopLevel) {
159               classModifiers.add("static");
160           }
161           buffer.printf("%s class %s implements %s {%n",
162               String.join(" ", classModifiers),
163               className,
164               String.join(", ", implementedInterfaces));
165       }

167       private void generateSubclasses(String className, StructSpec struct,
168               Versions parentVersions, boolean isSetElement) throws Exception {
169           for (FieldSpec field : struct.fields()) {
170               if (field.type().isStructArray()) {
171                   FieldType.ArrayType arrayType = (FieldType.ArrayType) field.type();
172                   if (!structRegistry.commonStructNames().contains(arrayType.elementName())) {
173                       generateClass(Optional.empty(),
174                               arrayType.elementType().toString(),
175                               structRegistry.findStruct(field),
176                               parentVersions.intersect(struct.versions()));
177                   }
178               } else if (field.type().isStruct()) {
179                   if (!structRegistry.commonStructNames().contains(field.typeString())) {
180                       generateClass(Optional.empty(),
181                               field.typeString(),
182                               structRegistry.findStruct(field),
183                               parentVersions.intersect(struct.versions()));
184                   }
185               }
186           }
187           if (isSetElement) {
188               generateHashSet(className, struct);
189           }
190       }

264       private void generateKeyElement(String className, StructSpec struct) {
265           buffer.printf("%s _key = new %s();%n", className, className);
266           for (FieldSpec field : struct.fields()) {
267               if (field.mapKey()) {
268                   buffer.printf("_key.set%s(%s);%n",
269                       field.capitalizedCamelCaseName(),
270                       field.camelCaseName());
271               }
272           }

276           return struct.fields().stream().

280               collect(Collectors.joining(", "));

408       private void generateClassReader(String className, StructSpec struct,
409                                        Versions parentVersions) {
410           headerGenerator.addImport(MessageGenerator.READABLE_CLASS);
411           buffer.printf("@Override%n");
412           buffer.printf("public void read(Readable _readable, short _version) {%n");
413           buffer.incrementIndent();
414           VersionConditional.forVersions(parentVersions, struct.versions()).
415               allowMembershipCheckAlwaysFalse(false).
416               ifNotMember(__ -> {
417                   headerGenerator.addImport(MessageGenerator.UNSUPPORTED_VERSION_EXCEPTION_CLASS);
418                   buffer.printf("throw new UnsupportedVersionException(\"Can't read " +
419                       "version \" + _version + \" of %s\");%n", className);
420               }).
421               generate(buffer);
422           Versions curVersions = parentVersions.intersect(struct.versions());
423           for (FieldSpec field : struct.fields()) {
424               Versions fieldFlexibleVersions = fieldFlexibleVersions(field);
425               if (!field.taggedVersions().intersect(fieldFlexibleVersions).equals(field.taggedVersions())) {
426                   throw new RuntimeException("Field " + field.name() + " specifies tagged " +
427                       "versions " + field.taggedVersions() + " that are not a subset of the " +
428                       "flexible versions " + fieldFlexibleVersions);
429               }
430               Versions mandatoryVersions = field.versions().subtract(field.taggedVersions());
431               VersionConditional.forVersions(mandatoryVersions, curVersions).
432                   alwaysEmitBlockScope(field.type().isVariableLength()).
433                   ifNotMember(__ -> {
434                       // If the field is not present, or is tagged, set it to its default here.
435                       buffer.printf("this.%s = %s;%n", field.camelCaseName(),
436                           field.fieldDefault(headerGenerator, structRegistry));
437                   }).
438                   ifMember(presentAndUntaggedVersions -> {
439                       if (field.type().isVariableLength() && !field.type().isStruct()) {
440                           ClauseGenerator callGenerateVariableLengthReader = versions -> {
441                               generateVariableLengthReader(fieldFlexibleVersions(field),
442                                   field.camelCaseName(),
443                                   field.type(),
444                                   versions,
445                                   field.nullableVersions(),
446                                   String.format("this.%s = ", field.camelCaseName()),
447                                   String.format(";%n"),
448                                   structRegistry.isStructArrayWithKeys(field),
449                                   field.zeroCopy());
450                           };
451                           // For arrays where the field type needs to be serialized differently in flexible
452                           // versions, lift the flexible version check outside of the array.
453                           // This may mean generating two separate 'for' loops-- one for flexible
454                           // versions, and one for regular versions.
455                           if (field.type().isArray() &&
456                               ((FieldType.ArrayType) field.type()).elementType().
457                                   serializationIsDifferentInFlexibleVersions()) {
458                               VersionConditional.forVersions(fieldFlexibleVersions(field),
459                                       presentAndUntaggedVersions).
460                                   ifMember(callGenerateVariableLengthReader).
461                                   ifNotMember(callGenerateVariableLengthReader).
462                                   generate(buffer);
463                           } else {
464                               callGenerateVariableLengthReader.generate(presentAndUntaggedVersions);
465                           }
466                       } else {
467                           buffer.printf("this.%s = %s;%n", field.camelCaseName(),
468                               primitiveReadExpression(field.type()));
469                       }
470                   }).
471                   generate(buffer);
472           }
473           buffer.printf("this._unknownTaggedFields = null;%n");
474           VersionConditional.forVersions(messageFlexibleVersions, curVersions).
475               ifMember(curFlexibleVersions -> {
476                   buffer.printf("int _numTaggedFields = _readable.readUnsignedVarint();%n");
477                   buffer.printf("for (int _i = 0; _i < _numTaggedFields; _i++) {%n");
478                   buffer.incrementIndent();
479                   buffer.printf("int _tag = _readable.readUnsignedVarint();%n");
480                   buffer.printf("int _size = _readable.readUnsignedVarint();%n");
481                   buffer.printf("switch (_tag) {%n");
482                   buffer.incrementIndent();
483                   for (FieldSpec field : struct.fields()) {
484                       Versions validTaggedVersions = field.versions().intersect(field.taggedVersions());
485                       if (!validTaggedVersions.empty()) {
486                           if (!field.tag().isPresent()) {
487                               throw new RuntimeException("Field " + field.name() + " has tagged versions, but no tag.");
488                           }
