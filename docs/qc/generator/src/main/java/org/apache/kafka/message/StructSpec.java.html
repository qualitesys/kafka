18   package org.apache.kafka.message;
19   
20   import com.fasterxml.jackson.annotation.JsonCreator;
21   import com.fasterxml.jackson.annotation.JsonProperty;
22   
23   import java.util.ArrayList;
24   import java.util.Collections;
25   import java.util.HashSet;
26   import java.util.List;
27   import java.util.Objects;
28   
29   public final class StructSpec {
30       private final String name;
31   
32       private final Versions versions;
33   
34       private final List<FieldSpec> fields;
35   
36       private final boolean hasKeys;
37   
38       @JsonCreator
39       public StructSpec(@JsonProperty("name") String name,
40                         @JsonProperty("versions") String versions,
41                         @JsonProperty("fields") List<FieldSpec> fields) {
42           this.name = Objects.requireNonNull(name);
43           this.versions = Versions.parse(versions, null);
44           if (this.versions == null) {
45               throw new RuntimeException("You must specify the version of the " +
46                       name + " structure.");
47           }
48           ArrayList<FieldSpec> newFields = new ArrayList<>();
49           if (fields != null) {
50               // Each field should have a unique tag ID (if the field has a tag ID).
51               HashSet<Integer> tags = new HashSet<>();
52               for (FieldSpec field : fields) {
53                   if (field.tag().isPresent()) {
54                       if (tags.contains(field.tag().get())) {
55                           throw new RuntimeException("In " + name + ", field " + field.name() +
56                               " has a duplicate tag ID " + field.tag().get() + ".  All tags IDs " +
57                               "must be unique.");
58                       }
59                       tags.add(field.tag().get());
60                   }
61                   newFields.add(field);
62               }
63               // Tag IDs should be contiguous and start at 0.  This optimizes space on the wire,
64               // since larger numbers take more space.
65               for (int i = 0; i < tags.size(); i++) {
66                   if (!tags.contains(i)) {
67                       throw new RuntimeException("In " + name + ", the tag IDs are not " +
68                           "contiguous.  Make use of tag " + i + " before using any " +
69                           "higher tag IDs.");
70                   }
71               }
72           }
73           this.fields = Collections.unmodifiableList(newFields);
74           this.hasKeys = this.fields.stream().anyMatch(f -> f.mapKey());
75       }
76   
77       @JsonProperty
78       public String name() {
79           return name;
80       }
81   
82       public Versions versions() {
83           return versions;
84       }
85   
86       @JsonProperty
87       public String versionsString() {
88           return versions.toString();
89       }
90   
91       @JsonProperty
92       public List<FieldSpec> fields() {
93           return fields;
94       }
95   
96       boolean hasKeys() {
97           return hasKeys;
98       }
99   }
