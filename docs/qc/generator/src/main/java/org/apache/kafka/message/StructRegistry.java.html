20   import java.util.Iterator;
21   import java.util.List;

27   /**
28    * Contains structure data for Kafka MessageData classes.
29    */
30   final class StructRegistry {
31       private final Map<String, StructInfo> structs;
32       private final Set<String> commonStructNames;
33   
34       static class StructInfo {
35           /**
36            * The specification for this structure.
37            */
38           private final StructSpec spec;
39   
40           /**
41            * The versions which the parent(s) of this structure can have.  If this is a
42            * top-level structure, this will be equal to the versions which the
43            * overall message can have.
44            */
45           private final Versions parentVersions;
46   
47           StructInfo(StructSpec spec, Versions parentVersions) {
48               this.spec = spec;
49               this.parentVersions = parentVersions;
50           }
51   
52           public StructSpec spec() {
53               return spec;
54           }
55   
56           public Versions parentVersions() {
57               return parentVersions;
58           }
59       }
60   
61       StructRegistry() {
62           this.structs = new TreeMap<>();
63           this.commonStructNames = new TreeSet<>();
64       }
65   
66       /**
67        * Register all the structures contained a message spec.
68        */
69       void register(MessageSpec message) throws Exception {
70           // Register common structures.
71           for (StructSpec struct : message.commonStructs()) {
72               if (!MessageGenerator.firstIsCapitalized(struct.name())) {
73                   throw new RuntimeException("Can't process structure " + struct.name() +
74                           ": the first letter of structure names must be capitalized.");
75               }
76               if (structs.containsKey(struct.name())) {
77                   throw new RuntimeException("Common struct " + struct.name() + " was specified twice.");
78               }
79               structs.put(struct.name(), new StructInfo(struct, struct.versions()));
80               commonStructNames.add(struct.name());
81           }
82           // Register inline structures.
83           addStructSpecs(message.validVersions(), message.fields());
84       }
85   
86       @SuppressWarnings("unchecked")
87       private void addStructSpecs(Versions parentVersions, List<FieldSpec> fields) {
88           for (FieldSpec field : fields) {
89               String typeName = null;
90               if (field.type().isStructArray()) {
91                   FieldType.ArrayType arrayType = (FieldType.ArrayType) field.type();
92                   typeName = arrayType.elementName();
93               } else if (field.type().isStruct()) {
94                   FieldType.StructType structType = (FieldType.StructType) field.type();
95                   typeName = structType.typeName();
96               }
97               if (typeName != null) {
98                   if (commonStructNames.contains(typeName)) {
99                       // If we're using a common structure, we can't specify its fields.
100                       // The fields should be specified in the commonStructs area.
101                       if (!field.fields().isEmpty()) {
102                           throw new RuntimeException("Can't re-specify the common struct " +
103                                   typeName + " as an inline struct.");
104                       }
105                   } else if (structs.containsKey(typeName)) {
106                       // Inline structures should only appear once.
107                       throw new RuntimeException("Struct " + typeName +
108                           " was specified twice.");
109                   } else {
110                       // Synthesize a StructSpec object out of the fields.
111                       StructSpec spec = new StructSpec(typeName,
112                           field.versions().toString(),
113                           field.fields());
114                       structs.put(typeName, new StructInfo(spec, parentVersions));
115                   }
116   
117                   addStructSpecs(parentVersions.intersect(field.versions()), field.fields());
118               }
119           }
120       }
121   
122       /**
123        * Locate the struct corresponding to a field.
124        */
125       @SuppressWarnings("unchecked")
126       StructSpec findStruct(FieldSpec field) {
127           String structFieldName;
128           if (field.type().isArray()) {
129               FieldType.ArrayType arrayType = (FieldType.ArrayType) field.type();
130               structFieldName = arrayType.elementName();
131           } else if (field.type().isStruct()) {
132               FieldType.StructType structType = (FieldType.StructType) field.type();
133               structFieldName = structType.typeName();
134           } else {
135               throw new RuntimeException("Field " + field.name() +
136                       " cannot be treated as a structure.");
137           }
138           StructInfo structInfo = structs.get(structFieldName);
139           if (structInfo == null) {
140               throw new RuntimeException("Unable to locate a specification for the structure " +
141                       structFieldName);
142           }
143           return structInfo.spec;
144       }
145   
146       /**
147        * Return true if the field is a struct array with keys.
148        */
149       @SuppressWarnings("unchecked")
150       boolean isStructArrayWithKeys(FieldSpec field) {
151           if (!field.type().isArray()) {
152               return false;
153           }
154           FieldType.ArrayType arrayType = (FieldType.ArrayType) field.type();
155           if (!arrayType.isStructArray()) {
156               return false;
157           }
158           StructInfo structInfo = structs.get(arrayType.elementName());
159           if (structInfo == null) {
160               throw new RuntimeException("Unable to locate a specification for the structure " +
161                       arrayType.elementName());
162           }
163           return structInfo.spec.hasKeys();
164       }
165   
166       Set<String> commonStructNames() {
167           return commonStructNames;
168       }
169   
170       /**
171        * Returns an iterator that will step through all the common structures.
172        */
173       Iterator<StructSpec> commonStructs() {
174           return new Iterator<StructSpec>() {
175               private final Iterator<String> iter = commonStructNames.iterator();
176   
177               @Override
178               public boolean hasNext() {
179                   return iter.hasNext();
180               }
181   
182               @Override
183               public StructSpec next() {
184                   return structs.get(iter.next()).spec;
185               }
186           };
187       }
188   
189       Iterator<StructInfo> structs() {
190           return structs.values().iterator();
191       }
