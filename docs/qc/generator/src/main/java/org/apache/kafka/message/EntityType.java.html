18   package org.apache.kafka.message;
19   
20   import com.fasterxml.jackson.annotation.JsonProperty;
21   
22   public enum EntityType {
23       @JsonProperty("unknown")
24       UNKNOWN(null),
25   
26       @JsonProperty("transactionalId")
27       TRANSACTIONAL_ID(FieldType.StringFieldType.INSTANCE),
28   
29       @JsonProperty("producerId")
30       PRODUCER_ID(FieldType.Int64FieldType.INSTANCE),
31   
32       @JsonProperty("groupId")
33       GROUP_ID(FieldType.StringFieldType.INSTANCE),
34   
35       @JsonProperty("topicName")
36       TOPIC_NAME(FieldType.StringFieldType.INSTANCE),
37   
38       @JsonProperty("brokerId")
39       BROKER_ID(FieldType.Int32FieldType.INSTANCE);
40   
41       private final FieldType baseType;
42   
43       EntityType(FieldType baseType) {
44           this.baseType = baseType;
45       }
46   
47       public void verifyTypeMatches(String fieldName, FieldType type) {
48           if (this == UNKNOWN) {
49               return;
50           }
51           if (type instanceof FieldType.ArrayType) {
52               FieldType.ArrayType arrayType = (FieldType.ArrayType) type;
53               verifyTypeMatches(fieldName, arrayType.elementType());
54           } else {
55               if (!type.toString().equals(baseType.toString())) {
56                   throw new RuntimeException("Field " + fieldName + " has entity type " +
57                       name() + ", but field type " + type.toString() + ", which does " +
58                       "not match.");
59               }
60           }
61       }
62   }
