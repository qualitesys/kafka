41   /**
42    * This class implements the protocol for Kafka Connect workers in a group. It includes the format of worker state used when
43    * joining the group and distributing assignments, and the format of assignments of connectors and tasks to workers.
44    */
45   public class ConnectProtocol {
46       public static final String VERSION_KEY_NAME = "version";
47       public static final String URL_KEY_NAME = "url";
48       public static final String CONFIG_OFFSET_KEY_NAME = "config-offset";
49       public static final String CONNECTOR_KEY_NAME = "connector";
50       public static final String LEADER_KEY_NAME = "leader";
51       public static final String LEADER_URL_KEY_NAME = "leader-url";
52       public static final String ERROR_KEY_NAME = "error";
53       public static final String TASKS_KEY_NAME = "tasks";
54       public static final String ASSIGNMENT_KEY_NAME = "assignment";
55       public static final int CONNECTOR_TASK = -1;
56   
57       public static final short CONNECT_PROTOCOL_V0 = 0;
58       public static final Schema CONNECT_PROTOCOL_HEADER_SCHEMA = new Schema(
59               new Field(VERSION_KEY_NAME, Type.INT16));
60   
61       /**
62        * Connect Protocol Header V0:
63        * <pre>
64        *   Version            => Int16
65        * </pre>
66        */
67       private static final Struct CONNECT_PROTOCOL_HEADER_V0 = new Struct(CONNECT_PROTOCOL_HEADER_SCHEMA)
68               .set(VERSION_KEY_NAME, CONNECT_PROTOCOL_V0);
69   
70       /**
71        * Config State V0:
72        * <pre>
73        *   Url                => [String]
74        *   ConfigOffset       => Int64
75        * </pre>
76        */
77       public static final Schema CONFIG_STATE_V0 = new Schema(
78               new Field(URL_KEY_NAME, Type.STRING),
79               new Field(CONFIG_OFFSET_KEY_NAME, Type.INT64));
80   
81       /**
82        * Connector Assignment V0:
83        * <pre>
84        *   Connector          => [String]
85        *   Tasks              => [Int32]
86        * </pre>
87        *
88        * <p>Assignments for each worker are a set of connectors and tasks. These are categorized by
89        * connector ID. A sentinel task ID (CONNECTOR_TASK) is used to indicate the connector itself
90        * (i.e. that the assignment includes responsibility for running the Connector instance in
91        * addition to any tasks it generates).</p>
92        */
93       public static final Schema CONNECTOR_ASSIGNMENT_V0 = new Schema(
94               new Field(CONNECTOR_KEY_NAME, Type.STRING),
95               new Field(TASKS_KEY_NAME, new ArrayOf(Type.INT32)));
96   
97       /**
98        * Assignment V0:
99        * <pre>
100        *   Error              => Int16
101        *   Leader             => [String]
102        *   LeaderUrl          => [String]
103        *   ConfigOffset       => Int64
104        *   Assignment         => [Connector Assignment]
105        * </pre>
106        */
107       public static final Schema ASSIGNMENT_V0 = new Schema(
108               new Field(ERROR_KEY_NAME, Type.INT16),
109               new Field(LEADER_KEY_NAME, Type.STRING),
110               new Field(LEADER_URL_KEY_NAME, Type.STRING),
111               new Field(CONFIG_OFFSET_KEY_NAME, Type.INT64),
112               new Field(ASSIGNMENT_KEY_NAME, new ArrayOf(CONNECTOR_ASSIGNMENT_V0)));
113   
114       /**
115        * The fields are serialized in sequence as follows:
116        * Subscription V0:
117        * <pre>
118        *   Version            => Int16
119        *   Url                => [String]
120        *   ConfigOffset       => Int64
121        * </pre>
122        *
123        * @param workerState the current state of the worker metadata
124        * @return the serialized state of the worker metadata
125        */
126       public static ByteBuffer serializeMetadata(WorkerState workerState) {
127           Struct struct = new Struct(CONFIG_STATE_V0);
128           struct.set(URL_KEY_NAME, workerState.url());
129           struct.set(CONFIG_OFFSET_KEY_NAME, workerState.offset());
130           ByteBuffer buffer = ByteBuffer.allocate(CONNECT_PROTOCOL_HEADER_V0.sizeOf() + CONFIG_STATE_V0.sizeOf(struct));
131           CONNECT_PROTOCOL_HEADER_V0.writeTo(buffer);
132           CONFIG_STATE_V0.write(buffer, struct);
133           buffer.flip();
134           return buffer;
135       }
136   
137       /**
138        * Returns the collection of Connect protocols that are supported by this version along
139        * with their serialized metadata. The protocols are ordered by preference.
140        *
141        * @param workerState the current state of the worker metadata
142        * @return the collection of Connect protocol metadata
143        */
144       public static JoinGroupRequestProtocolCollection metadataRequest(WorkerState workerState) {
145           return new JoinGroupRequestProtocolCollection(Collections.singleton(
146                   new JoinGroupRequestProtocol()
147                           .setName(EAGER.protocol())
148                           .setMetadata(ConnectProtocol.serializeMetadata(workerState).array()))
149                   .iterator());
150       }
151   
152       /**
153        * Given a byte buffer that contains protocol metadata return the deserialized form of the
154        * metadata.
155        *
156        * @param buffer A buffer containing the protocols metadata
157        * @return the deserialized metadata
158        * @throws SchemaException on incompatible Connect protocol version
159        */
160       public static WorkerState deserializeMetadata(ByteBuffer buffer) {
161           Struct header = CONNECT_PROTOCOL_HEADER_SCHEMA.read(buffer);
162           Short version = header.getShort(VERSION_KEY_NAME);
163           checkVersionCompatibility(version);
164           Struct struct = CONFIG_STATE_V0.read(buffer);
165           long configOffset = struct.getLong(CONFIG_OFFSET_KEY_NAME);
166           String url = struct.getString(URL_KEY_NAME);
167           return new WorkerState(url, configOffset);
168       }
169   
170       /**
171        * The fields are serialized in sequence as follows:
172        * Complete Assignment V0:
173        * <pre>
174        *   Version            => Int16
175        *   Error              => Int16
176        *   Leader             => [String]
177        *   LeaderUrl          => [String]
178        *   ConfigOffset       => Int64
179        *   Assignment         => [Connector Assignment]
180        * </pre>
181        */
182       public static ByteBuffer serializeAssignment(Assignment assignment) {
183           Struct struct = new Struct(ASSIGNMENT_V0);
184           struct.set(ERROR_KEY_NAME, assignment.error());
185           struct.set(LEADER_KEY_NAME, assignment.leader());
186           struct.set(LEADER_URL_KEY_NAME, assignment.leaderUrl());
187           struct.set(CONFIG_OFFSET_KEY_NAME, assignment.offset());
188           List<Struct> taskAssignments = new ArrayList<>();
189           for (Map.Entry<String, Collection<Integer>> connectorEntry : assignment.asMap().entrySet()) {
190               Struct taskAssignment = new Struct(CONNECTOR_ASSIGNMENT_V0);
191               taskAssignment.set(CONNECTOR_KEY_NAME, connectorEntry.getKey());
192               Collection<Integer> tasks = connectorEntry.getValue();
193               taskAssignment.set(TASKS_KEY_NAME, tasks.toArray());
194               taskAssignments.add(taskAssignment);
195           }
196           struct.set(ASSIGNMENT_KEY_NAME, taskAssignments.toArray());
197   
198           ByteBuffer buffer = ByteBuffer.allocate(CONNECT_PROTOCOL_HEADER_V0.sizeOf() + ASSIGNMENT_V0.sizeOf(struct));
199           CONNECT_PROTOCOL_HEADER_V0.writeTo(buffer);
200           ASSIGNMENT_V0.write(buffer, struct);
201           buffer.flip();
202           return buffer;
203       }

205       /**
206        * Given a byte buffer that contains an assignment as defined by this protocol, return the
207        * deserialized form of the assignment.
208        *
209        * @param buffer the buffer containing a serialized assignment
210        * @return the deserialized assignment
211        * @throws SchemaException on incompatible Connect protocol version
212        */
213       public static Assignment deserializeAssignment(ByteBuffer buffer) {
214           Struct header = CONNECT_PROTOCOL_HEADER_SCHEMA.read(buffer);
215           Short version = header.getShort(VERSION_KEY_NAME);
216           checkVersionCompatibility(version);
217           Struct struct = ASSIGNMENT_V0.read(buffer);
218           short error = struct.getShort(ERROR_KEY_NAME);
219           String leader = struct.getString(LEADER_KEY_NAME);
220           String leaderUrl = struct.getString(LEADER_URL_KEY_NAME);
221           long offset = struct.getLong(CONFIG_OFFSET_KEY_NAME);
222           List<String> connectorIds = new ArrayList<>();
223           List<ConnectorTaskId> taskIds = new ArrayList<>();
224           for (Object structObj : struct.getArray(ASSIGNMENT_KEY_NAME)) {
225               Struct assignment = (Struct) structObj;
226               String connector = assignment.getString(CONNECTOR_KEY_NAME);
227               for (Object taskIdObj : assignment.getArray(TASKS_KEY_NAME)) {
228                   Integer taskId = (Integer) taskIdObj;
229                   if (taskId == CONNECTOR_TASK)
230                       connectorIds.add(connector);
231                   else
232                       taskIds.add(new ConnectorTaskId(connector, taskId));
233               }
234           }
235           return new Assignment(error, leader, leaderUrl, offset, connectorIds, taskIds);
236       }

272       /**
273        * The basic assignment of connectors and tasks introduced with V0 version of the Connect protocol.
274        */
275       public static class Assignment {
276           public static final short NO_ERROR = 0;
277           // Configuration offsets mismatched in a way that the leader could not resolve. Workers should read to the end
278           // of the config log and try to re-join
279           public static final short CONFIG_MISMATCH = 1;
280   
281           private final short error;
282           private final String leader;
283           private final String leaderUrl;
284           private final long offset;
285           private final Collection<String> connectorIds;
286           private final Collection<ConnectorTaskId> taskIds;
287   
288           /**
289            * Create an assignment indicating responsibility for the given connector instances and task Ids.
290            *
291            * @param error error code for this assignment; {@code ConnectProtocol.Assignment.NO_ERROR}
292            *              indicates no error during assignment
293            * @param leader Connect group's leader Id; may be null only on the empty assignment
294            * @param leaderUrl Connect group's leader URL; may be null only on the empty assignment
295            * @param configOffset the most up-to-date configuration offset according to this assignment
296            * @param connectorIds list of connectors that the worker should instantiate and run; may not be null
297            * @param taskIds list of task IDs that the worker should instantiate and run; may not be null
298            */
299           public Assignment(short error, String leader, String leaderUrl, long configOffset,
300                             Collection<String> connectorIds, Collection<ConnectorTaskId> taskIds) {
301               this.error = error;
302               this.leader = leader;
303               this.leaderUrl = leaderUrl;
304               this.offset = configOffset;
305               this.connectorIds = Objects.requireNonNull(connectorIds,
306                       "Assigned connector IDs may be empty but not null");
307               this.taskIds = Objects.requireNonNull(taskIds,
308                       "Assigned task IDs may be empty but not null");
309           }
310   
311           /**
312            * Return the error code of this assignment; 0 signals successful assignment ({@code ConnectProtocol.Assignment.NO_ERROR}).
313            *
314            * @return the error code of the assignment
315            */
316           public short error() {
317               return error;
318           }
319   
320           /**
321            * Return the ID of the leader Connect worker in this assignment.
322            *
323            * @return the ID of the leader
324            */
325           public String leader() {
326               return leader;
327           }
328   
329           /**
330            * Return the URL to which the leader accepts requests from other members of the group.
331            *
332            * @return the leader URL
333            */
334           public String leaderUrl() {
335               return leaderUrl;
336           }
337   
338           /**
339            * Check if this assignment failed.
340            *
341            * @return true if this assignment failed; false otherwise
342            */

345           }
346   
347           /**
348            * Return the most up-to-date offset in the configuration topic according to this assignment

361           public Collection<String> connectors() {

363           }

374           @Override

376               return "Assignment{" +
377                       "error=" + error +
378                       ", leader='" + leader + '\'' +
379                       ", leaderUrl='" + leaderUrl + '\'' +
380                       ", offset=" + offset +

386           protected Map<String, Collection<Integer>> asMap() {
387               // Using LinkedHashMap preserves the ordering, which is helpful for tests and debugging
388               Map<String, Collection<Integer>> taskMap = new LinkedHashMap<>();
389               for (String connectorId : new HashSet<>(connectorIds)) {
390                   taskMap.computeIfAbsent(connectorId, key -> new ArrayList<>()).add(CONNECTOR_TASK);
391               }
392               for (ConnectorTaskId taskId : taskIds) {
393                   String connectorId = taskId.connector();
394                   taskMap.computeIfAbsent(connectorId, key -> new ArrayList<>()).add(taskId.task());
395               }
396               return taskMap;
397           }
398       }

400       private static void checkVersionCompatibility(short version) {
401           // check for invalid versions
402           if (version < CONNECT_PROTOCOL_V0)
403               throw new SchemaException("Unsupported subscription version: " + version);
404   
405           // otherwise, assume versions can be parsed as V0
406       }
