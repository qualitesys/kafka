61   @Path("/connector-plugins")
62   @Produces(MediaType.APPLICATION_JSON)
63   @Consumes(MediaType.APPLICATION_JSON)
64   public class ConnectorPluginsResource {
65   
66       private static final String ALIAS_SUFFIX = "Connector";
67       private final Herder herder;
68       private final List<PluginInfo> connectorPlugins;
69   
70       static final List<Class<? extends SinkConnector>> SINK_CONNECTOR_EXCLUDES = Arrays.asList(
71               VerifiableSinkConnector.class,
72               MockSinkConnector.class
73       );
74   
75       static final List<Class<? extends SourceConnector>> SOURCE_CONNECTOR_EXCLUDES = Arrays.asList(
76               VerifiableSourceConnector.class,
77               MockSourceConnector.class,
78               SchemaSourceConnector.class
79       );
80   
81       @SuppressWarnings({"unchecked", "rawtypes"})
82       static final List<Class<? extends Transformation<?>>> TRANSFORM_EXCLUDES = Collections.singletonList(
83               (Class) PredicatedTransformation.class
84       );
85   
86       public ConnectorPluginsResource(Herder herder) {
87           this.herder = herder;
88           this.connectorPlugins = new ArrayList<>();
89   
90           // TODO: improve once plugins are allowed to be added/removed during runtime.
91           addConnectorPlugins(herder.plugins().sinkConnectors(), SINK_CONNECTOR_EXCLUDES);
92           addConnectorPlugins(herder.plugins().sourceConnectors(), SOURCE_CONNECTOR_EXCLUDES);
93           addConnectorPlugins(herder.plugins().transformations(), TRANSFORM_EXCLUDES);
94           addConnectorPlugins(herder.plugins().predicates(), Collections.emptySet());
95           addConnectorPlugins(herder.plugins().converters(), Collections.emptySet());
96           addConnectorPlugins(herder.plugins().headerConverters(), Collections.emptySet());
97       }
98   
99       private <T> void addConnectorPlugins(Collection<PluginDesc<T>> plugins, Collection<Class<? extends T>> excludes) {
100           plugins.stream()
101                   .filter(p -> !excludes.contains(p.pluginClass()))
102                   .map(PluginInfo::new)
103                   .forEach(connectorPlugins::add);
104       }
105   
106       @PUT
107       @Path("/{pluginName}/config/validate")
108       @Operation(summary = "Validate the provided configuration against the configuration definition for the specified pluginName")
109       public ConfigInfos validateConfigs(
110           final @PathParam("pluginName") String pluginName,
111           final Map<String, String> connectorConfig
112       ) throws Throwable {
113           String includedConnType = connectorConfig.get(ConnectorConfig.CONNECTOR_CLASS_CONFIG);
114           if (includedConnType != null
115               && !normalizedPluginName(includedConnType).endsWith(normalizedPluginName(pluginName))) {
116               throw new BadRequestException(
117                   "Included connector type " + includedConnType + " does not match request type "
118                       + pluginName
119               );
120           }
121   
122           // the validated configs don't need to be logged
123           FutureCallback<ConfigInfos> validationCallback = new FutureCallback<>();
124           herder.validateConnectorConfig(connectorConfig, validationCallback, false);
125   
126           try {
127               return validationCallback.get(ConnectorsResource.REQUEST_TIMEOUT_MS, TimeUnit.MILLISECONDS);
128           } catch (TimeoutException e) {
129               // This timeout is for the operation itself. None of the timeout error codes are relevant, so internal server
130               // error is the best option
131               throw new ConnectRestException(Response.Status.INTERNAL_SERVER_ERROR.getStatusCode(), "Request timed out");
132           } catch (InterruptedException e) {
133               throw new ConnectRestException(Response.Status.INTERNAL_SERVER_ERROR.getStatusCode(), "Request interrupted");
134           }
135       }
136   
137       @GET
138       @Path("/")
139       @Operation(summary = "List all connector plugins installed")
140       public List<PluginInfo> listConnectorPlugins(
141               @DefaultValue("true") @QueryParam("connectorsOnly") @Parameter(description = "Whether to list only connectors instead of all plugins") boolean connectorsOnly
142       ) {
143           synchronized (this) {
144               if (connectorsOnly) {
145                   return Collections.unmodifiableList(connectorPlugins.stream()
146                           .filter(p -> PluginType.SINK.toString().equals(p.type()) || PluginType.SOURCE.toString().equals(p.type()))
147                           .collect(Collectors.toList()));
148               } else {
149                   return Collections.unmodifiableList(connectorPlugins);
150               }
151           }
152       }
153   
154       @GET
155       @Path("/{pluginName}/config")
156       @Operation(summary = "Get the configuration definition for the specified pluginName")
157       public List<ConfigKeyInfo> getConnectorConfigDef(final @PathParam("pluginName") String pluginName) {
158           synchronized (this) {
159               return herder.connectorPluginConfig(pluginName);
160           }
161       }
162   
163       private String normalizedPluginName(String pluginName) {
164           // Works for both full and simple class names. In the latter case, it generates the alias.
165           return pluginName.endsWith(ALIAS_SUFFIX) && pluginName.length() > ALIAS_SUFFIX.length()
166               ? pluginName.substring(0, pluginName.length() - ALIAS_SUFFIX.length())
167               : pluginName;
168       }
