88       /**
89        * Throw an exception if the passed-in properties do not constitute a valid sink.
90        * @param props sink configuration properties
91        */
92       public static void validate(Map<String, String> props) {
93           final boolean hasTopicsConfig = hasTopicsConfig(props);
94           final boolean hasTopicsRegexConfig = hasTopicsRegexConfig(props);
95           final boolean hasDlqTopicConfig = hasDlqTopicConfig(props);
96   
97           if (hasTopicsConfig && hasTopicsRegexConfig) {
98               throw new ConfigException(SinkTask.TOPICS_CONFIG + " and " + SinkTask.TOPICS_REGEX_CONFIG +
99                   " are mutually exclusive options, but both are set.");
100           }
101   
102           if (!hasTopicsConfig && !hasTopicsRegexConfig) {
103               throw new ConfigException("Must configure one of " +
104                   SinkTask.TOPICS_CONFIG + " or " + SinkTask.TOPICS_REGEX_CONFIG);
105           }
106   
107           if (hasDlqTopicConfig) {
108               String dlqTopic = props.get(DLQ_TOPIC_NAME_CONFIG).trim();
109               if (hasTopicsConfig) {
110                   List<String> topics = parseTopicsList(props);
111                   if (topics.contains(dlqTopic)) {
112                       throw new ConfigException(String.format("The DLQ topic '%s' may not be included in the list of "
113                               + "topics ('%s=%s') consumed by the connector", dlqTopic, SinkTask.TOPICS_REGEX_CONFIG, topics));
114                   }
115               }
116               if (hasTopicsRegexConfig) {
117                   String topicsRegexStr = props.get(SinkTask.TOPICS_REGEX_CONFIG);
118                   Pattern pattern = Pattern.compile(topicsRegexStr);
119                   if (pattern.matcher(dlqTopic).matches()) {
120                       throw new ConfigException(String.format("The DLQ topic '%s' may not be included in the regex matching the "
121                               + "topics ('%s=%s') consumed by the connector", dlqTopic, SinkTask.TOPICS_REGEX_CONFIG, topicsRegexStr));
122                   }
123               }
124           }
125       }

127       public static boolean hasTopicsConfig(Map<String, String> props) {
128           String topicsStr = props.get(TOPICS_CONFIG);
129           return !Utils.isBlank(topicsStr);
130       }
131   
132       public static boolean hasTopicsRegexConfig(Map<String, String> props) {
133           String topicsRegexStr = props.get(TOPICS_REGEX_CONFIG);
134           return !Utils.isBlank(topicsRegexStr);
135       }
136   
137       public static boolean hasDlqTopicConfig(Map<String, String> props) {
138           String dqlTopicStr = props.get(DLQ_TOPIC_NAME_CONFIG);
139           return !Utils.isBlank(dqlTopicStr);
140       }
141   
142       @SuppressWarnings("unchecked")
143       public static List<String> parseTopicsList(Map<String, String> props) {
144           List<String> topics = (List<String>) ConfigDef.parseType(TOPICS_CONFIG, props.get(TOPICS_CONFIG), Type.LIST);
145           if (topics == null) {
146               return Collections.emptyList();
147           }
148           return topics
149                   .stream()
150                   .filter(topic -> !topic.isEmpty())
151                   .distinct()
152                   .collect(Collectors.toList());
153       }
