49   /**
50    * This class implements a group protocol for Kafka Connect workers that support incremental and
51    * cooperative rebalancing of connectors and tasks. It includes the format of worker state used when
52    * joining the group and distributing assignments, and the format of assignments of connectors
53    * and tasks to workers.
54    */
55   public class IncrementalCooperativeConnectProtocol {
56       public static final String ALLOCATION_KEY_NAME = "allocation";
57       public static final String REVOKED_KEY_NAME = "revoked";
58       public static final String SCHEDULED_DELAY_KEY_NAME = "delay";
59       public static final short CONNECT_PROTOCOL_V1 = 1;
60       public static final short CONNECT_PROTOCOL_V2 = 2;
61       public static final boolean TOLERATE_MISSING_FIELDS_WITH_DEFAULTS = true;
62   
63       /**
64        * Connect Protocol Header V1:
65        * <pre>
66        *   Version            => Int16
67        * </pre>
68        */
69       private static final Struct CONNECT_PROTOCOL_HEADER_V1 = new Struct(CONNECT_PROTOCOL_HEADER_SCHEMA)
70               .set(VERSION_KEY_NAME, CONNECT_PROTOCOL_V1);
71   
72       /**
73        * Connect Protocol Header V2:
74        * <pre>
75        *   Version            => Int16
76        * </pre>
77        * The V2 protocol is schematically identical to V1, but is used to signify that internal request
78        * verification and distribution of session keys is enabled (for more information, see KIP-507:
79        * https://cwiki.apache.org/confluence/display/KAFKA/KIP-507%3A+Securing+Internal+Connect+REST+Endpoints)
80        */
81       private static final Struct CONNECT_PROTOCOL_HEADER_V2 = new Struct(CONNECT_PROTOCOL_HEADER_SCHEMA)
82           .set(VERSION_KEY_NAME, CONNECT_PROTOCOL_V2);
83   
84   
85       /**
86        * Config State V1:
87        * <pre>
88        *   Url                => [String]
89        *   ConfigOffset       => Int64
90        * </pre>
91        */
92       public static final Schema CONFIG_STATE_V1 = CONFIG_STATE_V0;
93   
94       /**
95        * Allocation V1
96        * <pre>
97        *   Current Assignment => [Byte]
98        * </pre>
99        */
100       public static final Schema ALLOCATION_V1 = new Schema(
101               TOLERATE_MISSING_FIELDS_WITH_DEFAULTS,
102               new Field(ALLOCATION_KEY_NAME, NULLABLE_BYTES, null, true, null));
103   
104       /**
105        *
106        * Connector Assignment V1:
107        * <pre>
108        *   Connector          => [String]
109        *   Tasks              => [Int32]
110        * </pre>
111        *
112        * <p>Assignments for each worker are a set of connectors and tasks. These are categorized by
113        * connector ID. A sentinel task ID (CONNECTOR_TASK) is used to indicate the connector itself
114        * (i.e. that the assignment includes responsibility for running the Connector instance in
115        * addition to any tasks it generates).</p>
116        */
117       public static final Schema CONNECTOR_ASSIGNMENT_V1 = CONNECTOR_ASSIGNMENT_V0;
118   
119       /**
120        * Raw (non versioned) assignment V1:
121        * <pre>
122        *   Error              => Int16
123        *   Leader             => [String]
124        *   LeaderUrl          => [String]
125        *   ConfigOffset       => Int64
126        *   Assignment         => [Connector Assignment]
127        *   Revoked            => [Connector Assignment]
128        *   ScheduledDelay     => Int32
129        * </pre>
130        */
131       public static final Schema ASSIGNMENT_V1 = new Schema(
132               TOLERATE_MISSING_FIELDS_WITH_DEFAULTS,
133               new Field(ERROR_KEY_NAME, Type.INT16),
134               new Field(LEADER_KEY_NAME, Type.STRING),
135               new Field(LEADER_URL_KEY_NAME, Type.STRING),
136               new Field(CONFIG_OFFSET_KEY_NAME, Type.INT64),
137               new Field(ASSIGNMENT_KEY_NAME, ArrayOf.nullable(CONNECTOR_ASSIGNMENT_V1), null, true, null),
138               new Field(REVOKED_KEY_NAME, ArrayOf.nullable(CONNECTOR_ASSIGNMENT_V1), null, true, null),
139               new Field(SCHEDULED_DELAY_KEY_NAME, Type.INT32, null, 0));
140   
141       /**
142        * The fields are serialized in sequence as follows:
143        * Subscription V1:
144        * <pre>
145        *   Version            => Int16
146        *   Url                => [String]
147        *   ConfigOffset       => Int64
148        *   Current Assignment => [Byte]
149        * </pre>
150        */
151       public static ByteBuffer serializeMetadata(ExtendedWorkerState workerState, boolean sessioned) {
152           Struct configState = new Struct(CONFIG_STATE_V1)
153                   .set(URL_KEY_NAME, workerState.url())
154                   .set(CONFIG_OFFSET_KEY_NAME, workerState.offset());
155           // Not a big issue if we embed the protocol version with the assignment in the metadata
156           Struct allocation = new Struct(ALLOCATION_V1)
157                   .set(ALLOCATION_KEY_NAME, serializeAssignment(workerState.assignment(), sessioned));
158           Struct connectProtocolHeader = sessioned ? CONNECT_PROTOCOL_HEADER_V2 : CONNECT_PROTOCOL_HEADER_V1;
159           ByteBuffer buffer = ByteBuffer.allocate(connectProtocolHeader.sizeOf()
160                                                   + CONFIG_STATE_V1.sizeOf(configState)
161                                                   + ALLOCATION_V1.sizeOf(allocation));
162           connectProtocolHeader.writeTo(buffer);
163           CONFIG_STATE_V1.write(buffer, configState);
164           ALLOCATION_V1.write(buffer, allocation);
165           buffer.flip();
166           return buffer;
167       }
168   
169       /**
170        * Returns the collection of Connect protocols that are supported by this version along
171        * with their serialized metadata. The protocols are ordered by preference.
172        *
173        * @param workerState the current state of the worker metadata
174        * @param sessioned whether the {@link ConnectProtocolCompatibility#SESSIONED} protocol should
175        *                  be included in the collection of supported protocols
176        * @return the collection of Connect protocol metadata
177        */
178       public static JoinGroupRequestProtocolCollection metadataRequest(ExtendedWorkerState workerState, boolean sessioned) {
179           // Order matters in terms of protocol preference
180           List<JoinGroupRequestProtocol> joinGroupRequestProtocols = new ArrayList<>();
181           if (sessioned) {
182               joinGroupRequestProtocols.add(new JoinGroupRequestProtocol()
183                   .setName(SESSIONED.protocol())
184                   .setMetadata(IncrementalCooperativeConnectProtocol.serializeMetadata(workerState, true).array())
185               );
186           }
187           joinGroupRequestProtocols.add(new JoinGroupRequestProtocol()
188                           .setName(COMPATIBLE.protocol())
189                           .setMetadata(IncrementalCooperativeConnectProtocol.serializeMetadata(workerState, false).array())
190           );
191           joinGroupRequestProtocols.add(new JoinGroupRequestProtocol()
192                           .setName(EAGER.protocol())
193                           .setMetadata(ConnectProtocol.serializeMetadata(workerState).array())
194           );
195           return new JoinGroupRequestProtocolCollection(joinGroupRequestProtocols.iterator());
196       }

198       /**
199        * Given a byte buffer that contains protocol metadata return the deserialized form of the
200        * metadata.
201        *
202        * @param buffer A buffer containing the protocols metadata
203        * @return the deserialized metadata
204        * @throws SchemaException on incompatible Connect protocol version
205        */
206       public static ExtendedWorkerState deserializeMetadata(ByteBuffer buffer) {
207           Struct header = CONNECT_PROTOCOL_HEADER_SCHEMA.read(buffer);
208           Short version = header.getShort(VERSION_KEY_NAME);
209           checkVersionCompatibility(version);
210           Struct configState = CONFIG_STATE_V1.read(buffer);
211           long configOffset = configState.getLong(CONFIG_OFFSET_KEY_NAME);
212           String url = configState.getString(URL_KEY_NAME);
213           Struct allocation = ALLOCATION_V1.read(buffer);
214           // Protocol version is embedded with the assignment in the metadata
215           ExtendedAssignment assignment = deserializeAssignment(allocation.getBytes(ALLOCATION_KEY_NAME));
216           return new ExtendedWorkerState(url, configOffset, assignment);
217       }
218   
219       /**
220        * The fields are serialized in sequence as follows:
221        * Complete Assignment V1:
222        * <pre>
223        *   Version            => Int16
224        *   Error              => Int16
225        *   Leader             => [String]
226        *   LeaderUrl          => [String]
227        *   ConfigOffset       => Int64
228        *   Assignment         => [Connector Assignment]
229        *   Revoked            => [Connector Assignment]
230        *   ScheduledDelay     => Int32
231        * </pre>
232        */
233       public static ByteBuffer serializeAssignment(ExtendedAssignment assignment, boolean sessioned) {
234           // comparison depends on reference equality for now
235           if (assignment == null || ExtendedAssignment.empty().equals(assignment)) {
236               return null;
237           }
238           Struct struct = assignment.toStruct();
239           Struct protocolHeader = sessioned ? CONNECT_PROTOCOL_HEADER_V2 : CONNECT_PROTOCOL_HEADER_V1;
240           ByteBuffer buffer = ByteBuffer.allocate(protocolHeader.sizeOf()
241                                                   + ASSIGNMENT_V1.sizeOf(struct));
242           protocolHeader.writeTo(buffer);
243           ASSIGNMENT_V1.write(buffer, struct);
244           buffer.flip();
245           return buffer;
246       }
247   
248       /**
249        * Given a byte buffer that contains an assignment as defined by this protocol, return the
250        * deserialized form of the assignment.
251        *
252        * @param buffer the buffer containing a serialized assignment
253        * @return the deserialized assignment
254        * @throws SchemaException on incompatible Connect protocol version
255        */
256       public static ExtendedAssignment deserializeAssignment(ByteBuffer buffer) {
257           if (buffer == null) {
258               return null;
259           }
260           Struct header = CONNECT_PROTOCOL_HEADER_SCHEMA.read(buffer);
261           Short version = header.getShort(VERSION_KEY_NAME);
262           checkVersionCompatibility(version);
263           Struct struct = ASSIGNMENT_V1.read(buffer);
264           return ExtendedAssignment.fromStruct(version, struct);
265       }
266   
267       private static void checkVersionCompatibility(short version) {
268           // check for invalid versions
269           if (version < CONNECT_PROTOCOL_V0)
270               throw new SchemaException("Unsupported subscription version: " + version);
271   
272           // otherwise, assume versions can be parsed
273       }
