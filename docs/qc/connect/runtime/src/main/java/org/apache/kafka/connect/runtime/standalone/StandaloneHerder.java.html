104       @Override
105       public synchronized void stop() {
106           log.info("Herder stopping");
107           requestExecutorService.shutdown();
108           try {
109               if (!requestExecutorService.awaitTermination(30, TimeUnit.SECONDS))
110                   requestExecutorService.shutdownNow();
111           } catch (InterruptedException e) {
112               // ignore
113           }
114   
115           // There's no coordination/hand-off to do here since this is all standalone. Instead, we
116           // should just clean up the stuff we normally would, i.e. cleanly checkpoint and shutdown all
117           // the tasks.
118           for (String connName : connectors()) {
119               removeConnectorTasks(connName);
120               worker.stopAndAwaitConnector(connName);
121           }
122           stopServices();
123           running = false;
124           log.info("Herder stopped");
125       }

201       private synchronized void putConnectorConfig(String connName,
202                                                    final Map<String, String> config,
203                                                    boolean allowReplace,
204                                                    final Callback<Created<ConnectorInfo>> callback,
205                                                    ConfigInfos configInfos) {
206           try {
207               if (maybeAddConfigErrors(configInfos, callback)) {
208                   return;
209               }
210   
211               final boolean created;
212               if (configState.contains(connName)) {
213                   if (!allowReplace) {
214                       callback.onCompletion(new AlreadyExistsException("Connector " + connName + " already exists"), null);
215                       return;
216                   }
217                   worker.stopAndAwaitConnector(connName);
218                   created = false;
219               } else {
220                   created = true;
221               }
222   
223               configBackingStore.putConnectorConfig(connName, config);
224   
225               startConnector(connName, (error, result) -> {
226                   if (error != null) {
227                       callback.onCompletion(error, null);
228                       return;
229                   }
230   
231                   requestExecutorService.submit(() -> {
232                       updateConnectorTasks(connName);
233                       callback.onCompletion(null, new Created<>(created, createConnectorInfo(connName)));
234                   });
235               });
236           } catch (Throwable t) {
237               callback.onCompletion(t, null);
238           }
239       }

275           if (!configState.contains(taskId.connector()))

308       @Override
309       public synchronized void restartConnectorAndTasks(RestartRequest request, Callback<ConnectorStateInfo> cb) {
310           // Ensure the connector exists
311           String connectorName = request.connectorName();
312           if (!configState.contains(connectorName)) {
313               cb.onCompletion(new NotFoundException("Unknown connector: " + connectorName, null), null);
314               return;
315           }
316   
317           Optional<RestartPlan> maybePlan = buildRestartPlan(request);
318           if (!maybePlan.isPresent()) {
319               cb.onCompletion(new NotFoundException("Status for connector " + connectorName + " not found", null), null);
320               return;
321           }
322           RestartPlan plan = maybePlan.get();
323   
324           // If requested, stop the connector and any tasks, marking each as restarting
325           log.info("Received {}", plan);
326           if (plan.shouldRestartConnector()) {
327               worker.stopAndAwaitConnector(connectorName);
328               onRestart(connectorName);
329           }
330           if (plan.shouldRestartTasks()) {
331               // Stop the tasks and mark as restarting
332               worker.stopAndAwaitTasks(plan.taskIdsToRestart());
333               plan.taskIdsToRestart().forEach(this::onRestart);
334           }
335   
336           // Now restart the connector and tasks
337           if (plan.shouldRestartConnector()) {
338               log.debug("Restarting connector '{}'", connectorName);
339               startConnector(connectorName, (error, targetState) -> {
340                   if (error == null) {
341                       log.info("Connector '{}' restart successful", connectorName);
342                   } else {
343                       log.error("Connector '{}' restart failed", connectorName, error);
344                   }
345               });
346           }
347           if (plan.shouldRestartTasks()) {
348               log.debug("Restarting {} of {} tasks for {}", plan.restartTaskCount(), plan.totalTaskCount(), request);
349               createConnectorTasks(connectorName, plan.taskIdsToRestart());
350               log.debug("Restarted {} of {} tasks for {} as requested", plan.restartTaskCount(), plan.totalTaskCount(), request);
351           }
352           // Complete the restart request
353           log.info("Completed {}", plan);
354           cb.onCompletion(null, plan.restartConnectorStateInfo());
355       }

385       private boolean startTask(ConnectorTaskId taskId, Map<String, String> connProps) {
386           switch (connectorTypeForClass(connProps.get(ConnectorConfig.CONNECTOR_CLASS_CONFIG))) {
387               case SINK:
388                   return worker.startSinkTask(
389                           taskId,
390                           configState,
391                           connProps,
392                           configState.taskConfig(taskId),
393                           this,
394                           configState.targetState(taskId.connector())
395                   );
396               case SOURCE:
397                   return worker.startSourceTask(
398                           taskId,
399                           configState,
400                           connProps,
401                           configState.taskConfig(taskId),
402                           this,
403                           configState.targetState(taskId.connector())
404                   );
405               default:
406                   throw new ConnectException("Failed to start task " + taskId + " since it is not a recognizable type (source or sink)");
407           }
408       }

410       private void removeConnectorTasks(String connName) {
411           Collection<ConnectorTaskId> tasks = configState.tasks(connName);
412           if (!tasks.isEmpty()) {
413               worker.stopAndAwaitTasks(tasks);
414               configBackingStore.removeTaskConfigs(connName);
415               tasks.forEach(this::onDeletion);
416           }
417       }

419       private void updateConnectorTasks(String connName) {
420           if (!worker.isRunning(connName)) {
421               log.info("Skipping update of connector {} since it is not running", connName);
422               return;
423           }
424   
425           List<Map<String, String>> newTaskConfigs = recomputeTaskConfigs(connName);
426           List<Map<String, String>> oldTaskConfigs = configState.allTaskConfigs(connName);
427   
428           if (!newTaskConfigs.equals(oldTaskConfigs)) {
429               removeConnectorTasks(connName);
430               List<Map<String, String>> rawTaskConfigs = reverseTransform(connName, configState, newTaskConfigs);
431               configBackingStore.putTaskConfigs(connName, rawTaskConfigs);
432               createConnectorTasks(connName);
433           }
434       }
