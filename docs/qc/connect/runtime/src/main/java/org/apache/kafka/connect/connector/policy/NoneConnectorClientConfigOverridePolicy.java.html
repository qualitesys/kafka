18   package org.apache.kafka.connect.connector.policy;
19   
20   import org.apache.kafka.common.config.ConfigValue;
21   import org.slf4j.Logger;
22   import org.slf4j.LoggerFactory;
23   
24   import java.util.Map;
25   
26   /**
27    * Disallow any client configuration to be overridden via the connector configs by setting {@code connector.client.config.override.policy} to {@code None}.
28    * This is the default behavior.
29    */
30   public class NoneConnectorClientConfigOverridePolicy extends AbstractConnectorClientConfigOverridePolicy {
31       private static final Logger log = LoggerFactory.getLogger(NoneConnectorClientConfigOverridePolicy.class);
32   
33       @Override
34       protected String policyName() {
35           return "None";
36       }
37   
38       @Override
39       protected boolean isAllowed(ConfigValue configValue) {
40           return false;
41       }
42   
43       @Override
44       public void configure(Map<String, ?> configs) {
45           log.info("Setting up None Policy for ConnectorClientConfigOverride. This will disallow any client configuration to be overridden");
46       }
47   }
