115   /**
116    * <p>
117    *     Distributed "herder" that coordinates with other workers to spread work across multiple processes.
118    * </p>
119    * <p>
120    *     Under the hood, this is implemented as a group managed by Kafka's group membership facilities (i.e. the generalized
121    *     group/consumer coordinator). Each instance of DistributedHerder joins the group and indicates what it's current
122    *     configuration state is (where it is in the configuration log). The group coordinator selects one member to take
123    *     this information and assign each instance a subset of the active connectors & tasks to execute. This assignment
124    *     is currently performed in a simple round-robin fashion, but this is not guaranteed -- the herder may also choose
125    *     to, e.g., use a sticky assignment to avoid the usual start/stop costs associated with connectors and tasks. Once
126    *     an assignment is received, the DistributedHerder simply runs its assigned connectors and tasks in a Worker.
127    * </p>
128    * <p>
129    *     In addition to distributing work, the DistributedHerder uses the leader determined during the work assignment
130    *     to select a leader for this generation of the group who is responsible for other tasks that can only be performed
131    *     by a single node at a time. Most importantly, this includes writing updated configurations for connectors and tasks,
132    *     (and therefore, also for creating, destroy, and scaling up/down connectors).
133    * </p>
134    * <p>
135    *     The DistributedHerder uses a single thread for most of its processing. This includes processing
136    *     config changes, handling task rebalances and serving requests from the HTTP layer. The latter are pushed
137    *     into a queue until the thread has time to handle them. A consequence of this is that requests can get blocked
138    *     behind a worker rebalance. When the herder knows that a rebalance is expected, it typically returns an error
139    *     immediately to the request, but this is not always possible (in particular when another worker has requested
140    *     the rebalance). Similar to handling HTTP requests, config changes which are observed asynchronously by polling
141    *     the config log are batched for handling in the work thread.
142    * </p>
143    */

362       // public for testing
363       public void tick() {
364           // The main loop does two primary things: 1) drive the group membership protocol, responding to rebalance events
365           // as they occur, and 2) handle external requests targeted at the leader. All the "real" work of the herder is
366           // performed in this thread, which keeps synchronization straightforward at the cost of some operations possibly
367           // blocking up this thread (especially those in callbacks due to rebalance events).
368   
369           try {
370               // if we failed to read to end of log before, we need to make sure the issue was resolved before joining group
371               // Joining and immediately leaving for failure to read configs is exceedingly impolite
372               if (!canReadConfigs) {
373                   if (readConfigToEnd(workerSyncTimeoutMs)) {
374                       canReadConfigs = true;
375                   } else {
376                       return; // Safe to return and tick immediately because readConfigToEnd will do the backoff for us
377                   }
378               }
379   
380               log.debug("Ensuring group membership is still active");
381               member.ensureActive();
382               // Ensure we're in a good state in our group. If not restart and everything should be setup to rejoin
383               if (!handleRebalanceCompleted()) return;
384           } catch (WakeupException e) {
385               // May be due to a request from another thread, or might be stopping. If the latter, we need to check the
386               // flag immediately. If the former, we need to re-run the ensureActive call since we can't handle requests
387               // unless we're in the group.
388               log.trace("Woken up while ensure group membership is still active");
389               return;
390           }
391   
392           if (fencedFromConfigTopic) {
393               if (isLeader()) {
394                   // We were accidentally fenced out, possibly by a zombie leader
395                   try {
396                       log.debug("Reclaiming write privileges for config topic after being fenced out");
397                       configBackingStore.claimWritePrivileges();
398                       fencedFromConfigTopic = false;
399                       log.debug("Successfully reclaimed write privileges for config topic after being fenced out");
400                   } catch (Exception e) {
401                       log.warn("Unable to claim write privileges for config topic. Will backoff and possibly retry if still the leader", e);
402                       backoff(CONFIG_TOPIC_WRITE_PRIVILEGES_BACKOFF_MS);
403                       return;
404                   }
405               } else {
406                   log.trace("Relinquished write privileges for config topic after being fenced out, since worker is no longer the leader of the cluster");
407                   // We were meant to be fenced out because we fell out of the group and a new leader was elected
408                   fencedFromConfigTopic = false;
409               }
410           }
411   
412           long now = time.milliseconds();
413   
414           if (checkForKeyRotation(now)) {
415               log.debug("Distributing new session key");
416               keyExpiration = Long.MAX_VALUE;
417               try {
418                   SessionKey newSessionKey = new SessionKey(keyGenerator.generateKey(), now);
419                   writeToConfigTopicAsLeader(() -> configBackingStore.putSessionKey(newSessionKey));
420               } catch (Exception e) {
421                   log.info("Failed to write new session key to config topic; forcing a read to the end of the config topic before possibly retrying", e);
422                   canReadConfigs = false;
423                   return;
424               }
425           }
426   
427           // Process any external requests
428           // TODO: Some of these can be performed concurrently or even optimized away entirely.
429           //       For example, if three different connectors are slated to be restarted, it's fine to
430           //       restart all three at the same time instead.
431           //       Another example: if multiple configurations are submitted for the same connector,
432           //       the only one that actually has to be written to the config topic is the
433           //       most-recently one.
434           long nextRequestTimeoutMs = Long.MAX_VALUE;
435           while (true) {
436               final DistributedHerderRequest next = peekWithoutException();
437               if (next == null) {
438                   break;
439               } else if (now >= next.at) {
440                   requests.pollFirst();
441               } else {
442                   nextRequestTimeoutMs = next.at - now;
443                   break;
444               }
445   
446               runRequest(next.action(), next.callback());
447           }
448   
449           // Process all pending connector restart requests
450           processRestartRequests();
451   
452           if (scheduledRebalance < Long.MAX_VALUE) {
453               nextRequestTimeoutMs = Math.min(nextRequestTimeoutMs, Math.max(scheduledRebalance - now, 0));
454               rebalanceResolved = false;
455               log.debug("Scheduled rebalance at: {} (now: {} nextRequestTimeoutMs: {}) ",
456                       scheduledRebalance, now, nextRequestTimeoutMs);
457           }
458           if (isLeader() && internalRequestValidationEnabled() && keyExpiration < Long.MAX_VALUE) {
459               nextRequestTimeoutMs = Math.min(nextRequestTimeoutMs, Math.max(keyExpiration - now, 0));
460               log.debug("Scheduled next key rotation at: {} (now: {} nextRequestTimeoutMs: {}) ",
461                       keyExpiration, now, nextRequestTimeoutMs);
462           }
463   
464           // Process any configuration updates
465           AtomicReference<Set<String>> connectorConfigUpdatesCopy = new AtomicReference<>();
466           AtomicReference<Set<String>> connectorTargetStateChangesCopy = new AtomicReference<>();
467           AtomicReference<Set<ConnectorTaskId>> taskConfigUpdatesCopy = new AtomicReference<>();
468   
469           boolean shouldReturn;
470           if (member.currentProtocolVersion() == CONNECT_PROTOCOL_V0) {
471               shouldReturn = updateConfigsWithEager(connectorConfigUpdatesCopy,
472                       connectorTargetStateChangesCopy);
473               // With eager protocol we should return immediately if needsReconfigRebalance has
474               // been set to retain the old workflow
475               if (shouldReturn) {
476                   return;
477               }
478   
479               if (connectorConfigUpdatesCopy.get() != null) {
480                   processConnectorConfigUpdates(connectorConfigUpdatesCopy.get());
481               }
482   
483               if (connectorTargetStateChangesCopy.get() != null) {
484                   processTargetStateChanges(connectorTargetStateChangesCopy.get());
485               }
486           } else {
487               shouldReturn = updateConfigsWithIncrementalCooperative(connectorConfigUpdatesCopy,
488                       connectorTargetStateChangesCopy, taskConfigUpdatesCopy);
489   
490               if (connectorConfigUpdatesCopy.get() != null) {
491                   processConnectorConfigUpdates(connectorConfigUpdatesCopy.get());
492               }
493   
494               if (connectorTargetStateChangesCopy.get() != null) {
495                   processTargetStateChanges(connectorTargetStateChangesCopy.get());
496               }
497   
498               if (taskConfigUpdatesCopy.get() != null) {
499                   processTaskConfigUpdatesWithIncrementalCooperative(taskConfigUpdatesCopy.get());
500               }
501   
502               if (shouldReturn) {
503                   return;
504               }
505           }
506   
507           // Let the group take any actions it needs to
508           try {
509               log.trace("Polling for group activity; will wait for {}ms or until poll is interrupted by "
510                       + "either config backing store updates or a new external request",
511                       nextRequestTimeoutMs);
512               member.poll(nextRequestTimeoutMs);
513               // Ensure we're in a good state in our group. If not restart and everything should be setup to rejoin
514               handleRebalanceCompleted();
515           } catch (WakeupException e) { // FIXME should not be WakeupException
516               log.trace("Woken up while polling for group activity");
517               // Ignore. Just indicates we need to check the exit flag, for requested actions, etc.
518           }

521       private boolean checkForKeyRotation(long now) {
522           SecretKey key;
523           long expiration;
524           synchronized (this) {
525               // This happens on startup; the snapshot contains the session key,
526               // but no callback in the config update listener has been fired for it yet.
527               if (sessionKey == null && configState.sessionKey() != null) {
528                   sessionKey = configState.sessionKey().key();
529                   keyExpiration = configState.sessionKey().creationTimestamp() + keyRotationIntervalMs;
530               }
531               key = sessionKey;
532               expiration = keyExpiration;
533           }
534   
535           if (internalRequestValidationEnabled()) {
536               if (isLeader()) {
537                   if (key == null) {
538                       log.debug("Internal request signing is enabled but no session key has been distributed yet. "
539                           + "Distributing new key now.");
540                       return true;
541                   } else if (expiration <= now) {
542                       log.debug("Existing key has expired. Distributing new key now.");
543                       return true;
544                   } else if (!key.getAlgorithm().equals(keyGenerator.getAlgorithm())
545                           || key.getEncoded().length != keyGenerator.generateKey().getEncoded().length) {
546                       log.debug("Previously-distributed key uses different algorithm/key size "
547                           + "than required by current worker configuration. Distributing new key now.");
548                       return true;
549                   }
550               }
551           }
552           return false;
553       }

555       private synchronized boolean updateConfigsWithEager(AtomicReference<Set<String>> connectorConfigUpdatesCopy,
556                                                           AtomicReference<Set<String>> connectorTargetStateChangesCopy) {
557           // This branch is here to avoid creating a snapshot if not needed
558           if (needsReconfigRebalance
559                   || !connectorConfigUpdates.isEmpty()
560                   || !connectorTargetStateChanges.isEmpty()) {
561               log.trace("Handling config updates with eager rebalancing");
562               // Connector reconfigs only need local updates since there is no coordination between workers required.
563               // However, if connectors were added or removed, work needs to be rebalanced since we have more work
564               // items to distribute among workers.
565               configState = configBackingStore.snapshot();
566   
567               if (needsReconfigRebalance) {
568                   // Task reconfigs require a rebalance. Request the rebalance, clean out state, and then restart
569                   // this loop, which will then ensure the rebalance occurs without any other requests being
570                   // processed until it completes.
571                   log.debug("Requesting rebalance due to reconfiguration of tasks (needsReconfigRebalance: {})",
572                           needsReconfigRebalance);
573                   member.requestRejoin();
574                   needsReconfigRebalance = false;
575                   // Any connector config updates or target state changes will be addressed during the rebalance too
576                   connectorConfigUpdates.clear();
577                   connectorTargetStateChanges.clear();
578                   return true;
579               } else {
580                   if (!connectorConfigUpdates.isEmpty()) {
581                       // We can't start/stop while locked since starting connectors can cause task updates that will
582                       // require writing configs, which in turn make callbacks into this class from another thread that
583                       // require acquiring a lock. This leads to deadlock. Instead, just copy the info we need and process
584                       // the updates after unlocking.
585                       connectorConfigUpdatesCopy.set(connectorConfigUpdates);
586                       connectorConfigUpdates = new HashSet<>();
587                   }
588   
589                   if (!connectorTargetStateChanges.isEmpty()) {
590                       // Similarly for target state changes which can cause connectors to be restarted
591                       connectorTargetStateChangesCopy.set(connectorTargetStateChanges);
592                       connectorTargetStateChanges = new HashSet<>();
593                   }
594               }
595           } else {
596               log.trace("Skipping config updates with eager rebalancing "
597                   + "since no config rebalance is required "
598                   + "and there are no connector config, task config, or target state changes pending");
599           }
600           return false;
601       }

610                   || !connectorTargetStateChanges.isEmpty()

613               // Connector reconfigs only need local updates since there is no coordination between workers required.

622                   needsReconfigRebalance = false;
623                   retValue = true;
624               }

629                   // require acquiring a lock. This leads to deadlock. Instead, just copy the info we need and process

654       private void processConnectorConfigUpdates(Set<String> connectorConfigUpdates) {
655           // If we only have connector config updates, we can just bounce the updated connectors that are
656           // currently assigned to this worker.
657           Set<String> localConnectors = assignment == null ? Collections.emptySet() : new HashSet<>(assignment.connectors());
658           log.trace("Processing connector config updates; "
659                   + "currently-owned connectors are {}, and to-be-updated connectors are {}",
660                   localConnectors,
661                   connectorConfigUpdates);
662           for (String connectorName : connectorConfigUpdates) {
663               if (!localConnectors.contains(connectorName)) {
664                   log.trace("Skipping config update for connector {} as it is not owned by this worker",
665                           connectorName);
666                   continue;
667               }
668               boolean remains = configState.contains(connectorName);
669               log.info("Handling connector-only config update by {} connector {}",
670                       remains ? "restarting" : "stopping", connectorName);
671               worker.stopAndAwaitConnector(connectorName);

673               if (remains) {
674                   startConnector(connectorName, (error, result) -> {
675                       if (error != null) {
676                           log.error("Failed to start connector '" + connectorName + "'", error);
677                       }
678                   });
679               }
680           }
681       }

706               });
707           }
