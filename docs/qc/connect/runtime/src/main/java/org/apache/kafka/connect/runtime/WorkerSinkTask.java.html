196       @Override
197       public void execute() {
198           log.info("{} Executing sink task", this);
199           // Make sure any uncommitted data has been committed and the task has
200           // a chance to clean up its state
201           try (UncheckedCloseable suppressible = this::closeAllPartitions) {
202               while (!isStopping())
203                   iteration();
204           } catch (WakeupException e) {
205               log.trace("Consumer woken up during initial offset commit attempt, " 
206                   + "but succeeded during a later attempt");
207           }
208       }

210       protected void iteration() {
211           final long offsetCommitIntervalMs = workerConfig.getLong(WorkerConfig.OFFSET_COMMIT_INTERVAL_MS_CONFIG);
212   
213           try {
214               long now = time.milliseconds();
215   
216               // Maybe commit
217               if (!committing && (context.isCommitRequested() || now >= nextCommit)) {
218                   commitOffsets(now, false);
219                   nextCommit = now + offsetCommitIntervalMs;
220                   context.clearCommitRequest();
221               }
222   
223               final long commitTimeoutMs = commitStarted + workerConfig.getLong(WorkerConfig.OFFSET_COMMIT_TIMEOUT_MS_CONFIG);
224   
225               // Check for timed out commits
226               if (committing && now >= commitTimeoutMs) {
227                   log.warn("{} Commit of offsets timed out", this);
228                   commitFailures++;
229                   committing = false;
230               }
231   
232               // And process messages
233               long timeoutMs = Math.max(nextCommit - now, 0);
234               poll(timeoutMs);
235           } catch (WakeupException we) {
236               log.trace("{} Consumer woken up", this);
237   
238               if (isStopping())
239                   return;
240   
241               if (shouldPause()) {
242                   pauseAll();
243                   onPause();
244                   context.requestCommit();
245               } else if (!pausedForRedelivery) {
246                   resumeAll();
247                   onResume();
248               }
249           }
250       }

252       /**
253        * Respond to a previous commit attempt that may or may not have succeeded. Note that due to our use of async commits,
254        * these invocations may come out of order and thus the need for the commit sequence number.
255        *
256        * @param error            the error resulting from the commit, or null if the commit succeeded without error
257        * @param seqno            the sequence number at the time the commit was requested
258        * @param committedOffsets the offsets that were committed; may be null if the commit did not complete successfully
259        *                         or if no new offsets were committed
260        */
261       private void onCommitCompleted(Throwable error, long seqno, Map<TopicPartition, OffsetAndMetadata> committedOffsets) {
262           if (commitSeqno != seqno) {
263               log.debug("{} Received out of order commit callback for sequence number {}, but most recent sequence number is {}",
264                       this, seqno, commitSeqno);
265               sinkTaskMetricsGroup.recordOffsetCommitSkip();
266           } else {
267               long durationMillis = time.milliseconds() - commitStarted;
268               if (error != null) {
269                   log.error("{} Commit of offsets threw an unexpected exception for sequence number {}: {}",
270                           this, seqno, committedOffsets, error);
271                   commitFailures++;
272                   recordCommitFailure(durationMillis, error);
273               } else {
274                   log.debug("{} Finished offset commit successfully in {} ms for sequence number {}: {}",
275                           this, durationMillis, seqno, committedOffsets);
276                   if (committedOffsets != null) {
277                       log.trace("{} Adding to last committed offsets: {}", this, committedOffsets);
278                       lastCommittedOffsets.putAll(committedOffsets);
279                       log.debug("{} Last committed offsets are now {}", this, committedOffsets);
280                       sinkTaskMetricsGroup.recordCommittedOffsets(committedOffsets);
281                   }
282                   commitFailures = 0;
283                   recordCommitSuccess(durationMillis);
284               }
285               committing = false;
286           }
287       }

316       /**
317        * Poll for new messages with the given timeout. Should only be invoked by the worker thread.
318        */
319       protected void poll(long timeoutMs) {
320           rewind();
321           long retryTimeout = context.timeout();
322           if (retryTimeout > 0) {
323               timeoutMs = Math.min(timeoutMs, retryTimeout);
324               context.timeout(-1L);
325           }
326   
327           log.trace("{} Polling consumer with timeout {} ms", this, timeoutMs);
328           ConsumerRecords<byte[], byte[]> msgs = pollConsumer(timeoutMs);
329           assert messageBatch.isEmpty() || msgs.isEmpty();
330           log.trace("{} Polling returned {} messages", this, msgs.count());
331   
332           convertMessages(msgs);
333           deliverMessages();
334       }

341       private void doCommitSync(Map<TopicPartition, OffsetAndMetadata> offsets, int seqno) {
342           log.debug("{} Committing offsets synchronously using sequence number {}: {}", this, seqno, offsets);
343           try {
344               consumer.commitSync(offsets);
345               onCommitCompleted(null, seqno, offsets);
346           } catch (WakeupException e) {
347               // retry the commit to ensure offsets get pushed, then propagate the wakeup up to poll
348               doCommitSync(offsets, seqno);
349               throw e;
350           } catch (KafkaException e) {
351               onCommitCompleted(e, seqno, offsets);
352           }
353       }

355       private void doCommitAsync(Map<TopicPartition, OffsetAndMetadata> offsets, final int seqno) {
356           log.debug("{} Committing offsets asynchronously using sequence number {}: {}", this, seqno, offsets);
357           OffsetCommitCallback cb = (tpOffsets, error) -> onCommitCompleted(error, seqno, tpOffsets);
358           consumer.commitAsync(offsets, cb);
359       }

377       private void commitOffsets(long now, boolean closing, Collection<TopicPartition> topicPartitions) {
378           log.trace("Committing offsets for partitions {}", topicPartitions);
379           if (workerErrantRecordReporter != null) {
380               log.trace("Awaiting reported errors to be completed");
381               workerErrantRecordReporter.awaitFutures(topicPartitions);
382               log.trace("Completed reported errors");
383           }
384   
385           Map<TopicPartition, OffsetAndMetadata> offsetsToCommit = currentOffsets.entrySet().stream()
386               .filter(e -> topicPartitions.contains(e.getKey()))
387               .collect(Collectors.toMap(Map.Entry::getKey, Map.Entry::getValue));
388   
389           if (offsetsToCommit.isEmpty())
390               return;
391   
392           committing = true;
393           commitSeqno += 1;
394           commitStarted = now;
395           sinkTaskMetricsGroup.recordOffsetSequenceNumber(commitSeqno);
396   
397           Map<TopicPartition, OffsetAndMetadata> lastCommittedOffsetsForPartitions = this.lastCommittedOffsets.entrySet().stream()
398               .filter(e -> offsetsToCommit.containsKey(e.getKey()))
399               .collect(Collectors.toMap(Map.Entry::getKey, Map.Entry::getValue));
400   
401           final Map<TopicPartition, OffsetAndMetadata> taskProvidedOffsets;
402           try {
403               log.trace("{} Calling task.preCommit with current offsets: {}", this, offsetsToCommit);
404               taskProvidedOffsets = task.preCommit(new HashMap<>(offsetsToCommit));
405           } catch (Throwable t) {
406               if (closing) {
407                   log.warn("{} Offset commit failed during close", this);
408               } else {
409                   log.error("{} Offset commit failed, rewinding to last committed offsets", this, t);
410                   for (Map.Entry<TopicPartition, OffsetAndMetadata> entry : lastCommittedOffsetsForPartitions.entrySet()) {
411                       log.debug("{} Rewinding topic partition {} to offset {}", this, entry.getKey(), entry.getValue().offset());
412                       consumer.seek(entry.getKey(), entry.getValue().offset());
413                   }
414                   currentOffsets.putAll(lastCommittedOffsetsForPartitions);
415               }
416               onCommitCompleted(t, commitSeqno, null);
417               return;
418           } finally {
419               if (closing) {
420                   log.trace("{} Closing the task before committing the offsets: {}", this, offsetsToCommit);
421                   task.close(topicPartitions);
422               }
423           }
424   
425           if (taskProvidedOffsets.isEmpty()) {
426               log.debug("{} Skipping offset commit, task opted-out by returning no offsets from preCommit", this);
427               onCommitCompleted(null, commitSeqno, null);
428               return;
429           }
430   
431           Collection<TopicPartition> allAssignedTopicPartitions = consumer.assignment();
432           final Map<TopicPartition, OffsetAndMetadata> committableOffsets = new HashMap<>(lastCommittedOffsetsForPartitions);
433           for (Map.Entry<TopicPartition, OffsetAndMetadata> taskProvidedOffsetEntry : taskProvidedOffsets.entrySet()) {
434               final TopicPartition partition = taskProvidedOffsetEntry.getKey();
435               final OffsetAndMetadata taskProvidedOffset = taskProvidedOffsetEntry.getValue();
436               if (committableOffsets.containsKey(partition)) {
437                   long taskOffset = taskProvidedOffset.offset();
438                   long currentOffset = offsetsToCommit.get(partition).offset();
439                   if (taskOffset <= currentOffset) {
440                       committableOffsets.put(partition, taskProvidedOffset);
441                   } else {
442                       log.warn("{} Ignoring invalid task provided offset {}/{} -- not yet consumed, taskOffset={} currentOffset={}",
443                           this, partition, taskProvidedOffset, taskOffset, currentOffset);
444                   }
445               } else if (!allAssignedTopicPartitions.contains(partition)) {
446                   log.warn("{} Ignoring invalid task provided offset {}/{} -- partition not assigned, assignment={}",
447                           this, partition, taskProvidedOffset, allAssignedTopicPartitions);
448               } else {
449                   log.debug("{} Ignoring task provided offset {}/{} -- partition not requested, requested={}",
450                           this, partition, taskProvidedOffset, committableOffsets.keySet());
451               }
452           }

459   
460           doCommit(committableOffsets, closing, commitSeqno);

512       private SinkRecord convertAndTransformRecord(final ConsumerRecord<byte[], byte[]> msg) {
513           SchemaAndValue keyAndSchema = retryWithToleranceOperator.execute(() -> keyConverter.toConnectData(msg.topic(), msg.headers(), msg.key()),
514                   Stage.KEY_CONVERTER, keyConverter.getClass());
515   
516           SchemaAndValue valueAndSchema = retryWithToleranceOperator.execute(() -> valueConverter.toConnectData(msg.topic(), msg.headers(), msg.value()),
517                   Stage.VALUE_CONVERTER, valueConverter.getClass());
518   
519           Headers headers = retryWithToleranceOperator.execute(() -> convertHeadersFor(msg), Stage.HEADER_CONVERTER, headerConverter.getClass());
520   
521           if (retryWithToleranceOperator.failed()) {
522               return null;
523           }
524   
525           Long timestamp = ConnectUtils.checkAndConvertTimestamp(msg.timestamp());
526           SinkRecord origRecord = new SinkRecord(msg.topic(), msg.partition(),
527                   keyAndSchema.schema(), keyAndSchema.value(),
528                   valueAndSchema.schema(), valueAndSchema.value(),
529                   msg.offset(),
530                   timestamp,
531                   msg.timestampType(),
532                   headers);
533           log.trace("{} Applying transformations to record in topic '{}' partition {} at offset {} and timestamp {} with key {} and value {}",
534                   this, msg.topic(), msg.partition(), msg.offset(), timestamp, keyAndSchema.value(), valueAndSchema.value());
535           if (isTopicTrackingEnabled) {
536               recordActiveTopic(origRecord.topic());
537           }
538   
539           // Apply the transformations
540           SinkRecord transformedRecord = transformationChain.apply(origRecord);
541           if (transformedRecord == null) {
542               return null;
543           }
544           // Error reporting will need to correlate each sink record with the original consumer record
545           return new InternalSinkRecord(msg, transformedRecord);
546       }

575       private void deliverMessages() {
576           // Finally, deliver this batch to the sink
577           try {
578               // Since we reuse the messageBatch buffer, ensure we give the task its own copy
579               log.trace("{} Delivering batch of {} messages to task", this, messageBatch.size());
580               long start = time.milliseconds();
581               task.put(new ArrayList<>(messageBatch));
582               // if errors raised from the operator were swallowed by the task implementation, an
583               // exception needs to be thrown to kill the task indicating the tolerance was exceeded
584               if (retryWithToleranceOperator.failed() && !retryWithToleranceOperator.withinToleranceLimits()) {
585                   throw new ConnectException("Tolerance exceeded in error handler",
586                       retryWithToleranceOperator.error());
587               }
588               recordBatch(messageBatch.size());
589               sinkTaskMetricsGroup.recordPut(time.milliseconds() - start);
590               currentOffsets.putAll(origOffsets);
591               messageBatch.clear();
592               // If we had paused all consumer topic partitions to try to redeliver data, then we should resume any that
593               // the task had not explicitly paused
594               if (pausedForRedelivery) {
595                   if (!shouldPause())
596                       resumeAll();
597                   pausedForRedelivery = false;
598               }
599           } catch (RetriableException e) {
600               log.error("{} RetriableException from SinkTask:", this, e);
601               if (!pausedForRedelivery) {
602                   // If we're retrying a previous batch, make sure we've paused all topic partitions so we don't get new data,
603                   // but will still be able to poll in order to handle user-requested timeouts, keep group membership, etc.
604                   pausedForRedelivery = true;
605                   pauseAll();
606               }
607               // Let this exit normally, the batch will be reprocessed on the next loop.
608           } catch (Throwable t) {
609               log.error("{} Task threw an uncaught and unrecoverable exception. Task is being killed and will not "
610                       + "recover until manually restarted. Error: {}", this, t.getMessage(), t);
611               throw new ConnectException("Exiting WorkerSinkTask due to unrecoverable exception.", t);
612           }
613       }

615       private void rewind() {
616           Map<TopicPartition, Long> offsets = context.offsets();
617           if (offsets.isEmpty()) {
618               return;
619           }
620           for (Map.Entry<TopicPartition, Long> entry: offsets.entrySet()) {
621               TopicPartition tp = entry.getKey();
622               Long offset = entry.getValue();

624                   log.trace("{} Rewind {} to offset {}", this, tp, offset);
625                   consumer.seek(tp, offset);
626                   lastCommittedOffsets.put(tp, new OffsetAndMetadata(offset));
627                   currentOffsets.put(tp, new OffsetAndMetadata(offset));
628               } else {
629                   log.warn("{} Cannot rewind {} to null offset", this, tp);
630               }
631           }
632           context.clearOffsets();
633       }

691       private class HandleRebalance implements ConsumerRebalanceListener {
692           @Override
693           public void onPartitionsAssigned(Collection<TopicPartition> partitions) {
694               log.debug("{} Partitions assigned {}", WorkerSinkTask.this, partitions);
695   
696               for (TopicPartition tp : partitions) {
697                   long pos = consumer.position(tp);
698                   lastCommittedOffsets.put(tp, new OffsetAndMetadata(pos));
699                   currentOffsets.put(tp, new OffsetAndMetadata(pos));
700                   log.debug("{} Assigned topic partition {} with offset {}", WorkerSinkTask.this, tp, pos);
701               }
702               sinkTaskMetricsGroup.assignedOffsets(currentOffsets);
703   
704               boolean wasPausedForRedelivery = pausedForRedelivery;
705               pausedForRedelivery = wasPausedForRedelivery && !messageBatch.isEmpty();
706               if (pausedForRedelivery) {
707                   // Re-pause here in case we picked up new partitions in the rebalance
708                   pauseAll();
709               } else {
710                   // If we paused everything for redelivery and all partitions for the failed deliveries have been revoked, make
711                   // sure anything we paused that the task didn't request to be paused *and* which we still own is resumed.
712                   // Also make sure our tracking of paused partitions is updated to remove any partitions we no longer own.
713                   if (wasPausedForRedelivery) {
714                       resumeAll();
715                   }
716                   // Ensure that the paused partitions contains only assigned partitions and repause as necessary
717                   context.pausedPartitions().retainAll(consumer.assignment());
718                   if (shouldPause())
719                       pauseAll();
720                   else if (!context.pausedPartitions().isEmpty())
721                       consumer.pause(context.pausedPartitions());
722               }
723   
724               if (partitions.isEmpty()) {
725                   return;
726               }
727   
728               // Instead of invoking the assignment callback on initialization, we guarantee the consumer is ready upon
729               // task start. Since this callback gets invoked during that initial setup before we've started the task, we
730               // need to guard against invoking the user's callback method during that period.
731               if (rebalanceException == null || rebalanceException instanceof WakeupException) {
732                   try {
733                       openPartitions(partitions);
734                       // Rewind should be applied only if openPartitions succeeds.
735                       rewind();
736                   } catch (RuntimeException e) {
737                       // The consumer swallows exceptions raised in the rebalance listener, so we need to store
738                       // exceptions and rethrow when poll() returns.
739                       rebalanceException = e;
740                   }
741               }
742           }

745           public void onPartitionsRevoked(Collection<TopicPartition> partitions) {
746               onPartitionsRemoved(partitions, false);
747           }
748   
749           @Override
750           public void onPartitionsLost(Collection<TopicPartition> partitions) {
751               onPartitionsRemoved(partitions, true);
752           }

754           private void onPartitionsRemoved(Collection<TopicPartition> partitions, boolean lost) {
755               if (taskStopped) {
756                   log.trace("Skipping partition revocation callback as task has already been stopped");
757                   return;
758               }
759               log.debug("{} Partitions {}: {}", WorkerSinkTask.this, lost ? "lost" : "revoked", partitions);
760   
761               if (partitions.isEmpty())
762                   return;
763   
764               try {
765                   closePartitions(partitions, lost);
766                   sinkTaskMetricsGroup.clearOffsets(partitions);
767               } catch (RuntimeException e) {
768                   // The consumer swallows exceptions raised in the rebalance listener, so we need to store
769                   // exceptions and rethrow when poll() returns.
770                   rebalanceException = e;
771               }
772   
773               // Make sure we don't have any leftover data since offsets for these partitions will be reset to committed positions
774               messageBatch.removeIf(record -> partitions.contains(new TopicPartition(record.topic(), record.kafkaPartition())));
775           }
776       }

838               Map<TopicPartition, OffsetAndMetadata> consumed = this.consumedOffsets;
839               Map<TopicPartition, OffsetAndMetadata> committed = this.committedOffsets;
