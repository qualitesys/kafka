17   package org.apache.kafka.connect.storage;
18   
19   import org.apache.kafka.connect.runtime.ConnectorStatus;
20   import org.apache.kafka.connect.runtime.TaskStatus;
21   import org.apache.kafka.connect.runtime.TopicStatus;
22   import org.apache.kafka.connect.runtime.WorkerConfig;
23   import org.apache.kafka.connect.util.ConnectorTaskId;
24   import org.apache.kafka.connect.util.Table;
25   
26   import java.util.Collection;
27   import java.util.Collections;
28   import java.util.HashMap;
29   import java.util.HashSet;
30   import java.util.Map;
31   import java.util.Objects;
32   import java.util.Set;
33   import java.util.concurrent.ConcurrentHashMap;
34   import java.util.concurrent.ConcurrentMap;
35   
36   public class MemoryStatusBackingStore implements StatusBackingStore {
37       private final Table<String, Integer, TaskStatus> tasks;
38       private final Map<String, ConnectorStatus> connectors;
39       private final ConcurrentMap<String, ConcurrentMap<String, TopicStatus>> topics;
40   
41       public MemoryStatusBackingStore() {
42           this.tasks = new Table<>();
43           this.connectors = new HashMap<>();
44           this.topics = new ConcurrentHashMap<>();
45       }
46   
47       @Override
48       public void configure(WorkerConfig config) {
49   
50       }
51   
52       @Override
53       public void start() {
54   
55       }
56   
57       @Override
58       public void stop() {
59   
60       }
61   
62       @Override
63       public synchronized void put(ConnectorStatus status) {
64           if (status.state() == ConnectorStatus.State.DESTROYED)
65               connectors.remove(status.id());
66           else
67               connectors.put(status.id(), status);
68       }
69   
70       @Override
71       public synchronized void putSafe(ConnectorStatus status) {
72           put(status);
73       }
74   
75       @Override
76       public synchronized void put(TaskStatus status) {
77           if (status.state() == TaskStatus.State.DESTROYED)
78               tasks.remove(status.id().connector(), status.id().task());
79           else
80               tasks.put(status.id().connector(), status.id().task(), status);
81       }
82   
83       @Override
84       public synchronized void putSafe(TaskStatus status) {
85           put(status);
86       }
87   
88       @Override
89       public void put(final TopicStatus status) {
90           topics.computeIfAbsent(status.connector(), k -> new ConcurrentHashMap<>())
91                   .put(status.topic(), status);
92       }
93   
94       @Override
95       public synchronized TaskStatus get(ConnectorTaskId id) {
96           return tasks.get(id.connector(), id.task());
97       }
98   
99       @Override
100       public synchronized ConnectorStatus get(String connector) {
101           return connectors.get(connector);
102       }
103   
104       @Override
105       public synchronized Collection<TaskStatus> getAll(String connector) {
106           return new HashSet<>(tasks.row(connector).values());
107       }
108   
109       @Override
110       public TopicStatus getTopic(String connector, String topic) {
111           ConcurrentMap<String, TopicStatus> activeTopics = topics.get(Objects.requireNonNull(connector));
112           return activeTopics != null ? activeTopics.get(Objects.requireNonNull(topic)) : null;
113       }
114   
115       @Override
116       public Collection<TopicStatus> getAllTopics(String connector) {
117           ConcurrentMap<String, TopicStatus> activeTopics = topics.get(Objects.requireNonNull(connector));
118           return activeTopics != null
119                  ? Collections.unmodifiableCollection(activeTopics.values())
120                  : Collections.emptySet();
121       }
122   
123       @Override
124       public void deleteTopic(String connector, String topic) {
125           ConcurrentMap<String, TopicStatus> activeTopics = topics.get(Objects.requireNonNull(connector));
126           if (activeTopics != null) {
127               activeTopics.remove(Objects.requireNonNull(topic));
128           }
129       }
130   
131       @Override
132       public synchronized Set<String> connectors() {
133           return new HashSet<>(connectors.keySet());
134       }
135   
136       @Override
137       public void flush() {
138   
139       }
140   
141   }
