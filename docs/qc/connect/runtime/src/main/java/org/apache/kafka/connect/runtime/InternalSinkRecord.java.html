18   package org.apache.kafka.connect.runtime;
19   
20   import org.apache.kafka.clients.consumer.ConsumerRecord;
21   import org.apache.kafka.common.record.TimestampType;
22   import org.apache.kafka.connect.data.Schema;
23   import org.apache.kafka.connect.header.Header;
24   import org.apache.kafka.connect.sink.SinkRecord;
25   
26   /**
27    * A specialization of {@link SinkRecord} that allows a {@link WorkerSinkTask} to track the
28    * original {@link ConsumerRecord} for each {@link SinkRecord}. It is used internally and not
29    * exposed to connectors.
30    */
31   public class InternalSinkRecord extends SinkRecord {
32   
33       private final ConsumerRecord<byte[], byte[]> originalRecord;
34   
35       public InternalSinkRecord(ConsumerRecord<byte[], byte[]> originalRecord, SinkRecord record) {
36           super(record.topic(), record.kafkaPartition(), record.keySchema(), record.key(),
37               record.valueSchema(), record.value(), record.kafkaOffset(), record.timestamp(),
38               record.timestampType(), record.headers());
39           this.originalRecord = originalRecord;
40       }
41   
42       protected InternalSinkRecord(ConsumerRecord<byte[], byte[]> originalRecord, String topic,
43                                    int partition, Schema keySchema, Object key, Schema valueSchema,
44                                    Object value, long kafkaOffset, Long timestamp,
45                                    TimestampType timestampType, Iterable<Header> headers) {
46           super(topic, partition, keySchema, key, valueSchema, value, kafkaOffset, timestamp, timestampType, headers);
47           this.originalRecord = originalRecord;
48       }
49   
50       @Override
51       public SinkRecord newRecord(String topic, Integer kafkaPartition, Schema keySchema, Object key,
52                                   Schema valueSchema, Object value, Long timestamp,
53                                   Iterable<Header> headers) {
54           return new InternalSinkRecord(originalRecord, topic, kafkaPartition, keySchema, key,
55               valueSchema, value, kafkaOffset(), timestamp, timestampType(), headers());
56       }
57   
58       @Override
59       public boolean equals(Object o) {
60           return super.equals(o);
61       }
62   
63       @Override
64       public int hashCode() {
65           return super.hashCode();
66       }
67   
68       @Override
69       public String toString() {
70           return super.toString();
71       }
72   
73       /**
74       * Return the original consumer record that this sink record represents.
75       *
76       * @return the original consumer record; never null
77       */
78       public ConsumerRecord<byte[], byte[]> originalRecord() {
79           return originalRecord;
80       }
81   }
