17   package org.apache.kafka.connect.runtime.distributed;
18   
19   import java.util.Arrays;
20   import java.util.Locale;
21   
22   import static org.apache.kafka.connect.runtime.distributed.ConnectProtocol.CONNECT_PROTOCOL_V0;
23   import static org.apache.kafka.connect.runtime.distributed.IncrementalCooperativeConnectProtocol.CONNECT_PROTOCOL_V1;
24   import static org.apache.kafka.connect.runtime.distributed.IncrementalCooperativeConnectProtocol.CONNECT_PROTOCOL_V2;
25   
26   /**
27    * An enumeration of the modes available to the worker to signal which Connect protocols are
28    * enabled at any time.
29    *
30    * {@code EAGER} signifies that this worker only supports prompt release of assigned connectors
31    * and tasks in every rebalance. Corresponds to Connect protocol V0.
32    *
33    * {@code COMPATIBLE} signifies that this worker supports both eager and incremental cooperative
34    * Connect protocols and will use the version that is elected by the Kafka broker coordinator
35    * during rebalance.
36    *
37    * {@code SESSIONED} signifies that this worker supports all of the above protocols in addition to
38    * a protocol that uses incremental cooperative rebalancing for worker assignment and uses session
39    * keys distributed via the config topic to verify internal REST requests
40    */
41   public enum ConnectProtocolCompatibility {
42       EAGER {
43           @Override
44           public String protocol() {
45               return "default";
46           }
47   
48           @Override
49           public short protocolVersion() {
50               return CONNECT_PROTOCOL_V0;
51           }
52       },
53   
54       COMPATIBLE {
55           @Override
56           public String protocol() {
57               return "compatible";
58           }
59   
60           @Override
61           public short protocolVersion() {
62               return CONNECT_PROTOCOL_V1;
63           }
64       },
65   
66       SESSIONED {
67           @Override
68           public String protocol() {
69               return "sessioned";
70           }
71   
72           @Override
73           public short protocolVersion() {
74               return CONNECT_PROTOCOL_V2;
75           }
76       };
77   
78       /**
79        * Return the enum that corresponds to the name that is given as an argument;
80        * if no mapping is found {@code IllegalArgumentException} is thrown.
81        *
82        * @param name the name of the protocol compatibility mode
83        * @return the enum that corresponds to the protocol compatibility mode
84        */
85       public static ConnectProtocolCompatibility compatibility(String name) {
86           return Arrays.stream(ConnectProtocolCompatibility.values())
87                   .filter(mode -> mode.name().equalsIgnoreCase(name))
88                   .findFirst()
89                   .orElseThrow(() -> new IllegalArgumentException(
90                           "Unknown Connect protocol compatibility mode: " + name));
91       }
92   
93       /**
94        * Return the enum that corresponds to the Connect protocol version that is given as an argument;
95        * if no mapping is found {@code IllegalArgumentException} is thrown.
96        *
97        * @param protocolVersion the version of the protocol; for example,
98        * {@link ConnectProtocol#CONNECT_PROTOCOL_V0 CONNECT_PROTOCOL_V0}. May not be null
99        * @return the enum that corresponds to the protocol compatibility mode
100        */
101       public static ConnectProtocolCompatibility fromProtocolVersion(short protocolVersion) {
102           switch (protocolVersion) {
103               case CONNECT_PROTOCOL_V0:
104                   return EAGER;
105               case CONNECT_PROTOCOL_V1:
106                   return COMPATIBLE;
107               case CONNECT_PROTOCOL_V2:
108                   return SESSIONED;
109               default:
110                   throw new IllegalArgumentException("Unknown Connect protocol version: " + protocolVersion);
111           }
112       }
113   
114       @Override
115       public String toString() {
116           return name().toLowerCase(Locale.ROOT);
117       }
118   
119       /**
120        * Return the version of the protocol for this mode.
121        *
122        * @return the protocol version
123        */
124       public abstract short protocolVersion();
125   
126       /**
127        * Return the name of the protocol that this mode will use in {@code ProtocolMetadata}.
128        *
129        * @return the protocol name
130        */
131       public abstract String protocol();
132   
133       /**
134        * Return the enum that corresponds to the protocol name that is given as an argument;
135        * if no mapping is found {@code IllegalArgumentException} is thrown.
136        *
137        * @param protocolName the name of the connect protocol
138        * @return the enum that corresponds to the protocol compatibility mode that supports the
139        * given protocol
140        */
141       public static ConnectProtocolCompatibility fromProtocol(String protocolName) {
142           return Arrays.stream(ConnectProtocolCompatibility.values())
143                   .filter(mode -> mode.protocol().equalsIgnoreCase(protocolName))
144                   .findFirst()
145                   .orElseThrow(() -> new IllegalArgumentException(
146                           "Not found Connect protocol compatibility mode for protocol: " + protocolName));
147       }
148   }
