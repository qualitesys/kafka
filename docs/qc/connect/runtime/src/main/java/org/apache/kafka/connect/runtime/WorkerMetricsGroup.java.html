27   class WorkerMetricsGroup {
28       private final ConnectMetrics.MetricGroup metricGroup;
29       private final Sensor connectorStartupAttempts;
30       private final Sensor connectorStartupSuccesses;
31       private final Sensor connectorStartupFailures;
32       private final Sensor connectorStartupResults;
33       private final Sensor taskStartupAttempts;
34       private final Sensor taskStartupSuccesses;
35       private final Sensor taskStartupFailures;
36       private final Sensor taskStartupResults;
37   
38       public WorkerMetricsGroup(final Map<String, WorkerConnector> connectors, Map<ConnectorTaskId, WorkerTask> tasks, ConnectMetrics connectMetrics) {
39           ConnectMetricsRegistry registry = connectMetrics.registry();
40           metricGroup = connectMetrics.group(registry.workerGroupName());
41   
42           metricGroup.addValueMetric(registry.connectorCount, now -> (double) connectors.size());
43           metricGroup.addValueMetric(registry.taskCount, now -> (double) tasks.size());
44   
45           MetricName connectorFailurePct = metricGroup.metricName(registry.connectorStartupFailurePercentage);
46           MetricName connectorSuccessPct = metricGroup.metricName(registry.connectorStartupSuccessPercentage);
47           Frequencies connectorStartupResultFrequencies = Frequencies.forBooleanValues(connectorFailurePct, connectorSuccessPct);
48           connectorStartupResults = metricGroup.sensor("connector-startup-results");
49           connectorStartupResults.add(connectorStartupResultFrequencies);
50   
51           connectorStartupAttempts = metricGroup.sensor("connector-startup-attempts");
52           connectorStartupAttempts.add(metricGroup.metricName(registry.connectorStartupAttemptsTotal), new CumulativeSum());
53   
54           connectorStartupSuccesses = metricGroup.sensor("connector-startup-successes");
55           connectorStartupSuccesses.add(metricGroup.metricName(registry.connectorStartupSuccessTotal), new CumulativeSum());
56   
57           connectorStartupFailures = metricGroup.sensor("connector-startup-failures");
58           connectorStartupFailures.add(metricGroup.metricName(registry.connectorStartupFailureTotal), new CumulativeSum());
59   
60           MetricName taskFailurePct = metricGroup.metricName(registry.taskStartupFailurePercentage);
61           MetricName taskSuccessPct = metricGroup.metricName(registry.taskStartupSuccessPercentage);
62           Frequencies taskStartupResultFrequencies = Frequencies.forBooleanValues(taskFailurePct, taskSuccessPct);
63           taskStartupResults = metricGroup.sensor("task-startup-results");
64           taskStartupResults.add(taskStartupResultFrequencies);
65   
66           taskStartupAttempts = metricGroup.sensor("task-startup-attempts");
67           taskStartupAttempts.add(metricGroup.metricName(registry.taskStartupAttemptsTotal), new CumulativeSum());
68   
69           taskStartupSuccesses = metricGroup.sensor("task-startup-successes");
70           taskStartupSuccesses.add(metricGroup.metricName(registry.taskStartupSuccessTotal), new CumulativeSum());
71   
72           taskStartupFailures = metricGroup.sensor("task-startup-failures");
73           taskStartupFailures.add(metricGroup.metricName(registry.taskStartupFailureTotal), new CumulativeSum());
74       }
75   
76       void close() {
77           metricGroup.close();
78       }
79   
80       void recordConnectorStartupFailure() {
81           connectorStartupAttempts.record(1.0);
82           connectorStartupFailures.record(1.0);
83           connectorStartupResults.record(0.0);
84       }
85   
86       void recordConnectorStartupSuccess() {
87           connectorStartupAttempts.record(1.0);
88           connectorStartupSuccesses.record(1.0);
89           connectorStartupResults.record(1.0);
90       }
91   
92       void recordTaskFailure() {
93           taskStartupAttempts.record(1.0);
94           taskStartupFailures.record(1.0);
95           taskStartupResults.record(0.0);
96       }
97   
98       void recordTaskSuccess() {
99           taskStartupAttempts.record(1.0);
100           taskStartupSuccesses.record(1.0);
101           taskStartupResults.record(1.0);
102       }
103   
104       protected ConnectMetrics.MetricGroup metricGroup() {
105           return metricGroup;
106       }

116       class ConnectorStatusListener implements ConnectorStatus.Listener {
117           private final ConnectorStatus.Listener delegateListener;
118           private volatile boolean startupSucceeded = false;
119   
120           ConnectorStatusListener(ConnectorStatus.Listener delegateListener) {
121               this.delegateListener = delegateListener;
122           }
123   
124           @Override
125           public void onStartup(final String connector) {
126               startupSucceeded = true;
127               recordConnectorStartupSuccess();
128               delegateListener.onStartup(connector);
129           }
130   
131           @Override
132           public void onPause(final String connector) {
133               delegateListener.onPause(connector);
134           }
135   
136           @Override
137           public void onResume(final String connector) {
138               delegateListener.onResume(connector);
139           }
140   
141           @Override
142           public void onFailure(final String connector, final Throwable cause) {
143               if (!startupSucceeded) {
144                   recordConnectorStartupFailure();
145               }
146               delegateListener.onFailure(connector, cause);
147           }
148   
149           @Override
150           public void onRestart(String connector) {
151               delegateListener.onRestart(connector);
152           }
153   
154           @Override
155           public void onShutdown(final String connector) {
156               delegateListener.onShutdown(connector);
157           }
158   
159           @Override
160           public void onDeletion(final String connector) {
161               delegateListener.onDeletion(connector);
162           }
163       }

196           }
