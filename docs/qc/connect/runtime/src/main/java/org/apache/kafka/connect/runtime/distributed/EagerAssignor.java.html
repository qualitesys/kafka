38   /**
39    * An assignor that computes a unweighted round-robin distribution of connectors and tasks. The
40    * connectors are assigned to the workers first, followed by the tasks. This is to avoid
41    * load imbalance when several 1-task connectors are running, given that a connector is usually
42    * more lightweight than a task.
43    *
44    * Note that this class is NOT thread-safe.
45    */
46   public class EagerAssignor implements ConnectAssignor {
47       private final Logger log;
48   
49       public EagerAssignor(LogContext logContext) {
50           this.log = logContext.logger(EagerAssignor.class);
51       }
52   
53       @Override
54       public Map<String, ByteBuffer> performAssignment(String leaderId, String protocol,
55                                                        List<JoinGroupResponseMember> allMemberMetadata,
56                                                        WorkerCoordinator coordinator) {
57           log.debug("Performing task assignment");
58           Map<String, ExtendedWorkerState> memberConfigs = new HashMap<>();
59           for (JoinGroupResponseMember member : allMemberMetadata)
60               memberConfigs.put(member.memberId(), IncrementalCooperativeConnectProtocol.deserializeMetadata(ByteBuffer.wrap(member.metadata())));
61   
62           long maxOffset = findMaxMemberConfigOffset(memberConfigs, coordinator);
63           Long leaderOffset = ensureLeaderConfig(maxOffset, coordinator);
64           if (leaderOffset == null)
65               return fillAssignmentsAndSerialize(memberConfigs.keySet(), Assignment.CONFIG_MISMATCH,
66                       leaderId, memberConfigs.get(leaderId).url(), maxOffset,
67                       new HashMap<>(), new HashMap<>());
68           return performTaskAssignment(leaderId, leaderOffset, memberConfigs, coordinator);
69       }
70   
71       private Long ensureLeaderConfig(long maxOffset, WorkerCoordinator coordinator) {
72           // If this leader is behind some other members, we can't do assignment
73           if (coordinator.configSnapshot().offset() < maxOffset) {
74               // We might be able to take a new snapshot to catch up immediately and avoid another round of syncing here.
75               // Alternatively, if this node has already passed the maximum reported by any other member of the group, it
76               // is also safe to use this newer state.
77               ClusterConfigState updatedSnapshot = coordinator.configFreshSnapshot();
78               if (updatedSnapshot.offset() < maxOffset) {
79                   log.info("Was selected to perform assignments, but do not have latest config found in sync request. " +
80                           "Returning an empty configuration to trigger re-sync.");
81                   return null;
82               } else {
83                   coordinator.configSnapshot(updatedSnapshot);
84                   return updatedSnapshot.offset();
85               }
86           }
87           return maxOffset;
88       }

90       private Map<String, ByteBuffer> performTaskAssignment(String leaderId, long maxOffset,
91                                                             Map<String, ExtendedWorkerState> memberConfigs,
92                                                             WorkerCoordinator coordinator) {
93           Map<String, Collection<String>> connectorAssignments = new HashMap<>();
94           Map<String, Collection<ConnectorTaskId>> taskAssignments = new HashMap<>();
95   
96           // Perform round-robin task assignment. Assign all connectors and then all tasks because assigning both the
97           // connector and its tasks can lead to very uneven distribution of work in some common cases (e.g. for connectors
98           // that generate only 1 task each; in a cluster of 2 or an even # of nodes, only even nodes will be assigned
99           // connectors and only odd nodes will be assigned tasks, but tasks are, on average, actually more resource
100           // intensive than connectors).
101           List<String> connectorsSorted = sorted(coordinator.configSnapshot().connectors());
102           CircularIterator<String> memberIt = new CircularIterator<>(sorted(memberConfigs.keySet()));
103           for (String connectorId : connectorsSorted) {
104               String connectorAssignedTo = memberIt.next();
105               log.trace("Assigning connector {} to {}", connectorId, connectorAssignedTo);
106               Collection<String> memberConnectors = connectorAssignments.computeIfAbsent(connectorAssignedTo, k -> new ArrayList<>());
107               memberConnectors.add(connectorId);
108           }
109           for (String connectorId : connectorsSorted) {
110               for (ConnectorTaskId taskId : sorted(coordinator.configSnapshot().tasks(connectorId))) {
111                   String taskAssignedTo = memberIt.next();
112                   log.trace("Assigning task {} to {}", taskId, taskAssignedTo);
113                   Collection<ConnectorTaskId> memberTasks = taskAssignments.computeIfAbsent(taskAssignedTo, k -> new ArrayList<>());
114                   memberTasks.add(taskId);
115               }
116           }
117   
118           coordinator.leaderState(new LeaderState(memberConfigs, connectorAssignments, taskAssignments));
119   
120           return fillAssignmentsAndSerialize(memberConfigs.keySet(), Assignment.NO_ERROR,
121                   leaderId, memberConfigs.get(leaderId).url(), maxOffset, connectorAssignments, taskAssignments);
122       }
123   
124       private Map<String, ByteBuffer> fillAssignmentsAndSerialize(Collection<String> members,
125                                                                   short error,
126                                                                   String leaderId,
127                                                                   String leaderUrl,
128                                                                   long maxOffset,
129                                                                   Map<String, Collection<String>> connectorAssignments,
130                                                                   Map<String, Collection<ConnectorTaskId>> taskAssignments) {
131   
132           Map<String, ByteBuffer> groupAssignment = new HashMap<>();
133           for (String member : members) {
134               Collection<String> connectors = connectorAssignments.get(member);
135               if (connectors == null) {
136                   connectors = Collections.emptyList();
137               }
138               Collection<ConnectorTaskId> tasks = taskAssignments.get(member);
139               if (tasks == null) {
140                   tasks = Collections.emptyList();
141               }
142               Assignment assignment = new Assignment(error, leaderId, leaderUrl, maxOffset, connectors, tasks);
143               log.debug("Assignment: {} -> {}", member, assignment);
144               groupAssignment.put(member, ConnectProtocol.serializeAssignment(assignment));
145           }
146           log.debug("Finished assignment");
147           return groupAssignment;
148       }

150       private long findMaxMemberConfigOffset(Map<String, ExtendedWorkerState> memberConfigs,
151                                              WorkerCoordinator coordinator) {
152           // The new config offset is the maximum seen by any member. We always perform assignment using this offset,
153           // even if some members have fallen behind. The config offset used to generate the assignment is included in
154           // the response so members that have fallen behind will not use the assignment until they have caught up.
155           Long maxOffset = null;
156           for (Map.Entry<String, ExtendedWorkerState> stateEntry : memberConfigs.entrySet()) {
157               long memberRootOffset = stateEntry.getValue().offset();
158               if (maxOffset == null)
159                   maxOffset = memberRootOffset;
160               else
161                   maxOffset = Math.max(maxOffset, memberRootOffset);
162           }
163   
164           log.debug("Max config offset root: {}, local snapshot config offsets root: {}",
165                     maxOffset, coordinator.configSnapshot().offset());
166           return maxOffset;
167       }

169       private static <T extends Comparable<T>> List<T> sorted(Collection<T> members) {
170           List<T> res = new ArrayList<>(members);
171           Collections.sort(res);
172           return res;
173       }
