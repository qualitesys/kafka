59   /**
60    * <p>
61    *     KafkaBasedLog provides a generic implementation of a shared, compacted log of records stored in Kafka that all
62    *     clients need to consume and, at times, agree on their offset / that they have read to the end of the log.
63    * </p>
64    * <p>
65    *     This functionality is useful for storing different types of data that all clients may need to agree on --
66    *     offsets or config for example. This class runs a consumer in a background thread to continuously tail the target
67    *     topic, accepts write requests which it writes to the topic using an internal producer, and provides some helpful
68    *     utilities like checking the current log end offset and waiting until the current end of the log is reached.
69    * </p>
70    * <p>
71    *     To support different use cases, this class works with either single- or multi-partition topics.
72    * </p>
73    * <p>
74    *     Since this class is generic, it delegates the details of data storage via a callback that is invoked for each
75    *     record that is consumed from the topic. The invocation of callbacks is guaranteed to be serialized -- if the
76    *     calling class keeps track of state based on the log and only writes to it when consume callbacks are invoked
77    *     and only reads it in {@link #readToEnd(Callback)} callbacks then no additional synchronization will be required.
78    * </p>
79    */

221       public void start() {
222           log.info("Starting KafkaBasedLog with topic " + topic);
223   
224           // Create the topic admin client and initialize the topic ...
225           admin = topicAdminSupplier.get();   // may be null
226           if (admin == null && requireAdminForOffsets) {
227               throw new ConnectException(
228                       "Must provide a TopicAdmin to KafkaBasedLog when consumer is configured with "
229                               + ConsumerConfig.ISOLATION_LEVEL_CONFIG + " set to "
230                               + IsolationLevel.READ_COMMITTED.name().toLowerCase(Locale.ROOT)
231               );
232           }
233           initializer.accept(admin);
234   
235           // Then create the producer and consumer
236           producer = Optional.ofNullable(createProducer());
237           if (!producer.isPresent())
238               log.trace("Creating read-only KafkaBasedLog for topic " + topic);
239           consumer = createConsumer();
240   
241           List<TopicPartition> partitions = new ArrayList<>();
242   
243           // We expect that the topics will have been created either manually by the user or automatically by the herder
244           List<PartitionInfo> partitionInfos = consumer.partitionsFor(topic);
245           long started = time.nanoseconds();
246           long sleepMs = 100;
247           while (partitionInfos.isEmpty() && time.nanoseconds() - started < CREATE_TOPIC_TIMEOUT_NS) {
248               time.sleep(sleepMs);
249               sleepMs = Math.min(2 * sleepMs, MAX_SLEEP_MS);
250               partitionInfos = consumer.partitionsFor(topic);
251           }
252           if (partitionInfos.isEmpty())
253               throw new ConnectException("Could not look up partition metadata for offset backing store topic in" +
254                       " allotted period. This could indicate a connectivity issue, unavailable topic partitions, or if" +
255                       " this is your first use of the topic it may have taken too long to create.");
256   
257           for (PartitionInfo partition : partitionInfos)
258               partitions.add(new TopicPartition(partition.topic(), partition.partition()));
259           partitionCount = partitions.size();
260           consumer.assign(partitions);
261   
262           // Always consume from the beginning of all partitions. Necessary to ensure that we don't use committed offsets
263           // when a 'group.id' is specified (if offsets happen to have been committed unexpectedly).
264           consumer.seekToBeginning(partitions);
265   
266           readToLogEnd(true);
267   
268           thread = new WorkThread();
269           thread.start();
270   
271           log.info("Finished reading KafkaBasedLog for topic " + topic);
272   
273           log.info("Started KafkaBasedLog for topic " + topic);
274       }

304       /**
305        * Flushes any outstanding writes and then reads to the current end of the log and invokes the specified callback.
306        * Note that this checks the current offsets, reads to them, and invokes the callback regardless of whether
307        * additional records have been written to the log. If the caller needs to ensure they have truly reached the end
308        * of the log, they must ensure there are no other writers during this period.
309        *
310        * This waits until the end of all partitions has been reached.
311        *
312        * This method is asynchronous. If you need a synchronous version, pass an instance of
313        * {@link org.apache.kafka.connect.util.FutureCallback} as the {@param callback} parameter and wait on it to block.
314        *
315        * @param callback the callback to invoke once the end of the log has been reached.
316        */
317       public void readToEnd(Callback<Void> callback) {
318           log.trace("Starting read to end log for topic {}", topic);
319           flush();
320           synchronized (this) {
321               readLogEndOffsetCallbacks.add(callback);
322           }
323           consumer.wakeup();
324       }

357       protected Producer<K, V> createProducer() {

366       protected Consumer<K, V> createConsumer() {
367           // Always force reset to the beginning of the log since this class wants to consume all available log data

371           consumerConfigs.put(ConsumerConfig.ENABLE_AUTO_COMMIT_CONFIG, false);
372           return new KafkaConsumer<>(consumerConfigs);

375       private void poll(long timeoutMs) {
376           try {
377               ConsumerRecords<K, V> records = consumer.poll(Duration.ofMillis(timeoutMs));
378               for (ConsumerRecord<K, V> record : records)
379                   consumedCallback.onCompletion(null, record);
380           } catch (WakeupException e) {
381               // Expected on get() or stop(). The calling code should handle this
382               throw e;
383           } catch (KafkaException e) {
384               log.error("Error polling: " + e);
385           }
386       }
387   
388       /**
389        * This method finds the end offsets of the Kafka log's topic partitions, optionally retrying
390        * if the {@code listOffsets()} method of the admin client throws a {@link RetriableException}.
391        *
392        * @param shouldRetry Boolean flag to enable retry for the admin client {@code listOffsets()} call.
393        * @see TopicAdmin#retryEndOffsets
394        */
395       private void readToLogEnd(boolean shouldRetry) {
396           Set<TopicPartition> assignment = consumer.assignment();
397           Map<TopicPartition, Long> endOffsets = readEndOffsets(assignment, shouldRetry);
398           log.trace("Reading to end of log offsets {}", endOffsets);
399   
400           while (!endOffsets.isEmpty()) {
401               Iterator<Map.Entry<TopicPartition, Long>> it = endOffsets.entrySet().iterator();
402               while (it.hasNext()) {
403                   Map.Entry<TopicPartition, Long> entry = it.next();
404                   TopicPartition topicPartition = entry.getKey();
405                   long endOffset = entry.getValue();
406                   long lastConsumedOffset = consumer.position(topicPartition);
407                   if (lastConsumedOffset >= endOffset) {
408                       log.trace("Read to end offset {} for {}", endOffset, topicPartition);
409                       it.remove();
410                   } else {
411                       log.trace("Behind end offset {} for {}; last-read offset is {}",
412                               endOffset, topicPartition, lastConsumedOffset);
413                       poll(Integer.MAX_VALUE);
414                       break;
415                   }
416               }
417           }
418       }
419   
420       // Visible for testing
421       /**
422        * Read to the end of the given list of topic partitions
423        * @param assignment the topic partitions to read to the end of
424        * @param shouldRetry boolean flag to enable retry for the admin client {@code listOffsets()} call.
425        * @throws UnsupportedVersionException if the log's consumer is using the "read_committed" isolation level (and
426        * therefore a separate admin client is required to read end offsets for the topic), but the broker does not support
427        * reading end offsets using an admin client
428        */
429       Map<TopicPartition, Long> readEndOffsets(Set<TopicPartition> assignment, boolean shouldRetry) throws UnsupportedVersionException {
430           log.trace("Reading to end of offset log");
431   
432           // Note that we'd prefer to not use the consumer to find the end offsets for the assigned topic partitions.
433           // That is because it's possible that the consumer is already blocked waiting for new records to appear, when
434           // the consumer is already at the end. In such cases, using 'consumer.endOffsets(...)' will block until at least
435           // one more record becomes available, meaning we can't even check whether we're at the end offset.
436           // Since all we're trying to do here is get the end offset, we should use the supplied admin client
437           // (if available) to obtain the end offsets for the given topic partitions.
438   
439           // Deprecated constructors do not provide an admin supplier, so the admin is potentially null.
440           if (admin != null) {
441               // Use the admin client to immediately find the end offsets for the assigned topic partitions.
442               // Unlike using the consumer
443               try {
444                   if (shouldRetry) {
445                       return admin.retryEndOffsets(assignment,
446                               ADMIN_CLIENT_RETRY_DURATION,
447                               ADMIN_CLIENT_RETRY_BACKOFF_MS);
448                   }
449   
450                   return admin.endOffsets(assignment);
451               } catch (UnsupportedVersionException e) {
452                   // This may happen with really old brokers that don't support the auto topic creation
453                   // field in metadata requests
454                   if (requireAdminForOffsets) {
455                       // Should be handled by the caller during log startup
456                       throw e;
457                   }
458                   log.debug("Reading to end of log offsets with consumer since admin client is unsupported: {}", e.getMessage());
459                   // Forget the reference to the admin so that we won't even try to use the admin the next time this method is called
460                   admin = null;
461                   // continue and let the consumer handle the read
462               }
463               // Other errors, like timeouts and retriable exceptions are intentionally propagated
464           }
465           // The admin may be null if older deprecated constructor is used or if the admin client is using a broker that doesn't
466           // support getting the end offsets (e.g., 0.10.x). In such cases, we should use the consumer, which is not ideal (see above).
467           return consumer.endOffsets(assignment);
468       }

470       private class WorkThread extends Thread {
471           public WorkThread() {
472               super("KafkaBasedLog Work Thread - " + topic);
473           }
474   
475           @Override
476           public void run() {
477               try {
478                   log.trace("{} started execution", this);
479                   while (true) {
480                       int numCallbacks;
481                       synchronized (KafkaBasedLog.this) {
482                           if (stopRequested)
483                               break;
484                           numCallbacks = readLogEndOffsetCallbacks.size();
485                       }
486   
487                       if (numCallbacks > 0) {
488                           try {
489                               readToLogEnd(false);
490                               log.trace("Finished read to end log for topic {}", topic);
491                           } catch (TimeoutException e) {
492                               log.warn("Timeout while reading log to end for topic '{}'. Retrying automatically. " +
493                                        "This may occur when brokers are unavailable or unreachable. Reason: {}", topic, e.getMessage());
494                               continue;
495                           } catch (RetriableException | org.apache.kafka.connect.errors.RetriableException e) {
496                               log.warn("Retriable error while reading log to end for topic '{}'. Retrying automatically. " +
497                                        "Reason: {}", topic, e.getMessage());
498                               continue;
499                           } catch (WakeupException e) {
500                               // Either received another get() call and need to retry reading to end of log or stop() was
501                               // called. Both are handled by restarting this loop.
502                               continue;
503                           }
504                       }
505   
506                       synchronized (KafkaBasedLog.this) {
507                           // Only invoke exactly the number of callbacks we found before triggering the read to log end
508                           // since it is possible for another write + readToEnd to sneak in the meantime
509                           for (int i = 0; i < numCallbacks; i++) {
510                               Callback<Void> cb = readLogEndOffsetCallbacks.poll();
511                               cb.onCompletion(null, null);
512                           }
513                       }
514   
515                       try {
516                           poll(Integer.MAX_VALUE);
517                       } catch (WakeupException e) {
518                           // See previous comment, both possible causes of this wakeup are handled by starting this loop again
519                           continue;
520                       }
521                   }
522               } catch (Throwable t) {
523                   log.error("Unexpected exception in {}", this, t);
524               }
525           }
526       }
527   }
