26   /**
27    * A utility for defining Mapped Diagnostic Context (MDC) for SLF4J logs.
28    *
29    * <p>{@link LoggingContext} instances should be created in a try-with-resources block to ensure
30    * that the logging context is properly closed. The only exception is the logging context created
31    * upon thread creation that is to be used for the entire lifetime of the thread.
32    *
33    * <p>Any logger created on the thread will inherit the MDC context, so this mechanism is ideal for
34    * providing additional information in the log messages without requiring connector
35    * implementations to use a specific Connect API or SLF4J API. {@link LoggingContext#close()}
36    * will also properly restore the Connect MDC parameters to their state just prior to when the
37    * LoggingContext was created. Use {@link #clear()} to remove all MDC parameters from the
38    * current thread context.
39    *
40    * <p>Compare this approach to {@link org.apache.kafka.common.utils.LogContext}, which must be
41    * used to create a new {@link org.slf4j.Logger} instance pre-configured with the desired prefix.
42    * Currently LogContext does not allow the prefix to be changed, and it requires that all
43    * components use the LogContext to create their Logger instance.
44    */
45   public final class LoggingContext implements AutoCloseable {
46   
47       /**
48        * The name of the Mapped Diagnostic Context (MDC) key that defines the context for a connector.
49        */
50       public static final String CONNECTOR_CONTEXT = "connector.context";
51   
52       public static final Collection<String> ALL_CONTEXTS = Collections.singleton(CONNECTOR_CONTEXT);
53   
54       /**
55        * The Scope values used by Connect when specifying the context.
56        */
57       public enum Scope {
58           /**
59            * The scope value for the worker as it starts a connector.
60            */
61           WORKER("worker"),
62   
63           /**
64            * The scope value for Task implementations.
65            */
66           TASK("task"),
67   
68           /**
69            * The scope value for committing offsets.
70            */
71           OFFSETS("offsets"),
72   
73           /**
74            * The scope value for validating connector configurations.
75            */
76           VALIDATE("validate");
77   
78           private final String text;
79           Scope(String value) {
80               this.text = value;
81           }
82   
83           @Override
84           public String toString() {
85               return text;
86           }
87       }
88   
89       /**
90        * Clear all MDC parameters.
91        */
92       public static void clear() {
93           MDC.clear();
94       }
95   
96       /**
97        * Modify the current {@link MDC} logging context to set the {@link #CONNECTOR_CONTEXT connector context} to include the
98        * supplied name and the {@link Scope#WORKER} scope.
99        *
100        * @param connectorName the connector name; may not be null
101        */
102       public static LoggingContext forConnector(String connectorName) {
103           Objects.requireNonNull(connectorName);
104           LoggingContext context = new LoggingContext();
105           MDC.put(CONNECTOR_CONTEXT, prefixFor(connectorName, Scope.WORKER, null));
106           return context;
107       }
108   
109       /**
110        * Modify the current {@link MDC} logging context to set the {@link #CONNECTOR_CONTEXT connector context} to include the
111        * supplied connector name and the {@link Scope#VALIDATE} scope.
112        *
113        * @param connectorName the connector name
114        */
115       public static LoggingContext forValidation(String connectorName) {
116           LoggingContext context = new LoggingContext();
117           MDC.put(CONNECTOR_CONTEXT, prefixFor(connectorName, Scope.VALIDATE, null));
118           return context;
119       }
120   
121       /**
122        * Modify the current {@link MDC} logging context to set the {@link #CONNECTOR_CONTEXT connector context} to include the
123        * connector name and task number using the supplied {@link ConnectorTaskId}, and to set the scope to {@link Scope#TASK}.
124        *
125        * @param id the connector task ID; may not be null
126        */
127       public static LoggingContext forTask(ConnectorTaskId id) {
128           Objects.requireNonNull(id);
129           LoggingContext context = new LoggingContext();
130           MDC.put(CONNECTOR_CONTEXT, prefixFor(id.connector(), Scope.TASK, id.task()));
131           return context;
132       }
133   
134       /**
135        * Modify the current {@link MDC} logging context to set the {@link #CONNECTOR_CONTEXT connector context} to include the
136        * connector name and task number using the supplied {@link ConnectorTaskId}, and to set the scope to {@link Scope#OFFSETS}.
137        *
138        * @param id the connector task ID; may not be null
139        */
140       public static LoggingContext forOffsets(ConnectorTaskId id) {
141           Objects.requireNonNull(id);
142           LoggingContext context = new LoggingContext();
143           MDC.put(CONNECTOR_CONTEXT, prefixFor(id.connector(), Scope.OFFSETS, id.task()));
144           return context;
145       }
146   
147       /**
148        * Return the prefix that uses the specified connector name, task number, and scope. The
149        * format is as follows:
150        *
151        * <pre>
152        *     [&lt;connectorName>|&lt;scope>]&lt;sp>
153        * </pre>
154        *
155        * where "<code>&lt;connectorName></code>" is the name of the connector,
156        * "<code>&lt;sp></code>" indicates a trailing space, and
157        * "<code>&lt;scope></code>" is one of the following:
158        *
159        * <ul>
160        *   <li>"<code>task-n</code>" for the operation of the numbered task, including calling the
161        *      task methods and the producer/consumer; here "n" is the 0-based task number
162        *   <li>"<code>task-n|offset</code>" for the committing of source offsets for the numbered
163        *       task; here "n" is the * zero-based task number
164        *   <li>"<code>worker</code>" for the creation and usage of connector instances
165        * </ul>
166        *
167        * <p>The following are examples of the connector context for a connector named "my-connector":
168        *
169        * <ul>
170        *   <li>`[my-connector|worker]` - used on log messages where the Connect worker is
171        *     validating the configuration for or starting/stopping the "local-file-source" connector
172        *     via the SourceConnector / SinkConnector implementation methods.
173        *   <li>`[my-connector|task-0]` - used on log messages where the Connect worker is executing
174        *     task 0 of the "local-file-source" connector, including calling any of the SourceTask /
175        *     SinkTask implementation methods, processing the messages for/from the task, and
176        *     calling the task's * producer/consumer.
177        *   <li>`[my-connector|task-0|offsets]` - used on log messages where the Connect worker is
178        *       committing * source offsets for task 0 of the "local-file-source" connector.
179        * </ul>
180        *
181        * @param connectorName the name of the connector; may not be null
182        * @param scope the scope; may not be null
183        * @param taskNumber the 0-based task number; may be null if there is no associated task
184        * @return the prefix; never null
185        */
186       protected static String prefixFor(String connectorName, Scope scope, Integer taskNumber) {
187           StringBuilder sb = new StringBuilder();
188           sb.append("[");
189           sb.append(connectorName);
190           if (taskNumber != null) {
191               // There is a task number, so this is a task
192               sb.append("|");
193               sb.append(Scope.TASK);
194               sb.append("-");
195               sb.append(taskNumber);
196           }
197           // Append non-task scopes (e.g., worker and offset)
198           if (scope != Scope.TASK) {
199               sb.append("|");
200               sb.append(scope.toString());
201           }
202           sb.append("] ");
203           return sb.toString();
204       }
205   
206       private final Map<String, String> previous;
207   
208       private LoggingContext() {
209           previous = MDC.getCopyOfContextMap(); // may be null!
210       }
211   
212       /**
213        * Close this logging context, restoring the Connect {@link MDC} parameters back to the state
214        * just before this context was created. This does not affect other MDC parameters set by
215        * connectors or tasks.
216        */
217       @Override
218       public void close() {
219           for (String param : ALL_CONTEXTS) {
220               if (previous != null && previous.containsKey(param)) {
221                   MDC.put(param, previous.get(param));
222               } else {
223                   MDC.remove(param);
224               }
225           }
226       }
227   }
