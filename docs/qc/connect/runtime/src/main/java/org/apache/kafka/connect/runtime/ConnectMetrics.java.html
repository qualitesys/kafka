64       /**
65        * Create an instance.
66        *
67        * @param workerId the worker identifier; may not be null
68        * @param config   the worker configuration; may not be null
69        * @param time     the time; may not be null
70        * @param clusterId the Kafka cluster ID
71        */
72       public ConnectMetrics(String workerId, WorkerConfig config, Time time, String clusterId) {
73           this.workerId = workerId;
74           this.time = time;
75   
76           int numSamples = config.getInt(CommonClientConfigs.METRICS_NUM_SAMPLES_CONFIG);
77           long sampleWindowMs = config.getLong(CommonClientConfigs.METRICS_SAMPLE_WINDOW_MS_CONFIG);
78           String metricsRecordingLevel = config.getString(CommonClientConfigs.METRICS_RECORDING_LEVEL_CONFIG);
79           List<MetricsReporter> reporters = config.getConfiguredInstances(CommonClientConfigs.METRIC_REPORTER_CLASSES_CONFIG, MetricsReporter.class);
80   
81           MetricConfig metricConfig = new MetricConfig().samples(numSamples)
82                   .timeWindow(sampleWindowMs, TimeUnit.MILLISECONDS).recordLevel(
83                           Sensor.RecordingLevel.forName(metricsRecordingLevel));
84           JmxReporter jmxReporter = new JmxReporter();
85           jmxReporter.configure(config.originals());
86           reporters.add(jmxReporter);
87   
88           Map<String, Object> contextLabels = new HashMap<>();
89           contextLabels.putAll(config.originalsWithPrefix(CommonClientConfigs.METRICS_CONTEXT_PREFIX));
90           contextLabels.put(WorkerConfig.CONNECT_KAFKA_CLUSTER_ID, clusterId);
91           Object groupId = config.originals().get(DistributedConfig.GROUP_ID_CONFIG);
92           if (groupId != null) {
93               contextLabels.put(WorkerConfig.CONNECT_GROUP_ID, groupId);
94           }
95           MetricsContext metricsContext = new KafkaMetricsContext(JMX_PREFIX, contextLabels);
96           this.metrics = new Metrics(metricConfig, reporters, time, metricsContext);
97   
98           LOG.debug("Registering Connect metrics with JMX for worker '{}'", workerId);
99           AppInfoParser.registerAppInfo(JMX_PREFIX, workerId, metrics, time.milliseconds());
100       }

130       /**
131        * Get or create a {@link MetricGroup} with the specified group name and the given tags.
132        * Each group is uniquely identified by the name and tags.
133        *
134        * @param groupName    the name of the metric group; may not be null
135        * @param tagKeyValues pairs of tag name and values
136        * @return the {@link MetricGroup} that can be used to create metrics; never null
137        * @throws IllegalArgumentException if the group name is not valid
138        */
139       public MetricGroup group(String groupName, String... tagKeyValues) {
140           MetricGroupId groupId = groupId(groupName, tagKeyValues);
141           MetricGroup group = groupsByName.get(groupId);
142           if (group == null) {
143               group = new MetricGroup(groupId);
144               MetricGroup previous = groupsByName.putIfAbsent(groupId, group);
145               if (previous != null)
146                   group = previous;
147           }
148           return group;
149       }
150   
151       protected MetricGroupId groupId(String groupName, String... tagKeyValues) {
152           Map<String, String> tags = MetricsUtils.getTags(tagKeyValues);
153           return new MetricGroupId(groupName, tags);
154       }

174       public static class MetricGroupId {
175           private final String groupName;
176           private final Map<String, String> tags;
177           private final int hc;
178           private final String str;
179   
180           public MetricGroupId(String groupName, Map<String, String> tags) {
181               Objects.requireNonNull(groupName);
182               Objects.requireNonNull(tags);
183               this.groupName = groupName;
184               this.tags = Collections.unmodifiableMap(new LinkedHashMap<>(tags));
185               this.hc = Objects.hash(this.groupName, this.tags);
186               StringBuilder sb = new StringBuilder(this.groupName);
187               for (Map.Entry<String, String> entry : this.tags.entrySet()) {
188                   sb.append(";").append(entry.getKey()).append('=').append(entry.getValue());
189               }
190               this.str = sb.toString();
191           }

194            * Get the group name.
195            *
196            * @return the group name; never null
197            */

199               return groupName;
200           }

202           /**
203            * Get the immutable map of tag names and values.
204            *
205            * @return the tags; never null
206            */
207           public Map<String, String> tags() {
208               return tags;
209           }
210   
211           /**
212            * Determine if the supplied metric name is part of this group identifier.
213            *
214            * @param metricName the metric name
215            * @return true if the metric name's group and tags match this group identifier, or false otherwise
216            */
217           public boolean includes(MetricName metricName) {
218               return metricName != null && groupName.equals(metricName.group()) && tags.equals(metricName.tags());
219           }
220   
221           @Override
222           public int hashCode() {
223               return hc;
224           }
225   
226           @Override
227           public boolean equals(Object obj) {
228               if (obj == this)
229                   return true;
230               if (obj instanceof MetricGroupId) {
231                   MetricGroupId that = (MetricGroupId) obj;
232                   return this.groupName.equals(that.groupName) && this.tags.equals(that.tags);
233               }
234               return false;
235           }

237           @Override
238           public String toString() {
239               return str;
240           }

243       /**
244        * A group of metrics. Each group maps to a JMX MBean and each metric maps to an MBean attribute.
245        * <p>
246        * Sensors should be added via the {@code sensor} methods on this class, rather than directly through
247        * the {@link Metrics} class, so that the sensor names are made to be unique (based on the group name)
248        * and so the sensors are removed when this group is {@link #close() closed}.
249        */
250       public class MetricGroup implements AutoCloseable {
251           private final MetricGroupId groupId;
252           private final Set<String> sensorNames = new HashSet<>();
253           private final String sensorPrefix;
254   
255           /**
256            * Create a group of Connect metrics.
257            *
258            * @param groupId the identifier of the group; may not be null and must be valid
259            */
260           protected MetricGroup(MetricGroupId groupId) {
261               Objects.requireNonNull(groupId);
262               this.groupId = groupId;
263               sensorPrefix = "connect-sensor-group: " + groupId + ";";
264           }
265   
266           /**
267            * Get the group identifier.
268            *

270            */
271           public MetricGroupId groupId() {
272               return groupId;
273           }

275           /**
276            * Create the name of a metric that belongs to this group and has the group's tags.
277            *
278            * @param template the name template for the metric; may not be null
279            * @return the metric name; never null
280            * @throws IllegalArgumentException if the name is not valid
281            */
282           public MetricName metricName(MetricNameTemplate template) {
283               return metrics.metricInstance(template, groupId.tags());

295            * {@link #close() closed}. Metrics can be added directly, as long as the metric names are obtained from
296            * this group via the {@link #metricName(MetricNameTemplate)} method.

313           /**
314            * Add to this group an indicator metric with a function that returns the current value.
315            *
316            * @param nameTemplate the name template for the metric; may not be null
317            * @param supplier     the function used to determine the literal value of the metric; may not be null
318            * @throws IllegalArgumentException if the name is not valid
319            */
320           public <T> void addValueMetric(MetricNameTemplate nameTemplate, final LiteralSupplier<T> supplier) {
321               MetricName metricName = metricName(nameTemplate);
322               metrics().addMetricIfAbsent(metricName, null, (Gauge<T>) (config, now) -> supplier.metricValue(now));
323           }

330            * @throws IllegalArgumentException if the name is not valid

332           public <T> void addImmutableValueMetric(MetricNameTemplate nameTemplate, final T value) {
333               MetricName metricName = metricName(nameTemplate);
334               metrics().addMetricIfAbsent(metricName, null, (Gauge<T>) (config, now) -> value);
335           }

385           /**
386            * Get or create a sensor with the given unique name and zero or more parent sensors. All parent sensors will
387            * receive every value recorded with this sensor.
388            *
389            * @param name           The name of the sensor
390            * @param config         A default configuration to use for this sensor for metrics that don't have their own config
391            * @param recordingLevel The recording level.
392            * @param parents        The parent sensors
393            * @return The sensor that is created
394            */
395           public synchronized Sensor sensor(String name, MetricConfig config, Sensor.RecordingLevel recordingLevel, Sensor... parents) {
396               // We need to make sure that all sensor names are unique across all groups, so use the sensor prefix
397               Sensor result = metrics.sensor(sensorPrefix + name, config, Long.MAX_VALUE, recordingLevel, parents);
398               if (result != null)
399                   sensorNames.add(result.name());
400               return result;
401           }
402   
403           /**
404            * Remove all sensors and metrics associated with this group.
405            */
406           public synchronized void close() {
407               for (String sensorName : sensorNames) {
408                   metrics.removeSensor(sensorName);
409               }
410               sensorNames.clear();
411               for (MetricName metricName : new HashSet<>(metrics.metrics().keySet())) {
412                   if (groupId.includes(metricName)) {
413                       metrics.removeMetric(metricName);
414                   }
415               }
416           }
417       }
