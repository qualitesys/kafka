54   /**
55    * An assignor that computes a distribution of connectors and tasks according to the incremental
56    * cooperative strategy for rebalancing. {@see
57    * https://cwiki.apache.org/confluence/display/KAFKA/KIP-415%3A+Incremental+Cooperative
58    * +Rebalancing+in+Kafka+Connect} for a description of the assignment policy.
59    *
60    * Note that this class is NOT thread-safe.
61    */
62   public class IncrementalCooperativeAssignor implements ConnectAssignor {
63       private final Logger log;
64       private final Time time;
65       private final int maxDelay;
66       private ConnectorsAndTasks previousAssignment;
67       private final ConnectorsAndTasks previousRevocation;
68       private boolean canRevoke;
69       // visible for testing
70       protected final Set<String> candidateWorkersForReassignment;
71       protected long scheduledRebalance;
72       protected int delay;
73       protected int previousGenerationId;
74       protected Set<String> previousMembers;

76       public IncrementalCooperativeAssignor(LogContext logContext, Time time, int maxDelay) {
77           this.log = logContext.logger(IncrementalCooperativeAssignor.class);
78           this.time = time;
79           this.maxDelay = maxDelay;
80           this.previousAssignment = ConnectorsAndTasks.EMPTY;
81           this.previousRevocation = new ConnectorsAndTasks.Builder().build();
82           this.canRevoke = true;
83           this.scheduledRebalance = 0;
84           this.candidateWorkersForReassignment = new LinkedHashSet<>();
85           this.delay = 0;
86           this.previousGenerationId = -1;
87           this.previousMembers = Collections.emptySet();
88       }

90       @Override
91       public Map<String, ByteBuffer> performAssignment(String leaderId, String protocol,
92                                                        List<JoinGroupResponseMember> allMemberMetadata,
93                                                        WorkerCoordinator coordinator) {
94           log.debug("Performing task assignment");
95   
96           Map<String, ExtendedWorkerState> memberConfigs = new HashMap<>();
97           for (JoinGroupResponseMember member : allMemberMetadata) {
98               memberConfigs.put(
99                       member.memberId(),
100                       IncrementalCooperativeConnectProtocol.deserializeMetadata(ByteBuffer.wrap(member.metadata())));
101           }
102           log.debug("Member configs: {}", memberConfigs);
103   
104           // The new config offset is the maximum seen by any member. We always perform assignment using this offset,
105           // even if some members have fallen behind. The config offset used to generate the assignment is included in
106           // the response so members that have fallen behind will not use the assignment until they have caught up.
107           long maxOffset = memberConfigs.values().stream().map(ExtendedWorkerState::offset).max(Long::compare).get();
108           log.debug("Max config offset root: {}, local snapshot config offsets root: {}",
109                     maxOffset, coordinator.configSnapshot().offset());
110   
111           short protocolVersion = ConnectProtocolCompatibility.fromProtocol(protocol).protocolVersion();
112   
113           Long leaderOffset = ensureLeaderConfig(maxOffset, coordinator);
114           if (leaderOffset == null) {
115               Map<String, ExtendedAssignment> assignments = fillAssignments(
116                       memberConfigs.keySet(), Assignment.CONFIG_MISMATCH,
117                       leaderId, memberConfigs.get(leaderId).url(), maxOffset,
118                       ClusterAssignment.EMPTY, 0, protocolVersion);
119               return serializeAssignments(assignments, protocolVersion);
120           }
121           return performTaskAssignment(leaderId, leaderOffset, memberConfigs, coordinator, protocolVersion);
122       }
123   
124       private Long ensureLeaderConfig(long maxOffset, WorkerCoordinator coordinator) {
125           // If this leader is behind some other members, we can't do assignment
126           if (coordinator.configSnapshot().offset() < maxOffset) {
127               // We might be able to take a new snapshot to catch up immediately and avoid another round of syncing here.
128               // Alternatively, if this node has already passed the maximum reported by any other member of the group, it
129               // is also safe to use this newer state.
130               ClusterConfigState updatedSnapshot = coordinator.configFreshSnapshot();
131               if (updatedSnapshot.offset() < maxOffset) {
132                   log.info("Was selected to perform assignments, but do not have latest config found in sync request. "
133                            + "Returning an empty configuration to trigger re-sync.");
134                   return null;
135               } else {
136                   coordinator.configSnapshot(updatedSnapshot);
137                   return updatedSnapshot.offset();
138               }
139           }
140           return maxOffset;
141       }

143       /**
144        * Performs task assignment based on the incremental cooperative connect protocol.
145        * Read more on the design and implementation in:
146        * {@see https://cwiki.apache.org/confluence/display/KAFKA/KIP-415%3A+Incremental+Cooperative+Rebalancing+in+Kafka+Connect}
147        *
148        * @param leaderId the ID of the group leader
149        * @param maxOffset the latest known offset of the configuration topic
150        * @param memberConfigs the metadata of all the members of the group as gather in the current
151        * round of rebalancing
152        * @param coordinator the worker coordinator instance that provide the configuration snapshot
153        * and get assigned the leader state during this assignment
154        * @param protocolVersion the Connect subprotocol version
155        * @return the serialized assignment of tasks to the whole group, including assigned or
156        * revoked tasks
157        */
158       protected Map<String, ByteBuffer> performTaskAssignment(String leaderId, long maxOffset,
159                                                               Map<String, ExtendedWorkerState> memberConfigs,
160                                                               WorkerCoordinator coordinator, short protocolVersion) {
161           log.debug("Performing task assignment during generation: {} with memberId: {}",
162                   coordinator.generationId(), coordinator.memberId());
163           Map<String, ConnectorsAndTasks> memberAssignments = transformValues(
164                   memberConfigs,
165                   memberConfig -> new ConnectorsAndTasks.Builder()
166                           .with(memberConfig.assignment().connectors(), memberConfig.assignment().tasks())
167                           .build()
168           );
169           ClusterAssignment clusterAssignment = performTaskAssignment(
170                   coordinator.configSnapshot(),
171                   coordinator.lastCompletedGenerationId(),
172                   coordinator.generationId(),
173                   memberAssignments
174           );
175   
176           coordinator.leaderState(new LeaderState(memberConfigs, clusterAssignment.allAssignedConnectors(), clusterAssignment.allAssignedTasks()));
177   
178           Map<String, ExtendedAssignment> assignments =
179                   fillAssignments(memberConfigs.keySet(), Assignment.NO_ERROR, leaderId,
180                           memberConfigs.get(leaderId).url(), maxOffset,
181                           clusterAssignment,
182                           delay, protocolVersion);
183   
184           log.debug("Actual assignments: {}", assignments);
185           return serializeAssignments(assignments, protocolVersion);
186       }
187   
188       // Visible for testing
189       ClusterAssignment performTaskAssignment(
190               ClusterConfigState configSnapshot,
191               int lastCompletedGenerationId,
192               int currentGenerationId,
193               Map<String, ConnectorsAndTasks> memberAssignments
194       ) {
195           // Base set: The previous assignment of connectors-and-tasks is a standalone snapshot that
196           // can be used to calculate derived sets
197           log.debug("Previous assignments: {}", previousAssignment);
198           if (previousGenerationId != lastCompletedGenerationId) {
199               log.debug("Clearing the view of previous assignments due to generation mismatch between "
200                       + "previous generation ID {} and last completed generation ID {}. This can "
201                       + "happen if the leader fails to sync the assignment within a rebalancing round. "
202                       + "The following view of previous assignments might be outdated and will be "
203                       + "ignored by the leader in the current computation of new assignments. "
204                       + "Possibly outdated previous assignments: {}",
205                       previousGenerationId, lastCompletedGenerationId, previousAssignment);
206               this.previousAssignment = ConnectorsAndTasks.EMPTY;
207           }
208   
209           Set<String> configuredConnectors = new TreeSet<>(configSnapshot.connectors());
210           Set<ConnectorTaskId> configuredTasks = combineCollections(configuredConnectors, configSnapshot::tasks, Collectors.toSet());
211   
212           // Base set: The set of configured connectors-and-tasks is a standalone snapshot that can
213           // be used to calculate derived sets
214           ConnectorsAndTasks configured = new ConnectorsAndTasks.Builder()
215                   .with(configuredConnectors, configuredTasks).build();
216           log.debug("Configured assignments: {}", configured);
217   
218           // Base set: The set of active connectors-and-tasks is a standalone snapshot that can be
219           // used to calculate derived sets
220           ConnectorsAndTasks activeAssignments = assignment(memberAssignments);
221           log.debug("Active assignments: {}", activeAssignments);
222   
223           // This means that a previous revocation did not take effect. In this case, reset
224           // appropriately and be ready to re-apply revocation of tasks
225           if (!previousRevocation.isEmpty()) {
226               if (previousRevocation.connectors().stream().anyMatch(c -> activeAssignments.connectors().contains(c))
227                       || previousRevocation.tasks().stream().anyMatch(t -> activeAssignments.tasks().contains(t))) {
228                   previousAssignment = activeAssignments;
229                   canRevoke = true;
230               }
231               previousRevocation.connectors().clear();
232               previousRevocation.tasks().clear();
233           }
234   
235           // Derived set: The set of deleted connectors-and-tasks is a derived set from the set
236           // difference of previous - configured
237           ConnectorsAndTasks deleted = diff(previousAssignment, configured);
238           log.debug("Deleted assignments: {}", deleted);
239   
240           // Derived set: The set of remaining active connectors-and-tasks is a derived set from the
241           // set difference of active - deleted
242           ConnectorsAndTasks remainingActive = diff(activeAssignments, deleted);
243           log.debug("Remaining (excluding deleted) active assignments: {}", remainingActive);
244   
245           // Derived set: The set of lost or unaccounted connectors-and-tasks is a derived set from
246           // the set difference of previous - active - deleted
247           ConnectorsAndTasks lostAssignments = diff(previousAssignment, activeAssignments, deleted);
248           log.debug("Lost assignments: {}", lostAssignments);
249   
250           // Derived set: The set of new connectors-and-tasks is a derived set from the set
251           // difference of configured - previous - active
252           ConnectorsAndTasks newSubmissions = diff(configured, previousAssignment, activeAssignments);
253           log.debug("New assignments: {}", newSubmissions);
254   
255           // A collection of the complete assignment
256           List<WorkerLoad> completeWorkerAssignment = workerAssignment(memberAssignments, ConnectorsAndTasks.EMPTY);
257           log.debug("Complete (ignoring deletions) worker assignments: {}", completeWorkerAssignment);
258   
259           // Per worker connector assignments without removing deleted connectors yet
260           Map<String, Collection<String>> connectorAssignments =
261                   completeWorkerAssignment.stream().collect(Collectors.toMap(WorkerLoad::worker, WorkerLoad::connectors));
262           log.debug("Complete (ignoring deletions) connector assignments: {}", connectorAssignments);
263   
264           // Per worker task assignments without removing deleted connectors yet
265           Map<String, Collection<ConnectorTaskId>> taskAssignments =
266                   completeWorkerAssignment.stream().collect(Collectors.toMap(WorkerLoad::worker, WorkerLoad::tasks));
267           log.debug("Complete (ignoring deletions) task assignments: {}", taskAssignments);
268   
269           // A collection of the current assignment excluding the connectors-and-tasks to be deleted
270           List<WorkerLoad> currentWorkerAssignment = workerAssignment(memberAssignments, deleted);
271   
272           Map<String, ConnectorsAndTasks> toRevoke = computeDeleted(deleted, connectorAssignments, taskAssignments);
273           log.debug("Connector and task to delete assignments: {}", toRevoke);
274   
275           // Revoking redundant connectors/tasks if the workers have duplicate assignments
276           toRevoke.putAll(computeDuplicatedAssignments(memberAssignments, connectorAssignments, taskAssignments));
277           log.debug("Connector and task to revoke assignments (include duplicated assignments): {}", toRevoke);
278   
279           // Recompute the complete assignment excluding the deleted connectors-and-tasks
280           completeWorkerAssignment = workerAssignment(memberAssignments, deleted);
281           connectorAssignments =
282                   completeWorkerAssignment.stream().collect(Collectors.toMap(WorkerLoad::worker, WorkerLoad::connectors));
283           taskAssignments =
284                   completeWorkerAssignment.stream().collect(Collectors.toMap(WorkerLoad::worker, WorkerLoad::tasks));
285   
286           handleLostAssignments(lostAssignments, newSubmissions, completeWorkerAssignment);
287   
288           // Do not revoke resources for re-assignment while a delayed rebalance is active
289           // Also we do not revoke in two consecutive rebalances by the same leader
290           canRevoke = delay == 0 && canRevoke;
291   
292           // Compute the connectors-and-tasks to be revoked for load balancing without taking into
293           // account the deleted ones.
294           log.debug("Can leader revoke tasks in this assignment? {} (delay: {})", canRevoke, delay);
295           if (canRevoke) {
296               Map<String, ConnectorsAndTasks> toExplicitlyRevoke =
297                       performTaskRevocation(activeAssignments, currentWorkerAssignment);
298   
299               log.debug("Connector and task to revoke assignments: {}", toRevoke);
300   
301               toExplicitlyRevoke.forEach(
302                   (worker, assignment) -> {
303                       ConnectorsAndTasks existing = toRevoke.computeIfAbsent(
304                           worker,
305                           v -> new ConnectorsAndTasks.Builder().build());
306                       existing.connectors().addAll(assignment.connectors());
307                       existing.tasks().addAll(assignment.tasks());
308                   }
309               );
310               canRevoke = toExplicitlyRevoke.size() == 0;
311           } else {
312               canRevoke = delay == 0;
313           }
314   
315           assignConnectors(completeWorkerAssignment, newSubmissions.connectors());
316           assignTasks(completeWorkerAssignment, newSubmissions.tasks());
317           log.debug("Current complete assignments: {}", currentWorkerAssignment);
318           log.debug("New complete assignments: {}", completeWorkerAssignment);
319   
320           Map<String, Collection<String>> currentConnectorAssignments =
321                   currentWorkerAssignment.stream().collect(Collectors.toMap(WorkerLoad::worker, WorkerLoad::connectors));
322           Map<String, Collection<ConnectorTaskId>> currentTaskAssignments =
323                   currentWorkerAssignment.stream().collect(Collectors.toMap(WorkerLoad::worker, WorkerLoad::tasks));
324           Map<String, Collection<String>> incrementalConnectorAssignments =
325                   diff(connectorAssignments, currentConnectorAssignments);
326           Map<String, Collection<ConnectorTaskId>> incrementalTaskAssignments =
327                   diff(taskAssignments, currentTaskAssignments);
328   
329           previousAssignment = computePreviousAssignment(toRevoke, connectorAssignments, taskAssignments, lostAssignments);
330           previousGenerationId = currentGenerationId;
331           previousMembers = memberAssignments.keySet();
332   
333           log.debug("Incremental connector assignments: {}", incrementalConnectorAssignments);
334           log.debug("Incremental task assignments: {}", incrementalTaskAssignments);
335   
336           Map<String, Collection<String>> revokedConnectors = transformValues(toRevoke, ConnectorsAndTasks::connectors);
337           Map<String, Collection<ConnectorTaskId>> revokedTasks = transformValues(toRevoke, ConnectorsAndTasks::tasks);
338   
339           return new ClusterAssignment(
340                   incrementalConnectorAssignments,
341                   incrementalTaskAssignments,
342                   revokedConnectors,
343                   revokedTasks,
344                   diff(connectorAssignments, revokedConnectors),
345                   diff(taskAssignments, revokedTasks)
346           );
347       }

349       private Map<String, ConnectorsAndTasks> computeDeleted(ConnectorsAndTasks deleted,
350                                                              Map<String, Collection<String>> connectorAssignments,
351                                                              Map<String, Collection<ConnectorTaskId>> taskAssignments) {
352           // Connector to worker reverse lookup map
353           Map<String, String> connectorOwners = WorkerCoordinator.invertAssignment(connectorAssignments);
354           // Task to worker reverse lookup map
355           Map<ConnectorTaskId, String> taskOwners = WorkerCoordinator.invertAssignment(taskAssignments);
356   
357           Map<String, ConnectorsAndTasks> toRevoke = new HashMap<>();
358           // Add the connectors that have been deleted to the revoked set
359           deleted.connectors().forEach(c ->
360                   toRevoke.computeIfAbsent(
361                       connectorOwners.get(c),
362                       v -> new ConnectorsAndTasks.Builder().build()
363                   ).connectors().add(c));
364           // Add the tasks that have been deleted to the revoked set
365           deleted.tasks().forEach(t ->
366                   toRevoke.computeIfAbsent(
367                       taskOwners.get(t),
368                       v -> new ConnectorsAndTasks.Builder().build()
369                   ).tasks().add(t));
370           log.debug("Connectors and tasks to delete assignments: {}", toRevoke);
371           return toRevoke;
372       }

374       private ConnectorsAndTasks computePreviousAssignment(Map<String, ConnectorsAndTasks> toRevoke,
375                                                            Map<String, Collection<String>> connectorAssignments,
376                                                            Map<String, Collection<ConnectorTaskId>> taskAssignments,
377                                                            ConnectorsAndTasks lostAssignments) {
378           ConnectorsAndTasks previousAssignment = new ConnectorsAndTasks.Builder().with(
379                   ConnectUtils.combineCollections(connectorAssignments.values()),
380                   ConnectUtils.combineCollections(taskAssignments.values())
381           ).build();
382   
383           for (ConnectorsAndTasks revoked : toRevoke.values()) {
384               previousAssignment.connectors().removeAll(revoked.connectors());
385               previousAssignment.tasks().removeAll(revoked.tasks());
386               previousRevocation.connectors().addAll(revoked.connectors());
387               previousRevocation.tasks().addAll(revoked.tasks());
388           }
389   
390           // Depends on the previous assignment's collections being sets at the moment.
391           // TODO: make it independent
392           previousAssignment.connectors().addAll(lostAssignments.connectors());
393           previousAssignment.tasks().addAll(lostAssignments.tasks());
394   
395           return previousAssignment;
396       }

398       private ConnectorsAndTasks duplicatedAssignments(Map<String, ConnectorsAndTasks> memberAssignments) {
399           Map<String, Long> connectorInstanceCounts = combineCollections(
400                   memberAssignments.values(),
401                   ConnectorsAndTasks::connectors,
402                   Collectors.groupingBy(Function.identity(), Collectors.counting())
403           );

460       // visible for testing
461       protected void handleLostAssignments(ConnectorsAndTasks lostAssignments,
462                                            ConnectorsAndTasks newSubmissions,
463                                            List<WorkerLoad> completeWorkerAssignment) {
464           if (lostAssignments.isEmpty()) {
465               resetDelay();
466               return;
467           }
468   
469           final long now = time.milliseconds();
470           log.debug("Found the following connectors and tasks missing from previous assignments: "
471                   + lostAssignments);
472   
473           Set<String> activeMembers = completeWorkerAssignment.stream()
474                   .map(WorkerLoad::worker)
475                   .collect(Collectors.toSet());
476           if (scheduledRebalance <= 0 && activeMembers.containsAll(previousMembers)) {
477               log.debug("No worker seems to have departed the group during the rebalance. The "
478                       + "missing assignments that the leader is detecting are probably due to some "
479                       + "workers failing to receive the new assignments in the previous rebalance. "
480                       + "Will reassign missing tasks as new tasks");
481               newSubmissions.connectors().addAll(lostAssignments.connectors());
482               newSubmissions.tasks().addAll(lostAssignments.tasks());
483               return;
484           }
485   
486           if (scheduledRebalance > 0 && now >= scheduledRebalance) {
487               // delayed rebalance expired and it's time to assign resources
488               log.debug("Delayed rebalance expired. Reassigning lost tasks");
489               List<WorkerLoad> candidateWorkerLoad = Collections.emptyList();
490               if (!candidateWorkersForReassignment.isEmpty()) {
491                   candidateWorkerLoad = pickCandidateWorkerForReassignment(completeWorkerAssignment);
492               }
493   
494               if (!candidateWorkerLoad.isEmpty()) {
495                   log.debug("Assigning lost tasks to {} candidate workers: {}", 
496                           candidateWorkerLoad.size(),
497                           candidateWorkerLoad.stream().map(WorkerLoad::worker).collect(Collectors.joining(",")));
498                   Iterator<WorkerLoad> candidateWorkerIterator = candidateWorkerLoad.iterator();
499                   for (String connector : lostAssignments.connectors()) {
500                       // Loop over the candidate workers as many times as it takes
501                       if (!candidateWorkerIterator.hasNext()) {
502                           candidateWorkerIterator = candidateWorkerLoad.iterator();
503                       }
504                       WorkerLoad worker = candidateWorkerIterator.next();
505                       log.debug("Assigning connector id {} to member {}", connector, worker.worker());
506                       worker.assign(connector);
507                   }
508                   candidateWorkerIterator = candidateWorkerLoad.iterator();
509                   for (ConnectorTaskId task : lostAssignments.tasks()) {
510                       if (!candidateWorkerIterator.hasNext()) {
511                           candidateWorkerIterator = candidateWorkerLoad.iterator();
512                       }
513                       WorkerLoad worker = candidateWorkerIterator.next();
514                       log.debug("Assigning task id {} to member {}", task, worker.worker());
515                       worker.assign(task);
516                   }
517               } else {
518                   log.debug("No single candidate worker was found to assign lost tasks. Treating lost tasks as new tasks");
519                   newSubmissions.connectors().addAll(lostAssignments.connectors());
520                   newSubmissions.tasks().addAll(lostAssignments.tasks());
521               }
522               resetDelay();
523           } else {
524               candidateWorkersForReassignment
525                       .addAll(candidateWorkersForReassignment(completeWorkerAssignment));
526               if (now < scheduledRebalance) {
527                   // a delayed rebalance is in progress, but it's not yet time to reassign
528                   // unaccounted resources
529                   delay = calculateDelay(now);
530                   log.debug("Delayed rebalance in progress. Task reassignment is postponed. New computed rebalance delay: {}", delay);
531               } else {
532                   // This means scheduledRebalance == 0
533                   // We could also extract the current minimum delay from the group, to make
534                   // independent of consecutive leader failures, but this optimization is skipped
535                   // at the moment
536                   delay = maxDelay;
537                   log.debug("Resetting rebalance delay to the max: {}. scheduledRebalance: {} now: {} diff scheduledRebalance - now: {}",
538                           delay, scheduledRebalance, now, scheduledRebalance - now);
539               }
540               scheduledRebalance = now + delay;
541           }
542       }

548               log.debug("Resetting delay from previous value: {} to 0", delay);

569       /**
570        * Task revocation is based on a rough estimation of the lower average number of tasks before
571        * and after new workers join the group. If no new workers join, no revocation takes place.
572        * Based on this estimation, tasks are revoked until the new floor average is reached for
573        * each existing worker. The revoked tasks, once assigned to the new workers will maintain
574        * a balanced load among the group.
575        *
576        * @param activeAssignments
577        * @param completeWorkerAssignment
578        * @return
579        */
580       private Map<String, ConnectorsAndTasks> performTaskRevocation(ConnectorsAndTasks activeAssignments,
581                                                                     Collection<WorkerLoad> completeWorkerAssignment) {
582           int totalActiveConnectorsNum = activeAssignments.connectors().size();
583           int totalActiveTasksNum = activeAssignments.tasks().size();
584           Collection<WorkerLoad> existingWorkers = completeWorkerAssignment.stream()
585                   .filter(wl -> wl.size() > 0)
586                   .collect(Collectors.toList());
587           int existingWorkersNum = existingWorkers.size();
588           int totalWorkersNum = completeWorkerAssignment.size();
589           int newWorkersNum = totalWorkersNum - existingWorkersNum;
590   
591           if (log.isDebugEnabled()) {
592               completeWorkerAssignment.forEach(wl -> log.debug(
593                       "Per worker current load size; worker: {} connectors: {} tasks: {}",
594                       wl.worker(), wl.connectorsSize(), wl.tasksSize()));
595           }
596   
597           Map<String, ConnectorsAndTasks> revoking = new HashMap<>();
598           // If there are no new workers, or no existing workers to revoke tasks from return early
599           // after logging the status
600           if (!(newWorkersNum > 0 && existingWorkersNum > 0)) {
601               log.debug("No task revocation required; workers with existing load: {} workers with "
602                       + "no load {} total workers {}",
603                       existingWorkersNum, newWorkersNum, totalWorkersNum);
604               // This is intentionally empty but mutable, because the map is used to include deleted
605               // connectors and tasks as well
606               return revoking;
607           }
608   
609           log.debug("Task revocation is required; workers with existing load: {} workers with "
610                   + "no load {} total workers {}",
611                   existingWorkersNum, newWorkersNum, totalWorkersNum);
612   
613           // We have at least one worker assignment (the leader itself) so totalWorkersNum can't be 0
614           log.debug("Previous rounded down (floor) average number of connectors per worker {}", totalActiveConnectorsNum / existingWorkersNum);
615           int floorConnectors = totalActiveConnectorsNum / totalWorkersNum;
616           int ceilConnectors = floorConnectors + ((totalActiveConnectorsNum % totalWorkersNum == 0) ? 0 : 1);
617           log.debug("New average number of connectors per worker rounded down (floor) {} and rounded up (ceil) {}", floorConnectors, ceilConnectors);
618   
619   
620           log.debug("Previous rounded down (floor) average number of tasks per worker {}", totalActiveTasksNum / existingWorkersNum);
621           int floorTasks = totalActiveTasksNum / totalWorkersNum;
622           int ceilTasks = floorTasks + ((totalActiveTasksNum % totalWorkersNum == 0) ? 0 : 1);
623           log.debug("New average number of tasks per worker rounded down (floor) {} and rounded up (ceil) {}", floorTasks, ceilTasks);
624           int numToRevoke;
625   
626           for (WorkerLoad existing : existingWorkers) {
627               Iterator<String> connectors = existing.connectors().iterator();
628               numToRevoke = existing.connectorsSize() - ceilConnectors;
629               for (int i = existing.connectorsSize(); i > floorConnectors && numToRevoke > 0; --i, --numToRevoke) {
630                   ConnectorsAndTasks resources = revoking.computeIfAbsent(
631                       existing.worker(),
632                       w -> new ConnectorsAndTasks.Builder().build());
633                   resources.connectors().add(connectors.next());
634               }
635           }
636   
637           for (WorkerLoad existing : existingWorkers) {
638               Iterator<ConnectorTaskId> tasks = existing.tasks().iterator();
639               numToRevoke = existing.tasksSize() - ceilTasks;
640               log.debug("Tasks on worker {} is higher than ceiling, so revoking {} tasks", existing, numToRevoke);
641               for (int i = existing.tasksSize(); i > floorTasks && numToRevoke > 0; --i, --numToRevoke) {
642                   ConnectorsAndTasks resources = revoking.computeIfAbsent(
643                       existing.worker(),
644                       w -> new ConnectorsAndTasks.Builder().build());
645                   resources.tasks().add(tasks.next());
646               }
647           }
648   
649           return revoking;
650       }

718       private int calculateDelay(long now) {

723       /**
724        * Perform a round-robin assignment of connectors to workers with existing worker load. This
725        * assignment tries to balance the load between workers, by assigning connectors to workers
726        * that have equal load, starting with the least loaded workers.
727        *
728        * @param workerAssignment the current worker assignment; assigned connectors are added to this list
729        * @param connectors the connectors to be assigned
730        */
731       protected void assignConnectors(List<WorkerLoad> workerAssignment, Collection<String> connectors) {
732           workerAssignment.sort(WorkerLoad.connectorComparator());
733           WorkerLoad first = workerAssignment.get(0);
734   
735           Iterator<String> load = connectors.iterator();
736           while (load.hasNext()) {
737               int firstLoad = first.connectorsSize();
738               int upTo = IntStream.range(0, workerAssignment.size())
739                       .filter(i -> workerAssignment.get(i).connectorsSize() > firstLoad)
740                       .findFirst()
741                       .orElse(workerAssignment.size());
742               for (WorkerLoad worker : workerAssignment.subList(0, upTo)) {
743                   String connector = load.next();
744                   log.debug("Assigning connector {} to {}", connector, worker.worker());
745                   worker.assign(connector);
746                   if (!load.hasNext()) {
747                       break;
748                   }
749               }
750           }
751       }
752   
753       /**
754        * Perform a round-robin assignment of tasks to workers with existing worker load. This
755        * assignment tries to balance the load between workers, by assigning tasks to workers that
756        * have equal load, starting with the least loaded workers.
757        *
758        * @param workerAssignment the current worker assignment; assigned tasks are added to this list
759        * @param tasks the tasks to be assigned
760        */
761       protected void assignTasks(List<WorkerLoad> workerAssignment, Collection<ConnectorTaskId> tasks) {
762           workerAssignment.sort(WorkerLoad.taskComparator());
763           WorkerLoad first = workerAssignment.get(0);
764   
765           Iterator<ConnectorTaskId> load = tasks.iterator();
766           while (load.hasNext()) {
767               int firstLoad = first.tasksSize();
768               int upTo = IntStream.range(0, workerAssignment.size())
769                       .filter(i -> workerAssignment.get(i).tasksSize() > firstLoad)
770                       .findFirst()
771                       .orElse(workerAssignment.size());
772               for (WorkerLoad worker : workerAssignment.subList(0, upTo)) {
773                   ConnectorTaskId task = load.next();
774                   log.debug("Assigning task {} to {}", task, worker.worker());
775                   worker.assign(task);
776                   if (!load.hasNext()) {
777                       break;
778                   }
779               }
780           }
781       }
