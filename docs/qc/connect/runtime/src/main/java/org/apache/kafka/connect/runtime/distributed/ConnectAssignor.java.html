25   /**
26    * An assignor that computes a distribution of connectors and tasks among the workers of the group
27    * that performs rebalancing.
28    */
29   public interface ConnectAssignor {
30       /**
31        * Based on the member metadata and the information stored in the worker coordinator this
32        * method computes an assignment of connectors and tasks among the members of the worker group.
33        *
34        * @param leaderId the leader of the group
35        * @param protocol the protocol type; for Connect assignors this is "eager", "compatible", or "sessioned"
36        * @param allMemberMetadata the metadata of all the active workers of the group
37        * @param coordinator the worker coordinator that runs this assignor
38        * @return the assignment of connectors and tasks to workers
39        */
40       Map<String, ByteBuffer> performAssignment(String leaderId, String protocol,
41                                                 List<JoinGroupResponseData.JoinGroupResponseMember> allMemberMetadata,
42                                                 WorkerCoordinator coordinator);
43   }
