41   /**
42    * Handles processing for an individual task. This interface only provides the basic methods
43    * used by {@link Worker} to manage the tasks. Implementations combine a user-specified Task with
44    * Kafka to create a data flow.
45    *
46    * Note on locking: since the task runs in its own thread, special care must be taken to ensure
47    * that state transitions are reported correctly, in particular since some state transitions are
48    * asynchronous (e.g. pause/resume). For example, changing the state to paused could cause a race
49    * if the task fails at the same time. To protect from these cases, we synchronize status updates
50    * using the WorkerTask's monitor.
51    */
52   abstract class WorkerTask implements Runnable {
53       private static final Logger log = LoggerFactory.getLogger(WorkerTask.class);
54       private static final String THREAD_NAME_PREFIX = "task-thread-";
55

65       private volatile boolean cancelled;  // indicates whether the Worker has cancelled the task (e.g. because of slow shutdown)

106       private void triggerStop() {
107           synchronized (this) {
108               stopping = true;
109   
110               // wakeup any threads that are waiting for unpause
111               this.notifyAll();
112           }
113       }

115       /**
116        * Stop this task from processing messages. This method does not block, it only triggers
117        * shutdown. Use #{@link #awaitStop} to block until completion.
118        */
119       public void stop() {
120           triggerStop();
121       }
122   
123       /**
124        * Cancel this task. This won't actually stop it, but it will prevent the state from being
125        * updated when it eventually does shutdown.
126        */
127       public void cancel() {
128           cancelled = true;
129       }

131       /**
132        * Wait for this task to finish stopping.
133        *
134        * @param timeoutMs time in milliseconds to await stop
135        * @return true if successful, false if the timeout was reached
136        */
137       public boolean awaitStop(long timeoutMs) {
138           try {
139               return shutdownLatch.await(timeoutMs, TimeUnit.MILLISECONDS);
140           } catch (InterruptedException e) {
141               return false;
142           }
143       }

166       private void doClose() {
167           try {
168               close();
169           } catch (Throwable t) {
170               log.error("{} Task threw an uncaught and unrecoverable exception during shutdown", this, t);
171               throw t;
172           }
173       }
174   
175       private void doRun() throws InterruptedException {
176           try {
177               synchronized (this) {
178                   if (stopping)
179                       return;
180   
181                   if (targetState == TargetState.PAUSED) {
182                       onPause();
183                       if (!awaitUnpause()) return;
184                   }
185               }
186   
187               initializeAndStart();
188               statusListener.onStartup(id);
189               execute();
190           } catch (Throwable t) {
191               if (cancelled) {
192                   log.warn("{} After being scheduled for shutdown, the orphan task threw an uncaught exception. A newer instance of this task might be already running", this, t);
193               } else if (stopping) {
194                   log.warn("{} After being scheduled for shutdown, task threw an uncaught exception.", this, t);
195               } else {
196                   log.error("{} Task threw an uncaught and unrecoverable exception. Task is being killed and will not recover until manually restarted", this, t);
197                   throw t;
198               }
199           } finally {
200               doClose();
201           }
202       }

204       private void onShutdown() {
205           synchronized (this) {
206               triggerStop();
207   
208               // if we were cancelled, skip the status update since the task may have already been
209               // started somewhere else
210               if (!cancelled)
211                   statusListener.onShutdown(id);
212           }
213       }

216           synchronized (this) {
217               triggerStop();
218   
219               // if we were cancelled, skip the status update since the task may have already been
220               // started somewhere else
221               if (!cancelled)
222                   statusListener.onFailure(id, t);
223           }
224       }

234       @Override
235       public void run() {
236           // Clear all MDC parameters, in case this thread is being reused
237           LoggingContext.clear();
238   
239           try (LoggingContext loggingContext = LoggingContext.forTask(id())) {
240               ClassLoader savedLoader = Plugins.compareAndSwapLoaders(loader);
241               String savedName = Thread.currentThread().getName();
242               try {
243                   Thread.currentThread().setName(THREAD_NAME_PREFIX + id);
244                   doRun();
245                   onShutdown();
246               } catch (Throwable t) {
247                   onFailure(t);
248   
249                   if (t instanceof Error)
250                       throw (Error) t;
251               } finally {
252                   Thread.currentThread().setName(savedName);
253                   Plugins.compareAndSwapLoaders(savedLoader);
254                   shutdownLatch.countDown();
255               }
256           }
257       }

259       public boolean shouldPause() {

263       /**
264        * Await task resumption.
265        *
266        * @return true if the task's target state is not paused, false if the task is shutdown before resumption
267        * @throws InterruptedException
268        */
269       protected boolean awaitUnpause() throws InterruptedException {
270           synchronized (this) {
271               while (targetState == TargetState.PAUSED) {
272                   if (stopping)
273                       return false;
274                   this.wait();
275               }
276               return true;
277           }
278       }

337       static class TaskMetricsGroup implements TaskStatus.Listener {
338           private final TaskStatus.Listener delegateListener;
339           private final MetricGroup metricGroup;
340           private final Time time;
341           private final StateTracker taskStateTimer;
342           private final Sensor commitTime;
343           private final Sensor batchSize;
344           private final Sensor commitAttempts;
345   
346           public TaskMetricsGroup(ConnectorTaskId id, ConnectMetrics connectMetrics, TaskStatus.Listener statusListener) {
347               delegateListener = statusListener;
348               time = connectMetrics.time();
349               taskStateTimer = new StateTracker();
350               ConnectMetricsRegistry registry = connectMetrics.registry();
351               metricGroup = connectMetrics.group(registry.taskGroupName(),
352                       registry.connectorTagName(), id.connector(),
353                       registry.taskTagName(), Integer.toString(id.task()));
354               // prevent collisions by removing any previously created metrics in this group.
355               metricGroup.close();
356   
357               metricGroup.addValueMetric(registry.taskStatus, now ->
358                   taskStateTimer.currentState().toString().toLowerCase(Locale.getDefault())
359               );
360   
361               addRatioMetric(State.RUNNING, registry.taskRunningRatio);
362               addRatioMetric(State.PAUSED, registry.taskPauseRatio);
363   
364               commitTime = metricGroup.sensor("commit-time");
365               commitTime.add(metricGroup.metricName(registry.taskCommitTimeMax), new Max());
366               commitTime.add(metricGroup.metricName(registry.taskCommitTimeAvg), new Avg());
367   
368               batchSize = metricGroup.sensor("batch-size");
369               batchSize.add(metricGroup.metricName(registry.taskBatchSizeMax), new Max());
370               batchSize.add(metricGroup.metricName(registry.taskBatchSizeAvg), new Avg());
371   
372               MetricName offsetCommitFailures = metricGroup.metricName(registry.taskCommitFailurePercentage);
373               MetricName offsetCommitSucceeds = metricGroup.metricName(registry.taskCommitSuccessPercentage);
374               Frequencies commitFrequencies = Frequencies.forBooleanValues(offsetCommitFailures, offsetCommitSucceeds);
375               commitAttempts = metricGroup.sensor("offset-commit-completion");
376               commitAttempts.add(commitFrequencies);
377           }
378   
379           private void addRatioMetric(final State matchingState, MetricNameTemplate template) {
380               MetricName metricName = metricGroup.metricName(template);
381               metricGroup.metrics().addMetricIfAbsent(metricName, null, (Gauge<Double>) (config, now) ->
382                       taskStateTimer.durationRatio(matchingState, now));
383           }
