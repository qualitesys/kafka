270       /**
271        * Returns the initialized list of {@link Transformation} which are specified in {@link #TRANSFORMS_CONFIG}.
272        */
273       public <R extends ConnectRecord<R>> List<Transformation<R>> transformations() {
274           final List<String> transformAliases = getList(TRANSFORMS_CONFIG);
275   
276           final List<Transformation<R>> transformations = new ArrayList<>(transformAliases.size());
277           for (String alias : transformAliases) {
278               final String prefix = TRANSFORMS_CONFIG + "." + alias + ".";
279   
280               try {
281                   @SuppressWarnings("unchecked")
282                   final Transformation<R> transformation = Utils.newInstance(getClass(prefix + "type"), Transformation.class);
283                   Map<String, Object> configs = originalsWithPrefix(prefix);
284                   Object predicateAlias = configs.remove(PredicatedTransformation.PREDICATE_CONFIG);
285                   Object negate = configs.remove(PredicatedTransformation.NEGATE_CONFIG);
286                   transformation.configure(configs);
287                   if (predicateAlias != null) {
288                       String predicatePrefix = PREDICATES_PREFIX + predicateAlias + ".";
289                       @SuppressWarnings("unchecked")
290                       Predicate<R> predicate = Utils.newInstance(getClass(predicatePrefix + "type"), Predicate.class);
291                       predicate.configure(originalsWithPrefix(predicatePrefix));
292                       transformations.add(new PredicatedTransformation<>(predicate, negate == null ? false : Boolean.parseBoolean(negate.toString()), transformation));
293                   } else {
294                       transformations.add(transformation);
295                   }
296               } catch (Exception e) {
297                   throw new ConnectException(e);
298               }
299           }
300   
301           return transformations;
302       }

330               @Override

378       /**
379        * An abstraction over "enrichable plugins" ({@link Transformation}s and {@link Predicate}s) used for computing the
380        * contribution to a Connectors ConfigDef.
381        *
382        * This is not entirely elegant because
383        * although they basically use the same "alias prefix" configuration idiom there are some differences.
384        * The abstract method pattern is used to cope with this.
385        * @param <T> The type of plugin (either {@code Transformation} or {@code Predicate}).
386        */
387       static abstract class EnrichablePlugin<T> {
388   
389           private final String aliasKind;
390           private final String aliasConfig;
391           private final String aliasGroup;
392           private final Class<T> baseClass;
393           private final Map<String, String> props;
394           private final boolean requireFullConfig;
395   
396           public EnrichablePlugin(
397                   String aliasKind,
398                   String aliasConfig, String aliasGroup, Class<T> baseClass,
399                   Map<String, String> props, boolean requireFullConfig) {
400               this.aliasKind = aliasKind;
401               this.aliasConfig = aliasConfig;
402               this.aliasGroup = aliasGroup;
403               this.baseClass = baseClass;
404               this.props = props;
405               this.requireFullConfig = requireFullConfig;
406           }
407   
408           /** Add the configs for this alias to the given {@code ConfigDef}. */
409           void enrich(ConfigDef newDef) {
410               Object aliases = ConfigDef.parseType(aliasConfig, props.get(aliasConfig), Type.LIST);
411               if (!(aliases instanceof List)) {
412                   return;
413               }
414   
415               LinkedHashSet<?> uniqueAliases = new LinkedHashSet<>((List<?>) aliases);
416               for (Object o : uniqueAliases) {
417                   if (!(o instanceof String)) {
418                       throw new ConfigException("Item in " + aliasConfig + " property is not of "
419                               + "type String");
420                   }
421                   String alias = (String) o;
422                   final String prefix = aliasConfig + "." + alias + ".";
423                   final String group = aliasGroup + ": " + alias;
424                   int orderInGroup = 0;
425   
426                   final String typeConfig = prefix + "type";
427                   final ConfigDef.Validator typeValidator = ConfigDef.LambdaValidator.with(
428                       (String name, Object value) -> {
429                           validateProps(prefix);
430                           getConfigDefFromConfigProvidingClass(typeConfig, (Class<?>) value);
431                       },
432                       () -> "valid configs for " + alias + " " + aliasKind.toLowerCase(Locale.ENGLISH));
433                   newDef.define(typeConfig, Type.CLASS, ConfigDef.NO_DEFAULT_VALUE, typeValidator, Importance.HIGH,
434                           "Class for the '" + alias + "' " + aliasKind.toLowerCase(Locale.ENGLISH) + ".", group, orderInGroup++, Width.LONG,
435                           baseClass.getSimpleName() + " type for " + alias,
436                           Collections.emptyList(), new ClassRecommender());
437   
438                   final ConfigDef configDef = populateConfigDef(typeConfig);
439                   if (configDef == null) continue;
440                   newDef.embed(prefix, group, orderInGroup, configDef);
441               }
442           }

445           protected void validateProps(String prefix) { }

447           /**
448            * Populates the ConfigDef according to the configs returned from {@code configs()} method of class
449            * named in the {@code ...type} parameter of the {@code props}.
450            */
451           protected ConfigDef populateConfigDef(String typeConfig) {
452               final ConfigDef configDef = initialConfigDef();
453               try {
454                   configDefsForClass(typeConfig)
455                           .forEach(entry -> configDef.define(entry.getValue()));
456   
457               } catch (ConfigException e) {
458                   if (requireFullConfig) {
459                       throw e;
460                   } else {
461                       return null;
462                   }
463               }
464               return configDef;
465           }
466   
467           /**
468            * Return a stream of configs provided by the {@code configs()} method of class
469            * named in the {@code ...type} parameter of the {@code props}.
470            */
471           protected Stream<Map.Entry<String, ConfigDef.ConfigKey>> configDefsForClass(String typeConfig) {
472               final Class<?> cls = (Class<?>) ConfigDef.parseType(typeConfig, props.get(typeConfig), Type.CLASS);
473               return getConfigDefFromConfigProvidingClass(typeConfig, cls)
474                       .configKeys().entrySet().stream();
475           }

478           protected ConfigDef initialConfigDef() {
479               return new ConfigDef();
480           }

482           /**
483            * Return {@link ConfigDef} from {@code cls}, which is expected to be a non-null {@code Class<T>},
484            * by instantiating it and invoking {@link #config(T)}.
485            * @param key
486            * @param cls The subclass of the baseclass.
487            */
488           ConfigDef getConfigDefFromConfigProvidingClass(String key, Class<?> cls) {
489               if (cls == null || !baseClass.isAssignableFrom(cls)) {
490                   throw new ConfigException(key, String.valueOf(cls), "Not a " + baseClass.getSimpleName());
491               }
492               if (Modifier.isAbstract(cls.getModifiers())) {
493                   String childClassNames = Stream.of(cls.getClasses())
494                           .filter(cls::isAssignableFrom)
495                           .filter(c -> !Modifier.isAbstract(c.getModifiers()))
496                           .filter(c -> Modifier.isPublic(c.getModifiers()))
497                           .map(Class::getName)
498                           .collect(Collectors.joining(", "));
499                   String message = Utils.isBlank(childClassNames) ?
500                           aliasKind + " is abstract and cannot be created." :
501                           aliasKind + " is abstract and cannot be created. Did you mean " + childClassNames + "?";
502                   throw new ConfigException(key, String.valueOf(cls), message);
503               }
504               T transformation;
505               try {
506                   transformation = Utils.newInstance(cls, baseClass);
507               } catch (Exception e) {
508                   throw new ConfigException(key, String.valueOf(cls), "Error getting config definition from " + baseClass.getSimpleName() + ": " + e.getMessage());
509               }
510               ConfigDef configDef = config(transformation);
511               if (null == configDef) {
512                   throw new ConnectException(
513                       String.format(
514                           "%s.config() must return a ConfigDef that is not null.",
515                           cls.getName()
516                       )
517                   );
518               }
519               return configDef;
520           }

525            * {@link Predicate#config()}.
