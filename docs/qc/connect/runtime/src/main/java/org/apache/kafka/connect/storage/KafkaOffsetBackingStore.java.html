60   /**
61    * <p>
62    *     Implementation of OffsetBackingStore that uses a Kafka topic to store offset data.
63    * </p>
64    * <p>
65    *     Internally, this implementation both produces to and consumes from a Kafka topic which stores the offsets.
66    *     It accepts producer and consumer overrides via its configuration but forces some settings to specific values
67    *     to ensure correct behavior (e.g. acks, auto.offset.reset).
68    * </p>
69    */
70   public class KafkaOffsetBackingStore implements OffsetBackingStore {

167       @Override
168       public void configure(final WorkerConfig config) {
169           String topic = config.getString(DistributedConfig.OFFSET_STORAGE_TOPIC_CONFIG);
170           if (topic == null || topic.trim().length() == 0)
171               throw new ConfigException("Offset storage topic must be specified");
172   
173           this.exactlyOnce = config.exactlyOnceSourceEnabled();
174   
175           String clusterId = ConnectUtils.lookupKafkaClusterId(config);
176   
177           Map<String, Object> originals = config.originals();
178           Map<String, Object> producerProps = new HashMap<>(originals);
179           producerProps.put(ProducerConfig.KEY_SERIALIZER_CLASS_CONFIG, ByteArraySerializer.class.getName());
180           producerProps.put(ProducerConfig.VALUE_SERIALIZER_CLASS_CONFIG, ByteArraySerializer.class.getName());
181           producerProps.put(ProducerConfig.DELIVERY_TIMEOUT_MS_CONFIG, Integer.MAX_VALUE);
182           // By default, Connect disables idempotent behavior for all producers, even though idempotence became
183           // default for Kafka producers. This is to ensure Connect continues to work with many Kafka broker versions, including older brokers that do not support
184           // idempotent producers or require explicit steps to enable them (e.g. adding the IDEMPOTENT_WRITE ACL to brokers older than 2.8).
185           // These settings might change when https://cwiki.apache.org/confluence/display/KAFKA/KIP-318%3A+Make+Kafka+Connect+Source+idempotent
186           // gets approved and scheduled for release.
187           producerProps.put(ProducerConfig.ENABLE_IDEMPOTENCE_CONFIG, "false");
188           ConnectUtils.addMetricsContextProperties(producerProps, config, clusterId);
189   
190           Map<String, Object> consumerProps = new HashMap<>(originals);
191           consumerProps.put(ConsumerConfig.KEY_DESERIALIZER_CLASS_CONFIG, ByteArrayDeserializer.class.getName());
192           consumerProps.put(ConsumerConfig.VALUE_DESERIALIZER_CLASS_CONFIG, ByteArrayDeserializer.class.getName());
193           ConnectUtils.addMetricsContextProperties(consumerProps, config, clusterId);
194           if (config.exactlyOnceSourceEnabled()) {
195               ConnectUtils.ensureProperty(
196                       consumerProps, ConsumerConfig.ISOLATION_LEVEL_CONFIG, IsolationLevel.READ_COMMITTED.name().toLowerCase(Locale.ROOT),
197                       "for the worker offsets topic consumer when exactly-once source support is enabled",
198                       false
199               );
200           }
201   
202           Map<String, Object> adminProps = new HashMap<>(originals);
203           ConnectUtils.addMetricsContextProperties(adminProps, config, clusterId);
204           Supplier<TopicAdmin> adminSupplier;
205           if (topicAdminSupplier != null) {
206               adminSupplier = topicAdminSupplier;
207           } else {
208               // Create our own topic admin supplier that we'll close when we're stopped
209               this.ownTopicAdmin = new SharedTopicAdmin(adminProps);
210               adminSupplier = ownTopicAdmin;
211           }
212           NewTopic topicDescription = newTopicDescription(topic, config);
213   
214           this.offsetLog = createKafkaBasedLog(topic, producerProps, consumerProps, consumedCallback, topicDescription, adminSupplier);
215       }

217       private KafkaBasedLog<byte[], byte[]> createKafkaBasedLog(String topic, Map<String, Object> producerProps,
218                                                                 Map<String, Object> consumerProps,
219                                                                 Callback<ConsumerRecord<byte[], byte[]>> consumedCallback,
220                                                                 final NewTopic topicDescription, Supplier<TopicAdmin> adminSupplier) {
221           java.util.function.Consumer<TopicAdmin> createTopics = initialize(topic, topicDescription);
222           return new KafkaBasedLog<>(topic, producerProps, consumerProps, adminSupplier, consumedCallback, Time.SYSTEM, createTopics);
223       }

225       protected NewTopic newTopicDescription(final String topic, final WorkerConfig config) {
226           Map<String, Object> topicSettings = config instanceof DistributedConfig
227                   ? ((DistributedConfig) config).offsetStorageTopicSettings()
228                   : Collections.emptyMap();
229           return TopicAdmin.defineTopic(topic)
230                   .config(topicSettings) // first so that we override user-supplied settings as needed
231                   .compacted()
232                   .partitions(config.getInt(DistributedConfig.OFFSET_STORAGE_PARTITIONS_CONFIG))
233                   .replicationFactor(config.getShort(DistributedConfig.OFFSET_STORAGE_REPLICATION_FACTOR_CONFIG))
234                   .build();
235       }

237       protected java.util.function.Consumer<TopicAdmin> initialize(final String topic, final NewTopic topicDescription) {
238           return admin -> {
239               log.debug("Creating admin client to manage Connect internal offset topic");
240               // Create the topic if it doesn't exist
241               Set<String> newTopics = admin.createTopics(topicDescription);
242               if (!newTopics.contains(topic)) {
243                   // It already existed, so check that the topic cleanup policy is compact only and not delete
244                   log.debug("Using admin client to check cleanup policy for '{}' topic is '{}'", topic, TopicConfig.CLEANUP_POLICY_COMPACT);
245                   admin.verifyTopicCleanupPolicyOnlyCompact(topic,
246                           DistributedConfig.OFFSET_STORAGE_TOPIC_CONFIG, "source connector offsets");
247               }
248           };
249       }

251       @Override
252       public void start() {
253           log.info("Starting KafkaOffsetBackingStore");
254           try {
255               offsetLog.start();
256           } catch (UnsupportedVersionException e) {
257               String message;
258               if (exactlyOnce) {
259                   message = "Enabling exactly-once support for source connectors requires a Kafka broker version that allows "
260                           + "admin clients to read consumer offsets. Please either disable the worker's exactly-once "
261                           + "support for source connectors, or upgrade to a newer Kafka broker version.";
262               } else {
263                   message = "When " + ConsumerConfig.ISOLATION_LEVEL_CONFIG + "is set to "
264                           + IsolationLevel.READ_COMMITTED.name().toLowerCase(Locale.ROOT)
265                           + ", a Kafka broker version that allows admin clients to read consumer offsets is required. "
266                           + "Please either reconfigure the worker or connector, or upgrade to a newer Kafka broker version.";
267               }
268               throw new ConnectException(message, e);
269           }
270           log.info("Finished reading offsets topic and starting KafkaOffsetBackingStore");
271       }

273       /**
274        * Stop reading from and writing to the offsets topic, and relinquish resources allocated for interacting
275        * with it, including Kafka clients.
276        * <p>
277        * <b>Note:</b> if the now-deprecated {@link #KafkaOffsetBackingStore()} constructor was used to create
278        * this store, the underlying admin client allocated for interacting with the offsets topic will be closed.
279        * On the other hand, if the recommended {@link #KafkaOffsetBackingStore(Supplier)} constructor was used to
280        * create this store, the admin client derived from the given {@link Supplier} will not be closed and it is the
281        * caller's responsibility to manage its lifecycle accordingly.
282        */

284       public void stop() {
285           log.info("Stopping KafkaOffsetBackingStore");
286           try {
287               offsetLog.stop();
288           } finally {
289               if (ownTopicAdmin != null) {
290                   ownTopicAdmin.close();
291               }
292           }
293           log.info("Stopped KafkaOffsetBackingStore");
294       }

296       @Override
297       public Future<Map<ByteBuffer, ByteBuffer>> get(final Collection<ByteBuffer> keys) {
298           ConvertingFutureCallback<Void, Map<ByteBuffer, ByteBuffer>> future = new ConvertingFutureCallback<Void, Map<ByteBuffer, ByteBuffer>>() {
299               @Override
300               public Map<ByteBuffer, ByteBuffer> convert(Void result) {
301                   Map<ByteBuffer, ByteBuffer> values = new HashMap<>();
302                   for (ByteBuffer key : keys)
303                       values.put(key, data.get(key));
304                   return values;
305               }
306           };
307           // This operation may be relatively (but not too) expensive since it always requires checking end offsets, even
308           // if we've already read up to the end. However, it also should not be common (offsets should only be read when
309           // resetting a task). Always requiring that we read to the end is simpler than trying to differentiate when it
310           // is safe not to (which should only be if we *know* we've maintained ownership since the last write).
311           offsetLog.readToEnd(future);
312           return future;
313       }

315       @Override
316       public Future<Void> set(final Map<ByteBuffer, ByteBuffer> values, final Callback<Void> callback) {
317           SetCallbackFuture producerCallback = new SetCallbackFuture(values.size(), callback);
318   
319           for (Map.Entry<ByteBuffer, ByteBuffer> entry : values.entrySet()) {
320               ByteBuffer key = entry.getKey();
321               ByteBuffer value = entry.getValue();
322               offsetLog.send(key == null ? null : key.array(), value == null ? null : value.array(), producerCallback);
323           }
324   
325           return producerCallback;
326       }

328       protected final Callback<ConsumerRecord<byte[], byte[]>> consumedCallback = new Callback<ConsumerRecord<byte[], byte[]>>() {
329           @Override
330           public void onCompletion(Throwable error, ConsumerRecord<byte[], byte[]> record) {
331               ByteBuffer key = record.key() != null ? ByteBuffer.wrap(record.key()) : null;
332               ByteBuffer value = record.value() != null ? ByteBuffer.wrap(record.value()) : null;
333               data.put(key, value);
334           }

337       private static class SetCallbackFuture implements org.apache.kafka.clients.producer.Callback, Future<Void> {
338           private int numLeft;
339           private boolean completed = false;
340           private Throwable exception = null;
341           private final Callback<Void> callback;
342   
343           public SetCallbackFuture(int numRecords, Callback<Void> callback) {
344               numLeft = numRecords;
345               this.callback = callback;
346           }
347   
348           @Override
349           public synchronized void onCompletion(RecordMetadata metadata, Exception exception) {
350               if (exception != null) {
351                   if (!completed) {
352                       this.exception = exception;
353                       callback.onCompletion(exception, null);
354                       completed = true;
355                       this.notify();
356                   }
357                   return;
358               }
359   
360               numLeft -= 1;
361               if (numLeft == 0) {
362                   callback.onCompletion(null, null);
363                   completed = true;
364                   this.notify();
365               }
366           }
367   
368           @Override
369           public synchronized boolean cancel(boolean mayInterruptIfRunning) {
370               return false;
371           }
372   
373           @Override
374           public synchronized boolean isCancelled() {
375               return false;
376           }
377   
378           @Override
379           public synchronized boolean isDone() {
380               return completed;
381           }
382   
383           @Override
384           public synchronized Void get() throws InterruptedException, ExecutionException {
385               while (!completed) {
386                   this.wait();
387               }
388               if (exception != null)
389                   throw new ExecutionException(exception);
390               return null;
391           }
392   
393           @Override
394           public synchronized Void get(long timeout, TimeUnit unit) throws InterruptedException, ExecutionException, TimeoutException {
395               long started = System.currentTimeMillis();
396               long limit = started + unit.toMillis(timeout);
397               while (!completed) {
398                   long leftMs = limit - System.currentTimeMillis();
399                   if (leftMs < 0)
400                       throw new TimeoutException("KafkaOffsetBackingStore Future timed out.");
401                   this.wait(leftMs);
402               }
403               if (exception != null)
404                   throw new ExecutionException(exception);
405               return null;
406           }
407       }
408   }
