33   /**
34    * Used to track source records that have been (or are about to be) dispatched to a producer and their accompanying
35    * source offsets. Records are tracked in the order in which they are submitted, which should match the order they were
36    * returned from {@link SourceTask#poll()}. The latest-eligible offsets for each source partition can be retrieved via
37    * {@link #committableOffsets()}, where every record up to and including the record for each returned offset has been
38    * either {@link SubmittedRecord#ack() acknowledged} or {@link SubmittedRecord#drop dropped}.
39    * Note that this class is not thread-safe, though a {@link SubmittedRecord} can be
40    * {@link SubmittedRecord#ack() acknowledged} from a different thread.
41    */
42   class SubmittedRecords {
43   
44       private static final Logger log = LoggerFactory.getLogger(SubmittedRecords.class);
45   
46       // Visible for testing
47       final Map<Map<String, Object>, Deque<SubmittedRecord>> records = new HashMap<>();
48       private int numUnackedMessages = 0;
49       private CountDownLatch messageDrainLatch;
50   
51       public SubmittedRecords() {
52       }

54       /**
55        * Enqueue a new source record before dispatching it to a producer.
56        * The returned {@link SubmittedRecord} should either be {@link SubmittedRecord#ack() acknowledged} in the
57        * producer callback, or {@link SubmittedRecord#drop() dropped} if the record could not be successfully
58        * sent to the producer.
59        *
60        * @param record the record about to be dispatched; may not be null but may have a null
61        *               {@link SourceRecord#sourcePartition()} and/or {@link SourceRecord#sourceOffset()}
62        * @return a {@link SubmittedRecord} that can be either {@link SubmittedRecord#ack() acknowledged} once ack'd by
63        *         the producer, or {@link SubmittedRecord#drop() dropped} if synchronously rejected by the producer
64        */
65       @SuppressWarnings("unchecked")
66       public SubmittedRecord submit(SourceRecord record) {
67           return submit((Map<String, Object>) record.sourcePartition(), (Map<String, Object>) record.sourceOffset());
68       }
69   
70       // Convenience method for testing
71       SubmittedRecord submit(Map<String, Object> partition, Map<String, Object> offset) {
72           SubmittedRecord result = new SubmittedRecord(partition, offset);
73           records.computeIfAbsent(result.partition(), p -> new LinkedList<>())
74                   .add(result);
75           synchronized (this) {
76               numUnackedMessages++;
77           }
78           return result;
79       }

81       /**
82        * Clear out any acknowledged records at the head of the deques and return a {@link CommittableOffsets snapshot} of the offsets and offset metadata
83        * accrued between the last time this method was invoked and now. This snapshot can be {@link CommittableOffsets#updatedWith(CommittableOffsets) combined}
84        * with an existing snapshot if desired.
85        * Note that this may take some time to complete if a large number of records has built up, which may occur if a
86        * Kafka partition is offline and all records targeting that partition go unacknowledged while records targeting
87        * other partitions continue to be dispatched to the producer and sent successfully
88        * @return a fresh offset snapshot; never null
89        */
90       public CommittableOffsets committableOffsets() {
91           Map<Map<String, Object>, Map<String, Object>> offsets = new HashMap<>();
92           int totalCommittableMessages = 0;
93           int totalUncommittableMessages = 0;
94           int largestDequeSize = 0;
95           Map<String, Object> largestDequePartition = null;
96           for (Map.Entry<Map<String, Object>, Deque<SubmittedRecord>> entry : records.entrySet()) {
97               Map<String, Object> partition = entry.getKey();
98               Deque<SubmittedRecord> queuedRecords = entry.getValue();
99               int initialDequeSize = queuedRecords.size();
100               if (canCommitHead(queuedRecords)) {
101                   Map<String, Object> offset = committableOffset(queuedRecords);
102                   offsets.put(partition, offset);
103               }
104               int uncommittableMessages = queuedRecords.size();
105               int committableMessages = initialDequeSize - uncommittableMessages;
106               totalCommittableMessages += committableMessages;
107               totalUncommittableMessages += uncommittableMessages;
108               if (uncommittableMessages > largestDequeSize) {
109                   largestDequeSize = uncommittableMessages;
110                   largestDequePartition = partition;
111               }
112           }
113           // Clear out all empty deques from the map to keep it from growing indefinitely
114           records.values().removeIf(Deque::isEmpty);
115           return new CommittableOffsets(offsets, totalCommittableMessages, totalUncommittableMessages, records.size(), largestDequeSize, largestDequePartition);
116       }
117   
118       /**
119        * Wait for all currently in-flight messages to be acknowledged, up to the requested timeout.
120        * This method is expected to be called from the same thread that calls {@link #committableOffsets()}.
121        * @param timeout the maximum time to wait
122        * @param timeUnit the time unit of the timeout argument
123        * @return whether all in-flight messages were acknowledged before the timeout elapsed
124        */
125       public boolean awaitAllMessages(long timeout, TimeUnit timeUnit) {
126           // Create a new message drain latch as a local variable to avoid SpotBugs warnings about inconsistent synchronization
127           // on an instance variable when invoking CountDownLatch::await outside a synchronized block
128           CountDownLatch messageDrainLatch;
129           synchronized (this) {
130               messageDrainLatch = new CountDownLatch(numUnackedMessages);
131               this.messageDrainLatch = messageDrainLatch;
132           }
133           try {
134               return messageDrainLatch.await(timeout, timeUnit);
135           } catch (InterruptedException e) {
136               return false;
137           }
138       }
139   
140       // Note that this will return null if either there are no committable offsets for the given deque, or the latest
141       // committable offset is itself null. The caller is responsible for distinguishing between the two cases.
142       private Map<String, Object> committableOffset(Deque<SubmittedRecord> queuedRecords) {
143           Map<String, Object> result = null;
144           while (canCommitHead(queuedRecords)) {
145               result = queuedRecords.poll().offset();
146           }
147           return result;
148       }
149   
150       private boolean canCommitHead(Deque<SubmittedRecord> queuedRecords) {
151           return queuedRecords.peek() != null && queuedRecords.peek().acked();
152       }
153   
154       // Synchronize in order to ensure that the number of unacknowledged messages isn't modified in the middle of a call
155       // to awaitAllMessages (which might cause us to decrement first, then create a new message drain latch, then count down
156       // that latch here, effectively double-acking the message)
157       private synchronized void messageAcked() {
158           numUnackedMessages--;
159           if (messageDrainLatch != null) {
160               messageDrainLatch.countDown();
161           }
162       }

164       public class SubmittedRecord {
165           private final Map<String, Object> partition;
166           private final Map<String, Object> offset;
167           private final AtomicBoolean acked;
168   
169           public SubmittedRecord(Map<String, Object> partition, Map<String, Object> offset) {
170               this.partition = partition;
171               this.offset = offset;
172               this.acked = new AtomicBoolean(false);
173           }
174   
175           /**
176            * Acknowledge this record; signals that its offset may be safely committed.
177            * This is safe to be called from a different thread than what called {@link SubmittedRecords#submit(SourceRecord)}.
178            */
179           public void ack() {
180               if (this.acked.compareAndSet(false, true)) {
181                   messageAcked();
182               }
183           }
184   
185           /**
186            * Remove this record and do not take it into account any longer when tracking offsets.
187            * Useful if the record has been synchronously rejected by the producer.
188            * If multiple instances of this record have been submitted already, only the first one found
189            * (traversing from the end of the deque backward) will be removed.
190            * <p>
191            * This is <strong>not safe</strong> to be called from a different thread
192            * than what called {@link SubmittedRecords#submit(SourceRecord)}.
193            * @return whether this instance was dropped
194            */
195           public boolean drop() {
196               Deque<SubmittedRecord> deque = records.get(partition);
197               if (deque == null) {
198                   log.warn("Attempted to remove record from submitted queue for partition {}, but no records with that partition appear to have been submitted", partition);
199                   return false;
200               }
201               boolean result = deque.removeLastOccurrence(this);
202               if (deque.isEmpty()) {
203                   records.remove(partition);
204               }
205               if (result) {
206                   messageAcked();
207               } else {
208                   log.warn("Attempted to remove record from submitted queue for partition {}, but the record has not been submitted or has already been removed", partition);
209               }
210               return result;
211           }
212   
213           private boolean acked() {
214               return acked.get();
215           }
216   
217           private Map<String, Object> partition() {
218               return partition;
219           }
220   
221           private Map<String, Object> offset() {
222               return offset;
223           }
224       }

226       /**
227        * Contains a snapshot of offsets that can be committed for a source task and metadata for that offset commit
228        * (such as the number of messages for which offsets can and cannot be committed).
229        */
230       static class CommittableOffsets {
231   
232           /**
233            * An "empty" snapshot that contains no offsets to commit and whose metadata contains no committable or uncommitable messages.
234            */
235           public static final CommittableOffsets EMPTY = new CommittableOffsets(Collections.emptyMap(), 0, 0, 0, 0, null);
236   
237           private final Map<Map<String, Object>, Map<String, Object>> offsets;
238           private final int numCommittableMessages;
239           private final int numUncommittableMessages;
240           private final int numDeques;
241           private final int largestDequeSize;
242           private final Map<String, Object> largestDequePartition;
243   
244           CommittableOffsets(
245                   Map<Map<String, Object>, Map<String, Object>> offsets,
246                   int numCommittableMessages,
247                   int numUncommittableMessages,
248                   int numDeques,
249                   int largestDequeSize,
250                   Map<String, Object> largestDequePartition
251           ) {
252               this.offsets = offsets != null ? new HashMap<>(offsets) : Collections.emptyMap();
253               this.numCommittableMessages = numCommittableMessages;
254               this.numUncommittableMessages = numUncommittableMessages;
255               this.numDeques = numDeques;
256               this.largestDequeSize = largestDequeSize;
257               this.largestDequePartition = largestDequePartition;
258           }

260           /**
261            * @return the offsets that can be committed at the time of the snapshot
262            */
263           public Map<Map<String, Object>, Map<String, Object>> offsets() {
264               return Collections.unmodifiableMap(offsets);
265           }

267           /**
268            * @return the number of committable messages at the time of the snapshot, where a committable message is both
269            * acknowledged and not preceded by any unacknowledged messages in the deque for its source partition
270            */
271           public int numCommittableMessages() {
272               return numCommittableMessages;
273           }
274   
275           /**
276            * @return the number of uncommittable messages at the time of the snapshot, where an uncommittable message
277            * is either unacknowledged, or preceded in the deque for its source partition by an unacknowledged message
278            */
279           public int numUncommittableMessages() {
280               return numUncommittableMessages;
281           }
282   
283           /**
284            * @return the number of non-empty deques tracking uncommittable messages at the time of the snapshot
285            */
286           public int numDeques() {
287               return numDeques;
288           }
289   
290           /**
291            * @return the size of the largest deque at the time of the snapshot
292            */
293           public int largestDequeSize() {
294               return largestDequeSize;
295           }
296   
297           /**
298            * Get the partition for the deque with the most uncommitted messages at the time of the snapshot.
299            * @return the applicable partition, which may be null, or null if there are no uncommitted messages;
300            * it is the caller's responsibility to distinguish between these two cases via {@link #hasPending()}
301            */
302           public Map<String, Object> largestDequePartition() {
303               return largestDequePartition;
304           }
305   
306           /**
307            * @return whether there were any uncommittable messages at the time of the snapshot
308            */
309           public boolean hasPending() {
310               return numUncommittableMessages > 0;
311           }
312   
313           /**
314            * @return whether there were any committable or uncommittable messages at the time of the snapshot
315            */
316           public boolean isEmpty() {
317               return numCommittableMessages == 0 && numUncommittableMessages == 0 && offsets.isEmpty();
318           }
319   
320           /**
321            * Create a new snapshot by combining the data for this snapshot with newer data in a more recent snapshot.
322            * Offsets are combined (giving precedence to the newer snapshot in case of conflict), the total number of
323            * committable messages is summed across the two snapshots, and the newer snapshot's information on pending
324            * messages (num deques, largest deque size, etc.) is used.
325            * @param newerOffsets the newer snapshot to combine with this snapshot
326            * @return the new offset snapshot containing information from this snapshot and the newer snapshot; never null
327            */
328           public CommittableOffsets updatedWith(CommittableOffsets newerOffsets) {
329               Map<Map<String, Object>, Map<String, Object>> offsets = new HashMap<>(this.offsets);
330               offsets.putAll(newerOffsets.offsets);
331   
332               return new CommittableOffsets(
333                       offsets,
334                       this.numCommittableMessages + newerOffsets.numCommittableMessages,
335                       newerOffsets.numUncommittableMessages,
336                       newerOffsets.numDeques,
337                       newerOffsets.largestDequeSize,
338                       newerOffsets.largestDequePartition
339               );
340           }
341       }
