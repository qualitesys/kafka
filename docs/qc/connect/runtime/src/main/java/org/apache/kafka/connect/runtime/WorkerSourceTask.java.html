149       @Override
150       protected void producerSendFailed(
151               boolean synchronous,
152               ProducerRecord<byte[], byte[]> producerRecord,
153               SourceRecord preTransformRecord,
154               Exception e
155       ) {
156           if (synchronous) {
157               throw new ConnectException("Unrecoverable exception trying to send", e);
158           }
159   
160           String topic = producerRecord.topic();
161           if (retryWithToleranceOperator.getErrorToleranceType() == ToleranceType.ALL) {
162               log.trace(
163                       "Ignoring failed record send: {} failed to send record to {}: ",
164                       WorkerSourceTask.this,
165                       topic,
166                       e
167               );
168               // executeFailed here allows the use of existing logging infrastructure/configuration
169               retryWithToleranceOperator.executeFailed(
170                       Stage.KAFKA_PRODUCE,
171                       WorkerSourceTask.class,
172                       preTransformRecord,
173                       e
174               );
175               commitTaskRecord(preTransformRecord, null);
176           } else {
177               producerSendException.compareAndSet(null, e);
178           }
179       }

181       @Override
182       protected void finalOffsetCommit(boolean failed) {
183           // It should still be safe to commit offsets since any exception would have
184           // simply resulted in not getting more records but all the existing records should be ok to flush
185           // and commit offsets. Worst case, task.commit() will also throw an exception causing the offset
186           // commit to fail.
187           submittedRecords.awaitAllMessages(
188                   workerConfig.getLong(WorkerConfig.OFFSET_COMMIT_TIMEOUT_MS_CONFIG),
189                   TimeUnit.MILLISECONDS
190           );
191           updateCommittableOffsets();
192           commitOffsets();
193       }

195       public boolean commitOffsets() {
196           long commitTimeoutMs = workerConfig.getLong(WorkerConfig.OFFSET_COMMIT_TIMEOUT_MS_CONFIG);
197   
198           log.debug("{} Committing offsets", this);
199   
200           long started = time.milliseconds();
201           long timeout = started + commitTimeoutMs;
202   
203           CommittableOffsets offsetsToCommit;
204           synchronized (this) {
205               offsetsToCommit = this.committableOffsets;
206               this.committableOffsets = CommittableOffsets.EMPTY;
207           }
208   
209           if (committableOffsets.isEmpty()) {
210               log.debug("{} Either no records were produced by the task since the last offset commit, " 
211                       + "or every record has been filtered out by a transformation " 
212                       + "or dropped due to transformation or conversion errors.",
213                       this
214               );
215               // We continue with the offset commit process here instead of simply returning immediately
216               // in order to invoke SourceTask::commit and record metrics for a successful offset commit
217           } else {
218               log.info("{} Committing offsets for {} acknowledged messages", this, committableOffsets.numCommittableMessages());
219               if (committableOffsets.hasPending()) {
220                   log.debug("{} There are currently {} pending messages spread across {} source partitions whose offsets will not be committed. "
221                                   + "The source partition with the most pending messages is {}, with {} pending messages",
222                           this,
223                           committableOffsets.numUncommittableMessages(),
224                           committableOffsets.numDeques(),
225                           committableOffsets.largestDequePartition(),
226                           committableOffsets.largestDequeSize()
227                   );
228               } else {
229                   log.debug("{} There are currently no pending messages for this offset commit; "
230                                   + "all messages dispatched to the task's producer since the last commit have been acknowledged",
231                           this
232                   );
233               }
234           }
235   
236           // Update the offset writer with any new offsets for records that have been acked.
237           // The offset writer will continue to track all offsets until they are able to be successfully flushed.
238           // IOW, if the offset writer fails to flush, it keeps those offset for the next attempt,
239           // though we may update them here with newer offsets for acked records.
240           offsetsToCommit.offsets().forEach(offsetWriter::offset);
241   
242           if (!offsetWriter.beginFlush()) {
243               // There was nothing in the offsets to process, but we still mark a successful offset commit.
244               long durationMillis = time.milliseconds() - started;
245               recordCommitSuccess(durationMillis);
246               log.debug("{} Finished offset commitOffsets successfully in {} ms",
247                       this, durationMillis);
248   
249               commitSourceTask();
250               return true;
251           }
252   
253           // Now we can actually flush the offsets to user storage.
254           Future<Void> flushFuture = offsetWriter.doFlush((error, result) -> {
255               if (error != null) {
256                   log.error("{} Failed to flush offsets to storage: ", WorkerSourceTask.this, error);
257               } else {
258                   log.trace("{} Finished flushing offsets to storage", WorkerSourceTask.this);
259               }
260           });
261           // Very rare case: offsets were unserializable and we finished immediately, unable to store
262           // any data
263           if (flushFuture == null) {
264               offsetWriter.cancelFlush();
265               recordCommitFailure(time.milliseconds() - started, null);
266               return false;
267           }
268           try {
269               flushFuture.get(Math.max(timeout - time.milliseconds(), 0), TimeUnit.MILLISECONDS);
270               // There's a small race here where we can get the callback just as this times out (and log
271               // success), but then catch the exception below and cancel everything. This won't cause any
272               // errors, is only wasteful in this minor edge case, and the worst result is that the log
273               // could look a little confusing.
274           } catch (InterruptedException e) {
275               log.warn("{} Flush of offsets interrupted, cancelling", this);
276               offsetWriter.cancelFlush();
277               recordCommitFailure(time.milliseconds() - started, e);
278               return false;
279           } catch (ExecutionException e) {
280               log.error("{} Flush of offsets threw an unexpected exception: ", this, e);
281               offsetWriter.cancelFlush();
282               recordCommitFailure(time.milliseconds() - started, e);
283               return false;
284           } catch (TimeoutException e) {
285               log.error("{} Timed out waiting to flush offsets to storage; will try again on next flush interval with latest offsets", this);
286               offsetWriter.cancelFlush();
287               recordCommitFailure(time.milliseconds() - started, null);
288               return false;
289           }
290   
291           long durationMillis = time.milliseconds() - started;
292           recordCommitSuccess(durationMillis);
293           log.debug("{} Finished commitOffsets successfully in {} ms",
294                   this, durationMillis);
295   
296           commitSourceTask();
297   
298           return true;
299       }

301       private void updateCommittableOffsets() {
302           CommittableOffsets newOffsets = submittedRecords.committableOffsets();
303           synchronized (this) {
304               this.committableOffsets = this.committableOffsets.updatedWith(newOffsets);
305           }
306       }

308       private void maybeThrowProducerSendException() {
309           if (producerSendException.get() != null) {
310               throw new ConnectException(
311                       "Unrecoverable exception from producer send callback",
312                       producerSendException.get()
313               );
314           }
315       }
