112        * Adds Jetty connector for each configured listener
113        */
114       public void createConnectors(List<String> listeners, List<String> adminListeners) {
115           List<Connector> connectors = new ArrayList<>();
116   
117           for (String listener : listeners) {
118               Connector connector = createConnector(listener);
119               connectors.add(connector);
120               log.info("Added connector for {}", listener);
121           }
122

124   
125           if (adminListeners != null && !adminListeners.isEmpty()) {
126               for (String adminListener : adminListeners) {
127                   Connector conn = createConnector(adminListener, true);
128                   jettyServer.addConnector(conn);
129                   log.info("Added admin connector for {}", adminListener);
130               }
131           }
132       }

141       /**
142        * Creates Jetty connector according to configuration
143        */
144       public Connector createConnector(String listener, boolean isAdmin) {
145           Matcher listenerMatcher = LISTENER_PATTERN.matcher(listener);
146   
147           if (!listenerMatcher.matches())
148               throw new ConfigException("Listener doesn't have the right format (protocol://hostname:port).");
149   
150           String protocol = listenerMatcher.group(1).toLowerCase(Locale.ENGLISH);
151   
152           if (!PROTOCOL_HTTP.equals(protocol) && !PROTOCOL_HTTPS.equals(protocol))
153               throw new ConfigException(String.format("Listener protocol must be either \"%s\" or \"%s\".", PROTOCOL_HTTP, PROTOCOL_HTTPS));
154   
155           String hostname = listenerMatcher.group(2);
156           int port = Integer.parseInt(listenerMatcher.group(3));
157   
158           ServerConnector connector;
159   
160           if (PROTOCOL_HTTPS.equals(protocol)) {
161               SslContextFactory ssl;
162               if (isAdmin) {
163                   ssl = SSLUtils.createServerSideSslContextFactory(config, ADMIN_LISTENERS_HTTPS_CONFIGS_PREFIX);
164               } else {
165                   ssl = SSLUtils.createServerSideSslContextFactory(config);
166               }
167               connector = new ServerConnector(jettyServer, ssl);
168               if (!isAdmin) {
169                   connector.setName(String.format("%s_%s%d", PROTOCOL_HTTPS, hostname, port));
170               }
171           } else {
172               connector = new ServerConnector(jettyServer);
173               if (!isAdmin) {
174                   connector.setName(String.format("%s_%s%d", PROTOCOL_HTTP, hostname, port));
175               }
176           }
177   
178           if (isAdmin) {
179               connector.setName(ADMIN_SERVER_CONNECTOR_NAME);
180           }
181   
182           if (!hostname.isEmpty())
183               connector.setHost(hostname);
184   
185           connector.setPort(port);
186   
187           return connector;
188       }

210       public void initializeResources(Herder herder) {
211           log.info("Initializing REST resources");
212   
213           ResourceConfig resourceConfig = new ResourceConfig();
214           resourceConfig.register(new JacksonJsonProvider());
215   
216           this.resources = new ArrayList<>();
217           resources.add(new RootResource(herder));
218           resources.add(new ConnectorsResource(herder, config));
219           resources.add(new ConnectorPluginsResource(herder));
220           resources.forEach(resourceConfig::register);
221   
222           resourceConfig.register(ConnectExceptionMapper.class);
223           resourceConfig.property(ServerProperties.WADL_FEATURE_DISABLE, true);
224   
225           registerRestExtensions(herder, resourceConfig);
226   
227           List<String> adminListeners = config.getList(WorkerConfig.ADMIN_LISTENERS_CONFIG);
228           ResourceConfig adminResourceConfig;
229           if (adminListeners == null) {
230               log.info("Adding admin resources to main listener");
231               adminResourceConfig = resourceConfig;
232               LoggingResource loggingResource = new LoggingResource();
233               this.resources.add(loggingResource);
234               adminResourceConfig.register(loggingResource);
235           } else if (adminListeners.size() > 0) {
236               // TODO: we need to check if these listeners are same as 'listeners'
237               // TODO: the following code assumes that they are different
238               log.info("Adding admin resources to admin listener");
239               adminResourceConfig = new ResourceConfig();
240               adminResourceConfig.register(new JacksonJsonProvider());
241               LoggingResource loggingResource = new LoggingResource();
242               this.resources.add(loggingResource);
243               adminResourceConfig.register(loggingResource);
244               adminResourceConfig.register(ConnectExceptionMapper.class);
245           } else {
246               log.info("Skipping adding admin resources");
247               // set up adminResource but add no handlers to it
248               adminResourceConfig = resourceConfig;
249           }
250   
251           ServletContainer servletContainer = new ServletContainer(resourceConfig);
252           ServletHolder servletHolder = new ServletHolder(servletContainer);
253           List<Handler> contextHandlers = new ArrayList<>();
254   
255           ServletContextHandler context = new ServletContextHandler(ServletContextHandler.SESSIONS);
256           context.setContextPath("/");
257           context.addServlet(servletHolder, "/*");
258           contextHandlers.add(context);
259   
260           ServletContextHandler adminContext = null;
261           if (adminResourceConfig != resourceConfig) {
262               adminContext = new ServletContextHandler(ServletContextHandler.SESSIONS);
263               ServletHolder adminServletHolder = new ServletHolder(new ServletContainer(adminResourceConfig));
264               adminContext.setContextPath("/");
265               adminContext.addServlet(adminServletHolder, "/*");
266               adminContext.setVirtualHosts(new String[]{"@" + ADMIN_SERVER_CONNECTOR_NAME});
267               contextHandlers.add(adminContext);
268           }
269   
270           String allowedOrigins = config.getString(WorkerConfig.ACCESS_CONTROL_ALLOW_ORIGIN_CONFIG);
271           if (!Utils.isBlank(allowedOrigins)) {
272               FilterHolder filterHolder = new FilterHolder(new CrossOriginFilter());
273               filterHolder.setName("cross-origin");
274               filterHolder.setInitParameter(CrossOriginFilter.ALLOWED_ORIGINS_PARAM, allowedOrigins);
275               String allowedMethods = config.getString(WorkerConfig.ACCESS_CONTROL_ALLOW_METHODS_CONFIG);
276               if (!Utils.isBlank(allowedMethods)) {
277                   filterHolder.setInitParameter(CrossOriginFilter.ALLOWED_METHODS_PARAM, allowedMethods);
278               }
279               context.addFilter(filterHolder, "/*", EnumSet.of(DispatcherType.REQUEST));
280           }
281   
282           String headerConfig = config.getString(WorkerConfig.RESPONSE_HTTP_HEADERS_CONFIG);
283           if (!Utils.isBlank(headerConfig)) {
284               configureHttpResponsHeaderFilter(context);
285           }
286   
287           RequestLogHandler requestLogHandler = new RequestLogHandler();
288           Slf4jRequestLogWriter slf4jRequestLogWriter = new Slf4jRequestLogWriter();
289           slf4jRequestLogWriter.setLoggerName(RestServer.class.getCanonicalName());
290           CustomRequestLog requestLog = new CustomRequestLog(slf4jRequestLogWriter, CustomRequestLog.EXTENDED_NCSA_FORMAT + " %{ms}T");
291           requestLogHandler.setRequestLog(requestLog);
292   
293           contextHandlers.add(new DefaultHandler());
294           contextHandlers.add(requestLogHandler);
295   
296           handlers.setHandlers(contextHandlers.toArray(new Handler[0]));
297           try {
298               context.start();
299           } catch (Exception e) {
300               throw new ConnectException("Unable to initialize REST resources", e);
301           }
302   
303           if (adminResourceConfig != resourceConfig) {
304               try {
305                   log.debug("Starting admin context");
306                   adminContext.start();
307               } catch (Exception e) {
308                   throw new ConnectException("Unable to initialize Admin REST resources", e);
309               }
310           }
311   
312           log.info("REST resources initialized; server is started and ready to handle requests");
313       }

338       }

340       /**
341        * Get the URL to advertise to other workers and clients. This uses the default connector from the embedded Jetty
342        * server, unless overrides for advertised hostname and/or port are provided via configs. {@link #initializeServer()}
343        * must be invoked successfully before calling this method.
344        */
345       public URI advertisedUrl() {
346           UriBuilder builder = UriBuilder.fromUri(jettyServer.getURI());
347   
348           String advertisedSecurityProtocol = determineAdvertisedProtocol();
349           ServerConnector serverConnector = findConnector(advertisedSecurityProtocol);
350           builder.scheme(advertisedSecurityProtocol);
351   
352           String advertisedHostname = config.getString(WorkerConfig.REST_ADVERTISED_HOST_NAME_CONFIG);
353           if (advertisedHostname != null && !advertisedHostname.isEmpty())
354               builder.host(advertisedHostname);
355           else if (serverConnector != null && serverConnector.getHost() != null && serverConnector.getHost().length() > 0)
356               builder.host(serverConnector.getHost());
357   
358           Integer advertisedPort = config.getInt(WorkerConfig.REST_ADVERTISED_PORT_CONFIG);
359           if (advertisedPort != null)
360               builder.port(advertisedPort);
361           else if (serverConnector != null && serverConnector.getPort() > 0)
362               builder.port(serverConnector.getPort());
363   
364           log.info("Advertised URI: {}", builder.build());
365   
366           return builder.build();
367       }

369       /**
370        * @return the admin url for this worker. can be null if admin endpoints are disabled.
371        */
372       public URI adminUrl() {
373           ServerConnector adminConnector = null;
374           for (Connector connector : jettyServer.getConnectors()) {
375               if (ADMIN_SERVER_CONNECTOR_NAME.equals(connector.getName()))
376                   adminConnector = (ServerConnector) connector;
377           }
378   
379           if (adminConnector == null) {
380               List<String> adminListeners = config.getList(WorkerConfig.ADMIN_LISTENERS_CONFIG);
381               if (adminListeners == null) {
382                   return advertisedUrl();
383               } else if (adminListeners.isEmpty()) {
384                   return null;
385               } else {
386                   log.error("No admin connector found for listeners {}", adminListeners);
387                   return null;
388               }
389           }
390   
391           UriBuilder builder = UriBuilder.fromUri(jettyServer.getURI());
392           builder.port(adminConnector.getLocalPort());
393   
394           return builder.build();
395       }

402       String determineAdvertisedProtocol() {
403           String advertisedSecurityProtocol = config.getString(WorkerConfig.REST_ADVERTISED_LISTENER_CONFIG);
404           if (advertisedSecurityProtocol == null) {
405               String listeners = (String) config.originals().get(WorkerConfig.LISTENERS_CONFIG);
406   
407               if (listeners == null)
408                   return PROTOCOL_HTTP;
409               else
410                   listeners = listeners.toLowerCase(Locale.ENGLISH);
411   
412               if (listeners.contains(String.format("%s://", PROTOCOL_HTTP)))
413                   return PROTOCOL_HTTP;
414               else if (listeners.contains(String.format("%s://", PROTOCOL_HTTPS)))
415                   return PROTOCOL_HTTPS;
416               else
417                   return PROTOCOL_HTTP;
418           } else {
419               return advertisedSecurityProtocol.toLowerCase(Locale.ENGLISH);
420           }
421       }
422   
423       /**
424        * Locate a Jetty connector for the standard (non-admin) REST API that uses the given protocol.
425        * @param protocol the protocol for the connector (e.g., "http" or "https").
426        * @return a {@link ServerConnector} for the server that uses the requested protocol, or
427        * {@code null} if none exist.
428        */
429       ServerConnector findConnector(String protocol) {
430           for (Connector connector : jettyServer.getConnectors()) {
431               String connectorName = connector.getName();
432               // We set the names for these connectors when instantiating them, beginning with the
433               // protocol for the connector and then an underscore ("_"). We rely on that format here
434               // when trying to locate a connector with the requested protocol; if the naming format
435               // for the connectors we create is ever changed, we'll need to adjust the logic here
436               // accordingly.
437               if (connectorName.startsWith(protocol + "_") && !ADMIN_SERVER_CONNECTOR_NAME.equals(connectorName))
438                   return (ServerConnector) connector;
439           }
440   
441           return null;
442       }

444       void registerRestExtensions(Herder herder, ResourceConfig resourceConfig) {
445           connectRestExtensions = herder.plugins().newPlugins(
446               config.getList(WorkerConfig.REST_EXTENSION_CLASSES_CONFIG),
447               config, ConnectRestExtension.class);
448   
449           long herderRequestTimeoutMs = ConnectResource.DEFAULT_REST_REQUEST_TIMEOUT_MS;
450   
451           Integer rebalanceTimeoutMs = config.getRebalanceTimeout();
452   
453           if (rebalanceTimeoutMs != null) {
454               herderRequestTimeoutMs = Math.min(herderRequestTimeoutMs, rebalanceTimeoutMs.longValue());
455           }
456   
457           ConnectClusterDetails connectClusterDetails = new ConnectClusterDetailsImpl(
458               herder.kafkaClusterId()
459           );
460   
461           ConnectRestExtensionContext connectRestExtensionContext =
462               new ConnectRestExtensionContextImpl(
463                   new ConnectRestConfigurable(resourceConfig),
464                   new ConnectClusterStateImpl(herderRequestTimeoutMs, connectClusterDetails, herder)
465               );
466           for (ConnectRestExtension connectRestExtension : connectRestExtensions) {
467               connectRestExtension.register(connectRestExtensionContext);
468           }
469   
470       }
