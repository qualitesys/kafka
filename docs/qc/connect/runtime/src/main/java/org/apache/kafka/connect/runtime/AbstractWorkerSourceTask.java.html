123        * {@link Producer#send}. Does not guarantee that the record has been sent to Kafka or ack'd by the required number
124        * of brokers, but does guarantee that it will never be re-processed.

127       protected abstract void recordDispatched(SourceRecord record);
128   
129       /**
130        * Invoked when an entire batch of records returned from {@link SourceTask#poll} has been transformed, converted,
131        * and either discarded due to transform/conversion errors, filtered by a transform, or dispatched to the producer
132        * successfully via {@link Producer#send}. Does not guarantee that the records have been sent to Kafka or ack'd by the
133        * required number of brokers, but does guarantee that none of the records in the batch will ever be re-processed during
134        * the lifetime of this task. At most one record batch is polled from the task in between calls to this method.
135        */
136       protected abstract void batchDispatched();
137   
138       /**
139        * Invoked when a record has been sent and ack'd by the Kafka cluster. Note that this method may be invoked
140        *  concurrently and should therefore be made thread-safe.
141        * @param sourceRecord  the pre-transform {@link SourceRecord} that was successfully sent to Kafka; never null.
142        * @param producerRecord the {@link ProducerRecord} produced by transforming and converting the
143        * {@code sourceRecord}; never null;
144        * @param recordMetadata the {@link RecordMetadata} for the corresponding producer record; never null.
145        */
146       protected abstract void recordSent(
147               SourceRecord sourceRecord,
148               ProducerRecord<byte[], byte[]> producerRecord,
149               RecordMetadata recordMetadata
150       );
151   
152       /**
153        * Invoked when a record given to {@link Producer#send(ProducerRecord, Callback)} has failed with a non-retriable error.
154        * @param synchronous whether the error occurred during the invocation of {@link Producer#send(ProducerRecord, Callback)}.
155        *                    If {@code false}, indicates that the error was reported asynchronously by the producer by a {@link Callback}
156        * @param producerRecord the {@link ProducerRecord} that the producer failed to send; never null

158        * @param e the exception that was either thrown from {@link Producer#send(ProducerRecord, Callback)}, or reported by the producer
159        *          via {@link Callback} after the call to {@link Producer#send(ProducerRecord, Callback)} completed

256       @Override
257       protected void initializeAndStart() {
258           prepareToInitializeTask();
259           // If we try to start the task at all by invoking initialize, then count this as
260           // "started" and expect a subsequent call to the task's stop() method
261           // to properly clean up any resources allocated by its initialize() or
262           // start() methods. If the task throws an exception during stop(),
263           // the worst thing that happens is another exception gets logged for an already-
264           // failed task
265           started = true;
266           task.initialize(sourceTaskContext);
267           task.start(taskConfig);
268           log.info("{} Source task finished initialization and start", this);
269       }

271       @Override
272       public void cancel() {
273           super.cancel();
274           // Preemptively close the offset reader in case the task is blocked on an offset read.
275           offsetReader.close();
276           // We proactively close the producer here as the main work thread for the task may
277           // be blocked indefinitely in a call to Producer::send if automatic topic creation is
278           // not enabled on either the connector or the Kafka cluster. Closing the producer should
279           // unblock it in that case and allow shutdown to proceed normally.
280           // With a duration of 0, the producer's own shutdown logic should be fairly quick,
281           // but closing user-pluggable classes like interceptors may lag indefinitely. So, we
282           // call close on a separate thread in order to avoid blocking the herder's tick thread.
283           closeExecutor.execute(() -> closeProducer(Duration.ZERO));
284       }

321       @Override
322       public void execute() {
323           try {
324               prepareToEnterSendLoop();
325               while (!isStopping()) {
326                   beginSendIteration();
327   
328                   if (shouldPause()) {
329                       onPause();
330                       if (awaitUnpause()) {
331                           onResume();
332                           prepareToEnterSendLoop();
333                       }
334                       continue;
335                   }
336   
337                   if (toSend == null) {
338                       prepareToPollTask();
339   
340                       log.trace("{} Nothing to send to Kafka. Polling source for additional records", this);
341                       long start = time.milliseconds();
342                       toSend = poll();
343                       if (toSend != null) {
344                           recordPollReturned(toSend.size(), time.milliseconds() - start);
345                       }
346                   }
347                   if (toSend == null)
348                       continue;
349                   log.trace("{} About to send {} records to Kafka", this, toSend.size());
350                   if (sendRecords()) {
351                       batchDispatched();
352                   } else {
353                       stopRequestedLatch.await(SEND_FAILED_BACKOFF_MS, TimeUnit.MILLISECONDS);
354                   }
355               }
356           } catch (InterruptedException e) {
357               // Ignore and allow to exit.
358           } catch (RuntimeException e) {
359               try {
360                   finalOffsetCommit(true);
361               } catch (Exception offsetException) {
362                   log.error("Failed to commit offsets for already-failing task", offsetException);
363               }
364               throw e;
365           }
366           finalOffsetCommit(false);
367       }
368   
369       /**
370        * Try to send a batch of records. If a send fails and is retriable, this saves the remainder of the batch so it can
371        * be retried after backing off. If a send fails and is not retriable, this will throw a ConnectException.
372        * @return true if all messages were sent, false if some need to be retried
373        */
374       // Visible for testing
375       boolean sendRecords() {
376           int processed = 0;
377           recordBatch(toSend.size());
378           final SourceRecordWriteCounter counter =
379                   toSend.size() > 0 ? new SourceRecordWriteCounter(toSend.size(), sourceTaskMetricsGroup) : null;
380           for (final SourceRecord preTransformRecord : toSend) {
381               retryWithToleranceOperator.sourceRecord(preTransformRecord);
382               final SourceRecord record = transformationChain.apply(preTransformRecord);
383               final ProducerRecord<byte[], byte[]> producerRecord = convertTransformedRecord(record);
384               if (producerRecord == null || retryWithToleranceOperator.failed()) {
385                   counter.skipRecord();
386                   recordDropped(preTransformRecord);
387                   continue;
388               }
389   
390               log.trace("{} Appending record to the topic {} with key {}, value {}", this, record.topic(), record.key(), record.value());
391               Optional<SubmittedRecords.SubmittedRecord> submittedRecord = prepareToSendRecord(preTransformRecord, producerRecord);
392               try {
393                   final String topic = producerRecord.topic();
394                   maybeCreateTopic(topic);
395                   producer.send(
396                       producerRecord,
397                       (recordMetadata, e) -> {
398                           if (e != null) {
399                               log.debug("{} failed to send record to {}: ", AbstractWorkerSourceTask.this, topic, e);
400                               log.trace("{} Failed record: {}", AbstractWorkerSourceTask.this, preTransformRecord);
401                               producerSendFailed(false, producerRecord, preTransformRecord, e);
402                           } else {
403                               counter.completeRecord();
404                               log.trace("{} Wrote record successfully: topic {} partition {} offset {}",
405                                       AbstractWorkerSourceTask.this,
406                                       recordMetadata.topic(), recordMetadata.partition(),
407                                       recordMetadata.offset());
408                               recordSent(preTransformRecord, producerRecord, recordMetadata);
409                               submittedRecord.ifPresent(SubmittedRecords.SubmittedRecord::ack);
410                               if (topicTrackingEnabled) {
411                                   recordActiveTopic(producerRecord.topic());
412                               }
413                           }
414                       });
415                   // Note that this will cause retries to take place within a transaction
416               } catch (RetriableException | org.apache.kafka.common.errors.RetriableException e) {
417                   log.warn("{} Failed to send record to topic '{}' and partition '{}'. Backing off before retrying: ",
418                           this, producerRecord.topic(), producerRecord.partition(), e);
419                   toSend = toSend.subList(processed, toSend.size());
420                   submittedRecord.ifPresent(SubmittedRecords.SubmittedRecord::drop);
421                   counter.retryRemaining();
422                   return false;
423               } catch (ConnectException e) {
424                   log.warn("{} Failed to send record to topic '{}' and partition '{}' due to an unrecoverable exception: ",
425                           this, producerRecord.topic(), producerRecord.partition(), e);
426                   log.trace("{} Failed to send {} with unrecoverable exception: ", this, producerRecord, e);
427                   throw e;
428               } catch (KafkaException e) {
429                   producerSendFailed(true, producerRecord, preTransformRecord, e);
430               }
431               processed++;
432               recordDispatched(preTransformRecord);
433           }
434           toSend = null;
435           return true;
436       }

448       /**
449        * Convert the source record into a producer record.
450        *
451        * @param record the transformed record
452        * @return the producer record which can sent over to Kafka. A null is returned if the input is null or
453        * if an error was encountered during any of the converter stages.
454        */
455       protected ProducerRecord<byte[], byte[]> convertTransformedRecord(SourceRecord record) {
456           if (record == null) {
457               return null;
458           }
459   
460           RecordHeaders headers = retryWithToleranceOperator.execute(() -> convertHeaderFor(record), Stage.HEADER_CONVERTER, headerConverter.getClass());
461   
462           byte[] key = retryWithToleranceOperator.execute(() -> keyConverter.fromConnectData(record.topic(), headers, record.keySchema(), record.key()),
463                   Stage.KEY_CONVERTER, keyConverter.getClass());
464   
465           byte[] value = retryWithToleranceOperator.execute(() -> valueConverter.fromConnectData(record.topic(), headers, record.valueSchema(), record.value()),
466                   Stage.VALUE_CONVERTER, valueConverter.getClass());
467   
468           if (retryWithToleranceOperator.failed()) {
469               return null;
470           }
471   
472           return new ProducerRecord<>(record.topic(), record.kafkaPartition(),
473                   ConnectUtils.checkAndConvertTimestamp(record.timestamp()), key, value, headers);
474       }

476       // Due to transformations that may change the destination topic of a record (such as
477       // RegexRouter) topic creation can not be batched for multiple topics
478       private void maybeCreateTopic(String topic) {
479           if (!topicCreation.isTopicCreationRequired(topic)) {
480               log.trace("Topic creation by the connector is disabled or the topic {} was previously created." +
481                       "If auto.create.topics.enable is enabled on the broker, " +
482                       "the topic will be created with default settings", topic);
483               return;
484           }
485           log.info("The task will send records to topic '{}' for the first time. Checking "
486                   + "whether topic exists", topic);
487           Map<String, TopicDescription> existing = admin.describeTopics(topic);
488           if (!existing.isEmpty()) {
489               log.info("Topic '{}' already exists.", topic);
490               topicCreation.addTopic(topic);
491               return;
492           }
493   
494           log.info("Creating topic '{}'", topic);
495           TopicCreationGroup topicGroup = topicCreation.findFirstGroup(topic);
496           log.debug("Topic '{}' matched topic creation group: {}", topic, topicGroup);
497           NewTopic newTopic = topicGroup.newTopic(topic);
498   
499           TopicAdmin.TopicCreationResponse response = admin.createOrFindTopics(newTopic);
500           if (response.isCreated(newTopic.name())) {
501               topicCreation.addTopic(topic);
502               log.info("Created topic '{}' using creation group {}", newTopic, topicGroup);
503           } else if (response.isExisting(newTopic.name())) {
504               topicCreation.addTopic(topic);
505               log.info("Found existing topic '{}'", newTopic);
506           } else {
507               // The topic still does not exist and could not be created, so treat it as a task failure
508               log.warn("Request to create new topic '{}' failed", topic);
509               throw new ConnectException("Task failed to create new topic " + newTopic + ". Ensure "
510                       + "that the task is authorized to create topics or that the topic exists and "
511                       + "restart the task");
512           }
513       }

517           RecordHeaders result = new RecordHeaders();

523                   result.add(key, rawHeader);
524               }
525           }
526           return result;
527       }

553       static class SourceRecordWriteCounter {
554           private final SourceTaskMetricsGroup metricsGroup;
555           private final int batchSize;
556           private boolean completed = false;
557           private int counter;
558           public SourceRecordWriteCounter(int batchSize, SourceTaskMetricsGroup metricsGroup) {
559               assert batchSize > 0;
560               assert metricsGroup != null;
561               this.batchSize = batchSize;
562               counter = batchSize;
563               this.metricsGroup = metricsGroup;
564           }
565           public void skipRecord() {
566               if (counter > 0 && --counter == 0) {
567                   finishedAllWrites();
568               }
569           }
570           public void completeRecord() {
571               if (counter > 0 && --counter == 0) {
572                   finishedAllWrites();
573               }
574           }
575           public void retryRemaining() {
576               finishedAllWrites();
577           }
578           private void finishedAllWrites() {
579               if (!completed) {
580                   metricsGroup.recordWrite(batchSize - counter);
581                   completed = true;
582               }
583           }
584       }
