17   package org.apache.kafka.connect.runtime.isolation;
18   
19   import org.slf4j.Logger;
20   import org.slf4j.LoggerFactory;
21   
22   import java.net.URL;
23   import java.net.URLClassLoader;
24   
25   /**
26    * A custom classloader dedicated to loading Connect plugin classes in classloading isolation.
27    * <p>
28    * Under the current scheme for classloading isolation in Connect, a plugin classloader loads the
29    * classes that it finds in its urls. For classes that are either not found or are not supposed to
30    * be loaded in isolation, this plugin classloader delegates their loading to its parent. This makes
31    * this classloader a child-first classloader.
32    * <p>
33    * This class is thread-safe and parallel capable.
34    */
35   public class PluginClassLoader extends URLClassLoader {
36       private static final Logger log = LoggerFactory.getLogger(PluginClassLoader.class);
37       private final URL pluginLocation;
38   
39       static {
40           ClassLoader.registerAsParallelCapable();
41       }
42   
43       /**
44        * Constructor that accepts a specific classloader as parent.
45        *
46        * @param pluginLocation the top-level location of the plugin to be loaded in isolation by this
47        * classloader.
48        * @param urls the list of urls from which to load classes and resources for this plugin.
49        * @param parent the parent classloader to be used for delegation for classes that were
50        * not found or should not be loaded in isolation by this classloader.
51        */
52       public PluginClassLoader(URL pluginLocation, URL[] urls, ClassLoader parent) {
53           super(urls, parent);
54           this.pluginLocation = pluginLocation;
55       }
56   
57       /**
58        * Constructor that defines the system classloader as parent of this plugin classloader.
59        *
60        * @param pluginLocation the top-level location of the plugin to be loaded in isolation by this
61        * classloader.
62        * @param urls the list of urls from which to load classes and resources for this plugin.
63        */
64       public PluginClassLoader(URL pluginLocation, URL[] urls) {
65           super(urls);
66           this.pluginLocation = pluginLocation;
67       }
68   
69       /**
70        * Returns the top-level location of the classes and dependencies required by the plugin that
71        * is loaded by this classloader.
72        *
73        * @return the plugin location.
74        */
75       public String location() {
76           return pluginLocation.toString();
77       }
78   
79       @Override
80       public String toString() {
81           return "PluginClassLoader{pluginLocation=" + pluginLocation + "}";
82       }
83   
84       // This method needs to be thread-safe because it is supposed to be called by multiple
85       // Connect tasks. While findClass is thread-safe, defineClass called within loadClass of the
86       // base method is not. More on multithreaded classloaders in:
87       // https://docs.oracle.com/javase/7/docs/technotes/guides/lang/cl-mt.html
88       @Override
89       protected Class<?> loadClass(String name, boolean resolve) throws ClassNotFoundException {
90           synchronized (getClassLoadingLock(name)) {
91               Class<?> klass = findLoadedClass(name);
92               if (klass == null) {
93                   try {
94                       if (PluginUtils.shouldLoadInIsolation(name)) {
95                           klass = findClass(name);
96                       }
97                   } catch (ClassNotFoundException e) {
98                       // Not found in loader's path. Search in parents.
99                       log.trace("Class '{}' not found. Delegating to parent", name);
100                   }
101               }
102               if (klass == null) {
103                   klass = super.loadClass(name, false);
104               }
105               if (resolve) {
106                   resolveClass(klass);
107               }
108               return klass;
109           }
110       }
111   }
