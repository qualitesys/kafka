17   package org.apache.kafka.connect.tools;
18   
19   import com.fasterxml.jackson.core.JsonProcessingException;
20   import com.fasterxml.jackson.databind.ObjectMapper;
21   import org.apache.kafka.clients.consumer.OffsetAndMetadata;
22   import org.apache.kafka.common.TopicPartition;
23   import org.apache.kafka.connect.errors.ConnectException;
24   import org.apache.kafka.connect.sink.SinkRecord;
25   import org.apache.kafka.connect.sink.SinkTask;
26   
27   import java.util.ArrayList;
28   import java.util.Collection;
29   import java.util.HashMap;
30   import java.util.List;
31   import java.util.Map;
32   
33   /**
34    * Counterpart to {@link VerifiableSourceTask} that consumes records and logs information about each to stdout. This
35    * allows validation of processing of messages by sink tasks on distributed workers even in the face of worker restarts
36    * and failures. This task relies on the offset management provided by the Kafka Connect framework and therefore can detect
37    * bugs in its implementation.
38    */
39   public class VerifiableSinkTask extends SinkTask {
40       public static final String NAME_CONFIG = "name";
41       public static final String ID_CONFIG = "id";
42   
43       private static final ObjectMapper JSON_SERDE = new ObjectMapper();
44   
45       private String name; // Connector name
46       private int id; // Task ID
47   
48       private final Map<TopicPartition, List<Map<String, Object>>> unflushed = new HashMap<>();
49   
50       @Override
51       public String version() {
52           return new VerifiableSinkConnector().version();
53       }
54   
55       @Override
56       public void start(Map<String, String> props) {
57           try {
58               name = props.get(NAME_CONFIG);
59               id = Integer.parseInt(props.get(ID_CONFIG));
60           } catch (NumberFormatException e) {
61               throw new ConnectException("Invalid VerifiableSourceTask configuration", e);
62           }
63       }
64   
65       @Override
66       public void put(Collection<SinkRecord> records) {
67           long nowMs = System.currentTimeMillis();
68           for (SinkRecord record : records) {
69               Map<String, Object> data = new HashMap<>();
70               data.put("name", name);
71               data.put("task", record.key()); // VerifiableSourceTask's input task (source partition)
72               data.put("sinkTask", id);
73               data.put("topic", record.topic());
74               data.put("time_ms", nowMs);
75               data.put("seqno", record.value());
76               data.put("offset", record.kafkaOffset());
77               String dataJson;
78               try {
79                   dataJson = JSON_SERDE.writeValueAsString(data);
80               } catch (JsonProcessingException e) {
81                   dataJson = "Bad data can't be written as json: " + e.getMessage();
82               }
83               System.out.println(dataJson);
84               unflushed.computeIfAbsent(
85                       new TopicPartition(record.topic(), record.kafkaPartition()),
86                       tp -> new ArrayList<>()
87               ).add(data);
88           }
89       }
90   
91       @Override
92       public void flush(Map<TopicPartition, OffsetAndMetadata> offsets) {
93           long nowMs = System.currentTimeMillis();
94           for (TopicPartition topicPartition : offsets.keySet()) {
95               if (!unflushed.containsKey(topicPartition)) {
96                   continue;
97               }
98               for (Map<String, Object> data : unflushed.get(topicPartition)) {
99                   data.put("time_ms", nowMs);
100                   data.put("flushed", true);
101                   String dataJson;
102                   try {
103                       dataJson = JSON_SERDE.writeValueAsString(data);
104                   } catch (JsonProcessingException e) {
105                       dataJson = "Bad data can't be written as json: " + e.getMessage();
106                   }
107                   System.out.println(dataJson);
108               }
109               unflushed.remove(topicPartition);
110           }
111       }
112   
113       @Override
114       public void stop() {
115   
116       }
117   }
