48   public class WorkerErrantRecordReporter implements ErrantRecordReporter {
49   
50       private static final Logger log = LoggerFactory.getLogger(WorkerErrantRecordReporter.class);
51   
52       private final RetryWithToleranceOperator retryWithToleranceOperator;
53       private final Converter keyConverter;
54       private final Converter valueConverter;
55       private final HeaderConverter headerConverter;
56   
57       // Visible for testing
58       protected final ConcurrentMap<TopicPartition, List<Future<Void>>> futures;
59   
60       public WorkerErrantRecordReporter(
61           RetryWithToleranceOperator retryWithToleranceOperator,
62           Converter keyConverter,
63           Converter valueConverter,
64           HeaderConverter headerConverter
65       ) {
66           this.retryWithToleranceOperator = retryWithToleranceOperator;
67           this.keyConverter = keyConverter;
68           this.valueConverter = valueConverter;
69           this.headerConverter = headerConverter;
70           this.futures = new ConcurrentHashMap<>();
71       }
72   
73       @Override
74       public Future<Void> report(SinkRecord record, Throwable error) {
75           ConsumerRecord<byte[], byte[]> consumerRecord;
76   
77           // Most of the records will be an internal sink record, but the task could potentially
78           // report modified or new records, so handle both cases
79           if (record instanceof InternalSinkRecord) {
80               consumerRecord = ((InternalSinkRecord) record).originalRecord();
81           } else {
82               // Generate a new consumer record from the modified sink record. We prefer
83               // to send the original consumer record (pre-transformed) to the DLQ,
84               // but in this case we don't have one and send the potentially transformed
85               // record instead
86               String topic = record.topic();
87               byte[] key = keyConverter.fromConnectData(topic, record.keySchema(), record.key());
88               byte[] value = valueConverter.fromConnectData(topic,
89                   record.valueSchema(), record.value());
90   
91               RecordHeaders headers = new RecordHeaders();
92               if (record.headers() != null) {
93                   for (Header header : record.headers()) {
94                       String headerKey = header.key();
95                       byte[] rawHeader = headerConverter.fromConnectHeader(topic, headerKey,
96                           header.schema(), header.value());
97                       headers.add(headerKey, rawHeader);
98                   }
99               }
100   
101               int keyLength = key != null ? key.length : -1;
102               int valLength = value != null ? value.length : -1;
103   
104               consumerRecord = new ConsumerRecord<>(record.topic(), record.kafkaPartition(),
105                   record.kafkaOffset(), record.timestamp(), record.timestampType(), keyLength,
106                   valLength, key, value, headers, Optional.empty());
107           }
108   
109           Future<Void> future = retryWithToleranceOperator.executeFailed(Stage.TASK_PUT, SinkTask.class, consumerRecord, error);
110   
111           if (!future.isDone()) {
112               TopicPartition partition = new TopicPartition(consumerRecord.topic(), consumerRecord.partition());
113               futures.computeIfAbsent(partition, p -> new ArrayList<>()).add(future);
114           }
115           return future;
116       }
117   
118       /**
119        * Awaits the completion of all error reports for a given set of topic partitions
120        * @param topicPartitions the topic partitions to await reporter completion for
121        */
122       public void awaitFutures(Collection<TopicPartition> topicPartitions) {
123           futuresFor(topicPartitions).forEach(future -> {
124               try {
125                   future.get();
126               } catch (InterruptedException | ExecutionException e) {
127                   log.error("Encountered an error while awaiting an errant record future's completion.", e);
128                   throw new ConnectException(e);
129               }
130           });
131       }
132   
133       /**
134        * Cancels all active error reports for a given set of topic partitions
135        * @param topicPartitions the topic partitions to cancel reporting for
136        */
137       public void cancelFutures(Collection<TopicPartition> topicPartitions) {
138           futuresFor(topicPartitions).forEach(future -> {
139               try {
140                   future.cancel(true);
141               } catch (Exception e) {
142                   log.error("Encountered an error while cancelling an errant record future", e);
143                   // No need to throw the exception here; it's enough to log an error message
144               }
145           });
146       }
147   
148       // Removes and returns all futures for the given topic partitions from the set of currently-active futures
149       private Collection<Future<Void>> futuresFor(Collection<TopicPartition> topicPartitions) {
150           return topicPartitions.stream()
151                   .map(futures::remove)
152                   .filter(Objects::nonNull)
153                   .flatMap(List::stream)
154                   .collect(Collectors.toList());
155       }
156   
157       /**
158        * Wrapper class to aggregate producer futures and abstract away the record metadata from the
159        * Connect user.
160        */
161       public static class ErrantRecordFuture implements Future<Void> {
162   
163           private final List<Future<RecordMetadata>> futures;
164   
165           public ErrantRecordFuture(List<Future<RecordMetadata>> producerFutures) {
166               futures = producerFutures;
167           }
168   
169           public boolean cancel(boolean mayInterruptIfRunning) {
170               throw new UnsupportedOperationException("Reporting an errant record cannot be cancelled.");
171           }
172   
173           public boolean isCancelled() {
174               return false;
175           }
176   
177           public boolean isDone() {
178               return futures.stream().allMatch(Future::isDone);
179           }
180   
181           public Void get() throws InterruptedException, ExecutionException {
182               for (Future<RecordMetadata> future: futures) {
183                   future.get();
184               }
185               return null;
186           }
187   
188           public Void get(long timeout, TimeUnit unit)
189               throws InterruptedException, ExecutionException, TimeoutException {
190               for (Future<RecordMetadata> future: futures) {
191                   future.get(timeout, unit);
192               }
193               return null;
194           }
195       }
196   }
