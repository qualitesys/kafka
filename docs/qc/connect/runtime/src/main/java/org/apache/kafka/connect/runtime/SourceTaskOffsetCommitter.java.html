17   package org.apache.kafka.connect.runtime;
18   
19   import org.apache.kafka.connect.errors.ConnectException;
20   import org.apache.kafka.connect.util.ConnectorTaskId;
21   import org.apache.kafka.connect.util.LoggingContext;
22   import org.apache.kafka.common.utils.ThreadUtils;
23   import org.slf4j.Logger;
24   import org.slf4j.LoggerFactory;
25   
26   import java.util.concurrent.CancellationException;
27   import java.util.concurrent.ConcurrentHashMap;
28   import java.util.concurrent.ConcurrentMap;
29   import java.util.concurrent.ExecutionException;
30   import java.util.concurrent.Executors;
31   import java.util.concurrent.ScheduledExecutorService;
32   import java.util.concurrent.ScheduledFuture;
33   import java.util.concurrent.TimeUnit;
34   
35   /**
36    * <p>
37    * Manages offset commit scheduling and execution for SourceTasks.
38    * </p>
39    * <p>
40    * Unlike sink tasks which directly manage their offset commits in the main poll() thread since
41    * they drive the event loop and control (for all intents and purposes) the timeouts, source
42    * tasks are at the whim of the connector and cannot be guaranteed to wake up on the necessary
43    * schedule. Instead, this class tracks all the active tasks, their schedule for commits, and
44    * ensures they are invoked in a timely fashion.
45    * </p>
46    */
47   class SourceTaskOffsetCommitter {
48       private static final Logger log = LoggerFactory.getLogger(SourceTaskOffsetCommitter.class);
49   
50       private final WorkerConfig config;
51       private final ScheduledExecutorService commitExecutorService;
52       private final ConcurrentMap<ConnectorTaskId, ScheduledFuture<?>> committers;
53   
54       // visible for testing
55       SourceTaskOffsetCommitter(WorkerConfig config,
56                                 ScheduledExecutorService commitExecutorService,
57                                 ConcurrentMap<ConnectorTaskId, ScheduledFuture<?>> committers) {
58           this.config = config;
59           this.commitExecutorService = commitExecutorService;
60           this.committers = committers;
61       }
62   
63       public SourceTaskOffsetCommitter(WorkerConfig config) {
64           this(config, Executors.newSingleThreadScheduledExecutor(ThreadUtils.createThreadFactory(
65                   SourceTaskOffsetCommitter.class.getSimpleName() + "-%d", false)),
66                   new ConcurrentHashMap<>());
67       }
68   
69       public void close(long timeoutMs) {
70           commitExecutorService.shutdown();
71           try {
72               if (!commitExecutorService.awaitTermination(timeoutMs, TimeUnit.MILLISECONDS)) {
73                   log.error("Graceful shutdown of offset commitOffsets thread timed out.");
74               }
75           } catch (InterruptedException e) {
76               // ignore and allow to exit immediately
77           }
78       }
79   
80       public void schedule(final ConnectorTaskId id, final WorkerSourceTask workerTask) {
81           long commitIntervalMs = config.getLong(WorkerConfig.OFFSET_COMMIT_INTERVAL_MS_CONFIG);
82           ScheduledFuture<?> commitFuture = commitExecutorService.scheduleWithFixedDelay(() -> {
83               try (LoggingContext loggingContext = LoggingContext.forOffsets(id)) {
84                   commit(workerTask);
85               }
86           }, commitIntervalMs, commitIntervalMs, TimeUnit.MILLISECONDS);
87           committers.put(id, commitFuture);
88       }
89   
90       public void remove(ConnectorTaskId id) {
91           final ScheduledFuture<?> task = committers.remove(id);
92           if (task == null)
93               return;
94   
95           try (LoggingContext loggingContext = LoggingContext.forTask(id)) {
96               task.cancel(false);
97               if (!task.isDone())
98                   task.get();
99           } catch (CancellationException e) {
100               // ignore
101               log.trace("Offset commit thread was cancelled by another thread while removing connector task with id: {}", id);
102           } catch (ExecutionException | InterruptedException e) {
103               throw new ConnectException("Unexpected interruption in SourceTaskOffsetCommitter while removing task with id: " + id, e);
104           }
105       }
106   
107       private void commit(WorkerSourceTask workerTask) {
108           log.debug("{} Committing offsets", workerTask);
109           try {
110               if (workerTask.commitOffsets()) {
111                   return;
112               }
113               log.error("{} Failed to commit offsets", workerTask);
114           } catch (Throwable t) {
115               // We're very careful about exceptions here since any uncaught exceptions in the commit
116               // thread would cause the fixed interval schedule on the ExecutorService to stop running
117               // for that task
118               log.error("{} Unhandled exception when committing: ", workerTask, t);
119           }
120       }
121   }
