17   package org.apache.kafka.connect.storage;
18   
19   import org.apache.kafka.connect.runtime.WorkerConfig;
20   import org.apache.kafka.connect.util.Callback;
21   
22   import java.nio.ByteBuffer;
23   import java.util.Collection;
24   import java.util.Map;
25   import java.util.concurrent.Future;
26   
27   /**
28    * <p>
29    * OffsetBackingStore is an interface for storage backends that store key-value data. The backing
30    * store doesn't need to handle serialization or deserialization. It only needs to support
31    * reading/writing bytes. Since it is expected these operations will require network
32    * operations, only bulk operations are supported.
33    * </p>
34    * <p>
35    * Since OffsetBackingStore is a shared resource that may be used by many OffsetStorage instances
36    * that are associated with individual tasks, the caller must be sure keys include information about the
37    * connector so that the shared namespace does not result in conflicting keys.
38    * </p>
39    */
40   public interface OffsetBackingStore {
41   
42       /**
43        * Start this offset store.
44        */
45       void start();
46   
47       /**
48        * Stop the backing store. Implementations should attempt to shutdown gracefully, but not block
49        * indefinitely.
50        */
51       void stop();
52   
53       /**
54        * Get the values for the specified keys
55        * @param keys list of keys to look up
56        * @return future for the resulting map from key to value
57        */
58       Future<Map<ByteBuffer, ByteBuffer>> get(Collection<ByteBuffer> keys);
59   
60       /**
61        * Set the specified keys and values.
62        * @param values map from key to value
63        * @param callback callback to invoke on completion
64        * @return void future for the operation
65        */
66       Future<Void> set(Map<ByteBuffer, ByteBuffer> values, Callback<Void> callback);
67   
68       /**
69        * Configure class with the given key-value pairs
70        * @param config can be DistributedConfig or StandaloneConfig
71        */
72       void configure(WorkerConfig config);
73   }
