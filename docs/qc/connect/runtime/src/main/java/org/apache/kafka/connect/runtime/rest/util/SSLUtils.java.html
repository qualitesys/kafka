30   /**
31    * Helper class for setting up SSL for RestServer and RestClient
32    */
33   public class SSLUtils {
34   
35       private static final Pattern COMMA_WITH_WHITESPACE = Pattern.compile("\\s*,\\s*");
36   
37   
38       /**
39        * Configures SSL/TLS for HTTPS Jetty Server using configs with the given prefix
40        */
41       public static SslContextFactory createServerSideSslContextFactory(WorkerConfig config, String prefix) {
42           Map<String, Object> sslConfigValues = config.valuesWithPrefixAllOrNothing(prefix);
43   
44           final SslContextFactory.Server ssl = new SslContextFactory.Server();
45   
46           configureSslContextFactoryKeyStore(ssl, sslConfigValues);
47           configureSslContextFactoryTrustStore(ssl, sslConfigValues);
48           configureSslContextFactoryAlgorithms(ssl, sslConfigValues);
49           configureSslContextFactoryAuthentication(ssl, sslConfigValues);
50   
51           return ssl;
52       }
53   
54       /**
55        * Configures SSL/TLS for HTTPS Jetty Server
56        */
57       public static SslContextFactory createServerSideSslContextFactory(WorkerConfig config) {
58           return createServerSideSslContextFactory(config, "listeners.https.");
59       }
60   
61       /**
62        * Configures SSL/TLS for HTTPS Jetty Client
63        */
64       public static SslContextFactory createClientSideSslContextFactory(WorkerConfig config) {
65           Map<String, Object> sslConfigValues = config.valuesWithPrefixAllOrNothing("listeners.https.");
66   
67           final SslContextFactory.Client ssl = new SslContextFactory.Client();
68   
69           configureSslContextFactoryKeyStore(ssl, sslConfigValues);
70           configureSslContextFactoryTrustStore(ssl, sslConfigValues);
71           configureSslContextFactoryAlgorithms(ssl, sslConfigValues);
72           configureSslContextFactoryEndpointIdentification(ssl, sslConfigValues);
73   
74           return ssl;
75       }

77       /**
78        * Configures KeyStore related settings in SslContextFactory
79        */
80       protected static void configureSslContextFactoryKeyStore(SslContextFactory ssl, Map<String, Object> sslConfigValues) {
81           ssl.setKeyStoreType((String) getOrDefault(sslConfigValues, SslConfigs.SSL_KEYSTORE_TYPE_CONFIG, SslConfigs.DEFAULT_SSL_KEYSTORE_TYPE));
82   
83           String sslKeystoreLocation = (String) sslConfigValues.get(SslConfigs.SSL_KEYSTORE_LOCATION_CONFIG);
84           if (sslKeystoreLocation != null)
85               ssl.setKeyStorePath(sslKeystoreLocation);
86   
87           Password sslKeystorePassword = (Password) sslConfigValues.get(SslConfigs.SSL_KEYSTORE_PASSWORD_CONFIG);
88           if (sslKeystorePassword != null)
89               ssl.setKeyStorePassword(sslKeystorePassword.value());
90   
91           Password sslKeyPassword = (Password) sslConfigValues.get(SslConfigs.SSL_KEY_PASSWORD_CONFIG);
92           if (sslKeyPassword != null)
93               ssl.setKeyManagerPassword(sslKeyPassword.value());
94       }
95   
96       protected static Object getOrDefault(Map<String, Object> configMap, String key, Object defaultValue) {
97           if (configMap.containsKey(key))
98               return configMap.get(key);
99   
100           return defaultValue;
101       }

103       /**
104        * Configures TrustStore related settings in SslContextFactory
105        */
106       protected static void configureSslContextFactoryTrustStore(SslContextFactory ssl, Map<String, Object> sslConfigValues) {
107           ssl.setTrustStoreType((String) getOrDefault(sslConfigValues, SslConfigs.SSL_TRUSTSTORE_TYPE_CONFIG, SslConfigs.DEFAULT_SSL_TRUSTSTORE_TYPE));
108   
109           String sslTruststoreLocation = (String) sslConfigValues.get(SslConfigs.SSL_TRUSTSTORE_LOCATION_CONFIG);
110           if (sslTruststoreLocation != null)
111               ssl.setTrustStorePath(sslTruststoreLocation);
112   
113           Password sslTruststorePassword = (Password) sslConfigValues.get(SslConfigs.SSL_TRUSTSTORE_PASSWORD_CONFIG);
114           if (sslTruststorePassword != null)
115               ssl.setTrustStorePassword(sslTruststorePassword.value());
116       }
117   
118       /**
119        * Configures Protocol, Algorithm and Provider related settings in SslContextFactory
120        */
121       @SuppressWarnings("unchecked")
122       protected static void configureSslContextFactoryAlgorithms(SslContextFactory ssl, Map<String, Object> sslConfigValues) {
123           List<String> sslEnabledProtocols = (List<String>) getOrDefault(sslConfigValues, SslConfigs.SSL_ENABLED_PROTOCOLS_CONFIG, Arrays.asList(COMMA_WITH_WHITESPACE.split(SslConfigs.DEFAULT_SSL_ENABLED_PROTOCOLS)));
124           ssl.setIncludeProtocols(sslEnabledProtocols.toArray(new String[0]));
125   
126           String sslProvider = (String) sslConfigValues.get(SslConfigs.SSL_PROVIDER_CONFIG);
127           if (sslProvider != null)
128               ssl.setProvider(sslProvider);
129   
130           ssl.setProtocol((String) getOrDefault(sslConfigValues, SslConfigs.SSL_PROTOCOL_CONFIG, SslConfigs.DEFAULT_SSL_PROTOCOL));
131   
132           List<String> sslCipherSuites = (List<String>) sslConfigValues.get(SslConfigs.SSL_CIPHER_SUITES_CONFIG);
133           if (sslCipherSuites != null)
134               ssl.setIncludeCipherSuites(sslCipherSuites.toArray(new String[0]));
135   
136           ssl.setKeyManagerFactoryAlgorithm((String) getOrDefault(sslConfigValues, SslConfigs.SSL_KEYMANAGER_ALGORITHM_CONFIG, SslConfigs.DEFAULT_SSL_KEYMANGER_ALGORITHM));
137   
138           String sslSecureRandomImpl = (String) sslConfigValues.get(SslConfigs.SSL_SECURE_RANDOM_IMPLEMENTATION_CONFIG);
139           if (sslSecureRandomImpl != null)
140               ssl.setSecureRandomAlgorithm(sslSecureRandomImpl);
141   
142           ssl.setTrustManagerFactoryAlgorithm((String) getOrDefault(sslConfigValues, SslConfigs.SSL_TRUSTMANAGER_ALGORITHM_CONFIG, SslConfigs.DEFAULT_SSL_TRUSTMANAGER_ALGORITHM));
143       }

151               ssl.setEndpointIdentificationAlgorithm(sslEndpointIdentificationAlg);
152       }

154       /**
155        * Configures Authentication related settings in SslContextFactory
156        */
157       protected static void configureSslContextFactoryAuthentication(SslContextFactory.Server ssl, Map<String, Object> sslConfigValues) {
158           String sslClientAuth = (String) getOrDefault(sslConfigValues, BrokerSecurityConfigs.SSL_CLIENT_AUTH_CONFIG, "none");
159           switch (sslClientAuth) {
160               case "requested":
161                   ssl.setWantClientAuth(true);
162                   break;
163               case "required":
164                   ssl.setNeedClientAuth(true);
165                   break;
166               default:
167                   ssl.setNeedClientAuth(false);
168                   ssl.setWantClientAuth(false);
169           }
170       }
