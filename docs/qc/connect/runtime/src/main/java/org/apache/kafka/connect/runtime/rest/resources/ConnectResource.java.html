19   import java.util.concurrent.TimeUnit;
20   
21   /**
22    * This interface defines shared logic for all Connect REST resources.
23    */
24   public interface ConnectResource {
25   
26       // TODO: This should not be so long. However, due to potentially long rebalances that may have to wait a full
27       // session timeout to complete, during which we cannot serve some requests. Ideally we could reduce this, but
28       // we need to consider all possible scenarios this could fail. It might be ok to fail with a timeout in rare cases,
29       // but currently a worker simply leaving the group can take this long as well.
30       long DEFAULT_REST_REQUEST_TIMEOUT_MS = TimeUnit.SECONDS.toMillis(90);
31   
32       /**
33        * Set how long the resource will await the completion of each request before returning a 500 error.
34        * If the resource does not perform any operations that can be expected to block under reasonable
35        * circumstances, this can be implemented as a no-op.
36        * @param requestTimeoutMs the new timeout in milliseconds; must be positive
37        */
38       void requestTimeout(long requestTimeoutMs);
39   
40   }
