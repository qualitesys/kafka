55   /**
56    * This class manages the coordination process with brokers for the Connect cluster group membership. It ties together
57    * the Coordinator, which implements the group member protocol, with all the other pieces needed to drive the connection
58    * to the group coordinator broker. This isolates all the networking to a single thread managed by this class, with
59    * higher level operations in response to group membership events being handled by the herder.
60    */
61   public class WorkerGroupMember {
62       private static final String JMX_PREFIX = "kafka.connect";
63   
64       private final Logger log;
65       private final Time time;
66       private final String clientId;
67       private final ConsumerNetworkClient client;
68       private final Metrics metrics;
69       private final Metadata metadata;
70       private final long retryBackoffMs;
71       private final WorkerCoordinator coordinator;
72   
73       private boolean stopped = false;
74   
75       public WorkerGroupMember(DistributedConfig config,
76                                String restUrl,
77                                ConfigBackingStore configStorage,
78                                WorkerRebalanceListener listener,
79                                Time time,
80                                String clientId,
81                                LogContext logContext) {
82           try {
83               this.time = time;
84               this.clientId = clientId;
85               this.log = logContext.logger(WorkerGroupMember.class);
86   
87               Map<String, String> metricsTags = new LinkedHashMap<>();
88               metricsTags.put("client-id", clientId);
89               MetricConfig metricConfig = new MetricConfig().samples(config.getInt(CommonClientConfigs.METRICS_NUM_SAMPLES_CONFIG))
90                       .timeWindow(config.getLong(CommonClientConfigs.METRICS_SAMPLE_WINDOW_MS_CONFIG), TimeUnit.MILLISECONDS)
91                       .tags(metricsTags);
92               List<MetricsReporter> reporters = config.getConfiguredInstances(CommonClientConfigs.METRIC_REPORTER_CLASSES_CONFIG,
93                       MetricsReporter.class,
94                       Collections.singletonMap(CommonClientConfigs.CLIENT_ID_CONFIG, clientId));
95               JmxReporter jmxReporter = new JmxReporter();
96               jmxReporter.configure(config.originals());
97               reporters.add(jmxReporter);
98   
99               Map<String, Object> contextLabels = new HashMap<>();
100               contextLabels.putAll(config.originalsWithPrefix(CommonClientConfigs.METRICS_CONTEXT_PREFIX));
101               contextLabels.put(WorkerConfig.CONNECT_KAFKA_CLUSTER_ID, ConnectUtils.lookupKafkaClusterId(config));
102               contextLabels.put(WorkerConfig.CONNECT_GROUP_ID, config.getString(DistributedConfig.GROUP_ID_CONFIG));
103               MetricsContext metricsContext = new KafkaMetricsContext(JMX_PREFIX, contextLabels);
104   
105               this.metrics = new Metrics(metricConfig, reporters, time, metricsContext);
106               this.retryBackoffMs = config.getLong(CommonClientConfigs.RETRY_BACKOFF_MS_CONFIG);
107               this.metadata = new Metadata(retryBackoffMs, config.getLong(CommonClientConfigs.METADATA_MAX_AGE_CONFIG),
108                       logContext, new ClusterResourceListeners());
109               List<InetSocketAddress> addresses = ClientUtils.parseAndValidateAddresses(
110                       config.getList(CommonClientConfigs.BOOTSTRAP_SERVERS_CONFIG),
111                       config.getString(CommonClientConfigs.CLIENT_DNS_LOOKUP_CONFIG));
112               this.metadata.bootstrap(addresses);
113               String metricGrpPrefix = "connect";
114               ChannelBuilder channelBuilder = ClientUtils.createChannelBuilder(config, time, logContext);
115               NetworkClient netClient = new NetworkClient(
116                       new Selector(config.getLong(CommonClientConfigs.CONNECTIONS_MAX_IDLE_MS_CONFIG), metrics, time, metricGrpPrefix, channelBuilder, logContext),
117                       this.metadata,
118                       clientId,
119                       100, // a fixed large enough value will suffice
120                       config.getLong(CommonClientConfigs.RECONNECT_BACKOFF_MS_CONFIG),
121                       config.getLong(CommonClientConfigs.RECONNECT_BACKOFF_MAX_MS_CONFIG),
122                       config.getInt(CommonClientConfigs.SEND_BUFFER_CONFIG),
123                       config.getInt(CommonClientConfigs.RECEIVE_BUFFER_CONFIG),
124                       config.getInt(CommonClientConfigs.REQUEST_TIMEOUT_MS_CONFIG),
125                       config.getLong(CommonClientConfigs.SOCKET_CONNECTION_SETUP_TIMEOUT_MS_CONFIG),
126                       config.getLong(CommonClientConfigs.SOCKET_CONNECTION_SETUP_TIMEOUT_MAX_MS_CONFIG),
127                       time,
128                       true,
129                       new ApiVersions(),
130                       logContext);
131               this.client = new ConsumerNetworkClient(
132                       logContext,
133                       netClient,
134                       metadata,
135                       time,
136                       retryBackoffMs,
137                       config.getInt(CommonClientConfigs.REQUEST_TIMEOUT_MS_CONFIG),
138                       Integer.MAX_VALUE);
139               this.coordinator = new WorkerCoordinator(
140                       new GroupRebalanceConfig(config, GroupRebalanceConfig.ProtocolType.CONNECT),
141                       logContext,
142                       this.client,
143                       metrics,
144                       metricGrpPrefix,
145                       this.time,
146                       restUrl,
147                       configStorage,
148                       listener,
149                       ConnectProtocolCompatibility.compatibility(config.getString(DistributedConfig.CONNECT_PROTOCOL_CONFIG)),
150                       config.getInt(DistributedConfig.SCHEDULED_REBALANCE_MAX_DELAY_MS_CONFIG));
151   
152               AppInfoParser.registerAppInfo(JMX_PREFIX, clientId, metrics, time.milliseconds());
153               log.debug("Connect group member created");
154           } catch (Throwable t) {
155               // call close methods if internal objects are already constructed
156               // this is to prevent resource leak. see KAFKA-2121
157               stop(true);
158               // now propagate the exception
159               throw new KafkaException("Failed to construct kafka consumer", t);
160           }
161       }
162   
163       public void stop() {
164           if (stopped) return;
165           stop(false);
166       }
167   
168       /**
169        * Ensure that the connection to the broker coordinator is up and that the worker is an
170        * active member of the group.
171        */
172       public void ensureActive() {
173           coordinator.poll(0);
174       }
175   
176       public void poll(long timeout) {
177           if (timeout < 0)
178               throw new IllegalArgumentException("Timeout must not be negative");
179           coordinator.poll(timeout);
180       }
181   
182       /**
183        * Interrupt any running poll() calls, causing a WakeupException to be thrown in the thread invoking that method.
184        */
185       public void wakeup() {
186           this.client.wakeup();
187       }

190        * Get the member ID of this worker in the group of workers.
191        *
192        * This ID is the unique member ID automatically generated.
193        *
194        * @return the member ID
195        */
196       public String memberId() {
197           return coordinator.memberId();
198       }

201           coordinator.requestRejoin("connect worker requested rejoin");
202       }

205           coordinator.maybeLeaveGroup(leaveReason);
206       }

209           return coordinator.ownerUrl(connector);
210       }

214       }

216       /**
217        * Get the version of the connect protocol that is currently active in the group of workers.
218        *
219        * @return the current connect protocol version
220        */
221       public short currentProtocolVersion() {
222           return coordinator.currentProtocolVersion();
223       }
224   
225       public void revokeAssignment(ExtendedAssignment assignment) {
226           coordinator.revokeAssignment(assignment);
227       }

229       private void stop(boolean swallowException) {
230           log.trace("Stopping the Connect group member.");
231           AtomicReference<Throwable> firstException = new AtomicReference<>();
232           this.stopped = true;
233           Utils.closeQuietly(coordinator, "coordinator", firstException);
234           Utils.closeQuietly(metrics, "consumer metrics", firstException);
235           Utils.closeQuietly(client, "consumer network client", firstException);
236           AppInfoParser.unregisterAppInfo(JMX_PREFIX, clientId, metrics);
237           if (firstException.get() != null && !swallowException)
238               throw new KafkaException("Failed to stop the Connect group member", firstException.get());
239           else
240               log.debug("The Connect group member has stopped.");
241       }
242   
243       // Visible for testing
244       Metrics metrics() {
245           return this.metrics;
246       }
