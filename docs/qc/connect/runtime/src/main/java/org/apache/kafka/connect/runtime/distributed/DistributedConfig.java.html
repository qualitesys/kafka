285                       INTER_WORKER_VERIFICATION_ALGORITHMS_CONFIG

577               log.warn("Ignoring '{}{}={}' setting, since config topic partitions is always 1", prefix, PARTITIONS_SUFFIX, result.get("partitions"));

581               log.warn("Ignoring '{}cleanup.policy={}' setting, since compaction is always used", prefix, removedPolicy);

583           result.remove(TOPIC_SUFFIX);
584           result.remove(REPLICATION_FACTOR_SUFFIX);

586           return result;

618       private static void validateVerificationAlgorithms(String configName, List<String> algorithms) {
619           if (algorithms.isEmpty()) {

631               }
632           }
633       }

659       // Visible for testing
660       static Set<String> supportedAlgorithms(String type) {
661           Set<String> result = new HashSet<>();
662           for (Provider provider : Security.getProviders()) {
663               for (Provider.Service service : provider.getServices()) {
664                   if (type.equals(service.getType())) {
665                       result.add(service.getAlgorithm());
666                   }
667               }
668           }
669           return result;
670       }
