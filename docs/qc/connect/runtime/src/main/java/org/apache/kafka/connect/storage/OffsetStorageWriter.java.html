17   package org.apache.kafka.connect.storage;
18   
19   import org.apache.kafka.connect.errors.ConnectException;
20   import org.apache.kafka.connect.util.Callback;
21   import org.slf4j.Logger;
22   import org.slf4j.LoggerFactory;
23   
24   import java.nio.ByteBuffer;
25   import java.util.Arrays;
26   import java.util.HashMap;
27   import java.util.Map;
28   import java.util.concurrent.Future;
29   
30   /**
31    * <p>
32    * OffsetStorageWriter is a buffered writer that wraps the simple OffsetBackingStore interface.
33    * It maintains a copy of the key-value data in memory and buffers writes. It allows you to take
34    * a snapshot, which can then be asynchronously flushed to the backing store while new writes
35    * continue to be processed. This allows Kafka Connect to process offset commits in the background
36    * while continuing to process messages.
37    * </p>
38    * <p>
39    * Connect uses an OffsetStorage implementation to save state about the current progress of
40    * source (import to Kafka) jobs, which may have many input partitions and "offsets" may not be as
41    * simple as they are for Kafka partitions or files. Offset storage is not required for sink jobs
42    * because they can use Kafka's native offset storage (or the sink data store can handle offset
43    * storage to achieve exactly once semantics).
44    * </p>
45    * <p>
46    * Both partitions and offsets are generic data objects. This allows different connectors to use
47    * whatever representation they need, even arbitrarily complex records. These are translated
48    * internally into the serialized form the OffsetBackingStore uses.
49    * </p>
50    * <p>
51    * Note that this only provides write functionality. This is intentional to ensure stale data is
52    * never read. Offset data should only be read during startup or reconfiguration of a task. By
53    * always serving those requests by reading the values from the backing store, we ensure we never
54    * accidentally use stale data. (One example of how this can occur: a task is processing input
55    * partition A, writing offsets; reconfiguration causes partition A to be reassigned elsewhere;
56    * reconfiguration causes partition A to be reassigned to this node, but now the offset data is out
57    * of date). Since these offsets are created and managed by the connector itself, there's no way
58    * for the offset management layer to know which keys are "owned" by which tasks at any given
59    * time.
60    * </p>
61    * <p>
62    * This class is thread-safe.
63    * </p>
64    */
65   public class OffsetStorageWriter {
66       private static final Logger log = LoggerFactory.getLogger(OffsetStorageWriter.class);
67   
68       private final OffsetBackingStore backingStore;
69       private final Converter keyConverter;
70       private final Converter valueConverter;
71       private final String namespace;
72       // Offset data in Connect format
73       private Map<Map<String, Object>, Map<String, Object>> data = new HashMap<>();
74   
75       private Map<Map<String, Object>, Map<String, Object>> toFlush = null;
76       // Unique ID for each flush request to handle callbacks after timeouts
77       private long currentFlushId = 0;
78   
79       public OffsetStorageWriter(OffsetBackingStore backingStore,
80                                  String namespace, Converter keyConverter, Converter valueConverter) {
81           this.backingStore = backingStore;
82           this.namespace = namespace;
83           this.keyConverter = keyConverter;
84           this.valueConverter = valueConverter;
85       }
86   
87       /**
88        * Set an offset for a partition using Connect data values
89        * @param partition the partition to store an offset for
90        * @param offset the offset
91        */
92       @SuppressWarnings("unchecked")
93       public synchronized void offset(Map<String, ?> partition, Map<String, ?> offset) {
94           data.put((Map<String, Object>) partition, (Map<String, Object>) offset);
95       }
96   
97       private boolean flushing() {
98           return toFlush != null;
99       }
100   
101       /**
102        * Performs the first step of a flush operation, snapshotting the current state. This does not
103        * actually initiate the flush with the underlying storage.
104        *
105        * @return true if a flush was initiated, false if no data was available
106        */
107       public synchronized boolean beginFlush() {
108           if (flushing()) {
109               log.error("Invalid call to OffsetStorageWriter flush() while already flushing, the "
110                       + "framework should not allow this");
111               throw new ConnectException("OffsetStorageWriter is already flushing");
112           }
113   
114           if (data.isEmpty())
115               return false;
116   
117           toFlush = data;
118           data = new HashMap<>();
119           return true;
120       }
121   
122       /**
123        * @return whether there's anything to flush right now.
124        */
125       public synchronized boolean willFlush() {
126           return !data.isEmpty();
127       }
128   
129       /**
130        * Flush the current offsets and clear them from this writer. This is non-blocking: it
131        * moves the current set of offsets out of the way, serializes the data, and asynchronously
132        * writes the data to the backing store. If no offsets need to be written, the callback is
133        * still invoked, but no Future is returned.
134        *
135        * @return a Future, or null if there are no offsets to commitOffsets
136        */
137       public Future<Void> doFlush(final Callback<Void> callback) {
138   
139           final long flushId;
140           // Serialize
141           final Map<ByteBuffer, ByteBuffer> offsetsSerialized;
142   
143           synchronized (this) {
144               flushId = currentFlushId;
145   
146               try {
147                   offsetsSerialized = new HashMap<>(toFlush.size());
148                   for (Map.Entry<Map<String, Object>, Map<String, Object>> entry : toFlush.entrySet()) {
149                       // Offsets are specified as schemaless to the converter, using whatever internal schema is appropriate
150                       // for that data. The only enforcement of the format is here.
151                       OffsetUtils.validateFormat(entry.getKey());
152                       OffsetUtils.validateFormat(entry.getValue());
153                       // When serializing the key, we add in the namespace information so the key is [namespace, real key]
154                       byte[] key = keyConverter.fromConnectData(namespace, null, Arrays.asList(namespace, entry.getKey()));
155                       ByteBuffer keyBuffer = (key != null) ? ByteBuffer.wrap(key) : null;
156                       byte[] value = valueConverter.fromConnectData(namespace, null, entry.getValue());
157                       ByteBuffer valueBuffer = (value != null) ? ByteBuffer.wrap(value) : null;
158                       offsetsSerialized.put(keyBuffer, valueBuffer);
159                   }
160               } catch (Throwable t) {
161                   // Must handle errors properly here or the writer will be left mid-flush forever and be
162                   // unable to make progress.
163                   log.error("CRITICAL: Failed to serialize offset data, making it impossible to commit "
164                           + "offsets under namespace {}. This likely won't recover unless the "
165                           + "unserializable partition or offset information is overwritten.", namespace);
166                   log.error("Cause of serialization failure:", t);
167                   callback.onCompletion(t, null);
168                   return null;
169               }
170   
171               // And submit the data
172               log.debug("Submitting {} entries to backing store. The offsets are: {}", offsetsSerialized.size(), toFlush);
173           }
174   
175           return backingStore.set(offsetsSerialized, (error, result) -> {
176               boolean isCurrent = handleFinishWrite(flushId, error, result);
177               if (isCurrent && callback != null) {
178                   callback.onCompletion(error, result);
179               }
180           });
181       }
182   
183       /**
184        * Cancel a flush that has been initiated by {@link #beginFlush}. This should not be called if
185        * {@link #doFlush} has already been invoked. It should be used if an operation performed
186        * between beginFlush and doFlush failed.
187        */
188       public synchronized void cancelFlush() {
189           // Verify we're still flushing data to handle a race between cancelFlush() calls from up the
190           // call stack and callbacks from the write request to underlying storage
191           if (flushing()) {
192               // Just recombine the data and place it back in the primary storage
193               toFlush.putAll(data);
194               data = toFlush;
195               currentFlushId++;
196               toFlush = null;
197           }
198       }
199   
200       /**
201        * Handle completion of a write. Returns true if this callback is for the current flush
202        * operation, false if it's for an old operation that should now be ignored.
203        */
204       private synchronized boolean handleFinishWrite(long flushId, Throwable error, Void result) {
205           // Callbacks need to be handled carefully since the flush operation may have already timed
206           // out and been cancelled.
207           if (flushId != currentFlushId)
208               return false;
209   
210           if (error != null) {
211               cancelFlush();
212           } else {
213               currentFlushId++;
214               toFlush = null;
215           }
216           return true;
217       }
218   }
