46   /**
47    * Write the original consumed record into a dead letter queue. The dead letter queue is a Kafka topic located
48    * on the same cluster used by the worker to maintain internal topics. Each connector is typically configured
49    * with its own Kafka topic dead letter queue. By default, the topic name is not set, and if the
50    * connector config doesn't specify one, this feature is disabled.
51    */
52   public class DeadLetterQueueReporter implements ErrorReporter {
53   
54       private static final Logger log = LoggerFactory.getLogger(DeadLetterQueueReporter.class);
55   
56       private static final int DLQ_NUM_DESIRED_PARTITIONS = 1;
57   
58       public static final String HEADER_PREFIX = "__connect.errors.";
59       public static final String ERROR_HEADER_ORIG_TOPIC = HEADER_PREFIX + "topic";
60       public static final String ERROR_HEADER_ORIG_PARTITION = HEADER_PREFIX + "partition";
61       public static final String ERROR_HEADER_ORIG_OFFSET = HEADER_PREFIX + "offset";
62       public static final String ERROR_HEADER_CONNECTOR_NAME = HEADER_PREFIX + "connector.name";
63       public static final String ERROR_HEADER_TASK_ID = HEADER_PREFIX + "task.id";
64       public static final String ERROR_HEADER_STAGE = HEADER_PREFIX + "stage";
65       public static final String ERROR_HEADER_EXECUTING_CLASS = HEADER_PREFIX + "class.name";
66       public static final String ERROR_HEADER_EXCEPTION = HEADER_PREFIX + "exception.class.name";
67       public static final String ERROR_HEADER_EXCEPTION_MESSAGE = HEADER_PREFIX + "exception.message";
68       public static final String ERROR_HEADER_EXCEPTION_STACK_TRACE = HEADER_PREFIX + "exception.stacktrace";
69   
70       private final SinkConnectorConfig connConfig;
71       private final ConnectorTaskId connectorTaskId;
72       private final ErrorHandlingMetrics errorHandlingMetrics;
73       private final String dlqTopicName;
74   
75       private final KafkaProducer<byte[], byte[]> kafkaProducer;
76   
77       public static DeadLetterQueueReporter createAndSetup(Map<String, Object> adminProps,
78                                                            ConnectorTaskId id,
79                                                            SinkConnectorConfig sinkConfig, Map<String, Object> producerProps,
80                                                            ErrorHandlingMetrics errorHandlingMetrics) {
81           String topic = sinkConfig.dlqTopicName();
82   
83           try (Admin admin = Admin.create(adminProps)) {
84               if (!admin.listTopics().names().get().contains(topic)) {
85                   log.error("Topic {} doesn't exist. Will attempt to create topic.", topic);
86                   NewTopic schemaTopicRequest = new NewTopic(topic, DLQ_NUM_DESIRED_PARTITIONS, sinkConfig.dlqTopicReplicationFactor());
87                   admin.createTopics(singleton(schemaTopicRequest)).all().get();
88               }
89           } catch (InterruptedException e) {
90               throw new ConnectException("Could not initialize dead letter queue with topic=" + topic, e);
91           } catch (ExecutionException e) {
92               if (!(e.getCause() instanceof TopicExistsException)) {
93                   throw new ConnectException("Could not initialize dead letter queue with topic=" + topic, e);
94               }
95           }
96   
97           KafkaProducer<byte[], byte[]> dlqProducer = new KafkaProducer<>(producerProps);
98           return new DeadLetterQueueReporter(dlqProducer, sinkConfig, id, errorHandlingMetrics);
99       }
100   
101       /**
102        * Initialize the dead letter queue reporter with a {@link KafkaProducer}.
103        *
104        * @param kafkaProducer a Kafka Producer to produce the original consumed records.
105        */
106       // Visible for testing
107       DeadLetterQueueReporter(KafkaProducer<byte[], byte[]> kafkaProducer, SinkConnectorConfig connConfig,
108                               ConnectorTaskId id, ErrorHandlingMetrics errorHandlingMetrics) {
109           Objects.requireNonNull(kafkaProducer);
110           Objects.requireNonNull(connConfig);
111           Objects.requireNonNull(id);
112           Objects.requireNonNull(errorHandlingMetrics);
113   
114           this.kafkaProducer = kafkaProducer;
115           this.connConfig = connConfig;
116           this.connectorTaskId = id;
117           this.errorHandlingMetrics = errorHandlingMetrics;
118           this.dlqTopicName = connConfig.dlqTopicName().trim();
119       }
120   
121       /**
122        * Write the raw records into a Kafka topic and return the producer future.
123        *
124        * @param context processing context containing the raw record at {@link ProcessingContext#consumerRecord()}.
125        * @return the future associated with the writing of this record; never null
126        */
127       public Future<RecordMetadata> report(ProcessingContext context) {
128           if (dlqTopicName.isEmpty()) {
129               return CompletableFuture.completedFuture(null);
130           }
131           errorHandlingMetrics.recordDeadLetterQueueProduceRequest();
132   
133           ConsumerRecord<byte[], byte[]> originalMessage = context.consumerRecord();
134           if (originalMessage == null) {
135               errorHandlingMetrics.recordDeadLetterQueueProduceFailed();
136               return CompletableFuture.completedFuture(null);
137           }
138   
139           ProducerRecord<byte[], byte[]> producerRecord;
140           if (originalMessage.timestamp() == RecordBatch.NO_TIMESTAMP) {
141               producerRecord = new ProducerRecord<>(dlqTopicName, null,
142                       originalMessage.key(), originalMessage.value(), originalMessage.headers());
143           } else {
144               producerRecord = new ProducerRecord<>(dlqTopicName, null, originalMessage.timestamp(),
145                       originalMessage.key(), originalMessage.value(), originalMessage.headers());
146           }
147   
148           if (connConfig.isDlqContextHeadersEnabled()) {
149               populateContextHeaders(producerRecord, context);
150           }
151   
152           return this.kafkaProducer.send(producerRecord, (metadata, exception) -> {
153               if (exception != null) {
154                   log.error("Could not produce message to dead letter queue. topic=" + dlqTopicName, exception);
155                   errorHandlingMetrics.recordDeadLetterQueueProduceFailed();
156               }
157           });
158       }
159   
160       // Visible for testing
161       void populateContextHeaders(ProducerRecord<byte[], byte[]> producerRecord, ProcessingContext context) {
162           Headers headers = producerRecord.headers();
163           if (context.consumerRecord() != null) {
164               headers.add(ERROR_HEADER_ORIG_TOPIC, toBytes(context.consumerRecord().topic()));
165               headers.add(ERROR_HEADER_ORIG_PARTITION, toBytes(context.consumerRecord().partition()));
166               headers.add(ERROR_HEADER_ORIG_OFFSET, toBytes(context.consumerRecord().offset()));
167           }
168   
169           headers.add(ERROR_HEADER_CONNECTOR_NAME, toBytes(connectorTaskId.connector()));
170           headers.add(ERROR_HEADER_TASK_ID, toBytes(String.valueOf(connectorTaskId.task())));
171           headers.add(ERROR_HEADER_STAGE, toBytes(context.stage().name()));
172           headers.add(ERROR_HEADER_EXECUTING_CLASS, toBytes(context.executingClass().getName()));
173           if (context.error() != null) {
174               headers.add(ERROR_HEADER_EXCEPTION, toBytes(context.error().getClass().getName()));
175               headers.add(ERROR_HEADER_EXCEPTION_MESSAGE, toBytes(context.error().getMessage()));
176               byte[] trace;
177               if ((trace = stacktrace(context.error())) != null) {
178                   headers.add(ERROR_HEADER_EXCEPTION_STACK_TRACE, trace);
179               }
180           }
181       }
182   
183       private byte[] stacktrace(Throwable error) {
184           ByteArrayOutputStream bos = new ByteArrayOutputStream();
185           try {
186               PrintStream stream = new PrintStream(bos, true, StandardCharsets.UTF_8.name());
187               error.printStackTrace(stream);
188               bos.close();
189               return bos.toByteArray();
190           } catch (IOException e) {
191               log.error("Could not serialize stacktrace.", e);
192           }
193           return null;
194       }
195   
196       private byte[] toBytes(int value) {
197           return toBytes(String.valueOf(value));
198       }
199   
200       private byte[] toBytes(long value) {
201           return toBytes(String.valueOf(value));
202       }
203   
204       private byte[] toBytes(String value) {
205           if (value != null) {
206               return value.getBytes(StandardCharsets.UTF_8);
207           } else {
208               return null;
209           }
210       }

213       public void close() {
214           kafkaProducer.close();
215       }
