67   /**
68    * A custom classloader dedicated to loading Connect plugin classes in classloading isolation.
69    *
70    * <p>
71    * Under the current scheme for classloading isolation in Connect, the delegating classloader loads
72    * plugin classes that it finds in its child plugin classloaders. For classes that are not plugins,
73    * this delegating classloader delegates its loading to its parent. This makes this classloader a
74    * child-first classloader.
75    * <p>
76    * This class is thread-safe and parallel capable.
77    */
78   public class DelegatingClassLoader extends URLClassLoader {
79       private static final Logger log = LoggerFactory.getLogger(DelegatingClassLoader.class);
80       private static final String CLASSPATH_NAME = "classpath";

177       /**
178        * Retrieve the PluginClassLoader associated with a plugin class
179        * @param name The fully qualified class name of the plugin
180        * @return the PluginClassLoader that should be used to load this, or null if the plugin is not isolated.
181        */
182       public PluginClassLoader pluginClassLoader(String name) {
183           if (!PluginUtils.shouldLoadInIsolation(name)) {
184               return null;
185           }
186           SortedMap<PluginDesc<?>, ClassLoader> inner = pluginLoaders.get(name);
187           if (inner == null) {
188               return null;
189           }
190           ClassLoader pluginLoader = inner.get(inner.lastKey());
191           return pluginLoader instanceof PluginClassLoader
192                  ? (PluginClassLoader) pluginLoader
193                  : null;
194       }

222       private <T> void addPlugins(Collection<PluginDesc<T>> plugins, ClassLoader loader) {
223           for (PluginDesc<T> plugin : plugins) {
224               String pluginClassName = plugin.className();
225               SortedMap<PluginDesc<?>, ClassLoader> inner = pluginLoaders.get(pluginClassName);
226               if (inner == null) {
227                   inner = new TreeMap<>();
228                   pluginLoaders.put(pluginClassName, inner);
229                   // TODO: once versioning is enabled this line should be moved outside this if branch
230                   log.info("Added plugin '{}'", pluginClassName);
231               }
232               inner.put(plugin, loader);
233           }
234       }

245       private void initPluginLoader(String path) {
246           try {
247               if (CLASSPATH_NAME.equals(path)) {
248                   scanUrlsAndAddPlugins(
249                           getParent(),
250                           ClasspathHelper.forJavaClassPath().toArray(new URL[0])
251                   );
252               } else {
253                   Path pluginPath = Paths.get(path).toAbsolutePath();
254                   // Update for exception handling
255                   path = pluginPath.toString();
256                   // Currently 'plugin.paths' property is a list of top-level directories
257                   // containing plugins
258                   if (Files.isDirectory(pluginPath)) {
259                       for (Path pluginLocation : PluginUtils.pluginLocations(pluginPath)) {
260                           registerPlugin(pluginLocation);
261                       }
262                   } else if (PluginUtils.isArchive(pluginPath)) {
263                       registerPlugin(pluginPath);
264                   }
265               }
266           } catch (InvalidPathException | MalformedURLException e) {
267               log.error("Invalid path in plugin path: {}. Ignoring.", path, e);
268           } catch (IOException e) {
269               log.error("Could not get listing for plugin path: {}. Ignoring.", path, e);
270           } catch (ReflectiveOperationException e) {
271               log.error("Could not instantiate plugins in: {}. Ignoring.", path, e);
272           }
273       }

275       private void registerPlugin(Path pluginLocation)
276           throws IOException, ReflectiveOperationException {
277           log.info("Loading plugin from: {}", pluginLocation);
278           List<URL> pluginUrls = new ArrayList<>();
279           for (Path path : PluginUtils.pluginUrls(pluginLocation)) {
280               pluginUrls.add(path.toUri().toURL());
281           }
282           URL[] urls = pluginUrls.toArray(new URL[0]);
283           if (log.isDebugEnabled()) {
284               log.debug("Loading plugin urls: {}", Arrays.toString(urls));
285           }
286           PluginClassLoader loader = newPluginClassLoader(
287                   pluginLocation.toUri().toURL(),
288                   urls,
289                   this
290           );
291           scanUrlsAndAddPlugins(loader, urls);
292       }

294       private void scanUrlsAndAddPlugins(
295               ClassLoader loader,
296               URL[] urls
297       ) throws ReflectiveOperationException {
298           PluginScanResult plugins = scanPluginPath(loader, urls);
299           log.info("Registered loader: {}", loader);
300           if (!plugins.isEmpty()) {
301               addPlugins(plugins.sinkConnectors(), loader);
302               sinkConnectors.addAll(plugins.sinkConnectors());
303               addPlugins(plugins.sourceConnectors(), loader);
304               sourceConnectors.addAll(plugins.sourceConnectors());
305               addPlugins(plugins.converters(), loader);
306               converters.addAll(plugins.converters());
307               addPlugins(plugins.headerConverters(), loader);
308               headerConverters.addAll(plugins.headerConverters());
309               addPlugins(plugins.transformations(), loader);
310               transformations.addAll(plugins.transformations());
311               addPlugins(plugins.predicates(), loader);
312               predicates.addAll(plugins.predicates());
313               addPlugins(plugins.configProviders(), loader);
314               configProviders.addAll(plugins.configProviders());
315               addPlugins(plugins.restExtensions(), loader);
316               restExtensions.addAll(plugins.restExtensions());
317               addPlugins(plugins.connectorClientConfigPolicies(), loader);
318               connectorClientConfigPolicies.addAll(plugins.connectorClientConfigPolicies());
319           }

321           loadJdbcDrivers(loader);
322       }

324       private void loadJdbcDrivers(final ClassLoader loader) {
325           // Apply here what java.sql.DriverManager does to discover and register classes
326           // implementing the java.sql.Driver interface.
327           AccessController.doPrivileged(
328               (PrivilegedAction<Void>) () -> {
329                   ServiceLoader<Driver> loadedDrivers = ServiceLoader.load(
330                           Driver.class,
331                           loader
332                   );
333                   Iterator<Driver> driversIterator = loadedDrivers.iterator();
334                   try {
335                       while (driversIterator.hasNext()) {
336                           Driver driver = driversIterator.next();
337                           log.debug(
338                                   "Registered java.sql.Driver: {} to java.sql.DriverManager",
339                                   driver
340                           );
341                       }
342                   } catch (Throwable t) {
343                       log.debug(
344                               "Ignoring java.sql.Driver classes listed in resources but not"
345                                       + " present in class loader's classpath: ",
346                               t
347                       );
348                   }
349                   return null;
350               }
351           );
352       }

354       private PluginScanResult scanPluginPath(
355               ClassLoader loader,
356               URL[] urls
357       ) throws ReflectiveOperationException {
358           ConfigurationBuilder builder = new ConfigurationBuilder();
359           builder.setClassLoaders(new ClassLoader[]{loader});
360           builder.addUrls(urls);
361           builder.setScanners(new SubTypesScanner());
362           builder.useParallelExecutor();
363           Reflections reflections = new InternalReflections(builder);
364   
365           return new PluginScanResult(
366                   getPluginDesc(reflections, SinkConnector.class, loader),
367                   getPluginDesc(reflections, SourceConnector.class, loader),
368                   getPluginDesc(reflections, Converter.class, loader),
369                   getPluginDesc(reflections, HeaderConverter.class, loader),
370                   getTransformationPluginDesc(loader, reflections),
371                   getPredicatePluginDesc(loader, reflections),
372                   getServiceLoaderPluginDesc(ConfigProvider.class, loader),
373                   getServiceLoaderPluginDesc(ConnectRestExtension.class, loader),
374                   getServiceLoaderPluginDesc(ConnectorClientConfigOverridePolicy.class, loader)
375           );
376       }

388       private <T> Collection<PluginDesc<T>> getPluginDesc(
389               Reflections reflections,
390               Class<T> klass,
391               ClassLoader loader
392       ) throws ReflectiveOperationException {
393           Set<Class<? extends T>> plugins;
394           try {
395               plugins = reflections.getSubTypesOf(klass);
396           } catch (ReflectionsException e) {
397               log.debug("Reflections scanner could not find any classes for URLs: " +
398                       reflections.getConfiguration().getUrls(), e);
399               return Collections.emptyList();
400           }
401   
402           Collection<PluginDesc<T>> result = new ArrayList<>();
403           for (Class<? extends T> plugin : plugins) {
404               if (PluginUtils.isConcrete(plugin)) {
405                   result.add(pluginDesc(plugin, versionFor(plugin), loader));
406               } else {
407                   log.debug("Skipping {} as it is not concrete implementation", plugin);
408               }
409           }
410           return result;
411       }

418       @SuppressWarnings("unchecked")
419       private <T> Collection<PluginDesc<T>> getServiceLoaderPluginDesc(Class<T> klass, ClassLoader loader) {
420           ClassLoader savedLoader = Plugins.compareAndSwapLoaders(loader);
421           Collection<PluginDesc<T>> result = new ArrayList<>();
422           try {
423               ServiceLoader<T> serviceLoader = ServiceLoader.load(klass, loader);
424               for (T pluginImpl : serviceLoader) {
425                   result.add(pluginDesc((Class<? extends T>) pluginImpl.getClass(),
426                       versionFor(pluginImpl), loader));
427               }
428           } finally {
429               Plugins.compareAndSwapLoaders(savedLoader);
430           }
431           return result;
432       }

466       private <S> void addAliases(Collection<PluginDesc<S>> plugins) {
467           for (PluginDesc<S> plugin : plugins) {
468               if (PluginUtils.isAliasUnique(plugin, plugins)) {
469                   String simple = PluginUtils.simpleName(plugin);
470                   String pruned = PluginUtils.prunedName(plugin);
471                   aliases.put(simple, plugin.className());
472                   if (simple.equals(pruned)) {

481                       );
482                   }

504               }
505           }
