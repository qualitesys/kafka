43   /**
44    * A set of endpoints to adjust the log levels of runtime loggers.
45    */
46   @Path("/admin/loggers")
47   @Produces(MediaType.APPLICATION_JSON)
48   @Consumes(MediaType.APPLICATION_JSON)
49   public class LoggingResource {
50   
51       /**
52        * Log4j uses "root" (case insensitive) as name of the root logger.
53        */
54       private static final String ROOT_LOGGER_NAME = "root";
55   
56       /**
57        * List the current loggers that have their levels explicitly set and their log levels.
58        *
59        * @return a list of current loggers and their levels.
60        */
61       @GET
62       @Path("/")
63       @Operation(summary = "List the current loggers that have their levels explicitly set and their log levels")
64       public Response listLoggers() {
65           Map<String, Map<String, String>> loggers = new TreeMap<>();
66           Enumeration<Logger> enumeration = currentLoggers();
67           Collections.list(enumeration)
68                   .stream()
69                   .filter(logger -> logger.getLevel() != null)
70                   .forEach(logger -> loggers.put(logger.getName(), levelToMap(logger)));
71   
72           Logger root = rootLogger();
73           if (root.getLevel() != null) {
74               loggers.put(ROOT_LOGGER_NAME, levelToMap(root));
75           }
76   
77           return Response.ok(loggers).build();
78       }
79   
80       /**
81        * Get the log level of a named logger.
82        *
83        * @param namedLogger name of a logger
84        * @return level of the logger, effective level if the level was not explicitly set.
85        */
86       @GET
87       @Path("/{logger}")
88       @Operation(summary = "Get the log level for the specified logger")
89       public Response getLogger(final @PathParam("logger") String namedLogger) {
90           Objects.requireNonNull(namedLogger, "require non-null name");
91   
92           Logger logger = null;
93           if (ROOT_LOGGER_NAME.equalsIgnoreCase(namedLogger)) {
94               logger = rootLogger();
95           } else {
96               Enumeration<Logger> en = currentLoggers();
97               // search within existing loggers for the given name.
98               // using LogManger.getLogger() will create a logger if it doesn't exist
99               // (potential leak since these don't get cleaned up).
100               while (en.hasMoreElements()) {
101                   Logger l = en.nextElement();
102                   if (namedLogger.equals(l.getName())) {
103                       logger = l;
104                       break;
105                   }
106               }
107           }
108           if (logger == null) {
109               throw new NotFoundException("Logger " + namedLogger + " not found.");
110           } else {
111               return Response.ok(effectiveLevelToMap(logger)).build();
112           }
113       }
114   
115   
116       /**
117        * Adjust level of a named logger. if name corresponds to an ancestor, then the log level is applied to all child loggers.
118        *
119        * @param namedLogger name of the logger
120        * @param levelMap a map that is expected to contain one key 'level', and a value that is one of the log4j levels:
121        *                 DEBUG, ERROR, FATAL, INFO, TRACE, WARN
122        * @return names of loggers whose levels were modified
123        */
124       @PUT
125       @Path("/{logger}")
126       @Operation(summary = "Set the level for the specified logger")
127       public Response setLevel(final @PathParam("logger") String namedLogger,
128                                final Map<String, String> levelMap) {
129           String desiredLevelStr = levelMap.get("level");
130           if (desiredLevelStr == null) {
131               throw new BadRequestException("Desired 'level' parameter was not specified in request.");
132           }
133   
134           Level level = Level.toLevel(desiredLevelStr.toUpperCase(Locale.ROOT), null);
135           if (level == null) {
136               throw new NotFoundException("invalid log level '" + desiredLevelStr + "'.");
137           }
138   
139           List<Logger> childLoggers;
140           if (ROOT_LOGGER_NAME.equalsIgnoreCase(namedLogger)) {
141               childLoggers = Collections.list(currentLoggers());
142               childLoggers.add(rootLogger());
143           } else {
144               childLoggers = new ArrayList<>();
145               Logger ancestorLogger = lookupLogger(namedLogger);
146               Enumeration<Logger> en = currentLoggers();
147               boolean present = false;
148               while (en.hasMoreElements()) {
149                   Logger current = en.nextElement();
150                   if (current.getName().startsWith(namedLogger)) {
151                       childLoggers.add(current);
152                   }
153                   if (namedLogger.equals(current.getName())) {
154                       present = true;
155                   }
156               }
157               if (!present) {
158                   childLoggers.add(ancestorLogger);
159               }
160           }
161   
162           List<String> modifiedLoggerNames = new ArrayList<>();
163           for (Logger logger: childLoggers) {
164               logger.setLevel(level);
165               modifiedLoggerNames.add(logger.getName());
166           }
167           Collections.sort(modifiedLoggerNames);
168   
169           return Response.ok(modifiedLoggerNames).build();
170       }
171   
172       protected Logger lookupLogger(String namedLogger) {
173           return LogManager.getLogger(namedLogger);
174       }
175   
176       @SuppressWarnings("unchecked")
177       protected Enumeration<Logger> currentLoggers() {
178           return LogManager.getCurrentLoggers();
179       }
180   
181       protected Logger rootLogger() {
182           return LogManager.getRootLogger();
183       }
184   
185       /**
186        *
187        * Map representation of a logger's effective log level.
188        *
189        * @param logger a non-null log4j logger
190        * @return a singleton map whose key is level and the value is the string representation of the logger's effective log level.
191        */
192       private static Map<String, String> effectiveLevelToMap(Logger logger) {
193           Level level = logger.getLevel();
194           if (level == null) {
195               level = logger.getEffectiveLevel();
196           }
197           return Collections.singletonMap("level", String.valueOf(level));
198       }
199   
200       /**
201        *
202        * Map representation of a logger's log level.
203        *
204        * @param logger a non-null log4j logger
205        * @return a singleton map whose key is level and the value is the string representation of the logger's log level.
206        */
207       private static Map<String, String> levelToMap(Logger logger) {
208           return Collections.singletonMap("level", String.valueOf(logger.getLevel()));
209       }
210   }
