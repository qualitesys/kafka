17   package org.apache.kafka.connect.tools;

19   import com.fasterxml.jackson.core.JsonProcessingException;
20   import com.fasterxml.jackson.databind.ObjectMapper;
21   import org.apache.kafka.tools.ThroughputThrottler;
22   import org.apache.kafka.clients.producer.RecordMetadata;
23   import org.apache.kafka.connect.data.Schema;
24   import org.apache.kafka.connect.errors.ConnectException;
25   import org.apache.kafka.connect.source.SourceRecord;
26   import org.apache.kafka.connect.source.SourceTask;
27   import org.slf4j.Logger;
28   import org.slf4j.LoggerFactory;
29   
30   import java.util.Collections;
31   import java.util.HashMap;
32   import java.util.List;
33   import java.util.Map;

35   /**
36    * A connector primarily intended for system tests. The connector simply generates as many tasks as requested. The
37    * tasks print metadata in the form of JSON to stdout for each message generated, making externally visible which
38    * messages have been sent. Each message is also assigned a unique, increasing seqno that is passed to Kafka Connect; when
39    * tasks are started on new nodes, this seqno is used to resume where the task previously left off, allowing for
40    * testing of distributed Kafka Connect.
41    *
42    * If logging is left enabled, log output on stdout can be easily ignored by checking whether a given line is valid JSON.
43    */
44   public class VerifiableSourceTask extends SourceTask {
45       private static final Logger log = LoggerFactory.getLogger(VerifiableSourceTask.class);
46   
47       public static final String NAME_CONFIG = "name";
48       public static final String ID_CONFIG = "id";
49       public static final String TOPIC_CONFIG = "topic";
50       public static final String THROUGHPUT_CONFIG = "throughput";
51   
52       private static final String ID_FIELD = "id";
53       private static final String SEQNO_FIELD = "seqno";
54   
55       private static final ObjectMapper JSON_SERDE = new ObjectMapper();
56   
57       private String name; // Connector name
58       private int id; // Task ID
59       private String topic;
60       private Map<String, Integer> partition;
61       private long startingSeqno;
62       private long seqno;
63       private ThroughputThrottler throttler;
64   
65       @Override
66       public String version() {
67           return new VerifiableSourceConnector().version();
68       }
69   
70       @Override
71       public void start(Map<String, String> props) {
72           final long throughput;
73           try {
74               name = props.get(NAME_CONFIG);
75               id = Integer.parseInt(props.get(ID_CONFIG));
76               topic = props.get(TOPIC_CONFIG);
77               throughput = Long.parseLong(props.get(THROUGHPUT_CONFIG));
78           } catch (NumberFormatException e) {
79               throw new ConnectException("Invalid VerifiableSourceTask configuration", e);
80           }
81   
82           partition = Collections.singletonMap(ID_FIELD, id);
83           Map<String, Object> previousOffset = this.context.offsetStorageReader().offset(partition);
84           if (previousOffset != null)
85               seqno = (Long) previousOffset.get(SEQNO_FIELD) + 1;
86           else
87               seqno = 0;
88           startingSeqno = seqno;
89           throttler = new ThroughputThrottler(throughput, System.currentTimeMillis());
90   
91           log.info("Started VerifiableSourceTask {}-{} producing to topic {} resuming from seqno {}", name, id, topic, startingSeqno);
92       }
93   
94       @Override
95       public List<SourceRecord> poll() {
96           long sendStartMs = System.currentTimeMillis();
97           if (throttler.shouldThrottle(seqno - startingSeqno, sendStartMs))
98               throttler.throttle();
99   
100           long nowMs = System.currentTimeMillis();
101   
102           Map<String, Object> data = new HashMap<>();
103           data.put("name", name);
104           data.put("task", id);
105           data.put("topic", this.topic);
106           data.put("time_ms", nowMs);
107           data.put("seqno", seqno);
108           String dataJson;
109           try {
110               dataJson = JSON_SERDE.writeValueAsString(data);
111           } catch (JsonProcessingException e) {
112               dataJson = "Bad data can't be written as json: " + e.getMessage();
113           }
114           System.out.println(dataJson);
115   
116           Map<String, Long> ccOffset = Collections.singletonMap(SEQNO_FIELD, seqno);
117           SourceRecord srcRecord = new SourceRecord(partition, ccOffset, topic, Schema.INT32_SCHEMA, id, Schema.INT64_SCHEMA, seqno);
118           List<SourceRecord> result = Collections.singletonList(srcRecord);
119           seqno++;
120           return result;
121       }
122   
123       @Override
124       public void commitRecord(SourceRecord record, RecordMetadata metadata) {
125           Map<String, Object> data = new HashMap<>();
126           data.put("name", name);
127           data.put("task", id);
128           data.put("topic", this.topic);
129           data.put("time_ms", System.currentTimeMillis());
130           data.put("seqno", record.value());
131           data.put("committed", true);
132   
133           String dataJson;
134           try {
135               dataJson = JSON_SERDE.writeValueAsString(data);
136           } catch (JsonProcessingException e) {
137               dataJson = "Bad data can't be written as json: " + e.getMessage();
138           }
139           System.out.println(dataJson);
140       }
141   
142       @Override
143       public void stop() {
144           throttler.wakeup();
145       }
146   }
