145       private static final DirectoryStream.Filter<Path> PLUGIN_PATH_FILTER = path ->
146           Files.isDirectory(path) || isArchive(path) || isClassFile(path);

148       /**
149        * Return whether the class with the given name should be loaded in isolation using a plugin
150        * classloader.
151        *
152        * @param name the fully qualified name of the class.
153        * @return true if this class should be loaded in isolation, false otherwise.
154        */
155       public static boolean shouldLoadInIsolation(String name) {
156           return !(EXCLUDE.matcher(name).matches() && !INCLUDE.matcher(name).matches());
157       }

159       /**
160        * Verify the given class corresponds to a concrete class and not to an abstract class or
161        * interface.
162        * @param klass the class object.
163        * @return true if the argument is a concrete class, false if it's abstract or interface.
164        */
165       public static boolean isConcrete(Class<?> klass) {
166           int mod = klass.getModifiers();
167           return !Modifier.isAbstract(mod) && !Modifier.isInterface(mod);
168       }

191       public static List<Path> pluginLocations(Path topPath) throws IOException {
192           List<Path> locations = new ArrayList<>();
193           try (
194                   DirectoryStream<Path> listing = Files.newDirectoryStream(
195                           topPath,
196                           PLUGIN_PATH_FILTER
197                   )
198           ) {
199               for (Path dir : listing) {
200                   locations.add(dir);
201               }
202           }
203           return locations;
204       }

206       /**
207        * Given a top path in the filesystem, return a list of paths to archives (JAR or ZIP
208        * files) contained under this top path. If the top path contains only java class files,
209        * return the top path itself. This method follows symbolic links to discover archives and
210        * returns the such archives as absolute paths.
211        *
212        * @param topPath the path to use as root of plugin search.
213        * @return a list of potential plugin paths, or empty list if no such paths exist.
214        * @throws IOException
215        */
216       public static List<Path> pluginUrls(Path topPath) throws IOException {
217           boolean containsClassFiles = false;
218           Set<Path> archives = new TreeSet<>();
219           LinkedList<DirectoryEntry> dfs = new LinkedList<>();
220           Set<Path> visited = new HashSet<>();
221   
222           if (isArchive(topPath)) {
223               return Collections.singletonList(topPath);
224           }
225   
226           DirectoryStream<Path> topListing = Files.newDirectoryStream(
227                   topPath,
228                   PLUGIN_PATH_FILTER
229           );
230           dfs.push(new DirectoryEntry(topListing));
231           visited.add(topPath);
232           try {
233               while (!dfs.isEmpty()) {
234                   Iterator<Path> neighbors = dfs.peek().iterator;
235                   if (!neighbors.hasNext()) {
236                       dfs.pop().stream.close();
237                       continue;
238                   }
239   
240                   Path adjacent = neighbors.next();
241                   if (Files.isSymbolicLink(adjacent)) {
242                       try {
243                           Path symlink = Files.readSymbolicLink(adjacent);
244                           // if symlink is absolute resolve() returns the absolute symlink itself
245                           Path parent = adjacent.getParent();
246                           if (parent == null) {
247                               continue;
248                           }
249                           Path absolute = parent.resolve(symlink).toRealPath();
250                           if (Files.exists(absolute)) {
251                               adjacent = absolute;
252                           } else {
253                               continue;
254                           }
255                       } catch (IOException e) {
256                           // See https://issues.apache.org/jira/browse/KAFKA-6288 for a reported
257                           // failure. Such a failure at this stage is not easily reproducible and
258                           // therefore an exception is caught and ignored after issuing a
259                           // warning. This allows class scanning to continue for non-broken plugins.
260                           log.warn(
261                                   "Resolving symbolic link '{}' failed. Ignoring this path.",
262                                   adjacent,
263                                   e
264                           );
265                           continue;
266                       }
267                   }
268   
269                   if (!visited.contains(adjacent)) {
270                       visited.add(adjacent);
271                       if (isArchive(adjacent)) {
272                           archives.add(adjacent);
273                       } else if (isClassFile(adjacent)) {
274                           containsClassFiles = true;
275                       } else {
276                           DirectoryStream<Path> listing = Files.newDirectoryStream(
277                                   adjacent,
278                                   PLUGIN_PATH_FILTER
279                           );
280                           dfs.push(new DirectoryEntry(listing));
281                       }
282                   }
283               }
284           } finally {
285               while (!dfs.isEmpty()) {
286                   dfs.pop().stream.close();
287               }
288           }
289   
290           if (containsClassFiles) {
291               if (archives.isEmpty()) {
292                   return Collections.singletonList(topPath);
293               }
294               log.warn("Plugin path contains both java archives and class files. Returning only the"
295                       + " archives");
296           }
297           return Arrays.asList(archives.toArray(new Path[0]));
298       }

310       /**
311        * Remove the plugin type name at the end of a plugin class name, if such suffix is present.
312        * This method is meant to be used to extract plugin aliases.
313        *
314        * @param plugin the plugin descriptor.
315        * @return the pruned simple class name of the plugin.
316        */
317       public static String prunedName(PluginDesc<?> plugin) {
318           // It's currently simpler to switch on type than do pattern matching.
319           switch (plugin.type()) {
320               case SOURCE:
321               case SINK:
322                   return prunePluginName(plugin, "Connector");
323               default:
324                   return prunePluginName(plugin, plugin.type().simpleName());
325           }
326       }

328       /**
329        * Verify whether a given plugin's alias matches another alias in a collection of plugins.
330        *
331        * @param alias the plugin descriptor to test for alias matching.
332        * @param plugins the collection of plugins to test against.
333        * @param <U> the plugin type.
334        * @return false if a match was found in the collection, otherwise true.
335        */
336       public static <U> boolean isAliasUnique(
337               PluginDesc<U> alias,
338               Collection<PluginDesc<U>> plugins
339       ) {
340           boolean matched = false;
341           for (PluginDesc<U> plugin : plugins) {
342               if (simpleName(alias).equals(simpleName(plugin))
343                       || prunedName(alias).equals(prunedName(plugin))) {
344                   if (matched) {
345                       return false;
346                   }
347                   matched = true;
348               }
349           }
350           return true;
351       }
352   
353       private static String prunePluginName(PluginDesc<?> plugin, String suffix) {
354           String simple = plugin.pluginClass().getSimpleName();
355           int pos = simple.lastIndexOf(suffix);
356           if (pos > 0) {
357               return simple.substring(0, pos);
358           }
359           return simple;
360       }
