46       public static Long checkAndConvertTimestamp(Long timestamp) {
47           if (timestamp == null || timestamp >= 0)
48               return timestamp;
49           else if (timestamp == RecordBatch.NO_TIMESTAMP)
50               return null;
51           else
52               throw new InvalidRecordException(String.format("Invalid record timestamp %d", timestamp));
53       }

62       static String lookupKafkaClusterId(Admin adminClient) {
63           log.debug("Looking up Kafka cluster ID");
64           try {
65               KafkaFuture<String> clusterIdFuture = adminClient.describeCluster().clusterId();
66               if (clusterIdFuture == null) {
67                   log.info("Kafka cluster version is too old to return cluster ID");
68                   return null;
69               }
70               log.debug("Fetching Kafka cluster ID");
71               String kafkaClusterId = clusterIdFuture.get();
72               log.info("Kafka cluster ID: {}", kafkaClusterId);
73               return kafkaClusterId;
74           } catch (InterruptedException e) {
75               throw new ConnectException("Unexpectedly interrupted when looking up Kafka cluster info", e);
76           } catch (ExecutionException e) {
77               throw new ConnectException("Failed to connect to and describe Kafka cluster. "
78                                          + "Check worker's broker connection and security properties.", e);
79           }
80       }

82       /**
83        * Ensure that the {@link Map properties} contain an expected value for the given key, inserting the
84        * expected value into the properties if necessary.
85        *
86        * <p>If there is a pre-existing value for the key in the properties, log a warning to the user
87        * that this value will be ignored, and the expected value will be used instead.
88        *
89        * @param props the configuration properties provided by the user; may not be null
90        * @param key the name of the property to check on; may not be null
91        * @param expectedValue the expected value for the property; may not be null
92        * @param justification the reason the property cannot be overridden.
93        *                      Will follow the phrase "The value... for the... property will be ignored as it cannot be overridden ".
94        *                      For example, one might supply the message "in connectors with the DLQ feature enabled" for this parameter.
95        *                      May be null (in which case, no justification is given to the user in the logged warning message)
96        * @param caseSensitive whether the value should match case-insensitively
97        */
98       public static void ensureProperty(
99               Map<String, ? super String> props,
100               String key,
101               String expectedValue,
102               String justification,
103               boolean caseSensitive
104       ) {
105           ensurePropertyAndGetWarning(props, key, expectedValue, justification, caseSensitive).ifPresent(log::warn);
106       }
107   
108       // Visible for testing
109       /**
110        * Ensure that a given key has an expected value in the properties, inserting the expected value into the
111        * properties if necessary. If a user-supplied value is overridden, return a warning message that can
112        * be logged to the user notifying them of this fact.
113        *
114        * @return an {@link Optional} containing a warning that should be logged to the user if a value they
115        * supplied in the properties is being overridden, or {@link Optional#empty()} if no such override has
116        * taken place
117        */
118       static Optional<String> ensurePropertyAndGetWarning(
119               Map<String, ? super String> props,
120               String key,
121               String expectedValue,
122               String justification,
123               boolean caseSensitive) {
124           if (!props.containsKey(key)) {
125               // Insert the expected value
126               props.put(key, expectedValue);
127               // But don't issue a warning to the user
128               return Optional.empty();
129           }
130   
131           String value = Objects.toString(props.get(key));
132           boolean matchesExpectedValue = caseSensitive ? expectedValue.equals(value) : expectedValue.equalsIgnoreCase(value);
133           if (matchesExpectedValue) {
134               return Optional.empty();
135           }
136   
137           // Insert the expected value
138           props.put(key, expectedValue);
139   
140           justification = justification != null ? " " + justification : "";
141           // And issue a warning to the user
142           return Optional.of(String.format(
143                   "The value '%s' for the '%s' property will be ignored as it cannot be overridden%s. "
144                           + "The value '%s' will be used instead.",
145                   value, key, justification, expectedValue
146           ));
147       }

149       public static void addMetricsContextProperties(Map<String, Object> prop, WorkerConfig config, String clusterId) {
150           //add all properties predefined with "metrics.context."
151           prop.putAll(config.originalsWithPrefix(CommonClientConfigs.METRICS_CONTEXT_PREFIX, false));
152           //add connect properties
153           prop.put(CommonClientConfigs.METRICS_CONTEXT_PREFIX + WorkerConfig.CONNECT_KAFKA_CLUSTER_ID, clusterId);
154           Object groupId = config.originals().get(DistributedConfig.GROUP_ID_CONFIG);
155           if (groupId != null) {
156               prop.put(CommonClientConfigs.METRICS_CONTEXT_PREFIX + WorkerConfig.CONNECT_GROUP_ID, groupId);
157           }
158       }

168       public static <K, I, O> Map<K, O> transformValues(Map<K, I> map, Function<I, O> transformation) {
169           return map.entrySet().stream().collect(Collectors.toMap(
170                   Map.Entry::getKey,
171                   transformation.compose(Map.Entry::getValue)
172           ));
173       }

179       public static <I, T> List<T> combineCollections(Collection<I> collection, Function<I, Collection<T>> extractCollection) {
180           return combineCollections(collection, extractCollection, Collectors.toList());
181       }

183       public static <I, T, C> C combineCollections(
184               Collection<I> collection,
185               Function<I, Collection<T>> extractCollection,
186               Collector<T, ?, C> collector
187       ) {
188           return collection.stream()
189                   .map(extractCollection)
190                   .flatMap(Collection::stream)
191                   .collect(collector);
192       }

194       public static ConnectException maybeWrap(Throwable t, String message) {
195           if (t == null) {
196               return null;
197           }
198           if (t instanceof ConnectException) {
199               return (ConnectException) t;
200           }
201           return new ConnectException(message, t);
202       }
