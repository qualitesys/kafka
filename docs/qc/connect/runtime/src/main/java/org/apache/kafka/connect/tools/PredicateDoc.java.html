17   package org.apache.kafka.connect.tools;
18   
19   import org.apache.kafka.common.config.ConfigDef;
20   import org.apache.kafka.connect.runtime.isolation.Plugins;
21   import org.apache.kafka.connect.transforms.predicates.Predicate;
22   
23   import java.io.PrintStream;
24   import java.util.Collections;
25   import java.util.Comparator;
26   import java.util.List;
27   import java.util.stream.Collectors;
28   
29   public class PredicateDoc {
30   
31       private static final class DocInfo {
32           final String predicateName;
33           final String overview;
34           final ConfigDef configDef;
35   
36           private <P extends Predicate<?>> DocInfo(Class<P> predicateClass, String overview, ConfigDef configDef) {
37               this.predicateName = predicateClass.getName();
38               this.overview = overview;
39               this.configDef = configDef;
40           }
41       }
42   
43       private static final List<DocInfo> PREDICATES;
44       static {
45           List<DocInfo> collect = new Plugins(Collections.emptyMap()).predicates().stream()
46               .map(p -> {
47                   try {
48                       String overviewDoc = (String) p.pluginClass().getDeclaredField("OVERVIEW_DOC").get(null);
49                       ConfigDef configDef = (ConfigDef) p.pluginClass().getDeclaredField("CONFIG_DEF").get(null);
50                       return new DocInfo(p.pluginClass(), overviewDoc, configDef);
51                   } catch (ReflectiveOperationException e) {
52                       throw new RuntimeException("Predicate class " + p.pluginClass().getName() + " lacks either a `public static final String OVERVIEW_DOC` or `public static final ConfigDef CONFIG_DEF`");
53                   }
54               })
55               .sorted(Comparator.comparing(docInfo -> docInfo.predicateName))
56               .collect(Collectors.toList());
57           PREDICATES = collect;
58       }
59   
60       private static void printPredicateHtml(PrintStream out, DocInfo docInfo) {
61           out.println("<div id=\"" + docInfo.predicateName + "\">");
62   
63           out.print("<h5>");
64           out.print("<a href=\"#" + docInfo.predicateName + "\">" + docInfo.predicateName + "</a>");
65           out.println("</h5>");
66   
67           out.println(docInfo.overview);
68   
69           out.println("<p/>");
70   
71           out.println(docInfo.configDef.toHtml(6, key -> docInfo.predicateName + "_" + key));
72   
73           out.println("</div>");
74       }
75   
76       private static void printHtml(PrintStream out) {
77           for (final DocInfo docInfo : PREDICATES) {
78               printPredicateHtml(out, docInfo);
79           }
80       }
81   
82       public static void main(String... args) {
83           printHtml(System.out);
84       }
85   }
