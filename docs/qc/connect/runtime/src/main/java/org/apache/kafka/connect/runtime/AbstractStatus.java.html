17   package org.apache.kafka.connect.runtime;
18   
19   import java.util.Objects;
20   
21   public abstract class AbstractStatus<T> {
22   
23       public enum State {
24           UNASSIGNED,
25           RUNNING,
26           PAUSED,
27           FAILED,
28           DESTROYED,
29           RESTARTING,
30       }
31   
32       private final T id;
33       private final State state;
34       private final String trace;
35       private final String workerId;
36       private final int generation;
37   
38       public AbstractStatus(T id,
39                             State state,
40                             String workerId,
41                             int generation,
42                             String trace) {
43           this.id = id;
44           this.state = state;
45           this.workerId = workerId;
46           this.generation = generation;
47           this.trace = trace;
48       }
49   
50       public T id() {
51           return id;
52       }
53   
54       public State state() {
55           return state;
56       }
57   
58       public String trace() {
59           return trace;
60       }
61   
62       public String workerId() {
63           return workerId;
64       }
65   
66       public int generation() {
67           return generation;
68       }
69   
70       @Override
71       public String toString() {
72           return "Status{" +
73                   "id=" + id +
74                   ", state=" + state +
75                   ", workerId='" + workerId + '\'' +
76                   ", generation=" + generation +
77                   '}';
78       }
79   
80       @Override
81       public boolean equals(Object o) {
82           if (this == o) return true;
83           if (o == null || getClass() != o.getClass()) return false;
84   
85           AbstractStatus<?> that = (AbstractStatus<?>) o;
86   
87           return generation == that.generation
88                   && Objects.equals(id, that.id)
89                   && state == that.state
90                   && Objects.equals(trace, that.trace)
91                   && Objects.equals(workerId, that.workerId);
92       }
93   
94       @Override
95       public int hashCode() {
96           int result = id != null ? id.hashCode() : 0;
97           result = 31 * result + (state != null ? state.hashCode() : 0);
98           result = 31 * result + (trace != null ? trace.hashCode() : 0);
99           result = 31 * result + (workerId != null ? workerId.hashCode() : 0);
100           result = 31 * result + generation;
101           return result;
102       }
103   }
