17   package org.apache.kafka.connect.storage;
18   
19   import org.apache.kafka.connect.runtime.RestartRequest;
20   import org.apache.kafka.connect.runtime.SessionKey;
21   import org.apache.kafka.connect.runtime.TargetState;
22   import org.apache.kafka.connect.util.ConnectorTaskId;
23   
24   import java.util.Collection;
25   import java.util.List;
26   import java.util.Map;
27   import java.util.concurrent.TimeUnit;
28   import java.util.concurrent.TimeoutException;
29   
30   public interface ConfigBackingStore {
31   
32       void start();
33   
34       void stop();
35   
36       /**
37        * Get a snapshot of the current configuration state including all connector and task
38        * configurations.
39        * @return the cluster config state
40        */
41       ClusterConfigState snapshot();
42   
43       /**
44        * Check if the store has configuration for a connector.
45        * @param connector name of the connector
46        * @return true if the backing store contains configuration for the connector
47        */
48       boolean contains(String connector);
49   
50       /**
51        * Update the configuration for a connector.
52        * @param connector name of the connector
53        * @param properties the connector configuration
54        */
55       void putConnectorConfig(String connector, Map<String, String> properties);
56   
57       /**
58        * Remove configuration for a connector
59        * @param connector name of the connector
60        */
61       void removeConnectorConfig(String connector);
62   
63       /**
64        * Update the task configurations for a connector.
65        * @param connector name of the connector
66        * @param configs the new task configs for the connector
67        */
68       void putTaskConfigs(String connector, List<Map<String, String>> configs);
69   
70       /**
71        * Remove the task configs associated with a connector.
72        * @param connector name of the connector
73        */
74       void removeTaskConfigs(String connector);
75   
76       /**
77        * Refresh the backing store. This forces the store to ensure that it has the latest
78        * configs that have been written.
79        * @param timeout max time to wait for the refresh to complete
80        * @param unit unit of timeout
81        * @throws TimeoutException if the timeout expires before the refresh has completed
82        */
83       void refresh(long timeout, TimeUnit unit) throws TimeoutException;
84   
85       /**
86        * Transition a connector to a new target state (e.g. paused).
87        * @param connector name of the connector
88        * @param state the state to transition to
89        */
90       void putTargetState(String connector, TargetState state);
91   
92       /**
93        * Store a new {@link SessionKey} that can be used to validate internal (i.e., non-user-triggered) inter-worker communication.
94        * @param sessionKey the session key to store
95        */
96       void putSessionKey(SessionKey sessionKey);
97   
98       /**
99        * Request a restart of a connector and optionally its tasks.
100        * @param restartRequest the restart request details
101        */
102       void putRestartRequest(RestartRequest restartRequest);
103   
104       /**
105        * Record the number of tasks for the connector after a successful round of zombie fencing.
106        * @param connector name of the connector
107        * @param taskCount number of tasks used by the connector
108        */
109       void putTaskCountRecord(String connector, int taskCount);
110   
111       /**
112        * Prepare to write to the backing config store. May be required by some implementations (such as those that only permit a single
113        * writer at a time across a cluster of workers) before performing mutating operations like writing configurations, target states, etc.
114        * The default implementation is a no-op; it is the responsibility of the implementing class to override this and document any expectations for
115        * when it must be invoked.
116        */
117       default void claimWritePrivileges() {
118       }
119   
120       /**
121        * Set an update listener to get notifications when there are config/target state
122        * changes.
123        * @param listener non-null listener
124        */
125       void setUpdateListener(UpdateListener listener);
126   
127       interface UpdateListener {
128           /**
129            * Invoked when a connector configuration has been removed
130            * @param connector name of the connector
131            */
132           void onConnectorConfigRemove(String connector);
133   
134           /**
135            * Invoked when a connector configuration has been updated.
136            * @param connector name of the connector
137            */
138           void onConnectorConfigUpdate(String connector);
139   
140           /**
141            * Invoked when task configs are updated.
142            * @param tasks all the tasks whose configs have been updated
143            */
144           void onTaskConfigUpdate(Collection<ConnectorTaskId> tasks);
145   
146           /**
147            * Invoked when the user has set a new target state (e.g. paused)
148            * @param connector name of the connector
149            */
150           void onConnectorTargetStateChange(String connector);
151   
152           /**
153            * Invoked when the leader has distributed a new session key
154            * @param sessionKey the {@link SessionKey session key}
155            */
156           void onSessionKeyUpdate(SessionKey sessionKey);
157   
158           /**
159            * Invoked when a connector and possibly its tasks have been requested to be restarted.
160            * @param restartRequest the {@link RestartRequest restart request}
161            */
162           void onRestartRequest(RestartRequest restartRequest);
163       }
164   
165   }
