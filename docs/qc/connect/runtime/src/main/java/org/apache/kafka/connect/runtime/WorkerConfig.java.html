430       // Visible for testing
431       static void validateHttpResponseHeaderConfig(String config) {
432           try {
433               // validate format
434               String[] configTokens = config.trim().split("\\s+", 2);
435               if (configTokens.length != 2) {
436                   throw new ConfigException(String.format("Invalid format of header config '%s'. "
437                           + "Expected: '[ation] [header name]:[header value]'", config));
438               }
439   
440               // validate action
441               String method = configTokens[0].trim();
442               validateHeaderConfigAction(method);
443   
444               // validate header name and header value pair
445               String header = configTokens[1];
446               String[] headerTokens = header.trim().split(":");
447               if (headerTokens.length != 2) {
448                   throw new ConfigException(
449                           String.format("Invalid format of header name and header value pair '%s'. "
450                                   + "Expected: '[header name]:[header value]'", header));
451               }
452   
453               // validate header name
454               String headerName = headerTokens[0].trim();
455               if (headerName.isEmpty() || headerName.matches(".*\\s+.*")) {
456                   throw new ConfigException(String.format("Invalid header name '%s'. "
457                           + "The '[header name]' cannot contain whitespace", headerName));
458               }
459           } catch (ArrayIndexOutOfBoundsException e) {
460               throw new ConfigException(String.format("Invalid header config '%s'.", config), e);
461           }
462       }

468                       + "Expected one of %s", action, HEADER_ACTIONS));
