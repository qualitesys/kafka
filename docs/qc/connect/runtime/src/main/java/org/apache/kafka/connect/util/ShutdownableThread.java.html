17   package org.apache.kafka.connect.util;
18   
19   import org.slf4j.Logger;
20   import org.slf4j.LoggerFactory;
21   
22   import java.util.concurrent.CountDownLatch;
23   import java.util.concurrent.TimeUnit;
24   import java.util.concurrent.atomic.AtomicBoolean;
25   
26   /**
27    * <p>
28    * Thread class with support for triggering graceful and forcible shutdown. In graceful shutdown,
29    * a flag is set, which the thread should detect and try to exit gracefully from. In forcible
30    * shutdown, the thread is interrupted. These can be combined to give a thread a chance to exit
31    * gracefully, but then force it to exit if it takes too long.
32    * </p>
33    * <p>
34    * Implementations should override the {@link #execute} method and check {@link #getRunning} to
35    * determine whether they should try to gracefully exit.
36    * </p>
37    */
38   public abstract class ShutdownableThread extends Thread {
39       private static final Logger log = LoggerFactory.getLogger(ShutdownableThread.class);
40   
41       private final AtomicBoolean isRunning = new AtomicBoolean(true);
42       private final CountDownLatch shutdownLatch = new CountDownLatch(1);
43   
44       /**
45        * An UncaughtExceptionHandler to register on every instance of this class. This is useful for
46        * testing, where AssertionExceptions in the thread may not cause the test to fail. Since one
47        * instance is used for all threads, it must be thread-safe.
48        */
49       volatile public static UncaughtExceptionHandler funcaughtExceptionHandler = null;
50   
51       public ShutdownableThread(String name) {
52           // The default is daemon=true so that these threads will not prevent shutdown. We use this
53           // default because threads that are running user code that may not clean up properly, even
54           // when we attempt to forcibly shut them down.
55           this(name, true);
56       }
57   
58       public ShutdownableThread(String name, boolean daemon) {
59           super(name);
60           this.setDaemon(daemon);
61           if (funcaughtExceptionHandler != null)
62               this.setUncaughtExceptionHandler(funcaughtExceptionHandler);
63       }
64   
65       /**
66        * Implementations should override this method with the main body for the thread.
67        */
68       public abstract void execute();
69   
70       /**
71        * Returns true if the thread hasn't exited yet and none of the shutdown methods have been
72        * invoked
73        */
74       public boolean getRunning() {
75           return isRunning.get();
76       }
77   
78       @Override
79       public void run() {
80           try {
81               execute();
82           } catch (Error | RuntimeException e) {
83               log.error("Thread {} exiting with uncaught exception: ", getName(), e);
84               throw e;
85           } finally {
86               shutdownLatch.countDown();
87           }
88       }
89   
90       /**
91        * Shutdown the thread, first trying to shut down gracefully using the specified timeout, then
92        * forcibly interrupting the thread.
93        * @param gracefulTimeout the maximum time to wait for a graceful exit
94        * @param unit the time unit of the timeout argument
95        */
96       public void shutdown(long gracefulTimeout, TimeUnit unit)
97               throws InterruptedException {
98           boolean success = gracefulShutdown(gracefulTimeout, unit);
99           if (!success)
100               forceShutdown();
101       }
102   
103       /**
104        * Attempt graceful shutdown
105        * @param timeout the maximum time to wait
106        * @param unit the time unit of the timeout argument
107        * @return true if successful, false if the timeout elapsed
108        */
109       public boolean gracefulShutdown(long timeout, TimeUnit unit) throws InterruptedException {
110           startGracefulShutdown();
111           return awaitShutdown(timeout, unit);
112       }
113   
114       /**
115        * Start shutting down this thread gracefully, but do not block waiting for it to exit.
116        */
117       public void startGracefulShutdown() {
118           log.info("Starting graceful shutdown of thread {}", getName());
119           isRunning.set(false);
120       }
121   
122       /**
123        * Awaits shutdown of this thread, waiting up to the timeout.
124        * @param timeout the maximum time to wait
125        * @param unit the time unit of the timeout argument
126        * @return true if successful, false if the timeout elapsed
127        * @throws InterruptedException
128        */
129       public boolean awaitShutdown(long timeout, TimeUnit unit) throws InterruptedException {
130           return shutdownLatch.await(timeout, unit);
131       }
132   
133       /**
134        * Immediately tries to force the thread to shut down by interrupting it. This does not try to
135        * wait for the thread to truly exit because forcible shutdown is not always possible. By
136        * default, threads are marked as daemon threads so they will not prevent the process from
137        * exiting.
138        */
139       public void forceShutdown() {
140           log.info("Forcing shutdown of thread {}", getName());
141           isRunning.set(false);
142           interrupt();
143       }
144   }
