19   import org.apache.kafka.common.config.ConfigDef;
20   import org.apache.kafka.common.utils.AppInfoParser;

22   import org.apache.kafka.connect.connector.Task;
23   import org.slf4j.Logger;
24   import org.slf4j.LoggerFactory;

26   import java.util.Collections;
27   import java.util.List;
28   import java.util.Map;
29   import java.util.concurrent.Executors;
30   import java.util.concurrent.ScheduledExecutorService;
31   import java.util.concurrent.TimeUnit;
32   
33   /**
34    * This connector provides support for mocking certain connector behaviors. For example,
35    * this can be used to simulate connector or task failures. It works by passing a "mock mode"
36    * through configuration from the system test. New mock behavior can be implemented either
37    * in the connector or in the task by providing a new mode implementation.
38    *
39    * At the moment, this connector only supports a single task and shares configuration between
40    * the connector and its tasks.
41    *
42    * @see MockSinkConnector
43    * @see MockSourceConnector
44    */
45   public class MockConnector extends Connector {
46       public static final String MOCK_MODE_KEY = "mock_mode";
47       public static final String DELAY_MS_KEY = "delay_ms";
48   
49       public static final String CONNECTOR_FAILURE = "connector-failure";
50       public static final String TASK_FAILURE = "task-failure";
51   
52       public static final long DEFAULT_FAILURE_DELAY_MS = 15000;
53   
54       private static final Logger log = LoggerFactory.getLogger(MockConnector.class);
55   
56       private Map<String, String> config;
57       private ScheduledExecutorService executor;
58   
59       @Override
60       public String version() {
61           return AppInfoParser.getVersion();
62       }
63   
64       @Override
65       public void start(Map<String, String> config) {
66           this.config = config;
67   
68           if (CONNECTOR_FAILURE.equals(config.get(MOCK_MODE_KEY))) {
69               // Schedule this connector to raise an exception after some delay
70   
71               String delayMsString = config.get(DELAY_MS_KEY);
72               long delayMs = DEFAULT_FAILURE_DELAY_MS;
73               if (delayMsString != null)
74                   delayMs = Long.parseLong(delayMsString);
75   
76               log.debug("Started MockConnector with failure delay of {} ms", delayMs);
77               executor = Executors.newSingleThreadScheduledExecutor();
78               executor.schedule(() -> {
79                   log.debug("Triggering connector failure");
80                   context.raiseError(new RuntimeException());
81               }, delayMs, TimeUnit.MILLISECONDS);
82           }
83       }
84   
85       @Override
86       public Class<? extends Task> taskClass() {
87           throw new UnsupportedOperationException();
88       }
89   
90       @Override
91       public List<Map<String, String>> taskConfigs(int maxTasks) {
92           log.debug("Creating single task for MockConnector");
93           return Collections.singletonList(config);
94       }
95   
96       @Override
97       public void stop() {
98           if (executor != null) {
99               executor.shutdownNow();
100   
101               try {
102                   if (!executor.awaitTermination(20, TimeUnit.SECONDS))
103                       throw new RuntimeException("Failed timely termination of scheduler");
104               } catch (InterruptedException e) {
105                   throw new RuntimeException("Task was interrupted during shutdown");
106               }
107           }
108       }
109   
110       @Override
111       public ConfigDef config() {
112           return new ConfigDef();
113       }
114   
115   }
