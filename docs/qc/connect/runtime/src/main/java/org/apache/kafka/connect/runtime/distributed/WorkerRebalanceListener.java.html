17   package org.apache.kafka.connect.runtime.distributed;
18   
19   import org.apache.kafka.connect.util.ConnectorTaskId;
20   
21   import java.util.Collection;
22   
23   /**
24    * Listener for rebalance events in the worker group.
25    */
26   public interface WorkerRebalanceListener {
27       /**
28        * Invoked when a new assignment is created by joining the Connect worker group. This is
29        * invoked for both successful and unsuccessful assignments.
30        */
31       void onAssigned(ExtendedAssignment assignment, int generation);
32   
33       /**
34        * Invoked when a rebalance operation starts, revoking ownership for the set of connectors
35        * and tasks. Depending on the Connect protocol version, the collection of revoked connectors
36        * or tasks might refer to all or some of the connectors and tasks running on the worker.
37        */
38       void onRevoked(String leader, Collection<String> connectors, Collection<ConnectorTaskId> tasks);
39   }
