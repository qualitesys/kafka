348        * Attempt to create the topics described by the given definitions, returning all of the names of those topics that
349        * were created by this request. Any existing topics with the same name are unchanged, and the names of such topics
350        * are excluded from the result.
351        * <p>
352        * If multiple topic definitions have the same topic name, the last one with that name will be used.
353        * <p>
354        * Apache Kafka added support for creating topics in 0.10.1.0, so this method works as expected with that and later versions.
355        * With brokers older than 0.10.1.0, this method is unable to create topics and always returns an empty set.
356        *
357        * @param topics the specifications of the topics
358        * @return the {@link TopicCreationResponse} with the names of the newly created and existing topics;
359        *         never null but possibly empty
360        * @throws ConnectException if an error occurs, the operation takes too long, or the thread is interrupted while
361        *                          attempting to perform this operation
362        */
363       public TopicCreationResponse createOrFindTopics(NewTopic... topics) {
364           Map<String, NewTopic> topicsByName = new HashMap<>();
365           if (topics != null) {
366               for (NewTopic topic : topics) {
367                   if (topic != null) topicsByName.put(topic.name(), topic);
368               }
369           }
370           if (topicsByName.isEmpty()) return EMPTY_CREATION;
371           String topicNameList = Utils.join(topicsByName.keySet(), "', '");
372   
373           // Attempt to create any missing topics
374           CreateTopicsOptions args = new CreateTopicsOptions().validateOnly(false);
375           Map<String, KafkaFuture<Void>> newResults = admin.createTopics(topicsByName.values(), args).values();
376   
377           // Iterate over each future so that we can handle individual failures like when some topics already exist
378           Set<String> newlyCreatedTopicNames = new HashSet<>();
379           Set<String> existingTopicNames = new HashSet<>();
380           for (Map.Entry<String, KafkaFuture<Void>> entry : newResults.entrySet()) {
381               String topic = entry.getKey();
382               try {
383                   entry.getValue().get();
384                   if (logCreation) {
385                       log.info("Created topic {} on brokers at {}", topicsByName.get(topic), bootstrapServers);
386                   }
387                   newlyCreatedTopicNames.add(topic);
388               } catch (ExecutionException e) {
389                   Throwable cause = e.getCause();
390                   if (cause instanceof TopicExistsException) {
391                       log.debug("Found existing topic '{}' on the brokers at {}", topic, bootstrapServers);
392                       existingTopicNames.add(topic);
393                       continue;
394                   }
395                   if (cause instanceof UnsupportedVersionException) {
396                       log.debug("Unable to create topic(s) '{}' since the brokers at {} do not support the CreateTopics API." +
397                               " Falling back to assume topic(s) exist or will be auto-created by the broker.",
398                               topicNameList, bootstrapServers);
399                       return EMPTY_CREATION;
400                   }
401                   if (cause instanceof ClusterAuthorizationException) {
402                       log.debug("Not authorized to create topic(s) '{}' upon the brokers {}." +
403                               " Falling back to assume topic(s) exist or will be auto-created by the broker.",
404                               topicNameList, bootstrapServers);
405                       return EMPTY_CREATION;
406                   }
407                   if (cause instanceof TopicAuthorizationException) {
408                       log.debug("Not authorized to create topic(s) '{}' upon the brokers {}." +
409                                       " Falling back to assume topic(s) exist or will be auto-created by the broker.",
410                               topicNameList, bootstrapServers);
411                       return EMPTY_CREATION;
412                   }
413                   if (cause instanceof InvalidConfigurationException) {
414                       throw new ConnectException("Unable to create topic(s) '" + topicNameList + "': " + cause.getMessage(),
415                               cause);
416                   }
417                   if (cause instanceof TimeoutException) {
418                       // Timed out waiting for the operation to complete
419                       throw new ConnectException("Timed out while checking for or creating topic(s) '" + topicNameList + "'." +
420                               " This could indicate a connectivity issue, unavailable topic partitions, or if" +
421                               " this is your first use of the topic it may have taken too long to create.", cause);
422                   }

642       /**
643        * Fetch the most recent offset for each of the supplied {@link TopicPartition} objects.
644        *
645        * @param partitions the topic partitions
646        * @return the map of offset for each topic partition, or an empty map if the supplied partitions
647        *         are null or empty
648        * @throws UnsupportedVersionException if the admin client cannot read end offsets
649        * @throws TimeoutException if the offset metadata could not be fetched before the amount of time allocated
650        *         by {@code request.timeout.ms} expires, and this call can be retried
651        * @throws LeaderNotAvailableException if the leader was not available and this call can be retried
652        * @throws RetriableException if a retriable error occurs, or the thread is interrupted while attempting
653        *         to perform this operation
654        * @throws ConnectException if a non retriable error occurs
655        */
656       public Map<TopicPartition, Long> endOffsets(Set<TopicPartition> partitions) {
657           if (partitions == null || partitions.isEmpty()) {
658               return Collections.emptyMap();
659           }
660           Map<TopicPartition, OffsetSpec> offsetSpecMap = partitions.stream().collect(Collectors.toMap(Function.identity(), tp -> OffsetSpec.latest()));
661           ListOffsetsResult resultFuture = admin.listOffsets(offsetSpecMap, new ListOffsetsOptions(IsolationLevel.READ_UNCOMMITTED));
662           // Get the individual result for each topic partition so we have better error messages
663           Map<TopicPartition, Long> result = new HashMap<>();
664           for (TopicPartition partition : partitions) {
665               try {
666                   ListOffsetsResultInfo info = resultFuture.partitionResult(partition).get();
667                   result.put(partition, info.offset());
668               } catch (ExecutionException e) {
669                   Throwable cause = e.getCause();
670                   String topic = partition.topic();
671                   if (cause instanceof AuthorizationException) {
672                       String msg = String.format("Not authorized to get the end offsets for topic '%s' on brokers at %s", topic, bootstrapServers);
673                       throw new ConnectException(msg, e);
674                   } else if (cause instanceof UnsupportedVersionException) {
675                       // Should theoretically never happen, because this method is the same as what the consumer uses and therefore
676                       // should exist in the broker since before the admin client was added
677                       String msg = String.format("API to get the get the end offsets for topic '%s' is unsupported on brokers at %s", topic, bootstrapServers);
678                       throw new UnsupportedVersionException(msg, e);
679                   } else if (cause instanceof TimeoutException) {
680                       String msg = String.format("Timed out while waiting to get end offsets for topic '%s' on brokers at %s", topic, bootstrapServers);
681                       throw new TimeoutException(msg, e);
682                   } else if (cause instanceof LeaderNotAvailableException) {
683                       String msg = String.format("Unable to get end offsets during leader election for topic '%s' on brokers at %s", topic, bootstrapServers);
684                       throw new LeaderNotAvailableException(msg, e);
685                   } else if (cause instanceof org.apache.kafka.common.errors.RetriableException) {
686                       throw (org.apache.kafka.common.errors.RetriableException) cause;
687                   } else {
688                       String msg = String.format("Error while getting end offsets for topic '%s' on brokers at %s", topic, bootstrapServers);
689                       throw new ConnectException(msg, e);
690                   }
691               } catch (InterruptedException e) {
692                   Thread.interrupted();
693                   String msg = String.format("Interrupted while attempting to read end offsets for topic '%s' on brokers at %s", partition.topic(), bootstrapServers);
694                   throw new RetriableException(msg, e);
695               }
696           }
697           return result;
698       }
