17   package org.apache.kafka.connect.runtime.errors;
18   
19   import org.apache.kafka.clients.consumer.ConsumerRecord;
20   import org.apache.kafka.clients.producer.RecordMetadata;
21   import org.apache.kafka.common.record.TimestampType;
22   import org.apache.kafka.connect.errors.ConnectException;
23   import org.apache.kafka.connect.runtime.errors.WorkerErrantRecordReporter.ErrantRecordFuture;
24   import org.apache.kafka.connect.source.SourceRecord;
25   
26   import java.util.Collection;
27   import java.util.Collections;
28   import java.util.List;
29   import java.util.Objects;
30   import java.util.concurrent.CompletableFuture;
31   import java.util.concurrent.Future;
32   import java.util.stream.Collectors;

34   /**
35    * Contains all the metadata related to the currently evaluating operation. Only one instance of this class is meant
36    * to exist per task in a JVM.
37    */
38   class ProcessingContext implements AutoCloseable {
39   
40       private Collection<ErrorReporter> reporters = Collections.emptyList();
41   
42       private ConsumerRecord<byte[], byte[]> consumedMessage;
43       private SourceRecord sourceRecord;
44   
45       /**
46        * The following fields need to be reset every time a new record is seen.
47        */
48   
49       private Stage position;
50       private Class<?> klass;
51       private int attempt;
52       private Throwable error;
53   
54       /**
55        * Reset the internal fields before executing operations on a new record.
56        */
57       private void reset() {
58           attempt = 0;
59           position = null;
60           klass = null;
61           error = null;
62       }
63   
64       /**
65        * Set the record consumed from Kafka in a sink connector.
66        *
67        * @param consumedMessage the record
68        */
69       public void consumerRecord(ConsumerRecord<byte[], byte[]> consumedMessage) {
70           this.consumedMessage = consumedMessage;
71           reset();
72       }
73   
74       /**
75        * @return the record consumed from Kafka. could be null
76        */
77       public ConsumerRecord<byte[], byte[]> consumerRecord() {
78           return consumedMessage;
79       }
80   
81       /**
82        * @return the source record being processed.
83        */
84       public SourceRecord sourceRecord() {
85           return sourceRecord;
86       }
87   
88       /**
89        * Set the source record being processed in the connect pipeline.
90        *
91        * @param record the source record
92        */
93       public void sourceRecord(SourceRecord record) {
94           this.sourceRecord = record;
95           reset();
96       }
97   
98       /**
99        * Set the stage in the connector pipeline which is currently executing.
100        *
101        * @param position the stage
102        */
103       public void position(Stage position) {
104           this.position = position;
105       }
106   
107       /**
108        * @return the stage in the connector pipeline which is currently executing.
109        */
110       public Stage stage() {
111           return position;
112       }
113   
114       /**
115        * @return the class which is going to execute the current operation.
116        */
117       public Class<?> executingClass() {
118           return klass;
119       }
120   
121       /**
122        * @param klass set the class which is currently executing.
123        */
124       public void executingClass(Class<?> klass) {
125           this.klass = klass;
126       }
127   
128       /**
129        * A helper method to set both the stage and the class.
130        *
131        * @param stage the stage
132        * @param klass the class which will execute the operation in this stage.
133        */
134       public void currentContext(Stage stage, Class<?> klass) {
135           position(stage);
136           executingClass(klass);
137       }
138   
139       /**
140        * Report errors. Should be called only if an error was encountered while executing the operation.
141        *
142        * @return a errant record future that potentially aggregates the producer futures
143        */
144       public Future<Void> report() {
145           if (reporters.size() == 1) {
146               return new ErrantRecordFuture(Collections.singletonList(reporters.iterator().next().report(this)));
147           }
148   
149           List<Future<RecordMetadata>> futures = reporters.stream()
150                   .map(r -> r.report(this))
151                   .filter(f -> !f.isDone())
152                   .collect(Collectors.toList());
153           if (futures.isEmpty()) {
154               return CompletableFuture.completedFuture(null);
155           }
156           return new ErrantRecordFuture(futures);
157       }
158   
159       @Override
160       public String toString() {
161           return toString(false);
162       }
163   
164       public String toString(boolean includeMessage) {
165           StringBuilder builder = new StringBuilder();
166           builder.append("Executing stage '");
167           builder.append(stage().name());
168           builder.append("' with class '");
169           builder.append(executingClass() == null ? "null" : executingClass().getName());
170           builder.append('\'');
171           if (includeMessage && sourceRecord() != null) {
172               builder.append(", where source record is = ");
173               builder.append(sourceRecord());
174           } else if (includeMessage && consumerRecord() != null) {
175               ConsumerRecord<byte[], byte[]> msg = consumerRecord();
176               builder.append(", where consumed record is ");
177               builder.append("{topic='").append(msg.topic()).append('\'');
178               builder.append(", partition=").append(msg.partition());
179               builder.append(", offset=").append(msg.offset());
180               if (msg.timestampType() == TimestampType.CREATE_TIME || msg.timestampType() == TimestampType.LOG_APPEND_TIME) {
181                   builder.append(", timestamp=").append(msg.timestamp());
182                   builder.append(", timestampType=").append(msg.timestampType());
183               }
184               builder.append("}");
185           }
186           builder.append('.');
187           return builder.toString();
188       }
189   
190       /**
191        * @param attempt the number of attempts made to execute the current operation.
192        */
193       public void attempt(int attempt) {
194           this.attempt = attempt;
195       }
196   
197       /**
198        * @return the number of attempts made to execute the current operation.
199        */
200       public int attempt() {
201           return attempt;
202       }
203   
204       /**
205        * @return the error (if any) which was encountered while processing the current stage.
206        */
207       public Throwable error() {
208           return error;
209       }
210   
211       /**
212        * The error (if any) which was encountered while processing the current stage.
213        *
214        * @param error the error
215        */
216       public void error(Throwable error) {
217           this.error = error;
218       }
219   
220       /**
221        * @return true, if the last operation encountered an error; false otherwise
222        */
223       public boolean failed() {
224           return error() != null;
225       }
226   
227       /**
228        * Set the error reporters for this connector.
229        *
230        * @param reporters the error reporters (should not be null).
231        */
232       public void reporters(Collection<ErrorReporter> reporters) {
233           Objects.requireNonNull(reporters);
234           this.reporters = reporters;
235       }
236   
237       @Override
238       public void close() {
239           ConnectException e = null;
240           for (ErrorReporter reporter : reporters) {
241               try {
242                   reporter.close();
243               } catch (Throwable t) {
244                   e = e != null ? e : new ConnectException("Failed to close all reporters");
245                   e.addSuppressed(t);
246               }
247           }
248           if (e != null) {
249               throw e;
250           }
251       }
252   }
