45   /**
46    * <p>
47    * Command line utility that runs Kafka Connect as a standalone process. In this mode, work is not
48    * distributed. Instead, all the normal Connect machinery works within a single process. This is
49    * useful for ad hoc, small, or experimental jobs.
50    * </p>
51    * <p>
52    * By default, no job configs or offset data is persistent. You can make jobs persistent and
53    * fault tolerant by overriding the settings to use file storage for both.
54    * </p>
55    */
56   public class ConnectStandalone {
57       private static final Logger log = LoggerFactory.getLogger(ConnectStandalone.class);
58   
59       public static void main(String[] args) {
60   
61           if (args.length < 2 || Arrays.asList(args).contains("--help")) {
62               log.info("Usage: ConnectStandalone worker.properties connector1.properties [connector2.properties ...]");
63               Exit.exit(1);
64           }
65   
66           try {
67               Time time = Time.SYSTEM;
68               log.info("Kafka Connect standalone worker initializing ...");
69               long initStart = time.hiResClockMs();
70               WorkerInfo initInfo = new WorkerInfo();
71               initInfo.logAll();
72   
73               String workerPropsFile = args[0];
74               Map<String, String> workerProps = !workerPropsFile.isEmpty() ?
75                       Utils.propsToStringMap(Utils.loadProps(workerPropsFile)) : Collections.emptyMap();
76   
77               log.info("Scanning for plugin classes. This might take a moment ...");
78               Plugins plugins = new Plugins(workerProps);
79               plugins.compareAndSwapWithDelegatingLoader();
80               StandaloneConfig config = new StandaloneConfig(workerProps);
81   
82               String kafkaClusterId = ConnectUtils.lookupKafkaClusterId(config);
83               log.debug("Kafka cluster ID: {}", kafkaClusterId);
84   
85               RestServer rest = new RestServer(config);
86               rest.initializeServer();
87   
88               URI advertisedUrl = rest.advertisedUrl();
89               String workerId = advertisedUrl.getHost() + ":" + advertisedUrl.getPort();
90   
91               ConnectorClientConfigOverridePolicy connectorClientConfigOverridePolicy = plugins.newPlugin(
92                   config.getString(WorkerConfig.CONNECTOR_CLIENT_POLICY_CLASS_CONFIG),
93                   config, ConnectorClientConfigOverridePolicy.class);
94               Worker worker = new Worker(workerId, time, plugins, config, new FileOffsetBackingStore(),
95                                          connectorClientConfigOverridePolicy);
96   
97               Herder herder = new StandaloneHerder(worker, kafkaClusterId, connectorClientConfigOverridePolicy);
98               final Connect connect = new Connect(herder, rest);
99               log.info("Kafka Connect standalone worker initialization took {}ms", time.hiResClockMs() - initStart);
100   
101               try {
102                   connect.start();
103                   for (final String connectorPropsFile : Arrays.copyOfRange(args, 1, args.length)) {
104                       Map<String, String> connectorProps = Utils.propsToStringMap(Utils.loadProps(connectorPropsFile));
105                       FutureCallback<Herder.Created<ConnectorInfo>> cb = new FutureCallback<>((error, info) -> {
106                           if (error != null)
107                               log.error("Failed to create job for {}", connectorPropsFile);
108                           else
109                               log.info("Created connector {}", info.result().name());
110                       });
111                       herder.putConnectorConfig(
112                               connectorProps.get(ConnectorConfig.NAME_CONFIG),
113                               connectorProps, false, cb);
114                       cb.get();
115                   }
116               } catch (Throwable t) {
117                   log.error("Stopping after connector error", t);
118                   connect.stop();
119                   Exit.exit(3);
120               }
121   
122               // Shutdown will be triggered by Ctrl-C or via HTTP shutdown request
123               connect.awaitStop();
124   
125           } catch (Throwable t) {
126               log.error("Stopping due to error", t);
127               Exit.exit(2);
128           }
129       }
130   }
