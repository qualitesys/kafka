42   /**
43    * An {@link OffsetBackingStore} with support for reading from and writing to a worker-global
44    * offset backing store and/or a connector-specific offset backing store.
45    */
46   public class ConnectorOffsetBackingStore implements OffsetBackingStore {

50       /**
51        * Builds an offset store that uses a connector-specific offset topic as the primary store and
52        * the worker-global offset store as the secondary store.
53        *
54        * @param loggingContext a {@link Supplier} for the {@link LoggingContext} that should be used
55        *                       for messages logged by this offset store; may not be null, and may never return null
56        * @param workerStore the worker-global offset store; may not be null
57        * @param connectorStore the connector-specific offset store; may not be null
58        * @param connectorOffsetsTopic the name of the connector-specific offset topic; may not be null
59        * @param connectorStoreAdmin the topic admin to use for the connector-specific offset topic; may not be null
60        * @return an offset store backed primarily by the connector-specific offset topic and secondarily
61        * by the worker-global offset store; never null
62        */

64               Supplier<LoggingContext> loggingContext,

78                   connectorOffsetsTopic,
79                   workerStore,
80                   connectorStore,

141       ConnectorOffsetBackingStore(
142               Time time,
143               Supplier<LoggingContext> loggingContext,
144               String primaryOffsetsTopic,
145               OffsetBackingStore workerStore,

154           this.primaryOffsetsTopic = primaryOffsetsTopic;
155           this.workerStore = Optional.ofNullable(workerStore);

164       /**
165        * If configured to use a connector-specific offset store, {@link OffsetBackingStore#start() start} that store.
166        *
167        * <p>The worker-global offset store is not modified; it is the caller's responsibility to ensure that it is started
168        * before calls to {@link #get(Collection)} and {@link #set(Map, Callback)} take place.
169        */
170       @Override
171       public void start() {
172           // Worker offset store should already be started
173           connectorStore.ifPresent(OffsetBackingStore::start);
174       }

176       /**
177        * If configured to use a connector-specific offset store, {@link OffsetBackingStore#stop() stop} that store,
178        * and {@link TopicAdmin#close(Duration) close} the topic admin used by that store.
179        *
180        * <p>The worker-global offset store is not modified as it may be used for other connectors that either already exist,
181        * or will be created, on this worker.
182        */
183       @Override
184       public void stop() {
185           // Worker offset store should not be stopped as it may be used for multiple connectors
186           connectorStore.ifPresent(OffsetBackingStore::stop);
187           connectorStoreAdmin.ifPresent(TopicAdmin::close);
188       }

190       /**
191        * Get the offset values for the specified keys.
192        *
193        * <p>If configured to use a connector-specific offset store, priority is given to the values contained in that store,
194        * and the values in the worker-global offset store (if one is provided) are used as a fallback for keys that are not
195        * present in the connector-specific store.
196        *
197        * <p>If not configured to use a connector-specific offset store, only the values contained in the worker-global
198        * offset store are returned.
199   
200        * @param keys list of keys to look up
201        * @return future for the resulting map from key to value
202        */
203       @Override
204       public Future<Map<ByteBuffer, ByteBuffer>> get(Collection<ByteBuffer> keys) {
205           Future<Map<ByteBuffer, ByteBuffer>> workerGetFuture = getFromStore(workerStore, keys);
206           Future<Map<ByteBuffer, ByteBuffer>> connectorGetFuture = getFromStore(connectorStore, keys);
207   
208           return new Future<Map<ByteBuffer, ByteBuffer>>() {
209               @Override
210               public boolean cancel(boolean mayInterruptIfRunning) {
211                   // Note the use of | instead of || here; this causes cancel to be invoked on both futures,
212                   // even if the first call to cancel returns true
213                   return workerGetFuture.cancel(mayInterruptIfRunning)
214                           | connectorGetFuture.cancel(mayInterruptIfRunning);
215               }
216   
217               @Override
218               public boolean isCancelled() {
219                   return workerGetFuture.isCancelled()
220                           || connectorGetFuture.isCancelled();
221               }
222   
223               @Override
224               public boolean isDone() {
225                   return workerGetFuture.isDone()
226                           && connectorGetFuture.isDone();
227               }
228   
229               @Override
230               public Map<ByteBuffer, ByteBuffer> get() throws InterruptedException, ExecutionException {
231                   Map<ByteBuffer, ByteBuffer> result = new HashMap<>(workerGetFuture.get());
232                   result.putAll(connectorGetFuture.get());
233                   return result;
234               }
235   
236               @Override
237               public Map<ByteBuffer, ByteBuffer> get(long timeout, TimeUnit unit) throws InterruptedException, ExecutionException, TimeoutException {
238                   long timeoutMs = unit.toMillis(timeout);
239                   long endTime = time.milliseconds() + timeoutMs;
240                   Map<ByteBuffer, ByteBuffer> result = new HashMap<>(workerGetFuture.get(timeoutMs, unit));
241                   timeoutMs = Math.max(1, endTime - time.milliseconds());
242                   result.putAll(connectorGetFuture.get(timeoutMs, TimeUnit.MILLISECONDS));
243                   return result;
244               }
245           };
246       }
247   
248       /**
249        * Store the specified offset key/value pairs.
250        *
251        * <p>If configured to use a connector-specific offset store, the returned {@link Future} corresponds to a
252        * write to that store, and the passed-in {@link Callback} is invoked once that write completes. If a worker-global
253        * store is provided, a secondary write is made to that store if the write to the connector-specific store
254        * succeeds. Errors with this secondary write are not reflected in the returned {@link Future} or the passed-in
255        * {@link Callback}; they are only logged as a warning to users.
256        *
257        * <p>If not configured to use a connector-specific offset store, the returned {@link Future} corresponds to a
258        * write to the worker-global offset store, and the passed-in {@link Callback} is invoked once that write completes.
259   
260        * @param values map from key to value
261        * @param callback callback to invoke on completion of the primary write
262        * @return void future for the primary write
263        */
264       @Override
265       public Future<Void> set(Map<ByteBuffer, ByteBuffer> values, Callback<Void> callback) {
266           final OffsetBackingStore primaryStore;
267           final OffsetBackingStore secondaryStore;
268           if (connectorStore.isPresent()) {
269               primaryStore = connectorStore.get();
270               secondaryStore = workerStore.orElse(null);
271           } else if (workerStore.isPresent()) {
272               primaryStore = workerStore.get();
273               secondaryStore = null;
274           } else {
275               // Should never happen since we check for this case in the constructor, but just in case, this should
276               // be more informative than the NPE that would otherwise be thrown
277               throw new IllegalStateException("At least one non-null offset store must be provided");
278           }
279   
280           return primaryStore.set(values, (primaryWriteError, ignored) -> {
281               if (secondaryStore != null) {
282                   if (primaryWriteError != null) {
283                       log.trace("Skipping offsets write to secondary store because primary write has failed", primaryWriteError);
284                   } else {
285                       try {
286                           // Invoke OffsetBackingStore::set but ignore the resulting future; we don't block on writes to this
287                           // backing store.
288                           secondaryStore.set(values, (secondaryWriteError, ignored2) -> {
289                               try (LoggingContext context = loggingContext()) {
290                                   if (secondaryWriteError != null) {
291                                       log.warn("Failed to write offsets to secondary backing store", secondaryWriteError);
292                                   } else {
293                                       log.debug("Successfully flushed offsets to secondary backing store");
294                                   }
295                               }
296                           });
297                       } catch (Exception e) {
298                           log.warn("Failed to write offsets to secondary backing store", e);
299                       }
300                   }
301               }
302               try (LoggingContext context = loggingContext()) {
303                   callback.onCompletion(primaryWriteError, ignored);
304               }
305           });
306       }

308       /**
309        * If configured to use a connector-specific offset store,
310        * {@link OffsetBackingStore#configure(WorkerConfig) configure} that store.
311        *
312        * <p>The worker-global offset store is not modified; it is the caller's responsibility to ensure that it is configured
313        * before calls to {@link #start()}, {@link #get(Collection)} and {@link #set(Map, Callback)} take place.

318           connectorStore.ifPresent(store -> store.configure(config));
319       }

337       private static Future<Map<ByteBuffer, ByteBuffer>> getFromStore(Optional<? extends OffsetBackingStore> store, Collection<ByteBuffer> keys) {
