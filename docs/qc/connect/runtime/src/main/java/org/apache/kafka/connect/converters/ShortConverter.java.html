17   package org.apache.kafka.connect.converters;
18   
19   import org.apache.kafka.common.serialization.ShortDeserializer;
20   import org.apache.kafka.common.serialization.ShortSerializer;
21   import org.apache.kafka.connect.data.Schema;
22   import org.apache.kafka.connect.storage.Converter;
23   import org.apache.kafka.connect.storage.HeaderConverter;
24   
25   /**
26    * {@link Converter} and {@link HeaderConverter} implementation that only supports serializing to and deserializing from short values.
27    * It does support handling nulls. When converting from bytes to Kafka Connect format, the converter will always return an
28    * optional INT16 schema.
29    * <p>
30    * This implementation currently does nothing with the topic names or header names.
31    */
32   public class ShortConverter extends NumberConverter<Short> {
33   
34       public ShortConverter() {
35           super("short", Schema.OPTIONAL_INT16_SCHEMA, new ShortSerializer(), new ShortDeserializer());
36       }
37   }
