108       @GET
109       @Path("/")
110       @Operation(summary = "List all active connectors")
111       public Response listConnectors(
112           final @Context UriInfo uriInfo,
113           final @Context HttpHeaders headers
114       ) {
115           if (uriInfo.getQueryParameters().containsKey("expand")) {
116               Map<String, Map<String, Object>> out = new HashMap<>();
117               for (String connector : herder.connectors()) {
118                   try {
119                       Map<String, Object> connectorExpansions = new HashMap<>();
120                       for (String expansion : uriInfo.getQueryParameters().get("expand")) {
121                           switch (expansion) {
122                               case "status":
123                                   connectorExpansions.put("status", herder.connectorStatus(connector));
124                                   break;
125                               case "info":
126                                   connectorExpansions.put("info", herder.connectorInfo(connector));
127                                   break;
128                               default:
129                                   log.info("Ignoring unknown expansion type {}", expansion);
130                           }
131                       }
132                       out.put(connector, connectorExpansions);
133                   } catch (NotFoundException e) {
134                       // this likely means that a connector has been removed while we look its info up
135                       // we can just not include this connector in the return entity
136                       log.debug("Unable to get connector info for {} on this worker", connector);
137                   }
138   
139               }
140               return Response.ok(out).build();
141           } else {
142               return Response.ok(herder.connectors()).build();
143           }
144       }

378       // Check whether the connector name from the url matches the one (if there is one) provided in the connectorConfig
379       // object. Throw BadRequestException on mismatch, otherwise put connectorName in config
380       private void checkAndPutConnectorConfigName(String connectorName, Map<String, String> connectorConfig) {
381           String includedName = connectorConfig.get(ConnectorConfig.NAME_CONFIG);
382           if (includedName != null) {
383               if (!includedName.equals(connectorName))
384                   throw new BadRequestException("Connector name configuration (" + includedName + ") doesn't match connector name in the URL (" + connectorName + ")");
385           } else {
386               connectorConfig.put(ConnectorConfig.NAME_CONFIG, connectorName);
387           }
388       }
389   
390       // Wait for a FutureCallback to complete. If it succeeds, return the parsed response. If it fails, try to forward the
391       // request to the leader.
392       private <T, U> T completeOrForwardRequest(FutureCallback<T> cb,
393                                                 String path,
394                                                 String method,
395                                                 HttpHeaders headers,
396                                                 Map<String, String> queryParameters,
397                                                 Object body,
398                                                 TypeReference<U> resultType,
399                                                 Translator<T, U> translator,
400                                                 Boolean forward) throws Throwable {
401           try {
402               return cb.get(requestTimeoutMs, TimeUnit.MILLISECONDS);
403           } catch (ExecutionException e) {
404               Throwable cause = e.getCause();
405   
406               if (cause instanceof RequestTargetException) {
407                   if (forward == null || forward) {
408                       // the only time we allow recursive forwarding is when no forward flag has
409                       // been set, which should only be seen by the first worker to handle a user request.
410                       // this gives two total hops to resolve the request before giving up.
411                       boolean recursiveForward = forward == null;
412                       RequestTargetException targetException = (RequestTargetException) cause;
413                       String forwardedUrl = targetException.forwardUrl();
414                       if (forwardedUrl == null) {
415                           // the target didn't know of the leader at this moment.
416                           throw new ConnectRestException(Response.Status.CONFLICT.getStatusCode(),
417                                   "Cannot complete request momentarily due to no known leader URL, "
418                                   + "likely because a rebalance was underway.");
419                       }
420                       UriBuilder uriBuilder = UriBuilder.fromUri(forwardedUrl)
421                               .path(path)
422                               .queryParam("forward", recursiveForward);
423                       if (queryParameters != null) {
424                           queryParameters.forEach(uriBuilder::queryParam);
425                       }
426                       String forwardUrl = uriBuilder.build().toString();
427                       log.debug("Forwarding request {} {} {}", forwardUrl, method, body);
428                       return translator.translate(RestClient.httpRequest(forwardUrl, method, headers, body, resultType, config));
429                   } else {
430                       // we should find the right target for the query within two hops, so if
431                       // we don't, it probably means that a rebalance has taken place.
432                       throw new ConnectRestException(Response.Status.CONFLICT.getStatusCode(),
433                               "Cannot complete request because of a conflicting operation (e.g. worker rebalance)");
434                   }
435               } else if (cause instanceof RebalanceNeededException) {
436                   throw new ConnectRestException(Response.Status.CONFLICT.getStatusCode(),
437                           "Cannot complete request momentarily due to stale configuration (typically caused by a concurrent config change)");
438               }
439   
440               throw cause;
441           } catch (TimeoutException e) {
442               // This timeout is for the operation itself. None of the timeout error codes are relevant, so internal server
443               // error is the best option
444               throw new ConnectRestException(Response.Status.INTERNAL_SERVER_ERROR.getStatusCode(), "Request timed out");
445           } catch (InterruptedException e) {
446               throw new ConnectRestException(Response.Status.INTERNAL_SERVER_ERROR.getStatusCode(), "Request interrupted");
447           }
448       }
