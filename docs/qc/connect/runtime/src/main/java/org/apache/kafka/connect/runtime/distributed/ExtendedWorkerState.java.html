17   package org.apache.kafka.connect.runtime.distributed;
18   
19   /**
20    * A class that captures the deserialized form of a worker's metadata.
21    */
22   public class ExtendedWorkerState extends ConnectProtocol.WorkerState {
23       private final ExtendedAssignment assignment;
24   
25       public ExtendedWorkerState(String url, long offset, ExtendedAssignment assignment) {
26           super(url, offset);
27           this.assignment = assignment != null ? assignment : ExtendedAssignment.empty();
28       }
29   
30       /**
31        * This method returns which was the assignment of connectors and tasks on a worker at the
32        * moment that its state was captured by this class.
33        *
34        * @return the assignment of connectors and tasks
35        */
36       public ExtendedAssignment assignment() {
37           return assignment;
38       }
39   
40       @Override
41       public String toString() {
42           return "WorkerState{" +
43                   "url='" + url() + '\'' +
44                   ", offset=" + offset() +
45                   ", " + assignment +
46                   '}';
47       }
48   }
