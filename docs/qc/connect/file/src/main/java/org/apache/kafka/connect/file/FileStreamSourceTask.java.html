39   /**
40    * FileStreamSourceTask reads from stdin or a file.
41    */
42   public class FileStreamSourceTask extends SourceTask {
43       private static final Logger log = LoggerFactory.getLogger(FileStreamSourceTask.class);
44       public static final String FILENAME_FIELD = "filename";
45       public  static final String POSITION_FIELD = "position";
46       private static final Schema VALUE_SCHEMA = Schema.STRING_SCHEMA;
47   
48       private String filename;
49       private InputStream stream;
50       private BufferedReader reader = null;
51       private char[] buffer;
52       private int offset = 0;
53       private String topic = null;
54       private int batchSize = FileStreamSourceConnector.DEFAULT_TASK_BATCH_SIZE;
55   
56       private Long streamOffset;
57   
58       public FileStreamSourceTask() {
59           this(1024);
60       }
61   
62       /* visible for testing */
63       FileStreamSourceTask(int initialBufferSize) {
64           buffer = new char[initialBufferSize];
65       }
66   
67       @Override
68       public String version() {
69           return new FileStreamSourceConnector().version();
70       }
71   
72       @Override
73       public void start(Map<String, String> props) {
74           filename = props.get(FileStreamSourceConnector.FILE_CONFIG);
75           if (filename == null || filename.isEmpty()) {
76               stream = System.in;
77               // Tracking offset for stdin doesn't make sense
78               streamOffset = null;
79               reader = new BufferedReader(new InputStreamReader(stream, StandardCharsets.UTF_8));
80           }
81           // Missing topic or parsing error is not possible because we've parsed the config in the
82           // Connector
83           topic = props.get(FileStreamSourceConnector.TOPIC_CONFIG);
84           batchSize = Integer.parseInt(props.get(FileStreamSourceConnector.TASK_BATCH_SIZE_CONFIG));
85       }
86   
87       @Override
88       public List<SourceRecord> poll() throws InterruptedException {
89           if (stream == null) {
90               try {
91                   stream = Files.newInputStream(Paths.get(filename));
92                   Map<String, Object> offset = context.offsetStorageReader().offset(Collections.singletonMap(FILENAME_FIELD, filename));
93                   if (offset != null) {
94                       Object lastRecordedOffset = offset.get(POSITION_FIELD);
95                       if (lastRecordedOffset != null && !(lastRecordedOffset instanceof Long))
96                           throw new ConnectException("Offset position is the incorrect type");
97                       if (lastRecordedOffset != null) {
98                           log.debug("Found previous offset, trying to skip to file offset {}", lastRecordedOffset);
99                           long skipLeft = (Long) lastRecordedOffset;
100                           while (skipLeft > 0) {
101                               try {
102                                   long skipped = stream.skip(skipLeft);
103                                   skipLeft -= skipped;
104                               } catch (IOException e) {
105                                   log.error("Error while trying to seek to previous offset in file {}: ", filename, e);
106                                   throw new ConnectException(e);
107                               }
108                           }
109                           log.debug("Skipped to offset {}", lastRecordedOffset);
110                       }
111                       streamOffset = (lastRecordedOffset != null) ? (Long) lastRecordedOffset : 0L;
112                   } else {
113                       streamOffset = 0L;
114                   }
115                   reader = new BufferedReader(new InputStreamReader(stream, StandardCharsets.UTF_8));
116                   log.debug("Opened {} for reading", logFilename());
117               } catch (NoSuchFileException e) {
118                   log.warn("Couldn't find file {} for FileStreamSourceTask, sleeping to wait for it to be created", logFilename());
119                   synchronized (this) {
120                       this.wait(1000);
121                   }
122                   return null;
123               } catch (IOException e) {
124                   log.error("Error while trying to open file {}: ", filename, e);
125                   throw new ConnectException(e);
126               }
127           }
128   
129           // Unfortunately we can't just use readLine() because it blocks in an uninterruptible way.
130           // Instead we have to manage splitting lines ourselves, using simple backoff when no new data
131           // is available.
132           try {
133               final BufferedReader readerCopy;
134               synchronized (this) {
135                   readerCopy = reader;
136               }
137               if (readerCopy == null)
138                   return null;
139   
140               ArrayList<SourceRecord> records = null;
141   
142               int nread = 0;
143               while (readerCopy.ready()) {
144                   nread = readerCopy.read(buffer, offset, buffer.length - offset);
145                   log.trace("Read {} bytes from {}", nread, logFilename());
146   
147                   if (nread > 0) {
148                       offset += nread;
149                       String line;
150                       boolean foundOneLine = false;
151                       do {
152                           line = extractLine();
153                           if (line != null) {
154                               foundOneLine = true;
155                               log.trace("Read a line from {}", logFilename());
156                               if (records == null)
157                                   records = new ArrayList<>();
158                               records.add(new SourceRecord(offsetKey(filename), offsetValue(streamOffset), topic, null,
159                                       null, null, VALUE_SCHEMA, line, System.currentTimeMillis()));
160   
161                               if (records.size() >= batchSize) {
162                                   return records;
163                               }
164                           }
165                       } while (line != null);
166   
167                       if (!foundOneLine && offset == buffer.length) {
168                           char[] newbuf = new char[buffer.length * 2];
169                           System.arraycopy(buffer, 0, newbuf, 0, buffer.length);
170                           log.info("Increased buffer from {} to {}", buffer.length, newbuf.length);
171                           buffer = newbuf;
172                       }
173                   }
174               }
175   
176               if (nread <= 0)
177                   synchronized (this) {
178                       this.wait(1000);
179                   }
180   
181               return records;
182           } catch (IOException e) {
183               // Underlying stream was killed, probably as a result of calling stop. Allow to return
184               // null, and driving thread will handle any shutdown if necessary.
185           }
186           return null;
187       }
188   
189       private String extractLine() {
190           int until = -1, newStart = -1;
191           for (int i = 0; i < offset; i++) {
192               if (buffer[i] == '\n') {
193                   until = i;
194                   newStart = i + 1;
195                   break;
196               } else if (buffer[i] == '\r') {
197                   // We need to check for \r\n, so we must skip this if we can't check the next char
198                   if (i + 1 >= offset)
199                       return null;
200   
201                   until = i;
202                   newStart = (buffer[i + 1] == '\n') ? i + 2 : i + 1;
203                   break;
204               }
205           }
206   
207           if (until != -1) {
208               String result = new String(buffer, 0, until);
209               System.arraycopy(buffer, newStart, buffer, 0, buffer.length - newStart);
210               offset = offset - newStart;
211               if (streamOffset != null)
212                   streamOffset += newStart;
213               return result;
214           } else {
215               return null;
216           }
217       }
218   
219       @Override
220       public void stop() {
221           log.trace("Stopping");
222           synchronized (this) {
223               try {
224                   if (stream != null && stream != System.in) {
225                       stream.close();
226                       log.trace("Closed input stream");
227                   }
228               } catch (IOException e) {
229                   log.error("Failed to close FileStreamSourceTask stream: ", e);
230               }
231               this.notify();
232           }
233       }
234   
235       private Map<String, String> offsetKey(String filename) {
236           return Collections.singletonMap(FILENAME_FIELD, filename);
237       }
238   
239       private Map<String, Long> offsetValue(Long pos) {
240           return Collections.singletonMap(POSITION_FIELD, pos);
241       }
242   
243       private String logFilename() {
244           return filename == null ? "stdin" : filename;
245       }
246   
247       /* visible for testing */
248       int bufferSize() {
249           return buffer.length;
250       }
251   }
