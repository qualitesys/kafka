39   /** Replicate consumer group state between clusters. Emits checkpoint records.
40    *
41    *  @see MirrorConnectorConfig for supported config properties.
42    */
43   public class MirrorCheckpointConnector extends SourceConnector {
44   
45       private static final Logger log = LoggerFactory.getLogger(MirrorCheckpointConnector.class);
46   
47       private Scheduler scheduler;
48       private MirrorConnectorConfig config;
49       private GroupFilter groupFilter;
50       private AdminClient sourceAdminClient;
51       private SourceAndTarget sourceAndTarget;
52       private List<String> knownConsumerGroups = Collections.emptyList();
53   
54       public MirrorCheckpointConnector() {
55           // nop
56       }
57   
58       // visible for testing
59       MirrorCheckpointConnector(List<String> knownConsumerGroups, MirrorConnectorConfig config) {
60           this.knownConsumerGroups = knownConsumerGroups;
61           this.config = config;
62       }
63   
64       @Override
65       public void start(Map<String, String> props) {
66           config = new MirrorConnectorConfig(props);
67           if (!config.enabled()) {
68               return;
69           }
70           String connectorName = config.connectorName();
71           sourceAndTarget = new SourceAndTarget(config.sourceClusterAlias(), config.targetClusterAlias());
72           groupFilter = config.groupFilter();
73           sourceAdminClient = AdminClient.create(config.sourceAdminConfig());
74           scheduler = new Scheduler(MirrorCheckpointConnector.class, config.adminTimeout());
75           scheduler.execute(this::createInternalTopics, "creating internal topics");
76           scheduler.execute(this::loadInitialConsumerGroups, "loading initial consumer groups");
77           scheduler.scheduleRepeatingDelayed(this::refreshConsumerGroups, config.refreshGroupsInterval(),
78                   "refreshing consumer groups");
79           log.info("Started {} with {} consumer groups.", connectorName, knownConsumerGroups.size());
80           log.debug("Started {} with consumer groups: {}", connectorName, knownConsumerGroups);
81       }
82   
83       @Override
84       public void stop() {
85           if (!config.enabled()) {
86               return;
87           }
88           Utils.closeQuietly(scheduler, "scheduler");
89           Utils.closeQuietly(groupFilter, "group filter");
90           Utils.closeQuietly(sourceAdminClient, "source admin client");
91       }
92   
93       @Override
94       public Class<? extends Task> taskClass() {
95           return MirrorCheckpointTask.class;
96       }
97   
98       // divide consumer groups among tasks
99       @Override
100       public List<Map<String, String>> taskConfigs(int maxTasks) {
101           // if the replication is disabled, known consumer group is empty, or checkpoint emission is
102           // disabled by setting 'emit.checkpoints.enabled' to false, the interval of checkpoint emission
103           // will be negative and no 'MirrorHeartbeatTask' will be created
104           if (!config.enabled() || knownConsumerGroups.isEmpty()
105                   || config.emitCheckpointsInterval().isNegative()) {
106               return Collections.emptyList();
107           }
108           int numTasks = Math.min(maxTasks, knownConsumerGroups.size());
109           return ConnectorUtils.groupPartitions(knownConsumerGroups, numTasks).stream()
110                   .map(config::taskConfigForConsumerGroups)
111                   .collect(Collectors.toList());
112       }
113   
114       @Override
115       public ConfigDef config() {
116           return MirrorConnectorConfig.CONNECTOR_CONFIG_DEF;
117       }
118   
119       @Override
120       public String version() {
121           return AppInfoParser.getVersion();
122       }
123   
124       private void refreshConsumerGroups()
125               throws InterruptedException, ExecutionException {
126           List<String> consumerGroups = findConsumerGroups();
127           Set<String> newConsumerGroups = new HashSet<>();
128           newConsumerGroups.addAll(consumerGroups);
129           newConsumerGroups.removeAll(knownConsumerGroups);
130           Set<String> deadConsumerGroups = new HashSet<>();
131           deadConsumerGroups.addAll(knownConsumerGroups);
132           deadConsumerGroups.removeAll(consumerGroups);
133           if (!newConsumerGroups.isEmpty() || !deadConsumerGroups.isEmpty()) {
134               log.info("Found {} consumer groups for {}. {} are new. {} were removed. Previously had {}.",
135                       consumerGroups.size(), sourceAndTarget, newConsumerGroups.size(), deadConsumerGroups.size(),
136                       knownConsumerGroups.size());
137               log.debug("Found new consumer groups: {}", newConsumerGroups);
138               knownConsumerGroups = consumerGroups;
139               context.requestTaskReconfiguration();
140           }
141       }
142   
143       private void loadInitialConsumerGroups()
144               throws InterruptedException, ExecutionException {
145           knownConsumerGroups = findConsumerGroups();
146       }
147   
148       List<String> findConsumerGroups()
149               throws InterruptedException, ExecutionException {
150           return listConsumerGroups().stream()
151                   .map(ConsumerGroupListing::groupId)
152                   .filter(this::shouldReplicate)
153                   .collect(Collectors.toList());
154       }
155   
156       Collection<ConsumerGroupListing> listConsumerGroups()
157               throws InterruptedException, ExecutionException {
158           return sourceAdminClient.listConsumerGroups().valid().get();
159       }
160   
161       private void createInternalTopics() {
162           MirrorUtils.createSinglePartitionCompactedTopic(config.checkpointsTopic(),
163               config.checkpointsTopicReplicationFactor(), config.targetAdminConfig());
164       } 
165   
166       boolean shouldReplicate(String group) {
167           return groupFilter.shouldReplicateGroup(group);
168       }
169   }
