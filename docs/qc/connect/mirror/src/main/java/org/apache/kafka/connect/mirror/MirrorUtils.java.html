52       static Map<String, Object> wrapPartition(TopicPartition topicPartition, String sourceClusterAlias) {
53           Map<String, Object> wrapped = new HashMap<>();
54           wrapped.put("topic", topicPartition.topic());
55           wrapped.put("partition", topicPartition.partition());
56           wrapped.put("cluster", sourceClusterAlias);
57           return wrapped;
58       }

64       static TopicPartition unwrapPartition(Map<String, ?> wrapped) {
65           String topic = (String) wrapped.get("topic");
66           int partition = (Integer) wrapped.get("partition");
67           return new TopicPartition(topic, partition);
68       }

70       static Long unwrapOffset(Map<String, ?> wrapped) {
71           if (wrapped == null || wrapped.get("offset") == null) {
72               return -1L;
73           }
74           return (Long) wrapped.get("offset");
75       }

77       static TopicPartition decodeTopicPartition(String topicPartitionString) {
78           int sep = topicPartitionString.lastIndexOf('-');
79           String topic = topicPartitionString.substring(0, sep);
80           String partitionString = topicPartitionString.substring(sep + 1);
81           int partition = Integer.parseInt(partitionString);
82           return new TopicPartition(topic, partition);
83       }

85       // returns null if given empty list
86       static Pattern compilePatternList(List<String> fields) {
87           if (fields.isEmpty()) {
88               // The empty pattern matches _everything_, but a blank
89               // config property should match _nothing_.
90               return null;
91           } else {
92               String joined = String.join("|", fields);
93               return Pattern.compile(joined);
94           }
95       }
96   
97       static Pattern compilePatternList(String fields) {
98           return compilePatternList(Arrays.asList(fields.split("\\W*,\\W*")));
99       }

101       static void createCompactedTopic(String topicName, short partitions, short replicationFactor, Map<String, Object> adminProps) {
102           NewTopic topicDescription = TopicAdmin.defineTopic(topicName).
103                   compacted().
104                   partitions(partitions).
105                   replicationFactor(replicationFactor).
106                   build();
107   
108           try (TopicAdmin admin = new TopicAdmin(adminProps)) {
109               admin.createTopics(topicDescription);
110           }
111       }
112   
113       static void createSinglePartitionCompactedTopic(String topicName, short replicationFactor, Map<String, Object> adminProps) {
114           createCompactedTopic(topicName, (short) 1, replicationFactor, adminProps);
115       }
