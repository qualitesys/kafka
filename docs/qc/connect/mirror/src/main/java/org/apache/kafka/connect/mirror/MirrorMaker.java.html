62   /**
63    *  Entry point for "MirrorMaker 2.0".
64    *  <p>
65    *  MirrorMaker runs a set of Connectors between multiple clusters, in order to replicate data, configuration,
66    *  ACL rules, and consumer group state.
67    *  </p>
68    *  <p>
69    *  Configuration is via a top-level "mm2.properties" file, which supports per-cluster and per-replication
70    *  sub-configs. Each source->target replication must be explicitly enabled. For example:
71    *  </p>
72    *  <pre>
73    *    clusters = primary, backup
74    *    primary.bootstrap.servers = vip1:9092
75    *    backup.bootstrap.servers = vip2:9092
76    *    primary->backup.enabled = true
77    *    backup->primary.enabled = true
78    *  </pre>
79    *  <p>
80    *  Run as follows:
81    *  </p>
82    *  <pre>
83    *    ./bin/connect-mirror-maker.sh mm2.properties
84    *  </pre>
85    *  <p>
86    *  Additional information and example configurations are provided in ./connect/mirror/README.md
87    *  </p>
88    */
89   public class MirrorMaker {
90       private static final Logger log = LoggerFactory.getLogger(MirrorMaker.class);
91   
92       private static final long SHUTDOWN_TIMEOUT_SECONDS = 60L;
93       private static final ConnectorClientConfigOverridePolicy CLIENT_CONFIG_OVERRIDE_POLICY =
94               new AllConnectorClientConfigOverridePolicy();
95   
96       private static final List<Class<?>> CONNECTOR_CLASSES = Arrays.asList(
97           MirrorSourceConnector.class,
98           MirrorHeartbeatConnector.class,
99           MirrorCheckpointConnector.class);
100    
101       private final Map<SourceAndTarget, Herder> herders = new HashMap<>();
102       private CountDownLatch startLatch;
103       private CountDownLatch stopLatch;
104       private final AtomicBoolean shutdown = new AtomicBoolean(false);
105       private final ShutdownHook shutdownHook;
106       private final String advertisedBaseUrl;
107       private final Time time;
108       private final MirrorMakerConfig config;
109       private final Set<String> clusters;
110       private final Set<SourceAndTarget> herderPairs;
111   
112       /**
113        * @param config    MM2 configuration from mm2.properties file
114        * @param clusters  target clusters for this node. These must match cluster
115        *                  aliases as defined in the config. If null or empty list,
116        *                  uses all clusters in the config.
117        * @param time      time source
118        */
119       public MirrorMaker(MirrorMakerConfig config, List<String> clusters, Time time) {
120           log.debug("Kafka MirrorMaker instance created");
121           this.time = time;
122           this.advertisedBaseUrl = "NOTUSED";
123           this.config = config;
124           if (clusters != null && !clusters.isEmpty()) {
125               this.clusters = new HashSet<>(clusters);
126           } else {
127               // default to all clusters
128               this.clusters = config.clusters();
129           }
130           log.info("Targeting clusters {}", this.clusters);
131           this.herderPairs = config.clusterPairs().stream()
132               .filter(x -> this.clusters.contains(x.target()))
133               .collect(Collectors.toSet());
134           if (herderPairs.isEmpty()) {
135               throw new IllegalArgumentException("No source->target replication flows.");
136           }
137           this.herderPairs.forEach(this::addHerder);
138           shutdownHook = new ShutdownHook();
139       }
140   
141       /**
142        * @param config    MM2 configuration from mm2.properties file
143        * @param clusters  target clusters for this node. These must match cluster
144        *                  aliases as defined in the config. If null or empty list,
145        *                  uses all clusters in the config.
146        * @param time      time source
147        */
148       public MirrorMaker(Map<String, String> config, List<String> clusters, Time time) {
149           this(new MirrorMakerConfig(config), clusters, time);
150       }

153           this(props, clusters, Time.SYSTEM);
154       }

156       public MirrorMaker(Map<String, String> props) {
157           this(props, null);
158       }

161       public void start() {
162           log.info("Kafka MirrorMaker starting with {} herders.", herders.size());
163           if (startLatch != null) {
164               throw new IllegalStateException("MirrorMaker instance already started");
165           }
166           startLatch = new CountDownLatch(herders.size());
167           stopLatch = new CountDownLatch(herders.size());
168           Exit.addShutdownHook("mirror-maker-shutdown-hook", shutdownHook);
169           for (Herder herder : herders.values()) {
170               try {
171                   herder.start();
172               } finally {
173                   startLatch.countDown();
174               }
175           }
176           log.info("Configuring connectors...");
177           herderPairs.forEach(this::configureConnectors);
178           log.info("Kafka MirrorMaker started");
179       }
180   
181       public void stop() {
182           boolean wasShuttingDown = shutdown.getAndSet(true);
183           if (!wasShuttingDown) {
184               log.info("Kafka MirrorMaker stopping");
185               for (Herder herder : herders.values()) {
186                   try {
187                       herder.stop();
188                   } finally {
189                       stopLatch.countDown();
190                   }
191               }
192               log.info("Kafka MirrorMaker stopped.");
193           }
194       }
195   
196       public void awaitStop() {
197           try {
198               stopLatch.await();
199           } catch (InterruptedException e) {
200               log.error("Interrupted waiting for MirrorMaker to shutdown");
201           }
202       }
203   
204       private void configureConnector(SourceAndTarget sourceAndTarget, Class<?> connectorClass) {
205           checkHerder(sourceAndTarget);
206           Map<String, String> connectorProps = config.connectorBaseConfig(sourceAndTarget, connectorClass);
207           herders.get(sourceAndTarget)
208                   .putConnectorConfig(connectorClass.getSimpleName(), connectorProps, true, (e, x) -> {
209                       if (e instanceof NotLeaderException) {
210                           // No way to determine if the connector is a leader or not beforehand.
211                           log.info("Connector {} is a follower. Using existing configuration.", sourceAndTarget);
212                       } else {
213                           log.info("Connector {} configured.", sourceAndTarget, e);
214                       }
215                   });
216       }
217   
218       private void checkHerder(SourceAndTarget sourceAndTarget) {
219           if (!herders.containsKey(sourceAndTarget)) {
220               throw new IllegalArgumentException("No herder for " + sourceAndTarget.toString());
221           }
222       }
223   
224       private void configureConnectors(SourceAndTarget sourceAndTarget) {
225           CONNECTOR_CLASSES.forEach(x -> configureConnector(sourceAndTarget, x));
226       }
227   
228       private void addHerder(SourceAndTarget sourceAndTarget) {
229           log.info("creating herder for " + sourceAndTarget.toString());
230           Map<String, String> workerProps = config.workerConfig(sourceAndTarget);
231           String advertisedUrl = advertisedBaseUrl + "/" + sourceAndTarget.source();
232           String workerId = sourceAndTarget.toString();
233           Plugins plugins = new Plugins(workerProps);
234           plugins.compareAndSwapWithDelegatingLoader();
235           DistributedConfig distributedConfig = new DistributedConfig(workerProps);
236           String kafkaClusterId = ConnectUtils.lookupKafkaClusterId(distributedConfig);
237           // Create the admin client to be shared by all backing stores for this herder
238           Map<String, Object> adminProps = new HashMap<>(distributedConfig.originals());
239           ConnectUtils.addMetricsContextProperties(adminProps, distributedConfig, kafkaClusterId);
240           SharedTopicAdmin sharedAdmin = new SharedTopicAdmin(adminProps);
241           KafkaOffsetBackingStore offsetBackingStore = new KafkaOffsetBackingStore(sharedAdmin);
242           offsetBackingStore.configure(distributedConfig);
243           Worker worker = new Worker(workerId, time, plugins, distributedConfig, offsetBackingStore, CLIENT_CONFIG_OVERRIDE_POLICY);
244           WorkerConfigTransformer configTransformer = worker.configTransformer();
245           Converter internalValueConverter = worker.getInternalValueConverter();
246           StatusBackingStore statusBackingStore = new KafkaStatusBackingStore(time, internalValueConverter, sharedAdmin);
247           statusBackingStore.configure(distributedConfig);
248           ConfigBackingStore configBackingStore = new KafkaConfigBackingStore(
249                   internalValueConverter,
250                   distributedConfig,
251                   configTransformer,
252                   sharedAdmin);
253           // Pass the shared admin to the distributed herder as an additional AutoCloseable object that should be closed when the
254           // herder is stopped. MirrorMaker has multiple herders, and having the herder own the close responsibility is much easier than
255           // tracking the various shared admin objects in this class.
256           Herder herder = new DistributedHerder(distributedConfig, time, worker,
257                   kafkaClusterId, statusBackingStore, configBackingStore,
258                   advertisedUrl, CLIENT_CONFIG_OVERRIDE_POLICY, sharedAdmin);
259           herders.put(sourceAndTarget, herder);
260       }
261   
262       private class ShutdownHook extends Thread {
263           @Override
264           public void run() {
265               try {
266                   if (!startLatch.await(SHUTDOWN_TIMEOUT_SECONDS, TimeUnit.SECONDS)) {
267                       log.error("Timed out in shutdown hook waiting for MirrorMaker startup to finish. Unable to shutdown cleanly.");
268                   }
269               } catch (InterruptedException e) {
270                   log.error("Interrupted in shutdown hook while waiting for MirrorMaker startup to finish. Unable to shutdown cleanly.");
271               } finally {
272                   MirrorMaker.this.stop();
273               }
274           }
275       }

277       public static void main(String[] args) {
278           ArgumentParser parser = ArgumentParsers.newArgumentParser("connect-mirror-maker");
279           parser.description("MirrorMaker 2.0 driver");
280           parser.addArgument("config").type(Arguments.fileType().verifyCanRead())
281               .metavar("mm2.properties").required(true)
282               .help("MM2 configuration file.");
283           parser.addArgument("--clusters").nargs("+").metavar("CLUSTER").required(false)
284               .help("Target cluster to use for this node.");
285           Namespace ns;
286           try {
287               ns = parser.parseArgs(args);
288           } catch (ArgumentParserException e) {
289               parser.handleError(e);
290               Exit.exit(-1);
291               return;
292           }
293           File configFile = ns.get("config");
294           List<String> clusters = ns.getList("clusters");
295           try {
296               log.info("Kafka MirrorMaker initializing ...");
297   
298               Properties props = Utils.loadProps(configFile.getPath());
299               Map<String, String> config = Utils.propsToStringMap(props);
300               MirrorMaker mirrorMaker = new MirrorMaker(config, clusters, Time.SYSTEM);
301               
302               try {
303                   mirrorMaker.start();
304               } catch (Exception e) {
305                   log.error("Failed to start MirrorMaker", e);
306                   mirrorMaker.stop();
307                   Exit.exit(3);
308               }
309   
310               mirrorMaker.awaitStop();
311   
312           } catch (Throwable t) {
313               log.error("Stopping due to error", t);
314               Exit.exit(2);
315           }
316       }
