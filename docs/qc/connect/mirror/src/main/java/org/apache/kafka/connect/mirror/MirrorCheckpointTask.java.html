133       @Override
134       public List<SourceRecord> poll() throws InterruptedException {
135           try {
136               long deadline = System.currentTimeMillis() + interval.toMillis();
137               while (!stopping && System.currentTimeMillis() < deadline) {
138                   offsetSyncStore.update(pollTimeout);
139               }
140               List<SourceRecord> records = new ArrayList<>();
141               for (String group : consumerGroups) {
142                   records.addAll(sourceRecordsForGroup(group));
143               }
144               if (records.isEmpty()) {
145                   // WorkerSourceTask expects non-zero batches or null
146                   return null;
147               } else {
148                   return records;
149               }
150           } catch (Throwable e) {
151               log.warn("Failure polling consumer state for checkpoints.", e);
152               return null;
153           }
154       }

157       private List<SourceRecord> sourceRecordsForGroup(String group) throws InterruptedException {
158           try {
159               long timestamp = System.currentTimeMillis();
160               List<Checkpoint> checkpoints = checkpointsForGroup(group);
161               checkpointsPerConsumerGroup.put(group, checkpoints);
162               return checkpoints.stream()
163                   .map(x -> checkpointRecord(x, timestamp))
164                   .collect(Collectors.toList());
165           } catch (ExecutionException e) {
166               log.error("Error querying offsets for consumer group {} on cluster {}.",  group, sourceClusterAlias, e);
167               return Collections.emptyList();
168           }
169       }
170   
171       private List<Checkpoint> checkpointsForGroup(String group) throws ExecutionException, InterruptedException {
172           return listConsumerGroupOffsets(group).entrySet().stream()
173               .filter(x -> shouldCheckpointTopic(x.getKey().topic()))
174               .map(x -> checkpoint(group, x.getKey(), x.getValue()))
175               .flatMap(o -> o.map(Stream::of).orElseGet(Stream::empty)) // do not emit checkpoints for partitions that don't have offset-syncs
176               .filter(x -> x.downstreamOffset() >= 0)  // ignore offsets we cannot translate accurately
177               .collect(Collectors.toList());
178       }

180       private Map<TopicPartition, OffsetAndMetadata> listConsumerGroupOffsets(String group)
181               throws InterruptedException, ExecutionException {
182           if (stopping) {
183               // short circuit if stopping
184               return Collections.emptyMap();
185           }
186           return sourceAdminClient.listConsumerGroupOffsets(group).partitionsToOffsetAndMetadata().get();
187       }

189       Optional<Checkpoint> checkpoint(String group, TopicPartition topicPartition,
190                                       OffsetAndMetadata offsetAndMetadata) {
191           long upstreamOffset = offsetAndMetadata.offset();
192           OptionalLong downstreamOffset = offsetSyncStore.translateDownstream(topicPartition, upstreamOffset);
193           if (downstreamOffset.isPresent()) {
194               return Optional.of(new Checkpoint(group, renameTopicPartition(topicPartition),
195                       upstreamOffset, downstreamOffset.getAsLong(), offsetAndMetadata.metadata()));
196           } else {
197               return Optional.empty();
198           }
199       }
200   
201       SourceRecord checkpointRecord(Checkpoint checkpoint, long timestamp) {
202           return new SourceRecord(
203               checkpoint.connectPartition(), MirrorUtils.wrapOffset(0),
204               checkpointsTopic, 0,
205               Schema.BYTES_SCHEMA, checkpoint.recordKey(),
206               Schema.BYTES_SCHEMA, checkpoint.recordValue(),
207               timestamp);
208       }

210       TopicPartition renameTopicPartition(TopicPartition upstreamTopicPartition) {
211           if (targetClusterAlias.equals(replicationPolicy.topicSource(upstreamTopicPartition.topic()))) {
212               // this topic came from the target cluster, so we rename like us-west.topic1 -> topic1
213               return new TopicPartition(replicationPolicy.originalTopic(upstreamTopicPartition.topic()),
214                   upstreamTopicPartition.partition());
215           } else {
216               // rename like topic1 -> us-west.topic1
217               return new TopicPartition(replicationPolicy.formatRemoteTopic(sourceClusterAlias,
218                   upstreamTopicPartition.topic()), upstreamTopicPartition.partition());
219           }
220       }

222       boolean shouldCheckpointTopic(String topic) {
223           return topicFilter.shouldReplicateTopic(topic);
224       }

233       private void refreshIdleConsumerGroupOffset() {
234           Map<String, KafkaFuture<ConsumerGroupDescription>> consumerGroupsDesc = targetAdminClient
235               .describeConsumerGroups(consumerGroups).describedGroups();
236   
237           for (String group : consumerGroups) {
238               try {
239                   ConsumerGroupDescription consumerGroupDesc = consumerGroupsDesc.get(group).get();
240                   ConsumerGroupState consumerGroupState = consumerGroupDesc.state();
241                   // sync offset to the target cluster only if the state of current consumer group is:
242                   // (1) idle: because the consumer at target is not actively consuming the mirrored topic
243                   // (2) dead: the new consumer that is recently created at source and never existed at target
244                   if (consumerGroupState == ConsumerGroupState.EMPTY) {
245                       idleConsumerGroupsOffset.put(group, targetAdminClient.listConsumerGroupOffsets(group)
246                           .partitionsToOffsetAndMetadata().get());
247                   }
248                   // new consumer upstream has state "DEAD" and will be identified during the offset sync-up
249               } catch (InterruptedException | ExecutionException e) {
250                   log.error("Error querying for consumer group {} on cluster {}.", group, targetClusterAlias, e);
251               }
252           }
253       }

255       Map<String, Map<TopicPartition, OffsetAndMetadata>> syncGroupOffset() {
256           Map<String, Map<TopicPartition, OffsetAndMetadata>> offsetToSyncAll = new HashMap<>();
257   
258           // first, sync offsets for the idle consumers at target
259           for (Entry<String, Map<TopicPartition, OffsetAndMetadata>> group : getConvertedUpstreamOffset().entrySet()) {
260               String consumerGroupId = group.getKey();
261               // for each idle consumer at target, read the checkpoints (converted upstream offset)
262               // from the pre-populated map
263               Map<TopicPartition, OffsetAndMetadata> convertedUpstreamOffset = group.getValue();
264   
265               Map<TopicPartition, OffsetAndMetadata> offsetToSync = new HashMap<>();
266               Map<TopicPartition, OffsetAndMetadata> targetConsumerOffset = idleConsumerGroupsOffset.get(consumerGroupId);
267               if (targetConsumerOffset == null) {
268                   // this is a new consumer, just sync the offset to target
269                   syncGroupOffset(consumerGroupId, convertedUpstreamOffset);
270                   offsetToSyncAll.put(consumerGroupId, convertedUpstreamOffset);
271                   continue;
272               }
273   
274               for (Entry<TopicPartition, OffsetAndMetadata> convertedEntry : convertedUpstreamOffset.entrySet()) {
275   
276                   TopicPartition topicPartition = convertedEntry.getKey();
277                   OffsetAndMetadata convertedOffset = convertedUpstreamOffset.get(topicPartition);
278                   if (!targetConsumerOffset.containsKey(topicPartition)) {
279                       // if is a new topicPartition from upstream, just sync the offset to target
280                       offsetToSync.put(topicPartition, convertedOffset);
281                       continue;
282                   }
283   
284                   // if translated offset from upstream is smaller than the current consumer offset
285                   // in the target, skip updating the offset for that partition
286                   long latestDownstreamOffset = targetConsumerOffset.get(topicPartition).offset();
287                   if (latestDownstreamOffset >= convertedOffset.offset()) {
288                       log.trace("latestDownstreamOffset {} is larger than or equal to convertedUpstreamOffset {} for "
289                           + "TopicPartition {}", latestDownstreamOffset, convertedOffset.offset(), topicPartition);
290                       continue;
291                   }
292                   offsetToSync.put(topicPartition, convertedOffset);
293               }
294   
295               if (offsetToSync.size() == 0) {
296                   log.trace("skip syncing the offset for consumer group: {}", consumerGroupId);
297                   continue;
298               }
299               syncGroupOffset(consumerGroupId, offsetToSync);
300   
301               offsetToSyncAll.put(consumerGroupId, offsetToSync);
302           }
303           idleConsumerGroupsOffset.clear();
304           return offsetToSyncAll;
305       }
306   
307       void syncGroupOffset(String consumerGroupId, Map<TopicPartition, OffsetAndMetadata> offsetToSync) {
308           if (targetAdminClient != null) {
309               targetAdminClient.alterConsumerGroupOffsets(consumerGroupId, offsetToSync);
310               log.trace("sync-ed the offset for consumer group: {} with {} number of offset entries",
311                         consumerGroupId, offsetToSync.size());
312           }
313       }
314   
315       Map<String, Map<TopicPartition, OffsetAndMetadata>> getConvertedUpstreamOffset() {
316           Map<String, Map<TopicPartition, OffsetAndMetadata>> result = new HashMap<>();
317   
318           for (Entry<String, List<Checkpoint>> entry : checkpointsPerConsumerGroup.entrySet()) {
319               String consumerId = entry.getKey();
320               Map<TopicPartition, OffsetAndMetadata> convertedUpstreamOffset = new HashMap<>();
321               for (Checkpoint checkpoint : entry.getValue()) {
322                   convertedUpstreamOffset.put(checkpoint.topicPartition(), checkpoint.offsetAndMetadata());
323               }
324               result.put(consumerId, convertedUpstreamOffset);
325           }
326           return result;
327       }
