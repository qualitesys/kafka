48   /** Replicates a set of topic-partitions. */
49   public class MirrorSourceTask extends SourceTask {

53       private static final int MAX_OUTSTANDING_OFFSET_SYNCS = 10;
54

81       @Override
82       public void start(Map<String, String> props) {
83           MirrorTaskConfig config = new MirrorTaskConfig(props);
84           outstandingOffsetSyncs = new Semaphore(MAX_OUTSTANDING_OFFSET_SYNCS);
85           consumerAccess = new Semaphore(1);  // let one thread at a time access the consumer
86           sourceClusterAlias = config.sourceClusterAlias();
87           metrics = config.metrics();
88           pollTimeout = config.consumerPollTimeout();
89           maxOffsetLag = config.maxOffsetLag();
90           replicationPolicy = config.replicationPolicy();
91           partitionStates = new HashMap<>();
92           offsetSyncsTopic = config.offsetSyncsTopic();
93           consumer = MirrorUtils.newConsumer(config.sourceConsumerConfig());
94           offsetProducer = MirrorUtils.newProducer(config.offsetSyncsTopicProducerConfig());
95           Set<TopicPartition> taskTopicPartitions = config.taskTopicPartitions();
96           Map<TopicPartition, Long> topicPartitionOffsets = loadOffsets(taskTopicPartitions);
97           consumer.assign(topicPartitionOffsets.keySet());
98           log.info("Starting with {} previously uncommitted partitions.", topicPartitionOffsets.entrySet().stream()
99               .filter(x -> x.getValue() == 0L).count());
100           log.trace("Seeking offsets: {}", topicPartitionOffsets);
101           topicPartitionOffsets.forEach(consumer::seek);
102           log.info("{} replicating {} topic-partitions {}->{}: {}.", Thread.currentThread().getName(),
103               taskTopicPartitions.size(), sourceClusterAlias, config.targetClusterAlias(), taskTopicPartitions);
104       }

111       @Override
112       public void stop() {
113           long start = System.currentTimeMillis();
114           stopping = true;
115           consumer.wakeup();
116           try {
117               consumerAccess.acquire();
118           } catch (InterruptedException e) {
119               log.warn("Interrupted waiting for access to consumer. Will try closing anyway."); 
120           }
121           Utils.closeQuietly(consumer, "source consumer");
122           Utils.closeQuietly(offsetProducer, "offset producer");
123           Utils.closeQuietly(metrics, "metrics");
124           log.info("Stopping {} took {} ms.", Thread.currentThread().getName(), System.currentTimeMillis() - start);
125       }

132       @Override
133       public List<SourceRecord> poll() {
134           if (!consumerAccess.tryAcquire()) {
135               return null;
136           }
137           if (stopping) {
138               return null;
139           }
140           try {
141               ConsumerRecords<byte[], byte[]> records = consumer.poll(pollTimeout);
142               List<SourceRecord> sourceRecords = new ArrayList<>(records.count());
143               for (ConsumerRecord<byte[], byte[]> record : records) {
144                   SourceRecord converted = convertRecord(record);
145                   sourceRecords.add(converted);
146                   TopicPartition topicPartition = new TopicPartition(converted.topic(), converted.kafkaPartition());
147                   metrics.recordAge(topicPartition, System.currentTimeMillis() - record.timestamp());
148                   metrics.recordBytes(topicPartition, byteSize(record.value()));
149               }
150               if (sourceRecords.isEmpty()) {
151                   // WorkerSourceTasks expects non-zero batch size
152                   return null;
153               } else {
154                   log.trace("Polled {} records from {}.", sourceRecords.size(), records.partitions());
155                   return sourceRecords;
156               }
157           } catch (WakeupException e) {
158               return null;
159           } catch (KafkaException e) {
160               log.warn("Failure during poll.", e);
161               return null;
162           } catch (Throwable e)  {
163               log.error("Failure during poll.", e);
164               // allow Connect to deal with the exception
165               throw e;
166           } finally {
167               consumerAccess.release();
168           }
169       }

171       @Override
172       public void commitRecord(SourceRecord record, RecordMetadata metadata) {
173           try {
174               if (stopping) {
175                   return;
176               }
177               if (!metadata.hasOffset()) {
178                   log.error("RecordMetadata has no offset -- can't sync offsets for {}.", record.topic());
179                   return;
180               }
181               TopicPartition topicPartition = new TopicPartition(record.topic(), record.kafkaPartition());
182               long latency = System.currentTimeMillis() - record.timestamp();
183               metrics.countRecord(topicPartition);
184               metrics.replicationLatency(topicPartition, latency);
185               TopicPartition sourceTopicPartition = MirrorUtils.unwrapPartition(record.sourcePartition());
186               long upstreamOffset = MirrorUtils.unwrapOffset(record.sourceOffset());
187               long downstreamOffset = metadata.offset();
188               maybeSyncOffsets(sourceTopicPartition, upstreamOffset, downstreamOffset);
189           } catch (Throwable e) {
190               log.warn("Failure committing record.", e);
191           }
192       }
193   
194       // updates partition state and sends OffsetSync if necessary
195       private void maybeSyncOffsets(TopicPartition topicPartition, long upstreamOffset,
196               long downstreamOffset) {
197           PartitionState partitionState =
198               partitionStates.computeIfAbsent(topicPartition, x -> new PartitionState(maxOffsetLag));
199           if (partitionState.update(upstreamOffset, downstreamOffset)) {
200               sendOffsetSync(topicPartition, upstreamOffset, downstreamOffset);
201           }
202       }
203   
204       // sends OffsetSync record upstream to internal offsets topic
205       private void sendOffsetSync(TopicPartition topicPartition, long upstreamOffset,
206               long downstreamOffset) {
207           if (!outstandingOffsetSyncs.tryAcquire()) {
208               // Too many outstanding offset syncs.
209               return;
210           }
211           OffsetSync offsetSync = new OffsetSync(topicPartition, upstreamOffset, downstreamOffset);
212           ProducerRecord<byte[], byte[]> record = new ProducerRecord<>(offsetSyncsTopic, 0,
213                   offsetSync.recordKey(), offsetSync.recordValue());
214           offsetProducer.send(record, (x, e) -> {
215               if (e != null) {
216                   log.error("Failure sending offset sync.", e);
217               } else {
218                   log.trace("Sync'd offsets for {}: {}=={}", topicPartition,
219                       upstreamOffset, downstreamOffset);
220               }
221               outstandingOffsetSyncs.release();
222           });
223       }

225       private Map<TopicPartition, Long> loadOffsets(Set<TopicPartition> topicPartitions) {
226           return topicPartitions.stream().collect(Collectors.toMap(x -> x, this::loadOffset));
227       }
228   
229       private Long loadOffset(TopicPartition topicPartition) {
230           Map<String, Object> wrappedPartition = MirrorUtils.wrapPartition(topicPartition, sourceClusterAlias);
231           Map<String, Object> wrappedOffset = context.offsetStorageReader().offset(wrappedPartition);
232           return MirrorUtils.unwrapOffset(wrappedOffset) + 1;
233       }

235       // visible for testing 
236       SourceRecord convertRecord(ConsumerRecord<byte[], byte[]> record) {
237           String targetTopic = formatRemoteTopic(record.topic());
238           Headers headers = convertHeaders(record);
239           return new SourceRecord(
240                   MirrorUtils.wrapPartition(new TopicPartition(record.topic(), record.partition()), sourceClusterAlias),
241                   MirrorUtils.wrapOffset(record.offset()),
242                   targetTopic, record.partition(),
243                   Schema.OPTIONAL_BYTES_SCHEMA, record.key(),
244                   Schema.BYTES_SCHEMA, record.value(),
245                   record.timestamp(), headers);
246       }
247   
248       private Headers convertHeaders(ConsumerRecord<byte[], byte[]> record) {
249           ConnectHeaders headers = new ConnectHeaders();
250           for (Header header : record.headers()) {
251               headers.addBytes(header.key(), header.value());
252           }
253           return headers;
254       }

256       private String formatRemoteTopic(String topic) {
257           return replicationPolicy.formatRemoteTopic(sourceClusterAlias, topic);
258       }

260       private static int byteSize(byte[] bytes) {
261           if (bytes == null) {
262               return 0;
263           } else {
264               return bytes.length;
265           }

268       static class PartitionState {
269           long previousUpstreamOffset = -1L;
270           long previousDownstreamOffset = -1L;
271           long lastSyncUpstreamOffset = -1L;
272           long lastSyncDownstreamOffset = -1L;
273           long maxOffsetLag;
274   
275           PartitionState(long maxOffsetLag) {
276               this.maxOffsetLag = maxOffsetLag;
277           }
278   
279           // true if we should emit an offset sync
280           boolean update(long upstreamOffset, long downstreamOffset) {
281               boolean shouldSyncOffsets = false;
282               long upstreamStep = upstreamOffset - lastSyncUpstreamOffset;
283               long downstreamTargetOffset = lastSyncDownstreamOffset + upstreamStep;
284               if (lastSyncDownstreamOffset == -1L
285                       || downstreamOffset - downstreamTargetOffset >= maxOffsetLag
286                       || upstreamOffset - previousUpstreamOffset != 1L
287                       || downstreamOffset < previousDownstreamOffset) {
288                   lastSyncUpstreamOffset = upstreamOffset;
289                   lastSyncDownstreamOffset = downstreamOffset;
290                   shouldSyncOffsets = true;
291               }
292               previousUpstreamOffset = upstreamOffset;
293               previousDownstreamOffset = downstreamOffset;
294               return shouldSyncOffsets;
295           }
