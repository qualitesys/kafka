31   /** Checkpoint records emitted from MirrorCheckpointConnector. Encodes remote consumer group state. */
32   public class Checkpoint {
33       public static final String TOPIC_KEY = "topic";
34       public static final String PARTITION_KEY = "partition";
35       public static final String CONSUMER_GROUP_ID_KEY = "group";
36       public static final String UPSTREAM_OFFSET_KEY = "upstreamOffset";
37       public static final String DOWNSTREAM_OFFSET_KEY = "offset";
38       public static final String METADATA_KEY = "metadata";
39       public static final String VERSION_KEY = "version";
40       public static final short VERSION = 0;
41   
42       public static final Schema VALUE_SCHEMA_V0 = new Schema(
43               new Field(UPSTREAM_OFFSET_KEY, Type.INT64),
44               new Field(DOWNSTREAM_OFFSET_KEY, Type.INT64),
45               new Field(METADATA_KEY, Type.STRING));
46   
47       public static final Schema KEY_SCHEMA = new Schema(
48               new Field(CONSUMER_GROUP_ID_KEY, Type.STRING),
49               new Field(TOPIC_KEY, Type.STRING),
50               new Field(PARTITION_KEY, Type.INT32));
51   
52       public static final Schema HEADER_SCHEMA = new Schema(
53               new Field(VERSION_KEY, Type.INT16));
54   
55       private final String consumerGroupId;
56       private final TopicPartition topicPartition;
57       private final long upstreamOffset;
58       private final long downstreamOffset;
59       private final String metadata;
60   
61       public Checkpoint(String consumerGroupId, TopicPartition topicPartition, long upstreamOffset,
62               long downstreamOffset, String metadata) {
63           this.consumerGroupId = consumerGroupId;
64           this.topicPartition = topicPartition;
65           this.upstreamOffset = upstreamOffset;
66           this.downstreamOffset = downstreamOffset;
67           this.metadata = metadata;
68       }

71           return consumerGroupId;
72       }

75           return topicPartition;
76       }
77

79           return upstreamOffset;
80       }
81

83           return downstreamOffset;
84       }

86       public String metadata() {
87           return metadata;
88       }
89   
90       public OffsetAndMetadata offsetAndMetadata() {
91           return new OffsetAndMetadata(downstreamOffset, metadata);
92       }

94       @Override
95       public String toString() {
96           return String.format("Checkpoint{consumerGroupId=%s, topicPartition=%s, "
97               + "upstreamOffset=%d, downstreamOffset=%d, metatadata=%s}",
98               consumerGroupId, topicPartition, upstreamOffset, downstreamOffset, metadata);
99       }

101       ByteBuffer serializeValue(short version) {
102           Struct header = headerStruct(version);
103           Schema valueSchema = valueSchema(version);
104           Struct valueStruct = valueStruct(valueSchema);
105           ByteBuffer buffer = ByteBuffer.allocate(HEADER_SCHEMA.sizeOf(header) + valueSchema.sizeOf(valueStruct));
106           HEADER_SCHEMA.write(buffer, header);
107           valueSchema.write(buffer, valueStruct);
108           buffer.flip();
109           return buffer;
110       }
111   
112       ByteBuffer serializeKey() {
113           Struct struct = keyStruct();
114           ByteBuffer buffer = ByteBuffer.allocate(KEY_SCHEMA.sizeOf(struct));
115           KEY_SCHEMA.write(buffer, struct);
116           buffer.flip();
117           return buffer;
118       }
119   
120       public static Checkpoint deserializeRecord(ConsumerRecord<byte[], byte[]> record) {
121           ByteBuffer value = ByteBuffer.wrap(record.value());
122           Struct header = HEADER_SCHEMA.read(value);
123           short version = header.getShort(VERSION_KEY);
124           Schema valueSchema = valueSchema(version);
125           Struct valueStruct = valueSchema.read(value);
126           long upstreamOffset = valueStruct.getLong(UPSTREAM_OFFSET_KEY);
127           long downstreamOffset = valueStruct.getLong(DOWNSTREAM_OFFSET_KEY);
128           String metadata = valueStruct.getString(METADATA_KEY);
129           Struct keyStruct = KEY_SCHEMA.read(ByteBuffer.wrap(record.key()));
130           String group = keyStruct.getString(CONSUMER_GROUP_ID_KEY);
131           String topic = keyStruct.getString(TOPIC_KEY);
132           int partition = keyStruct.getInt(PARTITION_KEY);
133           return new Checkpoint(group, new TopicPartition(topic, partition), upstreamOffset,
134               downstreamOffset, metadata);
135       }
136   
137       private static Schema valueSchema(short version) {
138           assert version == 0;
139           return VALUE_SCHEMA_V0;
140       }
141   
142       private Struct valueStruct(Schema schema) {
143           Struct struct = new Struct(schema);
144           struct.set(UPSTREAM_OFFSET_KEY, upstreamOffset);
145           struct.set(DOWNSTREAM_OFFSET_KEY, downstreamOffset);
146           struct.set(METADATA_KEY, metadata);
147           return struct;
148       }
149   
150       private Struct keyStruct() {
151           Struct struct = new Struct(KEY_SCHEMA);
152           struct.set(CONSUMER_GROUP_ID_KEY, consumerGroupId);
153           struct.set(TOPIC_KEY, topicPartition.topic());
154           struct.set(PARTITION_KEY, topicPartition.partition());
155           return struct;
156       }
157   
158       private Struct headerStruct(short version) {
159           Struct struct = new Struct(HEADER_SCHEMA);
160           struct.set(VERSION_KEY, version);
161           return struct;
162       }
163   
164       Map<String, ?> connectPartition() {
165           Map<String, Object> partition = new HashMap<>();
166           partition.put(CONSUMER_GROUP_ID_KEY, consumerGroupId);
167           partition.put(TOPIC_KEY, topicPartition.topic());
168           partition.put(PARTITION_KEY, topicPartition.partition());
169           return partition;
170       }

172       static String unwrapGroup(Map<String, ?> connectPartition) {
173           return connectPartition.get(CONSUMER_GROUP_ID_KEY).toString();
174       }

180       byte[] recordValue() {
181           return serializeValue(VERSION).array();
