45   /** Interprets MM2's internal topics (checkpoints, heartbeats) on a given cluster.
46    *  <p> 
47    *  Given a top-level "mm2.properties" configuration file, MirrorClients can be constructed
48    *  for individual clusters as follows:
49    *  </p> 
50    *  <pre>
51    *    MirrorMakerConfig mmConfig = new MirrorMakerConfig(props);
52    *    MirrorClientConfig mmClientConfig = mmConfig.clientConfig("some-cluster");
53    *    MirrorClient mmClient = new Mirrorclient(mmClientConfig);
54    *  </pre>
55    */
56   public class MirrorClient implements AutoCloseable {
57       private static final Logger log = LoggerFactory.getLogger(MirrorClient.class);
58   
59       private final AdminClient adminClient;
60       private final ReplicationPolicy replicationPolicy;
61       private final Map<String, Object> consumerConfig;
62   
63       public MirrorClient(Map<String, Object> props) {
64           this(new MirrorClientConfig(props));
65       }

68           adminClient = AdminClient.create(config.adminConfig());
69           consumerConfig = config.consumerConfig();
70           replicationPolicy = config.replicationPolicy();
71       }

86       /** Get the ReplicationPolicy instance used to interpret remote topics. This instance is constructed based on
87        *  relevant configuration properties, including {@code replication.policy.class}. */
88       public ReplicationPolicy replicationPolicy() {
89           return replicationPolicy;
90       }

92       /** Compute shortest number of hops from an upstream source cluster.
93        *  For example, given replication flow A-&gt;B-&gt;C, there are two hops from A to C.
94        *  Returns -1 if upstream cluster is unreachable.
95        */
96       public int replicationHops(String upstreamClusterAlias) throws InterruptedException {
97           return heartbeatTopics().stream()
98               .map(x -> countHopsForTopic(x, upstreamClusterAlias))
99               .filter(x -> x != -1)
100               .mapToInt(x -> x)
101               .min()
102               .orElse(-1);
103       }
104   
105       /** Find all heartbeat topics on this cluster. Heartbeat topics are replicated from other clusters. */
106       public Set<String> heartbeatTopics() throws InterruptedException {
107           return listTopics().stream()
108               .filter(this::isHeartbeatTopic)
109               .collect(Collectors.toSet());
110       }
111   
112       /** Find all checkpoint topics on this cluster. */
113       public Set<String> checkpointTopics() throws InterruptedException {
114           return listTopics().stream()
115               .filter(this::isCheckpointTopic)
116               .collect(Collectors.toSet());
117       }
118   
119       /** Find upstream clusters, which may be multiple hops away, based on incoming heartbeats. */
120       public Set<String> upstreamClusters() throws InterruptedException {
121           return listTopics().stream()
122               .filter(this::isHeartbeatTopic)
123               .flatMap(x -> allSources(x).stream())
124               .collect(Collectors.toSet());
125       }
126   
127       /** Find all remote topics on this cluster. This does not include internal topics (heartbeats, checkpoints). */
128       public Set<String> remoteTopics() throws InterruptedException {
129           return listTopics().stream()
130               .filter(this::isRemoteTopic)
131               .collect(Collectors.toSet());
132       }
133   
134       /** Find all remote topics that have been replicated directly from the given source cluster. */
135       public Set<String> remoteTopics(String source) throws InterruptedException {
136           return listTopics().stream()
137               .filter(this::isRemoteTopic)
138               .filter(x -> source.equals(replicationPolicy.topicSource(x)))
139               .collect(Collectors.toSet());
140       }

142       /** Translate a remote consumer group's offsets into corresponding local offsets. Topics are automatically
143        *  renamed according to the ReplicationPolicy.
144        *  @param consumerGroupId group ID of remote consumer group
145        *  @param remoteClusterAlias alias of remote cluster
146        *  @param timeout timeout
147        */
148       public Map<TopicPartition, OffsetAndMetadata> remoteConsumerOffsets(String consumerGroupId,
149               String remoteClusterAlias, Duration timeout) {
150           long deadline = System.currentTimeMillis() + timeout.toMillis();
151           Map<TopicPartition, OffsetAndMetadata> offsets = new HashMap<>();
152   
153           try (KafkaConsumer<byte[], byte[]> consumer = new KafkaConsumer<>(consumerConfig,
154                   new ByteArrayDeserializer(), new ByteArrayDeserializer())) {
155               // checkpoint topics are not "remote topics", as they are not replicated. So we don't need
156               // to use ReplicationPolicy to create the checkpoint topic here.
157               String checkpointTopic = replicationPolicy.checkpointsTopic(remoteClusterAlias);
158               List<TopicPartition> checkpointAssignment =
159                   Collections.singletonList(new TopicPartition(checkpointTopic, 0));
160               consumer.assign(checkpointAssignment);
161               consumer.seekToBeginning(checkpointAssignment);
162               while (System.currentTimeMillis() < deadline && !endOfStream(consumer, checkpointAssignment)) {
163                   ConsumerRecords<byte[], byte[]> records = consumer.poll(timeout);
164                   for (ConsumerRecord<byte[], byte[]> record : records) {
165                       try {
166                           Checkpoint checkpoint = Checkpoint.deserializeRecord(record);
167                           if (checkpoint.consumerGroupId().equals(consumerGroupId)) {
168                               offsets.put(checkpoint.topicPartition(), checkpoint.offsetAndMetadata());
169                           }
170                       } catch (SchemaException e) {
171                           log.info("Could not deserialize record. Skipping.", e);
172                       }
173                   }
174               }
175               log.info("Consumed {} checkpoint records for {} from {}.", offsets.size(),
176                   consumerGroupId, checkpointTopic);
177           }
178           return offsets;
179       }

181       Set<String> listTopics() throws InterruptedException {
182           try {
183               return adminClient.listTopics().names().get();
184           } catch (ExecutionException e) {
185               throw new KafkaException(e.getCause());
186           }
187       }

189       int countHopsForTopic(String topic, String sourceClusterAlias) {
190           int hops = 0;
191           Set<String> visited = new HashSet<>();
192           while (true) {
193               hops++;
194               String source = replicationPolicy.topicSource(topic);
195               if (source == null) {
196                   return -1;
197               }
198               if (source.equals(sourceClusterAlias)) {
199                   return hops;
200               }
201               if (visited.contains(source)) {
202                   // Extra check for IdentityReplicationPolicy and similar impls that cannot prevent cycles.
203                   // We assume we're stuck in a cycle and will never find sourceClusterAlias.
204                   return -1;
205               }
206               visited.add(source);
207               topic = replicationPolicy.upstreamTopic(topic);
208           }
209       }

211       boolean isHeartbeatTopic(String topic) {
212           return replicationPolicy.isHeartbeatsTopic(topic);
213       }
214   
215       boolean isCheckpointTopic(String topic) {
216           return replicationPolicy.isCheckpointsTopic(topic);
217       }
218   
219       boolean isRemoteTopic(String topic) {
220           return !replicationPolicy.isInternalTopic(topic)
221               && replicationPolicy.topicSource(topic) != null;
222       }
223   
224       Set<String> allSources(String topic) {
225           Set<String> sources = new HashSet<>();
226           String source = replicationPolicy.topicSource(topic);
227           while (source != null && !sources.contains(source)) {
228               // The extra Set.contains above is for ReplicationPolicies that cannot prevent cycles.
229               sources.add(source);
230               topic = replicationPolicy.upstreamTopic(topic);
231               source = replicationPolicy.topicSource(topic);
232           }
233           return sources;
234       }

236       static private boolean endOfStream(Consumer<?, ?> consumer, Collection<TopicPartition> assignments) {
237           Map<TopicPartition, Long> endOffsets = consumer.endOffsets(assignments);
238           for (TopicPartition topicPartition : assignments) {
239               if (consumer.position(topicPartition) < endOffsets.get(topicPartition)) {
240                   return false;
241               }
242           }
243           return true;
244       }
