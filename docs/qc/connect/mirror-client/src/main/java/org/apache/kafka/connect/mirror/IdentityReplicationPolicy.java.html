17   package org.apache.kafka.connect.mirror;
18   
19   import java.util.Map;
20   
21   import org.slf4j.Logger;
22   import org.slf4j.LoggerFactory;
23   
24   /** IdentityReplicationPolicy does not rename remote topics. This is useful for migrating
25     * from legacy MM1, or for any use-case involving one-way replication.
26     *
27     * N.B. MirrorMaker is not able to prevent cycles when using this class, so take care that
28     * your replication topology is acyclic. If migrating from MirrorMaker v1, this will likely
29     * already be the case.
30     */
31   public class IdentityReplicationPolicy extends DefaultReplicationPolicy {
32       private static final Logger log = LoggerFactory.getLogger(IdentityReplicationPolicy.class);
33   
34       public static final String SOURCE_CLUSTER_ALIAS_CONFIG = "source.cluster.alias";
35   
36       private String sourceClusterAlias = null;
37   
38       @Override
39       public void configure(Map<String, ?> props) {
40           super.configure(props);
41           if (props.containsKey(SOURCE_CLUSTER_ALIAS_CONFIG)) {
42               sourceClusterAlias = (String) props.get(SOURCE_CLUSTER_ALIAS_CONFIG);
43               log.info("Using source cluster alias `{}`.", sourceClusterAlias);
44           }
45       }
46   
47       /** Unlike DefaultReplicationPolicy, IdentityReplicationPolicy does not include the source
48         * cluster alias in the remote topic name. Instead, topic names are unchanged.
49         *
50         * In the special case of heartbeats, we defer to DefaultReplicationPolicy.
51         */
52       @Override
53       public String formatRemoteTopic(String sourceClusterAlias, String topic) {
54           if (looksLikeHeartbeat(topic)) {
55               return super.formatRemoteTopic(sourceClusterAlias, topic);
56           } else {
57               return topic;
58           }
59       }
60   
61       /** Unlike DefaultReplicationPolicy, IdentityReplicationPolicy cannot know the source of
62         * a remote topic based on its name alone. If `source.cluster.alias` is provided,
63         * `topicSource` will return that.
64         *
65         * In the special case of heartbeats, we defer to DefaultReplicationPolicy.
66         */
67       @Override
68       public String topicSource(String topic) {
69           if (looksLikeHeartbeat(topic)) {
70               return super.topicSource(topic);
71           } else {
72               return sourceClusterAlias;
73           }
74       }
75   
76       /** Since any topic may be a "remote topic", this just returns `topic`.
77         *
78         * In the special case of heartbeats, we defer to DefaultReplicationPolicy.
79         */
80       @Override
81       public String upstreamTopic(String topic) {
82           if (looksLikeHeartbeat(topic)) {
83               return super.upstreamTopic(topic);
84           } else {
85               return topic;
86           }
87       }
88   
89       private boolean looksLikeHeartbeat(String topic) {
90           return topic != null && topic.endsWith(heartbeatsTopic());
91       }
92   }
