26   /**
27    * <p>
28    * Base class for records containing data to be copied to/from Kafka. This corresponds closely to
29    * Kafka's {@link org.apache.kafka.clients.producer.ProducerRecord ProducerRecord} and {@link org.apache.kafka.clients.consumer.ConsumerRecord ConsumerRecord} classes, and holds the data that may be used by both
30    * sources and sinks (topic, kafkaPartition, key, value). Although both implementations include a
31    * notion of offset, it is not included here because they differ in type.
32    * </p>
33    */
34   public abstract class ConnectRecord<R extends ConnectRecord<R>> {
35       private final String topic;
36       private final Integer kafkaPartition;
37       private final Schema keySchema;
38       private final Object key;
39       private final Schema valueSchema;
40       private final Object value;
41       private final Long timestamp;
42       private final Headers headers;
43   
44       public ConnectRecord(String topic, Integer kafkaPartition,
45                            Schema keySchema, Object key,
46                            Schema valueSchema, Object value,
47                            Long timestamp) {
48           this(topic, kafkaPartition, keySchema, key, valueSchema, value, timestamp, new ConnectHeaders());
49       }
50   
51       public ConnectRecord(String topic, Integer kafkaPartition,
52                            Schema keySchema, Object key,
53                            Schema valueSchema, Object value,
54                            Long timestamp, Iterable<Header> headers) {
55           this.topic = topic;
56           this.kafkaPartition = kafkaPartition;
57           this.keySchema = keySchema;
58           this.key = key;
59           this.valueSchema = valueSchema;
60           this.value = value;
61           this.timestamp = timestamp;
62           if (headers instanceof ConnectHeaders) {
63               this.headers = (ConnectHeaders) headers;
64           } else {
65               this.headers = new ConnectHeaders(headers);
66           }
67       }
68   
69       public String topic() {
70           return topic;
71       }
72   
73       public Integer kafkaPartition() {
74           return kafkaPartition;
75       }
76   
77       public Object key() {
78           return key;
79       }
80   
81       public Schema keySchema() {
82           return keySchema;
83       }
84   
85       public Object value() {
86           return value;
87       }
88   
89       public Schema valueSchema() {
90           return valueSchema;
91       }
92   
93       public Long timestamp() {
94           return timestamp;
95       }
96   
97       /**
98        * Get the headers for this record.
99        *
100        * @return the headers; never null
101        */
102       public Headers headers() {
103           return headers;
104       }
105   
106       /**
107        * Create a new record of the same type as itself, with the specified parameter values. All other fields in this record will be copied
108        * over to the new record. Since the headers are mutable, the resulting record will have a copy of this record's headers.
109        *
110        * @param topic the name of the topic; may be null
111        * @param kafkaPartition the partition number for the Kafka topic; may be null
112        * @param keySchema the schema for the key; may be null
113        * @param key the key; may be null
114        * @param valueSchema the schema for the value; may be null
115        * @param value the value; may be null
116        * @param timestamp the timestamp; may be null
117        * @return the new record
118        */
119       public abstract R newRecord(String topic, Integer kafkaPartition, Schema keySchema, Object key, Schema valueSchema, Object value, Long timestamp);
120   
121       /**
122        * Create a new record of the same type as itself, with the specified parameter values. All other fields in this record will be copied
123        * over to the new record.
124        *
125        * @param topic the name of the topic; may be null
126        * @param kafkaPartition the partition number for the Kafka topic; may be null
127        * @param keySchema the schema for the key; may be null
128        * @param key the key; may be null
129        * @param valueSchema the schema for the value; may be null
130        * @param value the value; may be null
131        * @param timestamp the timestamp; may be null
132        * @param headers the headers; may be null or empty
133        * @return the new record
134        */
135       public abstract R newRecord(String topic, Integer kafkaPartition, Schema keySchema, Object key, Schema valueSchema, Object value, Long timestamp, Iterable<Header> headers);
136   
137       @Override
138       public String toString() {
139           return "ConnectRecord{" +
140                   "topic='" + topic + '\'' +
141                   ", kafkaPartition=" + kafkaPartition +
142                   ", key=" + key +
143                   ", keySchema=" + keySchema +
144                   ", value=" + value +
145                   ", valueSchema=" + valueSchema +
146                   ", timestamp=" + timestamp +
147                   ", headers=" + headers +
148                   '}';
149       }
150   
151       @Override
152       public boolean equals(Object o) {
153           if (this == o)
154               return true;
155           if (o == null || getClass() != o.getClass())
156               return false;
157   
158           ConnectRecord<?> that = (ConnectRecord<?>) o;
159   
160           return Objects.equals(kafkaPartition, that.kafkaPartition)
161                  && Objects.equals(topic, that.topic)
162                  && Objects.equals(keySchema, that.keySchema)
163                  && Objects.equals(key, that.key)
164                  && Objects.equals(valueSchema, that.valueSchema)
165                  && Objects.equals(value, that.value)
166                  && Objects.equals(timestamp, that.timestamp)
167                  && Objects.equals(headers, that.headers);
168       }
169   
170       @Override
171       public int hashCode() {
172           int result = topic != null ? topic.hashCode() : 0;
173           result = 31 * result + (kafkaPartition != null ? kafkaPartition.hashCode() : 0);
174           result = 31 * result + (keySchema != null ? keySchema.hashCode() : 0);
175           result = 31 * result + (key != null ? key.hashCode() : 0);
176           result = 31 * result + (valueSchema != null ? valueSchema.hashCode() : 0);
177           result = 31 * result + (value != null ? value.hashCode() : 0);
178           result = 31 * result + (timestamp != null ? timestamp.hashCode() : 0);
179           result = 31 * result + headers.hashCode();
180           return result;
181       }
