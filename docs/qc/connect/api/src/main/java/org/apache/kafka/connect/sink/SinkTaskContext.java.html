17   package org.apache.kafka.connect.sink;
18   
19   import org.apache.kafka.common.TopicPartition;
20   
21   import java.util.Map;
22   import java.util.Set;
23   
24   /**
25    * Context passed to SinkTasks, allowing them to access utilities in the Kafka Connect runtime.
26    */
27   public interface SinkTaskContext {
28   
29       /**
30        * Get the Task configuration.  This is the latest configuration and may differ from that passed on startup.
31        *
32        * For example, this method can be used to obtain the latest configuration if an external secret has changed,
33        * and the configuration is using variable references such as those compatible with
34        * {@link org.apache.kafka.common.config.ConfigTransformer}.
35        */
36       Map<String, String> configs();
37   
38       /**
39        * Reset the consumer offsets for the given topic partitions. SinkTasks should use this if they manage offsets
40        * in the sink data store rather than using Kafka consumer offsets. For example, an HDFS connector might record
41        * offsets in HDFS to provide exactly once delivery. When the SinkTask is started or a rebalance occurs, the task
42        * would reload offsets from HDFS and use this method to reset the consumer to those offsets.
43        *
44        * SinkTasks that do not manage their own offsets do not need to use this method.
45        *
46        * @param offsets map of offsets for topic partitions
47        */
48       void offset(Map<TopicPartition, Long> offsets);
49   
50       /**
51        * Reset the consumer offsets for the given topic partition. SinkTasks should use if they manage offsets
52        * in the sink data store rather than using Kafka consumer offsets. For example, an HDFS connector might record
53        * offsets in HDFS to provide exactly once delivery. When the topic partition is recovered the task
54        * would reload offsets from HDFS and use this method to reset the consumer to the offset.
55        *
56        * SinkTasks that do not manage their own offsets do not need to use this method.
57        *
58        * @param tp the topic partition to reset offset.
59        * @param offset the offset to reset to.
60        */
61       void offset(TopicPartition tp, long offset);
62   
63       /**
64        * Set the timeout in milliseconds. SinkTasks should use this to indicate that they need to retry certain
65        * operations after the timeout. SinkTasks may have certain operations on external systems that may need
66        * to retry in case of failures. For example, append a record to an HDFS file may fail due to temporary network
67        * issues. SinkTasks use this method to set how long to wait before retrying.
68        * @param timeoutMs the backoff timeout in milliseconds.
69        */
70       void timeout(long timeoutMs);
71   
72       /**
73        * Get the current set of assigned TopicPartitions for this task.
74        * @return the set of currently assigned TopicPartitions
75        */
76       Set<TopicPartition> assignment();
77   
78       /**
79        * Pause consumption of messages from the specified TopicPartitions.
80        * @param partitions the partitions which should be paused
81        */
82       void pause(TopicPartition... partitions);
83   
84       /**
85        * Resume consumption of messages from previously paused TopicPartitions.
86        * @param partitions the partitions to resume
87        */
88       void resume(TopicPartition... partitions);
89   
90       /**
91        * Request an offset commit. Sink tasks can use this to minimize the potential for redelivery
92        * by requesting an offset commit as soon as they flush data to the destination system.
93        *
94        * It is only a hint to the runtime and no timing guarantee should be assumed.
95        */
96       void requestCommit();
97   
98       /**
99        * Get the reporter to which the sink task can report problematic or failed {@link SinkRecord records}
100        * passed to the {@link SinkTask#put(java.util.Collection)} method. When reporting a failed record,
101        * the sink task will receive a {@link java.util.concurrent.Future} that the task can optionally use to wait until
102        * the failed record and exception have been written to Kafka. Note that the result of
103        * this method may be null if this connector has not been configured to use a reporter.
104        *
105        * <p>This method was added in Apache Kafka 2.6. Sink tasks that use this method but want to
106        * maintain backward compatibility so they can also be deployed to older Connect runtimes
107        * should guard the call to this method with a try-catch block, since calling this method will result in a
108        * {@link NoSuchMethodException} or {@link NoClassDefFoundError} when the sink connector is deployed to
109        * Connect runtimes older than Kafka 2.6. For example:
110        * <pre>
111        *     ErrantRecordReporter reporter;
112        *     try {
113        *         reporter = context.errantRecordReporter();
114        *     } catch (NoSuchMethodError | NoClassDefFoundError e) {
115        *         reporter = null;
116        *     }
117        * </pre>
118        *
119        * @return the reporter; null if no error reporter has been configured for the connector
120        * @since 2.6
121        */
122       default ErrantRecordReporter errantRecordReporter() {
123           return null;
124       }
125   
126   }
