22   import java.util.AbstractMap;

31   /**
32    * <p>
33    *     SchemaProjector is utility to project a value between compatible schemas and throw exceptions
34    *     when non compatible schemas are provided.
35    * </p>
36    */
37   
38   public class SchemaProjector {
39   
40       private static Set<AbstractMap.SimpleImmutableEntry<Type, Type>> promotable = new HashSet<>();
41   
42       static {
43           Type[] promotableTypes = {Type.INT8, Type.INT16, Type.INT32, Type.INT64, Type.FLOAT32, Type.FLOAT64};
44           for (int i = 0; i < promotableTypes.length; ++i) {
45               for (int j = i; j < promotableTypes.length; ++j) {
46                   promotable.add(new AbstractMap.SimpleImmutableEntry<>(promotableTypes[i], promotableTypes[j]));
47               }
48           }
49       }
50   
51       /**
52        * This method project a value between compatible schemas and throw exceptions when non compatible schemas are provided
53        * @param source the schema used to construct the record
54        * @param record the value to project from source schema to target schema
55        * @param target the schema to project the record to
56        * @return the projected value with target schema
57        * @throws SchemaProjectorException
58        */
59       public static Object project(Schema source, Object record, Schema target) throws SchemaProjectorException {
60           checkMaybeCompatible(source, target);
61           if (source.isOptional() && !target.isOptional()) {
62               if (target.defaultValue() != null) {
63                   if (record != null) {
64                       return projectRequiredSchema(source, record, target);
65                   } else {
66                       return target.defaultValue();
67                   }
68               } else {
69                   throw new SchemaProjectorException("Writer schema is optional, however, target schema does not provide a default value.");
70               }
71           } else {
72               if (record != null) {
73                   return projectRequiredSchema(source, record, target);
74               } else {
75                   return null;
76               }
77           }
78       }
79   
80       private static Object projectRequiredSchema(Schema source, Object record, Schema target) throws SchemaProjectorException {
81           switch (target.type()) {
82               case INT8:
83               case INT16:
84               case INT32:
85               case INT64:
86               case FLOAT32:
87               case FLOAT64:
88               case BOOLEAN:
89               case BYTES:
90               case STRING:
91                   return projectPrimitive(source, record, target);
92               case STRUCT:
93                   return projectStruct(source, (Struct) record, target);
94               case ARRAY:
95                   return projectArray(source, record, target);
96               case MAP:
97                   return projectMap(source, record, target);
98           }
99           return null;
100       }
101   
102       private static Object projectStruct(Schema source, Struct sourceStruct, Schema target) throws SchemaProjectorException {
103           Struct targetStruct = new Struct(target);
104           for (Field targetField : target.fields()) {
105               String fieldName = targetField.name();
106               Field sourceField = source.field(fieldName);
107               if (sourceField != null) {
108                   Object sourceFieldValue = sourceStruct.get(fieldName);
109                   try {
110                       Object targetFieldValue = project(sourceField.schema(), sourceFieldValue, targetField.schema());
111                       targetStruct.put(fieldName, targetFieldValue);
112                   } catch (SchemaProjectorException e) {
113                       throw new SchemaProjectorException("Error projecting " + sourceField.name(), e);
114                   }
115               } else if (targetField.schema().isOptional()) {
116                   // Ignore missing field
117               } else if (targetField.schema().defaultValue() != null) {
118                   targetStruct.put(fieldName, targetField.schema().defaultValue());
119               } else {
120                   throw new SchemaProjectorException("Required field `" +  fieldName + "` is missing from source schema: " + source);
121               }
122           }
123           return targetStruct;
124       }

127       private static void checkMaybeCompatible(Schema source, Schema target) {
128           if (source.type() != target.type() && !isPromotable(source.type(), target.type())) {
129               throw new SchemaProjectorException("Schema type mismatch. source type: " + source.type() + " and target type: " + target.type());
130           } else if (!Objects.equals(source.name(), target.name())) {
131               throw new SchemaProjectorException("Schema name mismatch. source name: " + source.name() + " and target name: " + target.name());
132           } else if (!Objects.equals(source.parameters(), target.parameters())) {
133               throw new SchemaProjectorException("Schema parameters not equal. source parameters: " + source.parameters() + " and target parameters: " + target.parameters());
134           }
135       }

137       private static Object projectArray(Schema source, Object record, Schema target) throws SchemaProjectorException {
138           List<?> array = (List<?>) record;
139           List<Object> retArray = new ArrayList<>();
140           for (Object entry : array) {
141               retArray.add(project(source.valueSchema(), entry, target.valueSchema()));
142           }
143           return retArray;
144       }
145   
146       private static Object projectMap(Schema source, Object record, Schema target) throws SchemaProjectorException {
147           Map<?, ?> map = (Map<?, ?>) record;
148           Map<Object, Object> retMap = new HashMap<>();
149           for (Map.Entry<?, ?> entry : map.entrySet()) {
150               Object key = entry.getKey();
151               Object value = entry.getValue();
152               Object retKey = project(source.keySchema(), key, target.keySchema());
153               Object retValue = project(source.valueSchema(), value, target.valueSchema());
154               retMap.put(retKey, retValue);
155           }
156           return retMap;
157       }

159       private static Object projectPrimitive(Schema source, Object record, Schema target) throws SchemaProjectorException {
160           assert source.type().isPrimitive();
161           assert target.type().isPrimitive();
162           Object result;
163           if (isPromotable(source.type(), target.type()) && record instanceof Number) {
164               Number numberRecord = (Number) record;
165               switch (target.type()) {
166                   case INT8:
167                       result = numberRecord.byteValue();
168                       break;
169                   case INT16:
170                       result = numberRecord.shortValue();
171                       break;
172                   case INT32:
173                       result = numberRecord.intValue();
174                       break;
175                   case INT64:
176                       result = numberRecord.longValue();
177                       break;
178                   case FLOAT32:
179                       result = numberRecord.floatValue();
180                       break;
181                   case FLOAT64:
182                       result = numberRecord.doubleValue();
183                       break;
184                   default:
185                       throw new SchemaProjectorException("Not promotable type.");
186               }
187           } else {
188               result = record;
189           }
190           return result;
191       }
192   
193       private static boolean isPromotable(Type sourceType, Type targetType) {
194           return promotable.contains(new AbstractMap.SimpleImmutableEntry<>(sourceType, targetType));
195       }
