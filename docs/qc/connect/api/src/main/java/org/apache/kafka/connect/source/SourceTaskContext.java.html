17   package org.apache.kafka.connect.source;
18   
19   import org.apache.kafka.connect.storage.OffsetStorageReader;
20   
21   import java.util.Map;
22   
23   /**
24    * SourceTaskContext is provided to SourceTasks to allow them to interact with the underlying
25    * runtime.
26    */
27   public interface SourceTaskContext {
28       /**
29        * Get the Task configuration.  This is the latest configuration and may differ from that passed on startup.
30        *
31        * For example, this method can be used to obtain the latest configuration if an external secret has changed,
32        * and the configuration is using variable references such as those compatible with
33        * {@link org.apache.kafka.common.config.ConfigTransformer}.
34        */
35       Map<String, String> configs();
36   
37       /**
38        * Get the OffsetStorageReader for this SourceTask.
39        */
40       OffsetStorageReader offsetStorageReader();
41   
42       /**
43        * Get a {@link TransactionContext} that can be used to define producer transaction boundaries
44        * when exactly-once support is enabled for the connector.
45        *
46        * <p>This method was added in Apache Kafka 3.2. Source tasks that use this method but want to
47        * maintain backward compatibility so they can also be deployed to older Connect runtimes
48        * should guard the call to this method with a try-catch block, since calling this method will result in a
49        * {@link NoSuchMethodException} or {@link NoClassDefFoundError} when the source connector is deployed to
50        * Connect runtimes older than Kafka 3.2. For example:
51        * <pre>
52        *     TransactionContext transactionContext;
53        *     try {
54        *         transactionContext = context.transactionContext();
55        *     } catch (NoSuchMethodError | NoClassDefFoundError e) {
56        *         transactionContext = null;
57        *     }
58        * </pre>
59        *
60        * @return the transaction context, or null if the connector was not configured to specify transaction boundaries
61        * @since 3.3
62        */
63       default TransactionContext transactionContext() {
64           return null;
65       }
66   }
