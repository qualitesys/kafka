20   import org.apache.kafka.common.Configurable;
21   import org.apache.kafka.connect.components.Versioned;
22   import org.apache.kafka.connect.health.ConnectClusterState;

24   import java.io.Closeable;
25   import java.util.Map;
26   
27   /**
28    * A plugin interface to allow registration of new JAX-RS resources like Filters, REST endpoints, providers, etc. The implementations will
29    * be discovered using the standard Java {@link java.util.ServiceLoader} mechanism by  Connect's plugin class loading mechanism.
30    *
31    * <p>The extension class(es) must be packaged as a plugin, with one JAR containing the implementation classes and a {@code
32    * META-INF/services/org.apache.kafka.connect.rest.extension.ConnectRestExtension} file that contains the fully qualified name of the
33    * class(es) that implement the ConnectRestExtension interface. The plugin should also include the JARs of all dependencies except those
34    * already provided by the Connect framework.
35    *
36    * <p>To install into a Connect installation, add a directory named for the plugin and containing the plugin's JARs into a directory that is
37    * on Connect's {@code plugin.path}, and (re)start the Connect worker.
38    *
39    * <p>When the Connect worker process starts up, it will read its configuration and instantiate all of the REST extension implementation
40    * classes that are specified in the `rest.extension.classes` configuration property. Connect will then pass its configuration to each
41    * extension via the {@link Configurable#configure(Map)} method, and will then call {@link #register} with a provided context.
42    *
43    * <p>When the Connect worker shuts down, it will call the extension's {@link #close} method to allow the implementation to release all of
44    * its resources.
45    */
46   public interface ConnectRestExtension extends Configurable, Versioned, Closeable {
47   
48       /**
49        * ConnectRestExtension implementations can register custom JAX-RS resources via the {@link #register(ConnectRestExtensionContext)}
50        * method. The Connect framework will invoke this method after registering the default Connect resources. If the implementations attempt
51        * to re-register any of the Connect resources, it will be be ignored and will be logged.
52        *
53        * @param restPluginContext The context provides access to JAX-RS {@link javax.ws.rs.core.Configurable} and {@link
54        *                          ConnectClusterState}.The custom JAX-RS resources can be registered via the {@link
55        *                          ConnectRestExtensionContext#configurable()}
56        */
57       void register(ConnectRestExtensionContext restPluginContext);
58   }
