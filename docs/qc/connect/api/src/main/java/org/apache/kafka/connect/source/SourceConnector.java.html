19   import org.apache.kafka.connect.connector.Connector;

21   import java.util.Map;

23   /**
24    * SourceConnectors implement the connector interface to pull data from another system and send
25    * it to Kafka.
26    */
27   public abstract class SourceConnector extends Connector {
28   
29       @Override
30       protected SourceConnectorContext context() {
31           return (SourceConnectorContext) context;
32       }
33   
34       /**
35        * Signals whether the connector supports exactly-once delivery guarantees with a proposed configuration.
36        * Connector authors can assume that worker-level exactly-once support is enabled when this method is invoked.
37        *
38        * <p>For backwards compatibility, the default implementation will return {@code null}, but connector authors are
39        * strongly encouraged to override this method to return a non-null value such as
40        * {@link ExactlyOnceSupport#SUPPORTED SUPPORTED} or {@link ExactlyOnceSupport#UNSUPPORTED UNSUPPORTED}.
41        *
42        * <p>Similar to {@link #validate(Map) validate}, this method may be called by the runtime before the
43        * {@link #start(Map) start} method is invoked when the connector will be run with exactly-once support.
44        *
45        * @param connectorConfig the configuration that will be used for the connector.
46        * @return {@link ExactlyOnceSupport#SUPPORTED} if the connector can provide exactly-once support with the given
47        * configuration, and {@link ExactlyOnceSupport#UNSUPPORTED} if it cannot. If this method is overridden by a
48        * connector, should not be {@code null}, but if {@code null}, it will be assumed that the connector cannot provide
49        * exactly-once guarantees.
50        * @since 3.3
51        */
52       public ExactlyOnceSupport exactlyOnceSupport(Map<String, String> connectorConfig) {
53           return null;
54       }
55   
56       /**
57        * Signals whether the connector implementation is capable of defining the transaction boundaries for a
58        * connector with the given configuration. This method is called before {@link #start(Map)}, only when the
59        * runtime supports exactly-once and the connector configuration includes {@code transaction.boundary=connector}.
60        *
61        * <p>This method need not be implemented if the connector implementation does not support defining
62        * transaction boundaries.
63        *
64        * @param connectorConfig the configuration that will be used for the connector
65        * @return {@link ConnectorTransactionBoundaries#SUPPORTED} if the connector will define its own transaction boundaries,
66        * or {@link ConnectorTransactionBoundaries#UNSUPPORTED} otherwise; may never be {@code null}. The default implementation
67        * returns {@link ConnectorTransactionBoundaries#UNSUPPORTED}.
68        * @since 3.3
69        * @see TransactionContext
70        */
71       public ConnectorTransactionBoundaries canDefineTransactionBoundaries(Map<String, String> connectorConfig) {
72           return ConnectorTransactionBoundaries.UNSUPPORTED;
73       }
74   }
