28   /**
29    * <p>
30    *     SchemaBuilder provides a fluent API for constructing {@link Schema} objects. It allows you to set each of the
31    *     properties for the schema and each call returns the SchemaBuilder so the calls can be chained. When nested types
32    *     are required, use one of the predefined schemas from {@link Schema} or use a second SchemaBuilder inline.
33    * </p>
34    * <p>
35    *     Here is an example of building a struct schema:
36    *     <pre>
37    *     Schema dateSchema = SchemaBuilder.struct()
38    *         .name("com.example.CalendarDate").version(2).doc("A calendar date including month, day, and year.")
39    *         .field("month", Schema.STRING_SCHEMA)
40    *         .field("day", Schema.INT8_SCHEMA)
41    *         .field("year", Schema.INT16_SCHEMA)
42    *         .build();
43    *     </pre>
44    * </p>
45    * <p>
46    *     Here is an example of using a second SchemaBuilder to construct complex, nested types:
47    *     <pre>
48    *     Schema userListSchema = SchemaBuilder.array(
49    *         SchemaBuilder.struct().name("com.example.User").field("username", Schema.STRING_SCHEMA).field("id", Schema.INT64_SCHEMA).build()
50    *     ).build();
51    *     </pre>
52    * </p>
53    */
54   public class SchemaBuilder implements Schema {
55       private static final String TYPE_FIELD = "type";
56       private static final String OPTIONAL_FIELD = "optional";
57       private static final String DEFAULT_FIELD = "default";
58       private static final String NAME_FIELD = "name";
59       private static final String VERSION_FIELD = "version";
60       private static final String DOC_FIELD = "doc";
61

69       private Schema valueSchema = null;

76       private Map<String, String> parameters;

78       public SchemaBuilder(Type type) {
79           if (null == type)
80               throw new SchemaBuilderException("type cannot be null");
81           this.type = type;
82           if (type == Type.STRUCT) {
83               fields = new LinkedHashMap<>();
84           }
85       }

119       /**
120        * Set the default value for this schema. The value is validated against the schema type, throwing a
121        * {@link SchemaBuilderException} if it does not match.
122        * @param value the default value
123        * @return the SchemaBuilder
124        */
125       public SchemaBuilder defaultValue(Object value) {
126           checkCanSet(DEFAULT_FIELD, defaultValue, value);
127           checkNotNull(TYPE_FIELD, type, DEFAULT_FIELD);
128           try {
129               ConnectSchema.validateValue(this, value);
130           } catch (DataException e) {
131               throw new SchemaBuilderException("Invalid default value", e);
132           }
133           defaultValue = value;
134           return this;
135       }

191       /**
192        * Set a schema parameter.
193        * @param propertyName name of the schema property to define
194        * @param propertyValue value of the schema property to define, as a String
195        * @return the SchemaBuilder
196        */
197       public SchemaBuilder parameter(String propertyName, String propertyValue) {
198           // Preserve order of insertion with a LinkedHashMap. This isn't strictly necessary, but is nice if logical types
199           // can print their properties in a consistent order.
200           if (parameters == null)
201               parameters = new LinkedHashMap<>();
202           parameters.put(propertyName, propertyValue);
203           return this;
204       }
205   
206       /**
207        * Set schema parameters. This operation is additive; it does not remove existing parameters that do not appear in
208        * the set of properties pass to this method.
209        * @param props Map of properties to set
210        * @return the SchemaBuilder
211        */
212       public SchemaBuilder parameters(Map<String, String> props) {
213           // Avoid creating an empty set of properties so we never have an empty map
214           if (props.isEmpty())
215               return this;
216           if (parameters == null)
217               parameters = new LinkedHashMap<>();
218           parameters.putAll(props);
219           return this;
220       }

315       /**
316        * Add a field to this struct schema. Throws a SchemaBuilderException if this is not a struct schema.
317        * @param fieldName the name of the field to add
318        * @param fieldSchema the Schema for the field's value
319        * @return the SchemaBuilder
320        */
321       public SchemaBuilder field(String fieldName, Schema fieldSchema) {
322           if (type != Type.STRUCT)
323               throw new SchemaBuilderException("Cannot create fields on type " + type);
324           if (null == fieldName || fieldName.isEmpty())
325               throw new SchemaBuilderException("fieldName cannot be null.");
326           if (null == fieldSchema)
327               throw new SchemaBuilderException("fieldSchema for field " + fieldName + " cannot be null.");
328           int fieldIndex = fields.size();
329           if (fields.containsKey(fieldName))
330               throw new SchemaBuilderException("Cannot create field because of field name duplication " + fieldName);
331           fields.put(fieldName, new Field(fieldName, fieldIndex, fieldSchema));
332           return this;
333       }
334   
335       /**
336        * Get the list of fields for this Schema. Throws a DataException if this schema is not a struct.
337        * @return the list of fields for this Schema
338        */
339       @Override
340       public List<Field> fields() {
341           if (type != Type.STRUCT)
342               throw new DataException("Cannot list fields on non-struct type");
343           return new ArrayList<>(fields.values());
344       }

346       @Override
347       public Field field(String fieldName) {
348           if (type != Type.STRUCT)
349               throw new DataException("Cannot look up fields on non-struct type");
350           return fields.get(fieldName);
351       }

435       private static void checkCanSet(String fieldName, Object fieldVal, Object val) {
436           if (fieldVal != null && fieldVal != val)
437               throw new SchemaBuilderException("Invalid SchemaBuilder call: " + fieldName + " has already been set.");
438       }
439   
440       private static void checkNotNull(String fieldName, Object val, String fieldToSet) {
441           if (val == null)
442               throw new SchemaBuilderException("Invalid SchemaBuilder call: " + fieldName + " must be specified to set " + fieldToSet);
443       }
