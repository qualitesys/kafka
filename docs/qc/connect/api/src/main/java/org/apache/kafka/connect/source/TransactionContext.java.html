17   package org.apache.kafka.connect.source;
18   
19   /**
20    * Provided to source tasks to allow them to define their own producer transaction boundaries when
21    * exactly-once support is enabled.
22    */
23   public interface TransactionContext {
24   
25       /**
26        * Request a transaction commit after the next batch of records from {@link SourceTask#poll()}
27        * is processed.
28        */
29       void commitTransaction();
30   
31       /**
32        * Request a transaction commit after a source record is processed. The source record will be the
33        * last record in the committed transaction.
34        * @param record the record to commit the transaction after; may not be null.
35        */
36       void commitTransaction(SourceRecord record);
37   
38       /**
39        * Requests a transaction abort after the next batch of records from {@link SourceTask#poll()}. All of
40        * the records in that transaction will be discarded and will not appear in a committed transaction.
41        * However, offsets for that transaction will still be committed so than the records in that transaction
42        * are not reprocessed. If the data should instead be reprocessed, the task should not invoke this method
43        * and should instead throw an exception.
44        */
45       void abortTransaction();
46   
47       /**
48        * Requests a transaction abort after a source record is processed. The source record will be the
49        * last record in the aborted transaction. All of the records in that transaction will be discarded
50        * and will not appear in a committed transaction. However, offsets for that transaction will still
51        * be committed so that the records in that transaction are not reprocessed. If the data should be
52        * reprocessed, the task should not invoke this method and should instead throw an exception.
53        * @param record the record to abort the transaction after; may not be null.
54        */
55       void abortTransaction(SourceRecord record);
56   }
