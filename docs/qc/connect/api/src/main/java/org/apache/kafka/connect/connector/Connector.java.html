17   package org.apache.kafka.connect.connector;
18   
19   import org.apache.kafka.common.config.Config;
20   import org.apache.kafka.common.config.ConfigDef;
21   import org.apache.kafka.common.config.ConfigValue;
22   import org.apache.kafka.connect.errors.ConnectException;
23   import org.apache.kafka.connect.components.Versioned;
24   
25   import java.util.List;
26   import java.util.Map;
27   
28   /**
29    * <p>
30    * Connectors manage integration of Kafka Connect with another system, either as an input that ingests
31    * data into Kafka or an output that passes data to an external system. Implementations should
32    * not use this class directly; they should inherit from {@link org.apache.kafka.connect.source.SourceConnector SourceConnector}
33    * or {@link org.apache.kafka.connect.sink.SinkConnector SinkConnector}.
34    * </p>
35    * <p>
36    * Connectors have two primary tasks. First, given some configuration, they are responsible for
37    * creating configurations for a set of {@link Task}s that split up the data processing. For
38    * example, a database Connector might create Tasks by dividing the set of tables evenly among
39    * tasks. Second, they are responsible for monitoring inputs for changes that require
40    * reconfiguration and notifying the Kafka Connect runtime via the {@link ConnectorContext}. Continuing the
41    * previous example, the connector might periodically check for new tables and notify Kafka Connect of
42    * additions and deletions. Kafka Connect will then request new configurations and update the running
43    * Tasks.
44    * </p>
45    */
46   public abstract class Connector implements Versioned {
47   
48       protected ConnectorContext context;
49   
50   
51       /**
52        * Initialize this connector, using the provided ConnectorContext to notify the runtime of
53        * input configuration changes.
54        * @param ctx context object used to interact with the Kafka Connect runtime
55        */
56       public void initialize(ConnectorContext ctx) {
57           context = ctx;
58       }
59   
60       /**
61        * <p>
62        * Initialize this connector, using the provided ConnectorContext to notify the runtime of
63        * input configuration changes and using the provided set of Task configurations.
64        * This version is only used to recover from failures.
65        * </p>
66        * <p>
67        * The default implementation ignores the provided Task configurations. During recovery, Kafka Connect will request
68        * an updated set of configurations and update the running Tasks appropriately. However, Connectors should
69        * implement special handling of this case if it will avoid unnecessary changes to running Tasks.
70        * </p>
71        *
72        * @param ctx context object used to interact with the Kafka Connect runtime
73        * @param taskConfigs existing task configurations, which may be used when generating new task configs to avoid
74        *                    churn in partition to task assignments
75        */
76       public void initialize(ConnectorContext ctx, List<Map<String, String>> taskConfigs) {
77           context = ctx;
78           // Ignore taskConfigs. May result in more churn of tasks during recovery if updated configs
79           // are very different, but reduces the difficulty of implementing a Connector
80       }
81   
82       /**
83        * Returns the context object used to interact with the Kafka Connect runtime.
84        *
85        * @return the context for this Connector.
86        */
87       protected ConnectorContext context() {
88           return context;
89       }
90   
91       /**
92        * Start this Connector. This method will only be called on a clean Connector, i.e. it has
93        * either just been instantiated and initialized or {@link #stop()} has been invoked.
94        *
95        * @param props configuration settings
96        */
97       public abstract void start(Map<String, String> props);
98   
99       /**
100        * Reconfigure this Connector. Most implementations will not override this, using the default
101        * implementation that calls {@link #stop()} followed by {@link #start(Map)}.
102        * Implementations only need to override this if they want to handle this process more
103        * efficiently, e.g. without shutting down network connections to the external system.
104        *
105        * @param props new configuration settings
106        */
107       public void reconfigure(Map<String, String> props) {
108           stop();
109           start(props);
110       }
111   
112       /**
113        * Returns the Task implementation for this Connector.
114        */
115       public abstract Class<? extends Task> taskClass();
116   
117       /**
118        * Returns a set of configurations for Tasks based on the current configuration,
119        * producing at most count configurations.
120        *
121        * @param maxTasks maximum number of configurations to generate
122        * @return configurations for Tasks
123        */
124       public abstract List<Map<String, String>> taskConfigs(int maxTasks);
125   
126       /**
127        * Stop this connector.
128        */
129       public abstract void stop();
130   
131       /**
132        * Validate the connector configuration values against configuration definitions.
133        * @param connectorConfigs the provided configuration values
134        * @return List of Config, each Config contains the updated configuration information given
135        * the current configuration values.
136        */
137       public Config validate(Map<String, String> connectorConfigs) {
138           ConfigDef configDef = config();
139           if (null == configDef) {
140               throw new ConnectException(
141                   String.format("%s.config() must return a ConfigDef that is not null.", this.getClass().getName())
142               );
143           }
144           List<ConfigValue> configValues = configDef.validate(connectorConfigs);
145           return new Config(configValues);
146       }
147   
148       /**
149        * Define the configuration for the connector.
150        * @return The ConfigDef for this connector; may not be null.
151        */
152       public abstract ConfigDef config();
153   }
