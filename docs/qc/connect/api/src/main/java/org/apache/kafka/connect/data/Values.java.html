47   /**
48    * Utility for converting from one Connect value to a different form. This is useful when the caller expects a value of a particular type
49    * but is uncertain whether the actual value is one that isn't directly that type but can be converted into that type.
50    *
51    * <p>For example, a caller might expects a particular {@link org.apache.kafka.connect.header.Header} to contain an {@link Type#INT64}
52    * value, when in fact that header contains a string representation of a 32-bit integer. Here, the caller can use the methods in this
53    * class to convert the value to the desired type:
54    * <pre>
55    *     Header header = ...
56    *     long value = Values.convertToLong(header.schema(), header.value());
57    * </pre>
58    *
59    * <p>This class is able to convert any value to a string representation as well as parse those string representations back into most of
60    * the types. The only exception is {@link Struct} values that require a schema and thus cannot be parsed from a simple string.
61    */

308       /**
309        * If possible infer a schema for the given value.
310        *
311        * @param value the value whose schema is to be inferred; may be null
312        * @return the inferred schema, or null if the value is null or no schema could be inferred
313        */
314       public static Schema inferSchema(Object value) {
315           if (value instanceof String) {
316               return Schema.STRING_SCHEMA;
317           }
318           if (value instanceof Boolean) {
319               return Schema.BOOLEAN_SCHEMA;
320           }
321           if (value instanceof Byte) {
322               return Schema.INT8_SCHEMA;
323           }
324           if (value instanceof Short) {
325               return Schema.INT16_SCHEMA;
326           }
327           if (value instanceof Integer) {
328               return Schema.INT32_SCHEMA;
329           }
330           if (value instanceof Long) {
331               return Schema.INT64_SCHEMA;
332           }
333           if (value instanceof Float) {
334               return Schema.FLOAT32_SCHEMA;
335           }
336           if (value instanceof Double) {
337               return Schema.FLOAT64_SCHEMA;
338           }
339           if (value instanceof byte[] || value instanceof ByteBuffer) {
340               return Schema.BYTES_SCHEMA;
341           }
342           if (value instanceof List) {
343               List<?> list = (List<?>) value;
344               if (list.isEmpty()) {
345                   return null;
346               }
347               SchemaDetector detector = new SchemaDetector();
348               for (Object element : list) {
349                   if (!detector.canDetect(element)) {
350                       return null;
351                   }
352               }
353               return SchemaBuilder.array(detector.schema()).build();
354           }
355           if (value instanceof Map) {
356               Map<?, ?> map = (Map<?, ?>) value;

358                   return null;
359               }

365                   }
366               }
367               return SchemaBuilder.map(keyDetector.schema(), valueDetector.schema()).build();
368           }

370               return ((Struct) value).schema();
371           }

373       }

388               return new SchemaAndValue(Schema.STRING_SCHEMA, value);

394       /**
395        * Convert the value to the desired type.
396        *
397        * @param toSchema   the schema for the desired type; may not be null
398        * @param fromSchema the schema for the supplied value; may be null if not known
399        * @return the converted value; never null
400        * @throws DataException if the value could not be converted to the desired type
401        */
402       protected static Object convertTo(Schema toSchema, Schema fromSchema, Object value) throws DataException {
403           if (value == null) {
404               if (toSchema.isOptional()) {
405                   return null;
406               }
407               throw new DataException("Unable to convert a null value to a schema that requires a value");
408           }
409           switch (toSchema.type()) {
410               case BYTES:
411                   if (Decimal.LOGICAL_NAME.equals(toSchema.name())) {
412                       if (value instanceof ByteBuffer) {
413                           value = Utils.toArray((ByteBuffer) value);
414                       }
415                       if (value instanceof byte[]) {
416                           return Decimal.toLogical(toSchema, (byte[]) value);
417                       }
418                       if (value instanceof BigDecimal) {
419                           return value;
420                       }
421                       if (value instanceof Number) {
422                           // Not already a decimal, so treat it as a double ...
423                           double converted = ((Number) value).doubleValue();
424                           return BigDecimal.valueOf(converted);
425                       }
426                       if (value instanceof String) {
427                           return new BigDecimal(value.toString()).doubleValue();
428                       }
429                   }
430                   if (value instanceof ByteBuffer) {
431                       return Utils.toArray((ByteBuffer) value);
432                   }
433                   if (value instanceof byte[]) {
434                       return value;
435                   }
436                   if (value instanceof BigDecimal) {
437                       return Decimal.fromLogical(toSchema, (BigDecimal) value);
438                   }
439                   break;
440               case STRING:
441                   StringBuilder sb = new StringBuilder();
442                   append(sb, value, false);
443                   return sb.toString();
444               case BOOLEAN:
445                   if (value instanceof Boolean) {
446                       return value;
447                   }
448                   if (value instanceof String) {
449                       SchemaAndValue parsed = parseString(value.toString());
450                       if (parsed.value() instanceof Boolean) {
451                           return parsed.value();
452                       }
453                   }
454                   return asLong(value, fromSchema, null) == 0L ? Boolean.FALSE : Boolean.TRUE;
455               case INT8:
456                   if (value instanceof Byte) {
457                       return value;
458                   }
459                   return (byte) asLong(value, fromSchema, null);
460               case INT16:
461                   if (value instanceof Short) {
462                       return value;
463                   }
464                   return (short) asLong(value, fromSchema, null);
465               case INT32:
466                   if (Date.LOGICAL_NAME.equals(toSchema.name())) {
467                       if (value instanceof String) {
468                           SchemaAndValue parsed = parseString(value.toString());
469                           value = parsed.value();
470                       }
471                       if (value instanceof java.util.Date) {
472                           if (fromSchema != null) {
473                               String fromSchemaName = fromSchema.name();
474                               if (Date.LOGICAL_NAME.equals(fromSchemaName)) {
475                                   return value;
476                               }
477                               if (Timestamp.LOGICAL_NAME.equals(fromSchemaName)) {
478                                   // Just get the number of days from this timestamp
479                                   long millis = ((java.util.Date) value).getTime();
480                                   int days = (int) (millis / MILLIS_PER_DAY); // truncates
481                                   return Date.toLogical(toSchema, days);
482                               }
483                           } else {
484                               // There is no fromSchema, so no conversion is needed
485                               return value;
486                           }
487                       }
488                       long numeric = asLong(value, fromSchema, null);
489                       return Date.toLogical(toSchema, (int) numeric);
490                   }
491                   if (Time.LOGICAL_NAME.equals(toSchema.name())) {
492                       if (value instanceof String) {
493                           SchemaAndValue parsed = parseString(value.toString());
494                           value = parsed.value();
495                       }
496                       if (value instanceof java.util.Date) {
497                           if (fromSchema != null) {
498                               String fromSchemaName = fromSchema.name();
499                               if (Time.LOGICAL_NAME.equals(fromSchemaName)) {
500                                   return value;
501                               }
502                               if (Timestamp.LOGICAL_NAME.equals(fromSchemaName)) {
503                                   // Just get the time portion of this timestamp
504                                   Calendar calendar = Calendar.getInstance(UTC);
505                                   calendar.setTime((java.util.Date) value);
506                                   calendar.set(Calendar.YEAR, 1970);
507                                   calendar.set(Calendar.MONTH, 0); // Months are zero-based
508                                   calendar.set(Calendar.DAY_OF_MONTH, 1);
509                                   return Time.toLogical(toSchema, (int) calendar.getTimeInMillis());
510                               }
511                           } else {
512                               // There is no fromSchema, so no conversion is needed
513                               return value;
514                           }
515                       }
516                       long numeric = asLong(value, fromSchema, null);
517                       return Time.toLogical(toSchema, (int) numeric);
518                   }
519                   if (value instanceof Integer) {
520                       return value;
521                   }
522                   return (int) asLong(value, fromSchema, null);
523               case INT64:
524                   if (Timestamp.LOGICAL_NAME.equals(toSchema.name())) {
525                       if (value instanceof String) {
526                           SchemaAndValue parsed = parseString(value.toString());
527                           value = parsed.value();
528                       }
529                       if (value instanceof java.util.Date) {
530                           java.util.Date date = (java.util.Date) value;
531                           if (fromSchema != null) {
532                               String fromSchemaName = fromSchema.name();
533                               if (Date.LOGICAL_NAME.equals(fromSchemaName)) {
534                                   int days = Date.fromLogical(fromSchema, date);
535                                   long millis = days * MILLIS_PER_DAY;
536                                   return Timestamp.toLogical(toSchema, millis);
537                               }
538                               if (Time.LOGICAL_NAME.equals(fromSchemaName)) {
539                                   long millis = Time.fromLogical(fromSchema, date);
540                                   return Timestamp.toLogical(toSchema, millis);
541                               }
542                               if (Timestamp.LOGICAL_NAME.equals(fromSchemaName)) {
543                                   return value;
544                               }
545                           } else {
546                               // There is no fromSchema, so no conversion is needed
547                               return value;
548                           }
549                       }
550                       long numeric = asLong(value, fromSchema, null);
551                       return Timestamp.toLogical(toSchema, numeric);
552                   }
553                   if (value instanceof Long) {
554                       return value;
555                   }
556                   return asLong(value, fromSchema, null);
557               case FLOAT32:
558                   if (value instanceof Float) {
559                       return value;
560                   }
561                   return (float) asDouble(value, fromSchema, null);
562               case FLOAT64:
563                   if (value instanceof Double) {
564                       return value;
565                   }
566                   return asDouble(value, fromSchema, null);
567               case ARRAY:
568                   if (value instanceof String) {
569                       SchemaAndValue schemaAndValue = parseString(value.toString());
570                       value = schemaAndValue.value();
571                   }
572                   if (value instanceof List) {
573                       return value;
574                   }
575                   break;
576               case MAP:
577                   if (value instanceof String) {
578                       SchemaAndValue schemaAndValue = parseString(value.toString());
579                       value = schemaAndValue.value();
580                   }
581                   if (value instanceof Map) {
582                       return value;
583                   }
584                   break;
585               case STRUCT:
586                   if (value instanceof Struct) {
587                       Struct struct = (Struct) value;
588                       return struct;
589                   }
590           }
591           throw new DataException("Unable to convert " + value + " (" + value.getClass() + ") to " + toSchema);
592       }

594       /**
595        * Convert the specified value to the desired scalar value type.
596        *
597        * @param value      the value to be converted; may not be null
598        * @param fromSchema the schema for the current value type; may not be null
599        * @param error      any previous error that should be included in an exception message; may be null

601        * @throws DataException if the value could not be converted to a long
602        */
603       protected static long asLong(Object value, Schema fromSchema, Throwable error) {
604           try {
605               if (value instanceof Number) {
606                   Number number = (Number) value;
607                   return number.longValue();
608               }
609               if (value instanceof String) {
610                   return new BigDecimal(value.toString()).longValue();
611               }
612           } catch (NumberFormatException e) {
613               error = e;
614               // fall through
615           }
616           if (fromSchema != null) {
617               String schemaName = fromSchema.name();
618               if (value instanceof java.util.Date) {
619                   if (Date.LOGICAL_NAME.equals(schemaName)) {
620                       return Date.fromLogical(fromSchema, (java.util.Date) value);
621                   }
622                   if (Time.LOGICAL_NAME.equals(schemaName)) {
623                       return Time.fromLogical(fromSchema, (java.util.Date) value);
624                   }
625                   if (Timestamp.LOGICAL_NAME.equals(schemaName)) {
626                       return Timestamp.fromLogical(fromSchema, (java.util.Date) value);
627                   }
628               }
629               throw new DataException("Unable to convert " + value + " (" + value.getClass() + ") to " + fromSchema, error);
630           }
631           throw new DataException("Unable to convert " + value + " (" + value.getClass() + ") to a number", error);
632       }

651               }

659       protected static void append(StringBuilder sb, Object value, boolean embedded) {
660           if (value == null) {
661               sb.append(NULL_VALUE);
662           } else if (value instanceof Number) {
663               sb.append(value);
664           } else if (value instanceof Boolean) {
665               sb.append(value);
666           } else if (value instanceof String) {
667               if (embedded) {
668                   String escaped = escape((String) value);
669                   sb.append('"').append(escaped).append('"');
670               } else {
671                   sb.append(value);
672               }
673           } else if (value instanceof byte[]) {
674               value = Base64.getEncoder().encodeToString((byte[]) value);
675               if (embedded) {
676                   sb.append('"').append(value).append('"');
677               } else {
678                   sb.append(value);
679               }
680           } else if (value instanceof ByteBuffer) {
681               byte[] bytes = Utils.readBytes((ByteBuffer) value);
682               append(sb, bytes, embedded);
683           } else if (value instanceof List) {
684               List<?> list = (List<?>) value;
685               sb.append('[');
686               appendIterable(sb, list.iterator());
687               sb.append(']');
688           } else if (value instanceof Map) {
689               Map<?, ?> map = (Map<?, ?>) value;
690               sb.append('{');
691               appendIterable(sb, map.entrySet().iterator());
692               sb.append('}');
693           } else if (value instanceof Struct) {
694               Struct struct = (Struct) value;
695               Schema schema = struct.schema();
696               boolean first = true;
697               sb.append('{');
698               for (Field field : schema.fields()) {
699                   if (first) {
700                       first = false;
701                   } else {
702                       sb.append(',');
703                   }
704                   append(sb, field.name(), true);
705                   sb.append(':');
706                   append(sb, struct.get(field), true);
707               }
708               sb.append('}');
709           } else if (value instanceof Map.Entry) {
710               Map.Entry<?, ?> entry = (Map.Entry<?, ?>) value;
711               append(sb, entry.getKey(), true);
712               sb.append(':');
713               append(sb, entry.getValue(), true);
714           } else if (value instanceof java.util.Date) {
715               java.util.Date dateValue = (java.util.Date) value;
716               String formatted = dateFormatFor(dateValue).format(dateValue);
717               sb.append(formatted);
718           } else {
719               throw new DataException("Failed to serialize unexpected value type " + value.getClass().getName() + ": " + value);
720           }
721       }

726               while (iter.hasNext()) {

729               }
730           }
731       }

748       protected static boolean canParseSingleTokenLiteral(Parser parser, boolean embedded, String tokenLiteral) {
749           int startPosition = parser.mark();
750           // If the next token is what we expect, then either...
751           if (parser.canConsume(tokenLiteral)) {
752               //   ...we're reading an embedded value, in which case the next token will be handled appropriately
753               //      by the caller if it's something like an end delimiter for a map or array, or a comma to
754               //      separate multiple embedded values...
755               //   ...or it's being parsed as part of a top-level string, in which case, any other tokens should
756               //      cause use to stop parsing this single-token literal as such and instead just treat it like
757               //      a string. For example, the top-level string "true}" will be tokenized as the tokens "true" and
758               //      "}", but should ultimately be parsed as just the string "true}" instead of the boolean true.
759               if (embedded || !parser.hasNext()) {
760                   return true;
761               }
762           }
763           parser.rewindTo(startPosition);
764           return false;
765       }
766   
767       protected static SchemaAndValue parse(Parser parser, boolean embedded) throws NoSuchElementException {
768           if (!parser.hasNext()) {
769               return null;
770           }
771           if (embedded) {
772               if (parser.canConsume(QUOTE_DELIMITER)) {
773                   StringBuilder sb = new StringBuilder();
774                   while (parser.hasNext()) {
775                       if (parser.canConsume(QUOTE_DELIMITER)) {
776                           break;
777                       }
778                       sb.append(parser.next());
779                   }
780                   String content = sb.toString();
781                   // We can parse string literals as temporal logical types, but all others
782                   // are treated as strings
783                   SchemaAndValue parsed = parseString(content);
784                   if (parsed != null && TEMPORAL_LOGICAL_TYPE_NAMES.contains(parsed.schema().name())) {
785                       return parsed;
786                   }
787                   return new SchemaAndValue(Schema.STRING_SCHEMA, content);
788               }
789           }
790   
791           if (canParseSingleTokenLiteral(parser, embedded, NULL_VALUE)) {
792               return null;
793           }
794           if (canParseSingleTokenLiteral(parser, embedded, TRUE_LITERAL)) {
795               return TRUE_SCHEMA_AND_VALUE;
796           }
797           if (canParseSingleTokenLiteral(parser, embedded, FALSE_LITERAL)) {
798               return FALSE_SCHEMA_AND_VALUE;
799           }
800   
801           int startPosition = parser.mark();
802   
803           try {
804               if (parser.canConsume(ARRAY_BEGIN_DELIMITER)) {
805                   List<Object> result = new ArrayList<>();
806                   Schema elementSchema = null;
807                   while (parser.hasNext()) {
808                       if (parser.canConsume(ARRAY_END_DELIMITER)) {
809                           Schema listSchema;
810                           if (elementSchema != null) {
811                               listSchema = SchemaBuilder.array(elementSchema).schema();
812                               result = alignListEntriesWithSchema(listSchema, result);
813                           } else {
814                               // Every value is null
815                               listSchema = SchemaBuilder.arrayOfNull().build();
816                           }
817                           return new SchemaAndValue(listSchema, result);
818                       }
819
