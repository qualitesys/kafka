54       public ConnectHeaders(Iterable<Header> original) {
55           if (original == null) {
56               return;
57           }
58           if (original instanceof ConnectHeaders) {
59               ConnectHeaders originalHeaders = (ConnectHeaders) original;
60               if (!originalHeaders.isEmpty()) {
61                   headers = new LinkedList<>(originalHeaders.headers);
62               }
63           } else {
64               headers = new LinkedList<>();
65               for (Header header : original) {
66                   Objects.requireNonNull(header, "Unable to add a null header.");
67                   headers.add(header);
68               }
69           }
70       }

281       @Override
282       public Headers retainLatest(String key) {
283           checkKey(key);

285               boolean found = false;
286               ListIterator<Header> iter = headers.listIterator(headers.size());
287               while (iter.hasPrevious()) {
288                   String headerKey = iter.previous().key();
289                   if (key.equals(headerKey)) {
290                       if (found)
291                           iter.remove();
292                       found = true;
293                   }
294               }
295           }
296           return this;

299       @Override
300       public Headers apply(String key, HeaderTransform transform) {
301           checkKey(key);
302           if (!isEmpty()) {
303               ListIterator<Header> iter = headers.listIterator();
304               while (iter.hasNext()) {
305                   Header orig = iter.next();
306                   if (orig.key().equals(key)) {
307                       Header updated = transform.apply(orig);
308                       if (updated != null) {
309                           iter.set(updated);
310                       } else {
311                           iter.remove();
312                       }
313                   }
314               }
315           }
316           return this;
317       }

341       @Override
342       public boolean equals(Object obj) {
343           if (obj == this) {
344               return true;
345           }
346           if (obj instanceof Headers) {
347               Headers that = (Headers) obj;
348               Iterator<Header> thisIter = this.iterator();
349               Iterator<Header> thatIter = that.iterator();
350               while (thisIter.hasNext() && thatIter.hasNext()) {
351                   if (!Objects.equals(thisIter.next(), thatIter.next()))
352                       return false;
353               }
354               return !thisIter.hasNext() && !thatIter.hasNext();
355           }
356           return false;
357       }

379       /**
380        * Check the {@link Schema#type() schema's type} matches the specified type.
381        *
382        * @param schema the schema; never null
383        * @param type   the expected type
384        * @throws DataException if the schema's type does not match the expected type
385        */
386       private void checkSchemaType(Schema schema, Type type) {
387           if (schema.type() != type) {
388               throw new DataException("Expecting " + type + " but instead found " + schema.type());
389           }
390       }
391   
392       /**
393        * Check that the value and its schema are compatible.
394        *
395        * @param schemaAndValue the schema and value pair
396        * @throws DataException if the schema is not compatible with the value
397        */
398       // visible for testing
399       void checkSchemaMatches(SchemaAndValue schemaAndValue) {
400           if (schemaAndValue != null) {
401               Schema schema = schemaAndValue.schema();
402               if (schema == null)
403                   return;
404               schema = schema.schema(); // in case a SchemaBuilder is used
405               Object value = schemaAndValue.value();
406               if (value == null && !schema.isOptional()) {
407                   throw new DataException("A null value requires an optional schema but was " + schema);
408               }
409               if (value != null) {
410                   switch (schema.type()) {
411                       case BYTES:
412                           if (value instanceof ByteBuffer)
413                               return;
414                           if (value instanceof byte[])
415                               return;
416                           if (value instanceof BigDecimal && Decimal.LOGICAL_NAME.equals(schema.name()))
417                               return;
418                           break;
419                       case STRING:
420                           if (value instanceof String)
421                               return;
422                           break;
423                       case BOOLEAN:
424                           if (value instanceof Boolean)
425                               return;
426                           break;
427                       case INT8:
428                           if (value instanceof Byte)
429                               return;
430                           break;
431                       case INT16:
432                           if (value instanceof Short)
433                               return;
434                           break;
435                       case INT32:
436                           if (value instanceof Integer)
437                               return;
438                           if (value instanceof java.util.Date && Date.LOGICAL_NAME.equals(schema.name()))
439                               return;
440                           if (value instanceof java.util.Date && Time.LOGICAL_NAME.equals(schema.name()))
441                               return;
442                           break;
443                       case INT64:
444                           if (value instanceof Long)
445                               return;
446                           if (value instanceof java.util.Date && Timestamp.LOGICAL_NAME.equals(schema.name()))
447                               return;
448                           break;
449                       case FLOAT32:
450                           if (value instanceof Float)
451                               return;
452                           break;
453                       case FLOAT64:
454                           if (value instanceof Double)
455                               return;
456                           break;
457                       case ARRAY:
458                           if (value instanceof List)
459                               return;
460                           break;
461                       case MAP:
462                           if (value instanceof Map)
463                               return;
464                           break;
465                       case STRUCT:
466                           if (value instanceof Struct)
467                               return;
468                           break;
469                   }
470                   throw new DataException("The value " + value + " is not compatible with the schema " + schema);
471               }
472           }
473       }

475       private static final class FilterByKeyIterator extends AbstractIterator<Header> {
476   
477           private final Iterator<Header> original;
478           private final String key;
479   
480           private FilterByKeyIterator(Iterator<Header> original, String key) {
481               this.original = original;
482               this.key = key;
483           }
484   
485           @Override
486           protected Header makeNext() {
487               while (original.hasNext()) {
488                   Header header = original.next();
489                   if (!header.key().equals(key)) {
490                       continue;
491                   }
492                   return header;
493               }
494               return this.allDone();
495           }
496       }
