17   package org.apache.kafka.connect.connector;
18   
19   import java.util.Map;
20   
21   /**
22    * <p>
23    * Tasks contain the code that actually copies data to/from another system. They receive
24    * a configuration from their parent Connector, assigning them a fraction of a Kafka Connect job's work.
25    * The Kafka Connect framework then pushes/pulls data from the Task. The Task must also be able to
26    * respond to reconfiguration requests.
27    * </p>
28    * <p>
29    * Task only contains the minimal shared functionality between
30    * {@link org.apache.kafka.connect.source.SourceTask} and
31    * {@link org.apache.kafka.connect.sink.SinkTask}.
32    * </p>
33    */
34   public interface Task {
35       /**
36        * Get the version of this task. Usually this should be the same as the corresponding {@link Connector} class's version.
37        *
38        * @return the version, formatted as a String
39        */
40       String version();
41   
42       /**
43        * Start the Task
44        * @param props initial configuration
45        */
46       void start(Map<String, String> props);
47   
48       /**
49        * Stop this task.
50        */
51       void stop();
52   }
