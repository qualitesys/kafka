31   public class ConnectSchema implements Schema {
32       /**
33        * Maps Schema.Types to a list of Java classes that can be used to represent them.
34        */
35       private static final Map<Type, List<Class<?>>> SCHEMA_TYPE_CLASSES = new EnumMap<>(Type.class);
36       /**
37        * Maps known logical types to a list of Java classes that can be used to represent them.
38        */
39       private static final Map<String, List<Class<?>>> LOGICAL_TYPE_CLASSES = new HashMap<>();
40   
41       /**
42        * Maps the Java classes to the corresponding Schema.Type.
43        */
44       private static final Map<Class<?>, Type> JAVA_CLASS_SCHEMA_TYPES = new HashMap<>();
45   
46       static {
47           SCHEMA_TYPE_CLASSES.put(Type.INT8, Collections.singletonList(Byte.class));
48           SCHEMA_TYPE_CLASSES.put(Type.INT16, Collections.singletonList(Short.class));
49           SCHEMA_TYPE_CLASSES.put(Type.INT32, Collections.singletonList(Integer.class));
50           SCHEMA_TYPE_CLASSES.put(Type.INT64, Collections.singletonList(Long.class));
51           SCHEMA_TYPE_CLASSES.put(Type.FLOAT32, Collections.singletonList(Float.class));
52           SCHEMA_TYPE_CLASSES.put(Type.FLOAT64, Collections.singletonList(Double.class));
53           SCHEMA_TYPE_CLASSES.put(Type.BOOLEAN, Collections.singletonList(Boolean.class));
54           SCHEMA_TYPE_CLASSES.put(Type.STRING, Collections.singletonList(String.class));
55           // Bytes are special and have 2 representations. byte[] causes problems because it doesn't handle equals() and
56           // hashCode() like we want objects to, so we support both byte[] and ByteBuffer. Using plain byte[] can cause
57           // those methods to fail, so ByteBuffers are recommended
58           SCHEMA_TYPE_CLASSES.put(Type.BYTES, Arrays.asList(byte[].class, ByteBuffer.class));
59           SCHEMA_TYPE_CLASSES.put(Type.ARRAY, Collections.singletonList(List.class));
60           SCHEMA_TYPE_CLASSES.put(Type.MAP, Collections.singletonList(Map.class));
61           SCHEMA_TYPE_CLASSES.put(Type.STRUCT, Collections.singletonList(Struct.class));
62   
63           for (Map.Entry<Type, List<Class<?>>> schemaClasses : SCHEMA_TYPE_CLASSES.entrySet()) {
64               for (Class<?> schemaClass : schemaClasses.getValue())
65                   JAVA_CLASS_SCHEMA_TYPES.put(schemaClass, schemaClasses.getKey());
66           }
67   
68           LOGICAL_TYPE_CLASSES.put(Decimal.LOGICAL_NAME, Collections.singletonList(BigDecimal.class));
69           LOGICAL_TYPE_CLASSES.put(Date.LOGICAL_NAME, Collections.singletonList(java.util.Date.class));
70           LOGICAL_TYPE_CLASSES.put(Time.LOGICAL_NAME, Collections.singletonList(java.util.Date.class));
71           LOGICAL_TYPE_CLASSES.put(Timestamp.LOGICAL_NAME, Collections.singletonList(java.util.Date.class));
72           // We don't need to put these into JAVA_CLASS_SCHEMA_TYPES since that's only used to determine schemas for
73           // schemaless data and logical types will have ambiguous schemas (e.g. many of them use the same Java class) so
74           // they should not be used without schemas.
75       }

102        */
103       public ConnectSchema(Type type, boolean optional, Object defaultValue, String name, Integer version, String doc, Map<String, String> parameters, List<Field> fields, Schema keySchema, Schema valueSchema) {

105           this.optional = optional;
106           this.defaultValue = defaultValue;
107           this.name = name;
108           this.version = version;
109           this.doc = doc;
110           this.parameters = parameters;

115               for (Field field : this.fields)
116                   fieldsByName.put(field.name(), field);
117           } else {
118               this.fields = null;
119               this.fieldsByName = null;
120           }
121   
122           this.keySchema = keySchema;
123           this.valueSchema = valueSchema;
124       }

206       /**
207        * Validate that the value can be used with the schema, i.e. that its type matches the schema type and nullability
208        * requirements. Throws a DataException if the value is invalid.
209        * @param schema Schema to test
210        * @param value value to test
211        */
212       public static void validateValue(Schema schema, Object value) {
213           validateValue(null, schema, value);
214       }

216       public static void validateValue(String name, Schema schema, Object value) {
217           if (value == null) {
218               if (!schema.isOptional())
219                   throw new DataException("Invalid value: null used for required field: \"" + name
220                           + "\", schema type: " + schema.type());
221               return;
222           }
223   
224           List<Class<?>> expectedClasses = expectedClassesFor(schema);
225           boolean foundMatch = false;
226           for (Class<?> expectedClass : expectedClasses) {
227               if (expectedClass.isInstance(value)) {
228                   foundMatch = true;
229                   break;
230               }
231           }
232   
233           if (!foundMatch) {
234               StringBuilder exceptionMessage = new StringBuilder("Invalid Java object for schema");
235               if (schema.name() != null) {
236                   exceptionMessage.append(" \"").append(schema.name()).append("\"");
237               }
238               exceptionMessage.append(" with type ").append(schema.type()).append(": ").append(value.getClass());
239               if (name != null) {
240                   exceptionMessage.append(" for field: \"").append(name).append("\"");
241               }
242               throw new DataException(exceptionMessage.toString());
243           }
244   
245           switch (schema.type()) {
246               case STRUCT:
247                   Struct struct = (Struct) value;
248                   if (!struct.schema().equals(schema))
249                       throw new DataException("Struct schemas do not match.");
250                   struct.validate();
251                   break;
252               case ARRAY:
253                   List<?> array = (List<?>) value;
254                   for (Object entry : array)
255                       validateValue(schema.valueSchema(), entry);
256                   break;
257               case MAP:
258                   Map<?, ?> map = (Map<?, ?>) value;
259                   for (Map.Entry<?, ?> entry : map.entrySet()) {
260                       validateValue(schema.keySchema(), entry.getKey());
261                       validateValue(schema.valueSchema(), entry.getValue());
262                   }
263                   break;
264           }
265       }
266   
267       private static List<Class<?>> expectedClassesFor(Schema schema) {
268           List<Class<?>> expectedClasses = LOGICAL_TYPE_CLASSES.get(schema.name());
269           if (expectedClasses == null)
270               expectedClasses = SCHEMA_TYPE_CLASSES.getOrDefault(schema.type(), Collections.emptyList());
271           return expectedClasses;
272       }

274       /**
275        * Validate that the value can be used for this schema, i.e. that its type matches the schema type and optional
276        * requirements. Throws a DataException if the value is invalid.
277        * @param value the value to validate
278        */
279       public void validateValue(Object value) {
280           validateValue(this, value);
281       }

289       @Override
290       public boolean equals(Object o) {
291           if (this == o) return true;
292           if (o == null || getClass() != o.getClass()) return false;
293           ConnectSchema schema = (ConnectSchema) o;
294           return Objects.equals(optional, schema.optional) &&
295                   Objects.equals(version, schema.version) &&
296                   Objects.equals(name, schema.name) &&
297                   Objects.equals(doc, schema.doc) &&
298                   Objects.equals(type, schema.type) &&
299                   Objects.deepEquals(defaultValue, schema.defaultValue) &&
300                   Objects.equals(fields, schema.fields) &&
301                   Objects.equals(keySchema, schema.keySchema) &&
302                   Objects.equals(valueSchema, schema.valueSchema) &&
303                   Objects.equals(parameters, schema.parameters);
304       }

324       /**
325        * Get the {@link Schema.Type} associated with the given class.
326        *
327        * @param klass the Class to
328        * @return the corresponding type, or null if there is no matching type
329        */
330       public static Type schemaType(Class<?> klass) {
331           synchronized (JAVA_CLASS_SCHEMA_TYPES) {
332               Type schemaType = JAVA_CLASS_SCHEMA_TYPES.get(klass);
333               if (schemaType != null)
334                   return schemaType;
335   
336               // Since the lookup only checks the class, we need to also try
337               for (Map.Entry<Class<?>, Type> entry : JAVA_CLASS_SCHEMA_TYPES.entrySet()) {
338                   try {
339                       klass.asSubclass(entry.getKey());
340                       // Cache this for subsequent lookups
341                       JAVA_CLASS_SCHEMA_TYPES.put(klass, entry.getValue());
342                       return entry.getValue();
343                   } catch (ClassCastException e) {
344                       // Expected, ignore
345                   }
346               }
347           }
348           return null;
349       }
