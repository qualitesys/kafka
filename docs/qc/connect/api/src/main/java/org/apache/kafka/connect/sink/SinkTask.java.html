17   package org.apache.kafka.connect.sink;
18   
19   import org.apache.kafka.clients.consumer.OffsetAndMetadata;
20   import org.apache.kafka.common.TopicPartition;
21   import org.apache.kafka.connect.connector.Task;
22   
23   import java.util.Collection;
24   import java.util.Map;
25   
26   /**
27    * SinkTask is a Task that takes records loaded from Kafka and sends them to another system. Each task
28    * instance is assigned a set of partitions by the Connect framework and will handle all records received
29    * from those partitions. As records are fetched from Kafka, they will be passed to the sink task using the
30    * {@link #put(Collection)} API, which should either write them to the downstream system or batch them for
31    * later writing. Periodically, Connect will call {@link #flush(Map)} to ensure that batched records are
32    * actually pushed to the downstream system..
33    *
34    * Below we describe the lifecycle of a SinkTask.
35    *
36    * <ol>
37    *     <li><b>Initialization:</b> SinkTasks are first initialized using {@link #initialize(SinkTaskContext)}
38    *     to prepare the task's context and {@link #start(Map)} to accept configuration and start any services
39    *     needed for processing.</li>
40    *     <li><b>Partition Assignment:</b> After initialization, Connect will assign the task a set of partitions
41    *     using {@link #open(Collection)}. These partitions are owned exclusively by this task until they
42    *     have been closed with {@link #close(Collection)}.</li>
43    *     <li><b>Record Processing:</b> Once partitions have been opened for writing, Connect will begin forwarding
44    *     records from Kafka using the {@link #put(Collection)} API. Periodically, Connect will ask the task
45    *     to flush records using {@link #flush(Map)} as described above.</li>
46    *     <li><b>Partition Rebalancing:</b> Occasionally, Connect will need to change the assignment of this task.
47    *     When this happens, the currently assigned partitions will be closed with {@link #close(Collection)} and
48    *     the new assignment will be opened using {@link #open(Collection)}.</li>
49    *     <li><b>Shutdown:</b> When the task needs to be shutdown, Connect will close active partitions (if there
50    *     are any) and stop the task using {@link #stop()}</li>
51     * </ol>
52    *
53    */
54   public abstract class SinkTask implements Task {
55   
56       /**
57        * <p>
58        * The configuration key that provides the list of topics that are inputs for this
59        * SinkTask.
60        * </p>
61        */
62       public static final String TOPICS_CONFIG = "topics";
63   
64       /**
65        * <p>
66        * The configuration key that provides a regex specifying which topics to include as inputs
67        * for this SinkTask.
68        * </p>
69        */
70       public static final String TOPICS_REGEX_CONFIG = "topics.regex";
71   
72       protected SinkTaskContext context;
73   
74       /**
75        * Initialize the context of this task. Note that the partition assignment will be empty until
76        * Connect has opened the partitions for writing with {@link #open(Collection)}.
77        * @param context The sink task's context
78        */
79       public void initialize(SinkTaskContext context) {
80           this.context = context;
81       }
82   
83       /**
84        * Start the Task. This should handle any configuration parsing and one-time setup of the task.
85        * @param props initial configuration
86        */
87       @Override
88       public abstract void start(Map<String, String> props);
89   
90       /**
91        * Put the records in the sink. Usually this should send the records to the sink asynchronously
92        * and immediately return.
93        *
94        * If this operation fails, the SinkTask may throw a {@link org.apache.kafka.connect.errors.RetriableException} to
95        * indicate that the framework should attempt to retry the same call again. Other exceptions will cause the task to
96        * be stopped immediately. {@link SinkTaskContext#timeout(long)} can be used to set the maximum time before the
97        * batch will be retried.
98        *
99        * @param records the set of records to send
100        */
101       public abstract void put(Collection<SinkRecord> records);
102   
103       /**
104        * Flush all records that have been {@link #put(Collection)} for the specified topic-partitions.
105        *
106        * @param currentOffsets the current offset state as of the last call to {@link #put(Collection)}},
107        *                       provided for convenience but could also be determined by tracking all offsets included in the {@link SinkRecord}s
108        *                       passed to {@link #put}.
109        */
110       public void flush(Map<TopicPartition, OffsetAndMetadata> currentOffsets) {
111       }
112   
113       /**
114        * Pre-commit hook invoked prior to an offset commit.
115        *
116        * The default implementation simply invokes {@link #flush(Map)} and is thus able to assume all {@code currentOffsets} are safe to commit.
117        *
118        * @param currentOffsets the current offset state as of the last call to {@link #put(Collection)}},
119        *                       provided for convenience but could also be determined by tracking all offsets included in the {@link SinkRecord}s
120        *                       passed to {@link #put}.
121        *
122        * @return an empty map if Connect-managed offset commit is not desired, otherwise a map of offsets by topic-partition that are safe to commit.
123        */
124       public Map<TopicPartition, OffsetAndMetadata> preCommit(Map<TopicPartition, OffsetAndMetadata> currentOffsets) {
125           flush(currentOffsets);
126           return currentOffsets;
127       }
128   
129       /**
130        * The SinkTask use this method to create writers for newly assigned partitions in case of partition
131        * rebalance. This method will be called after partition re-assignment completes and before the SinkTask starts
132        * fetching data. Note that any errors raised from this method will cause the task to stop.
133        * @param partitions The list of partitions that are now assigned to the task (may include
134        *                   partitions previously assigned to the task)
135        */
136       public void open(Collection<TopicPartition> partitions) {
137           this.onPartitionsAssigned(partitions);
138       }
139   
140       /**
141        * @deprecated Use {@link #open(Collection)} for partition initialization.
142        */
143       @Deprecated
144       public void onPartitionsAssigned(Collection<TopicPartition> partitions) {
145       }
146   
147       /**
148        * The SinkTask use this method to close writers for partitions that are no
149        * longer assigned to the SinkTask. This method will be called before a rebalance operation starts
150        * and after the SinkTask stops fetching data. After being closed, Connect will not write
151        * any records to the task until a new set of partitions has been opened. Note that any errors raised
152        * from this method will cause the task to stop.
153        * @param partitions The list of partitions that should be closed
154        */
155       public void close(Collection<TopicPartition> partitions) {
156           this.onPartitionsRevoked(partitions);
157       }
158   
159       /**
160        * @deprecated Use {@link #close(Collection)} instead for partition cleanup.
161        */
162       @Deprecated
163       public void onPartitionsRevoked(Collection<TopicPartition> partitions) {
164       }
165   
166       /**
167        * Perform any cleanup to stop this task. In SinkTasks, this method is invoked only once outstanding calls to other
168        * methods have completed (e.g., {@link #put(Collection)} has returned) and a final {@link #flush(Map)} and offset
169        * commit has completed. Implementations of this method should only need to perform final cleanup operations, such
170        * as closing network connections to the sink system.
171        */
172       @Override
173       public abstract void stop();
174   }
