17   package org.apache.kafka.connect.source;
18   
19   import org.apache.kafka.clients.producer.RecordMetadata;
20   import org.apache.kafka.connect.connector.Task;
21   
22   import java.util.List;
23   import java.util.Locale;
24   import java.util.Map;
25   import java.util.Objects;
26   
27   /**
28    * SourceTask is a Task that pulls records from another system for storage in Kafka.
29    */
30   public abstract class SourceTask implements Task {
31   
32       /**
33        * The configuration key that determines how source tasks will define transaction boundaries
34        * when exactly-once support is enabled.
35        */
36       public static final String TRANSACTION_BOUNDARY_CONFIG = "transaction.boundary";
37   
38       /**
39        * Represents the permitted values for the {@link #TRANSACTION_BOUNDARY_CONFIG} property.
40        */
41       public enum TransactionBoundary {
42           /**
43            * A new transaction will be started and committed for every batch of records returned by {@link #poll()}.
44            */
45           POLL,
46           /**
47            * Transactions will be started and committed on a user-defined time interval.
48            */
49           INTERVAL,
50           /**
51            * Transactions will be defined by the connector itself, via a {@link TransactionContext}.
52            */
53           CONNECTOR;
54   
55           /**
56            * The default transaction boundary style that will be used for source connectors when no style is explicitly
57            * configured.
58            */
59           public static final TransactionBoundary DEFAULT = POLL;
60   
61           /**
62            * Parse a {@link TransactionBoundary} from the given string.
63            * @param property the string to parse; should not be null
64            * @return the {@link TransactionBoundary} whose name matches the given string
65            * @throws IllegalArgumentException if there is no transaction boundary type with the given name
66            */
67           public static TransactionBoundary fromProperty(String property) {
68               Objects.requireNonNull(property, "Value for transaction boundary property may not be null");
69               return TransactionBoundary.valueOf(property.toUpperCase(Locale.ROOT).trim());
70           }
71   
72           @Override
73           public String toString() {
74               return name().toLowerCase(Locale.ROOT);
75           }
76       }
77   
78       protected SourceTaskContext context;
79   
80       /**
81        * Initialize this SourceTask with the specified context object.
82        */
83       public void initialize(SourceTaskContext context) {
84           this.context = context;
85       }
86   
87       /**
88        * Start the Task. This should handle any configuration parsing and one-time setup of the task.
89        * @param props initial configuration
90        */
91       @Override
92       public abstract void start(Map<String, String> props);
93   
94       /**
95        * Poll this source task for new records. If no data is currently available, this method
96        * should block but return control to the caller regularly (by returning {@code null}) in
97        * order for the task to transition to the {@code PAUSED} state if requested to do so.
98        * <p>
99        * The task will be {@link #stop() stopped} on a separate thread, and when that happens
100        * this method is expected to unblock, quickly finish up any remaining processing, and
101        * return.
102        *
103        * @return a list of source records
104        */
105       public abstract List<SourceRecord> poll() throws InterruptedException;
106   
107       /**
108        * <p>
109        * Commit the offsets, up to the offsets that have been returned by {@link #poll()}. This
110        * method should block until the commit is complete.
111        * <p>
112        * SourceTasks are not required to implement this functionality; Kafka Connect will record offsets
113        * automatically. This hook is provided for systems that also need to store offsets internally
114        * in their own system.
115        */
116       public void commit() throws InterruptedException {
117           // This space intentionally left blank.
118       }
119   
120       /**
121        * Signal this SourceTask to stop. In SourceTasks, this method only needs to signal to the task that it should stop
122        * trying to poll for new data and interrupt any outstanding poll() requests. It is not required that the task has
123        * fully stopped. Note that this method necessarily may be invoked from a different thread than {@link #poll()} and
124        * {@link #commit()}.
125        *
126        * For example, if a task uses a {@link java.nio.channels.Selector} to receive data over the network, this method
127        * could set a flag that will force {@link #poll()} to exit immediately and invoke
128        * {@link java.nio.channels.Selector#wakeup() wakeup()} to interrupt any ongoing requests.
129        */
130       @Override
131       public abstract void stop();
132   
133       /**
134        * <p>
135        * Commit an individual {@link SourceRecord} when the callback from the producer client is received. This method is
136        * also called when a record is filtered by a transformation, and thus will never be ACK'd by a broker.
137        * <p>
138        * This is an alias for {@link #commitRecord(SourceRecord, RecordMetadata)} for backwards compatibility. The default
139        * implementation of {@link #commitRecord(SourceRecord, RecordMetadata)} just calls this method. It is not necessary
140        * to override both methods.
141        * <p>
142        * SourceTasks are not required to implement this functionality; Kafka Connect will record offsets
143        * automatically. This hook is provided for systems that also need to store offsets internally
144        * in their own system.
145        *
146        * @param record {@link SourceRecord} that was successfully sent via the producer or filtered by a transformation
147        * @throws InterruptedException
148        * @deprecated Use {@link #commitRecord(SourceRecord, RecordMetadata)} instead.
149        */
150       @Deprecated
151       public void commitRecord(SourceRecord record) throws InterruptedException {
152           // This space intentionally left blank.
153       }
154   
155       /**
156        * <p>
157        * Commit an individual {@link SourceRecord} when the callback from the producer client is received. This method is
158        * also called when a record is filtered by a transformation or when "errors.tolerance" is set to "all"
159        * and thus will never be ACK'd by a broker.
160        * In both cases {@code metadata} will be null.
161        * <p>
162        * SourceTasks are not required to implement this functionality; Kafka Connect will record offsets
163        * automatically. This hook is provided for systems that also need to store offsets internally
164        * in their own system.
165        * <p>
166        * The default implementation just calls {@link #commitRecord(SourceRecord)}, which is a nop by default. It is
167        * not necessary to implement both methods.
168        *
169        * @param record {@link SourceRecord} that was successfully sent via the producer, filtered by a transformation, or dropped on producer exception
170        * @param metadata {@link RecordMetadata} record metadata returned from the broker, or null if the record was filtered or if producer exceptions are ignored
171        * @throws InterruptedException
172        */
173       public void commitRecord(SourceRecord record, RecordMetadata metadata)
174               throws InterruptedException {
175           // by default, just call other method for backwards compatibility
176           commitRecord(record);
177       }
178   }
