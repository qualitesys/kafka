57   /**
58    * Implementation of Converter that uses JSON to store schemas and objects. By default this converter will serialize Connect keys, values,
59    * and headers with schemas, although this can be disabled with {@link JsonConverterConfig#SCHEMAS_ENABLE_CONFIG schemas.enable}
60    * configuration option.
61    *
62    * This implementation currently does nothing with the topic names or header names.
63    */
64   public class JsonConverter implements Converter, HeaderConverter {

66       private static final Map<Schema.Type, JsonToConnectTypeConverter> TO_CONNECT_CONVERTERS = new EnumMap<>(Schema.Type.class);
67   
68       static {
69           TO_CONNECT_CONVERTERS.put(Schema.Type.BOOLEAN, (schema, value) -> value.booleanValue());
70           TO_CONNECT_CONVERTERS.put(Schema.Type.INT8, (schema, value) -> (byte) value.intValue());
71           TO_CONNECT_CONVERTERS.put(Schema.Type.INT16, (schema, value) -> (short) value.intValue());
72           TO_CONNECT_CONVERTERS.put(Schema.Type.INT32, (schema, value) -> value.intValue());
73           TO_CONNECT_CONVERTERS.put(Schema.Type.INT64, (schema, value) -> value.longValue());
74           TO_CONNECT_CONVERTERS.put(Schema.Type.FLOAT32, (schema, value) -> value.floatValue());
75           TO_CONNECT_CONVERTERS.put(Schema.Type.FLOAT64, (schema, value) -> value.doubleValue());
76           TO_CONNECT_CONVERTERS.put(Schema.Type.BYTES, (schema, value) -> {
77               try {
78                   return value.binaryValue();
79               } catch (IOException e) {
80                   throw new DataException("Invalid bytes field", e);
81               }
82           });
83           TO_CONNECT_CONVERTERS.put(Schema.Type.STRING, (schema, value) -> value.textValue());
84           TO_CONNECT_CONVERTERS.put(Schema.Type.ARRAY, (schema, value) -> {
85               Schema elemSchema = schema == null ? null : schema.valueSchema();
86               ArrayList<Object> result = new ArrayList<>();
87               for (JsonNode elem : value) {
88                   result.add(convertToConnect(elemSchema, elem));
89               }
90               return result;
91           });
92           TO_CONNECT_CONVERTERS.put(Schema.Type.MAP, (schema, value) -> {
93               Schema keySchema = schema == null ? null : schema.keySchema();
94               Schema valueSchema = schema == null ? null : schema.valueSchema();
95   
96               // If the map uses strings for keys, it should be encoded in the natural JSON format. If it uses other
97               // primitive types or a complex type as a key, it will be encoded as a list of pairs. If we don't have a
98               // schema, we default to encoding in a Map.
99               Map<Object, Object> result = new HashMap<>();
100               if (schema == null || keySchema.type() == Schema.Type.STRING) {
101                   if (!value.isObject())
102                       throw new DataException("Maps with string fields should be encoded as JSON objects, but found " + value.getNodeType());
103                   Iterator<Map.Entry<String, JsonNode>> fieldIt = value.fields();
104                   while (fieldIt.hasNext()) {
105                       Map.Entry<String, JsonNode> entry = fieldIt.next();
106                       result.put(entry.getKey(), convertToConnect(valueSchema, entry.getValue()));
107                   }
108               } else {
109                   if (!value.isArray())
110                       throw new DataException("Maps with non-string fields should be encoded as JSON array of tuples, but found " + value.getNodeType());
111                   for (JsonNode entry : value) {
112                       if (!entry.isArray())
113                           throw new DataException("Found invalid map entry instead of array tuple: " + entry.getNodeType());
114                       if (entry.size() != 2)
115                           throw new DataException("Found invalid map entry, expected length 2 but found :" + entry.size());
116                       result.put(convertToConnect(keySchema, entry.get(0)),
117                               convertToConnect(valueSchema, entry.get(1)));
118                   }
119               }
120               return result;
121           });
122           TO_CONNECT_CONVERTERS.put(Schema.Type.STRUCT, (schema, value) -> {
123               if (!value.isObject())
124                   throw new DataException("Structs should be encoded as JSON objects, but found " + value.getNodeType());
125   
126               // We only have ISchema here but need Schema, so we need to materialize the actual schema. Using ISchema
127               // avoids having to materialize the schema for non-Struct types but it cannot be avoided for Structs since
128               // they require a schema to be provided at construction. However, the schema is only a SchemaBuilder during
129               // translation of schemas to JSON; during the more common translation of data to JSON, the call to schema.schema()
130               // just returns the schema Object and has no overhead.
131               Struct result = new Struct(schema.schema());
132               for (Field field : schema.fields())
133                   result.put(field, convertToConnect(field.schema(), value.get(field.name())));
134   
135               return result;
136           });
137       }

298       @Override
299       public byte[] fromConnectData(String topic, Schema schema, Object value) {
300           if (schema == null && value == null) {
301               return null;
302           }
303   
304           JsonNode jsonValue = config.schemasEnabled() ? convertToJsonWithEnvelope(schema, value) : convertToJsonWithoutEnvelope(schema, value);
305           try {
306               return serializer.serialize(topic, jsonValue);
307           } catch (SerializationException e) {
308               throw new DataException("Converting Kafka Connect data to byte[] failed due to serialization error: ", e);
309           }
310       }
311   
312       @Override
313       public SchemaAndValue toConnectData(String topic, byte[] value) {
314           JsonNode jsonValue;
315   
316           // This handles a tombstone message
317           if (value == null) {
318               return SchemaAndValue.NULL;
319           }
320   
321           try {
322               jsonValue = deserializer.deserialize(topic, value);
323           } catch (SerializationException e) {
324               throw new DataException("Converting byte[] to Kafka Connect data failed due to serialization error: ", e);
325           }
326   
327           if (config.schemasEnabled() && (!jsonValue.isObject() || jsonValue.size() != 2 || !jsonValue.has(JsonSchema.ENVELOPE_SCHEMA_FIELD_NAME) || !jsonValue.has(JsonSchema.ENVELOPE_PAYLOAD_FIELD_NAME)))
328               throw new DataException("JsonConverter with schemas.enable requires \"schema\" and \"payload\" fields and may not contain additional fields." +
329                       " If you are trying to deserialize plain JSON data, set schemas.enable=false in your converter configuration.");
330   
331           // The deserialized data should either be an envelope object containing the schema and the payload or the schema
332           // was stripped during serialization and we need to fill in an all-encompassing schema.
333           if (!config.schemasEnabled()) {
334               ObjectNode envelope = JSON_NODE_FACTORY.objectNode();
335               envelope.set(JsonSchema.ENVELOPE_SCHEMA_FIELD_NAME, null);
336               envelope.set(JsonSchema.ENVELOPE_PAYLOAD_FIELD_NAME, jsonValue);
337               jsonValue = envelope;
338           }
339   
340           Schema schema = asConnectSchema(jsonValue.get(JsonSchema.ENVELOPE_SCHEMA_FIELD_NAME));
341           return new SchemaAndValue(
342                   schema,
343                   convertToConnect(schema, jsonValue.get(JsonSchema.ENVELOPE_PAYLOAD_FIELD_NAME))
344           );
345       }

347       public ObjectNode asJsonSchema(Schema schema) {
348           if (schema == null)
349               return null;
350   
351           ObjectNode cached = fromConnectSchemaCache.get(schema);
352           if (cached != null)
353               return cached;
354   
355           final ObjectNode jsonSchema;
356           switch (schema.type()) {
357               case BOOLEAN:
358                   jsonSchema = JsonSchema.BOOLEAN_SCHEMA.deepCopy();
359                   break;
360               case BYTES:
361                   jsonSchema = JsonSchema.BYTES_SCHEMA.deepCopy();
362                   break;
363               case FLOAT64:
364                   jsonSchema = JsonSchema.DOUBLE_SCHEMA.deepCopy();
365                   break;
366               case FLOAT32:
367                   jsonSchema = JsonSchema.FLOAT_SCHEMA.deepCopy();
368                   break;
369               case INT8:
370                   jsonSchema = JsonSchema.INT8_SCHEMA.deepCopy();
371                   break;
372               case INT16:
373                   jsonSchema = JsonSchema.INT16_SCHEMA.deepCopy();
374                   break;
375               case INT32:
376                   jsonSchema = JsonSchema.INT32_SCHEMA.deepCopy();
377                   break;
378               case INT64:
379                   jsonSchema = JsonSchema.INT64_SCHEMA.deepCopy();
380                   break;
381               case STRING:
382                   jsonSchema = JsonSchema.STRING_SCHEMA.deepCopy();
383                   break;
384               case ARRAY:
385                   jsonSchema = JSON_NODE_FACTORY.objectNode().put(JsonSchema.SCHEMA_TYPE_FIELD_NAME, JsonSchema.ARRAY_TYPE_NAME);
386                   jsonSchema.set(JsonSchema.ARRAY_ITEMS_FIELD_NAME, asJsonSchema(schema.valueSchema()));
387                   break;
388               case MAP:
389                   jsonSchema = JSON_NODE_FACTORY.objectNode().put(JsonSchema.SCHEMA_TYPE_FIELD_NAME, JsonSchema.MAP_TYPE_NAME);
390                   jsonSchema.set(JsonSchema.MAP_KEY_FIELD_NAME, asJsonSchema(schema.keySchema()));
391                   jsonSchema.set(JsonSchema.MAP_VALUE_FIELD_NAME, asJsonSchema(schema.valueSchema()));
392                   break;
393               case STRUCT:
394                   jsonSchema = JSON_NODE_FACTORY.objectNode().put(JsonSchema.SCHEMA_TYPE_FIELD_NAME, JsonSchema.STRUCT_TYPE_NAME);
395                   ArrayNode fields = JSON_NODE_FACTORY.arrayNode();
396                   for (Field field : schema.fields()) {
397                       ObjectNode fieldJsonSchema = asJsonSchema(field.schema()).deepCopy();
398                       fieldJsonSchema.put(JsonSchema.STRUCT_FIELD_NAME_FIELD_NAME, field.name());
399                       fields.add(fieldJsonSchema);
400                   }
401                   jsonSchema.set(JsonSchema.STRUCT_FIELDS_FIELD_NAME, fields);
402                   break;
403               default:
404                   throw new DataException("Couldn't translate unsupported schema type " + schema + ".");
405           }
406   
407           jsonSchema.put(JsonSchema.SCHEMA_OPTIONAL_FIELD_NAME, schema.isOptional());
408           if (schema.name() != null)
409               jsonSchema.put(JsonSchema.SCHEMA_NAME_FIELD_NAME, schema.name());
410           if (schema.version() != null)
411               jsonSchema.put(JsonSchema.SCHEMA_VERSION_FIELD_NAME, schema.version());
412           if (schema.doc() != null)
413               jsonSchema.put(JsonSchema.SCHEMA_DOC_FIELD_NAME, schema.doc());
414           if (schema.parameters() != null) {
415               ObjectNode jsonSchemaParams = JSON_NODE_FACTORY.objectNode();
416               for (Map.Entry<String, String> prop : schema.parameters().entrySet())
417                   jsonSchemaParams.put(prop.getKey(), prop.getValue());
418               jsonSchema.set(JsonSchema.SCHEMA_PARAMETERS_FIELD_NAME, jsonSchemaParams);
419           }
420           if (schema.defaultValue() != null)
421               jsonSchema.set(JsonSchema.SCHEMA_DEFAULT_FIELD_NAME, convertToJson(schema, schema.defaultValue()));
422   
423           fromConnectSchemaCache.put(schema, jsonSchema);
424           return jsonSchema;
425       }

428       public Schema asConnectSchema(JsonNode jsonSchema) {
429           if (jsonSchema.isNull())
430               return null;
431   
432           Schema cached = toConnectSchemaCache.get(jsonSchema);
433           if (cached != null)
434               return cached;
435   
436           JsonNode schemaTypeNode = jsonSchema.get(JsonSchema.SCHEMA_TYPE_FIELD_NAME);
437           if (schemaTypeNode == null || !schemaTypeNode.isTextual())
438               throw new DataException("Schema must contain 'type' field");
439   
440           final SchemaBuilder builder;
441           switch (schemaTypeNode.textValue()) {
442               case JsonSchema.BOOLEAN_TYPE_NAME:
443                   builder = SchemaBuilder.bool();
444                   break;
445               case JsonSchema.INT8_TYPE_NAME:
446                   builder = SchemaBuilder.int8();
447                   break;
448               case JsonSchema.INT16_TYPE_NAME:
449                   builder = SchemaBuilder.int16();
450                   break;
451               case JsonSchema.INT32_TYPE_NAME:
452                   builder = SchemaBuilder.int32();
453                   break;
454               case JsonSchema.INT64_TYPE_NAME:
455                   builder = SchemaBuilder.int64();
456                   break;
457               case JsonSchema.FLOAT_TYPE_NAME:
458                   builder = SchemaBuilder.float32();
459                   break;
460               case JsonSchema.DOUBLE_TYPE_NAME:
461                   builder = SchemaBuilder.float64();
462                   break;
463               case JsonSchema.BYTES_TYPE_NAME:
464                   builder = SchemaBuilder.bytes();
465                   break;
466               case JsonSchema.STRING_TYPE_NAME:
467                   builder = SchemaBuilder.string();
468                   break;
469               case JsonSchema.ARRAY_TYPE_NAME:
470                   JsonNode elemSchema = jsonSchema.get(JsonSchema.ARRAY_ITEMS_FIELD_NAME);
471                   if (elemSchema == null || elemSchema.isNull())
472                       throw new DataException("Array schema did not specify the element type");
473                   builder = SchemaBuilder.array(asConnectSchema(elemSchema));
474                   break;
475               case JsonSchema.MAP_TYPE_NAME:
476                   JsonNode keySchema = jsonSchema.get(JsonSchema.MAP_KEY_FIELD_NAME);
477                   if (keySchema == null)
478                       throw new DataException("Map schema did not specify the key type");
479                   JsonNode valueSchema = jsonSchema.get(JsonSchema.MAP_VALUE_FIELD_NAME);
480                   if (valueSchema == null)
481                       throw new DataException("Map schema did not specify the value type");
482                   builder = SchemaBuilder.map(asConnectSchema(keySchema), asConnectSchema(valueSchema));
483                   break;
484               case JsonSchema.STRUCT_TYPE_NAME:
485                   builder = SchemaBuilder.struct();
486                   JsonNode fields = jsonSchema.get(JsonSchema.STRUCT_FIELDS_FIELD_NAME);
487                   if (fields == null || !fields.isArray())
488                       throw new DataException("Struct schema's \"fields\" argument is not an array.");
489                   for (JsonNode field : fields) {
490                       JsonNode jsonFieldName = field.get(JsonSchema.STRUCT_FIELD_NAME_FIELD_NAME);
491                       if (jsonFieldName == null || !jsonFieldName.isTextual())
492                           throw new DataException("Struct schema's field name not specified properly");
493                       builder.field(jsonFieldName.asText(), asConnectSchema(field));
494                   }
495                   break;
496               default:
497                   throw new DataException("Unknown schema type: " + schemaTypeNode.textValue());
498           }
499   
500   
501           JsonNode schemaOptionalNode = jsonSchema.get(JsonSchema.SCHEMA_OPTIONAL_FIELD_NAME);
502           if (schemaOptionalNode != null && schemaOptionalNode.isBoolean() && schemaOptionalNode.booleanValue())
503               builder.optional();
504           else
505               builder.required();
506   
507           JsonNode schemaNameNode = jsonSchema.get(JsonSchema.SCHEMA_NAME_FIELD_NAME);
508           if (schemaNameNode != null && schemaNameNode.isTextual())
509               builder.name(schemaNameNode.textValue());
510   
511           JsonNode schemaVersionNode = jsonSchema.get(JsonSchema.SCHEMA_VERSION_FIELD_NAME);
512           if (schemaVersionNode != null && schemaVersionNode.isIntegralNumber()) {
513               builder.version(schemaVersionNode.intValue());
514           }
515   
516           JsonNode schemaDocNode = jsonSchema.get(JsonSchema.SCHEMA_DOC_FIELD_NAME);
517           if (schemaDocNode != null && schemaDocNode.isTextual())
518               builder.doc(schemaDocNode.textValue());
519   
520           JsonNode schemaParamsNode = jsonSchema.get(JsonSchema.SCHEMA_PARAMETERS_FIELD_NAME);
521           if (schemaParamsNode != null && schemaParamsNode.isObject()) {
522               Iterator<Map.Entry<String, JsonNode>> paramsIt = schemaParamsNode.fields();
523               while (paramsIt.hasNext()) {
524                   Map.Entry<String, JsonNode> entry = paramsIt.next();
525                   JsonNode paramValue = entry.getValue();
526                   if (!paramValue.isTextual())
527                       throw new DataException("Schema parameters must have string values.");
528                   builder.parameter(entry.getKey(), paramValue.textValue());
529               }
530           }
531   
532           JsonNode schemaDefaultNode = jsonSchema.get(JsonSchema.SCHEMA_DEFAULT_FIELD_NAME);
533           if (schemaDefaultNode != null)
534               builder.defaultValue(convertToConnect(builder, schemaDefaultNode));
535   
536           Schema result = builder.build();
537           toConnectSchemaCache.put(jsonSchema, result);
538           return result;
539       }

557       /**
558        * Convert this object, in the org.apache.kafka.connect.data format, into a JSON object, returning both the schema
559        * and the converted object.
560        */
561       private JsonNode convertToJson(Schema schema, Object value) {
562           if (value == null) {
563               if (schema == null) // Any schema is valid and we don't have a default, so treat this as an optional schema
564                   return null;
565               if (schema.defaultValue() != null)
566                   return convertToJson(schema, schema.defaultValue());
567               if (schema.isOptional())
568                   return JSON_NODE_FACTORY.nullNode();
569               throw new DataException("Conversion error: null value for field that is required and has no default value");
570           }
571   
572           if (schema != null && schema.name() != null) {
573               LogicalTypeConverter logicalConverter = LOGICAL_CONVERTERS.get(schema.name());
574               if (logicalConverter != null)
575                   return logicalConverter.toJson(schema, value, config);
576           }
577   
578           try {
579               final Schema.Type schemaType;
580               if (schema == null) {
581                   schemaType = ConnectSchema.schemaType(value.getClass());
582                   if (schemaType == null)
583                       throw new DataException("Java class " + value.getClass() + " does not have corresponding schema type.");
584               } else {
585                   schemaType = schema.type();
586               }
587               switch (schemaType) {
588                   case INT8:
589                       return JSON_NODE_FACTORY.numberNode((Byte) value);
590                   case INT16:
591                       return JSON_NODE_FACTORY.numberNode((Short) value);
592                   case INT32:
593                       return JSON_NODE_FACTORY.numberNode((Integer) value);
594                   case INT64:
595                       return JSON_NODE_FACTORY.numberNode((Long) value);
596                   case FLOAT32:
597                       return JSON_NODE_FACTORY.numberNode((Float) value);
598                   case FLOAT64:
599                       return JSON_NODE_FACTORY.numberNode((Double) value);
600                   case BOOLEAN:
601                       return JSON_NODE_FACTORY.booleanNode((Boolean) value);
602                   case STRING:
603                       CharSequence charSeq = (CharSequence) value;
604                       return JSON_NODE_FACTORY.textNode(charSeq.toString());
605                   case BYTES:
606                       if (value instanceof byte[])
607                           return JSON_NODE_FACTORY.binaryNode((byte[]) value);
608                       else if (value instanceof ByteBuffer)
609                           return JSON_NODE_FACTORY.binaryNode(((ByteBuffer) value).array());
610                       else
611                           throw new DataException("Invalid type for bytes type: " + value.getClass());
612                   case ARRAY: {
613                       Collection<?> collection = (Collection<?>) value;
614                       ArrayNode list = JSON_NODE_FACTORY.arrayNode();
615                       for (Object elem : collection) {
616                           Schema valueSchema = schema == null ? null : schema.valueSchema();
617                           JsonNode fieldValue = convertToJson(valueSchema, elem);
618                           list.add(fieldValue);
619                       }
620                       return list;
621                   }
622                   case MAP: {
623                       Map<?, ?> map = (Map<?, ?>) value;
624                       // If true, using string keys and JSON object; if false, using non-string keys and Array-encoding
625                       boolean objectMode;
626                       if (schema == null) {
627                           objectMode = true;
628                           for (Map.Entry<?, ?> entry : map.entrySet()) {
629                               if (!(entry.getKey() instanceof String)) {
630                                   objectMode = false;
631                                   break;
632                               }
633                           }
634                       } else {
635                           objectMode = schema.keySchema().type() == Schema.Type.STRING;
636                       }
637                       ObjectNode obj = null;
638                       ArrayNode list = null;
639                       if (objectMode)
640                           obj = JSON_NODE_FACTORY.objectNode();
641                       else
642                           list = JSON_NODE_FACTORY.arrayNode();
643                       for (Map.Entry<?, ?> entry : map.entrySet()) {
644                           Schema keySchema = schema == null ? null : schema.keySchema();
645                           Schema valueSchema = schema == null ? null : schema.valueSchema();
646                           JsonNode mapKey = convertToJson(keySchema, entry.getKey());
647                           JsonNode mapValue = convertToJson(valueSchema, entry.getValue());
648   
649                           if (objectMode)
650                               obj.set(mapKey.asText(), mapValue);
651                           else
652                               list.add(JSON_NODE_FACTORY.arrayNode().add(mapKey).add(mapValue));
653                       }
654                       return objectMode ? obj : list;
655                   }
656                   case STRUCT: {
657                       Struct struct = (Struct) value;
658                       if (!struct.schema().equals(schema))
659                           throw new DataException("Mismatching schema.");
660                       ObjectNode obj = JSON_NODE_FACTORY.objectNode();
661                       for (Field field : schema.fields()) {
662                           obj.set(field.name(), convertToJson(field.schema(), struct.get(field)));
663                       }
664                       return obj;
665                   }
666               }
667   
668               throw new DataException("Couldn't convert " + value + " to JSON.");
669           } catch (ClassCastException e) {
670               String schemaTypeStr = (schema != null) ? schema.type().toString() : "unknown schema";
671               throw new DataException("Invalid type for " + schemaTypeStr + ": " + value.getClass());
672           }
673       }

676       private static Object convertToConnect(Schema schema, JsonNode jsonValue) {
677           final Schema.Type schemaType;
678           if (schema != null) {
679               schemaType = schema.type();
680               if (jsonValue == null || jsonValue.isNull()) {
681                   if (schema.defaultValue() != null)
682                       return schema.defaultValue(); // any logical type conversions should already have been applied
683                   if (schema.isOptional())
684                       return null;
685                   throw new DataException("Invalid null value for required " + schemaType +  " field");
686               }
687           } else {
688               switch (jsonValue.getNodeType()) {
689                   case NULL:
690                   case MISSING:
691                       // Special case. With no schema
692                       return null;
693                   case BOOLEAN:
694                       schemaType = Schema.Type.BOOLEAN;
695                       break;
696                   case NUMBER:
697                       if (jsonValue.isIntegralNumber())
698                           schemaType = Schema.Type.INT64;
699                       else
700                           schemaType = Schema.Type.FLOAT64;
701                       break;
702                   case ARRAY:
703                       schemaType = Schema.Type.ARRAY;
704                       break;
705                   case OBJECT:
706                       schemaType = Schema.Type.MAP;
707                       break;
708                   case STRING:
709                       schemaType = Schema.Type.STRING;
710                       break;
711   
712                   case BINARY:
713                   case POJO:
714                   default:
715                       schemaType = null;
716                       break;
717               }
718           }
719   
720           final JsonToConnectTypeConverter typeConverter = TO_CONNECT_CONVERTERS.get(schemaType);
721           if (typeConverter == null)
722               throw new DataException("Unknown schema type: " + schemaType);
723   
724           if (schema != null && schema.name() != null) {
725               LogicalTypeConverter logicalConverter = LOGICAL_CONVERTERS.get(schema.name());
726               if (logicalConverter != null)
727                   return logicalConverter.toConnect(schema, jsonValue);
728           }
729   
730           return typeConverter.convert(schema, jsonValue);
731       }
