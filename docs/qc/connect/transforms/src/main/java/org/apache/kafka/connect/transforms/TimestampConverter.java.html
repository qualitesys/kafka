368       private R applyWithSchema(R record) {
369           final Schema schema = operatingSchema(record);
370           if (config.field.isEmpty()) {
371               Object value = operatingValue(record);
372               // New schema is determined by the requested target timestamp type
373               Schema updatedSchema = TRANSLATORS.get(config.type).typeSchema(schema.isOptional());
374               return newRecord(record, updatedSchema, convertTimestamp(value, timestampTypeFromSchema(schema)));
375           } else {
376               final Struct value = requireStructOrNull(operatingValue(record), PURPOSE);
377               Schema updatedSchema = schemaUpdateCache.get(schema);
378               if (updatedSchema == null) {
379                   SchemaBuilder builder = SchemaUtil.copySchemaBasics(schema, SchemaBuilder.struct());
380                   for (Field field : schema.fields()) {
381                       if (field.name().equals(config.field)) {
382                           builder.field(field.name(), TRANSLATORS.get(config.type).typeSchema(field.schema().isOptional()));
383                       } else {
384                           builder.field(field.name(), field.schema());
385                       }
386                   }
387                   if (schema.isOptional())
388                       builder.optional();
389                   if (schema.defaultValue() != null) {
390                       Struct updatedDefaultValue = applyValueWithSchema((Struct) schema.defaultValue(), builder);
391                       builder.defaultValue(updatedDefaultValue);
392                   }
393   
394                   updatedSchema = builder.build();
395                   schemaUpdateCache.put(schema, updatedSchema);
396               }
397   
398               Struct updatedValue = applyValueWithSchema(value, updatedSchema);
399               return newRecord(record, updatedSchema, updatedValue);
400           }
401       }
402   
403       private Struct applyValueWithSchema(Struct value, Schema updatedSchema) {
404           if (value == null) {
405               return null;
406           }
407           Struct updatedValue = new Struct(updatedSchema);
408           for (Field field : value.schema().fields()) {
409               final Object updatedFieldValue;
410               if (field.name().equals(config.field)) {
411                   updatedFieldValue = convertTimestamp(value.get(field), timestampTypeFromSchema(field.schema()));
412               } else {
413                   updatedFieldValue = value.get(field);
414               }
415               updatedValue.put(field.name(), updatedFieldValue);
416           }
417           return updatedValue;
418       }
419   
420       private R applySchemaless(R record) {
421           Object rawValue = operatingValue(record);
422           if (rawValue == null || config.field.isEmpty()) {
423               return newRecord(record, null, convertTimestamp(rawValue));
424           } else {
425               final Map<String, Object> value = requireMap(rawValue, PURPOSE);
426               final HashMap<String, Object> updatedValue = new HashMap<>(value);
427               updatedValue.put(config.field, convertTimestamp(value.get(config.field)));
428               return newRecord(record, null, updatedValue);
429           }
430       }

432       /**
433        * Determine the type/format of the timestamp based on the schema
434        */
435       private String timestampTypeFromSchema(Schema schema) {
436           if (Timestamp.LOGICAL_NAME.equals(schema.name())) {
437               return TYPE_TIMESTAMP;
438           } else if (org.apache.kafka.connect.data.Date.LOGICAL_NAME.equals(schema.name())) {
439               return TYPE_DATE;
440           } else if (Time.LOGICAL_NAME.equals(schema.name())) {
441               return TYPE_TIME;
442           } else if (schema.type().equals(Schema.Type.STRING)) {
443               // If not otherwise specified, string == user-specified string format for timestamps
444               return TYPE_STRING;
445           } else if (schema.type().equals(Schema.Type.INT64)) {
446               // If not otherwise specified, long == unix time
447               return TYPE_UNIX;
448           }
449           throw new ConnectException("Schema " + schema + " does not correspond to a known timestamp type format");
450       }
451   
452       /**
453        * Infer the type/format of the timestamp based on the raw Java type
454        */
455       private String inferTimestampType(Object timestamp) {
456           // Note that we can't infer all types, e.g. Date/Time/Timestamp all have the same runtime representation as a
457           // java.util.Date
458           if (timestamp instanceof Date) {
459               return TYPE_TIMESTAMP;
460           } else if (timestamp instanceof Long) {
461               return TYPE_UNIX;
462           } else if (timestamp instanceof String) {
463               return TYPE_STRING;
464           }
465           throw new DataException("TimestampConverter does not support " + timestamp.getClass() + " objects as timestamps");
466       }
467   
468       /**
469        * Convert the given timestamp to the target timestamp format.
470        * @param timestamp the input timestamp, may be null
471        * @param timestampFormat the format of the timestamp, or null if the format should be inferred
472        * @return the converted timestamp
473        */
474       private Object convertTimestamp(Object timestamp, String timestampFormat) {
475           if (timestamp == null) {
476               return null;
477           }
478           if (timestampFormat == null) {
479               timestampFormat = inferTimestampType(timestamp);
480           }
481   
482           TimestampTranslator sourceTranslator = TRANSLATORS.get(timestampFormat);
483           if (sourceTranslator == null) {
484               throw new ConnectException("Unsupported timestamp type: " + timestampFormat);
485           }
486           Date rawTimestamp = sourceTranslator.toRaw(config, timestamp);
487   
488           TimestampTranslator targetTranslator = TRANSLATORS.get(config.type);
489           if (targetTranslator == null) {
490               throw new ConnectException("Unsupported timestamp type: " + config.type);
491           }
492           return targetTranslator.toType(config, rawTimestamp);
493       }
494   
495       private Object convertTimestamp(Object timestamp) {
496           return convertTimestamp(timestamp, null);
497       }
