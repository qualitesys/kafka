17   package org.apache.kafka.connect.transforms;
18   
19   import org.apache.kafka.common.config.ConfigDef;
20   import org.apache.kafka.connect.connector.ConnectRecord;
21   import org.apache.kafka.connect.transforms.util.RegexValidator;
22   import org.apache.kafka.connect.transforms.util.SimpleConfig;
23   import org.slf4j.Logger;
24   import org.slf4j.LoggerFactory;
25   
26   import java.util.Map;
27   import java.util.regex.Matcher;
28   import java.util.regex.Pattern;
29   
30   public class RegexRouter<R extends ConnectRecord<R>> implements Transformation<R> {
31   
32       private static final Logger log = LoggerFactory.getLogger(RegexRouter.class);
33   
34       public static final String OVERVIEW_DOC = "Update the record topic using the configured regular expression and replacement string."
35               + "<p/>Under the hood, the regex is compiled to a <code>java.util.regex.Pattern</code>. "
36               + "If the pattern matches the input topic, <code>java.util.regex.Matcher#replaceFirst()</code> is used with the replacement string to obtain the new topic.";
37   
38       public static final ConfigDef CONFIG_DEF = new ConfigDef()
39               .define(ConfigName.REGEX, ConfigDef.Type.STRING, ConfigDef.NO_DEFAULT_VALUE, new RegexValidator(), ConfigDef.Importance.HIGH,
40                       "Regular expression to use for matching.")
41               .define(ConfigName.REPLACEMENT, ConfigDef.Type.STRING, ConfigDef.NO_DEFAULT_VALUE, ConfigDef.Importance.HIGH,
42                       "Replacement string.");
43   
44       private interface ConfigName {
45           String REGEX = "regex";
46           String REPLACEMENT = "replacement";
47       }
48   
49       private Pattern regex;
50       private String replacement;
51   
52       @Override
53       public void configure(Map<String, ?> props) {
54           final SimpleConfig config = new SimpleConfig(CONFIG_DEF, props);
55           regex = Pattern.compile(config.getString(ConfigName.REGEX));
56           replacement = config.getString(ConfigName.REPLACEMENT);
57       }
58   
59       @Override
60       public R apply(R record) {
61           final Matcher matcher = regex.matcher(record.topic());
62           if (matcher.matches()) {
63               final String topic = matcher.replaceFirst(replacement);
64               log.trace("Rerouting from topic '{}' to new topic '{}'", record.topic(), topic);
65               return record.newRecord(topic, record.kafkaPartition(), record.keySchema(), record.key(), record.valueSchema(), record.value(), record.timestamp());
66           } else {
67               log.trace("Not rerouting topic '{}' as it does not match the configured regex", record.topic());
68           }
69           return record;
70       }
71   
72       @Override
73       public void close() {
74       }
75   
76       @Override
77       public ConfigDef config() {
78           return CONFIG_DEF;
79       }
80   
81   }
