17   package org.apache.kafka.connect.transforms;
18   
19   import org.apache.kafka.common.config.ConfigDef;
20   import org.apache.kafka.connect.connector.ConnectRecord;
21   import org.apache.kafka.connect.errors.DataException;
22   import org.apache.kafka.connect.transforms.util.SimpleConfig;
23   
24   import java.text.SimpleDateFormat;
25   import java.util.Date;
26   import java.util.Map;
27   import java.util.TimeZone;
28   import java.util.regex.Matcher;
29   import java.util.regex.Pattern;
30   
31   public class TimestampRouter<R extends ConnectRecord<R>> implements Transformation<R>, AutoCloseable {
32   
33       private static final Pattern TOPIC = Pattern.compile("${topic}", Pattern.LITERAL);
34   
35       private static final Pattern TIMESTAMP = Pattern.compile("${timestamp}", Pattern.LITERAL);
36   
37       public static final String OVERVIEW_DOC =
38               "Update the record's topic field as a function of the original topic value and the record timestamp."
39                       + "<p/>"
40                       + "This is mainly useful for sink connectors, since the topic field is often used to determine the equivalent entity name in the destination system"
41                       + "(e.g. database table or search index name).";
42   
43       public static final ConfigDef CONFIG_DEF = new ConfigDef()
44               .define(ConfigName.TOPIC_FORMAT, ConfigDef.Type.STRING, "${topic}-${timestamp}", ConfigDef.Importance.HIGH,
45                       "Format string which can contain <code>${topic}</code> and <code>${timestamp}</code> as placeholders for the topic and timestamp, respectively.")
46               .define(ConfigName.TIMESTAMP_FORMAT, ConfigDef.Type.STRING, "yyyyMMdd", ConfigDef.Importance.HIGH,
47                       "Format string for the timestamp that is compatible with <code>java.text.SimpleDateFormat</code>.");
48   
49       private interface ConfigName {
50           String TOPIC_FORMAT = "topic.format";
51           String TIMESTAMP_FORMAT = "timestamp.format";
52       }
53   
54       private String topicFormat;
55       private ThreadLocal<SimpleDateFormat> timestampFormat;
56   
57       @Override
58       public void configure(Map<String, ?> props) {
59           final SimpleConfig config = new SimpleConfig(CONFIG_DEF, props);
60   
61           topicFormat = config.getString(ConfigName.TOPIC_FORMAT);
62   
63           final String timestampFormatStr = config.getString(ConfigName.TIMESTAMP_FORMAT);
64           timestampFormat = ThreadLocal.withInitial(() -> {
65               final SimpleDateFormat fmt = new SimpleDateFormat(timestampFormatStr);
66               fmt.setTimeZone(TimeZone.getTimeZone("UTC"));
67               return fmt;
68           });
69       }
70   
71       @Override
72       public R apply(R record) {
73           final Long timestamp = record.timestamp();
74           if (timestamp == null) {
75               throw new DataException("Timestamp missing on record: " + record);
76           }
77           final String formattedTimestamp = timestampFormat.get().format(new Date(timestamp));
78   
79           final String replace1 = TOPIC.matcher(topicFormat).replaceAll(Matcher.quoteReplacement(record.topic()));
80           final String updatedTopic = TIMESTAMP.matcher(replace1).replaceAll(Matcher.quoteReplacement(formattedTimestamp));
81           return record.newRecord(
82                   updatedTopic, record.kafkaPartition(),
83                   record.keySchema(), record.key(),
84                   record.valueSchema(), record.value(),
85                   record.timestamp()
86           );
87       }
88   
89       @Override
90       public void close() {
91           timestampFormat.remove();
92       }
93   
94       @Override
95       public ConfigDef config() {
96           return CONFIG_DEF;
97       }
98   
99   }
