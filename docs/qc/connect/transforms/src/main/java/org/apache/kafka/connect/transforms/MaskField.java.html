43   public abstract class MaskField<R extends ConnectRecord<R>> implements Transformation<R> {
44   
45       public static final String OVERVIEW_DOC =
46               "Mask specified fields with a valid null value for the field type (i.e. 0, false, empty string, and so on)."
47                       + "<p/>For numeric and string fields, an optional replacement value can be specified that is converted to the correct type."
48                       + "<p/>Use the concrete transformation type designed for the record key (<code>" + Key.class.getName()
49                       + "</code>) or value (<code>" + Value.class.getName() + "</code>).";
50   
51       private static final String FIELDS_CONFIG = "fields";
52       private static final String REPLACEMENT_CONFIG = "replacement";
53   
54       public static final ConfigDef CONFIG_DEF = new ConfigDef()
55               .define(FIELDS_CONFIG, ConfigDef.Type.LIST, ConfigDef.NO_DEFAULT_VALUE, new NonEmptyListValidator(),
56                       ConfigDef.Importance.HIGH, "Names of fields to mask.")
57               .define(REPLACEMENT_CONFIG, ConfigDef.Type.STRING, null, new ConfigDef.NonEmptyString(),
58                       ConfigDef.Importance.LOW, "Custom value replacement, that will be applied to all"
59                               + " 'fields' values (numeric or non-empty string values only).");
60   
61       private static final String PURPOSE = "mask fields";
62   
63       private static final Map<Class<?>, Function<String, ?>> REPLACEMENT_MAPPING_FUNC = new HashMap<>();
64       private static final Map<Class<?>, Object> PRIMITIVE_VALUE_MAPPING = new HashMap<>();
65   
66       static {
67           PRIMITIVE_VALUE_MAPPING.put(Boolean.class, Boolean.FALSE);
68           PRIMITIVE_VALUE_MAPPING.put(Byte.class, (byte) 0);
69           PRIMITIVE_VALUE_MAPPING.put(Short.class, (short) 0);
70           PRIMITIVE_VALUE_MAPPING.put(Integer.class, 0);
71           PRIMITIVE_VALUE_MAPPING.put(Long.class, 0L);
72           PRIMITIVE_VALUE_MAPPING.put(Float.class, 0f);
73           PRIMITIVE_VALUE_MAPPING.put(Double.class, 0d);
74           PRIMITIVE_VALUE_MAPPING.put(BigInteger.class, BigInteger.ZERO);
75           PRIMITIVE_VALUE_MAPPING.put(BigDecimal.class, BigDecimal.ZERO);
76           PRIMITIVE_VALUE_MAPPING.put(Date.class, new Date(0));
77           PRIMITIVE_VALUE_MAPPING.put(String.class, "");
78   
79           REPLACEMENT_MAPPING_FUNC.put(Byte.class, v -> Values.convertToByte(null, v));
80           REPLACEMENT_MAPPING_FUNC.put(Short.class, v -> Values.convertToShort(null, v));
81           REPLACEMENT_MAPPING_FUNC.put(Integer.class, v -> Values.convertToInteger(null, v));
82           REPLACEMENT_MAPPING_FUNC.put(Long.class, v -> Values.convertToLong(null, v));
83           REPLACEMENT_MAPPING_FUNC.put(Float.class, v -> Values.convertToFloat(null, v));
84           REPLACEMENT_MAPPING_FUNC.put(Double.class, v -> Values.convertToDouble(null, v));
85           REPLACEMENT_MAPPING_FUNC.put(String.class, Function.identity());
86           REPLACEMENT_MAPPING_FUNC.put(BigDecimal.class, BigDecimal::new);
87           REPLACEMENT_MAPPING_FUNC.put(BigInteger.class, BigInteger::new);
88       }
89   
90       private Set<String> maskedFields;
91       private String replacement;
92   
93       @Override
94       public void configure(Map<String, ?> props) {
95           final SimpleConfig config = new SimpleConfig(CONFIG_DEF, props);
96           maskedFields = new HashSet<>(config.getList(FIELDS_CONFIG));
97           replacement = config.getString(REPLACEMENT_CONFIG);
98       }
99   
100       @Override
101       public R apply(R record) {
102           if (operatingSchema(record) == null) {
103               return applySchemaless(record);
104           } else {
105               return applyWithSchema(record);
106           }
107       }
108   
109       private R applySchemaless(R record) {
110           final Map<String, Object> value = requireMap(operatingValue(record), PURPOSE);
111           final HashMap<String, Object> updatedValue = new HashMap<>(value);
112           for (String field : maskedFields) {
113               updatedValue.put(field, masked(value.get(field)));
114           }
115           return newRecord(record, updatedValue);
116       }
117   
118       private R applyWithSchema(R record) {
119           final Struct value = requireStruct(operatingValue(record), PURPOSE);
120           final Struct updatedValue = new Struct(value.schema());
121           for (Field field : value.schema().fields()) {
122               final Object origFieldValue = value.get(field);
123               updatedValue.put(field, maskedFields.contains(field.name()) ? masked(origFieldValue) : origFieldValue);
124           }
125           return newRecord(record, updatedValue);
126       }
127   
128       private Object masked(Object value) {
129           if (value == null) {
130               return null;
131           }
132           return replacement == null ? maskWithNullValue(value) : maskWithCustomReplacement(value, replacement);
133       }
134   
135       private static Object maskWithCustomReplacement(Object value, String replacement) {
136           Function<String, ?> replacementMapper = REPLACEMENT_MAPPING_FUNC.get(value.getClass());
137           if (replacementMapper == null) {
138               throw new DataException("Cannot mask value of type " + value.getClass() + " with custom replacement.");
139           }
140           try {
141               return replacementMapper.apply(replacement);
142           } catch (NumberFormatException ex) {
143               throw new DataException("Unable to convert " + replacement + " (" + replacement.getClass() + ") to number", ex);
144           }
145       }
146   
147       private static Object maskWithNullValue(Object value) {
148           Object maskedValue = PRIMITIVE_VALUE_MAPPING.get(value.getClass());
149           if (maskedValue == null) {
150               if (value instanceof List)
151                   maskedValue = Collections.emptyList();
152               else if (value instanceof Map)
153                   maskedValue = Collections.emptyMap();
154               else
155                   throw new DataException("Cannot mask value of type: " + value.getClass());
156           }
157           return maskedValue;
158       }

161       public ConfigDef config() {
162           return CONFIG_DEF;
163       }

166       public void close() {
167       }
