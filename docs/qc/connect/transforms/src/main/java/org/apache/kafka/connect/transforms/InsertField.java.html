41   public abstract class InsertField<R extends ConnectRecord<R>> implements Transformation<R> {
42   
43       public static final String OVERVIEW_DOC =
44               "Insert field(s) using attributes from the record metadata or a configured static value."
45                       + "<p/>Use the concrete transformation type designed for the record key (<code>" + Key.class.getName() + "</code>) "
46                       + "or value (<code>" + Value.class.getName() + "</code>).";
47   
48       private interface ConfigName {
49           String TOPIC_FIELD = "topic.field";
50           String PARTITION_FIELD = "partition.field";
51           String OFFSET_FIELD = "offset.field";
52           String TIMESTAMP_FIELD = "timestamp.field";
53           String STATIC_FIELD = "static.field";
54           String STATIC_VALUE = "static.value";
55       }
56   
57       private static final String OPTIONALITY_DOC = "Suffix with <code>!</code> to make this a required field, or <code>?</code> to keep it optional (the default).";
58   
59       public static final ConfigDef CONFIG_DEF = new ConfigDef()
60               .define(ConfigName.TOPIC_FIELD, ConfigDef.Type.STRING, null, ConfigDef.Importance.MEDIUM,
61                       "Field name for Kafka topic. " + OPTIONALITY_DOC)
62               .define(ConfigName.PARTITION_FIELD, ConfigDef.Type.STRING, null, ConfigDef.Importance.MEDIUM,
63                       "Field name for Kafka partition. " + OPTIONALITY_DOC)
64               .define(ConfigName.OFFSET_FIELD, ConfigDef.Type.STRING, null, ConfigDef.Importance.MEDIUM,
65                       "Field name for Kafka offset - only applicable to sink connectors.<br/>" + OPTIONALITY_DOC)
66               .define(ConfigName.TIMESTAMP_FIELD, ConfigDef.Type.STRING, null, ConfigDef.Importance.MEDIUM,
67                       "Field name for record timestamp. " + OPTIONALITY_DOC)
68               .define(ConfigName.STATIC_FIELD, ConfigDef.Type.STRING, null, ConfigDef.Importance.MEDIUM,
69                       "Field name for static data field. " + OPTIONALITY_DOC)
70               .define(ConfigName.STATIC_VALUE, ConfigDef.Type.STRING, null, ConfigDef.Importance.MEDIUM,
71                       "Static field value, if field name configured.");
72   
73       private static final String PURPOSE = "field insertion";
74   
75       private static final Schema OPTIONAL_TIMESTAMP_SCHEMA = Timestamp.builder().optional().build();
76   
77       private static final class InsertionSpec {
78           final String name;
79           final boolean optional;
80   
81           private InsertionSpec(String name, boolean optional) {
82               this.name = name;
83               this.optional = optional;
84           }
85   
86           public static InsertionSpec parse(String spec) {
87               if (spec == null) return null;
88               if (spec.endsWith("?")) {
89                   return new InsertionSpec(spec.substring(0, spec.length() - 1), true);
90               }
91               if (spec.endsWith("!")) {
92                   return new InsertionSpec(spec.substring(0, spec.length() - 1), false);
93               }
94               return new InsertionSpec(spec, true);
95           }
96       }
97   
98       private InsertionSpec topicField;
99       private InsertionSpec partitionField;
100       private InsertionSpec offsetField;
101       private InsertionSpec timestampField;
102       private InsertionSpec staticField;
103       private String staticValue;
104   
105       private Cache<Schema, Schema> schemaUpdateCache;
106   
107       @Override
108       public void configure(Map<String, ?> props) {
109           final SimpleConfig config = new SimpleConfig(CONFIG_DEF, props);
110           topicField = InsertionSpec.parse(config.getString(ConfigName.TOPIC_FIELD));
111           partitionField = InsertionSpec.parse(config.getString(ConfigName.PARTITION_FIELD));
112           offsetField = InsertionSpec.parse(config.getString(ConfigName.OFFSET_FIELD));
113           timestampField = InsertionSpec.parse(config.getString(ConfigName.TIMESTAMP_FIELD));
114           staticField = InsertionSpec.parse(config.getString(ConfigName.STATIC_FIELD));
115           staticValue = config.getString(ConfigName.STATIC_VALUE);
116   
117           if (topicField == null && partitionField == null && offsetField == null && timestampField == null && staticField == null) {
118               throw new ConfigException("No field insertion configured");
119           }
120   
121           if (staticField != null && staticValue == null) {
122               throw new ConfigException(ConfigName.STATIC_VALUE, null, "No value specified for static field: " + staticField);
123           }
124   
125           schemaUpdateCache = new SynchronizedCache<>(new LRUCache<>(16));
126       }
127   
128       @Override
129       public R apply(R record) {
130           if (operatingValue(record) == null) {
131               return record;
132           } else if (operatingSchema(record) == null) {
133               return applySchemaless(record);
134           } else {
135               return applyWithSchema(record);
136           }
137       }
138   
139       private R applySchemaless(R record) {
140           final Map<String, Object> value = requireMap(operatingValue(record), PURPOSE);
141   
142           final Map<String, Object> updatedValue = new HashMap<>(value);
143   
144           if (topicField != null) {
145               updatedValue.put(topicField.name, record.topic());
146           }
147           if (partitionField != null && record.kafkaPartition() != null) {
148               updatedValue.put(partitionField.name, record.kafkaPartition());
149           }
150           if (offsetField != null) {
151               updatedValue.put(offsetField.name, requireSinkRecord(record, PURPOSE).kafkaOffset());
152           }
153           if (timestampField != null && record.timestamp() != null) {
154               updatedValue.put(timestampField.name, record.timestamp());
155           }
156           if (staticField != null && staticValue != null) {
157               updatedValue.put(staticField.name, staticValue);
158           }
159   
160           return newRecord(record, null, updatedValue);
161       }
162   
163       private R applyWithSchema(R record) {
164           final Struct value = requireStruct(operatingValue(record), PURPOSE);
165   
166           Schema updatedSchema = schemaUpdateCache.get(value.schema());
167           if (updatedSchema == null) {
168               updatedSchema = makeUpdatedSchema(value.schema());
169               schemaUpdateCache.put(value.schema(), updatedSchema);
170           }
171   
172           final Struct updatedValue = new Struct(updatedSchema);
173   
174           for (Field field : value.schema().fields()) {
175               updatedValue.put(field.name(), value.get(field));
176           }
177   
178           if (topicField != null) {
179               updatedValue.put(topicField.name, record.topic());
180           }
181           if (partitionField != null && record.kafkaPartition() != null) {
182               updatedValue.put(partitionField.name, record.kafkaPartition());
183           }
184           if (offsetField != null) {
185               updatedValue.put(offsetField.name, requireSinkRecord(record, PURPOSE).kafkaOffset());
186           }
187           if (timestampField != null && record.timestamp() != null) {
188               updatedValue.put(timestampField.name, new Date(record.timestamp()));
189           }
190           if (staticField != null && staticValue != null) {
191               updatedValue.put(staticField.name, staticValue);
192           }
193   
194           return newRecord(record, updatedSchema, updatedValue);
195       }
196   
197       private Schema makeUpdatedSchema(Schema schema) {
198           final SchemaBuilder builder = SchemaUtil.copySchemaBasics(schema, SchemaBuilder.struct());
199   
200           for (Field field : schema.fields()) {
201               builder.field(field.name(), field.schema());
202           }
203   
204           if (topicField != null) {
205               builder.field(topicField.name, topicField.optional ? Schema.OPTIONAL_STRING_SCHEMA : Schema.STRING_SCHEMA);
206           }
207           if (partitionField != null) {
208               builder.field(partitionField.name, partitionField.optional ? Schema.OPTIONAL_INT32_SCHEMA : Schema.INT32_SCHEMA);
209           }
210           if (offsetField != null) {
211               builder.field(offsetField.name, offsetField.optional ? Schema.OPTIONAL_INT64_SCHEMA : Schema.INT64_SCHEMA);
212           }
213           if (timestampField != null) {
214               builder.field(timestampField.name, timestampField.optional ? OPTIONAL_TIMESTAMP_SCHEMA : Timestamp.SCHEMA);
215           }
216           if (staticField != null) {
217               builder.field(staticField.name, staticField.optional ? Schema.OPTIONAL_STRING_SCHEMA : Schema.STRING_SCHEMA);
218           }
219   
220           return builder.build();
221       }
222   
223       @Override
224       public void close() {
225           schemaUpdateCache = null;
226       }
227   
228       @Override
229       public ConfigDef config() {
230           return CONFIG_DEF;
231       }

249           }

254           }
