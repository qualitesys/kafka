42   public abstract class ReplaceField<R extends ConnectRecord<R>> implements Transformation<R> {
43   
44       public static final String OVERVIEW_DOC = "Filter or rename fields."
45               + "<p/>Use the concrete transformation type designed for the record key (<code>" + Key.class.getName() + "</code>) "
46               + "or value (<code>" + Value.class.getName() + "</code>).";
47   
48       interface ConfigName {
49           String EXCLUDE = "exclude";
50           String INCLUDE = "include";
51   
52           // for backwards compatibility
53           String INCLUDE_ALIAS = "whitelist";
54           String EXCLUDE_ALIAS = "blacklist";
55   
56           String RENAME = "renames";
57       }
58   
59       public static final ConfigDef CONFIG_DEF = new ConfigDef()
60               .define(ConfigName.EXCLUDE, ConfigDef.Type.LIST, Collections.emptyList(), ConfigDef.Importance.MEDIUM,
61                       "Fields to exclude. This takes precedence over the fields to include.")
62               .define("blacklist", ConfigDef.Type.LIST, null, Importance.LOW,
63                       "Deprecated. Use " + ConfigName.EXCLUDE + " instead.")
64               .define(ConfigName.INCLUDE, ConfigDef.Type.LIST, Collections.emptyList(), ConfigDef.Importance.MEDIUM,
65                       "Fields to include. If specified, only these fields will be used.")
66               .define("whitelist", ConfigDef.Type.LIST, null, Importance.LOW,
67                       "Deprecated. Use " + ConfigName.INCLUDE + " instead.")
68               .define(ConfigName.RENAME, ConfigDef.Type.LIST, Collections.emptyList(), new ConfigDef.Validator() {
69                   @SuppressWarnings("unchecked")
70                   @Override
71                   public void ensureValid(String name, Object value) {
72                       parseRenameMappings((List<String>) value);
73                   }
74   
75                   @Override
76                   public String toString() {
77                       return "list of colon-delimited pairs, e.g. <code>foo:bar,abc:xyz</code>";
78                   }
79               }, ConfigDef.Importance.MEDIUM, "Field rename mappings.");
80   
81       private static final String PURPOSE = "field replacement";
82   
83       private List<String> exclude;
84       private List<String> include;
85       private Map<String, String> renames;
86       private Map<String, String> reverseRenames;
87   
88       private Cache<Schema, Schema> schemaUpdateCache;
89   
90       @Override
91       public void configure(Map<String, ?> configs) {
92           final SimpleConfig config = new SimpleConfig(CONFIG_DEF, ConfigUtils.translateDeprecatedConfigs(configs, new String[][]{
93               {ConfigName.INCLUDE, "whitelist"},
94               {ConfigName.EXCLUDE, "blacklist"},
95           }));
96   
97           exclude = config.getList(ConfigName.EXCLUDE);
98           include = config.getList(ConfigName.INCLUDE);
99           renames = parseRenameMappings(config.getList(ConfigName.RENAME));
100           reverseRenames = invert(renames);
101   
102           schemaUpdateCache = new SynchronizedCache<>(new LRUCache<>(16));
103       }
104   
105       static Map<String, String> parseRenameMappings(List<String> mappings) {
106           final Map<String, String> m = new HashMap<>();
107           for (String mapping : mappings) {
108               final String[] parts = mapping.split(":");
109               if (parts.length != 2) {
110                   throw new ConfigException(ConfigName.RENAME, mappings, "Invalid rename mapping: " + mapping);
111               }
112               m.put(parts[0], parts[1]);
113           }
114           return m;
115       }
116   
117       static Map<String, String> invert(Map<String, String> source) {
118           final Map<String, String> m = new HashMap<>();
119           for (Map.Entry<String, String> e : source.entrySet()) {
120               m.put(e.getValue(), e.getKey());
121           }
122           return m;
123       }
124   
125       boolean filter(String fieldName) {
126           return !exclude.contains(fieldName) && (include.isEmpty() || include.contains(fieldName));
127       }
128   
129       String renamed(String fieldName) {
130           final String mapping = renames.get(fieldName);
131           return mapping == null ? fieldName : mapping;
132       }
133   
134       String reverseRenamed(String fieldName) {
135           final String mapping = reverseRenames.get(fieldName);
136           return mapping == null ? fieldName : mapping;
137       }
138   
139       @Override
140       public R apply(R record) {
141           if (operatingValue(record) == null) {
142               return record;
143           } else if (operatingSchema(record) == null) {
144               return applySchemaless(record);
145           } else {
146               return applyWithSchema(record);
147           }
148       }
149   
150       private R applySchemaless(R record) {
151           final Map<String, Object> value = requireMap(operatingValue(record), PURPOSE);
152   
153           final Map<String, Object> updatedValue = new HashMap<>(value.size());
154   
155           for (Map.Entry<String, Object> e : value.entrySet()) {
156               final String fieldName = e.getKey();
157               if (filter(fieldName)) {
158                   final Object fieldValue = e.getValue();
159                   updatedValue.put(renamed(fieldName), fieldValue);
160               }
161           }
162   
163           return newRecord(record, null, updatedValue);
164       }
165   
166       private R applyWithSchema(R record) {
167           final Struct value = requireStruct(operatingValue(record), PURPOSE);
168   
169           Schema updatedSchema = schemaUpdateCache.get(value.schema());
170           if (updatedSchema == null) {
171               updatedSchema = makeUpdatedSchema(value.schema());
172               schemaUpdateCache.put(value.schema(), updatedSchema);
173           }
174   
175           final Struct updatedValue = new Struct(updatedSchema);
176   
177           for (Field field : updatedSchema.fields()) {
178               final Object fieldValue = value.get(reverseRenamed(field.name()));
179               updatedValue.put(field.name(), fieldValue);
180           }
181   
182           return newRecord(record, updatedSchema, updatedValue);
183       }
184   
185       private Schema makeUpdatedSchema(Schema schema) {
186           final SchemaBuilder builder = SchemaUtil.copySchemaBasics(schema, SchemaBuilder.struct());
187           for (Field field : schema.fields()) {
188               if (filter(field.name())) {
189                   builder.field(renamed(field.name()), field.schema());
190               }
191           }
192           return builder.build();
193       }
194   
195       @Override
196       public ConfigDef config() {
197           return CONFIG_DEF;
198       }
199   
200       @Override
201       public void close() {
202           schemaUpdateCache = null;
203       }
