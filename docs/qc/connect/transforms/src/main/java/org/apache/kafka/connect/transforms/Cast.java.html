144       private R applySchemaless(R record) {
145           if (wholeValueCastType != null) {
146               return newRecord(record, null, castValueToType(null, operatingValue(record), wholeValueCastType));
147           }
148   
149           final Map<String, Object> value = requireMap(operatingValue(record), PURPOSE);
150           final HashMap<String, Object> updatedValue = new HashMap<>(value);
151           for (Map.Entry<String, Schema.Type> fieldSpec : casts.entrySet()) {
152               String field = fieldSpec.getKey();
153               updatedValue.put(field, castValueToType(null, value.get(field), fieldSpec.getValue()));
154           }
155           return newRecord(record, null, updatedValue);
156       }

158       private R applyWithSchema(R record) {
159           Schema valueSchema = operatingSchema(record);
160           Schema updatedSchema = getOrBuildSchema(valueSchema);
161   
162           // Whole-record casting
163           if (wholeValueCastType != null)
164               return newRecord(record, updatedSchema, castValueToType(valueSchema, operatingValue(record), wholeValueCastType));
165   
166           // Casting within a struct
167           final Struct value = requireStruct(operatingValue(record), PURPOSE);
168   
169           final Struct updatedValue = new Struct(updatedSchema);
170           for (Field field : value.schema().fields()) {
171               final Object origFieldValue = value.get(field);
172               final Schema.Type targetType = casts.get(field.name());
173               final Object newFieldValue = targetType != null ? castValueToType(field.schema(), origFieldValue, targetType) : origFieldValue;
174               log.trace("Cast field '{}' from '{}' to '{}'", field.name(), origFieldValue, newFieldValue);
175               updatedValue.put(updatedSchema.field(field.name()), newFieldValue);
176           }
177           return newRecord(record, updatedSchema, updatedValue);
178       }

180       private Schema getOrBuildSchema(Schema valueSchema) {
181           Schema updatedSchema = schemaUpdateCache.get(valueSchema);
182           if (updatedSchema != null)
183               return updatedSchema;
184   
185           final SchemaBuilder builder;
186           if (wholeValueCastType != null) {
187               builder = SchemaUtil.copySchemaBasics(valueSchema, convertFieldType(wholeValueCastType));
188           } else {
189               builder = SchemaUtil.copySchemaBasics(valueSchema, SchemaBuilder.struct());
190               for (Field field : valueSchema.fields()) {
191                   if (casts.containsKey(field.name())) {
192                       SchemaBuilder fieldBuilder = convertFieldType(casts.get(field.name()));
193                       if (field.schema().isOptional())
194                           fieldBuilder.optional();
195                       if (field.schema().defaultValue() != null) {
196                           Schema fieldSchema = field.schema();
197                           fieldBuilder.defaultValue(castValueToType(fieldSchema, fieldSchema.defaultValue(), fieldBuilder.type()));
198                       }
199                       builder.field(field.name(), fieldBuilder.build());
200                   } else {
201                       builder.field(field.name(), field.schema());
202                   }
203               }
204           }
205   
206           if (valueSchema.isOptional())
207               builder.optional();
208           if (valueSchema.defaultValue() != null)
209               builder.defaultValue(castValueToType(valueSchema, valueSchema.defaultValue(), builder.type()));
210   
211           updatedSchema = builder.build();
212           schemaUpdateCache.put(valueSchema, updatedSchema);
213           return updatedSchema;
214       }

216       private SchemaBuilder convertFieldType(Schema.Type type) {
217           switch (type) {
218               case INT8:
219                   return SchemaBuilder.int8();
220               case INT16:
221                   return SchemaBuilder.int16();
222               case INT32:
223                   return SchemaBuilder.int32();
224               case INT64:
225                   return SchemaBuilder.int64();
226               case FLOAT32:
227                   return SchemaBuilder.float32();
228               case FLOAT64:
229                   return SchemaBuilder.float64();
230               case BOOLEAN:
231                   return SchemaBuilder.bool();
232               case STRING:
233                   return SchemaBuilder.string();
234               default:
235                   throw new DataException("Unexpected type in Cast transformation: " + type);
236           }
237       }

239       private static Object encodeLogicalType(Schema schema, Object value) {
240           switch (schema.name()) {
241               case Date.LOGICAL_NAME:
242                   return Date.fromLogical(schema, (java.util.Date) value);
243               case Time.LOGICAL_NAME:
244                   return Time.fromLogical(schema, (java.util.Date) value);
245               case Timestamp.LOGICAL_NAME:
246                   return Timestamp.fromLogical(schema, (java.util.Date) value);
247           }
248           return value;
249       }

251       private static Object castValueToType(Schema schema, Object value, Schema.Type targetType) {
252           try {
253               if (value == null) return null;
254   
255               Schema.Type inferredType = schema == null ? ConnectSchema.schemaType(value.getClass()) :
256                       schema.type();
257               if (inferredType == null) {
258                   throw new DataException("Cast transformation was passed a value of type " + value.getClass()
259                           + " which is not supported by Connect's data API");
260               }
261               // Ensure the type we are trying to cast from is supported
262               validCastType(inferredType, FieldType.INPUT);
263   
264               // Perform logical type encoding to their internal representation.
265               if (schema != null && schema.name() != null && targetType != Type.STRING) {
266                   value = encodeLogicalType(schema, value);
267               }
268   
269               switch (targetType) {
270                   case INT8:
271                       return castToInt8(value);
272                   case INT16:
273                       return castToInt16(value);
274                   case INT32:
275                       return castToInt32(value);
276                   case INT64:
277                       return castToInt64(value);
278                   case FLOAT32:
279                       return castToFloat32(value);
280                   case FLOAT64:
281                       return castToFloat64(value);
282                   case BOOLEAN:
283                       return castToBoolean(value);
284                   case STRING:
285                       return castToString(value);
286                   default:
287                       throw new DataException(targetType.toString() + " is not supported in the Cast transformation.");
288               }
289           } catch (NumberFormatException e) {
290               throw new DataException("Value (" + value.toString() + ") was out of range for requested data type", e);
291           }
292       }

371       private static String castToString(Object value) {
372           if (value instanceof java.util.Date) {
373               java.util.Date dateValue = (java.util.Date) value;
374               return Values.dateFormatFor(dateValue).format(dateValue);
375           } else if (value instanceof ByteBuffer) {
376               ByteBuffer byteBuffer = (ByteBuffer) value;
377               return Base64.getEncoder().encodeToString(Utils.readBytes(byteBuffer));
378           } else if (value instanceof byte[]) {
379               byte[] rawBytes = (byte[]) value;
380               return Base64.getEncoder().encodeToString(rawBytes);
381           } else {
382               return value.toString();
383           }

392       private static Map<String, Schema.Type> parseFieldTypes(List<String> mappings) {
393           final Map<String, Schema.Type> m = new HashMap<>();
394           boolean isWholeValueCast = false;
395           for (String mapping : mappings) {
396               final String[] parts = mapping.split(":");
397               if (parts.length > 2) {
398                   throw new ConfigException(ReplaceField.ConfigName.RENAME, mappings, "Invalid rename mapping: " + mapping);
399               }
400               if (parts.length == 1) {
401                   Schema.Type targetType = Schema.Type.valueOf(parts[0].trim().toUpperCase(Locale.ROOT));
402                   m.put(WHOLE_VALUE_CAST, validCastType(targetType, FieldType.OUTPUT));
403                   isWholeValueCast = true;
404               } else {
405                   Schema.Type type;
406                   try {
407                       type = Schema.Type.valueOf(parts[1].trim().toUpperCase(Locale.ROOT));
408                   } catch (IllegalArgumentException e) {
409                       throw new ConfigException("Invalid type found in casting spec: " + parts[1].trim(), e);
410                   }
411                   m.put(parts[0].trim(), validCastType(type, FieldType.OUTPUT));
412               }
413           }
414           if (isWholeValueCast && mappings.size() > 1) {
415               throw new ConfigException("Cast transformations that specify a type to cast the entire value to "
416                       + "may ony specify a single cast in their spec");
417           }
418           return m;
419       }

421       private enum FieldType {
422           INPUT, OUTPUT
423       }
424   
425       private static Schema.Type validCastType(Schema.Type type, FieldType fieldType) {
426           switch (fieldType) {
427               case INPUT:
428                   if (!SUPPORTED_CAST_INPUT_TYPES.contains(type)) {
429                       throw new DataException("Cast transformation does not support casting from " +
430                           type + "; supported types are " + SUPPORTED_CAST_INPUT_TYPES);
431                   }
432                   break;
433               case OUTPUT:
434                   if (!SUPPORTED_CAST_OUTPUT_TYPES.contains(type)) {
435                       throw new ConfigException("Cast transformation does not support casting to " +
436                           type + "; supported types are " + SUPPORTED_CAST_OUTPUT_TYPES);
437                   }
438                   break;
439           }
440           return type;
441       }
