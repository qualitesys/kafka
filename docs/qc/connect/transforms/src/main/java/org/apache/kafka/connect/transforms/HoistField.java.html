17   package org.apache.kafka.connect.transforms;
18   
19   import org.apache.kafka.common.cache.Cache;
20   import org.apache.kafka.common.cache.LRUCache;
21   import org.apache.kafka.common.cache.SynchronizedCache;
22   import org.apache.kafka.common.config.ConfigDef;
23   import org.apache.kafka.connect.connector.ConnectRecord;
24   import org.apache.kafka.connect.data.Schema;
25   import org.apache.kafka.connect.data.SchemaBuilder;
26   import org.apache.kafka.connect.data.Struct;
27   import org.apache.kafka.connect.transforms.util.SimpleConfig;
28   
29   import java.util.Collections;
30   import java.util.Map;
31   
32   public abstract class HoistField<R extends ConnectRecord<R>> implements Transformation<R> {
33   
34       public static final String OVERVIEW_DOC =
35               "Wrap data using the specified field name in a Struct when schema present, or a Map in the case of schemaless data."
36                       + "<p/>Use the concrete transformation type designed for the record key (<code>" + Key.class.getName() + "</code>) "
37                       + "or value (<code>" + Value.class.getName() + "</code>).";
38   
39       private static final String FIELD_CONFIG = "field";
40   
41       public static final ConfigDef CONFIG_DEF = new ConfigDef()
42               .define(FIELD_CONFIG, ConfigDef.Type.STRING, ConfigDef.NO_DEFAULT_VALUE, ConfigDef.Importance.MEDIUM,
43                       "Field name for the single field that will be created in the resulting Struct or Map.");
44   
45       private Cache<Schema, Schema> schemaUpdateCache;
46   
47       private String fieldName;
48   
49       @Override
50       public void configure(Map<String, ?> props) {
51           final SimpleConfig config = new SimpleConfig(CONFIG_DEF, props);
52           fieldName = config.getString("field");
53           schemaUpdateCache = new SynchronizedCache<>(new LRUCache<>(16));
54       }
55   
56       @Override
57       public R apply(R record) {
58           final Schema schema = operatingSchema(record);
59           final Object value = operatingValue(record);
60   
61           if (schema == null) {
62               return newRecord(record, null, Collections.singletonMap(fieldName, value));
63           } else {
64               Schema updatedSchema = schemaUpdateCache.get(schema);
65               if (updatedSchema == null) {
66                   updatedSchema = SchemaBuilder.struct().field(fieldName, schema).build();
67                   schemaUpdateCache.put(schema, updatedSchema);
68               }
69   
70               final Struct updatedValue = new Struct(updatedSchema).put(fieldName, value);
71   
72               return newRecord(record, updatedSchema, updatedValue);
73           }
74       }
75   
76       @Override
77       public void close() {
78           schemaUpdateCache = null;
79       }
80   
81       @Override
82       public ConfigDef config() {
83           return CONFIG_DEF;
84       }
85   
86       protected abstract Schema operatingSchema(R record);
87   
88       protected abstract Object operatingValue(R record);
89   
90       protected abstract R newRecord(R record, Schema updatedSchema, Object updatedValue);
91   
92       public static class Key<R extends ConnectRecord<R>> extends HoistField<R> {
93           @Override
94           protected Schema operatingSchema(R record) {
95               return record.keySchema();
96           }
97   
98           @Override
99           protected Object operatingValue(R record) {
100               return record.key();
101           }
102   
103           @Override
104           protected R newRecord(R record, Schema updatedSchema, Object updatedValue) {
105               return record.newRecord(record.topic(), record.kafkaPartition(), updatedSchema, updatedValue, record.valueSchema(), record.value(), record.timestamp());
106           }
107       }
108   
109       public static class Value<R extends ConnectRecord<R>> extends HoistField<R> {
110           @Override
111           protected Schema operatingSchema(R record) {
112               return record.valueSchema();
113           }
114   
115           @Override
116           protected Object operatingValue(R record) {
117               return record.value();
118           }
119   
120           @Override
121           protected R newRecord(R record, Schema updatedSchema, Object updatedValue) {
122               return record.newRecord(record.topic(), record.kafkaPartition(), record.keySchema(), record.key(), updatedSchema, updatedValue, record.timestamp());
123           }
124       }
125   
126   }
