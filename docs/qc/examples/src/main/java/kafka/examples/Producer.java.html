17   package kafka.examples;
18   
19   import org.apache.kafka.clients.producer.Callback;
20   import org.apache.kafka.clients.producer.KafkaProducer;
21   import org.apache.kafka.clients.producer.ProducerRecord;
22   import org.apache.kafka.clients.producer.RecordMetadata;
23   import org.apache.kafka.clients.producer.ProducerConfig;
24   import org.apache.kafka.common.serialization.IntegerSerializer;
25   import org.apache.kafka.common.serialization.StringSerializer;
26   
27   import java.util.Properties;
28   import java.util.concurrent.CountDownLatch;
29   import java.util.concurrent.ExecutionException;
30   
31   public class Producer extends Thread {
32       private final KafkaProducer<Integer, String> producer;
33       private final String topic;
34       private final Boolean isAsync;
35       private int numRecords;
36       private final CountDownLatch latch;
37   
38       public Producer(final String topic,
39                       final Boolean isAsync,
40                       final String transactionalId,
41                       final boolean enableIdempotency,
42                       final int numRecords,
43                       final int transactionTimeoutMs,
44                       final CountDownLatch latch) {
45           Properties props = new Properties();
46           props.put(ProducerConfig.BOOTSTRAP_SERVERS_CONFIG, KafkaProperties.KAFKA_SERVER_URL + ":" + KafkaProperties.KAFKA_SERVER_PORT);
47           props.put(ProducerConfig.CLIENT_ID_CONFIG, "DemoProducer");
48           props.put(ProducerConfig.KEY_SERIALIZER_CLASS_CONFIG, IntegerSerializer.class.getName());
49           props.put(ProducerConfig.VALUE_SERIALIZER_CLASS_CONFIG, StringSerializer.class.getName());
50           if (transactionTimeoutMs > 0) {
51               props.put(ProducerConfig.TRANSACTION_TIMEOUT_CONFIG, transactionTimeoutMs);
52           }
53           if (transactionalId != null) {
54               props.put(ProducerConfig.TRANSACTIONAL_ID_CONFIG, transactionalId);
55           }
56           props.put(ProducerConfig.ENABLE_IDEMPOTENCE_CONFIG, enableIdempotency);
57   
58           producer = new KafkaProducer<>(props);
59           this.topic = topic;
60           this.isAsync = isAsync;
61           this.numRecords = numRecords;
62           this.latch = latch;
63       }
64   
65       KafkaProducer<Integer, String> get() {
66           return producer;
67       }
68   
69       @Override
70       public void run() {
71           int messageKey = 0;
72           int recordsSent = 0;
73           while (recordsSent < numRecords) {
74               String messageStr = "Message_" + messageKey;
75               long startTime = System.currentTimeMillis();
76               if (isAsync) { // Send asynchronously
77                   producer.send(new ProducerRecord<>(topic,
78                       messageKey,
79                       messageStr), new DemoCallBack(startTime, messageKey, messageStr));
80               } else { // Send synchronously
81                   try {
82                       producer.send(new ProducerRecord<>(topic,
83                           messageKey,
84                           messageStr)).get();
85                       System.out.println("Sent message: (" + messageKey + ", " + messageStr + ")");
86                   } catch (InterruptedException | ExecutionException e) {
87                       e.printStackTrace();
88                   }
89               }
90               messageKey += 2;
91               recordsSent += 1;
92           }
93           System.out.println("Producer sent " + numRecords + " records successfully");
94           latch.countDown();
95       }
96   }
97   
98   class DemoCallBack implements Callback {
99   
100       private final long startTime;
101       private final int key;
102       private final String message;
103   
104       public DemoCallBack(long startTime, int key, String message) {
105           this.startTime = startTime;
106           this.key = key;
107           this.message = message;
108       }
109   
110       /**
111        * A callback method the user can implement to provide asynchronous handling of request completion. This method will
112        * be called when the record sent to the server has been acknowledged. When exception is not null in the callback,
113        * metadata will contain the special -1 value for all fields except for topicPartition, which will be valid.
114        *
115        * @param metadata  The metadata for the record that was sent (i.e. the partition and offset). An empty metadata
116        *                  with -1 value for all fields except for topicPartition will be returned if an error occurred.
117        * @param exception The exception thrown during processing of this record. Null if no error occurred.
118        */
119       public void onCompletion(RecordMetadata metadata, Exception exception) {
120           long elapsedTime = System.currentTimeMillis() - startTime;
121           if (metadata != null) {
122               System.out.println(
123                   "message(" + key + ", " + message + ") sent to partition(" + metadata.partition() +
124                       "), " +
125                       "offset(" + metadata.offset() + ") in " + elapsedTime + " ms");
126           } else {
127               exception.printStackTrace();
128           }
129       }
130   }
