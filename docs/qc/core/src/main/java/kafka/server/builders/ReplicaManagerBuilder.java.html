148       public ReplicaManager build() {
149           if (config == null) config = new KafkaConfig(Collections.emptyMap());
150           if (metrics == null) metrics = new Metrics();
151           if (logManager == null) throw new RuntimeException("You must set logManager");
152           if (metadataCache == null) throw new RuntimeException("You must set metadataCache");
153           if (logDirFailureChannel == null) throw new RuntimeException("You must set logDirFailureChannel");
154           if (alterPartitionManager == null) throw new RuntimeException("You must set alterIsrManager");
155           return new ReplicaManager(config,
156                                metrics,
157                                time,
158                                scheduler,
159                                logManager,
160                                quotaManagers,
161                                metadataCache,
162                                logDirFailureChannel,
163                                alterPartitionManager,
164                                brokerTopicStats,
165                                isShuttingDown,
166                                OptionConverters.toScala(zkClient),
167                                OptionConverters.toScala(delayedProducePurgatory),
168                                OptionConverters.toScala(delayedFetchPurgatory),
169                                OptionConverters.toScala(delayedDeleteRecordsPurgatory),
170                                OptionConverters.toScala(delayedElectLeaderPurgatory),
171                                OptionConverters.toScala(threadNamePrefix));
172       }
