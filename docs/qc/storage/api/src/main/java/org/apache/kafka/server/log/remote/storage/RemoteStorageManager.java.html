17   package org.apache.kafka.server.log.remote.storage;
18   
19   import org.apache.kafka.common.Configurable;
20   import org.apache.kafka.common.annotation.InterfaceStability;
21   
22   import java.io.Closeable;
23   import java.io.InputStream;
24   
25   /**
26    * This interface provides the lifecycle of remote log segments that includes copy, fetch, and delete from remote
27    * storage.
28    * <p>
29    * Each upload or copy of a segment is initiated with {@link RemoteLogSegmentMetadata} containing {@link RemoteLogSegmentId}
30    * which is universally unique even for the same topic partition and offsets.
31    * <p>
32    * {@link RemoteLogSegmentMetadata} is stored in {@link RemoteLogMetadataManager} before and after copy/delete operations on
33    * {@link RemoteStorageManager} with the respective {@link RemoteLogSegmentState}. {@link RemoteLogMetadataManager} is
34    * responsible for storing and fetching metadata about the remote log segments in a strongly consistent manner.
35    * This allows {@link RemoteStorageManager} to have eventual consistency on metadata (although the data is stored
36    * in strongly consistent semantics).
37    */
38   @InterfaceStability.Evolving
39   public interface RemoteStorageManager extends Configurable, Closeable {
40   
41       /**
42        * Type of the index file.
43        */
44       enum IndexType {
45           /**
46            * Represents offset index.
47            */
48           OFFSET,
49   
50           /**
51            * Represents timestamp index.
52            */
53           TIMESTAMP,
54   
55           /**
56            * Represents producer snapshot index.
57            */
58           PRODUCER_SNAPSHOT,
59   
60           /**
61            * Represents transaction index.
62            */
63           TRANSACTION,
64   
65           /**
66            * Represents leader epoch index.
67            */
68           LEADER_EPOCH,
69       }
70   
71       /**
72        * Copies the given {@link LogSegmentData} provided for the given {@code remoteLogSegmentMetadata}. This includes
73        * log segment and its auxiliary indexes like offset index, time index, transaction index, leader epoch index, and
74        * producer snapshot index.
75        * <p>
76        * Invoker of this API should always send a unique id as part of {@link RemoteLogSegmentMetadata#remoteLogSegmentId()}
77        * even when it retries to invoke this method for the same log segment data.
78        *
79        * @param remoteLogSegmentMetadata metadata about the remote log segment.
80        * @param logSegmentData           data to be copied to tiered storage.
81        * @throws RemoteStorageException if there are any errors in storing the data of the segment.
82        */
83       void copyLogSegmentData(RemoteLogSegmentMetadata remoteLogSegmentMetadata,
84                               LogSegmentData logSegmentData)
85               throws RemoteStorageException;
86   
87       /**
88        * Returns the remote log segment data file/object as InputStream for the given {@link RemoteLogSegmentMetadata}
89        * starting from the given startPosition. The stream will end at the end of the remote log segment data file/object.
90        *
91        * @param remoteLogSegmentMetadata metadata about the remote log segment.
92        * @param startPosition            start position of log segment to be read, inclusive.
93        * @return input stream of the requested log segment data.
94        * @throws RemoteStorageException          if there are any errors while fetching the desired segment.
95        * @throws RemoteResourceNotFoundException when there are no resources associated with the given remoteLogSegmentMetadata.
96        */
97       InputStream fetchLogSegment(RemoteLogSegmentMetadata remoteLogSegmentMetadata,
98                                   int startPosition) throws RemoteStorageException;
99   
100       /**
101        * Returns the remote log segment data file/object as InputStream for the given {@link RemoteLogSegmentMetadata}
102        * starting from the given startPosition. The stream will end at the smaller of endPosition and the end of the
103        * remote log segment data file/object.
104        *
105        * @param remoteLogSegmentMetadata metadata about the remote log segment.
106        * @param startPosition            start position of log segment to be read, inclusive.
107        * @param endPosition              end position of log segment to be read, inclusive.
108        * @return input stream of the requested log segment data.
109        * @throws RemoteStorageException          if there are any errors while fetching the desired segment.
110        * @throws RemoteResourceNotFoundException when there are no resources associated with the given remoteLogSegmentMetadata.
111        */
112       InputStream fetchLogSegment(RemoteLogSegmentMetadata remoteLogSegmentMetadata,
113                                   int startPosition,
114                                   int endPosition) throws RemoteStorageException;
115   
116       /**
117        * Returns the index for the respective log segment of {@link RemoteLogSegmentMetadata}.
118        *
119        * @param remoteLogSegmentMetadata metadata about the remote log segment.
120        * @param indexType                type of the index to be fetched for the segment.
121        * @return input stream of the requested index.
122        * @throws RemoteStorageException          if there are any errors while fetching the index.
123        * @throws RemoteResourceNotFoundException when there are no resources associated with the given remoteLogSegmentMetadata.
124        */
125       InputStream fetchIndex(RemoteLogSegmentMetadata remoteLogSegmentMetadata,
126                              IndexType indexType) throws RemoteStorageException;
127   
128       /**
129        * Deletes the resources associated with the given {@code remoteLogSegmentMetadata}. Deletion is considered as
130        * successful if this call returns successfully without any errors. It will throw {@link RemoteStorageException} if
131        * there are any errors in deleting the file.
132        * <p>
133        *
134        * @param remoteLogSegmentMetadata metadata about the remote log segment to be deleted.
135        * @throws RemoteResourceNotFoundException if the requested resource is not found
136        * @throws RemoteStorageException          if there are any storage related errors occurred.
137        * @throws RemoteResourceNotFoundException when there are no resources associated with the given remoteLogSegmentMetadata.
138        */
139       void deleteLogSegmentData(RemoteLogSegmentMetadata remoteLogSegmentMetadata) throws RemoteStorageException;
140   }
