30   /**
31    * This interface provides storing and fetching remote log segment metadata with strongly consistent semantics.
32    * <p>
33    * This class can be plugged in to Kafka cluster by adding the implementation class as
34    * <code>remote.log.metadata.manager.class.name</code> property value. There is an inbuilt implementation backed by
35    * topic storage in the local cluster. This is used as the default implementation if
36    * remote.log.metadata.manager.class.name is not configured.
37    * </p>
38    * <p>
39    * <code>remote.log.metadata.manager.class.path</code> property is about the class path of the RemoteLogStorageManager
40    * implementation. If specified, the RemoteLogStorageManager implementation and its dependent libraries will be loaded
41    * by a dedicated classloader which searches this class path before the Kafka broker class path. The syntax of this
42    * parameter is same with the standard Java class path string.
43    * </p>
44    * <p>
45    * <code>remote.log.metadata.manager.listener.name</code> property is about listener name of the local broker to which
46    * it should get connected if needed by RemoteLogMetadataManager implementation. When this is configured all other
47    * required properties can be passed as properties with prefix of 'remote.log.metadata.manager.listener.
48    * </p>
49    * "cluster.id", "broker.id" and all other properties prefixed with "remote.log.metadata." are passed when
50    * {@link #configure(Map)} is invoked on this instance.
51    * <p>
52    */
53   @InterfaceStability.Evolving
54   public interface RemoteLogMetadataManager extends Configurable, Closeable {
55   
56       /**
57        * This method is used to add {@link RemoteLogSegmentMetadata} asynchronously with the containing {@link RemoteLogSegmentId} into {@link RemoteLogMetadataManager}.
58        * <p>
59        * RemoteLogSegmentMetadata is identified by RemoteLogSegmentId and it should have the initial state which is {@link RemoteLogSegmentState#COPY_SEGMENT_STARTED}.
60        * <p>
61        * {@link #updateRemoteLogSegmentMetadata(RemoteLogSegmentMetadataUpdate)} should be used to update an existing RemoteLogSegmentMetadata.
62        *
63        * @param remoteLogSegmentMetadata metadata about the remote log segment.
64        * @throws RemoteStorageException   if there are any storage related errors occurred.
65        * @throws IllegalArgumentException if the given metadata instance does not have the state as {@link RemoteLogSegmentState#COPY_SEGMENT_STARTED}
66        * @return a CompletableFuture which will complete once this operation is finished.
67        */
68       CompletableFuture<Void> addRemoteLogSegmentMetadata(RemoteLogSegmentMetadata remoteLogSegmentMetadata) throws RemoteStorageException;
69   
70       /**
71        * This method is used to update the {@link RemoteLogSegmentMetadata} asynchronously. Currently, it allows to update with the new
72        * state based on the life cycle of the segment. It can go through the below state transitions.
73        * <p>
74        * <pre>
75        * +---------------------+            +----------------------+
76        * |COPY_SEGMENT_STARTED |-----------&gt;|COPY_SEGMENT_FINISHED |
77        * +-------------------+-+            +--+-------------------+
78        *                     |                 |
79        *                     |                 |
80        *                     v                 v
81        *                  +--+-----------------+-+
82        *                  |DELETE_SEGMENT_STARTED|
83        *                  +-----------+----------+
84        *                              |
85        *                              |
86        *                              v
87        *                  +-----------+-----------+
88        *                  |DELETE_SEGMENT_FINISHED|
89        *                  +-----------------------+
90        * </pre>
91        * <p>
92        * {@link RemoteLogSegmentState#COPY_SEGMENT_STARTED} - This state indicates that the segment copying to remote storage is started but not yet finished.
93        * {@link RemoteLogSegmentState#COPY_SEGMENT_FINISHED} - This state indicates that the segment copying to remote storage is finished.
94        * <br>
95        * The leader broker copies the log segments to the remote storage and puts the remote log segment metadata with the
96        * state as “COPY_SEGMENT_STARTED” and updates the state as “COPY_SEGMENT_FINISHED” once the copy is successful.
97        * <p></p>
98        * {@link RemoteLogSegmentState#DELETE_SEGMENT_STARTED} - This state indicates that the segment deletion is started but not yet finished.
99        * {@link RemoteLogSegmentState#DELETE_SEGMENT_FINISHED} - This state indicates that the segment is deleted successfully.
100        * <br>
101        * Leader partitions publish both the above delete segment events when remote log retention is reached for the
102        * respective segments. Remote Partition Removers also publish these events when a segment is deleted as part of
103        * the remote partition deletion.
104        *
105        * @param remoteLogSegmentMetadataUpdate update of the remote log segment metadata.
106        * @throws RemoteStorageException          if there are any storage related errors occurred.
107        * @throws RemoteResourceNotFoundException when there are no resources associated with the given remoteLogSegmentMetadataUpdate.
108        * @throws IllegalArgumentException        if the given metadata instance has the state as {@link RemoteLogSegmentState#COPY_SEGMENT_STARTED}
109        * @return a CompletableFuture which will complete once this operation is finished.
110        */
111       CompletableFuture<Void> updateRemoteLogSegmentMetadata(RemoteLogSegmentMetadataUpdate remoteLogSegmentMetadataUpdate)
112               throws RemoteStorageException;
113   
114       /**
115        * Returns {@link RemoteLogSegmentMetadata} if it exists for the given topic partition containing the offset with
116        * the given leader-epoch for the offset, else returns {@link Optional#empty()}.
117        *
118        * @param topicIdPartition topic partition
119        * @param epochForOffset   leader epoch for the given offset
120        * @param offset           offset
121        * @return the requested remote log segment metadata if it exists.
122        * @throws RemoteStorageException if there are any storage related errors occurred.
123        */
124       Optional<RemoteLogSegmentMetadata> remoteLogSegmentMetadata(TopicIdPartition topicIdPartition,
125                                                                   int epochForOffset,
126                                                                   long offset)
127               throws RemoteStorageException;
128   
129       /**
130        * Returns the highest log offset of topic partition for the given leader epoch in remote storage. This is used by
131        * remote log management subsystem to know up to which offset the segments have been copied to remote storage for
132        * a given leader epoch.
133        *
134        * @param topicIdPartition topic partition
135        * @param leaderEpoch      leader epoch
136        * @return the requested highest log offset if exists.
137        * @throws RemoteStorageException if there are any storage related errors occurred.
138        */
139       Optional<Long> highestOffsetForEpoch(TopicIdPartition topicIdPartition,
140                                            int leaderEpoch) throws RemoteStorageException;
141   
142       /**
143        * This method is used to update the metadata about remote partition delete event asynchronously. Currently, it allows updating the
144        * state ({@link RemotePartitionDeleteState}) of a topic partition in remote metadata storage. Controller invokes
145        * this method with {@link RemotePartitionDeleteMetadata} having state as {@link RemotePartitionDeleteState#DELETE_PARTITION_MARKED}.
146        * So, remote partition removers can act on this event to clean the respective remote log segments of the partition.
147        * <p><br>
148        * In the case of default RLMM implementation, remote partition remover processes {@link RemotePartitionDeleteState#DELETE_PARTITION_MARKED}
149        * <ul>
150        * <li> sends an event with state as {@link RemotePartitionDeleteState#DELETE_PARTITION_STARTED}
151        * <li> gets all the remote log segments and deletes them.
152        * <li> sends an event with state as {@link RemotePartitionDeleteState#DELETE_PARTITION_FINISHED} once all the remote log segments are
153        * deleted.
154        * </ul>
155        *
156        * @param remotePartitionDeleteMetadata update on delete state of a partition.
157        * @throws RemoteStorageException          if there are any storage related errors occurred.
158        * @throws RemoteResourceNotFoundException when there are no resources associated with the given remotePartitionDeleteMetadata.
159        * @return a CompletableFuture which will complete once this operation is finished.
160        */
161       CompletableFuture<Void> putRemotePartitionDeleteMetadata(RemotePartitionDeleteMetadata remotePartitionDeleteMetadata)
162               throws RemoteStorageException;
163   
164       /**
165        * Returns all the remote log segment metadata of the given topicIdPartition.
166        * <p>
167        * Remote Partition Removers uses this method to fetch all the segments for a given topic partition, so that they
168        * can delete them.
169        *
170        * @return Iterator of all the remote log segment metadata for the given topic partition.
171        */
172       Iterator<RemoteLogSegmentMetadata> listRemoteLogSegments(TopicIdPartition topicIdPartition)
173               throws RemoteStorageException;
174   
175       /**
176        * Returns iterator of remote log segment metadata, sorted by {@link RemoteLogSegmentMetadata#startOffset()} in
177        * ascending order which contains the given leader epoch. This is used by remote log retention management subsystem
178        * to fetch the segment metadata for a given leader epoch.
179        *
180        * @param topicIdPartition topic partition
181        * @param leaderEpoch      leader epoch
182        * @return Iterator of remote segments, sorted by start offset in ascending order.
183        */
184       Iterator<RemoteLogSegmentMetadata> listRemoteLogSegments(TopicIdPartition topicIdPartition,
185                                                                int leaderEpoch) throws RemoteStorageException;
186   
187       /**
188        * This method is invoked only when there are changes in leadership of the topic partitions that this broker is
189        * responsible for.
190        *
191        * @param leaderPartitions   partitions that have become leaders on this broker.
192        * @param followerPartitions partitions that have become followers on this broker.
193        */
194       void onPartitionLeadershipChanges(Set<TopicIdPartition> leaderPartitions,
195                                         Set<TopicIdPartition> followerPartitions);
196   
197       /**
198        * This method is invoked only when the topic partitions are stopped on this broker. This can happen when a
199        * partition is emigrated to other broker or a partition is deleted.
200        *
201        * @param partitions topic partitions that have been stopped.
202        */
203       void onStopPartitions(Set<TopicIdPartition> partitions);
204   }
