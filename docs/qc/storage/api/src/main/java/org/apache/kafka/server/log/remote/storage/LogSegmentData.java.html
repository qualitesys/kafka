17   package org.apache.kafka.server.log.remote.storage;
18   
19   import org.apache.kafka.common.annotation.InterfaceStability;
20   
21   import java.nio.ByteBuffer;
22   import java.nio.file.Path;
23   import java.util.Objects;
24   import java.util.Optional;
25   
26   /**
27    * This represents all the required data and indexes for a specific log segment that needs to be stored in the remote
28    * storage. This is passed with {@link RemoteStorageManager#copyLogSegmentData(RemoteLogSegmentMetadata, LogSegmentData)}
29    * while copying a specific log segment to the remote storage.
30    */
31   @InterfaceStability.Evolving
32   public class LogSegmentData {
33   
34       private final Path logSegment;
35       private final Path offsetIndex;
36       private final Path timeIndex;
37       private final Optional<Path> transactionIndex;
38       private final Path producerSnapshotIndex;
39       private final ByteBuffer leaderEpochIndex;
40   
41       /**
42        * Creates a LogSegmentData instance with data and indexes.
43        *
44        * @param logSegment            actual log segment file
45        * @param offsetIndex           offset index file
46        * @param timeIndex             time index file
47        * @param transactionIndex      transaction index file, which can be null
48        * @param producerSnapshotIndex producer snapshot until this segment
49        * @param leaderEpochIndex      leader-epoch-index until this segment
50        */
51       public LogSegmentData(Path logSegment,
52                             Path offsetIndex,
53                             Path timeIndex,
54                             Optional<Path> transactionIndex,
55                             Path producerSnapshotIndex,
56                             ByteBuffer leaderEpochIndex) {
57           this.logSegment = Objects.requireNonNull(logSegment, "logSegment can not be null");
58           this.offsetIndex = Objects.requireNonNull(offsetIndex, "offsetIndex can not be null");
59           this.timeIndex = Objects.requireNonNull(timeIndex, "timeIndex can not be null");
60           this.transactionIndex = Objects.requireNonNull(transactionIndex, "transactionIndex can not be null");
61           this.producerSnapshotIndex = Objects.requireNonNull(producerSnapshotIndex, "producerSnapshotIndex can not be null");
62           this.leaderEpochIndex = Objects.requireNonNull(leaderEpochIndex, "leaderEpochIndex can not be null");
63       }
64   
65       /**
66        * @return Log segment file of this segment.
67        */
68       public Path logSegment() {
69           return logSegment;
70       }
71   
72       /**
73        * @return Offset index file.
74        */
75       public Path offsetIndex() {
76           return offsetIndex;
77       }
78   
79       /**
80        * @return Time index file of this segment.
81        */
82       public Path timeIndex() {
83           return timeIndex;
84       }
85   
86       /**
87        * @return Transaction index file of this segment if it exists.
88        */
89       public Optional<Path> transactionIndex() {
90           return transactionIndex;
91       }
92   
93       /**
94        * @return Producer snapshot file until this segment.
95        */
96       public Path producerSnapshotIndex() {
97           return producerSnapshotIndex;
98       }
99   
100       /**
101        * @return Leader epoch index until this segment.
102        */
103       public ByteBuffer leaderEpochIndex() {
104           return leaderEpochIndex;
105       }
106   
107       @Override
108       public boolean equals(Object o) {
109           if (this == o) {
110               return true;
111           }
112           if (o == null || getClass() != o.getClass()) {
113               return false;
114           }
115           LogSegmentData that = (LogSegmentData) o;
116           return Objects.equals(logSegment, that.logSegment) &&
117                  Objects.equals(offsetIndex, that.offsetIndex) &&
118                  Objects.equals(timeIndex, that.timeIndex) &&
119                  Objects.equals(transactionIndex, that.transactionIndex) &&
120                  Objects.equals(producerSnapshotIndex, that.producerSnapshotIndex) &&
121                  Objects.equals(leaderEpochIndex, that.leaderEpochIndex);
122       }
123   
124       @Override
125       public int hashCode() {
126           return Objects.hash(logSegment, offsetIndex, timeIndex, transactionIndex, producerSnapshotIndex, leaderEpochIndex);
127       }
128   
129       @Override
130       public String toString() {
131           return "LogSegmentData{" +
132                  "logSegment=" + logSegment +
133                  ", offsetIndex=" + offsetIndex +
134                  ", timeIndex=" + timeIndex +
135                  ", txnIndex=" + transactionIndex +
136                  ", producerSnapshotIndex=" + producerSnapshotIndex +
137                  ", leaderEpochIndex=" + leaderEpochIndex +
138                  '}';
139       }
140   }
