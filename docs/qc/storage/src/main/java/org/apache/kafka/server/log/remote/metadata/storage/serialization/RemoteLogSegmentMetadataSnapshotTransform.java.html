17   package org.apache.kafka.server.log.remote.metadata.storage.serialization;
18   
19   import org.apache.kafka.server.common.ApiMessageAndVersion;
20   import org.apache.kafka.server.log.remote.metadata.storage.RemoteLogSegmentMetadataSnapshot;
21   import org.apache.kafka.server.log.remote.metadata.storage.generated.RemoteLogSegmentMetadataSnapshotRecord;
22   import org.apache.kafka.server.log.remote.storage.RemoteLogSegmentState;
23   
24   import java.util.HashMap;
25   import java.util.List;
26   import java.util.Map;
27   import java.util.stream.Collectors;
28   
29   public class RemoteLogSegmentMetadataSnapshotTransform implements RemoteLogMetadataTransform<RemoteLogSegmentMetadataSnapshot> {
30   
31       public ApiMessageAndVersion toApiMessageAndVersion(RemoteLogSegmentMetadataSnapshot segmentMetadata) {
32           RemoteLogSegmentMetadataSnapshotRecord record = new RemoteLogSegmentMetadataSnapshotRecord()
33                   .setSegmentId(segmentMetadata.segmentId())
34                   .setStartOffset(segmentMetadata.startOffset())
35                   .setEndOffset(segmentMetadata.endOffset())
36                   .setBrokerId(segmentMetadata.brokerId())
37                   .setEventTimestampMs(segmentMetadata.eventTimestampMs())
38                   .setMaxTimestampMs(segmentMetadata.maxTimestampMs())
39                   .setSegmentSizeInBytes(segmentMetadata.segmentSizeInBytes())
40                   .setSegmentLeaderEpochs(createSegmentLeaderEpochsEntry(segmentMetadata.segmentLeaderEpochs()))
41                   .setRemoteLogSegmentState(segmentMetadata.state().id());
42   
43           return new ApiMessageAndVersion(record, record.highestSupportedVersion());
44       }
45   
46       private List<RemoteLogSegmentMetadataSnapshotRecord.SegmentLeaderEpochEntry> createSegmentLeaderEpochsEntry(Map<Integer, Long> leaderEpochs) {
47           return leaderEpochs.entrySet().stream()
48                              .map(entry -> new RemoteLogSegmentMetadataSnapshotRecord.SegmentLeaderEpochEntry()
49                              .setLeaderEpoch(entry.getKey())
50                              .setOffset(entry.getValue()))
51                              .collect(Collectors.toList());
52       }
53   
54       @Override
55       public RemoteLogSegmentMetadataSnapshot fromApiMessageAndVersion(ApiMessageAndVersion apiMessageAndVersion) {
56           RemoteLogSegmentMetadataSnapshotRecord record = (RemoteLogSegmentMetadataSnapshotRecord) apiMessageAndVersion.message();
57           Map<Integer, Long> segmentLeaderEpochs = new HashMap<>();
58           for (RemoteLogSegmentMetadataSnapshotRecord.SegmentLeaderEpochEntry segmentLeaderEpoch : record.segmentLeaderEpochs()) {
59               segmentLeaderEpochs.put(segmentLeaderEpoch.leaderEpoch(), segmentLeaderEpoch.offset());
60           }
61   
62           return new RemoteLogSegmentMetadataSnapshot(record.segmentId(),
63                                                       record.startOffset(),
64                                                       record.endOffset(),
65                                                       record.maxTimestampMs(),
66                                                       record.brokerId(),
67                                                       record.eventTimestampMs(),
68                                                       record.segmentSizeInBytes(),
69                                                       RemoteLogSegmentState.forId(record.remoteLogSegmentState()),
70                                                       segmentLeaderEpochs);
71       }
72   
73   }
