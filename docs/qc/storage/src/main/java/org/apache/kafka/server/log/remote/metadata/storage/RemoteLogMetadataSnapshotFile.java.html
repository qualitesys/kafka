41   /**
42    * This class represents the remote log data snapshot stored in a file for a specific topic partition. This is used by
43    * {@link TopicBasedRemoteLogMetadataManager} to store the remote log metadata received for a specific partition from
44    * remote log metadata topic. This will avoid reading the remote log metadata messages from the topic again when a
45    * broker restarts.
46    */
47   public class RemoteLogMetadataSnapshotFile {
48       private static final Logger log = LoggerFactory.getLogger(RemoteLogMetadataSnapshotFile.class);
49   
50       public static final String COMMITTED_LOG_METADATA_SNAPSHOT_FILE_NAME = "remote_log_snapshot";
51   
52       // File format:
53       // <header>[<entry>...]
54       // header: <version:short><metadata-partition:int><metadata-partition-offset:long><entries-size:int>
55       // entry: <entry-length><entry-bytes>
56   
57       // header size: 2 (version) + 4 (partition num) + 8 (offset) + 4 (entries size) = 18
58       private static final int HEADER_SIZE = 18;
59   
60       private final File metadataStoreFile;
61       private final RemoteLogMetadataSerde serde = new RemoteLogMetadataSerde();
62   
63       /**
64        * Creates a CommittedLogMetadataSnapshotFile instance backed by a file with the name `remote_log_snapshot` in
65        * the given {@code metadataStoreDir}. It creates the file if it does not exist.
66        *
67        * @param metadataStoreDir directory in which the snapshot file to be created.
68        */
69       RemoteLogMetadataSnapshotFile(Path metadataStoreDir) {
70           this.metadataStoreFile = new File(metadataStoreDir.toFile(), COMMITTED_LOG_METADATA_SNAPSHOT_FILE_NAME);
71   
72           // Create an empty file if it does not exist.
73           try {
74               final boolean fileExists = Files.exists(metadataStoreFile.toPath());
75               if (!fileExists) {
76                   Files.createFile(metadataStoreFile.toPath());
77               }
78               log.info("Remote log metadata snapshot file: [{}], newFileCreated: [{}]", metadataStoreFile, !fileExists);
79           } catch (IOException e) {
80               throw new KafkaException(e);
81           }
82       }
83   
84       /**
85        * Writes the given snapshot replacing the earlier snapshot data.
86        *
87        * @param snapshot Snapshot to be stored.
88        * @throws IOException if there4 is any error in writing the given snapshot to the file.
89        */
90       public synchronized void write(Snapshot snapshot) throws IOException {
91           Path newMetadataSnapshotFilePath = new File(metadataStoreFile.getAbsolutePath() + ".tmp").toPath();
92           try (FileChannel fileChannel = FileChannel.open(newMetadataSnapshotFilePath,
93                                                           StandardOpenOption.CREATE, StandardOpenOption.READ, StandardOpenOption.WRITE)) {
94   
95               // header: <version:short><metadata-partition:int><metadata-partition-offset:long>
96               ByteBuffer headerBuffer = ByteBuffer.allocate(HEADER_SIZE);
97   
98               // Write version
99               headerBuffer.putShort(snapshot.version());
100   
101               // Write metadata partition and metadata partition offset
102               headerBuffer.putInt(snapshot.metadataPartition());
103   
104               // Write metadata partition offset
105               headerBuffer.putLong(snapshot.metadataPartitionOffset());
106   
107               // Write entries size
108               Collection<RemoteLogSegmentMetadataSnapshot> metadataSnapshots = snapshot.remoteLogSegmentMetadataSnapshots();
109               headerBuffer.putInt(metadataSnapshots.size());
110   
111               // Write header
112               headerBuffer.flip();
113               fileChannel.write(headerBuffer);
114   
115               // Write each entry
116               ByteBuffer lenBuffer = ByteBuffer.allocate(4);
117               for (RemoteLogSegmentMetadataSnapshot metadataSnapshot : metadataSnapshots) {
118                   final byte[] serializedBytes = serde.serialize(metadataSnapshot);
119                   // entry format: <entry-length><entry-bytes>
120   
121                   // Write entry length
122                   lenBuffer.putInt(serializedBytes.length);
123                   lenBuffer.flip();
124                   fileChannel.write(lenBuffer);
125                   lenBuffer.rewind();
126   
127                   // Write entry bytes
128                   fileChannel.write(ByteBuffer.wrap(serializedBytes));
129               }
130   
131               fileChannel.force(true);
132           }
133   
134           Utils.atomicMoveWithFallback(newMetadataSnapshotFilePath, metadataStoreFile.toPath());
135       }
136   
137       /**
138        * @return the Snapshot if it exists.
139        * @throws IOException if there is any error in reading the stored snapshot.
140        */
141       public synchronized Optional<Snapshot> read() throws IOException {
142   
143           // Checking for empty files.
144           if (metadataStoreFile.length() == 0) {
145               return Optional.empty();
146           }
147   
148           try (ReadableByteChannel channel = Channels.newChannel(new FileInputStream(metadataStoreFile))) {
149   
150               // header: <version:short><metadata-partition:int><metadata-partition-offset:long>
151               // Read header
152               ByteBuffer headerBuffer = ByteBuffer.allocate(HEADER_SIZE);
153               channel.read(headerBuffer);
154               headerBuffer.rewind();
155               short version = headerBuffer.getShort();
156               int metadataPartition = headerBuffer.getInt();
157               long metadataPartitionOffset = headerBuffer.getLong();
158               int metadataSnapshotsSize = headerBuffer.getInt();
159   
160               List<RemoteLogSegmentMetadataSnapshot> result = new ArrayList<>(metadataSnapshotsSize);
161               ByteBuffer lenBuffer = ByteBuffer.allocate(4);
162               int lenBufferReadCt;
163               while ((lenBufferReadCt = channel.read(lenBuffer)) > 0) {
164                   lenBuffer.rewind();
165   
166                   if (lenBufferReadCt != lenBuffer.capacity()) {
167                       throw new IOException("Invalid amount of data read for the length of an entry, file may have been corrupted.");
168                   }
169   
170                   // entry format: <entry-length><entry-bytes>
171   
172                   // Read the length of each entry
173                   final int len = lenBuffer.getInt();
174                   lenBuffer.rewind();
175   
176                   // Read the entry
177                   ByteBuffer data = ByteBuffer.allocate(len);
178                   final int read = channel.read(data);
179                   if (read != len) {
180                       throw new IOException("Invalid amount of data read, file may have been corrupted.");
181                   }
182   
183                   // We are always adding RemoteLogSegmentMetadata only as you can see in #write() method.
184                   // Did not add a specific serde for RemoteLogSegmentMetadata and reusing RemoteLogMetadataSerde
185                   final RemoteLogSegmentMetadataSnapshot remoteLogSegmentMetadata =
186                           (RemoteLogSegmentMetadataSnapshot) serde.deserialize(data.array());
187                   result.add(remoteLogSegmentMetadata);
188               }
189   
190               if (metadataSnapshotsSize != result.size()) {
191                   throw new IOException("Unexpected entries in the snapshot file. Expected size: " + metadataSnapshotsSize
192                                                 + ", but found: " + result.size());
193               }
194   
195               return Optional.of(new Snapshot(version, metadataPartition, metadataPartitionOffset, result));
196           }
197       }
198   
199       /**
200        * This class represents the collection of remote log metadata for a specific topic partition.
201        */
202       public static final class Snapshot {
203           private static final short CURRENT_VERSION = 0;
204   
205           private final short version;
206           private final int metadataPartition;
207           private final long metadataPartitionOffset;
208           private final Collection<RemoteLogSegmentMetadataSnapshot> remoteLogSegmentMetadataSnapshots;
209   
210           public Snapshot(int metadataPartition,
211                           long metadataPartitionOffset,
212                           Collection<RemoteLogSegmentMetadataSnapshot> remoteLogSegmentMetadataSnapshots) {
213               this(CURRENT_VERSION, metadataPartition, metadataPartitionOffset, remoteLogSegmentMetadataSnapshots);
214           }
215   
216           public Snapshot(short version,
217                           int metadataPartition,
218                           long metadataPartitionOffset,
219                           Collection<RemoteLogSegmentMetadataSnapshot> remoteLogSegmentMetadataSnapshots) {
220               // We will add multiple version support in future if needed. For now, the only supported version is CURRENT_VERSION viz 0.
221               if (version != CURRENT_VERSION) {
222                   throw new IllegalArgumentException("Unexpected version received: " + version);
223               }
224               this.version = version;
225               this.metadataPartition = metadataPartition;
226               this.metadataPartitionOffset = metadataPartitionOffset;
227               this.remoteLogSegmentMetadataSnapshots = remoteLogSegmentMetadataSnapshots;
228           }
229   
230           public short version() {
231               return version;
232           }
233   
234           public int metadataPartition() {
235               return metadataPartition;
236           }
237   
238           public long metadataPartitionOffset() {
239               return metadataPartitionOffset;
240           }
241   
242           public Collection<RemoteLogSegmentMetadataSnapshot> remoteLogSegmentMetadataSnapshots() {
243               return remoteLogSegmentMetadataSnapshots;
244           }
245   
246           @Override
247           public boolean equals(Object o) {
248               if (this == o) return true;
249               if (!(o instanceof Snapshot)) return false;
250               Snapshot snapshot = (Snapshot) o;
251               return version == snapshot.version && metadataPartition == snapshot.metadataPartition
252                       && metadataPartitionOffset == snapshot.metadataPartitionOffset
253                       && Objects.equals(remoteLogSegmentMetadataSnapshots, snapshot.remoteLogSegmentMetadataSnapshots);
254           }
255   
256           @Override
257           public int hashCode() {
258               return Objects.hash(version, metadataPartition, metadataPartitionOffset, remoteLogSegmentMetadataSnapshots);
259           }
260   
261           @Override
262           public String toString() {
263               return "Snapshot{" +
264                       "version=" + version +
265                       ", metadataPartition=" + metadataPartition +
266                       ", metadataPartitionOffset=" + metadataPartitionOffset +
267                       ", remoteLogSegmentMetadataSnapshotsSize" + remoteLogSegmentMetadataSnapshots.size() +
268                       '}';
269           }
270       }
271   }
