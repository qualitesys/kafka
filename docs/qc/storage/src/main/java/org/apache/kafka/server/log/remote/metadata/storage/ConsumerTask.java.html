48   /**
49    * This class is responsible for consuming messages from remote log metadata topic ({@link TopicBasedRemoteLogMetadataManagerConfig#REMOTE_LOG_METADATA_TOPIC_NAME})
50    * partitions and maintain the state of the remote log segment metadata. It gives an API to add or remove
51    * for what topic partition's metadata should be consumed by this instance using
52    * {{@link #addAssignmentsForPartitions(Set)}} and {@link #removeAssignmentsForPartitions(Set)} respectively.
53    * <p>
54    * When a broker is started, controller sends topic partitions that this broker is leader or follower for and the
55    * partitions to be deleted. This class receives those notifications with
56    * {@link #addAssignmentsForPartitions(Set)} and {@link #removeAssignmentsForPartitions(Set)} assigns consumer for the
57    * respective remote log metadata partitions by using {@link RemoteLogMetadataTopicPartitioner#metadataPartition(TopicIdPartition)}.
58    * Any leadership changes later are called through the same API. We will remove the partitions that are deleted from
59    * this broker which are received through {@link #removeAssignmentsForPartitions(Set)}.
60    * <p>
61    * After receiving these events it invokes {@link RemotePartitionMetadataEventHandler#handleRemoteLogSegmentMetadata(RemoteLogSegmentMetadata)},
62    * which maintains in-memory representation of the state of {@link RemoteLogSegmentMetadata}.
63    */
64   class ConsumerTask implements Runnable, Closeable {
65       private static final Logger log = LoggerFactory.getLogger(ConsumerTask.class);
66   
67       private static final long POLL_INTERVAL_MS = 100L;
68   
69       private final RemoteLogMetadataSerde serde = new RemoteLogMetadataSerde();
70       private final KafkaConsumer<byte[], byte[]> consumer;
71       private final RemotePartitionMetadataEventHandler remotePartitionMetadataEventHandler;
72       private final RemoteLogMetadataTopicPartitioner topicPartitioner;
73       private final Time time;
74   
75       // It indicates whether the closing process has been started or not. If it is set as true,
76       // consumer will stop consuming messages and it will not allow partition assignments to be updated.
77       private volatile boolean closing = false;
78   
79       // It indicates whether the consumer needs to assign the partitions or not. This is set when it is
80       // determined that the consumer needs to be assigned with the updated partitions.
81       private volatile boolean assignPartitions = false;
82   
83       // It represents a lock for any operations related to the assignedTopicPartitions.
84       private final Object assignPartitionsLock = new Object();
85   
86       // Remote log metadata topic partitions that consumer is assigned to.
87       private volatile Set<Integer> assignedMetaPartitions = Collections.emptySet();
88   
89       // User topic partitions that this broker is a leader/follower for.
90       private Set<TopicIdPartition> assignedTopicPartitions = Collections.emptySet();
91   
92       // Map of remote log metadata topic partition to consumed offsets. Received consumer records
93       // may or may not have been processed based on the assigned topic partitions.
94       private final Map<Integer, Long> partitionToConsumedOffsets = new ConcurrentHashMap<>();
95   
96       // Map of remote log metadata topic partition to processed offsets that were synced in committedOffsetsFile.
97       private Map<Integer, Long> lastSyncedPartitionToConsumedOffsets = Collections.emptyMap();
98   
99       private final long committedOffsetSyncIntervalMs;
100       private CommittedOffsetsFile committedOffsetsFile;
101       private long lastSyncedTimeMs;
102   
103       public ConsumerTask(KafkaConsumer<byte[], byte[]> consumer,
104                           RemotePartitionMetadataEventHandler remotePartitionMetadataEventHandler,
105                           RemoteLogMetadataTopicPartitioner topicPartitioner,
106                           Path committedOffsetsPath,
107                           Time time,
108                           long committedOffsetSyncIntervalMs) {
109           this.consumer = Objects.requireNonNull(consumer);
110           this.remotePartitionMetadataEventHandler = Objects.requireNonNull(remotePartitionMetadataEventHandler);
111           this.topicPartitioner = Objects.requireNonNull(topicPartitioner);
112           this.time = Objects.requireNonNull(time);
113           this.committedOffsetSyncIntervalMs = committedOffsetSyncIntervalMs;
114   
115           initializeConsumerAssignment(committedOffsetsPath);
116       }
117   
118       private void initializeConsumerAssignment(Path committedOffsetsPath) {
119           try {
120               committedOffsetsFile = new CommittedOffsetsFile(committedOffsetsPath.toFile());
121           } catch (IOException e) {
122               throw new KafkaException(e);
123           }
124   
125           Map<Integer, Long> committedOffsets = Collections.emptyMap();
126           try {
127               // Load committed offset and assign them in the consumer.
128               committedOffsets = committedOffsetsFile.readEntries();
129           } catch (IOException e) {
130               // Ignore the error and consumer consumes from the earliest offset.
131               log.error("Encountered error while building committed offsets from the file. " +
132                                 "Consumer will consume from the earliest offset for the assigned partitions.", e);
133           }
134   
135           if (!committedOffsets.isEmpty()) {
136               // Assign topic partitions from the earlier committed offsets file.
137               Set<Integer> earlierAssignedPartitions = committedOffsets.keySet();
138               assignedMetaPartitions = Collections.unmodifiableSet(earlierAssignedPartitions);
139               Set<TopicPartition> metadataTopicPartitions = earlierAssignedPartitions.stream()
140                                                                                      .map(x -> new TopicPartition(REMOTE_LOG_METADATA_TOPIC_NAME, x))
141                                                                                      .collect(Collectors.toSet());
142               consumer.assign(metadataTopicPartitions);
143   
144               // Seek to the committed offsets
145               for (Map.Entry<Integer, Long> entry : committedOffsets.entrySet()) {
146                   partitionToConsumedOffsets.put(entry.getKey(), entry.getValue());
147                   consumer.seek(new TopicPartition(REMOTE_LOG_METADATA_TOPIC_NAME, entry.getKey()), entry.getValue());
148               }
149   
150               lastSyncedPartitionToConsumedOffsets = Collections.unmodifiableMap(committedOffsets);
151           }
152       }
153   
154       @Override
155       public void run() {
156           log.info("Started Consumer task thread.");
157           lastSyncedTimeMs = time.milliseconds();
158           try {
159               while (!closing) {
160                   maybeWaitForPartitionsAssignment();
161   
162                   log.info("Polling consumer to receive remote log metadata topic records");
163                   ConsumerRecords<byte[], byte[]> consumerRecords = consumer.poll(Duration.ofMillis(POLL_INTERVAL_MS));
164                   for (ConsumerRecord<byte[], byte[]> record : consumerRecords) {
165                       processConsumerRecord(record);
166                   }
167   
168                   maybeSyncCommittedDataAndOffsets(false);
169               }
170           } catch (Exception e) {
171               log.error("Error occurred in consumer task, close:[{}]", closing, e);
172           } finally {
173               maybeSyncCommittedDataAndOffsets(true);
174               closeConsumer();
175               log.info("Exiting from consumer task thread");
176           }
177       }
178   
179       private void processConsumerRecord(ConsumerRecord<byte[], byte[]> record) {
180           // Taking assignPartitionsLock here as updateAssignmentsForPartitions changes assignedTopicPartitions
181           // and also calls remotePartitionMetadataEventHandler.clearTopicPartition(removedPartition) for the removed
182           // partitions.
183           RemoteLogMetadata remoteLogMetadata = serde.deserialize(record.value());
184           synchronized (assignPartitionsLock) {
185               if (assignedTopicPartitions.contains(remoteLogMetadata.topicIdPartition())) {
186                   remotePartitionMetadataEventHandler.handleRemoteLogMetadata(remoteLogMetadata);
187               } else {
188                   log.debug("This event {} is skipped as the topic partition is not assigned for this instance.", remoteLogMetadata);
189               }
190               partitionToConsumedOffsets.put(record.partition(), record.offset());
191           }
192       }
193   
194       private void maybeSyncCommittedDataAndOffsets(boolean forceSync) {
195           // Return immediately if there is no consumption from last time.
196           boolean noConsumedOffsetUpdates = partitionToConsumedOffsets.equals(lastSyncedPartitionToConsumedOffsets);
197           if (noConsumedOffsetUpdates || !forceSync && time.milliseconds() - lastSyncedTimeMs < committedOffsetSyncIntervalMs) {
198               log.debug("Skip syncing committed offsets, noConsumedOffsetUpdates: {}, forceSync: {}", noConsumedOffsetUpdates, forceSync);
199               return;
200           }
201   
202           try {
203               // Need to take lock on assignPartitionsLock as assignedTopicPartitions might
204               // get updated by other threads.
205               synchronized (assignPartitionsLock) {
206                   for (TopicIdPartition topicIdPartition : assignedTopicPartitions) {
207                       int metadataPartition = topicPartitioner.metadataPartition(topicIdPartition);
208                       Long offset = partitionToConsumedOffsets.get(metadataPartition);
209                       if (offset != null) {
210                           remotePartitionMetadataEventHandler.syncLogMetadataSnapshot(topicIdPartition, metadataPartition, offset);
211                       } else {
212                           log.debug("Skipping syncup of the remote-log-metadata-file for partition:{} , with remote log metadata partition{}, and no offset",
213                                   topicIdPartition, metadataPartition);
214                       }
215                   }
216   
217                   // Write partitionToConsumedOffsets into committed offsets file as we do not want to process them again
218                   // in case of restarts.
219                   committedOffsetsFile.writeEntries(partitionToConsumedOffsets);
220                   lastSyncedPartitionToConsumedOffsets = new HashMap<>(partitionToConsumedOffsets);
221               }
222   
223               lastSyncedTimeMs = time.milliseconds();
224           } catch (IOException e) {
225               throw new KafkaException("Error encountered while writing committed offsets to a local file", e);
226           }
227       }
228   
229       private void closeConsumer() {
230           log.info("Closing the consumer instance");
231           try {
232               consumer.close(Duration.ofSeconds(30));
233           } catch (Exception e) {
234               log.error("Error encountered while closing the consumer", e);
235           }
236       }

238       private void maybeWaitForPartitionsAssignment() {
239           Set<Integer> assignedMetaPartitionsSnapshot = Collections.emptySet();
240           synchronized (assignPartitionsLock) {
241               // If it is closing, return immediately. This should be inside the assignPartitionsLock as the closing is updated
242               // in close() method with in the same lock to avoid any race conditions.
243               if (closing) {
244                   return;
245               }
246   
247               while (assignedMetaPartitions.isEmpty()) {
248                   // If no partitions are assigned, wait until they are assigned.
249                   log.debug("Waiting for assigned remote log metadata partitions..");
250                   try {
251                       // No timeout is set here, as it is always notified. Even when it is closed, the race can happen
252                       // between the thread calling this method and the thread calling close(). We should have a check
253                       // for closing as that might have been set and notified with assignPartitionsLock by `close`
254                       // method.
255                       assignPartitionsLock.wait();
256   
257                       if (closing) {
258                           return;
259                       }
260                   } catch (InterruptedException e) {
261                       throw new KafkaException(e);
262                   }
263               }
264   
265               if (assignPartitions) {
266                   assignedMetaPartitionsSnapshot = new HashSet<>(assignedMetaPartitions);
267                   // Removing unassigned meta partitions from partitionToConsumedOffsets and partitionToCommittedOffsets
268                   partitionToConsumedOffsets.entrySet().removeIf(entry -> !assignedMetaPartitions.contains(entry.getKey()));
269   
270                   assignPartitions = false;
271               }
272           }
273   
274           if (!assignedMetaPartitionsSnapshot.isEmpty()) {
275               executeReassignment(assignedMetaPartitionsSnapshot);
276           }
277       }
278   
279       private void executeReassignment(Set<Integer> assignedMetaPartitionsSnapshot) {
280           Set<TopicPartition> assignedMetaTopicPartitions =
281                   assignedMetaPartitionsSnapshot.stream()
282                                                 .map(partitionNum -> new TopicPartition(REMOTE_LOG_METADATA_TOPIC_NAME, partitionNum))
283                                                 .collect(Collectors.toSet());
284           log.info("Reassigning partitions to consumer task [{}]", assignedMetaTopicPartitions);
285           consumer.assign(assignedMetaTopicPartitions);
286       }

288       public void addAssignmentsForPartitions(Set<TopicIdPartition> partitions) {
289           updateAssignmentsForPartitions(partitions, Collections.emptySet());
290       }
291   
292       public void removeAssignmentsForPartitions(Set<TopicIdPartition> partitions) {
293           updateAssignmentsForPartitions(Collections.emptySet(), partitions);
294       }
295   
296       private void updateAssignmentsForPartitions(Set<TopicIdPartition> addedPartitions,
297                                                   Set<TopicIdPartition> removedPartitions) {
298           log.info("Updating assignments for addedPartitions: {} and removedPartition: {}", addedPartitions, removedPartitions);
299   
300           Objects.requireNonNull(addedPartitions, "addedPartitions must not be null");
301           Objects.requireNonNull(removedPartitions, "removedPartitions must not be null");
302   
303           if (addedPartitions.isEmpty() && removedPartitions.isEmpty()) {
304               return;
305           }
306   
307           synchronized (assignPartitionsLock) {
308               Set<TopicIdPartition> updatedReassignedPartitions = new HashSet<>(assignedTopicPartitions);
309               updatedReassignedPartitions.addAll(addedPartitions);
310               updatedReassignedPartitions.removeAll(removedPartitions);
311               Set<Integer> updatedAssignedMetaPartitions = new HashSet<>();
312               for (TopicIdPartition tp : updatedReassignedPartitions) {
313                   updatedAssignedMetaPartitions.add(topicPartitioner.metadataPartition(tp));
314               }
315   
316               // Clear removed topic partitions from inmemory cache.
317               for (TopicIdPartition removedPartition : removedPartitions) {
318                   remotePartitionMetadataEventHandler.clearTopicPartition(removedPartition);
319               }
320   
321               assignedTopicPartitions = Collections.unmodifiableSet(updatedReassignedPartitions);
322               log.debug("Assigned topic partitions: {}", assignedTopicPartitions);
323   
324               if (!updatedAssignedMetaPartitions.equals(assignedMetaPartitions)) {
325                   assignedMetaPartitions = Collections.unmodifiableSet(updatedAssignedMetaPartitions);
326                   log.debug("Assigned metadata topic partitions: {}", assignedMetaPartitions);
327   
328                   assignPartitions = true;
329                   assignPartitionsLock.notifyAll();
330               } else {
331                   log.debug("No change in assigned metadata topic partitions: {}", assignedMetaPartitions);
332               }
333           }
334       }

336       public Optional<Long> receivedOffsetForPartition(int partition) {
337           return Optional.ofNullable(partitionToConsumedOffsets.get(partition));
338       }
339   
340       public boolean isPartitionAssigned(int partition) {
341           return assignedMetaPartitions.contains(partition);
342       }
343   
344       public void close() {
345           if (!closing) {
346               synchronized (assignPartitionsLock) {
347                   // Closing should be updated only after acquiring the lock to avoid race in
348                   // maybeWaitForPartitionsAssignment() where it waits on assignPartitionsLock. It should not wait
349                   // if the closing is already set.
350                   closing = true;
351                   consumer.wakeup();
352                   assignPartitionsLock.notifyAll();
353               }
354           }
355       }
