17   package org.apache.kafka.server.log.remote.metadata.storage;
18   
19   import org.apache.kafka.common.KafkaException;
20   import org.apache.kafka.common.TopicIdPartition;
21   import org.apache.kafka.server.log.remote.storage.RemoteLogSegmentId;
22   import org.apache.kafka.server.log.remote.storage.RemoteLogSegmentMetadata;
23   import org.slf4j.Logger;
24   import org.slf4j.LoggerFactory;
25   
26   import java.io.IOException;
27   import java.nio.file.Path;
28   import java.util.ArrayList;
29   import java.util.List;
30   
31   /**
32    * This is a wrapper around {@link RemoteLogMetadataCache} providing a file based snapshot of
33    * {@link RemoteLogMetadataCache} for the given {@code topicIdPartition}. Snapshot is stored in the given
34    * {@code partitionDir}.
35    */
36   public class FileBasedRemoteLogMetadataCache extends RemoteLogMetadataCache {
37       private static final Logger log = LoggerFactory.getLogger(FileBasedRemoteLogMetadataCache.class);
38       private final RemoteLogMetadataSnapshotFile snapshotFile;
39       private final TopicIdPartition topicIdPartition;
40   
41       public FileBasedRemoteLogMetadataCache(TopicIdPartition topicIdPartition,
42                                              Path partitionDir) {
43           if (!partitionDir.toFile().exists() || !partitionDir.toFile().isDirectory()) {
44               throw new KafkaException("Given partition directory:" + partitionDir + " must be an existing directory.");
45           }
46   
47           this.topicIdPartition = topicIdPartition;
48           snapshotFile = new RemoteLogMetadataSnapshotFile(partitionDir);
49   
50           try {
51               snapshotFile.read().ifPresent(snapshot -> loadRemoteLogSegmentMetadata(snapshot));
52           } catch (IOException e) {
53               throw new KafkaException(e);
54           }
55       }
56   
57       protected void loadRemoteLogSegmentMetadata(RemoteLogMetadataSnapshotFile.Snapshot snapshot) {
58           log.info("Loading snapshot for partition {} is: {}", topicIdPartition, snapshot);
59           for (RemoteLogSegmentMetadataSnapshot metadataSnapshot : snapshot.remoteLogSegmentMetadataSnapshots()) {
60               switch (metadataSnapshot.state()) {
61                   case COPY_SEGMENT_STARTED:
62                       addCopyInProgressSegment(createRemoteLogSegmentMetadata(metadataSnapshot));
63                       break;
64                   case COPY_SEGMENT_FINISHED:
65                       handleSegmentWithCopySegmentFinishedState(createRemoteLogSegmentMetadata(metadataSnapshot));
66                       break;
67                   case DELETE_SEGMENT_STARTED:
68                       handleSegmentWithDeleteSegmentStartedState(createRemoteLogSegmentMetadata(metadataSnapshot));
69                       break;
70                   case DELETE_SEGMENT_FINISHED:
71                   default:
72                       throw new IllegalArgumentException("Given remoteLogSegmentMetadata has invalid state: " + metadataSnapshot);
73               }
74           }
75       }
76   
77       private RemoteLogSegmentMetadata createRemoteLogSegmentMetadata(RemoteLogSegmentMetadataSnapshot snapshot) {
78           return new RemoteLogSegmentMetadata(new RemoteLogSegmentId(topicIdPartition, snapshot.segmentId()), snapshot.startOffset(),
79                                               snapshot.endOffset(), snapshot.maxTimestampMs(), snapshot.brokerId(), snapshot.eventTimestampMs(),
80                                               snapshot.segmentSizeInBytes(), snapshot.state(), snapshot.segmentLeaderEpochs());
81       }
82   
83       /**
84        * Flushes the in-memory state to the snapshot file.
85        *
86        * @param metadataPartition       remote log metadata partition from which the messages have been consumed for the given
87        *                                user topic partition.
88        * @param metadataPartitionOffset remote log metadata partition offset up to which the messages have been consumed.
89        * @throws IOException if any errors occurred while writing the snapshot to the file.
90        */
91       public void flushToFile(int metadataPartition,
92                               Long metadataPartitionOffset) throws IOException {
93           List<RemoteLogSegmentMetadataSnapshot> snapshots = new ArrayList<>(idToSegmentMetadata.size());
94           for (RemoteLogLeaderEpochState state : leaderEpochEntries.values()) {
95               // Add unreferenced segments first, as to maintain the order when these segments are again read from
96               // the snapshot to build RemoteLogMetadataCache.
97               for (RemoteLogSegmentId id : state.unreferencedSegmentIds()) {
98                   snapshots.add(RemoteLogSegmentMetadataSnapshot.create(idToSegmentMetadata.get(id)));
99               }
100               
101               // Add referenced segments.
102               for (RemoteLogSegmentId id : state.referencedSegmentIds()) {
103                   snapshots.add(RemoteLogSegmentMetadataSnapshot.create(idToSegmentMetadata.get(id)));
104               }
105           }
106   
107           snapshotFile.write(new RemoteLogMetadataSnapshotFile.Snapshot(metadataPartition, metadataPartitionOffset, snapshots));
108       }
109   }
