102       public CompletableFuture<Void> addRemoteLogSegmentMetadata(RemoteLogSegmentMetadata remoteLogSegmentMetadata)
103               throws RemoteStorageException {
104           Objects.requireNonNull(remoteLogSegmentMetadata, "remoteLogSegmentMetadata can not be null");
105   
106           // This allows gracefully rejecting the requests while closing of this instance is in progress, which triggers
107           // closing the producer/consumer manager instances.
108           lock.readLock().lock();
109           try {
110               ensureInitializedAndNotClosed();

112               // This method is allowed only to add remote log segment with the initial state(which is RemoteLogSegmentState.COPY_SEGMENT_STARTED)
113               // but not to update the existing remote log segment metadata.

116                           "Given remoteLogSegmentMetadata should have state as " + RemoteLogSegmentState.COPY_SEGMENT_STARTED
117                                   + " but it contains state as: " + remoteLogSegmentMetadata.state());
118               }
119   
120               // Publish the message to the topic.
121               return storeRemoteLogMetadata(remoteLogSegmentMetadata.remoteLogSegmentId().topicIdPartition(),
122                                             remoteLogSegmentMetadata);
123           } finally {
124               lock.readLock().unlock();

166       /**
167        * Returns {@link CompletableFuture} which will complete only after publishing of the given {@code remoteLogMetadata} into
168        * the remote log metadata topic and the internal consumer is caught up until the produced record's offset.
169        *
170        * @param topicIdPartition partition of the given remoteLogMetadata.
171        * @param remoteLogMetadata RemoteLogMetadata to be stored.
172        * @return
173        * @throws RemoteStorageException if there are any storage errors occur.
174        */
175       private CompletableFuture<Void> storeRemoteLogMetadata(TopicIdPartition topicIdPartition,
176                                                              RemoteLogMetadata remoteLogMetadata)
177               throws RemoteStorageException {
178           log.debug("Storing metadata for partition: [{}] with context: [{}]", topicIdPartition, remoteLogMetadata);
179   
180           try {
181               // Publish the message to the metadata topic.
182               CompletableFuture<RecordMetadata> produceFuture = producerManager.publishMessage(remoteLogMetadata);
183   
184               // Create and return a `CompletableFuture` instance which completes when the consumer is caught up with the produced record's offset.
185               return produceFuture.thenApplyAsync(recordMetadata -> {
186                   try {
187                       consumerManager.waitTillConsumptionCatchesUp(recordMetadata);
188                   } catch (TimeoutException e) {
189                       throw new KafkaException(e);
190                   }
191                   return null;
192               });
193           } catch (KafkaException e) {
194               if (e instanceof RetriableException) {
195                   throw e;
196               } else {
197                   throw new RemoteStorageException(e);
198               }
199           }
200       }

272       @Override
273       public void onPartitionLeadershipChanges(Set<TopicIdPartition> leaderPartitions,
274                                                Set<TopicIdPartition> followerPartitions) {
275           Objects.requireNonNull(leaderPartitions, "leaderPartitions can not be null");
276           Objects.requireNonNull(followerPartitions, "followerPartitions can not be null");
277   
278           log.info("Received leadership notifications with leader partitions {} and follower partitions {}",
279                    leaderPartitions, followerPartitions);
280   
281           lock.readLock().lock();
282           try {
283               if (closing.get()) {
284                   throw new IllegalStateException("This instance is in closing state");
285               }
286   
287               HashSet<TopicIdPartition> allPartitions = new HashSet<>(leaderPartitions);
288               allPartitions.addAll(followerPartitions);
289               if (!initialized.get()) {
290                   // If it is not yet initialized, then keep them as pending partitions and assign them
291                   // when it is initialized successfully in initializeResources().
292                   this.pendingAssignPartitions.addAll(allPartitions);
293               } else {
294                   assignPartitions(allPartitions);
295               }
296           } finally {
297               lock.readLock().unlock();
298           }
299       }

309       @Override
310       public void onStopPartitions(Set<TopicIdPartition> partitions) {
311           lock.readLock().lock();
312           try {
313               if (closing.get()) {
314                   throw new IllegalStateException("This instance is in closing state");
315               }
316   
317               if (!initialized.get()) {
318                   // If it is not yet initialized, then remove them from the pending partitions if any.
319                   if (!pendingAssignPartitions.isEmpty()) {
320                       pendingAssignPartitions.removeAll(partitions);
321                   }
322               } else {
323                   consumerManager.removeAssignmentsForPartitions(partitions);
324               }
325           } finally {
326               lock.readLock().unlock();
327           }

330       @Override
331       public void configure(Map<String, ?> configs) {
332           Objects.requireNonNull(configs, "configs can not be null.");
333   
334           lock.writeLock().lock();
335           try {
336               if (configured) {
337                   log.info("Skipping configure as it is already configured.");
338                   return;
339               }
340   
341               log.info("Started initializing with configs: {}", configs);
342   
343               rlmmConfig = new TopicBasedRemoteLogMetadataManagerConfig(configs);
344               rlmmTopicPartitioner = new RemoteLogMetadataTopicPartitioner(rlmmConfig.metadataTopicPartitionsCount());
345               remotePartitionMetadataStore = new RemotePartitionMetadataStore(new File(rlmmConfig.logDir()).toPath());
346               configured = true;
347               log.info("Successfully initialized with rlmmConfig: {}", rlmmConfig);
348   
349               // Scheduling the initialization producer/consumer managers in a separate thread. Required resources may
350               // not yet be available now. This thread makes sure that it is retried at regular intervals until it is
351               // successful.
352               initializationThread = KafkaThread.nonDaemon("RLMMInitializationThread", () -> initializeResources());
353               initializationThread.start();
354           } finally {
355               lock.writeLock().unlock();
356           }
357       }

359       private void initializeResources() {
360           log.info("Initializing the resources.");
361           final NewTopic remoteLogMetadataTopicRequest = createRemoteLogMetadataTopicRequest();
362           boolean topicCreated = false;
363           long startTimeMs = time.milliseconds();
364           AdminClient adminClient = null;
365           try {
366               adminClient = AdminClient.create(rlmmConfig.producerProperties());
367   
368               // Stop if it is already initialized or closing.
369               while (!(initialized.get() || closing.get())) {
370   
371                   // If it is timed out then raise an error to exit.
372                   if (time.milliseconds() - startTimeMs > rlmmConfig.initializationRetryMaxTimeoutMs()) {
373                       log.error("Timed out in initializing the resources, retried to initialize the resource for [{}] ms.",
374                                 rlmmConfig.initializationRetryMaxTimeoutMs());
375                       initializationFailed = true;
376                       return;
377                   }
378   
379                   if (!topicCreated) {
380                       topicCreated = createTopic(adminClient, remoteLogMetadataTopicRequest);
381                   }
382   
383                   if (!topicCreated) {
384                       // Sleep for INITIALIZATION_RETRY_INTERVAL_MS before trying to create the topic again.
385                       log.info("Sleep for : {} ms before it is retried again.", rlmmConfig.initializationRetryIntervalMs());
386                       Utils.sleep(rlmmConfig.initializationRetryIntervalMs());
387                       continue;
388                   } else {
389                       // If topic is already created, validate the existing topic partitions.
390                       try {
391                           String topicName = remoteLogMetadataTopicRequest.name();
392                           // If the existing topic partition size is not same as configured, mark initialization as failed and exit.
393                           if (!isPartitionsCountSameAsConfigured(adminClient, topicName)) {
394                               initializationFailed = true;
395                           }
396                       } catch (Exception e) {
397                           log.info("Sleep for : {} ms before it is retried again.", rlmmConfig.initializationRetryIntervalMs());
398                           Utils.sleep(rlmmConfig.initializationRetryIntervalMs());
399                           continue;
400                       }
401                   }
402   
403                   // Create producer and consumer managers.
404                   lock.writeLock().lock();
405                   try {
406                       producerManager = new ProducerManager(rlmmConfig, rlmmTopicPartitioner);
407                       consumerManager = new ConsumerManager(rlmmConfig, remotePartitionMetadataStore, rlmmTopicPartitioner, time);
408                       if (startConsumerThread) {
409                           consumerManager.startConsumerThread();
410                       } else {
411                           log.info("RLMM Consumer task thread is not configured to be started.");
412                       }
413   
414                       if (!pendingAssignPartitions.isEmpty()) {
415                           assignPartitions(pendingAssignPartitions);
416                           pendingAssignPartitions.clear();
417                       }
418   
419                       initialized.set(true);
420                       log.info("Initialized resources successfully.");
421                   } catch (Exception e) {
422                       log.error("Encountered error while initializing producer/consumer", e);
423                       return;
424                   } finally {
425                       lock.writeLock().unlock();
426                   }
427               }
428   
429           } finally {
430               if (adminClient != null) {
431                   try {
432                       adminClient.close(Duration.ofSeconds(10));
433                   } catch (Exception e) {
434                       // Ignore the error.
435                       log.debug("Error occurred while closing the admin client", e);
436                   }
437               }
438           }
439       }

441       private boolean isPartitionsCountSameAsConfigured(AdminClient adminClient,
442                                                         String topicName) throws InterruptedException, ExecutionException {
443           log.debug("Getting topic details to check for partition count and replication factor.");
444           TopicDescription topicDescription = adminClient.describeTopics(Collections.singleton(topicName))
445                                                          .topicNameValues().get(topicName).get();
446           int expectedPartitions = rlmmConfig.metadataTopicPartitionsCount();
447           int topicPartitionsSize = topicDescription.partitions().size();
448   
449           if (topicPartitionsSize != expectedPartitions) {
450               log.error("Existing topic partition count [{}] is not same as the expected partition count [{}]",
451                         topicPartitionsSize, expectedPartitions);
452               return false;
453           }
454   
455           return true;
456       }

458       private NewTopic createRemoteLogMetadataTopicRequest() {
459           Map<String, String> topicConfigs = new HashMap<>();
460           topicConfigs.put(TopicConfig.RETENTION_MS_CONFIG, Long.toString(rlmmConfig.metadataTopicRetentionMs()));
461           topicConfigs.put(TopicConfig.CLEANUP_POLICY_CONFIG, TopicConfig.CLEANUP_POLICY_DELETE);
462           return new NewTopic(rlmmConfig.remoteLogMetadataTopicName(),

464                               rlmmConfig.metadataTopicReplicationFactor()).configs(topicConfigs);

467       /**
468        * @param topic topic to be created.
469        * @return Returns true if the topic already exists or it is created successfully.
470        */
471       private boolean createTopic(AdminClient adminClient, NewTopic topic) {
472           boolean topicCreated = false;
473           try {
474               adminClient.createTopics(Collections.singleton(topic)).all().get();
475               topicCreated = true;

479                   topicCreated = true;
480               } else {
481                   log.error("Encountered error while creating remote log metadata topic.", e);
482               }
