18   package org.apache.kafka.image;
19   
20   import org.apache.kafka.common.config.ConfigResource;
21   import org.apache.kafka.common.metadata.ConfigRecord;
22   import org.apache.kafka.server.common.ApiMessageAndVersion;
23   
24   import java.util.ArrayList;
25   import java.util.Collections;
26   import java.util.List;
27   import java.util.Map;
28   import java.util.Objects;
29   import java.util.Properties;
30   import java.util.function.Consumer;
31   import java.util.stream.Collectors;
32   
33   import static org.apache.kafka.common.metadata.MetadataRecordType.CONFIG_RECORD;
34   
35   
36   /**
37    * Represents the configuration of a resource.
38    *
39    * This class is thread-safe.
40    */
41   public final class ConfigurationImage {
42       public static final ConfigurationImage EMPTY = new ConfigurationImage(Collections.emptyMap());
43   
44       private final Map<String, String> data;
45   
46       public ConfigurationImage(Map<String, String> data) {
47           this.data = data;
48       }
49   
50       Map<String, String> data() {
51           return data;
52       }
53   
54       public boolean isEmpty() {
55           return data.isEmpty();
56       }
57   
58       public Properties toProperties() {
59           Properties properties = new Properties();
60           properties.putAll(data);
61           return properties;
62       }
63   
64       public void write(ConfigResource configResource, Consumer<List<ApiMessageAndVersion>> out) {
65           List<ApiMessageAndVersion> records = new ArrayList<>();
66           for (Map.Entry<String, String> entry : data.entrySet()) {
67               records.add(new ApiMessageAndVersion(new ConfigRecord().
68                   setResourceType(configResource.type().id()).
69                   setResourceName(configResource.name()).
70                   setName(entry.getKey()).
71                   setValue(entry.getValue()), CONFIG_RECORD.highestSupportedVersion()));
72           }
73           out.accept(records);
74       }
75   
76       @Override
77       public boolean equals(Object o) {
78           if (!(o instanceof ConfigurationImage)) return false;
79           ConfigurationImage other = (ConfigurationImage) o;
80           return data.equals(other.data);
81       }
82   
83       @Override
84       public int hashCode() {
85           return Objects.hash(data);
86       }
87   
88       @Override
89       public String toString() {
90           return "ConfigurationImage(data=" + data.entrySet().stream().
91               map(e -> e.getKey() + ":" + e.getValue()).collect(Collectors.joining(", ")) +
92               ")";
93       }
94   }
