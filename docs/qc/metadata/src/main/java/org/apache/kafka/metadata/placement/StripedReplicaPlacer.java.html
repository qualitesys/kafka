33   /**
34    * The striped replica placer.
35    *
36    *
37    * GOALS
38    * The design of this placer attempts to satisfy a few competing goals.  Firstly, we want
39    * to spread the replicas as evenly as we can across racks.  In the simple case where
40    * broker racks have not been configured, this goal is a no-op, of course.  But it is the
41    * highest priority goal in multi-rack clusters.
42    *
43    * Our second goal is to spread the replicas evenly across brokers.  Since we are placing
44    * multiple partitions, we try to avoid putting each partition on the same set of
45    * replicas, even if it does satisfy the rack placement goal.  If any specific broker is
46    * fenced, we would like the new leaders to distributed evenly across the remaining
47    * brokers.
48    *
49    * However, we treat the rack placement goal as higher priority than this goal-- if you
50    * configure 10 brokers in rack A and B, and 1 broker in rack C, you will end up with a
51    * lot of partitions on that one broker in rack C.  If you were to place a lot of
52    * partitions with replication factor 3, each partition would try to get a replica there.
53    * In general racks are supposed to be about the same size -- if they aren't, this is a
54    * user error.
55    *
56    * Finally, we would prefer to place replicas on unfenced brokers, rather than on fenced
57    * brokers.
58    *
59    *
60    * CONSTRAINTS
61    * In addition to these goals, we have two constraints.  Unlike the goals, these are not
62    * optional -- they are mandatory.  Placement will fail if a constraint cannot be
63    * satisfied.  The first constraint is that we can't place more than one replica on the
64    * same broker.  This imposes an upper limit on replication factor-- for example, a 3-node
65    * cluster can't have any topics with replication factor 4.  This constraint comes from
66    * Kafka's internal design.
67    *
68    * The second constraint is that the leader of each partition must be an unfenced broker.
69    * This constraint is a bit arbitrary.  In theory, we could allow people to create
70    * new topics even if every broker were fenced.  However, this would be confusing for
71    * users.
72    *
73    *
74    * ALGORITHM
75    * The StripedReplicaPlacer constructor loads the broker data into rack objects.  Each
76    * rack object contains a sorted list of fenced brokers, and a separate sorted list of
77    * unfenced brokers.  The racks themselves are organized into a sorted list, stored inside
78    * the top-level RackList object.
79    *
80    * The general idea is that we place replicas on to racks in a round-robin fashion.  So if
81    * we had racks A, B, C, and D, and we were creating a new partition with replication
82    * factor 3, our first replica might come from A, our second from B, and our third from C.
83    * Of course our placement would not be very fair if we always started with rack A.
84    * Therefore, we generate a random starting offset when the RackList is created.  So one
85    * time we might go B, C, D.  Another time we might go C, D, A.  And so forth.
86    *
87    * Note that each partition we generate advances the starting offset by one.
88    * So in our 4-rack cluster, with 3 partitions, we might choose these racks:
89    *
90    * partition 1: A, B, C
91    * partition 2: B, C, A
92    * partition 3: C, A, B
93    *
94    * This is what generates the characteristic "striped" pattern of this placer.
95    *
96    * So far I haven't said anything about how we choose a replica from within a rack.  In
97    * fact, this is also done in a round-robin fashion.  So if rack A had replica A0, A1, A2,
98    * and A3, we might return A0 the first time, A1, the second, A2 the third, and so on.
99    * Just like with the racks, we add a random starting offset to mix things up a bit.
100    *
101    * So let's say you had a cluster with racks A, B, and C, and each rack had 3 replicas,
102    * for 9 nodes in total.
103    * If all the offsets were 0, you'd get placements like this:
104    *
105    * partition 1: A0, B0, C0
106    * partition 2: B1, C1, A1
107    * partition 3: C2, A2, B2
108    *
109    * One additional complication with choosing a replica within a rack is that we want to
110    * choose the unfenced replicas first.  In a big cluster with lots of nodes available,
111    * we'd prefer not to place a new partition on a node that is fenced.  Therefore, we
112    * actually maintain two lists, rather than the single list I described above.
113    * We only start using the fenced node list when the unfenced node list is totally
114    * exhausted.
115    *
116    * Furthermore, we cannot place the first replica (the leader) of a new partition on a
117    * fenced replica.  Therefore, we have some special logic to ensure that this doesn't
118    * happen.
119    */
120   public class StripedReplicaPlacer implements ReplicaPlacer {
121       /**
122        * A list of brokers that we can iterate through.
123        */
124       static class BrokerList {
125           final static BrokerList EMPTY = new BrokerList();
126           private final List<Integer> brokers = new ArrayList<>(0);
127   
128           /**
129            * How many brokers we have retrieved from the list during the current iteration epoch.
130            */
131           private int index = 0;
132   
133           /**
134            * The offset to add to the index in order to calculate the list entry to fetch.  The
135            * addition is done modulo the list size.
136            */
137           private int offset = 0;
138   
139           /**
140            * The last known iteration epoch. If we call next with a different epoch than this, the
141            * index and offset will be reset.
142            */
143           private int epoch = 0;
144   
145           BrokerList add(int broker) {
146               this.brokers.add(broker);
147               return this;
148           }
149   
150           /**
151            * Initialize this broker list by sorting it and randomizing the start offset.
152            *
153            * @param random    The random number generator.
154            */
155           void initialize(Random random) {
156               if (!brokers.isEmpty()) {
157                   brokers.sort(Integer::compareTo);
158                   this.offset = random.nextInt(brokers.size());
159               }
160           }
161   
162           /**
163            * Randomly shuffle the brokers in this list.
164            */
165           void shuffle(Random random) {
166               Collections.shuffle(brokers, random);
167           }
168   
169           /**
170            * @return          The number of brokers in this list.
171            */
172           int size() {
173               return brokers.size();
174           }
175   
176           /**
177            * Get the next broker in this list, or -1 if there are no more elements to be
178            * returned.
179            *
180            * @param epoch     The current iteration epoch.
181            *
182            * @return          The broker ID, or -1 if there are no more brokers to be
183            *                  returned in this epoch.
184            */
185           int next(int epoch) {
186               if (brokers.size() == 0) return -1;
187               if (this.epoch != epoch) {
188                   this.epoch = epoch;
189                   this.index = 0;
190                   this.offset = (offset + 1) % brokers.size();
191               }
192               if (index >= brokers.size()) return -1;
193               int broker = brokers.get((index + offset) % brokers.size());
194               index++;
195               return broker;
196           }
197       }

199       /**
200        * A rack in the cluster, which contains brokers.
201        */
202       static class Rack {
203           private final BrokerList fenced = new BrokerList();
204           private final BrokerList unfenced = new BrokerList();
205   
206           /**
207            * Initialize this rack.
208            *
209            * @param random    The random number generator.
210            */
211           void initialize(Random random) {
212               fenced.initialize(random);
213               unfenced.initialize(random);
214           }
215   
216           void shuffle(Random random) {
217               fenced.shuffle(random);
218               unfenced.shuffle(random);
219           }
220   
221           BrokerList fenced() {
222               return fenced;
223           }
224   
225           BrokerList unfenced() {
226               return unfenced;
227           }
228   
229           /**
230            * Get the next unfenced broker in this rack, or -1 if there are no more brokers
231            * to be returned.
232            *
233            * @param epoch     The current iteration epoch.
234            *
235            * @return          The broker ID, or -1 if there are no more brokers to be
236            *                  returned in this epoch.
237            */
238           int nextUnfenced(int epoch) {
239               return unfenced.next(epoch);
240           }
241   
242           /**
243            * Get the next broker in this rack, or -1 if there are no more brokers to be
244            * returned.
245            *
246            * @param epoch     The current iteration epoch.
247            *
248            * @return          The broker ID, or -1 if there are no more brokers to be
249            *                  returned in this epoch.
250            */
251           int next(int epoch) {
252               int result = unfenced.next(epoch);
253               if (result >= 0) return result;
254               return fenced.next(epoch);
255           }
256       }
257   
258       /**
259        * A list of racks that we can iterate through.
260        */
261       static class RackList {
262           /**
263            * The random number generator.
264            */
265           private final Random random;
266   
267           /**
268            * A map from rack names to the brokers contained within them.
269            */
270           private final Map<Optional<String>, Rack> racks = new HashMap<>();
271   
272           /**
273            * The names of all the racks in the cluster.
274            *
275            * Racks which have at least one unfenced broker come first (in sorted order),
276            * followed by racks which have only fenced brokers (also in sorted order).
277            */
278           private final List<Optional<String>> rackNames = new ArrayList<>();
279   
280           /**
281            * The total number of brokers in the cluster, both fenced and unfenced.
282            */
283           private final int numTotalBrokers;
284   
285           /**
286            * The total number of unfenced brokers in the cluster.
287            */
288           private final int numUnfencedBrokers;
289   
290           /**
291            * The iteration epoch.
292            */
293           private int epoch = 0;
294   
295           /**
296            * The offset we use to determine which rack is returned first.
297            */
298           private int offset;
299   
300           RackList(Random random, Iterator<UsableBroker> iterator) {
301               this.random = random;
302               int numTotalBrokersCount = 0, numUnfencedBrokersCount = 0;
303               while (iterator.hasNext()) {
304                   UsableBroker broker = iterator.next();
305                   Rack rack = racks.get(broker.rack());
306                   if (rack == null) {
307                       rackNames.add(broker.rack());
308                       rack = new Rack();
309                       racks.put(broker.rack(), rack);
310                   }
311                   if (broker.fenced()) {
312                       rack.fenced().add(broker.id());
313                   } else {
314                       numUnfencedBrokersCount++;
315                       rack.unfenced().add(broker.id());
316                   }
317                   numTotalBrokersCount++;
318               }
319               for (Rack rack : racks.values()) {
320                   rack.initialize(random);
321               }
322               this.rackNames.sort(OptionalStringComparator.INSTANCE);
323               this.numTotalBrokers = numTotalBrokersCount;
324               this.numUnfencedBrokers = numUnfencedBrokersCount;
325               this.offset = rackNames.isEmpty() ? 0 : random.nextInt(rackNames.size());
326           }
327   
328           int numTotalBrokers() {
329               return numTotalBrokers;
330           }
331   
332           int numUnfencedBrokers() {
333               return numUnfencedBrokers;
334           }
335   
336           // VisibleForTesting
337           List<Optional<String>> rackNames() {
338               return rackNames;
339           }
340   
341           List<Integer> place(int replicationFactor) {
342               throwInvalidReplicationFactorIfNonPositive(replicationFactor);
343               throwInvalidReplicationFactorIfTooFewBrokers(replicationFactor, numTotalBrokers());
344               throwInvalidReplicationFactorIfZero(numUnfencedBrokers());
345               // If we have returned as many assignments as there are unfenced brokers in
346               // the cluster, shuffle the rack list and broker lists to try to avoid
347               // repeating the same assignments again.
348               // But don't reset the iteration epoch for a single unfenced broker -- otherwise we would loop forever
349               if (epoch == numUnfencedBrokers && numUnfencedBrokers > 1) {
350                   shuffle();
351                   epoch = 0;
352               }
353               if (offset == rackNames.size()) {
354                   offset = 0;
355               }
356               List<Integer> brokers = new ArrayList<>(replicationFactor);
357               int firstRackIndex = offset;
358               while (true) {
359                   Optional<String> name = rackNames.get(firstRackIndex);
360                   Rack rack = racks.get(name);
361                   int result = rack.nextUnfenced(epoch);
362                   if (result >= 0) {
363                       brokers.add(result);
364                       break;
365                   }
366                   firstRackIndex++;
367                   if (firstRackIndex == rackNames.size()) {
368                       firstRackIndex = 0;
369                   }
370               }
371               int rackIndex = offset;
372               for (int replica = 1; replica < replicationFactor; replica++) {
373                   int result = -1;
374                   do {
375                       if (rackIndex == firstRackIndex) {
376                           firstRackIndex = -1;
377                       } else {
378                           Optional<String> rackName = rackNames.get(rackIndex);
379                           Rack rack = racks.get(rackName);
380                           result = rack.next(epoch);
381                       }
382                       rackIndex++;
383                       if (rackIndex == rackNames.size()) {
384                           rackIndex = 0;
385                       }
386                   } while (result < 0);
387                   brokers.add(result);
388               }
389               epoch++;
390               offset++;
391               return brokers;
392           }
393   
394           void shuffle() {
395               Collections.shuffle(rackNames, random);
396               for (Rack rack : racks.values()) {
397                   rack.shuffle(random);
398               }
399           }
400       }

402       private static void throwInvalidReplicationFactorIfNonPositive(int replicationFactor) {
403           if (replicationFactor <= 0) {
404               throw new InvalidReplicationFactorException("Invalid replication factor " +
405                       replicationFactor + ": the replication factor must be positive.");
406           }
407       }
408   
409       private static void throwInvalidReplicationFactorIfZero(int numUnfenced) {
410           if (numUnfenced == 0) {
411               throw new InvalidReplicationFactorException("All brokers are currently fenced.");
412           }
413       }
414   
415       private static void throwInvalidReplicationFactorIfTooFewBrokers(int replicationFactor, int numTotalBrokers) {
416           if (replicationFactor > numTotalBrokers) {
417               throw new InvalidReplicationFactorException("The target replication factor " +
418                       "of " + replicationFactor + " cannot be reached because only " +
419                       numTotalBrokers + " broker(s) are registered.");
420           }
421       }

423       private final Random random;
424   
425       public StripedReplicaPlacer(Random random) {
426           this.random = random;
427       }
428   
429       @Override
430       public List<List<Integer>> place(
431           PlacementSpec placement,
432           ClusterDescriber cluster
433       ) throws InvalidReplicationFactorException {
434           RackList rackList = new RackList(random, cluster.usableBrokers());
435           throwInvalidReplicationFactorIfNonPositive(placement.numReplicas());
436           throwInvalidReplicationFactorIfZero(rackList.numUnfencedBrokers());
437           throwInvalidReplicationFactorIfTooFewBrokers(placement.numReplicas(),
438               rackList.numTotalBrokers());
439           List<List<Integer>> placements = new ArrayList<>(placement.numPartitions());
440           for (int partition = 0; partition < placement.numPartitions(); partition++) {
441               placements.add(rackList.place(placement.numReplicas()));
442           }
443           return placements;
444       }
