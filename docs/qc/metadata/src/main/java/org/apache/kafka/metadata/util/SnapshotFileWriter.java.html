19   
20   import org.apache.kafka.common.record.CompressionType;
21   import org.apache.kafka.common.utils.Time;
22   import org.apache.kafka.common.utils.Utils;
23   import org.apache.kafka.metadata.MetadataRecordSerde;
24   import org.apache.kafka.raft.internals.BatchAccumulator;
25   import org.apache.kafka.raft.internals.BatchMemoryPool;
26   import org.apache.kafka.server.common.ApiMessageAndVersion;
27   
28   import java.io.IOException;
29   import java.nio.channels.FileChannel;
30   import java.nio.file.Path;
31   import java.nio.file.StandardOpenOption;
32   import java.util.Collections;
33   import java.util.List;
34   
35   import static org.apache.kafka.raft.KafkaRaftClient.MAX_BATCH_SIZE_BYTES;
36   
37   
38   /**
39    * Write an arbitrary set of metadata records into a Kafka metadata snapshot format. The resulting snapshot will be use
40    * epoch of zero and an initial offset of zero. This class should not be used for creating actual metadata snapshots.
41    */
42   public class SnapshotFileWriter implements AutoCloseable {
43       private final FileChannel channel;
44       private final BatchAccumulator<ApiMessageAndVersion> batchAccumulator;
45   
46       SnapshotFileWriter(FileChannel channel, BatchAccumulator<ApiMessageAndVersion> batchAccumulator) {
47           this.channel = channel;
48           this.batchAccumulator = batchAccumulator;
49       }
50   
51       public void append(ApiMessageAndVersion apiMessageAndVersion) {
52           batchAccumulator.append(0, Collections.singletonList(apiMessageAndVersion));
53       }
54   
55       public void append(List<ApiMessageAndVersion> messageBatch) {
56           batchAccumulator.append(0, messageBatch);
57       }
58   
59       public void close() throws IOException {
60           for (BatchAccumulator.CompletedBatch<ApiMessageAndVersion> batch : batchAccumulator.drain()) {
61               Utils.writeFully(channel, batch.data.buffer());
62           }
63           channel.close();
64       }
65   
66       public static SnapshotFileWriter open(Path snapshotPath) throws IOException {
67           BatchAccumulator<ApiMessageAndVersion> batchAccumulator = new BatchAccumulator<>(
68               0,
69               0,
70               Integer.MAX_VALUE,
71               MAX_BATCH_SIZE_BYTES,
72               new BatchMemoryPool(5, MAX_BATCH_SIZE_BYTES),
73               Time.SYSTEM,
74               CompressionType.NONE,
75               new MetadataRecordSerde());
76   
77           FileChannel channel = FileChannel.open(snapshotPath, StandardOpenOption.CREATE_NEW, StandardOpenOption.WRITE);
78   
79           return new SnapshotFileWriter(channel, batchAccumulator);
80       }
81   }
