38   /**
39    * Tracks information about the schema of configuration keys for brokers, topics, and other
40    * resources. Since this class does not depend on core, it is useful in the controller for
41    * determining the type of config keys (string, int, password, etc.)
42    */
43   public class KafkaConfigSchema {
44       public static final KafkaConfigSchema EMPTY = new KafkaConfigSchema(emptyMap(), emptyMap());
45   
46       private static final ConfigDef EMPTY_CONFIG_DEF = new ConfigDef();
47   
48       /**
49        * Translate a ConfigDef.Type to its equivalent for ConfigEntry.ConfigType.
50        *
51        * We do not want this code in ConfigEntry, since that is a public-facing API. On the
52        * other hand, putting this code in ConfigDef.Type would introduce an unwanted dependency
53        * from org.apache.kafka.common.config to org.apache.kafka.clients.admin. So it
54        * makes sense to put it here.
55        */
56       public static ConfigEntry.ConfigType translateConfigType(ConfigDef.Type type) {
57           switch (type) {
58               case BOOLEAN:
59                   return ConfigEntry.ConfigType.BOOLEAN;
60               case STRING:
61                   return ConfigEntry.ConfigType.STRING;
62               case INT:
63                   return ConfigEntry.ConfigType.INT;
64               case SHORT:
65                   return ConfigEntry.ConfigType.SHORT;
66               case LONG:
67                   return ConfigEntry.ConfigType.LONG;
68               case DOUBLE:
69                   return ConfigEntry.ConfigType.DOUBLE;
70               case LIST:
71                   return ConfigEntry.ConfigType.LIST;
72               case CLASS:
73                   return ConfigEntry.ConfigType.CLASS;
74               case PASSWORD:
75                   return ConfigEntry.ConfigType.PASSWORD;
76               default:
77                   return ConfigEntry.ConfigType.UNKNOWN;
78           }
79       }

81       private static final Map<ConfigEntry.ConfigSource, DescribeConfigsResponse.ConfigSource> TRANSLATE_CONFIG_SOURCE_MAP;
82   
83       static {
84           Map<ConfigEntry.ConfigSource, DescribeConfigsResponse.ConfigSource> map = new HashMap<>();
85           for (DescribeConfigsResponse.ConfigSource source : DescribeConfigsResponse.ConfigSource.values()) {
86               map.put(source.source(), source);
87           }
88           TRANSLATE_CONFIG_SOURCE_MAP = Collections.unmodifiableMap(map);
89       }
90   
91       /**
92        * Translate a ConfigEntry.ConfigSource enum to its equivalent for DescribeConfigsResponse.
93        *
94        * We do not want this code in ConfigEntry, since that is a public-facing API. On the
95        * other hand, putting this code in DescribeConfigsResponse would introduce an unwanted
96        * dependency from org.apache.kafka.common.requests to org.apache.kafka.clients.admin.
97        * So it makes sense to put it here.
98        */
99       public static DescribeConfigsResponse.ConfigSource translateConfigSource(ConfigEntry.ConfigSource configSource) {
100           DescribeConfigsResponse.ConfigSource result = TRANSLATE_CONFIG_SOURCE_MAP.get(configSource);
101           if (result != null) return result;
102           return DescribeConfigsResponse.ConfigSource.UNKNOWN;
103       }

115       /**
116        * Returns true if the configuration key specified is splittable (only lists are splittable.)
117        */
118       public boolean isSplittable(ConfigResource.Type type, String key) {
119           ConfigDef configDef = configDefs.get(type);
120           if (configDef == null) return false;
121           ConfigDef.ConfigKey configKey = configDef.configKeys().get(key);
122           if (configKey == null) return false;
123           return configKey.type == ConfigDef.Type.LIST;
124       }
125   
126       /**
127        * Returns true if the configuration key specified in this ConfigRecord is sensitive, or if
128        * we don't know whether it is sensitive.
129        */
130       public boolean isSensitive(ConfigRecord record) {
131           ConfigResource.Type type = ConfigResource.Type.forId(record.resourceType());
132           return isSensitive(type, record.name());
133       }
134   
135       /**
136        * Returns true if the configuration key specified is sensitive, or if we don't know whether
137        * it is sensitive.
138        */
139       public boolean isSensitive(ConfigResource.Type type, String key) {
140           ConfigDef configDef = configDefs.get(type);
141           if (configDef == null) return true;
142           ConfigDef.ConfigKey configKey = configDef.configKeys().get(key);
143           if (configKey == null) return true;
144           return configKey.type.isSensitive();
145       }

147       /**
148        * Get the default value of the configuration key, or null if no default is specified.
149        */
150       public String getDefault(ConfigResource.Type type, String key) {
151           ConfigDef configDef = configDefs.get(type);
152           if (configDef == null) return null;
153           ConfigDef.ConfigKey configKey = configDef.configKeys().get(key);
154           if (configKey == null || !configKey.hasDefault()) {
155               return null;
156           }
157           return ConfigDef.convertToString(configKey.defaultValue, configKey.type);
158       }

160       public Map<String, ConfigEntry> resolveEffectiveTopicConfigs(
161               Map<String, ? extends Object> staticNodeConfig,
162               Map<String, ? extends Object> dynamicClusterConfigs,
163               Map<String, ? extends Object> dynamicNodeConfigs,
164               Map<String, ? extends Object> dynamicTopicConfigs) {
165           ConfigDef configDef = configDefs.getOrDefault(ConfigResource.Type.TOPIC, EMPTY_CONFIG_DEF);
166           HashMap<String, ConfigEntry> effectiveConfigs = new HashMap<>();
167           for (ConfigDef.ConfigKey configKey : configDef.configKeys().values()) {
168               ConfigEntry entry = resolveEffectiveTopicConfig(configKey, staticNodeConfig,
169                   dynamicClusterConfigs, dynamicNodeConfigs, dynamicTopicConfigs);
170               effectiveConfigs.put(entry.name(), entry);
171           }
172           return effectiveConfigs;
173       }

175       private ConfigEntry resolveEffectiveTopicConfig(ConfigDef.ConfigKey configKey,
176               Map<String, ? extends Object> staticNodeConfig,
177               Map<String, ? extends Object> dynamicClusterConfigs,
178               Map<String, ? extends Object> dynamicNodeConfigs,
179               Map<String, ? extends Object> dynamicTopicConfigs) {
180           if (dynamicTopicConfigs.containsKey(configKey.name)) {
181               return toConfigEntry(configKey,
182                   dynamicTopicConfigs.get(configKey.name),
183                   ConfigSource.DYNAMIC_TOPIC_CONFIG, Function.identity());
184           }
185           List<ConfigSynonym> synonyms = logConfigSynonyms.getOrDefault(configKey.name, emptyList());
186           for (ConfigSynonym synonym : synonyms) {
187               if (dynamicNodeConfigs.containsKey(synonym.name())) {
188                   return toConfigEntry(configKey, dynamicNodeConfigs.get(synonym.name()),
189                       ConfigSource.DYNAMIC_BROKER_CONFIG, synonym.converter());
190               }
191           }
192           for (ConfigSynonym synonym : synonyms) {
193               if (dynamicClusterConfigs.containsKey(synonym.name())) {
194                   return toConfigEntry(configKey, dynamicClusterConfigs.get(synonym.name()),
195                       ConfigSource.DYNAMIC_DEFAULT_BROKER_CONFIG, synonym.converter());
196               }
197           }
198           for (ConfigSynonym synonym : synonyms) {
199               if (staticNodeConfig.containsKey(synonym.name())) {
200                   return toConfigEntry(configKey, staticNodeConfig.get(synonym.name()),
201                       ConfigSource.STATIC_BROKER_CONFIG, synonym.converter());
202               }
203           }
204           return toConfigEntry(configKey, configKey.hasDefault() ? configKey.defaultValue : null,
205               ConfigSource.DEFAULT_CONFIG, Function.identity());
206       }

208       private ConfigEntry toConfigEntry(ConfigDef.ConfigKey configKey,
209                                         Object value,
210                                         ConfigSource source,
211                                         Function<String, String> converter) {
212           // Convert the value into a nulllable string suitable for storing in ConfigEntry.
213           String stringValue = null;
214           if (value != null) {
215               if (value instanceof String) {
216                   // The value may already be a string if it's coming from a Map<String, String>.
217                   // Then it doesn't need to be converted.
218                   stringValue = (String) value;
219               } else if (value instanceof Password) {
220                   // We want the actual value here, not [hidden], which is what we'd get
221                   // from Password#toString. While we don't return sensitive config values
222                   // over the wire to users, we may need the real value internally.
223                   stringValue = ((Password) value).value();
224               } else {
225                   try {
226                       // Use the ConfigDef function here which will handle List, Class, etc.
227                       stringValue = ConfigDef.convertToString(value, configKey.type);
228                   } catch (Exception e) {
229                       throw new RuntimeException("Unable to convert " + configKey.name + " to string.", e);
230                   }
231               }
232           }
233           if (stringValue != null) {
234               stringValue = converter.apply(stringValue);
235           }
236           return new ConfigEntry(
237               configKey.name,
238               stringValue,
239               source,
240               configKey.type().isSensitive(),
241               false, // "readonly" is always false, for now.
242               emptyList(), // we don't populate synonyms, for now.
243               translateConfigType(configKey.type()),
244               configKey.documentation);
245       }
