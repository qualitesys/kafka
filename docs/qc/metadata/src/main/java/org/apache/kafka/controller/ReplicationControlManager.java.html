404       public void replay(TopicRecord record) {
405           topicsByName.put(record.name(), record.topicId());
406           if (Topic.hasCollisionChars(record.name())) {
407               String normalizedName = Topic.unifyCollisionChars(record.name());
408               TimelineHashSet<String> topicNames = topicsWithCollisionChars.get(normalizedName);
409               if (topicNames == null) {
410                   topicNames = new TimelineHashSet<>(snapshotRegistry, 1);
411                   topicsWithCollisionChars.put(normalizedName, topicNames);
412               }
413               topicNames.add(record.name());
414           }
415           topics.put(record.topicId(),
416               new TopicControlInfo(record.name(), snapshotRegistry, record.topicId()));
417           controllerMetrics.setGlobalTopicsCount(topics.size());
418           log.info("Created topic {} with topic ID {}.", record.name(), record.topicId());
419       }

421       public void replay(PartitionRecord record) {
422           TopicControlInfo topicInfo = topics.get(record.topicId());
423           if (topicInfo == null) {
424               throw new RuntimeException("Tried to create partition " + record.topicId() +
425                   ":" + record.partitionId() + ", but no topic with that ID was found.");
426           }
427           PartitionRegistration newPartInfo = new PartitionRegistration(record);
428           PartitionRegistration prevPartInfo = topicInfo.parts.get(record.partitionId());
429           String description = topicInfo.name + "-" + record.partitionId() +
430               " with topic ID " + record.topicId();
431           if (prevPartInfo == null) {
432               log.info("Created partition {} and {}.", description, newPartInfo);
433               topicInfo.parts.put(record.partitionId(), newPartInfo);
434               brokersToIsrs.update(record.topicId(), record.partitionId(), null,
435                   newPartInfo.isr, NO_LEADER, newPartInfo.leader);
436               globalPartitionCount.increment();
437               controllerMetrics.setGlobalPartitionCount(globalPartitionCount.get());
438               updateReassigningTopicsIfNeeded(record.topicId(), record.partitionId(),
439                       false,  newPartInfo.isReassigning());
440           } else if (!newPartInfo.equals(prevPartInfo)) {
441               newPartInfo.maybeLogPartitionChange(log, description, prevPartInfo);

645                   return new ApiError(INVALID_REQUEST,
646                       "A manual partition assignment was specified, but numPartitions " +
647                           "was not set to -1.");

650               for (CreatableReplicaAssignment assignment : topic.assignments()) {

672                       Replicas.toList(e.getValue().replicas)));

674                       topic.name(), null, null, assignments, creationConfigs);
675               });

679                   "Replication factor must be larger than 0, or -1 to use the default value.");

687                   defaultReplicationFactor : topic.replicationFactor();

690                       0,
691                       numPartitions,
692                       replicationFactor

697                           filter(clusterControl::active).collect(Collectors.toList());
