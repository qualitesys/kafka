40   /**
41    * An immutable class which represents broker registrations.
42    */
43   public class BrokerRegistration {
44       private static Map<String, Endpoint> listenersToMap(Collection<Endpoint> listeners) {
45           Map<String, Endpoint> listenersMap = new HashMap<>();
46           for (Endpoint endpoint : listeners) {
47               listenersMap.put(endpoint.listenerName().get(), endpoint);
48           }
49           return listenersMap;
50       }

56       private final Map<String, VersionRange> supportedFeatures;
57       private final Optional<String> rack;

61       public BrokerRegistration(int id,

64                                 List<Endpoint> listeners,

66                                 Optional<String> rack,
67                                 boolean fenced,

73       public BrokerRegistration(int id,
74                                 long epoch,
75                                 Uuid incarnationId,
76                                 Map<String, Endpoint> listeners,
77                                 Map<String, VersionRange> supportedFeatures,
78                                 Optional<String> rack,
79                                 boolean fenced,
80                                 boolean inControlledShutdown) {
81           this.id = id;
82           this.epoch = epoch;
83           this.incarnationId = incarnationId;
84           Map<String, Endpoint> newListeners = new HashMap<>(listeners.size());
85           for (Entry<String, Endpoint> entry : listeners.entrySet()) {
86               if (!entry.getValue().listenerName().isPresent()) {
87                   throw new IllegalArgumentException("Broker listeners must be named.");
88               }
89               newListeners.put(entry.getKey(), entry.getValue());
90           }
91           this.listeners = Collections.unmodifiableMap(newListeners);
92           Objects.requireNonNull(supportedFeatures);
93           this.supportedFeatures = new HashMap<>(supportedFeatures);
94           Objects.requireNonNull(rack);
95           this.rack = rack;
96           this.fenced = fenced;
97           this.inControlledShutdown = inControlledShutdown;
98       }

100       public static BrokerRegistration fromRecord(RegisterBrokerRecord record) {
101           Map<String, Endpoint> listeners = new HashMap<>();
102           for (BrokerEndpoint endpoint : record.endPoints()) {
103               listeners.put(endpoint.name(), new Endpoint(endpoint.name(),
104                   SecurityProtocol.forId(endpoint.securityProtocol()),
105                   endpoint.host(),
106                   endpoint.port()));
107           }
108           Map<String, VersionRange> supportedFeatures = new HashMap<>();
109           for (BrokerFeature feature : record.features()) {
110               supportedFeatures.put(feature.name(), VersionRange.of(
111                   feature.minSupportedVersion(), feature.maxSupportedVersion()));
112           }
113           return new BrokerRegistration(record.brokerId(),
114               record.brokerEpoch(),
115               record.incarnationId(),
116               listeners,
117               supportedFeatures,
118               Optional.ofNullable(record.rack()),
119               record.fenced(),
120               record.inControlledShutdown());
121       }

139       public Optional<Node> node(String listenerName) {
140           Endpoint endpoint = listeners().get(listenerName);
141           if (endpoint == null) {
142               return Optional.empty();
143           }
144           return Optional.of(new Node(id, endpoint.host(), endpoint.port(), rack.orElse(null)));
145       }

163       public ApiMessageAndVersion toRecord(MetadataVersion metadataVersion) {
164           RegisterBrokerRecord registrationRecord = new RegisterBrokerRecord().
165               setBrokerId(id).
166               setRack(rack.orElse(null)).
167               setBrokerEpoch(epoch).
168               setIncarnationId(incarnationId).
169               setFenced(fenced);
170   
171           if (metadataVersion.isInControlledShutdownStateSupported()) {
172               registrationRecord.setInControlledShutdown(inControlledShutdown);
173           }
174   
175           for (Entry<String, Endpoint> entry : listeners.entrySet()) {
176               Endpoint endpoint = entry.getValue();
177               registrationRecord.endPoints().add(new BrokerEndpoint().
178                   setName(entry.getKey()).
179                   setHost(endpoint.host()).
180                   setPort(endpoint.port()).
181                   setSecurityProtocol(endpoint.securityProtocol().id));
182           }
183   
184           for (Entry<String, VersionRange> entry : supportedFeatures.entrySet()) {
185               registrationRecord.features().add(new BrokerFeature().
186                   setName(entry.getKey()).
187                   setMinSupportedVersion(entry.getValue().min()).
188                   setMaxSupportedVersion(entry.getValue().max()));
189           }
190   
191           return new ApiMessageAndVersion(registrationRecord,
192               metadataVersion.registerBrokerRecordVersion());
193       }

201       @Override
202       public boolean equals(Object o) {
203           if (!(o instanceof BrokerRegistration)) return false;
204           BrokerRegistration other = (BrokerRegistration) o;
205           return other.id == id &&
206               other.epoch == epoch &&
207               other.incarnationId.equals(incarnationId) &&
208               other.listeners.equals(listeners) &&
209               other.supportedFeatures.equals(supportedFeatures) &&
210               other.rack.equals(rack) &&
211               other.fenced == fenced &&
212               other.inControlledShutdown == inControlledShutdown;
213       }

215       @Override
216       public String toString() {
217           StringBuilder bld = new StringBuilder();
218           bld.append("BrokerRegistration(id=").append(id);
219           bld.append(", epoch=").append(epoch);
220           bld.append(", incarnationId=").append(incarnationId);
221           bld.append(", listeners=[").append(
222               listeners.keySet().stream().sorted().
223                   map(n -> listeners.get(n).toString()).
224                   collect(Collectors.joining(", ")));
225           bld.append("], supportedFeatures={").append(
226               supportedFeatures.entrySet().stream().sorted().
227                   map(e -> e.getKey() + ": " + e.getValue()).
228                   collect(Collectors.joining(", ")));
229           bld.append("}");
230           bld.append(", rack=").append(rack);
231           bld.append(", fenced=").append(fenced);
232           bld.append(", inControlledShutdown=").append(inControlledShutdown);
233           bld.append(")");
234           return bld.toString();
235       }

237       public BrokerRegistration cloneWith(
238           Optional<Boolean> fencingChange,
239           Optional<Boolean> inControlledShutdownChange
240       ) {
241           boolean newFenced = fencingChange.orElse(fenced);
242           boolean newInControlledShutdownChange = inControlledShutdownChange.orElse(inControlledShutdown);
243   
244           if (newFenced == fenced && newInControlledShutdownChange == inControlledShutdown)
245               return this;
246   
247           return new BrokerRegistration(
248               id,
249               epoch,
250               incarnationId,
251               listeners,
252               supportedFeatures,
253               rack,
254               newFenced,
255               newInControlledShutdownChange
256           );
257       }
