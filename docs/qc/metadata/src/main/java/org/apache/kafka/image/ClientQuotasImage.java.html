48   /**
49    * Represents the client quotas in the metadata image.
50    *
51    * This class is thread-safe.
52    */
53   public final class ClientQuotasImage {
54       public final static ClientQuotasImage EMPTY = new ClientQuotasImage(Collections.emptyMap());

60       }

70       public void write(Consumer<List<ApiMessageAndVersion>> out) {
71           for (Entry<ClientQuotaEntity, ClientQuotaImage> entry : entities.entrySet()) {
72               ClientQuotaEntity entity = entry.getKey();
73               ClientQuotaImage clientQuotaImage = entry.getValue();
74               clientQuotaImage.write(entity, out);
75           }
76       }

78       public DescribeClientQuotasResponseData describe(DescribeClientQuotasRequestData request) {
79           DescribeClientQuotasResponseData response = new DescribeClientQuotasResponseData();
80           Map<String, String> exactMatch = new HashMap<>();
81           Set<String> typeMatch = new HashSet<>();
82           for (DescribeClientQuotasRequestData.ComponentData component : request.components()) {
83               if (component.entityType().isEmpty()) {
84                   throw new InvalidRequestException("Invalid empty entity type.");
85               } else if (exactMatch.containsKey(component.entityType()) ||
86                       typeMatch.contains(component.entityType())) {
87                   throw new InvalidRequestException("Entity type " + component.entityType() +
88                       " cannot appear more than once in the filter.");
89               }
90               if (!(component.entityType().equals(IP) || component.entityType().equals(USER) ||
91                       component.entityType().equals(CLIENT_ID))) {
92                   throw new UnsupportedVersionException("Unsupported entity type " +
93                           component.entityType());
94               }
95               switch (component.matchType()) {
96                   case MATCH_TYPE_EXACT:
97                       if (component.match() == null) {
98                           throw new InvalidRequestException("Request specified " +
99                               "MATCH_TYPE_EXACT, but set match string to null.");
100                       }
101                       exactMatch.put(component.entityType(), component.match());
102                       break;
103                   case MATCH_TYPE_DEFAULT:
104                       if (component.match() != null) {
105                           throw new InvalidRequestException("Request specified " +
106                               "MATCH_TYPE_DEFAULT, but also specified a match string.");
107                       }
108                       exactMatch.put(component.entityType(), null);
109                       break;
110                   case MATCH_TYPE_SPECIFIED:
111                       if (component.match() != null) {
112                           throw new InvalidRequestException("Request specified " +
113                               "MATCH_TYPE_SPECIFIED, but also specified a match string.");
114                       }
115                       typeMatch.add(component.entityType());
116                       break;
117                   default:
118                       throw new InvalidRequestException("Unknown match type " + component.matchType());
119               }
120           }
121           if (exactMatch.containsKey(IP) || typeMatch.contains(IP)) {
122               if ((exactMatch.containsKey(USER) || typeMatch.contains(USER)) ||
123                       (exactMatch.containsKey(CLIENT_ID) || typeMatch.contains(CLIENT_ID))) {
124                   throw new InvalidRequestException("Invalid entity filter component " +
125                       "combination. IP filter component should not be used with " +
126                       "user or clientId filter component.");
127               }
128           }
129           // TODO: this is O(N). We should add indexing here to speed it up. See KAFKA-13022.
130           for (Entry<ClientQuotaEntity, ClientQuotaImage> entry : entities.entrySet()) {
131               ClientQuotaEntity entity = entry.getKey();
132               ClientQuotaImage quotaImage = entry.getValue();
133               if (matches(entity, exactMatch, typeMatch, request.strict())) {
134                   response.entries().add(toDescribeEntry(entity, quotaImage));
135               }
136           }
137           return response;
138       }
139   
140       private static boolean matches(ClientQuotaEntity entity,
141                                      Map<String, String> exactMatch,
142                                      Set<String> typeMatch,
143                                      boolean strict) {
144           if (strict) {
145               if (entity.entries().size() != exactMatch.size() + typeMatch.size()) {
146                   return false;
147               }
148           }
149           for (Entry<String, String> entry : exactMatch.entrySet()) {
150               if (!entity.entries().containsKey(entry.getKey())) {
151                   return false;
152               }
153               if (!Objects.equals(entity.entries().get(entry.getKey()), entry.getValue())) {
154                   return false;
155               }
156           }
157           for (String type : typeMatch) {
158               if (!entity.entries().containsKey(type)) {
159                   return false;
160               }
161           }
162           return true;
163       }

165       private static EntryData toDescribeEntry(ClientQuotaEntity entity,
166                                                ClientQuotaImage quotaImage) {
167           EntryData data = new EntryData();
168           for (Entry<String, String> entry : entity.entries().entrySet()) {
169               data.entity().add(new EntityData().
170                   setEntityType(entry.getKey()).
171                   setEntityName(entry.getValue()));
172           }
173           data.setValues(quotaImage.toDescribeValues());
174           return data;
175       }
