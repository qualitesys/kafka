37   /**
38    * Represents changes to the topics in the metadata image.
39    */
40   public final class TopicsDelta {
41       private final TopicsImage image;
42   
43       /**
44        * A map from topic IDs to the topic deltas for each topic. Topics which have been
45        * deleted will not appear in this map.
46        */
47       private final Map<Uuid, TopicDelta> changedTopics = new HashMap<>();
48   
49       /**
50        * The IDs of topics that exist in the image but that have been deleted. Note that if
51        * a topic does not exist in the image, it will also not exist in this set. Topics
52        * that are created and then deleted within the same delta will leave no trace.
53        */
54       private final Set<Uuid> deletedTopicIds = new HashSet<>();
55   
56       public TopicsDelta(TopicsImage image) {
57           this.image = image;
58       }
59   
60       public TopicsImage image() {
61           return image;
62       }
63   
64       public Map<Uuid, TopicDelta> changedTopics() {
65           return changedTopics;
66       }
67   
68       public void replay(TopicRecord record) {
69           TopicDelta delta = new TopicDelta(
70               new TopicImage(record.name(), record.topicId(), Collections.emptyMap()));
71           changedTopics.put(record.topicId(), delta);
72       }
73   
74       TopicDelta getOrCreateTopicDelta(Uuid id) {
75           TopicDelta topicDelta = changedTopics.get(id);
76           if (topicDelta == null) {
77               topicDelta = new TopicDelta(image.getTopic(id));
78               changedTopics.put(id, topicDelta);
79           }
80           return topicDelta;
81       }
82   
83       public void replay(PartitionRecord record) {
84           TopicDelta topicDelta = getOrCreateTopicDelta(record.topicId());
85           topicDelta.replay(record);
86       }
87   
88       public void replay(PartitionChangeRecord record) {
89           TopicDelta topicDelta = getOrCreateTopicDelta(record.topicId());
90           topicDelta.replay(record);
91       }
92   
93       public String replay(RemoveTopicRecord record) {
94           TopicDelta topicDelta = changedTopics.remove(record.topicId());
95           String topicName;
96           if (topicDelta != null) {
97               topicName = topicDelta.image().name();
98               if (image.topicsById().containsKey(record.topicId())) {
99                   deletedTopicIds.add(record.topicId());
100               }
101           } else {
102               TopicImage topicImage = image.getTopic(record.topicId());
103               if (topicImage == null) {
104                   throw new RuntimeException("Unable to delete topic with id " +
105                       record.topicId() + ": no such topic found.");
106               }
107               topicName = topicImage.name();
108               deletedTopicIds.add(record.topicId());
109           }
110           return topicName;
111       }
112   
113       public void finishSnapshot() {
114           for (Uuid topicId : image.topicsById().keySet()) {
115               if (!changedTopics.containsKey(topicId)) {
116                   deletedTopicIds.add(topicId);
117               }
118           }
119       }
120   
121       public void handleMetadataVersionChange(MetadataVersion newVersion) {
122           // no-op
123       }
124   
125       public TopicsImage apply() {
126           Map<Uuid, TopicImage> newTopicsById = new HashMap<>(image.topicsById().size());
127           Map<String, TopicImage> newTopicsByName = new HashMap<>(image.topicsByName().size());
128           for (Entry<Uuid, TopicImage> entry : image.topicsById().entrySet()) {
129               Uuid id = entry.getKey();
130               TopicImage prevTopicImage = entry.getValue();
131               TopicDelta delta = changedTopics.get(id);
132               if (delta == null) {
133                   if (!deletedTopicIds.contains(id)) {
134                       newTopicsById.put(id, prevTopicImage);
135                       newTopicsByName.put(prevTopicImage.name(), prevTopicImage);
136                   }
137               } else {
138                   TopicImage newTopicImage = delta.apply();
139                   newTopicsById.put(id, newTopicImage);
140                   newTopicsByName.put(delta.name(), newTopicImage);
141               }
142           }
143           for (Entry<Uuid, TopicDelta> entry : changedTopics.entrySet()) {
144               if (!newTopicsById.containsKey(entry.getKey())) {
145                   TopicImage newTopicImage = entry.getValue().apply();
146                   newTopicsById.put(newTopicImage.id(), newTopicImage);
147                   newTopicsByName.put(newTopicImage.name(), newTopicImage);
148               }
149           }
150           return new TopicsImage(newTopicsById, newTopicsByName);
151       }
152   
153       public TopicDelta changedTopic(Uuid topicId) {
154           return changedTopics.get(topicId);
155       }
156   
157       /**
158        * Returns true if the topic with the given name was deleted. Note: this will return
159        * true even if a new topic with the same name was subsequently created.
160        */
161       public boolean topicWasDeleted(String topicName) {
162           TopicImage topicImage = image.getTopic(topicName);
163           if (topicImage == null) {
164               return false;
165           }
166           return deletedTopicIds.contains(topicImage.id());
167       }
168   
169       public Set<Uuid> deletedTopicIds() {
170           return deletedTopicIds;
171       }
172   
173       /**
174        * Find the topic partitions that have change based on the replica given.
175        *
176        * The changes identified are:
177        *   1. topic partitions for which the broker is not a replica anymore
178        *   2. topic partitions for which the broker is now the leader
179        *   3. topic partitions for which the broker is now a follower
180        *
181        * @param brokerId the broker id
182        * @return the list of topic partitions which the broker should remove, become leader or become follower.
183        */
184       public LocalReplicaChanges localChanges(int brokerId) {
185           Set<TopicPartition> deletes = new HashSet<>();
186           Map<TopicPartition, LocalReplicaChanges.PartitionInfo> leaders = new HashMap<>();
187           Map<TopicPartition, LocalReplicaChanges.PartitionInfo> followers = new HashMap<>();
188   
189           for (TopicDelta delta : changedTopics.values()) {
190               LocalReplicaChanges changes = delta.localChanges(brokerId);
191   
192               deletes.addAll(changes.deletes());
193               leaders.putAll(changes.leaders());
194               followers.putAll(changes.followers());
195           }
196   
197           // Add all of the removed topic partitions to the set of locally removed partitions
198           deletedTopicIds().forEach(topicId -> {
199               TopicImage topicImage = image().getTopic(topicId);
200               topicImage.partitions().forEach((partitionId, prevPartition) -> {
201                   if (Replicas.contains(prevPartition.replicas, brokerId)) {
202                       deletes.add(new TopicPartition(topicImage.name(), partitionId));
203                   }
204               });
205           });
206   
207           return new LocalReplicaChanges(deletes, leaders, followers);
208       }
209   
210       @Override
211       public String toString() {
212           return "TopicsDelta(" +
213               "changedTopics=" + changedTopics +
214               ", deletedTopicIds=" + deletedTopicIds +
215               ')';
216       }
217   }
