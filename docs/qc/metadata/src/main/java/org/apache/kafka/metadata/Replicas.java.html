33       /**
34        * Convert an array of integers to a list of ints.
35        *
36        * @param array         The input array.
37        * @return              The output list.
38        */
39       public static List<Integer> toList(int[] array) {
40           if (array == null) return null;
41           ArrayList<Integer> list = new ArrayList<>(array.length);
42           for (int i = 0; i < array.length; i++) {
43               list.add(array[i]);
44           }
45           return list;
46       }

75       /**
76        * Check that a replica set is valid.
77        *
78        * @param replicas      The replica set.
79        * @return              True if none of the replicas are negative, and there are no
80        *                      duplicates.
81        */
82       public static boolean validate(int[] replicas) {
83           if (replicas.length == 0) return true;
84           int[] sortedReplicas = clone(replicas);
85           Arrays.sort(sortedReplicas);
86           int prev = sortedReplicas[0];
87           if (prev < 0) return false;
88           for (int i = 1; i < sortedReplicas.length; i++) {
89               int replica = sortedReplicas[i];
90               if (prev == replica) return false;
91               prev = replica;
92           }
93           return true;
94       }

96       /**
97        * Check that an isr set is valid.
98        *
99        * @param replicas      The replica set.
100        * @param isr           The in-sync replica set.
101        * @return              True if none of the in-sync replicas are negative, there are
102        *                      no duplicates, and all in-sync replicas are also replicas.
103        */
104       public static boolean validateIsr(int[] replicas, int[] isr) {
105           if (isr.length == 0) return true;
106           if (replicas.length == 0) return false;
107           int[] sortedReplicas = clone(replicas);
108           Arrays.sort(sortedReplicas);
109           int[] sortedIsr = clone(isr);
110           Arrays.sort(sortedIsr);
111           int j = 0;
112           if (sortedIsr[0] < 0) return false;
113           int prevIsr = -1;
114           for (int i = 0; i < sortedIsr.length; i++) {
115               int curIsr = sortedIsr[i];
116               if (prevIsr == curIsr) return false;
117               prevIsr = curIsr;
118               while (true) {
119                   if (j == sortedReplicas.length) return false;
120                   int curReplica = sortedReplicas[j++];
121                   if (curReplica == curIsr) break;
122               }
123           }
124           return true;
125       }

127       /**
128        * Returns true if an array of replicas contains a specific value.
129        *
130        * @param replicas      The replica array.
131        * @param value         The value to look for.
132        *
133        * @return              True only if the value is found in the array.
134        */
135       public static boolean contains(int[] replicas, int value) {
136           for (int i = 0; i < replicas.length; i++) {
137               if (replicas[i] == value) return true;
138           }
139           return false;
140       }

142       /**
143        * Check if the first list of integers contains the second.
144        *
145        * @param a             The first list
146        * @param b             The second list
147        *
148        * @return              True only if the first contains the second.
149        */
150       public static boolean contains(List<Integer> a, int[] b) {
151           List<Integer> aSorted = new ArrayList<>(a);
152           aSorted.sort(Integer::compareTo);
153           List<Integer> bSorted = Replicas.toList(b);
154           bSorted.sort(Integer::compareTo);
155           int i = 0;
156           for (int replica : bSorted) {
157               while (true) {
158                   if (i >= aSorted.size()) return false;
159                   int replica2 = aSorted.get(i++);
160                   if (replica2 == replica) break;
161                   if (replica2 > replica) return false;
162               }
163           }
164           return true;
165       }

167       /**
168        * Copy a replica array without any occurrences of the given value.
169        *
170        * @param replicas      The replica array.
171        * @param value         The value to filter out.
172        *
173        * @return              A new array without the given value.
174        */
175       public static int[] copyWithout(int[] replicas, int value) {
176           int size = 0;
177           for (int i = 0; i < replicas.length; i++) {
178               if (replicas[i] != value) {
179                   size++;
180               }
181           }
182           int[] result = new int[size];
183           int j = 0;
184           for (int i = 0; i < replicas.length; i++) {
185               int replica = replicas[i];
186               if (replica != value) {
187                   result[j++] = replica;
188               }
189           }
190           return result;
191       }

193       /**
194        * Copy a replica array without any occurrences of the given values.
195        *
196        * @param replicas      The replica array.
197        * @param values        The values to filter out.
198        *
199        * @return              A new array without the given value.
200        */
201       public static int[] copyWithout(int[] replicas, int[] values) {
202           int size = 0;
203           for (int i = 0; i < replicas.length; i++) {
204               if (!Replicas.contains(values, replicas[i])) {
205                   size++;
206               }
207           }
208           int[] result = new int[size];
209           int j = 0;
210           for (int i = 0; i < replicas.length; i++) {
211               int replica = replicas[i];
212               if (!Replicas.contains(values, replica)) {
213                   result[j++] = replica;
214               }
215           }
216           return result;
217       }

221        *
222        * @param replicas      The replica array.
223        * @param value         The value to add.
224        *
225        * @return              A new array with the given value.

227       public static int[] copyWith(int[] replicas, int value) {
228           int[] newReplicas = new int[replicas.length + 1];
229           System.arraycopy(replicas, 0, newReplicas, 0, replicas.length);
230           newReplicas[newReplicas.length - 1] = value;
231           return newReplicas;
232       }
