51   public class ClientQuotaControlManager {
52       private final SnapshotRegistry snapshotRegistry;

54       final TimelineHashMap<ClientQuotaEntity, TimelineHashMap<String, Double>> clientQuotaData;

56       ClientQuotaControlManager(SnapshotRegistry snapshotRegistry) {
57           this.snapshotRegistry = snapshotRegistry;
58           this.clientQuotaData = new TimelineHashMap<>(snapshotRegistry, 0);
59       }

61       /**
62        * Determine the result of applying a batch of client quota alteration.  Note
63        * that this method does not change the contents of memory.  It just generates a
64        * result, that you can replay later if you wish using replay().
65        *
66        * @param quotaAlterations  List of client quota alterations to evaluate
67        * @return                  The result.
68        */
69       ControllerResult<Map<ClientQuotaEntity, ApiError>> alterClientQuotas(
70               Collection<ClientQuotaAlteration> quotaAlterations) {
71           List<ApiMessageAndVersion> outputRecords = new ArrayList<>();
72           Map<ClientQuotaEntity, ApiError> outputResults = new HashMap<>();
73   
74           quotaAlterations.forEach(quotaAlteration -> {
75               // Note that the values in this map may be null
76               Map<String, Double> alterations = new HashMap<>(quotaAlteration.ops().size());
77               quotaAlteration.ops().forEach(op -> {
78                   if (alterations.containsKey(op.key())) {
79                       outputResults.put(quotaAlteration.entity(), ApiError.fromThrowable(
80                               new InvalidRequestException("Duplicate quota key " + op.key() +
81                                   " not updating quota for this entity " + quotaAlteration.entity())));
82                   } else {
83                       alterations.put(op.key(), op.value());
84                   }
85               });
86               if (outputResults.containsKey(quotaAlteration.entity())) {
87                   outputResults.put(quotaAlteration.entity(), ApiError.fromThrowable(
88                           new InvalidRequestException("Ignoring duplicate entity " + quotaAlteration.entity())));
89               } else {
90                   alterClientQuotaEntity(quotaAlteration.entity(), alterations, outputRecords, outputResults);
91               }
92           });
93   
94           return ControllerResult.atomicOf(outputRecords, outputResults);
95       }

97       /**
98        * Apply a quota record to the in-memory state.
99        *
100        * @param record    A ClientQuotaRecord instance.
101        */
102       public void replay(ClientQuotaRecord record) {
103           Map<String, String> entityMap = new HashMap<>(2);
104           record.entity().forEach(entityData -> entityMap.put(entityData.entityType(), entityData.entityName()));
105           ClientQuotaEntity entity = new ClientQuotaEntity(entityMap);
106           TimelineHashMap<String, Double> quotas = clientQuotaData.get(entity);
107           if (quotas == null) {
108               quotas = new TimelineHashMap<>(snapshotRegistry, 0);
109               clientQuotaData.put(entity, quotas);
110           }
111           if (record.remove()) {
112               quotas.remove(record.key());
113               if (quotas.size() == 0) {
114                   clientQuotaData.remove(entity);
115               }
116           } else {
117               quotas.put(record.key(), record.value());
118           }
119       }

121       private void alterClientQuotaEntity(
122               ClientQuotaEntity entity,
123               Map<String, Double> newQuotaConfigs,
124               List<ApiMessageAndVersion> outputRecords,
125               Map<ClientQuotaEntity, ApiError> outputResults) {
126   
127           // Check entity types and sanitize the names
128           Map<String, String> validatedEntityMap = new HashMap<>(3);
129           ApiError error = validateEntity(entity, validatedEntityMap);
130           if (error.isFailure()) {
131               outputResults.put(entity, error);
132               return;
133           }
134   
135           // Check the combination of entity types and get the config keys
136           Map<String, ConfigDef.ConfigKey> configKeys = new HashMap<>(4);
137           error = configKeysForEntityType(validatedEntityMap, configKeys);
138           if (error.isFailure()) {
139               outputResults.put(entity, error);
140               return;
141           }
142   
143           // Don't share objects between different records
144           Supplier<List<EntityData>> recordEntitySupplier = () ->
145                   validatedEntityMap.entrySet().stream().map(mapEntry -> new EntityData()
146                           .setEntityType(mapEntry.getKey())
147                           .setEntityName(mapEntry.getValue()))
148                           .collect(Collectors.toList());
149   
150           List<ApiMessageAndVersion> newRecords = new ArrayList<>(newQuotaConfigs.size());
151           Map<String, Double> currentQuotas = clientQuotaData.containsKey(entity) ?
152                   clientQuotaData.get(entity) : Collections.emptyMap();
153           for (Map.Entry<String, Double> entry : newQuotaConfigs.entrySet()) {
154               String key = entry.getKey();
155               Double newValue = entry.getValue();
156               if (newValue == null) {
157                   if (currentQuotas.containsKey(key)) {
158                       // Null value indicates removal
159                       newRecords.add(new ApiMessageAndVersion(new ClientQuotaRecord()
160                               .setEntity(recordEntitySupplier.get())
161                               .setKey(key)
162                               .setRemove(true),
163                           CLIENT_QUOTA_RECORD.highestSupportedVersion()));
164                   }
165               } else {
166                   ApiError validationError = validateQuotaKeyValue(configKeys, key, newValue);
167                   if (validationError.isFailure()) {
168                       outputResults.put(entity, validationError);
169                       return;
170                   } else {
171                       final Double currentValue = currentQuotas.get(key);
172                       if (!Objects.equals(currentValue, newValue)) {
173                           // Only record the new value if it has changed
174                           newRecords.add(new ApiMessageAndVersion(new ClientQuotaRecord()
175                                   .setEntity(recordEntitySupplier.get())
176                                   .setKey(key)
177                                   .setValue(newValue),
178                               CLIENT_QUOTA_RECORD.highestSupportedVersion()));
179                       }
180                   }
181               }
182           }
183   
184           outputRecords.addAll(newRecords);
185           outputResults.put(entity, ApiError.NONE);
186       }

188       private ApiError configKeysForEntityType(Map<String, String> entity, Map<String, ConfigDef.ConfigKey> output) {
189           // We only allow certain combinations of quota entity types. Which type is in use determines which config
190           // keys are valid
191           boolean hasUser = entity.containsKey(ClientQuotaEntity.USER);
192           boolean hasClientId = entity.containsKey(ClientQuotaEntity.CLIENT_ID);
193           boolean hasIp = entity.containsKey(ClientQuotaEntity.IP);
194   
195           final Map<String, ConfigDef.ConfigKey> configKeys;
196           if (hasIp) {
197               if (hasUser || hasClientId) {
198                   return new ApiError(Errors.INVALID_REQUEST, "Invalid quota entity combination, IP entity should" +
199                       "not be combined with User or ClientId");
200               } else {
201                   if (isValidIpEntity(entity.get(ClientQuotaEntity.IP))) {
202                       configKeys = QuotaConfigs.ipConfigs().configKeys();
203                   } else {
204                       return new ApiError(Errors.INVALID_REQUEST, entity.get(ClientQuotaEntity.IP) + " is not a valid IP or resolvable host.");
205                   }
206               }
207           } else if (hasUser && hasClientId) {
208               configKeys = QuotaConfigs.userConfigs().configKeys();
209           } else if (hasUser) {
210               configKeys = QuotaConfigs.userConfigs().configKeys();
211           } else if (hasClientId) {
212               configKeys = QuotaConfigs.clientConfigs().configKeys();
213           } else {
214               return new ApiError(Errors.INVALID_REQUEST, "Invalid empty client quota entity");
215           }
216   
217           output.putAll(configKeys);
218           return ApiError.NONE;
219       }

221       private ApiError validateQuotaKeyValue(Map<String, ConfigDef.ConfigKey> validKeys, String key, Double value) {
222           // TODO can this validation be shared with alter configs?
223           // Ensure we have an allowed quota key
224           ConfigDef.ConfigKey configKey = validKeys.get(key);
225           if (configKey == null) {
226               return new ApiError(Errors.INVALID_REQUEST, "Invalid configuration key " + key);
227           }
228   
229           // Ensure the quota value is valid
230           switch (configKey.type()) {
231               case DOUBLE:
232                   break;
233               case SHORT:
234               case INT:
235               case LONG:
236                   Double epsilon = 1e-6;
237                   Long longValue = Double.valueOf(value + epsilon).longValue();
238                   if (Math.abs(longValue.doubleValue() - value) > epsilon) {
239                       return new ApiError(Errors.INVALID_REQUEST,
240                               "Configuration " + key + " must be a Long value");
241                   }
242                   break;
243               default:
244                   return new ApiError(Errors.UNKNOWN_SERVER_ERROR,
245                           "Unexpected config type " + configKey.type() + " should be Long or Double");
246           }
247           return ApiError.NONE;
248       }

250       // TODO move this somewhere common?
251       private boolean isValidIpEntity(String ip) {
252           if (Objects.nonNull(ip)) {
253               try {
254                   InetAddress.getByName(ip);
255                   return true;
256               } catch (UnknownHostException e) {
257                   return false;
258               }
259           } else {
260               return true;
261           }
262       }

264       private ApiError validateEntity(ClientQuotaEntity entity, Map<String, String> validatedEntityMap) {
265           // Given a quota entity (which is a mapping of entity type to entity name), validate it's types
266           if (entity.entries().isEmpty()) {
267               return new ApiError(Errors.INVALID_REQUEST, "Invalid empty client quota entity");
268           }
269   
270           for (Entry<String, String> entityEntry : entity.entries().entrySet()) {
271               String entityType = entityEntry.getKey();
272               String entityName = entityEntry.getValue();
273               if (validatedEntityMap.containsKey(entityType)) {
274                   return new ApiError(Errors.INVALID_REQUEST, "Invalid client quota entity, duplicate entity entry " + entityType);
275               }
276               if (Objects.equals(entityType, ClientQuotaEntity.USER)) {
277                   validatedEntityMap.put(ClientQuotaEntity.USER, entityName);
278               } else if (Objects.equals(entityType, ClientQuotaEntity.CLIENT_ID)) {
279                   validatedEntityMap.put(ClientQuotaEntity.CLIENT_ID, entityName);
280               } else if (Objects.equals(entityType, ClientQuotaEntity.IP)) {
281                   validatedEntityMap.put(ClientQuotaEntity.IP, entityName);
282               } else {
283                   return new ApiError(Errors.INVALID_REQUEST, "Unhandled client quota entity type: " + entityType);
284               }
285   
286               if (entityName != null && entityName.isEmpty()) {
287                   return new ApiError(Errors.INVALID_REQUEST, "Empty " + entityType + " not supported");
288               }
289           }
290   
291           return ApiError.NONE;
292       }

294       class ClientQuotaControlIterator implements Iterator<List<ApiMessageAndVersion>> {
295           private final long epoch;
296           private final Iterator<Entry<ClientQuotaEntity, TimelineHashMap<String, Double>>> iterator;
297   
298           ClientQuotaControlIterator(long epoch) {
299               this.epoch = epoch;
300               this.iterator = clientQuotaData.entrySet(epoch).iterator();
301           }
302   
303           @Override
304           public boolean hasNext() {
305               return iterator.hasNext();
306           }
307   
308           @Override
309           public List<ApiMessageAndVersion> next() {
310               if (!hasNext()) throw new NoSuchElementException();
311               Entry<ClientQuotaEntity, TimelineHashMap<String, Double>> entry = iterator.next();
312               ClientQuotaEntity entity = entry.getKey();
313               List<ApiMessageAndVersion> records = new ArrayList<>();
314               for (Entry<String, Double> quotaEntry : entry.getValue().entrySet(epoch)) {
315                   ClientQuotaRecord record = new ClientQuotaRecord();
316                   for (Entry<String, String> entityEntry : entity.entries().entrySet()) {
317                       record.entity().add(new EntityData().
318                           setEntityType(entityEntry.getKey()).
319                           setEntityName(entityEntry.getValue()));
320                   }
321                   record.setKey(quotaEntry.getKey());
322                   record.setValue(quotaEntry.getValue());
323                   record.setRemove(false);
324                   records.add(new ApiMessageAndVersion(record,
325                       CLIENT_QUOTA_RECORD.highestSupportedVersion()));
326               }
327               return records;
328           }
329       }
330   
331       ClientQuotaControlIterator iterator(long epoch) {
332           return new ClientQuotaControlIterator(epoch);
333       }
