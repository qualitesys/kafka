18   package org.apache.kafka.timeline;
19   
20   import java.util.ArrayList;
21   import java.util.HashMap;
22   import java.util.Iterator;
23   import java.util.List;
24   import java.util.stream.Collectors;
25   
26   import org.apache.kafka.common.utils.LogContext;
27   import org.slf4j.Logger;
28

30   /**
31    * A registry containing snapshots of timeline data structures.
32    * We generally expect a small number of snapshots-- perhaps 1 or 2 at a time.
33    * Therefore, we use ArrayLists here rather than a data structure with higher overhead.
34    */
35   public class SnapshotRegistry {
36       public final static long LATEST_EPOCH = Long.MAX_VALUE;
37   
38       /**
39        * Iterate through the list of snapshots in order of creation, such that older
40        * snapshots come first.
41        */
42       class SnapshotIterator implements Iterator<Snapshot> {
43           Snapshot cur;
44           Snapshot result = null;
45   
46           SnapshotIterator(Snapshot start) {
47               cur = start;
48           }
49   
50           @Override
51           public boolean hasNext() {
52               return cur != head;
53           }
54   
55           @Override
56           public Snapshot next() {
57               result = cur;
58               cur = cur.next();
59               return result;
60           }
61   
62           @Override
63           public void remove() {
64               if (result == null) {
65                   throw new IllegalStateException();
66               }
67               deleteSnapshot(result);
68               result = null;
69           }
70       }
71   
72       /**
73        * Iterate through the list of snapshots in reverse order of creation, such that
74        * the newest snapshot is first.
75        */
76       class ReverseSnapshotIterator implements Iterator<Snapshot> {
77           Snapshot cur;
78   
79           ReverseSnapshotIterator() {
80               cur = head.prev();
81           }
82   
83           @Override
84           public boolean hasNext() {
85               return cur != head;
86           }
87   
88           @Override
89           public Snapshot next() {
90               Snapshot result = cur;
91               cur = cur.prev();
92               return result;
93           }
94       }
95   
96       private final Logger log;
97   
98       /**
99        * A map from snapshot epochs to snapshot data structures.
100        */
101       private final HashMap<Long, Snapshot> snapshots = new HashMap<>();
102   
103       /**
104        * The head of a list of snapshots, sorted by epoch.
105        */
106       private final Snapshot head = new Snapshot(Long.MIN_VALUE);
107   
108       /**
109        * Collection of all Revertable registered with this registry
110        */
111       private final List<Revertable> revertables = new ArrayList<>();
112   
113       public SnapshotRegistry(LogContext logContext) {
114           this.log = logContext.logger(SnapshotRegistry.class);
115       }
116   
117       /**
118        * Returns a snapshot iterator that iterates from the snapshots with the
119        * lowest epoch to those with the highest.
120        */
121       public Iterator<Snapshot> iterator() {
122           return new SnapshotIterator(head.next());
123       }
124   
125       /**
126        * Returns a snapshot iterator that iterates from the snapshots with the
127        * lowest epoch to those with the highest, starting at the snapshot with the
128        * given epoch.
129        */
130       public Iterator<Snapshot> iterator(long epoch) {
131           return iterator(getSnapshot(epoch));
132       }
133   
134       /**
135        * Returns a snapshot iterator that iterates from the snapshots with the
136        * lowest epoch to those with the highest, starting at the given snapshot.
137        */
138       public Iterator<Snapshot> iterator(Snapshot snapshot) {
139           return new SnapshotIterator(snapshot);
140       }
141   
142       /**
143        * Returns a reverse snapshot iterator that iterates from the snapshots with the
144        * highest epoch to those with the lowest.
145        */
146       public Iterator<Snapshot> reverseIterator() {
147           return new ReverseSnapshotIterator();
148       }
149   
150       /**
151        * Returns a sorted list of snapshot epochs.
152        */
153       public List<Long> epochsList() {
154           List<Long> result = new ArrayList<>();
155           for (Iterator<Snapshot> iterator = iterator(); iterator.hasNext(); ) {
156               result.add(iterator.next().epoch());
157           }
158           return result;
159       }
160   
161       public boolean hasSnapshot(long epoch) {
162           return snapshots.containsKey(epoch);
163       }
164   
165       /**
166        * Gets the snapshot for a specific epoch.
167        */
168       public Snapshot getSnapshot(long epoch) {
169           Snapshot snapshot = snapshots.get(epoch);
170           if (snapshot == null) {
171               throw new RuntimeException("No snapshot for epoch " + epoch + ". Snapshot " +
172                   "epochs are: " + epochsList().stream().map(e -> e.toString()).
173                       collect(Collectors.joining(", ")));
174           }
175           return snapshot;
176       }
177   
178       /**
179        * Creates a new snapshot at the given epoch.
180        *
181        * If {@code epoch} already exists and it is the last snapshot then just return that snapshot.
182        *
183        * @param epoch             The epoch to create the snapshot at.  The current epoch
184        *                          will be advanced to one past this epoch.
185        */
186       public Snapshot getOrCreateSnapshot(long epoch) {
187           Snapshot last = head.prev();
188           if (last.epoch() > epoch) {
189               throw new RuntimeException("Can't create a new snapshot at epoch " + epoch +
190                   " because there is already a snapshot with epoch " + last.epoch());
191           } else if (last.epoch() == epoch) {
192               return last;
193           }
194           Snapshot snapshot = new Snapshot(epoch);
195           last.appendNext(snapshot);
196           snapshots.put(epoch, snapshot);
197           log.debug("Creating snapshot {}", epoch);
198           return snapshot;
199       }
200   
201       /**
202        * Reverts the state of all data structures to the state at the given epoch.
203        *
204        * @param targetEpoch       The epoch of the snapshot to revert to.
205        */
206       public void revertToSnapshot(long targetEpoch) {
207           Snapshot target = getSnapshot(targetEpoch);
208           Iterator<Snapshot> iterator = iterator(target);
209           iterator.next();
210           while (iterator.hasNext()) {
211               Snapshot snapshot = iterator.next();
212               log.debug("Deleting snapshot {} because we are reverting to {}",
213                   snapshot.epoch(), targetEpoch);
214               iterator.remove();
215           }
216           target.handleRevert();
217       }
218   
219       /**
220        * Deletes the snapshot with the given epoch.
221        *
222        * @param targetEpoch       The epoch of the snapshot to delete.
223        */
224       public void deleteSnapshot(long targetEpoch) {
225           deleteSnapshot(getSnapshot(targetEpoch));
226       }
227   
228       /**
229        * Deletes the given snapshot.
230        *
231        * @param snapshot          The snapshot to delete.
232        */
233       public void deleteSnapshot(Snapshot snapshot) {
234           Snapshot prev = snapshot.prev();
235           if (prev != head) {
236               prev.mergeFrom(snapshot);
237           } else {
238               snapshot.erase();
239           }
240           log.debug("Deleting snapshot {}", snapshot.epoch());
241           snapshots.remove(snapshot.epoch(), snapshot);
242       }
243   
244       /**
245        * Deletes all the snapshots up to the given epoch
246        *
247        * @param targetEpoch       The epoch to delete up to.
248        */
249       public void deleteSnapshotsUpTo(long targetEpoch) {
250           for (Iterator<Snapshot> iterator = iterator(); iterator.hasNext(); ) {
251               Snapshot snapshot = iterator.next();
252               if (snapshot.epoch() >= targetEpoch) {
253                   return;
254               }
255               log.debug("Deleting snapshot {}", snapshot.epoch());
256               iterator.remove();
257           }
258       }
259   
260       /**
261        * Return the latest epoch.
262        */
263       public long latestEpoch() {
264           return head.prev().epoch();
265       }
266   
267       /**
268        * Associate with this registry.
269        */
270       public void register(Revertable revertable) {
271           revertables.add(revertable);
272       }
273   
274       /**
275        * Delete all snapshots and resets all of the Revertable object registered.
276        */
277       public void reset() {
278           deleteSnapshotsUpTo(LATEST_EPOCH);
279   
280           for (Revertable revertable : revertables) {
281               revertable.reset();
282           }
283       }
284   }
