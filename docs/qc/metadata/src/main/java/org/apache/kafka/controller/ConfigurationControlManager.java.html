155       /**
156        * Determine the result of applying a batch of incremental configuration changes.  Note
157        * that this method does not change the contents of memory.  It just generates a
158        * result, that you can replay later if you wish using replay().
159        *
160        * Note that there can only be one result per ConfigResource.  So if you try to modify
161        * several keys and one modification fails, the whole ConfigKey fails and nothing gets
162        * changed.
163        *
164        * @param configChanges     Maps each resource to a map from config keys to
165        *                          operation data.
166        * @return                  The result.
167        */
168       ControllerResult<Map<ConfigResource, ApiError>> incrementalAlterConfigs(
169               Map<ConfigResource, Map<String, Entry<OpType, String>>> configChanges,
170               boolean newlyCreatedResource) {
171           List<ApiMessageAndVersion> outputRecords = new ArrayList<>();
172           Map<ConfigResource, ApiError> outputResults = new HashMap<>();
173           for (Entry<ConfigResource, Map<String, Entry<OpType, String>>> resourceEntry :
174                   configChanges.entrySet()) {
175               incrementalAlterConfigResource(resourceEntry.getKey(),
176                   resourceEntry.getValue(),
177                   newlyCreatedResource,
178                   outputRecords,
179                   outputResults);
180           }
181           return ControllerResult.atomicOf(outputRecords, outputResults);
182       }

184       private void incrementalAlterConfigResource(ConfigResource configResource,
185                                                   Map<String, Entry<OpType, String>> keysToOps,
186                                                   boolean newlyCreatedResource,
187                                                   List<ApiMessageAndVersion> outputRecords,
188                                                   Map<ConfigResource, ApiError> outputResults) {
189           List<ApiMessageAndVersion> newRecords = new ArrayList<>();
190           for (Entry<String, Entry<OpType, String>> keysToOpsEntry : keysToOps.entrySet()) {
191               String key = keysToOpsEntry.getKey();
192               String currentValue = null;
193               TimelineHashMap<String, String> currentConfigs = configData.get(configResource);
194               if (currentConfigs != null) {
195                   currentValue = currentConfigs.get(key);
196               }
197               String newValue = currentValue;
198               Entry<OpType, String> opTypeAndNewValue = keysToOpsEntry.getValue();
199               OpType opType = opTypeAndNewValue.getKey();
200               String opValue = opTypeAndNewValue.getValue();
201               switch (opType) {
202                   case SET:
203                       newValue = opValue;
204                       break;
205                   case DELETE:
206                       newValue = null;
207                       break;
208                   case APPEND:
209                   case SUBTRACT:
210                       if (!configSchema.isSplittable(configResource.type(), key)) {
211                           outputResults.put(configResource, new ApiError(
212                               INVALID_CONFIG, "Can't " + opType + " to " +
213                               "key " + key + " because its type is not LIST."));
214                           return;
215                       }
216                       List<String> oldValueList = getParts(newValue, key, configResource);
217                       if (opType == APPEND) {
218                           for (String value : opValue.split(",")) {
219                               if (!oldValueList.contains(value)) {
220                                   oldValueList.add(value);
221                               }
222                           }
223                       } else {
224                           for (String value : opValue.split(",")) {
225                               oldValueList.remove(value);
226                           }
227                       }
228                       newValue = String.join(",", oldValueList);
229                       break;
230               }
231               if (!Objects.equals(currentValue, newValue)) {
232                   newRecords.add(new ApiMessageAndVersion(new ConfigRecord().
233                       setResourceType(configResource.type().id()).
234                       setResourceName(configResource.name()).
235                       setName(key).
236                       setValue(newValue), CONFIG_RECORD.highestSupportedVersion()));
237               }
238           }
239           ApiError error = validateAlterConfig(configResource, newRecords, newlyCreatedResource);
240           if (error.isFailure()) {
241               outputResults.put(configResource, error);
242               return;
243           }
244           outputRecords.addAll(newRecords);
245           outputResults.put(configResource, ApiError.NONE);
246       }

248       private ApiError validateAlterConfig(ConfigResource configResource,
249                                            List<ApiMessageAndVersion> newRecords,
250                                            boolean newlyCreatedResource) {
251           Map<String, String> newConfigs = new HashMap<>();
252           TimelineHashMap<String, String> existingConfigs = configData.get(configResource);
253           if (existingConfigs != null) newConfigs.putAll(existingConfigs);
254           for (ApiMessageAndVersion newRecord : newRecords) {
255               ConfigRecord configRecord = (ConfigRecord) newRecord.message();
256               if (configRecord.value() == null) {
257                   newConfigs.remove(configRecord.name());
258               } else {
259                   newConfigs.put(configRecord.name(), configRecord.value());
260               }
261           }
262           try {
263               validator.validate(configResource, newConfigs);
264               if (!newlyCreatedResource) {
265                   existenceChecker.accept(configResource);
266               }
267               if (alterConfigPolicy.isPresent()) {
268                   alterConfigPolicy.get().validate(new RequestMetadata(configResource, newConfigs));
269               }
270           } catch (ConfigException e) {
271               return new ApiError(INVALID_CONFIG, e.getMessage());
272           } catch (Throwable e) {
273               return ApiError.fromThrowable(e);
274           }
275           return ApiError.NONE;
276       }

304       private void legacyAlterConfigResource(ConfigResource configResource,
305                                              Map<String, String> newConfigs,
306                                              boolean newlyCreatedResource,
307                                              List<ApiMessageAndVersion> outputRecords,
308                                              Map<ConfigResource, ApiError> outputResults) {
309           List<ApiMessageAndVersion> newRecords = new ArrayList<>();
310           Map<String, String> currentConfigs = configData.get(configResource);
311           if (currentConfigs == null) {
312               currentConfigs = Collections.emptyMap();
313           }
314           for (Entry<String, String> entry : newConfigs.entrySet()) {
315               String key = entry.getKey();
316               String newValue = entry.getValue();
317               String currentValue = currentConfigs.get(key);
318               if (!Objects.equals(newValue, currentValue)) {
319                   newRecords.add(new ApiMessageAndVersion(new ConfigRecord().
320                       setResourceType(configResource.type().id()).
321                       setResourceName(configResource.name()).
322                       setName(key).
323                       setValue(newValue), CONFIG_RECORD.highestSupportedVersion()));
324               }
325           }
326           for (String key : currentConfigs.keySet()) {
327               if (!newConfigs.containsKey(key)) {
328                   newRecords.add(new ApiMessageAndVersion(new ConfigRecord().
329                       setResourceType(configResource.type().id()).
330                       setResourceName(configResource.name()).
331                       setName(key).
332                       setValue(null), CONFIG_RECORD.highestSupportedVersion()));
333               }
334           }
335           ApiError error = validateAlterConfig(configResource, newRecords, newlyCreatedResource);
336           if (error.isFailure()) {
337               outputResults.put(configResource, error);
338               return;
339           }
340           outputRecords.addAll(newRecords);
341           outputResults.put(configResource, ApiError.NONE);
342       }

344       private List<String> getParts(String value, String key, ConfigResource configResource) {
345           if (value == null) {
346               value = configSchema.getDefault(configResource.type(), key);
347           }
348           List<String> parts = new ArrayList<>();
349           if (value == null) {
350               return parts;
351           }
352           String[] splitValues = value.split(",");
353           for (String splitValue : splitValues) {
354               if (!splitValue.isEmpty()) {
355                   parts.add(splitValue);
356               }
357           }
358           return parts;
359       }

362        * Apply a configuration record to the in-memory state.
363        *
364        * @param record            The ConfigRecord.
365        */
366       public void replay(ConfigRecord record) {
367           Type type = Type.forId(record.resourceType());
368           ConfigResource configResource = new ConfigResource(type, record.resourceName());
369           TimelineHashMap<String, String> configs = configData.get(configResource);
370           if (configs == null) {
371               configs = new TimelineHashMap<>(snapshotRegistry, 0);
372               configData.put(configResource, configs);
373           }
374           if (record.value() == null) {
375               configs.remove(record.name());
376           } else {
377               configs.put(record.name(), record.value());
378           }
379           if (configs.isEmpty()) {
380               configData.remove(configResource);
381           }
382           log.info("{}: set configuration {} to {}", configResource, record.name(), record.value());
383       }

395       public Map<ConfigResource, ResultOrError<Map<String, String>>> describeConfigs(
396               long lastCommittedOffset, Map<ConfigResource, Collection<String>> resources) {
397           Map<ConfigResource, ResultOrError<Map<String, String>>> results = new HashMap<>();
398           for (Entry<ConfigResource, Collection<String>> resourceEntry : resources.entrySet()) {
399               ConfigResource resource = resourceEntry.getKey();
400               try {
401                   validator.validate(resource);
402               } catch (Throwable e) {
403                   results.put(resource, new ResultOrError<>(ApiError.fromThrowable(e)));
404                   continue;
405               }
406               Map<String, String> foundConfigs = new HashMap<>();
407               TimelineHashMap<String, String> configs =
408                   configData.get(resource, lastCommittedOffset);
409               if (configs != null) {
410                   Collection<String> targetConfigs = resourceEntry.getValue();
411                   if (targetConfigs.isEmpty()) {
412                       Iterator<Entry<String, String>> iter =
413                           configs.entrySet(lastCommittedOffset).iterator();
414                       while (iter.hasNext()) {
415                           Entry<String, String> entry = iter.next();
416                           foundConfigs.put(entry.getKey(), entry.getValue());
417                       }
418                   } else {
419                       for (String key : targetConfigs) {
420                           String value = configs.get(key, lastCommittedOffset);
421                           if (value != null) {
422                               foundConfigs.put(key, value);
423                           }
424                       }
425                   }
426               }
427               results.put(resource, new ResultOrError<>(foundConfigs));
428           }
429           return results;
430       }

455       class ConfigurationControlIterator implements Iterator<List<ApiMessageAndVersion>> {
456           private final long epoch;
457           private final Iterator<Entry<ConfigResource, TimelineHashMap<String, String>>> iterator;
458   
459           ConfigurationControlIterator(long epoch) {
460               this.epoch = epoch;
461               this.iterator = configData.entrySet(epoch).iterator();
462           }
463   
464           @Override
465           public boolean hasNext() {
466               return iterator.hasNext();
467           }
468   
469           @Override
470           public List<ApiMessageAndVersion> next() {
471               if (!hasNext()) throw new NoSuchElementException();
472               List<ApiMessageAndVersion> records = new ArrayList<>();
473               Entry<ConfigResource, TimelineHashMap<String, String>> entry = iterator.next();
474               ConfigResource resource = entry.getKey();
475               for (Entry<String, String> configEntry : entry.getValue().entrySet(epoch)) {
476                   records.add(new ApiMessageAndVersion(new ConfigRecord().
477                       setResourceName(resource.name()).
478                       setResourceType(resource.type().id()).
479                       setName(configEntry.getKey()).
480                       setValue(configEntry.getValue()), CONFIG_RECORD.highestSupportedVersion()));
481               }
482               return records;
483           }
484       }
