37   /**
38    * Associates brokers with their in-sync partitions.
39    *
40    * This is useful when we need to remove a broker from all the ISRs, or move all leaders
41    * away from a broker.
42    *
43    * We also track all the partitions that currently have no leader.
44    *
45    * The core data structure is a map from broker IDs to topic maps.  Each topic map relates
46    * topic UUIDs to arrays of partition IDs.
47    *
48    * Each entry in the array has a high bit which indicates that the broker is the leader
49    * for the given partition, as well as 31 low bits which contain the partition id.  This
50    * works because partition IDs cannot be negative.
51    */
52   public class BrokersToIsrs {
53       private final static int LEADER_FLAG = 0x8000_0000;
54   
55       private final static int REPLICA_MASK = 0x7fff_ffff;
56   
57       static class PartitionsOnReplicaIterator implements Iterator<TopicIdPartition> {
58           private final Iterator<Entry<Uuid, int[]>> iterator;
59           private final boolean leaderOnly;
60           private int offset = 0;
61           Uuid uuid = Uuid.ZERO_UUID;
62           int[] replicas = NONE;
63           private TopicIdPartition next = null;
64   
65           PartitionsOnReplicaIterator(Map<Uuid, int[]> topicMap, boolean leaderOnly) {
66               this.iterator = topicMap.entrySet().iterator();
67               this.leaderOnly = leaderOnly;
68           }
69   
70           @Override
71           public boolean hasNext() {
72               if (next != null) return true;
73               while (true) {
74                   if (offset >= replicas.length) {
75                       if (!iterator.hasNext()) return false;
76                       offset = 0;
77                       Entry<Uuid, int[]> entry = iterator.next();
78                       uuid = entry.getKey();
79                       replicas = entry.getValue();
80                   }
81                   int replica = replicas[offset++];
82                   if ((!leaderOnly) || (replica & LEADER_FLAG) != 0) {
83                       next = new TopicIdPartition(uuid, replica & REPLICA_MASK);
84                       return true;
85                   }
86               }
87           }
88   
89           @Override
90           public TopicIdPartition next() {
91               if (!hasNext()) {
92                   throw new NoSuchElementException();
93               }
94               TopicIdPartition result = next;
95               next = null;
96               return result;
97           }
98       }
99   
100       private final SnapshotRegistry snapshotRegistry;
101   
102       /**
103        * A map of broker IDs to the partitions that the broker is in the ISR for.
104        * Partitions with no isr members appear in this map under id NO_LEADER.
105        */
106       private final TimelineHashMap<Integer, TimelineHashMap<Uuid, int[]>> isrMembers;
107       
108       private final TimelineInteger offlinePartitionCount;
109   
110       BrokersToIsrs(SnapshotRegistry snapshotRegistry) {
111           this.snapshotRegistry = snapshotRegistry;
112           this.isrMembers = new TimelineHashMap<>(snapshotRegistry, 0);
113           this.offlinePartitionCount = new TimelineInteger(snapshotRegistry);
114       }
115   
116       /**
117        * Update our records of a partition's ISR.
118        *
119        * @param topicId       The topic ID of the partition.
120        * @param partitionId   The partition ID of the partition.
121        * @param prevIsr       The previous ISR, or null if the partition is new.
122        * @param nextIsr       The new ISR, or null if the partition is being removed.
123        * @param prevLeader    The previous leader, or NO_LEADER if the partition had no leader.
124        * @param nextLeader    The new leader, or NO_LEADER if the partition now has no leader.
125        */
126       void update(Uuid topicId, int partitionId, int[] prevIsr, int[] nextIsr,
127                   int prevLeader, int nextLeader) {
128           int[] prev;
129           if (prevIsr == null) {
130               prev = NONE;
131           } else {
132               if (prevLeader == NO_LEADER) {
133                   prev = Replicas.copyWith(prevIsr, NO_LEADER);
134                   if (nextLeader != NO_LEADER) {
135                       offlinePartitionCount.decrement();
136                   }
137               } else {
138                   prev = Replicas.clone(prevIsr);
139               }
140               Arrays.sort(prev);
141           }
142           int[] next;
143           if (nextIsr == null) {
144               next = NONE;
145           } else {
146               if (nextLeader == NO_LEADER) {
147                   next = Replicas.copyWith(nextIsr, NO_LEADER);
148                   if (prevLeader != NO_LEADER) {
149                       offlinePartitionCount.increment();
150                   }
151               } else {
152                   next = Replicas.clone(nextIsr);
153               }
154               Arrays.sort(next);
155           }
156           int i = 0, j = 0;
157           while (true) {
158               if (i == prev.length) {
159                   if (j == next.length) {
160                       break;
161                   }
162                   int newReplica = next[j];
163                   add(newReplica, topicId, partitionId, newReplica == nextLeader);
164                   j++;
165               } else if (j == next.length) {
166                   int prevReplica = prev[i];
167                   remove(prevReplica, topicId, partitionId, prevReplica == prevLeader);
168                   i++;
169               } else {
170                   int prevReplica = prev[i];
171                   int newReplica = next[j];
172                   if (prevReplica < newReplica) {
173                       remove(prevReplica, topicId, partitionId, prevReplica == prevLeader);
174                       i++;
175                   } else if (prevReplica > newReplica) {
176                       add(newReplica, topicId, partitionId, newReplica == nextLeader);
177                       j++;
178                   } else {
179                       boolean wasLeader = prevReplica == prevLeader;
180                       boolean isLeader = prevReplica == nextLeader;
181                       if (wasLeader != isLeader) {
182                           change(prevReplica, topicId, partitionId, wasLeader, isLeader);
183                       }
184                       i++;
185                       j++;
186                   }
187               }
188           }
189       }

191       void removeTopicEntryForBroker(Uuid topicId, int brokerId) {
192           Map<Uuid, int[]> topicMap = isrMembers.get(brokerId);
193           if (topicMap != null) {
194               if (brokerId == NO_LEADER) {
195                   offlinePartitionCount.set(offlinePartitionCount.get() - topicMap.get(topicId).length);
196               }
197               topicMap.remove(topicId);
198           }
199       }

201       private void add(int brokerId, Uuid topicId, int newPartition, boolean leader) {
202           if (leader) {
203               newPartition = newPartition | LEADER_FLAG;
204           }
205           TimelineHashMap<Uuid, int[]> topicMap = isrMembers.get(brokerId);
206           if (topicMap == null) {
207               topicMap = new TimelineHashMap<>(snapshotRegistry, 0);
208               isrMembers.put(brokerId, topicMap);
209           }
210           int[] partitions = topicMap.get(topicId);
211           int[] newPartitions;
212           if (partitions == null) {
213               newPartitions = new int[1];
214           } else {
215               newPartitions = new int[partitions.length + 1];
216               System.arraycopy(partitions, 0, newPartitions, 0, partitions.length);
217           }
218           newPartitions[newPartitions.length - 1] = newPartition;
219           topicMap.put(topicId, newPartitions);
220       }
221   
222       private void change(int brokerId, Uuid topicId, int partition,
223                           boolean wasLeader, boolean isLeader) {
224           TimelineHashMap<Uuid, int[]> topicMap = isrMembers.get(brokerId);
225           if (topicMap == null) {
226               throw new RuntimeException("Broker " + brokerId + " has no isrMembers " +
227                   "entry, so we can't change " + topicId + ":" + partition);
228           }
229           int[] partitions = topicMap.get(topicId);
230           if (partitions == null) {
231               throw new RuntimeException("Broker " + brokerId + " has no " +
232                   "entry in isrMembers for topic " + topicId);
233           }
234           int[] newPartitions = new int[partitions.length];
235           int target = wasLeader ? partition | LEADER_FLAG : partition;
236           for (int i = 0; i < partitions.length; i++) {
237               int cur = partitions[i];
238               if (cur == target) {
239                   newPartitions[i] = isLeader ? partition | LEADER_FLAG : partition;
240               } else {
241                   newPartitions[i] = cur;
242               }
243           }
244           topicMap.put(topicId, newPartitions);
245       }

247       private void remove(int brokerId, Uuid topicId, int removedPartition, boolean leader) {
248           if (leader) {
249               removedPartition = removedPartition | LEADER_FLAG;
250           }
251           TimelineHashMap<Uuid, int[]> topicMap = isrMembers.get(brokerId);
252           if (topicMap == null) {
253               throw new RuntimeException("Broker " + brokerId + " has no isrMembers " +
254                   "entry, so we can't remove " + topicId + ":" + removedPartition);
255           }
256           int[] partitions = topicMap.get(topicId);
257           if (partitions == null) {
258               throw new RuntimeException("Broker " + brokerId + " has no " +
259                   "entry in isrMembers for topic " + topicId);
260           }
261           if (partitions.length == 1) {
262               if (partitions[0] != removedPartition) {
263                   throw new RuntimeException("Broker " + brokerId + " has no " +
264                       "entry in isrMembers for " + topicId + ":" + removedPartition);
265               }
266               topicMap.remove(topicId);
267               if (topicMap.isEmpty()) {
268                   isrMembers.remove(brokerId);
269               }
270           } else {
271               int[] newPartitions = new int[partitions.length - 1];
272               int j = 0;
273               for (int i = 0; i < partitions.length; i++) {
274                   int partition = partitions[i];
275                   if (partition != removedPartition) {
276                       newPartitions[j++] = partition;
277                   }
278               }
279               topicMap.put(topicId, newPartitions);
280           }
281       }
282   
283       PartitionsOnReplicaIterator iterator(int brokerId, boolean leadersOnly) {
284           Map<Uuid, int[]> topicMap = isrMembers.get(brokerId);
285           if (topicMap == null) {
286               topicMap = Collections.emptyMap();
287           }
288           return new PartitionsOnReplicaIterator(topicMap, leadersOnly);
289       }

291       PartitionsOnReplicaIterator partitionsWithNoLeader() {
292           return iterator(NO_LEADER, true);
293       }

295       PartitionsOnReplicaIterator partitionsLedByBroker(int brokerId) {
296           return iterator(brokerId, true);
297       }

299       PartitionsOnReplicaIterator partitionsWithBrokerInIsr(int brokerId) {
300           return iterator(brokerId, false);
301       }

304           return iterator(brokerId, true).hasNext();
305       }
