18   
19   import org.slf4j.Logger;
20   import org.slf4j.LoggerFactory;
21   
22   import java.util.concurrent.TimeUnit;
23   import java.util.function.Function;
24   
25   
26   /**
27    * Represents a synonym for a configuration plus a conversion function. The conversion
28    * function is necessary for cases where the synonym is denominated in different units
29    * (hours versus milliseconds, etc.)
30    */
31   public class ConfigSynonym {
32       private static final Logger log = LoggerFactory.getLogger(ConfigSynonym.class);
33   
34       public static final Function<String, String> HOURS_TO_MILLISECONDS = input -> {
35           int hours = valueToInt(input, 0, "hoursToMilliseconds");
36           return String.valueOf(TimeUnit.MILLISECONDS.convert(hours, TimeUnit.HOURS));
37       };
38   
39       public static final Function<String, String> MINUTES_TO_MILLISECONDS = input -> {
40           int hours = valueToInt(input, 0, "minutesToMilliseconds");
41           return String.valueOf(TimeUnit.MILLISECONDS.convert(hours, TimeUnit.MINUTES));
42       };
43   
44       private static int valueToInt(String input, int defaultValue, String what) {
45           if (input == null) return defaultValue;
46           String trimmedInput = input.trim();
47           if (trimmedInput.isEmpty()) {
48               return defaultValue;
49           }
50           try {
51               return Integer.parseInt(trimmedInput);
52           } catch (Exception e) {
53               log.error("{} failed: unable to parse '{}' as an integer.", what, trimmedInput, e);
54               return defaultValue;
55           }
56       }
57   
58       private final String name;
59       private final Function<String, String> converter;
60   
61       public ConfigSynonym(String name, Function<String, String> converter) {
62           this.name = name;
63           this.converter = converter;
64       }
65   
66       public ConfigSynonym(String name) {
67           this(name, Function.identity());
68       }
69   
70       public String name() {
71           return name;
72       }
73   
74       public Function<String, String> converter() {
75           return converter;
76       }
77   }
