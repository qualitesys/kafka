18   package org.apache.kafka.image;
19   
20   import org.apache.kafka.common.Uuid;
21   import org.apache.kafka.common.metadata.AccessControlEntryRecord;
22   import org.apache.kafka.common.metadata.RemoveAccessControlEntryRecord;
23   import org.apache.kafka.metadata.authorizer.StandardAcl;
24   import org.apache.kafka.metadata.authorizer.StandardAclWithId;
25   import org.apache.kafka.server.common.MetadataVersion;
26   
27   import java.util.HashMap;
28   import java.util.LinkedHashMap;
29   import java.util.Map;
30   import java.util.Map.Entry;
31   import java.util.Optional;
32   import java.util.stream.Collectors;
33   
34   
35   /**
36    * Represents changes to the ACLs in the metadata image.
37    */
38   public final class AclsDelta {
39       private final AclsImage image;
40       private final Map<Uuid, Optional<StandardAcl>> changes = new LinkedHashMap<>();
41       private boolean isSnapshotDelta = false;
42   
43       public AclsDelta(AclsImage image) {
44           this.image = image;
45       }
46   
47       /**
48        * Returns a Map of deltas from ACL ID to optional StandardAcl. An empty optional value indicates the ACL
49        * is for removal. An optional with a value indicates the ACL is to be added.
50        *
51        * @return Map of deltas.
52        */
53       public Map<Uuid, Optional<StandardAcl>> changes() {
54           return changes;
55       }
56   
57       void finishSnapshot() {
58           this.isSnapshotDelta = true;
59       }
60   
61       public void handleMetadataVersionChange(MetadataVersion newVersion) {
62           // no-op
63       }
64   
65       public boolean isSnapshotDelta() {
66           return isSnapshotDelta;
67       }
68   
69       public void replay(AccessControlEntryRecord record) {
70           StandardAclWithId aclWithId = StandardAclWithId.fromRecord(record);
71           changes.put(aclWithId.id(), Optional.of(aclWithId.acl()));
72       }
73   
74       /**
75        * This method replays a RemoveAccessControlEntryRecord record. If the current image contains the ACL
76        * the removal is stored as an Optional.empty() value in the Map. If the changes Map contains the ACL
77        * it means the ACL was recently applied and isn't in the image yet, in which case the ACL can be totally removed
78        * from the Map because there is no need to add it then delete it when the changes are applied.
79        *
80        * @param record Log metadata record to replay.
81        */
82       public void replay(RemoveAccessControlEntryRecord record) {
83           if (image.acls().containsKey(record.id())) {
84               changes.put(record.id(), Optional.empty());
85           } else if (changes.containsKey(record.id())) {
86               changes.remove(record.id());
87           } else {
88               throw new IllegalStateException("Failed to find existing ACL with ID " + record.id() + " in either image or changes");
89           }
90       }
91   
92       public AclsImage apply() {
93           Map<Uuid, StandardAcl> newAcls = new HashMap<>();
94           if (!isSnapshotDelta) {
95               for (Entry<Uuid, StandardAcl> entry : image.acls().entrySet()) {
96                   Optional<StandardAcl> change = changes.get(entry.getKey());
97                   if (change == null) {
98                       newAcls.put(entry.getKey(), entry.getValue());
99                   } else if (change.isPresent()) {
100                       newAcls.put(entry.getKey(), change.get());
101                   }
102               }
103           }
104           for (Entry<Uuid, Optional<StandardAcl>> entry : changes.entrySet()) {
105               if (!newAcls.containsKey(entry.getKey())) {
106                   if (entry.getValue().isPresent()) {
107                       newAcls.put(entry.getKey(), entry.getValue().get());
108                   }
109               }
110           }
111           return new AclsImage(newAcls);
112       }
113   
114       @Override
115       public String toString() {
116           return "AclsDelta(isSnapshotDelta=" + isSnapshotDelta +
117               ", changes=" + changes.entrySet().stream().
118                   map(e -> "" + e.getKey() + "=" + e.getValue()).
119                   collect(Collectors.joining(", ")) + ")";
120       }
121   }
