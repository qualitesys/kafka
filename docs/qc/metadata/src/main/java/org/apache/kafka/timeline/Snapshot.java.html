18   package org.apache.kafka.timeline;
19   
20   import java.util.IdentityHashMap;
21   import java.util.Map;
22   
23   /**
24    * A snapshot of some timeline data structures.
25    *
26    * The snapshot contains historical data for several timeline data structures.
27    * We use an IdentityHashMap to store this data.  This way, we can easily drop all of
28    * the snapshot data.
29    */
30   class Snapshot {
31       private final long epoch;
32       private IdentityHashMap<Revertable, Delta> map = new IdentityHashMap<>(4);
33       private Snapshot prev = this;
34       private Snapshot next = this;
35   
36       Snapshot(long epoch) {
37           this.epoch = epoch;
38       }
39   
40       long epoch() {
41           return epoch;
42       }
43   
44       @SuppressWarnings("unchecked")
45       <T extends Delta> T getDelta(Revertable owner) {
46           return (T) map.get(owner);
47       }
48   
49       void setDelta(Revertable owner, Delta delta) {
50           map.put(owner, delta);
51       }
52   
53       void handleRevert() {
54           for (Map.Entry<Revertable, Delta> entry : map.entrySet()) {
55               entry.getKey().executeRevert(epoch, entry.getValue());
56           }
57       }
58   
59       void mergeFrom(Snapshot source) {
60           // Merge the deltas from the source snapshot into this snapshot.
61           for (Map.Entry<Revertable, Delta> entry : source.map.entrySet()) {
62               // We first try to just copy over the object reference.  That will work if
63               //we have no entry at all for the given Revertable.
64               Delta destinationDelta = map.putIfAbsent(entry.getKey(), entry.getValue());
65               if (destinationDelta != null) {
66                   // If we already have an entry for the Revertable, we need to merge the
67                   // source delta into our delta.
68                   destinationDelta.mergeFrom(epoch, entry.getValue());
69               }
70           }
71           // Delete the source snapshot to make sure nobody tries to reuse it.  We might now
72           // share some delta entries with it.
73           source.erase();
74       }
75   
76       Snapshot prev() {
77           return prev;
78       }
79   
80       Snapshot next() {
81           return next;
82       }
83   
84       void appendNext(Snapshot newNext) {
85           newNext.prev = this;
86           newNext.next = next;
87           next.prev = newNext;
88           next = newNext;
89       }
90   
91       void erase() {
92           map = null;
93           next.prev = prev;
94           prev.next = next;
95           prev = this;
96           next = this;
97       }
98   }
