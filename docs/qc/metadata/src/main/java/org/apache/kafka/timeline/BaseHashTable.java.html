20   import java.util.ArrayList;
21   import java.util.List;

23   /**
24    * A hash table which uses separate chaining.
25    *
26    * In order to optimize memory consumption a bit, the common case where there is
27    * one element per slot is handled by simply placing the element in the slot,
28    * and the case where there are multiple elements is handled by creating an
29    * array and putting that in the slot.  Java is storing type info in memory
30    * about every object whether we want it or not, so let's get some benefit
31    * out of it.
32    *
33    * Arrays and null values cannot be inserted.
34    */
35   @SuppressWarnings("unchecked")
36   class BaseHashTable<T> {
37       /**
38        * The maximum load factor we will allow the hash table to climb to before expanding.
39        */
40       private final static double MAX_LOAD_FACTOR = 0.75f;
41   
42       /**
43        * The minimum number of slots we can have in the hash table.
44        */
45       final static int MIN_CAPACITY = 2;
46   
47       /**
48        * The maximum number of slots we can have in the hash table.
49        */
50       final static int MAX_CAPACITY = 1 << 30;
51   
52       private Object[] elements;
53       private int size = 0;
54   
55       BaseHashTable(int expectedSize) {
56           this.elements = new Object[expectedSizeToCapacity(expectedSize)];
57       }
58   
59       /**
60        * Calculate the capacity we should provision, given the expected size.
61        *
62        * Our capacity must always be a power of 2, and never less than 2 or more
63        * than MAX_CAPACITY.  We use 64-bit numbers here to avoid overflow
64        * concerns.
65        */
66       static int expectedSizeToCapacity(int expectedSize) {
67           long minCapacity = (long) Math.ceil((float) expectedSize / MAX_LOAD_FACTOR);
68           return Math.max(MIN_CAPACITY,
69                   (int) Math.min(MAX_CAPACITY, roundUpToPowerOfTwo(minCapacity)));
70       }
71   
72       private static long roundUpToPowerOfTwo(long i) {
73           if (i <= 0) {
74               return 0;
75           } else if (i > (1L << 62)) {
76               throw new ArithmeticException("There are no 63-bit powers of 2 higher than " +
77                       "or equal to " + i);
78           } else {
79               return 1L << -Long.numberOfLeadingZeros(i - 1);
80           }
81       }
82   
83       final int baseSize() {
84           return size;
85       }
86   
87       final Object[] baseElements() {
88           return elements;
89       }
90   
91       final T baseGet(Object key) {
92           int slot = findSlot(key, elements.length);
93           Object value = elements[slot];
94           if (value == null) {
95               return null;
96           } else if (value instanceof Object[]) {
97               T[] array = (T[]) value;
98               for (T object : array) {
99                   if (object.equals(key)) {
100                       return object;
101                   }
102               }
103               return null;
104           } else if (value.equals(key)) {
105               return (T) value;
106           } else {
107               return null;
108           }
109       }
110   
111       final T baseAddOrReplace(T newObject) {
112           if (((size + 1) * MAX_LOAD_FACTOR > elements.length) &&
113                   (elements.length < MAX_CAPACITY)) {
114               int newSize = elements.length * 2;
115               rehash(newSize);
116           }
117           int slot = findSlot(newObject, elements.length);
118           Object cur = elements[slot];
119           if (cur == null) {
120               size++;
121               elements[slot] = newObject;
122               return null;
123           } else if (cur instanceof Object[]) {
124               T[] curArray = (T[]) cur;
125               for (int i =  0; i < curArray.length; i++) {
126                   T value = curArray[i];
127                   if (value.equals(newObject)) {
128                       curArray[i] = newObject;
129                       return value;
130                   }
131               }
132               size++;
133               T[] newArray = (T[]) new Object[curArray.length + 1];
134               System.arraycopy(curArray, 0, newArray, 0, curArray.length);
135               newArray[curArray.length] = newObject;
136               elements[slot] = newArray;
137               return null;
138           } else if (cur.equals(newObject)) {
139               elements[slot] = newObject;
140               return (T) cur;
141           } else {
142               size++;
143               elements[slot] = new Object[] {cur, newObject};
144               return null;
145           }
146       }
147   
148       final T baseRemove(Object key) {
149           int slot = findSlot(key, elements.length);
150           Object object = elements[slot];
151           if (object == null) {
152               return null;
153           } else if (object instanceof Object[]) {
154               Object[] curArray = (Object[]) object;
155               for (int i = 0; i < curArray.length; i++) {
156                   if (curArray[i].equals(key)) {
157                       size--;
158                       if (curArray.length <= 2) {
159                           int j = i == 0 ? 1 : 0;
160                           elements[slot] = curArray[j];
161                       } else {
162                           Object[] newArray = new Object[curArray.length - 1];
163                           System.arraycopy(curArray, 0, newArray, 0, i);
164                           System.arraycopy(curArray, i + 1, newArray, i, curArray.length - 1 - i);
165                           elements[slot] = newArray;
166                       }
167                       return (T) curArray[i];
168                   }
169               }
170               return null;
171           } else if (object.equals(key)) {
172               size--;
173               elements[slot] = null;
174               return (T) object;
175           } else {
176               return null;
177           }
178       }
179   
180       /**
181        * Expand the hash table to a new size.  Existing elements will be copied to new slots.
182        */
183       final private void rehash(int newSize) {
184           Object[] prevElements = elements;
185           elements = new Object[newSize];
186           List<Object> ready = new ArrayList<>();
187           for (int slot = 0; slot < prevElements.length; slot++) {
188               unpackSlot(ready, prevElements, slot);
189               for (Object object : ready) {
190                   int newSlot = findSlot(object, elements.length);
191                   Object cur = elements[newSlot];
192                   if (cur == null) {
193                       elements[newSlot] = object;
194                   } else if (cur instanceof Object[]) {
195                       Object[] curArray = (Object[]) cur;
196                       Object[] newArray = new Object[curArray.length + 1];
197                       System.arraycopy(curArray, 0, newArray, 0, curArray.length);
198                       newArray[curArray.length] = object;
199                       elements[newSlot] = newArray;
200                   } else {
201                       elements[newSlot] = new Object[]{cur, object};
202                   }
203               }
204               ready.clear();
205           }
206       }
207   
208       /**
209        * Find the slot in the array that an element should go into.
210        */
211       static int findSlot(Object object, int numElements) {
212           // This performs a secondary hash using Knuth's multiplicative Fibonacci
213           // hashing.  Then, we choose some of the highest bits.  The number of bits
214           // we choose is based on the table size.  If the size is 2, we need 1 bit;
215           // if the size is 4, we need 2 bits, etc.
216           int objectHashCode = object.hashCode();
217           int log2size = 32 - Integer.numberOfLeadingZeros(numElements);
218           int shift = 65 - log2size;
219           return (int) ((objectHashCode * -7046029254386353131L) >>> shift);
220       }
221   
222       /**
223        * Copy any elements in the given slot into the output list.
224        */
225       static <T> void unpackSlot(List<T> out, Object[] elements, int slot) {
226           Object value = elements[slot];
227           if (value == null) {
228               return;
229           } else if (value instanceof Object[]) {
230               Object[] array = (Object[]) value;
231               for (Object object : array) {
232                   out.add((T) object);
233               }
234           } else {
235               out.add((T) value);
236           }
237       }

239       String baseToDebugString() {
240           StringBuilder bld = new StringBuilder();
241           bld.append("BaseHashTable{");
242           for (int i = 0; i < elements.length; i++) {
243               Object slotObject = elements[i];
244               bld.append(String.format("%n%d: ", i));
245               if (slotObject == null) {
246                   bld.append("null");
247               } else if (slotObject instanceof Object[]) {
248                   Object[] array = (Object[]) slotObject;
249                   String prefix = "";
250                   for (Object object : array) {
251                       bld.append(prefix);
252                       prefix = ", ";
253                       bld.append(object);
254                   }
255               } else {
256                   bld.append(slotObject);
257               }
258           }
259           bld.append(String.format("%n}"));
260           return bld.toString();
261       }
