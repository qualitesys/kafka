125   /**
126    * QuorumController implements the main logic of the KRaft (Kafka Raft Metadata) mode controller.
127    *
128    * The node which is the leader of the metadata log becomes the active controller.  All
129    * other nodes remain in standby mode.  Standby controllers cannot create new metadata log
130    * entries.  They just replay the metadata log entries that the current active controller
131    * has created.
132    *
133    * The QuorumController is single-threaded.  A single event handler thread performs most
134    * operations.  This avoids the need for complex locking.
135    *
136    * The controller exposes an asynchronous, futures-based API to the world.  This reflects
137    * the fact that the controller may have several operations in progress at any given
138    * point.  The future associated with each operation will not be completed until the
139    * results of the operation have been made durable to the metadata log.
140    *
141    * The QuorumController uses the "metadata.version" feature flag as a mechanism to control
142    * the usage of new log record schemas. Starting with 3.3, this version must be set before
143    * the controller can fully initialize.
144    */
145   public final class QuorumController implements Controller {

409       private Throwable handleEventException(String name,
410                                              OptionalLong startProcessingTimeNs,
411                                              Throwable exception) {
412           if (!startProcessingTimeNs.isPresent()) {
413               log.error("{}: unable to start processing because of {}.", name,
414                   exception.getClass().getSimpleName());
415               if (exception instanceof ApiException) {
416                   return exception;
417               } else {
418                   return new UnknownServerException(exception);
419               }
420           }
421           long endProcessingTime = time.nanoseconds();
422           long deltaNs = endProcessingTime - startProcessingTimeNs.getAsLong();
423           long deltaUs = MICROSECONDS.convert(deltaNs, NANOSECONDS);
424           if (exception instanceof ApiException) {
425               log.info("{}: failed with {} in {} us", name,
426                   exception.getClass().getSimpleName(), deltaUs);
427               return exception;
428           }
429           log.warn("{}: failed with unknown server exception {} at epoch {} in {} us.  " +
430                   "Reverting to last committed offset {}.",
431               name, exception.getClass().getSimpleName(), curClaimEpoch, deltaUs,
432               lastCommittedOffset, exception);
433           raftClient.resign(curClaimEpoch);
434           renounce();
435           return new UnknownServerException(exception);
436       }

689           ControllerWriteEvent(String name, ControllerWriteOperation<T> op) {
690               this.name = name;

700           @Override
701           public void run() throws Exception {
702               long now = time.nanoseconds();
703               controllerMetrics.updateEventQueueTime(NANOSECONDS.toMillis(now - eventCreatedTimeNs));
704               int controllerEpoch = curClaimEpoch;
705               if (controllerEpoch == -1) {
706                   throw newNotControllerException();
707               }
708               startProcessingTimeNs = OptionalLong.of(now);
709               ControllerResult<T> result = op.generateRecordsAndResult();
710               if (result.records().isEmpty()) {
711                   op.processBatchEndOffset(writeOffset);
712                   // If the operation did not return any records, then it was actually just
713                   // a read after all, and not a read + write.  However, this read was done
714                   // from the latest in-memory state, which might contain uncommitted data.
715                   OptionalLong maybeOffset = purgatory.highestPendingOffset();
716                   if (!maybeOffset.isPresent()) {
717                       // If the purgatory is empty, there are no pending operations and no
718                       // uncommitted state.  We can complete immediately.
719                       resultAndOffset = ControllerResultAndOffset.of(-1, result);
720                       log.debug("Completing read-only operation {} immediately because " +
721                           "the purgatory is empty.", this);
722                       complete(null);
723                   } else {
724                       // If there are operations in the purgatory, we want to wait for the latest
725                       // one to complete before returning our result to the user.
726                       resultAndOffset = ControllerResultAndOffset.of(maybeOffset.getAsLong(), result);
727                       log.debug("Read-only operation {} will be completed when the log " +
728                           "reaches offset {}", this, resultAndOffset.offset());
729                   }
730               } else {
731                   // If the operation returned a batch of records, those records need to be
732                   // written before we can return our result to the user.  Here, we hand off
733                   // the batch of records to the raft client.  They will be written out
734                   // asynchronously.
735                   final long offset;
736                   if (result.isAtomic()) {
737                       offset = raftClient.scheduleAtomicAppend(controllerEpoch, result.records());
738                   } else {
739                       offset = raftClient.scheduleAppend(controllerEpoch, result.records());
740                   }
741                   op.processBatchEndOffset(offset);
742                   updateWriteOffset(offset);
743                   resultAndOffset = ControllerResultAndOffset.of(offset, result);
744                   for (ApiMessageAndVersion message : result.records()) {
745                       replay(message.message(), Optional.empty(), offset);
746                   }
747                   snapshotRegistry.getOrCreateSnapshot(offset);
748   
749                   log.debug("Read-write operation {} will be completed when the log " +
750                       "reaches offset {}.", this, resultAndOffset.offset());
751               }
752   
753               // After every controller write event, schedule a leader rebalance if there are any topic partition
754               // with leader that is not the preferred leader.
755               maybeScheduleNextBalancePartitionLeaders();
756   
757               // Remember the latest offset and future if it is not already completed
758               if (!future.isDone()) {
759                   purgatory.add(resultAndOffset.offset(), this);
760               }
761           }
