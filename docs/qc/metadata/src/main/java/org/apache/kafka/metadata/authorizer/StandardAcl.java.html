33   /**
34    * A Kafka ACLs which is identified by a UUID and stored in the metadata log.
35    */
36   final public class StandardAcl implements Comparable<StandardAcl> {
37       public static StandardAcl fromRecord(AccessControlEntryRecord record) {
38           return new StandardAcl(
39               ResourceType.fromCode(record.resourceType()),
40               record.resourceName(),
41               PatternType.fromCode(record.patternType()),
42               record.principal(),
43               record.host(),
44               AclOperation.fromCode(record.operation()),
45               AclPermissionType.fromCode(record.permissionType()));
46       }
47   
48       public static StandardAcl fromAclBinding(AclBinding acl) {
49           return new StandardAcl(
50               acl.pattern().resourceType(),
51               acl.pattern().name(),
52               acl.pattern().patternType(),
53               acl.entry().principal(),
54               acl.entry().host(),
55               acl.entry().operation(),
56               acl.entry().permissionType());
57       }

59       private final ResourceType resourceType;
60       private final String resourceName;
61       private final PatternType patternType;
62       private final String principal;
63       private final String host;
64       private final AclOperation operation;
65       private final AclPermissionType permissionType;
66   
67       public StandardAcl(
68                   ResourceType resourceType,
69                   String resourceName,
70                   PatternType patternType,
71                   String principal,
72                   String host,
73                   AclOperation operation,
74                   AclPermissionType permissionType) {
75           this.resourceType = resourceType;
76           this.resourceName = resourceName;
77           this.patternType = patternType;
78           this.principal = principal;
79           this.host = host;
80           this.operation = operation;
81           this.permissionType = permissionType;
82       }

84       public ResourceType resourceType() {
85           return resourceType;
86       }
87   
88       public String resourceName() {
89           return resourceName;
90       }
91

93           return patternType;
94       }

97           return principal;
98       }
99   
100       public KafkaPrincipal kafkaPrincipal() {
101           int colonIndex = principal.indexOf(":");
102           if (colonIndex == -1) {
103               throw new IllegalStateException("Could not parse principal from `" + principal + "` " +
104                   "(no colon is present separating the principal type from the principal name)");
105           }
106           String principalType = principal.substring(0, colonIndex);
107           String principalName = principal.substring(colonIndex + 1);
108           return new KafkaPrincipal(principalType, principalName);
109       }
110   
111       public String host() {
112           return host;
113       }
114   
115       public AclOperation operation() {
116           return operation;
117       }
118   
119       public AclPermissionType permissionType() {
120           return permissionType;
121       }
122   
123       public AclBinding toBinding() {
124           ResourcePattern resourcePattern =
125               new ResourcePattern(resourceType, resourceName, patternType);
126           AccessControlEntry accessControlEntry =
127               new AccessControlEntry(principal, host, operation, permissionType);
128           return new AclBinding(resourcePattern, accessControlEntry);
129       }

131       @Override
132       public boolean equals(Object o) {
133           if (o == null || !o.getClass().equals(StandardAcl.class)) return false;
134           if (o == this) return true;
135           StandardAcl other = (StandardAcl) o;
136           return resourceType.equals(other.resourceType) &&
137               resourceName.equals(other.resourceName) &&
138               patternType.equals(other.patternType) &&
139               principal.equals(other.principal) &&
140               host.equals(other.host) &&
141               operation.equals(other.operation) &&
142               permissionType.equals(other.permissionType);
143       }

145       @Override

147           return Objects.hash(
148               resourceType,
149               resourceName,
150               patternType,
151               principal,
152               host,
153               operation,
154               permissionType);
155       }

157       /**
158        * Compare two StandardAcl objects. See {@link StandardAuthorizerData#authorize} for an
159        * explanation of why we want this particular sort order.
160        */
161       @Override
162       public int compareTo(StandardAcl other) {
163           int result;
164           result = resourceType.compareTo(other.resourceType);
165           if (result != 0) return result;
166           result = other.resourceName.compareTo(resourceName); // REVERSE sort by resource name.
167           if (result != 0) return result;
168           result = patternType.compareTo(other.patternType);
169           if (result != 0) return result;
170           result = operation.compareTo(other.operation);
171           if (result != 0) return result;
172           result = principal.compareTo(other.principal);
173           if (result != 0) return result;
174           result = host.compareTo(other.host);
175           if (result != 0) return result;
176           result = permissionType.compareTo(other.permissionType);
177           return result;
178       }

186               ", principal=" + principal +
187               ", host=" + host +
188               ", operation=" + operation +
189               ", permissionType=" + permissionType +
190               ")";
191       }
