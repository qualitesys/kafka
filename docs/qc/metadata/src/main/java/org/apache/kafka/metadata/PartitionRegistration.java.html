36   public class PartitionRegistration {
37       public final int[] replicas;

46       public static boolean electionWasClean(int newLeader, int[] isr) {
47           return newLeader == NO_LEADER || Replicas.contains(isr, newLeader);
48       }

50       public PartitionRegistration(PartitionRecord record) {
51           this(Replicas.toArray(record.replicas()),
52               Replicas.toArray(record.isr()),
53               Replicas.toArray(record.removingReplicas()),
54               Replicas.toArray(record.addingReplicas()),

56               LeaderRecoveryState.of(record.leaderRecoveryState()),
57               record.leaderEpoch(),
58               record.partitionEpoch());
59       }

61       public PartitionRegistration(int[] replicas, int[] isr, int[] removingReplicas,
62                                    int[] addingReplicas, int leader, LeaderRecoveryState leaderRecoveryState,
63                                    int leaderEpoch, int partitionEpoch) {
64           this.replicas = replicas;
65           this.isr = isr;

67           this.addingReplicas = addingReplicas;
68           this.leader = leader;
69           this.leaderRecoveryState = leaderRecoveryState;
70           this.leaderEpoch = leaderEpoch;
71           this.partitionEpoch = partitionEpoch;
72       }

74       public PartitionRegistration merge(PartitionChangeRecord record) {
75           int[] newReplicas = (record.replicas() == null) ?
76               replicas : Replicas.toArray(record.replicas());
77           int[] newIsr = (record.isr() == null) ? isr : Replicas.toArray(record.isr());
78           int[] newRemovingReplicas = (record.removingReplicas() == null) ?
79               removingReplicas : Replicas.toArray(record.removingReplicas());
80           int[] newAddingReplicas = (record.addingReplicas() == null) ?
81               addingReplicas : Replicas.toArray(record.addingReplicas());
82   
83           int newLeader;
84           int newLeaderEpoch;
85           if (record.leader() == NO_LEADER_CHANGE) {
86               newLeader = leader;
87               newLeaderEpoch = leaderEpoch;
88           } else {
89               newLeader = record.leader();
90               newLeaderEpoch = leaderEpoch + 1;
91           }
92   
93           LeaderRecoveryState newLeaderRecoveryState = leaderRecoveryState.changeTo(record.leaderRecoveryState());
94   
95           return new PartitionRegistration(newReplicas,
96               newIsr,
97               newRemovingReplicas,
98               newAddingReplicas,
99               newLeader,
100               newLeaderRecoveryState,
101               newLeaderEpoch,
102               partitionEpoch + 1);
103       }

105       public String diff(PartitionRegistration prev) {
106           StringBuilder builder = new StringBuilder();
107           String prefix = "";
108           if (!Arrays.equals(replicas, prev.replicas)) {
109               builder.append(prefix).append("replicas: ").
110                   append(Arrays.toString(prev.replicas)).
111                   append(" -> ").append(Arrays.toString(replicas));
112               prefix = ", ";
113           }
114           if (!Arrays.equals(isr, prev.isr)) {
115               builder.append(prefix).append("isr: ").
116                   append(Arrays.toString(prev.isr)).
117                   append(" -> ").append(Arrays.toString(isr));
118               prefix = ", ";
119           }
120           if (!Arrays.equals(removingReplicas, prev.removingReplicas)) {
121               builder.append(prefix).append("removingReplicas: ").
122                   append(Arrays.toString(prev.removingReplicas)).
123                   append(" -> ").append(Arrays.toString(removingReplicas));
124               prefix = ", ";
125           }
126           if (!Arrays.equals(addingReplicas, prev.addingReplicas)) {
127               builder.append(prefix).append("addingReplicas: ").
128                   append(Arrays.toString(prev.addingReplicas)).
129                   append(" -> ").append(Arrays.toString(addingReplicas));
130               prefix = ", ";
131           }
132           if (leader != prev.leader) {
133               builder.append(prefix).append("leader: ").
134                   append(prev.leader).append(" -> ").append(leader);
135               prefix = ", ";
136           }
137           if (leaderRecoveryState != prev.leaderRecoveryState) {
138               builder.append(prefix).append("leaderRecoveryState: ").
139                   append(prev.leaderRecoveryState).append(" -> ").append(leaderRecoveryState);
140               prefix = ", ";
141           }
142           if (leaderEpoch != prev.leaderEpoch) {
143               builder.append(prefix).append("leaderEpoch: ").
144                   append(prev.leaderEpoch).append(" -> ").append(leaderEpoch);
145               prefix = ", ";
146           }
147           if (partitionEpoch != prev.partitionEpoch) {
148               builder.append(prefix).append("partitionEpoch: ").
149                   append(prev.partitionEpoch).append(" -> ").append(partitionEpoch);
150           }
151           return builder.toString();
152       }

154       public void maybeLogPartitionChange(Logger log, String description, PartitionRegistration prev) {
155           if (!electionWasClean(leader, prev.isr)) {
156               log.info("UNCLEAN partition change for {}: {}", description, diff(prev));
157           } else if (log.isDebugEnabled()) {
158               log.debug("partition change for {}: {}", description, diff(prev));
159           }
160       }

170       public int preferredReplica() {
171           return replicas.length == 0 ? LeaderConstants.NO_LEADER : replicas[0];
172       }
