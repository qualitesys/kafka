65   @State(Scope.Benchmark)
66   @Fork(value = 1)
67   @Warmup(iterations = 5)
68   @Measurement(iterations = 15)
69   @BenchmarkMode(Mode.AverageTime)
70   @OutputTimeUnit(TimeUnit.MILLISECONDS)
71   public class AclAuthorizerBenchmark {
72       @Param({"10000", "50000", "200000"})
73       private int resourceCount;
74       //no. of. rules per resource
75       @Param({"10", "50"})
76       private int aclCount;
77   
78       @Param({"0", "20", "50", "90", "99", "99.9", "99.99", "100"})
79       private double denyPercentage;
80   
81       private final int hostPreCount = 1000;
82       private final String resourceNamePrefix = "foo-bar35_resource-";
83       private final AclAuthorizer aclAuthorizer = new AclAuthorizer();
84       private final KafkaPrincipal principal = new KafkaPrincipal(KafkaPrincipal.USER_TYPE, "test-user");
85       private List<Action> actions = new ArrayList<>();
86       private RequestContext authorizeContext;
87       private RequestContext authorizeByResourceTypeContext;
88       private String authorizeByResourceTypeHostName = "127.0.0.2";
89   
90       private HashMap<ResourcePattern, AclAuthorizer.VersionedAcls> aclToUpdate = new HashMap<>();
91   
92       Random rand = new Random(System.currentTimeMillis());
93       double eps = 1e-9;
94   
95       @Setup(Level.Trial)
96       public void setup() throws Exception {
97           prepareAclCache();
98           prepareAclToUpdate();
99           // By adding `-95` to the resource name prefix, we cause the `TreeMap.from/to` call to return
100           // most map entries. In such cases, we rely on the filtering based on `String.startsWith`
101           // to return the matching ACLs. Using a more efficient data structure (e.g. a prefix
102           // tree) should improve performance significantly).
103           actions = Collections.singletonList(new Action(AclOperation.WRITE,
104               new ResourcePattern(ResourceType.TOPIC, resourceNamePrefix + 95, PatternType.LITERAL),
105               1, true, true));
106           authorizeContext = new RequestContext(new RequestHeader(ApiKeys.PRODUCE, Integer.valueOf(1).shortValue(),
107               "someclient", 1), "1", InetAddress.getByName("127.0.0.1"), principal,
108               ListenerName.normalised("listener"), SecurityProtocol.PLAINTEXT, ClientInformation.EMPTY, false);
109           authorizeByResourceTypeContext = new RequestContext(new RequestHeader(ApiKeys.PRODUCE, Integer.valueOf(1).shortValue(),
110               "someclient", 1), "1", InetAddress.getByName(authorizeByResourceTypeHostName), principal,
111               ListenerName.normalised("listener"), SecurityProtocol.PLAINTEXT, ClientInformation.EMPTY, false);
112       }

114       private void prepareAclCache() {
115           Map<ResourcePattern, Set<AclEntry>> aclEntries = new HashMap<>();
116           for (int resourceId = 0; resourceId < resourceCount; resourceId++) {
117               ResourcePattern resource = new ResourcePattern(
118                   (resourceId % 10 == 0) ? ResourceType.GROUP : ResourceType.TOPIC,
119                   resourceNamePrefix + resourceId,
120                   (resourceId % 5 == 0) ? PatternType.PREFIXED : PatternType.LITERAL);
121   
122               Set<AclEntry> entries = aclEntries.computeIfAbsent(resource, k -> new HashSet<>());
123   
124               for (int aclId = 0; aclId < aclCount; aclId++) {
125                   // The principal in the request context we are using
126                   // is principal.toString without any suffix
127                   String principalName = principal.toString() + (aclId == 0 ? "" : aclId);
128                   AccessControlEntry allowAce = new AccessControlEntry(
129                       principalName, "*", AclOperation.READ, AclPermissionType.ALLOW);
130   
131                   entries.add(new AclEntry(allowAce));
132   
133                   if (shouldDeny()) {
134                       // dominantly deny the resource
135                       AccessControlEntry denyAce = new AccessControlEntry(
136                           principalName, "*", AclOperation.READ, AclPermissionType.DENY);
137                       entries.add(new AclEntry(denyAce));
138                   }
139               }
140           }
141   
142           ResourcePattern resourcePrefix = new ResourcePattern(ResourceType.TOPIC, resourceNamePrefix,
143               PatternType.PREFIXED);
144           Set<AclEntry> entriesPrefix = aclEntries.computeIfAbsent(resourcePrefix, k -> new HashSet<>());
145           for (int hostId = 0; hostId < hostPreCount; hostId++) {
146               AccessControlEntry allowAce = new AccessControlEntry(principal.toString(), "127.0.0." + hostId,
147                   AclOperation.READ, AclPermissionType.ALLOW);
148               entriesPrefix.add(new AclEntry(allowAce));
149   
150               if (shouldDeny()) {
151                   // dominantly deny the resource
152                   AccessControlEntry denyAce = new AccessControlEntry(principal.toString(), "127.0.0." + hostId,
153                       AclOperation.READ, AclPermissionType.DENY);
154                   entriesPrefix.add(new AclEntry(denyAce));
155               }
156           }
157   
158           ResourcePattern resourceWildcard = new ResourcePattern(ResourceType.TOPIC, ResourcePattern.WILDCARD_RESOURCE,
159               PatternType.LITERAL);
160           Set<AclEntry> entriesWildcard = aclEntries.computeIfAbsent(resourceWildcard, k -> new HashSet<>());
161           // get dynamic entries number for wildcard acl
162           for (int hostId = 0; hostId < resourceCount / 10; hostId++) {
163               String hostName = "127.0.0" + hostId;
164               // AuthorizeByResourceType is optimizing the wildcard deny case.
165               // If we didn't skip the host, we would end up having a biased short runtime.
166               if (hostName.equals(authorizeByResourceTypeHostName)) {
167                   continue;
168               }
169   
170               AccessControlEntry allowAce = new AccessControlEntry(principal.toString(), hostName,
171                   AclOperation.READ, AclPermissionType.ALLOW);
172               entriesWildcard.add(new AclEntry(allowAce));
173               if (shouldDeny()) {
174                   AccessControlEntry denyAce = new AccessControlEntry(principal.toString(), hostName,
175                       AclOperation.READ, AclPermissionType.DENY);
176                   entriesWildcard.add(new AclEntry(denyAce));
177               }
178           }
179   
180           for (Map.Entry<ResourcePattern, Set<AclEntry>> entryMap : aclEntries.entrySet()) {
181               aclAuthorizer.updateCache(entryMap.getKey(),
182                   new VersionedAcls(JavaConverters.asScalaSetConverter(entryMap.getValue()).asScala().toSet(), 1));
183           }
184       }
185   
186       private void prepareAclToUpdate() {
187           scala.collection.mutable.Set<AclEntry> entries = new scala.collection.mutable.HashSet<>();
188           for (int i = 0; i < resourceCount; i++) {
189               scala.collection.immutable.Set<AclEntry> immutable = new scala.collection.immutable.HashSet<>();
190               for (int j = 0; j < aclCount; j++) {
191                   entries.add(new AclEntry(new AccessControlEntry(
192                       principal.toString(), "127.0.0" + j, AclOperation.WRITE, AclPermissionType.ALLOW)));
193                   immutable = entries.toSet();
194               }
195               aclToUpdate.put(
196                   new ResourcePattern(ResourceType.TOPIC, randomResourceName(resourceNamePrefix), PatternType.LITERAL),
197                   new AclAuthorizer.VersionedAcls(immutable, i));
198           }
199       }

201       private String randomResourceName(String prefix) {
202           return prefix + UUID.randomUUID().toString().substring(0, 5);
203       }
204   
205       private Boolean shouldDeny() {
206           return rand.nextDouble() * 100.0 - eps < denyPercentage;
207       }
