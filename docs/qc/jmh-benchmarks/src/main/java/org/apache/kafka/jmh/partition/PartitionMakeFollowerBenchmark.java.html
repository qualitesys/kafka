73   @State(Scope.Benchmark)
74   @Fork(value = 1)
75   @Warmup(iterations = 5)
76   @Measurement(iterations = 15)
77   @BenchmarkMode(Mode.AverageTime)

80   public class PartitionMakeFollowerBenchmark {
81       private LogManager logManager;
82       private File logDir = new File(System.getProperty("java.io.tmpdir"), UUID.randomUUID().toString());
83       private KafkaScheduler scheduler = new KafkaScheduler(1, "scheduler", true);
84       private Partition partition;
85       private List<Integer> replicas = Arrays.asList(0, 1, 2);
86       private OffsetCheckpoints offsetCheckpoints = Mockito.mock(OffsetCheckpoints.class);
87       private DelayedOperations delayedOperations  = Mockito.mock(DelayedOperations.class);
88       private ExecutorService executorService = Executors.newSingleThreadExecutor();
89       private Option<Uuid> topicId;
90   
91       @Setup(Level.Trial)
92       public void setup() throws IOException {
93           if (!logDir.mkdir())
94               throw new IOException("error creating test directory");
95   
96           scheduler.startup();
97           LogConfig logConfig = createLogConfig();
98   
99           BrokerTopicStats brokerTopicStats = new BrokerTopicStats();
100           LogDirFailureChannel logDirFailureChannel = Mockito.mock(LogDirFailureChannel.class);
101           logManager = new LogManagerBuilder().
102               setLogDirs(Collections.singletonList(logDir)).
103               setInitialOfflineDirs(Collections.emptyList()).
104               setConfigRepository(new MockConfigRepository()).
105               setInitialDefaultConfig(logConfig).
106               setCleanerConfig(new CleanerConfig(0, 0, 0, 0, 0, 0.0, 0, false, "MD5")).
107               setRecoveryThreadsPerDataDir(1).
108               setFlushCheckMs(1000L).
109               setFlushRecoveryOffsetCheckpointMs(10000L).
110               setFlushStartOffsetCheckpointMs(10000L).
111               setRetentionCheckMs(1000L).
112               setMaxPidExpirationMs(60000).
113               setInterBrokerProtocolVersion(MetadataVersion.latest()).
114               setScheduler(scheduler).
115               setBrokerTopicStats(brokerTopicStats).
116               setLogDirFailureChannel(logDirFailureChannel).
117               setTime(Time.SYSTEM).setKeepPartitionMetadataFile(true).
118               build();
119   
120           TopicPartition tp = new TopicPartition("topic", 0);
121           topicId = OptionConverters.toScala(Optional.of(Uuid.randomUuid()));
122   
123           Mockito.when(offsetCheckpoints.fetch(logDir.getAbsolutePath(), tp)).thenReturn(Option.apply(0L));
124           AlterPartitionListener alterPartitionListener = Mockito.mock(AlterPartitionListener.class);
125           AlterPartitionManager alterPartitionManager = Mockito.mock(AlterPartitionManager.class);
126           partition = new Partition(tp, 100,
127               MetadataVersion.latest(), 0, Time.SYSTEM,
128               alterPartitionListener, delayedOperations,
129               Mockito.mock(MetadataCache.class), logManager, alterPartitionManager);
130           partition.createLogIfNotExists(true, false, offsetCheckpoints, topicId);
131           executorService.submit((Runnable) () -> {
132               SimpleRecord[] simpleRecords = new SimpleRecord[] {
133                   new SimpleRecord(1L, "foo".getBytes(StandardCharsets.UTF_8), "1".getBytes(StandardCharsets.UTF_8)),
134                   new SimpleRecord(2L, "bar".getBytes(StandardCharsets.UTF_8), "2".getBytes(StandardCharsets.UTF_8))
135               };
136               int initialOffSet = 0;
137               while (true) {
138                   MemoryRecords memoryRecords =  MemoryRecords.withRecords(initialOffSet, CompressionType.NONE, 0, simpleRecords);
139                   partition.appendRecordsToFollowerOrFutureReplica(memoryRecords, false);
140                   initialOffSet = initialOffSet + 2;
141               }
142           });
143       }
144   
145       @TearDown(Level.Trial)
146       public void tearDown() throws IOException {
147           executorService.shutdownNow();
148           logManager.shutdown();
149           scheduler.shutdown();
150           Utils.delete(logDir);
151       }

153       @Benchmark
154       public boolean testMakeFollower() {
155           LeaderAndIsrRequestData.LeaderAndIsrPartitionState partitionState = new LeaderAndIsrRequestData.LeaderAndIsrPartitionState()
156               .setControllerEpoch(0)
157               .setLeader(0)
158               .setLeaderEpoch(0)
159               .setIsr(replicas)
160               .setPartitionEpoch(1)
161               .setReplicas(replicas)
162               .setIsNew(true);
163           return partition.makeFollower(partitionState, offsetCheckpoints, topicId);
164       }
