45   @State(Scope.Benchmark)
46   @Fork(value = 1)
47   @Warmup(iterations = 5)
48   @Measurement(iterations = 15)
49   @BenchmarkMode(Mode.AverageTime)
50   @OutputTimeUnit(TimeUnit.MILLISECONDS)
51   public class SubscriptionStateBenchmark {
52       @Param({"5000"})
53       int topicCount;
54   
55       @Param({"50"})
56       int partitionCount;
57   
58       SubscriptionState subscriptionState;
59   
60       @Setup(Level.Trial)
61       public void setup() {
62           Set<TopicPartition> assignment = new HashSet<>(topicCount * partitionCount);
63           IntStream.range(0, topicCount).forEach(topicId ->
64               IntStream.range(0, partitionCount).forEach(partitionId ->
65                   assignment.add(new TopicPartition(String.format("topic-%04d", topicId), partitionId))
66               )
67           );
68           subscriptionState = new SubscriptionState(new LogContext(), OffsetResetStrategy.EARLIEST);
69           subscriptionState.assignFromUser(assignment);
70           SubscriptionState.FetchPosition position = new SubscriptionState.FetchPosition(
71               0L,
72               Optional.of(0),
73               new Metadata.LeaderAndEpoch(Optional.of(new Node(0, "host", 9092)), Optional.of(10))
74           );
75           assignment.forEach(topicPartition -> {
76               subscriptionState.seekUnvalidated(topicPartition, position);
77               subscriptionState.completeValidation(topicPartition);
78           });
79       }
80   
81       @Benchmark
82       public boolean testHasAllFetchPositions() {
83           return subscriptionState.hasAllFetchPositions();
84       }
85   
86       @Benchmark
87       public int testFetchablePartitions() {
88           return subscriptionState.fetchablePartitions(tp -> true).size();
89       }
90   
91       @Benchmark
92       public int testPartitionsNeedingValidation() {
93           return subscriptionState.partitionsNeedingValidation(0L).size();
94       }
95   }
