134       private void initializeMetadataCache() {
135           List<UpdateMetadataBroker> liveBrokers = new LinkedList<>();
136           List<UpdateMetadataPartitionState> partitionStates = new LinkedList<>();
137   
138           IntStream.range(0, 5).forEach(brokerId -> liveBrokers.add(
139               new UpdateMetadataBroker().setId(brokerId)
140                   .setEndpoints(endpoints(brokerId))
141                   .setRack("rack1")));
142   
143           IntStream.range(0, topicCount).forEach(topicId -> {
144               String topicName = "topic-" + topicId;
145   
146               IntStream.range(0, partitionCount).forEach(partitionId -> {
147                   partitionStates.add(
148                       new UpdateMetadataPartitionState().setTopicName(topicName)
149                           .setPartitionIndex(partitionId)
150                           .setControllerEpoch(1)
151                           .setLeader(partitionCount % 5)
152                           .setLeaderEpoch(0)
153                           .setIsr(Arrays.asList(0, 1, 3))
154                           .setZkVersion(1)
155                           .setReplicas(Arrays.asList(0, 1, 3)));
156               });
157           });
158   
159           UpdateMetadataRequest updateMetadataRequest = new UpdateMetadataRequest.Builder(
160               ApiKeys.UPDATE_METADATA.latestVersion(),
161               1, 1, 1,
162               partitionStates, liveBrokers, Collections.emptyMap()).build();
163           metadataCache.updateMetadata(100, updateMetadataRequest);
164       }

166       private List<UpdateMetadataEndpoint> endpoints(final int brokerId) {
167           return Collections.singletonList(
168               new UpdateMetadataEndpoint()
169                   .setHost("host_" + brokerId)
170                   .setPort(9092)
171                   .setSecurityProtocol(SecurityProtocol.PLAINTEXT.id)
172                   .setListener(ListenerName.forSecurityProtocol(SecurityProtocol.PLAINTEXT).value()));
173       }

175       private KafkaApis createKafkaApis() {
176           Properties kafkaProps =  new Properties();
177           kafkaProps.put(KafkaConfig$.MODULE$.ZkConnectProp(), "zk");
178           kafkaProps.put(KafkaConfig$.MODULE$.BrokerIdProp(), brokerId + "");
179           KafkaConfig config = new KafkaConfig(kafkaProps);
180           return new KafkaApisBuilder().
181               setRequestChannel(requestChannel).
182               setMetadataSupport(new ZkSupport(adminManager, kafkaController, kafkaZkClient, Option.empty(), metadataCache)).
183               setReplicaManager(replicaManager).
184               setGroupCoordinator(groupCoordinator).
185               setTxnCoordinator(transactionCoordinator).
186               setAutoTopicCreationManager(autoTopicCreationManager).
187               setBrokerId(brokerId).
188               setConfig(config).
189               setConfigRepository(new MockConfigRepository()).
190               setMetadataCache(metadataCache).
191               setMetrics(metrics).
192               setAuthorizer(Optional.empty()).
193               setQuotas(quotaManagers).
194               setFetchManager(fetchManager).
195               setBrokerTopicStats(brokerTopicStats).
196               setClusterId("clusterId").
197               setTime(Time.SYSTEM).
198               setTokenManager(null).
199               setApiVersionManager(new SimpleApiVersionManager(ApiMessageType.ListenerType.ZK_BROKER)).
200               build();
201       }

209       private RequestChannel.Request buildAllTopicMetadataRequest() {
210           MetadataRequest metadataRequest = MetadataRequest.Builder.allTopics().build();
211           RequestHeader header = new RequestHeader(metadataRequest.apiKey(), metadataRequest.version(), "", 0);
212           ByteBuffer bodyBuffer = metadataRequest.serialize();
213   
214           RequestContext context = new RequestContext(header, "1", null, principal,
215               ListenerName.forSecurityProtocol(SecurityProtocol.PLAINTEXT),
216               SecurityProtocol.PLAINTEXT, ClientInformation.EMPTY, false);
217           return new RequestChannel.Request(1, context, 0, MemoryPool.NONE, bodyBuffer, requestChannelMetrics, Option.empty());
218       }
