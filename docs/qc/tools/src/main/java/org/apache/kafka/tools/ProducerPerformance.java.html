54       void start(String[] args) throws IOException {
55           ArgumentParser parser = argParser();
56   
57           try {
58               Namespace res = parser.parseArgs(args);
59   
60               /* parse args */
61               String topicName = res.getString("topic");
62               long numRecords = res.getLong("numRecords");
63               Integer recordSize = res.getInt("recordSize");
64               int throughput = res.getInt("throughput");
65               List<String> producerProps = res.getList("producerConfig");
66               String producerConfig = res.getString("producerConfigFile");
67               String payloadFilePath = res.getString("payloadFile");
68               String transactionalId = res.getString("transactionalId");
69               boolean shouldPrintMetrics = res.getBoolean("printMetrics");
70               long transactionDurationMs = res.getLong("transactionDurationMs");
71               boolean transactionsEnabled =  0 < transactionDurationMs;
72   
73               // since default value gets printed with the help text, we are escaping \n there and replacing it with correct value here.
74               String payloadDelimiter = res.getString("payloadDelimiter").equals("\\n") ? "\n" : res.getString("payloadDelimiter");
75   
76               if (producerProps == null && producerConfig == null) {
77                   throw new ArgumentParserException("Either --producer-props or --producer.config must be specified.", parser);
78               }
79   
80               List<byte[]> payloadByteList = readPayloadFile(payloadFilePath, payloadDelimiter);
81   
82               Properties props = readProps(producerProps, producerConfig, transactionalId, transactionsEnabled);
83   
84               KafkaProducer<byte[], byte[]> producer = createKafkaProducer(props);
85   
86               if (transactionsEnabled)
87                   producer.initTransactions();
88   
89               /* setup perf test */
90               byte[] payload = null;
91               if (recordSize != null) {
92                   payload = new byte[recordSize];
93               }
94               Random random = new Random(0);
95               ProducerRecord<byte[], byte[]> record;
96               Stats stats = new Stats(numRecords, 5000);
97               long startMs = System.currentTimeMillis();
98   
99               ThroughputThrottler throttler = new ThroughputThrottler(throughput, startMs);
100   
101               int currentTransactionSize = 0;
102               long transactionStartTime = 0;
103               for (long i = 0; i < numRecords; i++) {
104   
105                   payload = generateRandomPayload(recordSize, payloadByteList, payload, random);
106   
107                   if (transactionsEnabled && currentTransactionSize == 0) {
108                       producer.beginTransaction();
109                       transactionStartTime = System.currentTimeMillis();
110                   }
111   
112                   record = new ProducerRecord<>(topicName, payload);
113   
114                   long sendStartMs = System.currentTimeMillis();
115                   Callback cb = stats.nextCompletion(sendStartMs, payload.length, stats);
116                   producer.send(record, cb);
117   
118                   currentTransactionSize++;
119                   if (transactionsEnabled && transactionDurationMs <= (sendStartMs - transactionStartTime)) {
120                       producer.commitTransaction();
121                       currentTransactionSize = 0;
122                   }
123   
124                   if (throttler.shouldThrottle(i, sendStartMs)) {
125                       throttler.throttle();
126                   }
127               }
128   
129               if (transactionsEnabled && currentTransactionSize != 0)
130                   producer.commitTransaction();
131   
132               if (!shouldPrintMetrics) {
133                   producer.close();
134   
135                   /* print final results */
136                   stats.printTotal();
137               } else {
138                   // Make sure all messages are sent before printing out the stats and the metrics
139                   // We need to do this in a different branch for now since tests/kafkatest/sanity_checks/test_performance_services.py
140                   // expects this class to work with older versions of the client jar that don't support flush().
141                   producer.flush();
142   
143                   /* print final results */
144                   stats.printTotal();
145   
146                   /* print out metrics */
147                   ToolsUtils.printMetrics(producer.metrics());
148                   producer.close();
149               }
150           } catch (ArgumentParserException e) {
151               if (args.length == 0) {
152                   parser.printHelp();
153                   Exit.exit(0);
154               } else {
155                   parser.handleError(e);
156                   Exit.exit(1);
157               }
158           }

166       static byte[] generateRandomPayload(Integer recordSize, List<byte[]> payloadByteList, byte[] payload,
167               Random random) {
168           if (!payloadByteList.isEmpty()) {
169               payload = payloadByteList.get(random.nextInt(payloadByteList.size()));
170           } else if (recordSize != null) {
171               for (int j = 0; j < payload.length; ++j)
172                   payload[j] = (byte) (random.nextInt(26) + 65);
173           } else {
174               throw new IllegalArgumentException("no payload File Path or record Size provided");
175           }
176           return payload;
177       }

179       static Properties readProps(List<String> producerProps, String producerConfig, String transactionalId,
180               boolean transactionsEnabled) throws IOException {
181           Properties props = new Properties();
182           if (producerConfig != null) {
183               props.putAll(Utils.loadProps(producerConfig));
184           }
185           if (producerProps != null)
186               for (String prop : producerProps) {
187                   String[] pieces = prop.split("=");
188                   if (pieces.length != 2)
189                       throw new IllegalArgumentException("Invalid property: " + prop);
190                   props.put(pieces[0], pieces[1]);
191               }
192   
193           props.put(ProducerConfig.KEY_SERIALIZER_CLASS_CONFIG, "org.apache.kafka.common.serialization.ByteArraySerializer");
194           props.put(ProducerConfig.VALUE_SERIALIZER_CLASS_CONFIG, "org.apache.kafka.common.serialization.ByteArraySerializer");
195           if (transactionsEnabled) props.put(ProducerConfig.TRANSACTIONAL_ID_CONFIG, transactionalId);
196           if (props.getProperty(ProducerConfig.CLIENT_ID_CONFIG) == null) {
197               props.put(ProducerConfig.CLIENT_ID_CONFIG, "perf-producer-client");
198           }
199           return props;
200       }

202       static List<byte[]> readPayloadFile(String payloadFilePath, String payloadDelimiter) throws IOException {
203           List<byte[]> payloadByteList = new ArrayList<>();
204           if (payloadFilePath != null) {
205               Path path = Paths.get(payloadFilePath);
206               System.out.println("Reading payloads from: " + path.toAbsolutePath());
207               if (Files.notExists(path) || Files.size(path) == 0)  {
208                   throw new IllegalArgumentException("File does not exist or empty file provided.");
209               }
210   
211               String[] payloadList = new String(Files.readAllBytes(path), StandardCharsets.UTF_8).split(payloadDelimiter);
212   
213               System.out.println("Number of messages read: " + payloadList.length);
214   
215               for (String payload : payloadList) {
216                   payloadByteList.add(payload.getBytes(StandardCharsets.UTF_8));
217               }
218           }
219           return payloadByteList;
220       }

331       private static class Stats {
332           private long start;
333           private long windowStart;
334           private int[] latencies;
335           private int sampling;
336           private int iteration;
337           private int index;
338           private long count;
339           private long bytes;
340           private int maxLatency;
341           private long totalLatency;
342           private long windowCount;
343           private int windowMaxLatency;
344           private long windowTotalLatency;
345           private long windowBytes;
346           private long reportingInterval;
347   
348           public Stats(long numRecords, int reportingInterval) {
349               this.start = System.currentTimeMillis();
350               this.windowStart = System.currentTimeMillis();
351               this.iteration = 0;
352               this.sampling = (int) (numRecords / Math.min(numRecords, 500000));
353               this.latencies = new int[(int) (numRecords / this.sampling) + 1];
354               this.index = 0;
355               this.maxLatency = 0;
356               this.totalLatency = 0;
357               this.windowCount = 0;
358               this.windowMaxLatency = 0;
359               this.windowTotalLatency = 0;
360               this.windowBytes = 0;
361               this.totalLatency = 0;
362               this.reportingInterval = reportingInterval;
363           }
364   
365           public void record(int iter, int latency, int bytes, long time) {
366               this.count++;
367               this.bytes += bytes;
368               this.totalLatency += latency;
369               this.maxLatency = Math.max(this.maxLatency, latency);
370               this.windowCount++;
371               this.windowBytes += bytes;
372               this.windowTotalLatency += latency;
373               this.windowMaxLatency = Math.max(windowMaxLatency, latency);
374               if (iter % this.sampling == 0) {
375                   this.latencies[index] = latency;
376                   this.index++;
377               }
378               /* maybe report the recent perf */
379               if (time - windowStart >= reportingInterval) {
380                   printWindow();
381                   newWindow();
382               }
383           }

385           public Callback nextCompletion(long start, int bytes, Stats stats) {
386               Callback cb = new PerfCallback(this.iteration, start, bytes, stats);
387               this.iteration++;
388               return cb;
389           }

391           public void printWindow() {
392               long elapsed = System.currentTimeMillis() - windowStart;
393               double recsPerSec = 1000.0 * windowCount / (double) elapsed;
394               double mbPerSec = 1000.0 * this.windowBytes / (double) elapsed / (1024.0 * 1024.0);
395               System.out.printf("%d records sent, %.1f records/sec (%.2f MB/sec), %.1f ms avg latency, %.1f ms max latency.%n",
396                                 windowCount,
397                                 recsPerSec,
398                                 mbPerSec,
399                                 windowTotalLatency / (double) windowCount,
400                                 (double) windowMaxLatency);
401           }

405               this.windowCount = 0;
406               this.windowMaxLatency = 0;
407               this.windowTotalLatency = 0;
408               this.windowBytes = 0;
409           }

411           public void printTotal() {
412               long elapsed = System.currentTimeMillis() - start;
413               double recsPerSec = 1000.0 * count / (double) elapsed;
414               double mbPerSec = 1000.0 * this.bytes / (double) elapsed / (1024.0 * 1024.0);
415               int[] percs = percentiles(this.latencies, index, 0.5, 0.95, 0.99, 0.999);
416               System.out.printf("%d records sent, %f records/sec (%.2f MB/sec), %.2f ms avg latency, %.2f ms max latency, %d ms 50th, %d ms 95th, %d ms 99th, %d ms 99.9th.%n",
417                                 count,
418                                 recsPerSec,
419                                 mbPerSec,
420                                 totalLatency / (double) count,
421                                 (double) maxLatency,
422                                 percs[0],
423                                 percs[1],
424                                 percs[2],
425                                 percs[3]);
426           }

428           private static int[] percentiles(int[] latencies, int count, double... percentiles) {
429               int size = Math.min(count, latencies.length);
430               Arrays.sort(latencies, 0, size);
431               int[] values = new int[percentiles.length];
432               for (int i = 0; i < percentiles.length; i++) {
433                   int index = (int) (percentiles[i] * size);
434                   values[i] = latencies[index];
435               }
436               return values;
437           }
438       }

440       private static final class PerfCallback implements Callback {
441           private final long start;
442           private final int iteration;
443           private final int bytes;
444           private final Stats stats;
445   
446           public PerfCallback(int iter, long start, int bytes, Stats stats) {
447               this.start = start;
448               this.stats = stats;
449               this.iteration = iter;
450               this.bytes = bytes;
451           }

453           public void onCompletion(RecordMetadata metadata, Exception exception) {
454               long now = System.currentTimeMillis();
455               int latency = (int) (now - start);
456               this.stats.record(iteration, latency, bytes, now);
457               if (exception != null)
458                   exception.printStackTrace();
459           }
460       }
