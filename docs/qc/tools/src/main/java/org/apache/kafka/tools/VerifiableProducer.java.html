46   /**
47    * Primarily intended for use with system testing, this producer prints metadata
48    * in the form of JSON to stdout on each "send" request. For example, this helps
49    * with end-to-end correctness tests by making externally visible which messages have been
50    * acked and which have not.
51    *
52    * When used as a command-line tool, it produces increasing integers. It will produce a
53    * fixed number of messages unless the default max-messages -1 is used, in which case
54    * it produces indefinitely.
55    *
56    * If logging is left enabled, log output on stdout can be easily ignored by checking
57    * whether a given line is valid JSON.
58    */
59   public class VerifiableProducer implements AutoCloseable {
60

62       private final String topic;
63       private final Producer<String, String> producer;
64       // If maxMessages < 0, produce until the process is killed externally
65       private long maxMessages = -1;
66   
67       // Number of messages for which acks were received
68       private long numAcked = 0;
69   
70       // Number of send attempts
71       private long numSent = 0;
72   
73       // Throttle message throughput if this is set >= 0
74       private final long throughput;
75

77       private boolean stopProducing = false;
78

82

84       // each message produced when number specified is reached

219       /** Construct a VerifiableProducer object from command-line arguments. */
220       public static VerifiableProducer createFromArgs(ArgumentParser parser, String[] args) throws ArgumentParserException {
221           Namespace res = parser.parseArgs(args);
222   
223           int maxMessages = res.getInt("maxMessages");
224           String topic = res.getString("topic");
225           int throughput = res.getInt("throughput");
226           String configFile = res.getString("producer.config");
227           Integer valuePrefix = res.getInt("valuePrefix");
228           Long createTime = res.getLong("createTime");
229           Integer repeatingKeys = res.getInt("repeatingKeys");
230   
231           if (createTime == -1L)
232               createTime = null;
233   
234           Properties producerProps = new Properties();
235   
236           if (res.get("bootstrapServer") != null) {
237               producerProps.put(ProducerConfig.BOOTSTRAP_SERVERS_CONFIG, res.getString("bootstrapServer"));
238           } else if (res.getString("brokerList") != null) {
239               producerProps.put(ProducerConfig.BOOTSTRAP_SERVERS_CONFIG, res.getString("brokerList"));
240           } else {
241               parser.printHelp();
242               // Can't use `Exit.exit` here because it didn't exist until 0.11.0.0.
243               System.exit(0);
244           }
245   
246           producerProps.put(ProducerConfig.KEY_SERIALIZER_CLASS_CONFIG,
247                   "org.apache.kafka.common.serialization.StringSerializer");
248           producerProps.put(ProducerConfig.VALUE_SERIALIZER_CLASS_CONFIG,
249                   "org.apache.kafka.common.serialization.StringSerializer");
250           producerProps.put(ProducerConfig.ACKS_CONFIG, Integer.toString(res.getInt("acks")));
251           // No producer retries
252           producerProps.put(ProducerConfig.RETRIES_CONFIG, "0");
253           if (configFile != null) {
254               try {
255                   producerProps.putAll(loadProps(configFile));
256               } catch (IOException e) {
257                   throw new ArgumentParserException(e.getMessage(), parser);
258               }
259           }
260   
261           StringSerializer serializer = new StringSerializer();
262           KafkaProducer<String, String> producer = new KafkaProducer<>(producerProps, serializer, serializer);
263   
264           return new VerifiableProducer(producer, topic, throughput, maxMessages, valuePrefix, createTime, repeatingKeys);
265       }

267       /** Produce a message with given key and value. */
268       public void send(String key, String value) {
269           ProducerRecord<String, String> record;
270   
271           // Older versions of ProducerRecord don't include the message create time in the constructor. So including
272           // even a 'null' argument results in a NoSuchMethodException. Thus we only include the create time if it is
273           // explicitly specified to remain fully backward compatible with older clients.
274           if (createTime != null) {
275               record = new ProducerRecord<>(topic, null, createTime, key, value);
276               createTime += System.currentTimeMillis() - startTime;
277           } else {
278               record = new ProducerRecord<>(topic, key, value);
279           }
280   
281           numSent++;
282           try {
283               producer.send(record, new PrintInfoCallback(key, value));
284           } catch (Exception e) {
285   
286               synchronized (System.out) {
287                   printJson(new FailedSend(key, value, topic, e));
288               }
289           }
290       }

495           public void onCompletion(RecordMetadata recordMetadata, Exception e) {
496               synchronized (System.out) {
497                   if (e == null) {
498                       VerifiableProducer.this.numAcked++;
499                       printJson(new SuccessfulSend(this.key, this.value, recordMetadata));
500                   } else {
501                       printJson(new FailedSend(this.key, this.value, topic, e));
502                   }
503               }
504           }
505       }

507       public void run(ThroughputThrottler throttler) {
508   
509           printJson(new StartupComplete());
510           // negative maxMessages (-1) means "infinite"
511           long maxMessages = (this.maxMessages < 0) ? Long.MAX_VALUE : this.maxMessages;
512   
513           for (long i = 0; i < maxMessages; i++) {
514               if (this.stopProducing) {
515                   break;
516               }
517               long sendStartMs = System.currentTimeMillis();
518   
519               this.send(this.getKey(), this.getValue(i));
520   
521               if (throttler.shouldThrottle(i, sendStartMs)) {
522                   throttler.throttle();
523               }
524           }
525       }
