153           private AbortTransactionSpec buildAbortSpec(
154               Admin admin,
155               TopicPartition topicPartition,
156               long startOffset
157           ) throws Exception {
158               final DescribeProducersResult.PartitionProducerState result;
159               try {
160                   result = admin.describeProducers(singleton(topicPartition))
161                       .partitionResult(topicPartition)
162                       .get();
163               } catch (ExecutionException e) {
164                   printErrorAndExit("Failed to validate producer state for partition "
165                       + topicPartition, e.getCause());
166                   return null;
167               }
168   
169               Optional<ProducerState> foundProducerState = result.activeProducers().stream()
170                   .filter(producerState -> {
171                       OptionalLong txnStartOffsetOpt = producerState.currentTransactionStartOffset();
172                       return txnStartOffsetOpt.isPresent() && txnStartOffsetOpt.getAsLong() == startOffset;
173                   })
174                   .findFirst();
175   
176               if (!foundProducerState.isPresent()) {
177                   printErrorAndExit("Could not find any open transactions starting at offset " +
178                       startOffset + " on partition " + topicPartition);
179                   return null;
180               }
181   
182               ProducerState producerState = foundProducerState.get();
183               return new AbortTransactionSpec(
184                   topicPartition,
185                   producerState.producerId(),
186                   (short) producerState.producerEpoch(),
187                   producerState.coordinatorEpoch().orElse(0)
188               );
189           }

191           private void abortTransaction(
192               Admin admin,
193               AbortTransactionSpec abortSpec
194           ) throws Exception {
195               try {
196                   admin.abortTransaction(abortSpec).all().get();
197               } catch (ExecutionException e) {
198                   TransactionsCommand.printErrorAndExit("Failed to abort transaction " + abortSpec, e.getCause());
199               }
200           }

202           @Override
203           void execute(Admin admin, Namespace ns, PrintStream out) throws Exception {
204               String topicName = ns.getString("topic");
205               Integer partitionId = ns.getInt("partition");
206               TopicPartition topicPartition = new TopicPartition(topicName, partitionId);
207   
208               Long startOffset = ns.getLong("start_offset");
209               Long producerId = ns.getLong("producer_id");
210   
211               if (startOffset == null && producerId == null) {
212                   printErrorAndExit("The transaction to abort must be identified either with " +
213                       "--start-offset (for brokers on 3.0 or above) or with " +
214                       "--producer-id, --producer-epoch, and --coordinator-epoch (for older brokers)");
215                   return;
216               }
217   
218               final AbortTransactionSpec abortSpec;
219               if (startOffset == null) {
220                   Short producerEpoch = ns.getShort("producer_epoch");
221                   if (producerEpoch == null) {
222                       printErrorAndExit("Missing required argument --producer-epoch");
223                       return;
224                   }
225   
226                   Integer coordinatorEpoch = ns.getInt("coordinator_epoch");
227                   if (coordinatorEpoch == null) {
228                       printErrorAndExit("Missing required argument --coordinator-epoch");
229                       return;
230                   }
231   
232                   // If a transaction was started by a new producerId and became hanging
233                   // before the initial commit/abort, then the coordinator epoch will be -1
234                   // as seen in the `DescribeProducers` output. In this case, we conservatively
235                   // use a coordinator epoch of 0, which is less than or equal to any possible
236                   // leader epoch.
237                   if (coordinatorEpoch < 0) {
238                       coordinatorEpoch = 0;
239                   }
240   
241                   abortSpec = new AbortTransactionSpec(
242                       topicPartition,
243                       producerId,
244                       producerEpoch,
245                       coordinatorEpoch
246                   );
247               } else {
248                   abortSpec = buildAbortSpec(admin, topicPartition, startOffset);
249               }
250   
251               abortTransaction(admin, abortSpec);
252           }
253       }

613           private List<OpenTransaction> filterHangingTransactions(
614               Map<Long, List<OpenTransaction>> openTransactionsByProducerId,
615               Map<Long, String> transactionalIds,
616               Map<String, TransactionDescription> descriptions
617           ) {
618               List<OpenTransaction> hangingTransactions = new ArrayList<>();
619   
620               openTransactionsByProducerId.forEach((producerId, openTransactions) -> {
621                   String transactionalId = transactionalIds.get(producerId);
622                   if (transactionalId == null) {
623                       // If we could not find the transactionalId corresponding to the
624                       // producerId of an open transaction, then the transaction is hanging.
625                       hangingTransactions.addAll(openTransactions);
626                   } else {
627                       // Otherwise, we need to check the current transaction state.
628                       TransactionDescription description = descriptions.get(transactionalId);
629                       if (description == null) {
630                           hangingTransactions.addAll(openTransactions);
631                       } else {
632                           for (OpenTransaction openTransaction : openTransactions) {
633                               // The `DescribeTransactions` API returns all partitions being
634                               // written to in an ongoing transaction and any partition which
635                               // does not yet have markers written when in the `PendingAbort` or
636                               // `PendingCommit` states. If the topic partition that we found is
637                               // among these, then we can still expect the coordinator to write
638                               // the marker. Otherwise, it is a hanging transaction.
639                               if (!description.topicPartitions().contains(openTransaction.topicPartition)) {
640                                   hangingTransactions.add(openTransaction);
641                               }
642                           }
643                       }
644                   }
645               });
646   
647               return hangingTransactions;
648           }

814               Admin admin,
815               Optional<Integer> brokerId,
816               long maxTransactionTimeoutMs,
817               List<TopicPartition> topicPartitions,
818               List<OpenTransaction> candidateTransactions
819           ) throws Exception {

834                                   candidateTransactions.add(new OpenTransaction(
835                                       topicPartition,
836                                       activeProducer
837                                   ));
838                               }
839                           }
840                       });
841                   });
842               } catch (ExecutionException e) {
843                   printErrorAndExit("Failed to describe producers for " + topicPartitions.size() +

889               ThrowableConsumer<List<T>> consumer
890           ) throws Exception {

896                       limitIndex,
897                       batchStartIndex + batchSize
898                   );
899   
900                   consumer.accept(list.subList(batchStartIndex, batchEndIndex));
901                   batchStartIndex = batchEndIndex;
902               }
903           }
