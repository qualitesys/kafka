36   /**
37    * Primarily intended for use with system testing, this appender produces message
38    * to Kafka on each "append" request. For example, this helps with end-to-end tests
39    * of KafkaLog4jAppender.
40    *
41    * When used as a command-line tool, it appends increasing integers. It will produce a
42    * fixed number of messages unless the default max-messages -1 is used, in which case
43    * it appends indefinitely.
44    */
45   
46   public class VerifiableLog4jAppender {
47       Logger logger = Logger.getLogger(VerifiableLog4jAppender.class);
48   
49       // If maxMessages < 0, log until the process is killed externally
50       private long maxMessages = -1;
51   
52       // Hook to trigger logging thread to stop logging messages
53       private volatile boolean stopLogging = false;

155       /**
156        * Read a properties file from the given path
157        * @param filename The path of the file to read
158        *
159        * Note: this duplication of org.apache.kafka.common.utils.Utils.loadProps is unfortunate
160        * but *intentional*. In order to use VerifiableProducer in compatibility and upgrade tests,
161        * we use VerifiableProducer from the development tools package, and run it against 0.8.X.X kafka jars.
162        * Since this method is not in Utils in the 0.8.X.X jars, we have to cheat a bit and duplicate.
163        */
164       public static Properties loadProps(String filename) throws IOException {
165           Properties props = new Properties();
166           try (InputStream propStream = Files.newInputStream(Paths.get(filename))) {
167               props.load(propStream);
168           }
169           return props;
170       }

172       /** Construct a VerifiableLog4jAppender object from command-line arguments. */
173       public static VerifiableLog4jAppender createFromArgs(String[] args) {
174           ArgumentParser parser = argParser();
175           VerifiableLog4jAppender producer = null;
176   
177           try {
178               Namespace res = parser.parseArgs(args);
179   
180               int maxMessages = res.getInt("maxMessages");
181               String topic = res.getString("topic");
182               String configFile = res.getString("appender.config");
183   
184               Properties props = new Properties();
185               props.setProperty("log4j.rootLogger", "INFO, KAFKA");
186               props.setProperty("log4j.appender.KAFKA", "org.apache.kafka.log4jappender.KafkaLog4jAppender");
187               props.setProperty("log4j.appender.KAFKA.layout", "org.apache.log4j.PatternLayout");
188               props.setProperty("log4j.appender.KAFKA.layout.ConversionPattern", "%-5p: %c - %m%n");
189               props.setProperty("log4j.appender.KAFKA.BrokerList", res.getString("brokerList"));
190               props.setProperty("log4j.appender.KAFKA.Topic", topic);
191               props.setProperty("log4j.appender.KAFKA.RequiredNumAcks", res.getString("acks"));
192               props.setProperty("log4j.appender.KAFKA.SyncSend", "true");
193               final String securityProtocol = res.getString("securityProtocol");
194               if (securityProtocol != null && !securityProtocol.equals(SecurityProtocol.PLAINTEXT.toString())) {
195                   props.setProperty("log4j.appender.KAFKA.SecurityProtocol", securityProtocol);
196               }
197               if (securityProtocol != null && securityProtocol.contains("SSL")) {
198                   props.setProperty("log4j.appender.KAFKA.SslTruststoreLocation", res.getString("sslTruststoreLocation"));
199                   props.setProperty("log4j.appender.KAFKA.SslTruststorePassword", res.getString("sslTruststorePassword"));
200               }
201               if (securityProtocol != null && securityProtocol.contains("SASL")) {
202                   props.setProperty("log4j.appender.KAFKA.SaslKerberosServiceName", res.getString("saslKerberosServiceName"));
203                   props.setProperty("log4j.appender.KAFKA.clientJaasConfPath", res.getString("clientJaasConfPath"));
204                   props.setProperty("log4j.appender.KAFKA.kerb5ConfPath", res.getString("kerb5ConfPath"));
205               }
206               props.setProperty("log4j.logger.kafka.log4j", "INFO, KAFKA");
207               // Changing log level from INFO to WARN as a temporary workaround for KAFKA-6415. This is to
208               // avoid deadlock in system tests when producer network thread appends to log while updating metadata.
209               props.setProperty("log4j.logger.org.apache.kafka.clients.Metadata", "WARN, KAFKA");
210   
211               if (configFile != null) {
212                   try {
213                       props.putAll(loadProps(configFile));
214                   } catch (IOException e) {
215                       throw new ArgumentParserException(e.getMessage(), parser);
216                   }
217               }
218   
219               producer = new VerifiableLog4jAppender(props, maxMessages);
220           } catch (ArgumentParserException e) {
221               if (args.length == 0) {
222                   parser.printHelp();
223                   Exit.exit(0);
224               } else {
225                   parser.handleError(e);
226                   Exit.exit(1);
227               }
228           }
229   
230           return producer;
231       }

234       public VerifiableLog4jAppender(Properties props, int maxMessages) {
235           this.maxMessages = maxMessages;
236           PropertyConfigurator.configure(props);
237       }
238   
239       public static void main(String[] args) {
240   
241           final VerifiableLog4jAppender appender = createFromArgs(args);
242           boolean infinite = appender.maxMessages < 0;
243   
244           // Trigger main thread to stop producing messages when shutting down
245           Exit.addShutdownHook("verifiable-log4j-appender-shutdown-hook", () -> appender.stopLogging = true);
246   
247           long maxMessages = infinite ? Long.MAX_VALUE : appender.maxMessages;
248           for (long i = 0; i < maxMessages; i++) {
249               if (appender.stopLogging) {
250                   break;
251               }
252               appender.append(String.format("%d", i));
253           }
254       }
255   
256       private void append(String msg) {
257           logger.info(msg);
258       }
259   }
