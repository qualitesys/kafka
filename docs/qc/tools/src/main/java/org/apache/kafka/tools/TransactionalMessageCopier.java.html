62    * This class is primarily meant for use with system tests. It copies messages from an input partition to an output
63    * topic transactionally, committing the offsets and messages together.

179       private static KafkaProducer<String, String> createProducer(Namespace parsedArgs) {
180           Properties props = new Properties();
181           props.put(ProducerConfig.BOOTSTRAP_SERVERS_CONFIG, parsedArgs.getString("brokerList"));
182           props.put(ProducerConfig.TRANSACTIONAL_ID_CONFIG, parsedArgs.getString("transactionalId"));
183           props.put(ProducerConfig.KEY_SERIALIZER_CLASS_CONFIG,
184                   "org.apache.kafka.common.serialization.StringSerializer");
185           props.put(ProducerConfig.VALUE_SERIALIZER_CLASS_CONFIG,
186                   "org.apache.kafka.common.serialization.StringSerializer");
187           // We set a small batch size to ensure that we have multiple inflight requests per transaction.
188           // If it is left at the default, each transaction will have only one batch per partition, hence not testing
189           // the case with multiple inflights.
190           props.put(ProducerConfig.BATCH_SIZE_CONFIG, "512");
191           props.put(ProducerConfig.MAX_IN_FLIGHT_REQUESTS_PER_CONNECTION, "5");
192           props.put(ProducerConfig.TRANSACTION_TIMEOUT_CONFIG, parsedArgs.getInt("transactionTimeout"));
193   
194           return new KafkaProducer<>(props);
195       }

225       private static Map<TopicPartition, OffsetAndMetadata> consumerPositions(KafkaConsumer<String, String> consumer) {
226           Map<TopicPartition, OffsetAndMetadata> positions = new HashMap<>();
227           for (TopicPartition topicPartition : consumer.assignment()) {
228               positions.put(topicPartition, new OffsetAndMetadata(consumer.position(topicPartition), null));
229           }
230           return positions;
231       }

233       private static void resetToLastCommittedPositions(KafkaConsumer<String, String> consumer) {
234           final Map<TopicPartition, OffsetAndMetadata> committed = consumer.committed(consumer.assignment());
235           consumer.assignment().forEach(tp -> {
236               OffsetAndMetadata offsetAndMetadata = committed.get(tp);
237               if (offsetAndMetadata != null)
238                   consumer.seek(tp, offsetAndMetadata.offset());
239               else
240                   consumer.seekToBeginning(singleton(tp));
241           });
242       }

244       private static long messagesRemaining(KafkaConsumer<String, String> consumer, TopicPartition partition) {
245           long currentPosition = consumer.position(partition);
246           Map<TopicPartition, Long> endOffsets = consumer.endOffsets(singleton(partition));
247           if (endOffsets.containsKey(partition)) {
248               return endOffsets.get(partition) - currentPosition;
249           }
250           return 0;
251       }

253       private static String toJsonString(Map<String, Object> data) {
254           String json;
255           try {
256               ObjectMapper mapper = new ObjectMapper();
257               json = mapper.writeValueAsString(data);
258           } catch (JsonProcessingException e) {
259               json = "Bad data can't be written as json: " + e.getMessage();
260           }
261           return json;
262       }

264       private static synchronized String statusAsJson(
265           String stage,
266           long totalProcessed,
267           long consumedSinceLastRebalanced,
268           long remaining,
269           String transactionalId
270       ) {
271           Map<String, Object> statusData = new LinkedHashMap<>();
272           statusData.put("transactionalId", transactionalId);
273           statusData.put("stage", stage);
274           statusData.put("time", FORMAT.format(new Date()));
275           statusData.put("totalProcessed", totalProcessed);

289       public static void main(String[] args) {
290           Namespace parsedArgs = argParser().parseArgsOrFail(args);
291           try {
292               runEventLoop(parsedArgs);
293               Exit.exit(0);
294           } catch (Exception e) {
295               log.error("Shutting down after unexpected error in event loop", e);
296               System.err.println("Shutting down after unexpected error " + e.getClass().getSimpleName()
297                   + ": " + e.getMessage() + " (see the log for additional detail)");
298               Exit.exit(1);
299           }

302       public static void runEventLoop(Namespace parsedArgs) {
303           final String transactionalId = parsedArgs.getString("transactionalId");
304           final String outputTopic = parsedArgs.getString("outputTopic");
305   
306           String consumerGroup = parsedArgs.getString("consumerGroup");
307   
308           final KafkaProducer<String, String> producer = createProducer(parsedArgs);
309           final KafkaConsumer<String, String> consumer = createConsumer(parsedArgs);
310   
311           final AtomicLong remainingMessages = new AtomicLong(
312               parsedArgs.getInt("maxMessages") == -1 ? Long.MAX_VALUE : parsedArgs.getInt("maxMessages"));
313   
314           boolean groupMode = parsedArgs.getBoolean("groupMode");
315           String topicName = parsedArgs.getString("inputTopic");
316           final AtomicLong numMessagesProcessedSinceLastRebalance = new AtomicLong(0);
317           final AtomicLong totalMessageProcessed = new AtomicLong(0);
318           if (groupMode) {
319               consumer.subscribe(Collections.singleton(topicName), new ConsumerRebalanceListener() {
320                   @Override
321                   public void onPartitionsRevoked(Collection<TopicPartition> partitions) {
322                   }
323   
324                   @Override
325                   public void onPartitionsAssigned(Collection<TopicPartition> partitions) {
326                       remainingMessages.set(partitions.stream()
327                           .mapToLong(partition -> messagesRemaining(consumer, partition)).sum());
328                       numMessagesProcessedSinceLastRebalance.set(0);
329                       // We use message cap for remaining here as the remainingMessages are not set yet.
330                       System.out.println(statusAsJson(
331                           "RebalanceComplete",
332                           totalMessageProcessed.get(),
333                           numMessagesProcessedSinceLastRebalance.get(),
334                           remainingMessages.get(),
335                           transactionalId
336                       ));
337                   }
338               });
339           } else {
340               TopicPartition inputPartition = new TopicPartition(topicName, parsedArgs.getInt("inputPartition"));
341               consumer.assign(singleton(inputPartition));
342               remainingMessages.set(Math.min(messagesRemaining(consumer, inputPartition), remainingMessages.get()));
343           }
344   
345           final boolean enableRandomAborts = parsedArgs.getBoolean("enableRandomAborts");
346   
347           producer.initTransactions();
348   
349           final AtomicBoolean isShuttingDown = new AtomicBoolean(false);
350   
351           Exit.addShutdownHook("transactional-message-copier-shutdown-hook", () -> {
352               isShuttingDown.set(true);
353               consumer.wakeup();
354               System.out.println(statusAsJson(
355                   "ShutdownComplete",
356                   totalMessageProcessed.get(),
357                   numMessagesProcessedSinceLastRebalance.get(),
358                   remainingMessages.get(),
359                   transactionalId
360               ));
361           });
362   
363           final boolean useGroupMetadata = parsedArgs.getBoolean("useGroupMetadata");
364           try {
365               Random random = new Random();
366               while (!isShuttingDown.get() && remainingMessages.get() > 0) {
367                   System.out.println(statusAsJson(
368                       "ProcessLoop",
369                       totalMessageProcessed.get(),
370                       numMessagesProcessedSinceLastRebalance.get(),
371                       remainingMessages.get(),
372                       transactionalId
373                   ));
374   
375                   ConsumerRecords<String, String> records = consumer.poll(Duration.ofMillis(200));
376                   if (records.count() > 0) {
377                       try {
378                           producer.beginTransaction();
379   
380                           for (ConsumerRecord<String, String> record : records) {
381                               producer.send(producerRecordFromConsumerRecord(outputTopic, record));
382                           }
383   
384                           long messagesSentWithinCurrentTxn = records.count();
385   
386                           ConsumerGroupMetadata groupMetadata = useGroupMetadata ? consumer.groupMetadata() : new ConsumerGroupMetadata(consumerGroup);
387                           producer.sendOffsetsToTransaction(consumerPositions(consumer), groupMetadata);
388   
389                           if (enableRandomAborts && random.nextInt() % 3 == 0) {
390                               abortTransactionAndResetPosition(producer, consumer);
391                           } else {
392                               producer.commitTransaction();
393                               remainingMessages.getAndAdd(-messagesSentWithinCurrentTxn);
394                               numMessagesProcessedSinceLastRebalance.getAndAdd(messagesSentWithinCurrentTxn);
395                               totalMessageProcessed.getAndAdd(messagesSentWithinCurrentTxn);
396                           }
397                       } catch (ProducerFencedException e) {
398                           throw new KafkaException(String.format("The transactional.id %s has been claimed by another process", transactionalId), e);
399                       } catch (KafkaException e) {
400                           log.debug("Aborting transaction after catching exception", e);
401                           abortTransactionAndResetPosition(producer, consumer);
402                       }
403                   }
404               }
405           } catch (WakeupException e) {
406               if (!isShuttingDown.get()) {
407                   // Let the exception propagate if the exception was not raised
408                   // as part of shutdown.
409                   throw e;
410               }
411           } finally {
412               Utils.closeQuietly(producer, "producer");
413               Utils.closeQuietly(consumer, "consumer");
414           }
415       }
416   }
