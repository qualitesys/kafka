118   
119       public static void main(String[] args) {
120           ArgsHandler argsHandler = new ArgsHandler();
121           Namespace namespace;
122   
123           try {
124               namespace = argsHandler.parseArgs(args);
125           } catch (ArgumentParserException e) {
126               Exit.exit(1);
127               return;
128           }
129   
130           ConfigHandler configHandler = new ConfigHandler(namespace);
131   
132           Map<String, ?> configs = configHandler.getConfigs();
133           Map<String, Object> jaasConfigs = configHandler.getJaasOptions();
134   
135           try {
136               String accessToken;
137   
138               {
139                   // Client side...
140                   try (AccessTokenRetriever atr = AccessTokenRetrieverFactory.create(configs, jaasConfigs)) {
141                       atr.init();
142                       AccessTokenValidator atv = AccessTokenValidatorFactory.create(configs);
143                       System.out.println("PASSED 1/5: client configuration");
144   
145                       accessToken = atr.retrieve();
146                       System.out.println("PASSED 2/5: client JWT retrieval");
147   
148                       atv.validate(accessToken);
149                       System.out.println("PASSED 3/5: client JWT validation");
150                   }
151               }
152   
153               {
154                   // Broker side...
155                   try (CloseableVerificationKeyResolver vkr = VerificationKeyResolverFactory.create(configs, jaasConfigs)) {
156                       vkr.init();
157                       AccessTokenValidator atv = AccessTokenValidatorFactory.create(configs, vkr);
158                       System.out.println("PASSED 4/5: broker configuration");
159   
160                       atv.validate(accessToken);
161                       System.out.println("PASSED 5/5: broker JWT validation");
162                   }
163               }
164   
165               System.out.println("SUCCESS");
166               Exit.exit(0);
167           } catch (Throwable t) {
168               System.out.println("FAILED:");
169               t.printStackTrace();
170   
171               if (t instanceof ConfigException) {
172                   System.out.printf("%n");
173                   argsHandler.parser.printHelp();
174               }
175   
176               Exit.exit(1);
177           }
178       }

256           private Argument addArgument(String option, String help, Class<?> clazz) {
257               // Change foo.bar into --foo.bar.

262                   .metavar(option)
263                   .dest(option)
264                   .help(help);
