28   /**
29    * {@code TestInputTopic} is used to pipe records to topic in {@link TopologyTestDriver}.
30    * To use {@code TestInputTopic} create a new instance via
31    * {@link TopologyTestDriver#createInputTopic(String, Serializer, Serializer)}.
32    * In actual test code, you can pipe new record values, keys and values or list of {@link KeyValue} pairs.
33    * If you have multiple source topics, you need to create a {@code TestInputTopic} for each.
34    *
35    * <h2>Processing messages</h2>
36    * <pre>{@code
37    *     private TestInputTopic<Long, String> inputTopic;
38    *     ...
39    *     inputTopic = testDriver.createInputTopic(INPUT_TOPIC, longSerializer, stringSerializer);
40    *     ...
41    *     inputTopic.pipeInput("Hello");
42    * }</pre>
43    *
44    * @param <K> the type of the record key
45    * @param <V> the type of the record value
46    * @see TopologyTestDriver
47    */
48   
49   public class TestInputTopic<K, V> {
50       private final TopologyTestDriver driver;
51       private final String topic;
52       private final Serializer<K> keySerializer;
53       private final Serializer<V> valueSerializer;
54   
55       //Timing
56       private Instant currentTime;
57       private final Duration advanceDuration;
58   
59       TestInputTopic(final TopologyTestDriver driver,
60                      final String topicName,
61                      final Serializer<K> keySerializer,
62                      final Serializer<V> valueSerializer,
63                      final Instant startTimestamp,
64                      final Duration autoAdvance) {
65           Objects.requireNonNull(driver, "TopologyTestDriver cannot be null");
66           Objects.requireNonNull(topicName, "topicName cannot be null");
67           Objects.requireNonNull(keySerializer, "keySerializer cannot be null");
68           Objects.requireNonNull(valueSerializer, "valueSerializer cannot be null");
69           Objects.requireNonNull(startTimestamp, "startTimestamp cannot be null");
70           Objects.requireNonNull(autoAdvance, "autoAdvance cannot be null");
71           this.driver = driver;
72           this.topic = topicName;
73           this.keySerializer = keySerializer;
74           this.valueSerializer = valueSerializer;
75           this.currentTime = startTimestamp;
76           if (autoAdvance.isNegative()) {
77               throw new IllegalArgumentException("autoAdvance must be positive");
78           }
79           this.advanceDuration = autoAdvance;
80       }
81   
82       /**
83        * Advances the internally tracked event time of this input topic.
84        * Each time a record without explicitly defined timestamp is piped,
85        * the current topic event time is used as record timestamp.
86        * <p>
87        * Note: advancing the event time on the input topic, does not advance the tracked stream time in
88        * {@link TopologyTestDriver} as long as no new input records are piped.
89        * Furthermore, it does not advance the wall-clock time of {@link TopologyTestDriver}.
90        *
91        * @param advance the duration of time to advance
92        */
93       public void advanceTime(final Duration advance) {
94           if (advance.isNegative()) {
95               throw new IllegalArgumentException("advance must be positive");
96           }
97           currentTime = currentTime.plus(advance);
98       }
99   
100       private Instant getTimestampAndAdvance() {
101           final Instant timestamp = currentTime;
102           currentTime = currentTime.plus(advanceDuration);
103           return timestamp;
104       }
105   
106       /**
107        * Send an input record with the given record on the topic and then commit the records.
108        * May auto advance topic time.
109        *
110        * @param record the record to sent
111        */
112       public void pipeInput(final TestRecord<K, V> record) {
113           //if record timestamp not set get timestamp and advance
114           final Instant timestamp = (record.getRecordTime() == null) ? getTimestampAndAdvance() : record.getRecordTime();
115           driver.pipeRecord(topic, record, keySerializer, valueSerializer, timestamp);
116       }

118       /**
119        * Send an input record with the given value on the topic and then commit the records.
120        * May auto advance topic time.
121        *
122        * @param value the record value
123        */
124       public void pipeInput(final V value) {
125           pipeInput(new TestRecord<>(value));
126       }

129        * Send an input record with the given key and value on the topic and then commit the records.

131        *
132        * @param key   the record key
133        * @param value the record value
134        */
135       public void pipeInput(final K key,
136                             final V value) {
137           pipeInput(new TestRecord<>(key, value));
138       }

140       /**
141        * Send an input record with the given value and timestamp on the topic and then commit the records.
142        * Does not auto advance internally tracked time.
143        *
144        * @param value       the record value
145        * @param timestamp the record timestamp
146        */
147       public void pipeInput(final V value,

149           pipeInput(new TestRecord<>(null, value, timestamp));
150       }

152       /**
153        * Send an input record with the given key, value and timestamp on the topic and then commit the records.
154        * Does not auto advance internally tracked time.
155        *
156        * @param key         the record key
157        * @param value       the record value
158        * @param timestampMs the record timestamp
159        */
160       public void pipeInput(final K key,
161                             final V value,
162                             final long timestampMs) {
163           pipeInput(new TestRecord<>(key, value, null, timestampMs));
164       }

168        * Does not auto advance internally tracked time.
169        *
170        * @param key         the record key
171        * @param value       the record value

174       public void pipeInput(final K key,
175                             final V value,
176                             final Instant timestamp) {

178       }

180       /**
181        * Send input records with the given KeyValue  list on the topic  then commit each record individually.
182        * The timestamp will be generated based on the constructor provided start time and time will auto advance.
183        *
184        * @param records the list of TestRecord records
185        */
186       public void pipeRecordList(final List<? extends TestRecord<K, V>> records) {
187           for (final TestRecord<K, V> record : records) {
188               pipeInput(record);
189           }
190       }

192       /**
193        * Send input records with the given KeyValue list on the topic then commit each record individually.
194        * The timestamp will be generated based on the constructor provided start time and time will auto advance based on
195        * {@link #TestInputTopic(TopologyTestDriver, String, Serializer, Serializer, Instant, Duration) autoAdvance} setting.
196        *
197        * @param keyValues the {@link List} of {@link KeyValue} records
198        */
199       public void pipeKeyValueList(final List<KeyValue<K, V>> keyValues) {
200           for (final KeyValue<K, V> keyValue : keyValues) {
201               pipeInput(keyValue.key, keyValue.value);
202           }
203       }

205       /**
206        * Send input records with the given value list on the topic then commit each record individually.
207        * The timestamp will be generated based on the constructor provided start time and time will auto advance based on
208        * {@link #TestInputTopic(TopologyTestDriver, String, Serializer, Serializer, Instant, Duration) autoAdvance} setting.
209        *
210        * @param values the {@link List} of {@link KeyValue} records
211        */
212       public void pipeValueList(final List<V> values) {
213           for (final V value : values) {
214               pipeInput(value);
215           }
216       }
217   
218       /**
219        * Send input records with the given {@link KeyValue} list on the topic then commit each record individually.
220        * Does not auto advance internally tracked time.
221        *
222        * @param keyValues      the {@link List} of {@link KeyValue} records
223        * @param startTimestamp the timestamp for the first generated record
224        * @param advance        the time difference between two consecutive generated records
225        */
226       public void pipeKeyValueList(final List<KeyValue<K, V>> keyValues,
227                                    final Instant startTimestamp,
228                                    final Duration advance) {
229           Instant recordTime = startTimestamp;
230           for (final KeyValue<K, V> keyValue : keyValues) {
231               pipeInput(keyValue.key, keyValue.value, recordTime);
232               recordTime = recordTime.plus(advance);
233           }
234       }
235   
236       /**
237        * Send input records with the given value list on the topic then commit each record individually.
238        * The timestamp will be generated based on the constructor provided start time and time will auto advance based on
239        * {@link #TestInputTopic(TopologyTestDriver, String, Serializer, Serializer, Instant, Duration) autoAdvance} setting.
240        *
241        * @param values         the {@link List} of values
242        * @param startTimestamp the timestamp for the first generated record
243        * @param advance        the time difference between two consecutive generated records
244        */
245       public void pipeValueList(final List<V> values,
246                                 final Instant startTimestamp,
247                                 final Duration advance) {
248           Instant recordTime = startTimestamp;
249           for (final V value : values) {
250               pipeInput(value, recordTime);
251               recordTime = recordTime.plus(advance);
252           }
253       }

259                   .add("keySerializer=" + keySerializer.getClass().getSimpleName())

262       }
