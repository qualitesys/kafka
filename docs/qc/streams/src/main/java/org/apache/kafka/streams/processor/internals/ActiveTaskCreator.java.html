71       // Tasks may have been assigned for a NamedTopology that is not yet known by this host. When that occurs we stash
72       // these unknown tasks until either the corresponding NamedTopology is added and we can create them at last, or
73       // we receive a new assignment and they are revoked from the thread.
74       private final Map<TaskId, Set<TopicPartition>> unknownTasksToBeCreated = new HashMap<>();

76       ActiveTaskCreator(final TopologyMetadata topologyMetadata,
77                         final StreamsConfig applicationConfig,
78                         final StreamsMetricsImpl streamsMetrics,
79                         final StateDirectory stateDirectory,
80                         final ChangelogReader storeChangelogReader,
81                         final ThreadCache cache,
82                         final Time time,
83                         final KafkaClientSupplier clientSupplier,
84                         final String threadId,
85                         final UUID processId,
86                         final Logger log) {
87           this.topologyMetadata = topologyMetadata;
88           this.applicationConfig = applicationConfig;
89           this.streamsMetrics = streamsMetrics;
90           this.stateDirectory = stateDirectory;
91           this.storeChangelogReader = storeChangelogReader;
92           this.cache = cache;
93           this.time = time;
94           this.clientSupplier = clientSupplier;
95           this.threadId = threadId;
96           this.log = log;
97   
98           createTaskSensor = ThreadMetrics.createTaskSensor(threadId, streamsMetrics);
99           processingMode = processingMode(applicationConfig);
100   
101           if (processingMode == EXACTLY_ONCE_ALPHA) {
102               threadProducer = null;
103               taskProducers = new HashMap<>();
104           } else { // non-eos and eos-v2
105               log.info("Creating thread producer client");
106   
107               final String threadIdPrefix = String.format("stream-thread [%s] ", Thread.currentThread().getName());
108               final LogContext logContext = new LogContext(threadIdPrefix);
109   
110               threadProducer = new StreamsProducer(
111                   applicationConfig,
112                   threadId,
113                   clientSupplier,
114                   null,
115                   processId,
116                   logContext,
117                   time);
118               taskProducers = Collections.emptyMap();
119           }
120       }

126       StreamsProducer streamsProducerForTask(final TaskId taskId) {
127           if (processingMode != EXACTLY_ONCE_ALPHA) {
128               throw new IllegalStateException("Expected EXACTLY_ONCE to be enabled, but the processing mode was " + processingMode);
129           }
130   
131           final StreamsProducer taskProducer = taskProducers.get(taskId);
132           if (taskProducer == null) {
133               throw new IllegalStateException("Unknown TaskId: " + taskId);
134           }
135           return taskProducer;
136       }

138       StreamsProducer threadProducer() {
139           if (processingMode != EXACTLY_ONCE_V2) {
140               throw new IllegalStateException("Expected EXACTLY_ONCE_V2 to be enabled, but the processing mode was " + processingMode);
141           }
142           return threadProducer;
143       }

153       // TODO: change return type to `StreamTask`
154       public Collection<Task> createTasks(final Consumer<byte[], byte[]> consumer,
155                                    final Map<TaskId, Set<TopicPartition>> tasksToBeCreated) {
156           // TODO: change type to `StreamTask`
157           final List<Task> createdTasks = new ArrayList<>();
158           final Map<TaskId, Set<TopicPartition>> newUnknownTasks = new HashMap<>();
159   
160           for (final Map.Entry<TaskId, Set<TopicPartition>> newTaskAndPartitions : tasksToBeCreated.entrySet()) {
161               final TaskId taskId = newTaskAndPartitions.getKey();
162               final Set<TopicPartition> partitions = newTaskAndPartitions.getValue();
163   
164               final LogContext logContext = getLogContext(taskId);
165   
166               // task belongs to a named topology that hasn't been added yet, wait until it has to create this
167               if (taskId.topologyName() != null && !topologyMetadata.namedTopologiesView().contains(taskId.topologyName())) {
168                   newUnknownTasks.put(taskId, partitions);
169                   continue;
170               }
171   
172               final ProcessorTopology topology = topologyMetadata.buildSubtopology(taskId);
173   
174               final ProcessorStateManager stateManager = new ProcessorStateManager(
175                   taskId,
176                   Task.TaskType.ACTIVE,
177                   eosEnabled(applicationConfig),
178                   logContext,
179                   stateDirectory,
180                   storeChangelogReader,
181                   topology.storeToChangelogTopic(),
182                   partitions
183               );
184   
185               final InternalProcessorContext context = new ProcessorContextImpl(
186                   taskId,
187                   applicationConfig,
188                   stateManager,
189                   streamsMetrics,
190                   cache
191               );
192   
193               createdTasks.add(
194                   createActiveTask(
195                       taskId,
196                       partitions,
197                       consumer,
198                       logContext,
199                       topology,
200                       stateManager,
201                       context
202                   )
203               );
204               unknownTasksToBeCreated.remove(taskId);
205           }
206           if (!newUnknownTasks.isEmpty()) {
207               log.info("Delaying creation of tasks not yet known by this instance: {}", newUnknownTasks.keySet());
208               unknownTasksToBeCreated.putAll(newUnknownTasks);
209           }
210           return createdTasks;
211       }

214       StreamTask createActiveTaskFromStandby(final StandbyTask standbyTask,
215                                              final Set<TopicPartition> inputPartitions,
216                                              final Consumer<byte[], byte[]> consumer) {
217           final InternalProcessorContext context = standbyTask.processorContext();
218           final ProcessorStateManager stateManager = standbyTask.stateMgr;
219           final LogContext logContext = getLogContext(standbyTask.id);
220   
221           standbyTask.closeCleanAndRecycleState();
222           stateManager.transitionTaskType(TaskType.ACTIVE, logContext);
223   
224           return createActiveTask(
225               standbyTask.id,
226               inputPartitions,
227               consumer,
228               logContext,
229               topologyMetadata.buildSubtopology(standbyTask.id),
230               stateManager,
231               context
232           );
233       }

235       private StreamTask createActiveTask(final TaskId taskId,
236                                           final Set<TopicPartition> inputPartitions,
237                                           final Consumer<byte[], byte[]> consumer,
238                                           final LogContext logContext,
239                                           final ProcessorTopology topology,
240                                           final ProcessorStateManager stateManager,
241                                           final InternalProcessorContext context) {
242           final StreamsProducer streamsProducer;
243           if (processingMode == ProcessingMode.EXACTLY_ONCE_ALPHA) {
244               log.info("Creating producer client for task {}", taskId);
245               streamsProducer = new StreamsProducer(
246                   applicationConfig,
247                   threadId,
248                   clientSupplier,
249                   taskId,
250                   null,
251                   logContext,
252                   time);
253               taskProducers.put(taskId, streamsProducer);
254           } else {
255               streamsProducer = threadProducer;
256           }
257   
258           final RecordCollector recordCollector = new RecordCollectorImpl(
259               logContext,
260               taskId,
261               streamsProducer,
262               applicationConfig.defaultProductionExceptionHandler(),
263               streamsMetrics,
264               topology
265           );
266   
267           final StreamTask task = new StreamTask(
268               taskId,
269               inputPartitions,
270               topology,
271               consumer,
272               topologyMetadata.getTaskConfigFor(taskId),
273               streamsMetrics,
274               stateDirectory,
275               cache,
276               time,
277               stateManager,
278               recordCollector,
279               context,
280               logContext
281           );
282   
283           log.trace("Created task {} with assigned partitions {}", taskId, inputPartitions);
284           createTaskSensor.record();
285           return task;
286       }

298       void closeAndRemoveTaskProducerIfNeeded(final TaskId id) {
299           final StreamsProducer taskProducer = taskProducers.remove(id);
300           if (taskProducer != null) {
301               try {
302                   taskProducer.close();
303               } catch (final RuntimeException e) {

305               }
306           }

309       Map<MetricName, Metric> producerMetrics() {
310           // When EOS is turned on, each task will have its own producer client
311           // and the producer object passed in here will be null. We would then iterate through
312           // all the active tasks and add their metrics to the output metrics map.
313           final Collection<StreamsProducer> producers = threadProducer != null ?
314               Collections.singleton(threadProducer) :
315               taskProducers.values();
316           return ClientUtils.producerMetrics(producers);
317       }

319       Set<String> producerClientIds() {
320           if (threadProducer != null) {
321               return Collections.singleton(getThreadProducerClientId(threadId));
322           } else {
323               return taskProducers.keySet()
324                                   .stream()
325                                   .map(taskId -> getTaskProducerClientId(threadId, taskId))
326                                   .collect(Collectors.toSet());
327           }
328       }

330       private LogContext getLogContext(final TaskId taskId) {
331           final String threadIdPrefix = String.format("stream-thread [%s] ", Thread.currentThread().getName());
332           final String logPrefix = threadIdPrefix + String.format("%s [%s] ", "task", taskId);
333           return new LogContext(logPrefix);
334       }
