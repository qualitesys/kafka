18   
19   import org.apache.kafka.common.serialization.Serializer;
20   import org.apache.kafka.streams.errors.InvalidStateStoreException;
21   
22   /**
23    * A key-value store that only supports read operations.
24    * Implementations should be thread-safe as concurrent reads and writes are expected.
25    * <p>
26    * Please note that this contract defines the thread-safe read functionality only; it does not
27    * guarantee anything about whether the actual instance is writable by another thread, or
28    * whether it uses some locking mechanism under the hood. For this reason, making dependencies
29    * between the read and write operations on different StateStore instances can cause concurrency
30    * problems like deadlock.
31    *
32    * @param <K> the key type
33    * @param <V> the value type
34    */
35   public interface ReadOnlyKeyValueStore<K, V> {
36   
37       /**
38        * Get the value corresponding to this key.
39        *
40        * @param key The key to fetch
41        * @return The value or null if no value is found.
42        * @throws NullPointerException       If null is used for key.
43        * @throws InvalidStateStoreException if the store is not initialized
44        */
45       V get(K key);
46   
47       /**
48        * Get an iterator over a given range of keys. This iterator must be closed after use.
49        * The returned iterator must be safe from {@link java.util.ConcurrentModificationException}s
50        * and must not return null values.
51        * Order is not guaranteed as bytes lexicographical ordering might not represent key order.
52        *
53        * @param from The first key that could be in the range, where iteration starts from.
54        *             A null value indicates that the range starts with the first element in the store.
55        * @param to   The last key that could be in the range, where iteration ends.
56        *             A null value indicates that the range ends with the last element in the store.
57        * @return The iterator for this range, from smallest to largest bytes.
58        * @throws InvalidStateStoreException if the store is not initialized
59        */
60       KeyValueIterator<K, V> range(K from, K to);
61   
62       /**
63        * Get a reverse iterator over a given range of keys. This iterator must be closed after use.
64        * The returned iterator must be safe from {@link java.util.ConcurrentModificationException}s
65        * and must not return null values.
66        * Order is not guaranteed as bytes lexicographical ordering might not represent key order.
67        *
68        * @param from The first key that could be in the range, where iteration ends.
69        *             A null value indicates that the range starts with the first element in the store.
70        * @param to   The last key that could be in the range, where iteration starts from.
71        *             A null value indicates that the range ends with the last element in the store.
72        * @return The reverse iterator for this range, from largest to smallest key bytes.
73        * @throws InvalidStateStoreException if the store is not initialized
74        */
75       default KeyValueIterator<K, V> reverseRange(K from, K to) {
76           throw new UnsupportedOperationException();
77       }
78   
79       /**
80        * Return an iterator over all keys in this store. This iterator must be closed after use.
81        * The returned iterator must be safe from {@link java.util.ConcurrentModificationException}s
82        * and must not return null values.
83        * Order is not guaranteed as bytes lexicographical ordering might not represent key order.
84        *
85        * @return An iterator of all key/value pairs in the store, from smallest to largest bytes.
86        * @throws InvalidStateStoreException if the store is not initialized
87        */
88       KeyValueIterator<K, V> all();
89   
90       /**
91        * Return a reverse iterator over all keys in this store. This iterator must be closed after use.
92        * The returned iterator must be safe from {@link java.util.ConcurrentModificationException}s
93        * and must not return null values.
94        * Order is not guaranteed as bytes lexicographical ordering might not represent key order.
95        *
96        * @return An reverse iterator of all key/value pairs in the store, from largest to smallest key bytes.
97        * @throws InvalidStateStoreException if the store is not initialized
98        */
99       default KeyValueIterator<K, V> reverseAll() {
100           throw new UnsupportedOperationException();
101       }
102   
103       /**
104        * Return an iterator over all keys with the specified prefix.
105        * Since the type of the prefix can be different from that of the key, a serializer to convert the
106        * prefix into the format in which the keys are stored in the stores needs to be passed to this method.
107        * The returned iterator must be safe from {@link java.util.ConcurrentModificationException}s
108        * and must not return null values.
109        * Since {@code prefixScan()} relies on byte lexicographical ordering and not on the ordering of the key type, results for some types might be unexpected.
110        * For example, if the key type is {@code Integer}, and the store contains keys [1, 2, 11, 13],
111        * then running {@code store.prefixScan(1, new IntegerSerializer())} will return [1] and not [1,11,13]. 
112        * In contrast, if the key type is {@code String} the keys will be sorted [1, 11, 13, 2] in the store and {@code store.prefixScan(1, new StringSerializer())} will return [1,11,13]. 
113        * In both cases {@code prefixScan()} starts the scan at 1 and stops at 2.
114        *
115        * @param prefix The prefix.
116        * @param prefixKeySerializer Serializer for the Prefix key type
117        * @param <PS> Prefix Serializer type
118        * @param <P> Prefix Type.
119        * @return The iterator for keys having the specified prefix.
120        */
121       default <PS extends Serializer<P>, P> KeyValueIterator<K, V> prefixScan(P prefix, PS prefixKeySerializer) {
122           throw new UnsupportedOperationException();
123       }
124   
125       /**
126        * Return an approximate count of key-value mappings in this store.
127        * <p>
128        * The count is not guaranteed to be exact in order to accommodate stores
129        * where an exact count is expensive to calculate.
130        *
131        * @return an approximate count of key-value mappings in the store.
132        * @throws InvalidStateStoreException if the store is not initialized
133        */
134       long approximateNumEntries();
135   }
