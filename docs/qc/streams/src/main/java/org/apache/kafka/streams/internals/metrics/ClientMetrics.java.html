17   package org.apache.kafka.streams.internals.metrics;
18   
19   import org.apache.kafka.common.metrics.Gauge;
20   import org.apache.kafka.common.metrics.Sensor;

23   import org.apache.kafka.streams.processor.internals.metrics.StreamsMetricsImpl;
24   import org.slf4j.Logger;
25   import org.slf4j.LoggerFactory;
26   
27   import java.io.InputStream;
28   import java.util.Properties;
29   
30   import static org.apache.kafka.streams.processor.internals.metrics.StreamsMetricsImpl.CLIENT_LEVEL_GROUP;
31   import static org.apache.kafka.streams.processor.internals.metrics.StreamsMetricsImpl.addSumMetricToSensor;
32   
33   public class ClientMetrics {
34       private ClientMetrics() {}
35   
36       private static final Logger log = LoggerFactory.getLogger(ClientMetrics.class);
37       private static final String VERSION = "version";
38       private static final String COMMIT_ID = "commit-id";
39       private static final String APPLICATION_ID = "application-id";
40       private static final String TOPOLOGY_DESCRIPTION = "topology-description";
41       private static final String STATE = "state";
42       private static final String ALIVE_STREAM_THREADS = "alive-stream-threads";
43       private static final String VERSION_FROM_FILE;
44       private static final String COMMIT_ID_FROM_FILE;
45       private static final String DEFAULT_VALUE = "unknown";
46       private static final String FAILED_STREAM_THREADS = "failed-stream-threads";
47   
48       static {
49           final Properties props = new Properties();
50           try (InputStream resourceStream = ClientMetrics.class.getResourceAsStream(
51               "/kafka/kafka-streams-version.properties")) {
52   
53               props.load(resourceStream);
54           } catch (final Exception exception) {
55               log.warn("Error while loading kafka-streams-version.properties", exception);
56           }
57           VERSION_FROM_FILE = props.getProperty("version", DEFAULT_VALUE).trim();
58           COMMIT_ID_FROM_FILE = props.getProperty("commitId", DEFAULT_VALUE).trim();
59       }
60   
61       private static final String VERSION_DESCRIPTION = "The version of the Kafka Streams client";
62       private static final String COMMIT_ID_DESCRIPTION = "The version control commit ID of the Kafka Streams client";
63       private static final String APPLICATION_ID_DESCRIPTION = "The application ID of the Kafka Streams client";
64       private static final String TOPOLOGY_DESCRIPTION_DESCRIPTION =
65           "The description of the topology executed in the Kafka Streams client";
66       private static final String STATE_DESCRIPTION = "The state of the Kafka Streams client";
67       private static final String ALIVE_STREAM_THREADS_DESCRIPTION = "The current number of alive stream threads that are running or participating in rebalance";
68       private static final String FAILED_STREAM_THREADS_DESCRIPTION = "The number of failed stream threads since the start of the Kafka Streams client";
69   
70       public static String version() {
71           return VERSION_FROM_FILE;
72       }
73   
74       public static String commitId() {
75           return COMMIT_ID_FROM_FILE;
76       }
77   
78       public static void addVersionMetric(final StreamsMetricsImpl streamsMetrics) {
79           streamsMetrics.addClientLevelImmutableMetric(
80               VERSION,
81               VERSION_DESCRIPTION,
82               RecordingLevel.INFO,
83               VERSION_FROM_FILE
84           );
85       }
86   
87       public static void addCommitIdMetric(final StreamsMetricsImpl streamsMetrics) {
88           streamsMetrics.addClientLevelImmutableMetric(
89               COMMIT_ID,
90               COMMIT_ID_DESCRIPTION,
91               RecordingLevel.INFO,
92               COMMIT_ID_FROM_FILE
93           );
94       }
95   
96       public static void addApplicationIdMetric(final StreamsMetricsImpl streamsMetrics, final String applicationId) {
97           streamsMetrics.addClientLevelImmutableMetric(
98               APPLICATION_ID,
99               APPLICATION_ID_DESCRIPTION,
100               RecordingLevel.INFO,
101               applicationId
102           );
103       }
104   
105       public static void addTopologyDescriptionMetric(final StreamsMetricsImpl streamsMetrics,
106                                                       final Gauge<String> topologyDescription) {
107           streamsMetrics.addClientLevelMutableMetric(
108               TOPOLOGY_DESCRIPTION,
109               TOPOLOGY_DESCRIPTION_DESCRIPTION,
110               RecordingLevel.INFO,
111               topologyDescription
112           );
113       }
114   
115       public static void addStateMetric(final StreamsMetricsImpl streamsMetrics,
116                                         final Gauge<State> stateProvider) {
117           streamsMetrics.addClientLevelMutableMetric(
118               STATE,
119               STATE_DESCRIPTION,
120               RecordingLevel.INFO,
121               stateProvider
122           );
123       }
124   
125       public static void addNumAliveStreamThreadMetric(final StreamsMetricsImpl streamsMetrics,
126                                                        final Gauge<Integer> stateProvider) {
127           streamsMetrics.addClientLevelMutableMetric(
128               ALIVE_STREAM_THREADS,
129               ALIVE_STREAM_THREADS_DESCRIPTION,
130               RecordingLevel.INFO,
131               stateProvider
132           );
133       }
134   
135       public static Sensor failedStreamThreadSensor(final StreamsMetricsImpl streamsMetrics) {
136           final Sensor sensor = streamsMetrics.clientLevelSensor(FAILED_STREAM_THREADS, RecordingLevel.INFO);
137           addSumMetricToSensor(
138               sensor,
139               CLIENT_LEVEL_GROUP,
140               streamsMetrics.clientLevelTagMap(),
141               FAILED_STREAM_THREADS,
142               false,
143               FAILED_STREAM_THREADS_DESCRIPTION
144           );
145           return sensor;
146       }
147   }
