20   import org.apache.kafka.common.serialization.Serializer;
21   import org.apache.kafka.common.utils.Bytes;
22   import org.apache.kafka.streams.kstream.Window;
23   import org.apache.kafka.streams.kstream.Windowed;
24   import org.apache.kafka.streams.kstream.internals.SessionWindow;

26   import static org.apache.kafka.streams.state.StateSerdes.TIMESTAMP_SIZE;
27   
28   import java.nio.ByteBuffer;
29   import java.util.List;

32   public class SessionKeySchema implements SegmentedBytesStore.KeySchema {
33   
34       private static final int SUFFIX_SIZE = 2 * TIMESTAMP_SIZE;
35       private static final byte[] MIN_SUFFIX = new byte[SUFFIX_SIZE];
36   
37       public static int keyByteLength(final Bytes key) {
38           return (key == null ? 0 : key.get().length) + 2 * TIMESTAMP_SIZE;
39       }
40   
41       @Override
42       public Bytes upperRangeFixedSize(final Bytes key, final long to) {
43           final Windowed<Bytes> sessionKey = upperRangeFixedWindow(key, to);
44           return SessionKeySchema.toBinary(sessionKey);
45       }
46   
47       public static <K> Windowed<K> upperRangeFixedWindow(final K key, final long to) {
48           return new Windowed<K>(key, new SessionWindow(to, Long.MAX_VALUE));
49       }
50   
51       @Override
52       public Bytes lowerRangeFixedSize(final Bytes key, final long from) {
53           final Windowed<Bytes> sessionKey = lowerRangeFixedWindow(key, from);
54           return SessionKeySchema.toBinary(sessionKey);
55       }
56   
57       public static <K> Windowed<K> lowerRangeFixedWindow(final K key, final long from) {
58           return new Windowed<K>(key, new SessionWindow(0, Math.max(0, from)));
59       }
60   
61       @Override
62       public Bytes upperRange(final Bytes key, final long to) {
63           if (key == null) {
64               return null;
65           }
66           final byte[] maxSuffix = ByteBuffer.allocate(SUFFIX_SIZE)
67               // the end timestamp can be as large as possible as long as it's larger than start time
68               .putLong(Long.MAX_VALUE)
69               // this is the start timestamp
70               .putLong(to)
71               .array();
72           return OrderedBytes.upperRange(key, maxSuffix);
73       }
74   
75       @Override
76       public Bytes lowerRange(final Bytes key, final long from) {
77           if (key == null) {
78               return null;
79           }
80           return OrderedBytes.lowerRange(key, MIN_SUFFIX);
81       }
82   
83       @Override
84       public long segmentTimestamp(final Bytes key) {
85           return SessionKeySchema.extractEndTimestamp(key.get());
86       }
87   
88       @Override
89       public HasNextCondition hasNextCondition(final Bytes binaryKeyFrom, final Bytes binaryKeyTo, final long from, final long to, final boolean forward) {
90           return iterator -> {
91               while (iterator.hasNext()) {
92                   final Bytes bytes = iterator.peekNextKey();
93                   final Windowed<Bytes> windowedKey = SessionKeySchema.from(bytes);
94                   if ((binaryKeyFrom == null || windowedKey.key().compareTo(binaryKeyFrom) >= 0)
95                       && (binaryKeyTo == null || windowedKey.key().compareTo(binaryKeyTo) <= 0)
96                       && windowedKey.window().end() >= from
97                       && windowedKey.window().start() <= to) {
98                       return true;
99                   }
100                   iterator.next();
101               }
102               return false;
103           };
104       }
105   
106       @Override
107       public <S extends Segment> List<S> segmentsToSearch(final Segments<S> segments,
108                                                           final long from,
109                                                           final long to,
110                                                           final boolean forward) {
111           return segments.segments(from, Long.MAX_VALUE, forward);
112       }
113   
114       private static <K> K extractKey(final byte[] binaryKey,
115                                       final Deserializer<K> deserializer,
116                                       final String topic) {
117           return deserializer.deserialize(topic, extractKeyBytes(binaryKey));
118       }
119   
120       static byte[] extractKeyBytes(final byte[] binaryKey) {
121           final byte[] bytes = new byte[binaryKey.length - 2 * TIMESTAMP_SIZE];
122           System.arraycopy(binaryKey, 0, bytes, 0, bytes.length);
123           return bytes;
124       }
125   
126       static long extractEndTimestamp(final byte[] binaryKey) {
127           return ByteBuffer.wrap(binaryKey).getLong(binaryKey.length - 2 * TIMESTAMP_SIZE);
128       }
129   
130       static long extractStartTimestamp(final byte[] binaryKey) {
131           return ByteBuffer.wrap(binaryKey).getLong(binaryKey.length - TIMESTAMP_SIZE);
132       }
133   
134       static Window extractWindow(final byte[] binaryKey) {
135           final ByteBuffer buffer = ByteBuffer.wrap(binaryKey);
136           final long start = buffer.getLong(binaryKey.length - TIMESTAMP_SIZE);
137           final long end = buffer.getLong(binaryKey.length - 2 * TIMESTAMP_SIZE);
138           return new SessionWindow(start, end);
139       }
140   
141       public static <K> Windowed<K> from(final byte[] binaryKey,
142                                          final Deserializer<K> keyDeserializer,
143                                          final String topic) {
144           final K key = extractKey(binaryKey, keyDeserializer, topic);
145           final Window window = extractWindow(binaryKey);
146           return new Windowed<>(key, window);
147       }
148   
149       public static Windowed<Bytes> from(final Bytes bytesKey) {
150           final byte[] binaryKey = bytesKey.get();
151           final Window window = extractWindow(binaryKey);
152           return new Windowed<>(Bytes.wrap(extractKeyBytes(binaryKey)), window);
153       }
154   
155       public static <K> Windowed<K> from(final Windowed<Bytes> keyBytes,
156                                          final Deserializer<K> keyDeserializer,
157                                          final String topic) {
158           final K key = keyDeserializer.deserialize(topic, keyBytes.key().get());
159           return new Windowed<>(key, keyBytes.window());
160       }
161   
162       public static <K> byte[] toBinary(final Windowed<K> sessionKey,
163                                         final Serializer<K> serializer,
164                                         final String topic) {
165           final byte[] bytes = serializer.serialize(topic, sessionKey.key());
166           return toBinary(Bytes.wrap(bytes), sessionKey.window().start(), sessionKey.window().end()).get();
167       }
168   
169       public static Bytes toBinary(final Windowed<Bytes> sessionKey) {
170           return toBinary(sessionKey.key(), sessionKey.window().start(), sessionKey.window().end());
171       }
172   
173       public static Bytes toBinary(final Bytes key,
174                                    final long startTime,
175                                    final long endTime) {
176           final ByteBuffer buf = ByteBuffer.allocate(keyByteLength(key));
177           writeBinary(buf, key, startTime, endTime);
178           return Bytes.wrap(buf.array());
179       }
180   
181       public static void writeBinary(final ByteBuffer buf, final Windowed<Bytes> sessionKey) {
182           writeBinary(buf, sessionKey.key(), sessionKey.window().start(), sessionKey.window().end());
183       }
184   
185       public static void writeBinary(final ByteBuffer buf,
186                                      final Bytes key,
187                                      final long startTime,
188                                      final long endTime) {
189           // we search for the session window that can overlap with the [ESET, LSST] range
190           // since the session window length can vary, we define the search boundary as:
191           // lower: [0, ESET]
192           // upper: [LSST, INF]
193           // and by puting the end time first and then the start time, the serialized search boundary
194           // is: [(ESET-0), (INF-LSST)]
195           buf.put(key.get());
196           buf.putLong(endTime);
197           buf.putLong(startTime);
198       }
