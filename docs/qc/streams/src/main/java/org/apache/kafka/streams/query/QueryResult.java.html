17   package org.apache.kafka.streams.query;
18   
19   
20   import org.apache.kafka.streams.processor.StateStore;
21   import org.apache.kafka.streams.query.internals.FailedQueryResult;
22   import org.apache.kafka.streams.query.internals.SucceededQueryResult;
23   
24   import java.util.List;
25   
26   /**
27    * Container for a single partition's result when executing a {@link StateQueryRequest}.
28    *
29    * @param <R> The result type of the query.
30    */
31   public interface QueryResult<R> {
32       /**
33        * Static factory method to create a result object for a successful query. Used by StateStores
34        * to respond to a {@link StateStore#query(Query, PositionBound, boolean)}.
35        */
36       static <R> QueryResult<R> forResult(final R result) {
37           return new SucceededQueryResult<>(result);
38       }
39   
40       /**
41        * Static factory method to create a result object for a failed query. Used by StateStores to
42        * respond to a {@link StateStore#query(Query, PositionBound, boolean)}.
43        */
44       static <R> QueryResult<R> forFailure(
45           final FailureReason failureReason,
46           final String failureMessage) {
47   
48           return new FailedQueryResult<>(failureReason, failureMessage);
49       }
50   
51       /**
52        * Static factory method to create a failed query result object to indicate that the store does
53        * not know how to handle the query.
54        * <p>
55        * Used by StateStores to respond to a {@link StateStore#query(Query, PositionBound, boolean)}.
56        */
57       static <R> QueryResult<R> forUnknownQueryType(
58           final Query<R> query,
59           final StateStore store) {
60   
61           return forFailure(
62               FailureReason.UNKNOWN_QUERY_TYPE,
63               "This store (" + store.getClass() + ") doesn't know how to execute "
64                   + "the given query (" + query + ")." +
65                   " Contact the store maintainer if you need support for a new query type.");
66       }
67   
68       /**
69        * Static factory method to create a failed query result object to indicate that the store has
70        * not yet caught up to the requested position bound.
71        * <p>
72        * Used by StateStores to respond to a {@link StateStore#query(Query, PositionBound, boolean)}.
73        */
74       static <R> QueryResult<R> notUpToBound(
75           final Position currentPosition,
76           final PositionBound positionBound,
77           final Integer partition) {
78   
79           if (partition == null) {
80               return new FailedQueryResult<>(
81                   FailureReason.NOT_UP_TO_BOUND,
82                   "The store is not initialized yet, so it is not yet up to the bound "
83                       + positionBound
84               );
85           } else {
86               return new FailedQueryResult<>(
87                   FailureReason.NOT_UP_TO_BOUND,
88                   "For store partition " + partition + ", the current position "
89                       + currentPosition + " is not yet up to the bound "
90                       + positionBound
91               );
92           }
93       }
94   
95       /**
96        * Used by stores to add detailed execution information (if requested) during query execution.
97        */
98       void addExecutionInfo(final String message);
99   
100       /**
101        * Used by stores to report what exact position in the store's history it was at when it
102        * executed the query.
103        */
104       void setPosition(final Position position);
105   
106       /**
107        * True iff the query was successfully executed. The response is available in {@link
108        * this#getResult()}.
109        */
110       boolean isSuccess();
111   
112   
113       /**
114        * True iff the query execution failed. More information about the failure is available in
115        * {@link this#getFailureReason()} and {@link this#getFailureMessage()}.
116        */
117       boolean isFailure();
118   
119       /**
120        * If detailed execution information was requested in {@link StateQueryRequest#enableExecutionInfo()},
121        * this method returned the execution details for this partition's result.
122        */
123       List<String> getExecutionInfo();
124   
125       /**
126        * This state partition's exact position in its history when this query was executed. Can be
127        * used in conjunction with subsequent queries via {@link StateQueryRequest#withPositionBound(PositionBound)}.
128        * <p>
129        * Note: stores are encouraged, but not required to set this property.
130        */
131       Position getPosition();
132   
133       /**
134        * If this partition failed to execute the query, returns the reason.
135        *
136        * @throws IllegalArgumentException if this is not a failed result.
137        */
138       FailureReason getFailureReason();
139   
140       /**
141        * If this partition failed to execute the query, returns the failure message.
142        *
143        * @throws IllegalArgumentException if this is not a failed result.
144        */
145       String getFailureMessage();
146   
147       /**
148        * Returns the result of executing the query on one partition. The result type is determined by
149        * the query. Note: queries may choose to return {@code null} for a successful query, so {@link
150        * this#isSuccess()} and {@link this#isFailure()} must be used to determine whether the query
151        * was successful of failed on this partition.
152        *
153        * @throws IllegalArgumentException if this is not a successful query.
154        */
155       R getResult();
156   }
