17   package org.apache.kafka.streams.processor.internals;
18   
19   import org.apache.kafka.common.header.Headers;
20   import org.apache.kafka.common.metrics.Sensor;
21   import org.apache.kafka.common.serialization.Deserializer;
22   import org.apache.kafka.streams.processor.TimestampExtractor;
23   import org.apache.kafka.streams.processor.api.Record;
24   import org.apache.kafka.streams.processor.internals.metrics.ProcessorNodeMetrics;
25   
26   import static org.apache.kafka.streams.kstream.internals.WrappingNullableUtils.prepareKeyDeserializer;
27   import static org.apache.kafka.streams.kstream.internals.WrappingNullableUtils.prepareValueDeserializer;
28   
29   public class SourceNode<KIn, VIn> extends ProcessorNode<KIn, VIn, KIn, VIn> {
30   
31       private InternalProcessorContext<KIn, VIn> context;
32       private Deserializer<KIn> keyDeserializer;
33       private Deserializer<VIn> valDeserializer;
34       private final TimestampExtractor timestampExtractor;
35       private Sensor processAtSourceSensor;
36   
37       public SourceNode(final String name,
38                         final TimestampExtractor timestampExtractor,
39                         final Deserializer<KIn> keyDeserializer,
40                         final Deserializer<VIn> valDeserializer) {
41           super(name);
42           this.timestampExtractor = timestampExtractor;
43           this.keyDeserializer = keyDeserializer;
44           this.valDeserializer = valDeserializer;
45       }
46   
47       public SourceNode(final String name,
48                         final Deserializer<KIn> keyDeserializer,
49                         final Deserializer<VIn> valDeserializer) {
50           this(name, null, keyDeserializer, valDeserializer);
51       }
52   
53       KIn deserializeKey(final String topic, final Headers headers, final byte[] data) {
54           return keyDeserializer.deserialize(topic, headers, data);
55       }
56   
57       VIn deserializeValue(final String topic, final Headers headers, final byte[] data) {
58           return valDeserializer.deserialize(topic, headers, data);
59       }
60   
61       @Override
62       public void init(final InternalProcessorContext<KIn, VIn> context) {
63           // It is important to first create the sensor before calling init on the
64           // parent object. Otherwise due to backwards compatibility an empty sensor
65           // without parent is created with the same name.
66           // Once the backwards compatibility is not needed anymore it might be possible to
67           // change this.
68           processAtSourceSensor = ProcessorNodeMetrics.processAtSourceSensor(
69               Thread.currentThread().getName(),
70               context.taskId().toString(),
71               context.currentNode().name(),
72               context.metrics()
73           );
74           super.init(context);
75           this.context = context;
76   
77           keyDeserializer = prepareKeyDeserializer(keyDeserializer, context, name());
78           valDeserializer = prepareValueDeserializer(valDeserializer, context, name());
79       }
80   
81   
82       @Override
83       public void process(final Record<KIn, VIn> record) {
84           context.forward(record);
85           processAtSourceSensor.record(1.0d, context.currentSystemTimeMs());
86       }
87   
88       /**
89        * @return a string representation of this node, useful for debugging.
90        */
91       @Override
92       public String toString() {
93           return toString("");
94       }
95   
96       public TimestampExtractor getTimestampExtractor() {
97           return timestampExtractor;
98       }
99   }
