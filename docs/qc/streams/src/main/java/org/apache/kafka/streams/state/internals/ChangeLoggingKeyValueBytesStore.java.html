17   package org.apache.kafka.streams.state.internals;
18   
19   import org.apache.kafka.common.serialization.Serializer;
20   import org.apache.kafka.common.utils.Bytes;
21   import org.apache.kafka.streams.KeyValue;
22   import org.apache.kafka.streams.processor.ProcessorContext;
23   import org.apache.kafka.streams.processor.StateStore;
24   import org.apache.kafka.streams.processor.StateStoreContext;
25   import org.apache.kafka.streams.processor.internals.InternalProcessorContext;
26   import org.apache.kafka.streams.state.KeyValueIterator;
27   import org.apache.kafka.streams.state.KeyValueStore;
28   
29   import java.util.List;
30   
31   import static org.apache.kafka.streams.processor.internals.ProcessorContextUtils.asInternalProcessorContext;
32   
33   public class ChangeLoggingKeyValueBytesStore
34           extends WrappedStateStore<KeyValueStore<Bytes, byte[]>, byte[], byte[]>
35           implements KeyValueStore<Bytes, byte[]> {
36   
37       InternalProcessorContext context;
38   
39       ChangeLoggingKeyValueBytesStore(final KeyValueStore<Bytes, byte[]> inner) {
40           super(inner);
41       }
42   
43       @Deprecated
44       @Override
45       public void init(final ProcessorContext context,
46                        final StateStore root) {
47           this.context = asInternalProcessorContext(context);
48           super.init(context, root);
49           maybeSetEvictionListener();
50       }
51   
52       @Override
53       public void init(final StateStoreContext context,
54                        final StateStore root) {
55           this.context = asInternalProcessorContext(context);
56           super.init(context, root);
57           maybeSetEvictionListener();
58       }
59   
60       private void maybeSetEvictionListener() {
61           // if the inner store is an LRU cache, add the eviction listener to log removed record
62           if (wrapped() instanceof MemoryLRUCache) {
63               ((MemoryLRUCache) wrapped()).setWhenEldestRemoved((key, value) -> {
64                   // pass null to indicate removal
65                   log(key, null);
66               });
67           }
68       }
69   
70       @Override
71       public long approximateNumEntries() {
72           return wrapped().approximateNumEntries();
73       }
74   
75       @Override
76       public void put(final Bytes key,
77                       final byte[] value) {
78           wrapped().put(key, value);
79           log(key, value);
80       }
81   
82       @Override
83       public byte[] putIfAbsent(final Bytes key,
84                                 final byte[] value) {
85           final byte[] previous = wrapped().putIfAbsent(key, value);
86           if (previous == null) {
87               // then it was absent
88               log(key, value);
89           }
90           return previous;
91       }
92   
93       @Override
94       public void putAll(final List<KeyValue<Bytes, byte[]>> entries) {
95           wrapped().putAll(entries);
96           for (final KeyValue<Bytes, byte[]> entry : entries) {
97               log(entry.key, entry.value);
98           }
99       }
100   
101       @Override
102       public <PS extends Serializer<P>, P> KeyValueIterator<Bytes, byte[]> prefixScan(final P prefix,
103                                                                                       final PS prefixKeySerializer) {
104           return wrapped().prefixScan(prefix, prefixKeySerializer);
105       }
106   
107       @Override
108       public byte[] delete(final Bytes key) {
109           final byte[] oldValue = wrapped().delete(key);
110           log(key, null);
111           return oldValue;
112       }
113   
114       @Override
115       public byte[] get(final Bytes key) {
116           return wrapped().get(key);
117       }
118   
119       @Override
120       public KeyValueIterator<Bytes, byte[]> range(final Bytes from,
121                                                    final Bytes to) {
122           return wrapped().range(from, to);
123       }
124   
125       @Override
126       public KeyValueIterator<Bytes, byte[]> reverseRange(final Bytes from,
127                                                           final Bytes to) {
128           return wrapped().reverseRange(from, to);
129       }
130   
131       @Override
132       public KeyValueIterator<Bytes, byte[]> all() {
133           return wrapped().all();
134       }
135   
136       @Override
137       public KeyValueIterator<Bytes, byte[]> reverseAll() {
138           return wrapped().reverseAll();
139       }
140   
141       void log(final Bytes key, final byte[] value) {
142           context.logChange(name(), key, value, context.timestamp(), wrapped().getPosition());
143       }
144   }
