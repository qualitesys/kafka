19   import org.apache.kafka.common.utils.Bytes;

26   /**
27    * The interface representing a StateStore that has 1 or more segments that are based
28    * on time.
29    * @see RocksDBSegmentedBytesStore
30    */
31   public interface SegmentedBytesStore extends StateStore {
32   
33       /**
34        * Fetch all records from the segmented store with the provided key and time range
35        * from all existing segments
36        * @param key       the key to match
37        * @param from      earliest time to match
38        * @param to        latest time to match
39        * @return  an iterator over key-value pairs
40        */
41       KeyValueIterator<Bytes, byte[]> fetch(final Bytes key, final long from, final long to);
42   
43       /**
44        * Fetch all records from the segmented store with the provided key and time range
45        * from all existing segments in backward order (from latest to earliest)
46        * @param key       the key to match
47        * @param from      earliest time to match
48        * @param to        latest time to match
49        * @return  an iterator over key-value pairs
50        */
51       KeyValueIterator<Bytes, byte[]> backwardFetch(final Bytes key, final long from, final long to);
52   
53       /**
54        * Fetch all records from the segmented store in the provided key range and time range
55        * from all existing segments
56        * @param keyFrom   The first key that could be in the range
57        * @param keyTo     The last key that could be in the range
58        * @param from      earliest time to match
59        * @param to        latest time to match
60        * @return  an iterator over key-value pairs
61        */
62       KeyValueIterator<Bytes, byte[]> fetch(final Bytes keyFrom, final Bytes keyTo, final long from, final long to);
63   
64       /**
65        * Fetch all records from the segmented store in the provided key range and time range
66        * from all existing segments in backward order (from latest to earliest)
67        * @param keyFrom   The first key that could be in the range
68        * @param keyTo     The last key that could be in the range
69        * @param from      earliest time to match
70        * @param to        latest time to match
71        * @return  an iterator over key-value pairs
72        */
73       KeyValueIterator<Bytes, byte[]> backwardFetch(final Bytes keyFrom, final Bytes keyTo, final long from, final long to);
74   
75       /**
76        * Gets all the key-value pairs in the existing windows.
77        *
78        * @return an iterator over windowed key-value pairs {@code <Windowed<K>, value>}
79        * @throws InvalidStateStoreException if the store is not initialized
80        */
81       KeyValueIterator<Bytes, byte[]> all();
82   
83       /**
84        * Gets all the key-value pairs in the existing windows in backward order (from latest to earliest).
85        *
86        * @return an iterator over windowed key-value pairs {@code <Windowed<K>, value>}
87        * @throws InvalidStateStoreException if the store is not initialized
88        */
89       KeyValueIterator<Bytes, byte[]> backwardAll();
90   
91       /**
92        * Gets all the key-value pairs that belong to the windows within in the given time range.
93        *
94        * @param from the beginning of the time slot from which to search (inclusive)
95        * @param to   the end of the time slot from which to search (inclusive)
96        * @return an iterator over windowed key-value pairs {@code <Windowed<K>, value>}
97        * @throws InvalidStateStoreException if the store is not initialized
98        * @throws NullPointerException if null is used for any key
99        */
100       KeyValueIterator<Bytes, byte[]> fetchAll(final long from, final long to);
101   
102       KeyValueIterator<Bytes, byte[]> backwardFetchAll(final long from, final long to);
103   
104       /**
105        * Remove the record with the provided key. The key
106        * should be a composite of the record key, and the timestamp information etc
107        * as described by the {@link KeySchema}
108        * @param key   the segmented key to remove
109        */
110       void remove(Bytes key);
111   
112       /**
113        * Remove all duplicated records with the provided key in the specified timestamp.
114        *
115        * @param key   the segmented key to remove
116        * @param timestamp  the timestamp to match
117        */
118       void remove(Bytes key, long timestamp);
119   
120       /**
121        * Write a new value to the store with the provided key. The key
122        * should be a composite of the record key, and the timestamp information etc
123        * as described by the {@link KeySchema}
124        * @param key
125        * @param value
126        */
127       void put(Bytes key, byte[] value);
128   
129       /**
130        * Get the record from the store with the given key. The key
131        * should be a composite of the record key, and the timestamp information etc
132        * as described by the {@link KeySchema}
133        * @param key
134        * @return
135        */
136       byte[] get(Bytes key);
137   
138       interface KeySchema {
139   
140           /**
141            * Given a range of record keys and a time, construct a Segmented key that represents
142            * the upper range of keys to search when performing range queries.
143            * @see SessionKeySchema#upperRange
144            * @see WindowKeySchema#upperRange
145            * @param key
146            * @param to
147            * @return      The key that represents the upper range to search for in the store
148            */
149           Bytes upperRange(final Bytes key, final long to);
150   
151           /**
152            * Given a range of record keys and a time, construct a Segmented key that represents
153            * the lower range of keys to search when performing range queries.
154            * @see SessionKeySchema#lowerRange
155            * @see WindowKeySchema#lowerRange
156            * @param key
157            * @param from
158            * @return      The key that represents the lower range to search for in the store
159            */
160           Bytes lowerRange(final Bytes key, final long from);
161   
162           /**
163            * Given a record key and a time, construct a Segmented key to search when performing
164            * prefixed queries.
165            *
166            * @param key
167            * @param timestamp
168            * @return  The key that represents the prefixed Segmented key in bytes.
169            */
170           default Bytes toStoreBinaryKeyPrefix(final Bytes key, long timestamp) {
171               throw new UnsupportedOperationException();
172           }
173   
174           /**
175            * Given a range of fixed size record keys and a time, construct a Segmented key that represents
176            * the upper range of keys to search when performing range queries.
177            * @see SessionKeySchema#upperRange
178            * @see WindowKeySchema#upperRange
179            * @param key the last key in the range
180            * @param to the last timestamp in the range
181            * @return The key that represents the upper range to search for in the store
182            */
183           Bytes upperRangeFixedSize(final Bytes key, final long to);
184   
185           /**
186            * Given a range of fixed size record keys and a time, construct a Segmented key that represents
187            * the lower range of keys to search when performing range queries.
188            * @see SessionKeySchema#lowerRange
189            * @see WindowKeySchema#lowerRange
190            * @param key the first key in the range
191            * @param from the first timestamp in the range
192            * @return      The key that represents the lower range to search for in the store
193            */
194           Bytes lowerRangeFixedSize(final Bytes key, final long from);
195   
196           /**
197            * Extract the timestamp of the segment from the key. The key is a composite of
198            * the record-key, any timestamps, plus any additional information.
199            * @see SessionKeySchema#lowerRange
200            * @see WindowKeySchema#lowerRange
201            * @param key
202            * @return
203            */
204           long segmentTimestamp(final Bytes key);
205   
206           /**
207            * Create an implementation of {@link HasNextCondition} that knows when
208            * to stop iterating over the KeyValueSegments. Used during {@link SegmentedBytesStore#fetch(Bytes, Bytes, long, long)} operations
209            * @param binaryKeyFrom the first key in the range
210            * @param binaryKeyTo   the last key in the range
211            * @param from          starting time range
212            * @param to            ending time range
213            * @param forward       forward or backward
214            * @return
215            */
216           HasNextCondition hasNextCondition(final Bytes binaryKeyFrom, final Bytes binaryKeyTo, final long from, final long to, final boolean forward);
217   
218           /**
219            * Used during {@link SegmentedBytesStore#fetch(Bytes, long, long)} operations to determine
220            * which segments should be scanned.
221            * @param segments
222            * @param from
223            * @param to
224            * @return  List of segments to search
225            */
226           <S extends Segment> List<S> segmentsToSearch(Segments<S> segments, long from, long to, boolean forward);
227       }
228   }
