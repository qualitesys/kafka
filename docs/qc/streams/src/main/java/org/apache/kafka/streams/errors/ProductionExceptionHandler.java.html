17   package org.apache.kafka.streams.errors;
18   
19   import org.apache.kafka.clients.producer.ProducerRecord;
20   import org.apache.kafka.common.Configurable;
21   
22   /**
23    * Interface that specifies how an exception when attempting to produce a result to
24    * Kafka should be handled.
25    */
26   public interface ProductionExceptionHandler extends Configurable {
27       /**
28        * Inspect a record that we attempted to produce, and the exception that resulted
29        * from attempting to produce it and determine whether or not to continue processing.
30        *
31        * @param record The record that failed to produce
32        * @param exception The exception that occurred during production
33        */
34       ProductionExceptionHandlerResponse handle(final ProducerRecord<byte[], byte[]> record,
35                                                 final Exception exception);
36   
37       enum ProductionExceptionHandlerResponse {
38           /* continue processing */
39           CONTINUE(0, "CONTINUE"),
40           /* fail processing */
41           FAIL(1, "FAIL");
42   
43           /**
44            * an english description of the api--this is for debugging and can change
45            */
46           public final String name;
47   
48           /**
49            * the permanent and immutable id of an API--this can't change ever
50            */
51           public final int id;
52   
53           ProductionExceptionHandlerResponse(final int id,
54                                              final String name) {
55               this.id = id;
56               this.name = name;
57           }
58       }
59   }
