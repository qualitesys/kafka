19   import org.apache.kafka.streams.kstream.Named;
20   
21   public class NamedInternal extends Named {
22   
23       public static NamedInternal empty() {
24           return new NamedInternal((String) null);
25       }
26   
27       public static NamedInternal with(final String name) {
28           return new NamedInternal(name);
29       }
30   
31       /**
32        * Creates a new {@link NamedInternal} instance.
33        *
34        * @param internal  the internal name.
35        */
36       NamedInternal(final Named internal) {
37           super(internal);
38       }
39   
40       /**
41        * Creates a new {@link NamedInternal} instance.
42        *
43        * @param internal the internal name.
44        */
45       NamedInternal(final String internal) {
46           super(internal);
47       }
48   
49       /**
50        * @return a string name.
51        */
52       public String name() {
53           return name;
54       }
55   
56       @Override
57       public NamedInternal withName(final String name) {
58           return new NamedInternal(name);
59       }
60       
61       String suffixWithOrElseGet(final String suffix, final String other) {
62           if (name != null) {
63               return name + suffix;
64           } else {
65               return other;
66           }
67       }
68   
69       String suffixWithOrElseGet(final String suffix, final InternalNameProvider provider, final String prefix) {
70           // We actually do not need to generate processor names for operation if a name is specified.
71           // But before returning, we still need to burn index for the operation to keep topology backward compatibility.
72           if (name != null) {
73               provider.newProcessorName(prefix);
74   
75               final String suffixed = name + suffix;
76               // Re-validate generated name as suffixed string could be too large.
77               Named.validate(suffixed);
78   
79               return suffixed;
80           } else {
81               return provider.newProcessorName(prefix);
82           }
83       }
84   
85       String orElseGenerateWithPrefix(final InternalNameProvider provider, final String prefix) {
86           // We actually do not need to generate processor names for operation if a name is specified.
87           // But before returning, we still need to burn index for the operation to keep topology backward compatibility.
88           if (name != null) {
89               provider.newProcessorName(prefix);
90               return name;
91           }  else {
92               return provider.newProcessorName(prefix);
93           }
94       }
95   
96   }
