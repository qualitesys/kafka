19   /**
20    * Processor context interface for {@link Record}.
21    *
22    * @param <KForward> a bound on the types of keys that may be forwarded
23    * @param <VForward> a bound on the types of values that may be forwarded
24    */
25   public interface ProcessorContext<KForward, VForward> extends ProcessingContext {
26       /**
27        * Forward a record to all child processors.
28        * <p>
29        * Note that the forwarded {@link Record} is shared between the parent and child
30        * processors. And of course, the parent may forward the same object to multiple children,
31        * and the child may forward it to grandchildren, etc. Therefore, you should be mindful
32        * of mutability.
33        * <p>
34        * The {@link Record} class itself is immutable (all the setter-style methods return an
35        * independent copy of the instance). However, the key, value, and headers referenced by
36        * the Record may themselves be mutable.
37        * <p>
38        * Some programs may opt to make use of this mutability for high performance, in which case
39        * the input record may be mutated and then forwarded by each {@link Processor}. However,
40        * most applications should instead favor safety.
41        * <p>
42        * Forwarding records safely simply means to make a copy of the record before you mutate it.
43        * This is trivial when using the {@link Record#withKey(Object)}, {@link Record#withValue(Object)},
44        * and {@link Record#withTimestamp(long)} methods, as each of these methods make a copy of the
45        * record as a matter of course. But a little extra care must be taken with headers, since
46        * the {@link org.apache.kafka.common.header.Header} class is mutable. The easiest way to
47        * safely handle headers is to use the {@link Record} constructors to make a copy before
48        * modifying headers.
49        * <p>
50        * In other words, this would be considered unsafe:
51        * <code>
52        *     process(Record inputRecord) {
53        *         inputRecord.headers().add(...);
54        *         context.forward(inputRecord);
55        *     }
56        * </code>
57        * This is unsafe because the parent, and potentially siblings, grandparents, etc.,
58        * all will see this modification to their shared Headers reference. This is a violation
59        * of causality and could lead to undefined behavior.
60        * <p>
61        * A safe usage would look like this:
62        * <code>
63        *     process(Record inputRecord) {
64        *         // makes a copy of the headers
65        *         Record toForward = inputRecord.withHeaders(inputRecord.headers());
66        *         // Other options to create a safe copy are:
67        *         // * use any copy-on-write method, which makes a copy of all fields:
68        *         //   toForward = inputRecord.withValue();
69        *         // * explicitly copy all fields:
70        *         //   toForward = new Record(inputRecord.key(), inputRecord.value(), inputRecord.timestamp(), inputRecord.headers());
71        *         // * create a fresh, empty Headers:
72        *         //   toForward = new Record(inputRecord.key(), inputRecord.value(), inputRecord.timestamp());
73        *         // * etc.
74        *
75        *         // now, we are modifying our own independent copy of the headers.
76        *         toForward.headers().add(...);
77        *         context.forward(toForward);
78        *     }
79        * </code>
80        * @param record The record to forward to all children
81        */
82       <K extends KForward, V extends VForward> void forward(Record<K, V> record);
83   
84       /**
85        * Forward a record to the specified child processor.
86        * See {@link ProcessorContext#forward(Record)} for considerations.
87        *
88        * @param record The record to forward
89        * @param childName The name of the child processor to receive the record
90        * @see ProcessorContext#forward(Record)
91        */
92       <K extends KForward, V extends VForward> void forward(Record<K, V> record, final String childName);
93   }
