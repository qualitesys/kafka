248       private void maybeAddNodeForOptimizationMetadata(final GraphNode node) {
249           node.setBuildPriority(buildPriorityIndex.getAndIncrement());
250   
251           if (node.parentNodes().isEmpty() && !node.nodeName().equals(TOPOLOGY_ROOT)) {
252               throw new IllegalStateException(
253                   "Nodes should not have a null parent node.  Name: " + node.nodeName() + " Type: "
254                   + node.getClass().getSimpleName());
255           }
256   
257           if (node.isKeyChangingOperation()) {
258               keyChangingOperationsToOptimizableRepartitionNodes.put(node, new LinkedHashSet<>());
259           } else if (node instanceof OptimizableRepartitionNode) {
260               final GraphNode parentNode = getKeyChangingParentNode(node);
261               if (parentNode != null) {
262                   keyChangingOperationsToOptimizableRepartitionNodes.get(parentNode).add((OptimizableRepartitionNode<?, ?>) node);
263               }
264           } else if (node.isMergeNode()) {
265               mergeNodes.add(node);
266           } else if (node instanceof TableSourceNode) {
267               tableSourceNodes.add(node);
268           }
269       }

276       public void buildAndOptimizeTopology(final boolean optimizeTopology) {
277   
278           mergeDuplicateSourceNodes();
279           if (optimizeTopology) {
280               LOG.debug("Optimizing the Kafka Streams graph for repartition nodes");
281               optimizeKTableSourceTopics();
282               maybeOptimizeRepartitionOperations();
283           }
284   
285           final PriorityQueue<GraphNode> graphNodePriorityQueue = new PriorityQueue<>(5, Comparator.comparing(GraphNode::buildPriority));
286   
287           graphNodePriorityQueue.offer(root);
288   
289           while (!graphNodePriorityQueue.isEmpty()) {
290               final GraphNode streamGraphNode = graphNodePriorityQueue.remove();
291   
292               if (LOG.isDebugEnabled()) {
293                   LOG.debug("Adding nodes to topology {} child nodes {}", streamGraphNode, streamGraphNode.children());
294               }
295   
296               if (streamGraphNode.allParentsWrittenToTopology() && !streamGraphNode.hasWrittenToTopology()) {
297                   streamGraphNode.writeToTopology(internalTopologyBuilder);
298                   streamGraphNode.setHasWrittenToTopology(true);
299               }
300   
301               for (final GraphNode graphNode : streamGraphNode.children()) {
302                   graphNodePriorityQueue.offer(graphNode);
303               }
304           }
305           internalTopologyBuilder.validateCopartition();
306       }

308       private void mergeDuplicateSourceNodes() {
309           final Map<String, StreamSourceNode<?, ?>> topicsToSourceNodes = new HashMap<>();
310   
311           // We don't really care about the order here, but since Pattern does not implement equals() we can't rely on
312           // a regular HashMap and containsKey(Pattern). But for our purposes it's sufficient to compare the compiled
313           // string and flags to determine if two pattern subscriptions can be merged into a single source node
314           final Map<Pattern, StreamSourceNode<?, ?>> patternsToSourceNodes =
315               new TreeMap<>(Comparator.comparing(Pattern::pattern).thenComparing(Pattern::flags));
316   
317           for (final GraphNode graphNode : root.children()) {
318               if (graphNode instanceof StreamSourceNode) {
319                   final StreamSourceNode<?, ?> currentSourceNode = (StreamSourceNode<?, ?>) graphNode;
320   
321                   if (currentSourceNode.topicPattern().isPresent()) {
322                       final Pattern topicPattern = currentSourceNode.topicPattern().get();
323                       if (!patternsToSourceNodes.containsKey(topicPattern)) {
324                           patternsToSourceNodes.put(topicPattern, currentSourceNode);
325                       } else {
326                           final StreamSourceNode<?, ?> mainSourceNode = patternsToSourceNodes.get(topicPattern);
327                           mainSourceNode.merge(currentSourceNode);
328                           root.removeChild(graphNode);
329                       }
330                   } else {
331                       if (currentSourceNode.topicNames().isPresent()) {
332                           for (final String topic : currentSourceNode.topicNames().get()) {
333                               if (!topicsToSourceNodes.containsKey(topic)) {
334                                   topicsToSourceNodes.put(topic, currentSourceNode);
335                               } else {
336                                   final StreamSourceNode<?, ?> mainSourceNode = topicsToSourceNodes.get(
337                                       topic);
338                                   // TODO we only merge source nodes if the subscribed topic(s) are an exact match, so it's still not
339                                   // possible to subscribe to topicA in one KStream and topicA + topicB in another. We could achieve
340                                   // this by splitting these source nodes into one topic per node and routing to the subscribed children
341                                   if (!mainSourceNode.topicNames()
342                                       .equals(currentSourceNode.topicNames())) {
343                                       LOG.error(
344                                           "Topic {} was found in  subscription for non-equal source nodes {} and {}",
345                                           topic, mainSourceNode, currentSourceNode);
346                                       throw new TopologyException(
347                                           "Two source nodes are subscribed to overlapping but not equal input topics");
348                                   }
349                                   mainSourceNode.merge(currentSourceNode);
350                                   root.removeChild(graphNode);
351                               }
352                           }
353                       }
354                   }
355               }
356           }
357       }

364       private void maybeOptimizeRepartitionOperations() {
365           maybeUpdateKeyChangingRepartitionNodeMap();
366           final Iterator<Entry<GraphNode, LinkedHashSet<OptimizableRepartitionNode<?, ?>>>> entryIterator =
367               keyChangingOperationsToOptimizableRepartitionNodes.entrySet().iterator();
368   
369           while (entryIterator.hasNext()) {
370               final Map.Entry<GraphNode, LinkedHashSet<OptimizableRepartitionNode<?, ?>>> entry = entryIterator.next();
371   
372               final GraphNode keyChangingNode = entry.getKey();
373   
374               if (entry.getValue().isEmpty()) {
375                   continue;
376               }
377   
378               final GroupedInternal<?, ?> groupedInternal = new GroupedInternal<>(getRepartitionSerdes(entry.getValue()));
379   
380               final String repartitionTopicName = getFirstRepartitionTopicName(entry.getValue());
381               //passing in the name of the first repartition topic, re-used to create the optimized repartition topic
382               final GraphNode optimizedSingleRepartition = createRepartitionNode(repartitionTopicName,
383                                                                                  groupedInternal.keySerde(),
384                                                                                  groupedInternal.valueSerde());
385   
386               // re-use parent buildPriority to make sure the single repartition graph node is evaluated before downstream nodes
387               optimizedSingleRepartition.setBuildPriority(keyChangingNode.buildPriority());
388   
389               for (final OptimizableRepartitionNode<?, ?> repartitionNodeToBeReplaced : entry.getValue()) {
390   
391                   final GraphNode keyChangingNodeChild = findParentNodeMatching(repartitionNodeToBeReplaced, gn -> gn.parentNodes().contains(keyChangingNode));
392   
393                   if (keyChangingNodeChild == null) {
394                       throw new StreamsException(String.format("Found a null keyChangingChild node for %s", repartitionNodeToBeReplaced));
395                   }
396   
397                   LOG.debug("Found the child node of the key changer {} from the repartition {}.", keyChangingNodeChild, repartitionNodeToBeReplaced);
398   
399                   // need to add children of key-changing node as children of optimized repartition
400                   // in order to process records from re-partitioning
401                   optimizedSingleRepartition.addChild(keyChangingNodeChild);
402   
403                   LOG.debug("Removing {} from {}  children {}", keyChangingNodeChild, keyChangingNode, keyChangingNode.children());
404                   // now remove children from key-changing node
405                   keyChangingNode.removeChild(keyChangingNodeChild);
406   
407                   // now need to get children of repartition node so we can remove repartition node
408                   final Collection<GraphNode> repartitionNodeToBeReplacedChildren = repartitionNodeToBeReplaced.children();
409                   final Collection<GraphNode> parentsOfRepartitionNodeToBeReplaced = repartitionNodeToBeReplaced.parentNodes();
410   
411                   for (final GraphNode repartitionNodeToBeReplacedChild : repartitionNodeToBeReplacedChildren) {
412                       for (final GraphNode parentNode : parentsOfRepartitionNodeToBeReplaced) {
413                           parentNode.addChild(repartitionNodeToBeReplacedChild);
414                       }
415                   }
416   
417                   for (final GraphNode parentNode : parentsOfRepartitionNodeToBeReplaced) {
418                       parentNode.removeChild(repartitionNodeToBeReplaced);
419                   }
420                   repartitionNodeToBeReplaced.clearChildren();
421   
422                   // if replaced repartition node is part of any copartition group,
423                   // we need to update it with the new node name so that co-partitioning won't break.
424                   internalTopologyBuilder.maybeUpdateCopartitionSourceGroups(repartitionNodeToBeReplaced.nodeName(),
425                                                                              optimizedSingleRepartition.nodeName());
426   
427                   LOG.debug("Updated node {} children {}", optimizedSingleRepartition, optimizedSingleRepartition.children());
428               }
429   
430               keyChangingNode.addChild(optimizedSingleRepartition);
431               entryIterator.remove();
432           }
433       }

435       private void maybeUpdateKeyChangingRepartitionNodeMap() {
436           final Map<GraphNode, Set<GraphNode>> mergeNodesToKeyChangers = new HashMap<>();
437           final Set<GraphNode> mergeNodeKeyChangingParentsToRemove = new HashSet<>();
438           for (final GraphNode mergeNode : mergeNodes) {
439               mergeNodesToKeyChangers.put(mergeNode, new LinkedHashSet<>());
440               final Set<Map.Entry<GraphNode, LinkedHashSet<OptimizableRepartitionNode<?, ?>>>> entrySet = keyChangingOperationsToOptimizableRepartitionNodes.entrySet();
441               for (final Map.Entry<GraphNode, LinkedHashSet<OptimizableRepartitionNode<?, ?>>> entry : entrySet) {
442                   if (mergeNodeHasRepartitionChildren(mergeNode, entry.getValue())) {
443                       final GraphNode maybeParentKey = findParentNodeMatching(mergeNode, node -> node.parentNodes().contains(entry.getKey()));
444                       if (maybeParentKey != null) {
445                           mergeNodesToKeyChangers.get(mergeNode).add(entry.getKey());
446                       }
447                   }
448               }
449           }
450   
451           for (final Map.Entry<GraphNode, Set<GraphNode>> entry : mergeNodesToKeyChangers.entrySet()) {
452               final GraphNode mergeKey = entry.getKey();
453               final Collection<GraphNode> keyChangingParents = entry.getValue();
454               final LinkedHashSet<OptimizableRepartitionNode<?, ?>> repartitionNodes = new LinkedHashSet<>();
455               for (final GraphNode keyChangingParent : keyChangingParents) {
456                   repartitionNodes.addAll(keyChangingOperationsToOptimizableRepartitionNodes.get(keyChangingParent));
457                   mergeNodeKeyChangingParentsToRemove.add(keyChangingParent);
458               }
459               keyChangingOperationsToOptimizableRepartitionNodes.put(mergeKey, repartitionNodes);
460           }
461   
462           for (final GraphNode mergeNodeKeyChangingParent : mergeNodeKeyChangingParentsToRemove) {
463               keyChangingOperationsToOptimizableRepartitionNodes.remove(mergeNodeKeyChangingParent);
464           }
465       }

532       private GraphNode findParentNodeMatching(final GraphNode startSeekingNode,
533                                                final Predicate<GraphNode> parentNodePredicate) {
534           if (parentNodePredicate.test(startSeekingNode)) {
535               return startSeekingNode;
536           }
537           GraphNode foundParentNode = null;
538   
539           for (final GraphNode parentNode : startSeekingNode.parentNodes()) {
540               if (parentNodePredicate.test(parentNode)) {
541                   return parentNode;
542               }
543               foundParentNode = findParentNodeMatching(parentNode, parentNodePredicate);
544           }
545           return foundParentNode;
