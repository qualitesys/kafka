21   import org.apache.kafka.streams.kstream.SlidingWindows;

24   import org.apache.kafka.streams.kstream.internals.KStreamSlidingWindowAggregate;
25   import org.apache.kafka.streams.kstream.internals.KStreamWindowAggregate;

28   public final class GraphGraceSearchUtil {
29       private GraphGraceSearchUtil() {}
30   
31       public static long findAndVerifyWindowGrace(final GraphNode graphNode) {
32           return findAndVerifyWindowGrace(graphNode, "");
33       }
34   
35       private static long findAndVerifyWindowGrace(final GraphNode graphNode, final String chain) {
36           // error base case: we traversed off the end of the graph without finding a window definition
37           if (graphNode == null) {
38               throw new TopologyException(
39                   "Window close time is only defined for windowed computations. Got [" + chain + "]."
40               );
41           }
42           // base case: return if this node defines a grace period.
43           {
44               final Long gracePeriod = extractGracePeriod(graphNode);
45               if (gracePeriod != null) {
46                   return gracePeriod;
47               }
48           }
49   
50           final String newChain = chain.equals("") ? graphNode.nodeName() : graphNode.nodeName() + "->" + chain;
51   
52           if (graphNode.parentNodes().isEmpty()) {
53               // error base case: we traversed to the end of the graph without finding a window definition
54               throw new TopologyException(
55                   "Window close time is only defined for windowed computations. Got [" + newChain + "]."
56               );
57           }
58   
59           // recursive case: all parents must define a grace period, and we use the max of our parents' graces.
60           long inheritedGrace = -1;
61           for (final GraphNode parentNode : graphNode.parentNodes()) {
62               final long parentGrace = findAndVerifyWindowGrace(parentNode, newChain);
63               inheritedGrace = Math.max(inheritedGrace, parentGrace);
64           }
65   
66           if (inheritedGrace == -1) {
67               throw new IllegalStateException(); // shouldn't happen, and it's not a legal grace period
68           }
69   
70           return inheritedGrace;
71       }
72   
73       @SuppressWarnings("rawtypes")
74       private static Long extractGracePeriod(final GraphNode node) {
75           if (node instanceof StatefulProcessorNode) {
76               final ProcessorSupplier processorSupplier = ((StatefulProcessorNode) node).processorParameters().processorSupplier();
77               if (processorSupplier instanceof KStreamWindowAggregate) {
78                   final KStreamWindowAggregate kStreamWindowAggregate = (KStreamWindowAggregate) processorSupplier;
79                   final Windows windows = kStreamWindowAggregate.windows();
80                   return windows.gracePeriodMs();
81               } else if (processorSupplier instanceof KStreamSessionWindowAggregate) {
82                   final KStreamSessionWindowAggregate kStreamSessionWindowAggregate = (KStreamSessionWindowAggregate) processorSupplier;
83                   final SessionWindows windows = kStreamSessionWindowAggregate.windows();
84                   return windows.gracePeriodMs() + windows.inactivityGap();
85               } else if (processorSupplier instanceof KStreamSlidingWindowAggregate) {
86                   final KStreamSlidingWindowAggregate kStreamSlidingWindowAggregate = (KStreamSlidingWindowAggregate) processorSupplier;
87                   final SlidingWindows windows = kStreamSlidingWindowAggregate.windows();
88                   return windows.gracePeriodMs();
89               } else {
90                   return null;
91               }
92           } else {
93               return null;
94           }
95       }
96   }
