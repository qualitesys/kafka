113       /**
114        * @throws StreamsException if the consumer throws an exception
115        * @throws org.apache.kafka.common.errors.TimeoutException if the request times out
116        */
117       public static Map<TopicPartition, Long> fetchCommittedOffsets(final Set<TopicPartition> partitions,
118                                                                     final Consumer<byte[], byte[]> consumer) {
119           if (partitions.isEmpty()) {
120               return Collections.emptyMap();
121           }
122   
123           final Map<TopicPartition, Long> committedOffsets;
124           try {
125               // those which do not have a committed offset would default to 0
126               committedOffsets = consumer.committed(partitions).entrySet().stream()
127                   .collect(Collectors.toMap(Map.Entry::getKey, e -> e.getValue() == null ? 0L : e.getValue().offset()));
128           } catch (final TimeoutException timeoutException) {
129               LOG.warn("The committed offsets request timed out, try increasing the consumer client's default.api.timeout.ms", timeoutException);
130               throw timeoutException;
131           } catch (final KafkaException fatal) {
132               LOG.warn("The committed offsets request failed.", fatal);
133               throw new StreamsException(String.format("Failed to retrieve end offsets for %s", partitions), fatal);
134           }
135   
136           return committedOffsets;
137       }

139       public static KafkaFuture<Map<TopicPartition, ListOffsetsResultInfo>> fetchEndOffsetsFuture(final Collection<TopicPartition> partitions,
140                                                                                                   final Admin adminClient) {
141           return adminClient.listOffsets(
142               partitions.stream().collect(Collectors.toMap(Function.identity(), tp -> OffsetSpec.latest()))
143           ).all();
144       }

146       /**
147        * A helper method that wraps the {@code Future#get} call and rethrows any thrown exception as a StreamsException
148        * @throws StreamsException if the admin client request throws an exception
149        */
150       public static Map<TopicPartition, ListOffsetsResultInfo> getEndOffsets(final KafkaFuture<Map<TopicPartition, ListOffsetsResultInfo>> endOffsetsFuture) {
151           try {
152               return endOffsetsFuture.get();
153           } catch (final RuntimeException | InterruptedException | ExecutionException e) {
154               LOG.warn("The listOffsets request failed.", e);
155               throw new StreamsException("Unable to obtain end offsets from kafka", e);
156           }
157       }

160        * @throws StreamsException if the admin client request throws an exception
161        */
162       public static Map<TopicPartition, ListOffsetsResultInfo> fetchEndOffsets(final Collection<TopicPartition> partitions,
163                                                                                final Admin adminClient) {
164           if (partitions.isEmpty()) {
165               return Collections.emptyMap();
166           }
167           return getEndOffsets(fetchEndOffsetsFuture(partitions, adminClient));
168       }

193       private static long recordSizeInBytes(final long keyBytes,
194                                             final long valueBytes,
195                                             final String topic,
196                                             final Headers headers) {
197           long headerSizeInBytes = 0L;
198   
199           if (headers != null) {
200               for (final Header header : headers.toArray()) {
201                   headerSizeInBytes += Utils.utf8(header.key()).length;
202                   if (header.value() != null) {
203                       headerSizeInBytes += header.value().length;
204                   }
205               }
206           }
207   
208           return keyBytes +
209               valueBytes +
210               8L + // timestamp
211               8L + // offset
212               Utils.utf8(topic).length +
213               4L + // partition
214               headerSizeInBytes;
215       }
