17   package org.apache.kafka.streams.kstream.internals;
18   
19   import org.apache.kafka.common.serialization.Deserializer;
20   import org.apache.kafka.common.serialization.Serde;
21   import org.apache.kafka.common.serialization.Serializer;
22   import org.apache.kafka.streams.processor.api.ProcessorContext;
23   import org.apache.kafka.streams.processor.internals.SerdeGetter;
24   
25   /**
26    * If a component's serdes are Wrapping serdes, then they require a little extra setup
27    * to be fully initialized at run time.
28    */
29   public class WrappingNullableUtils {
30   
31       @SuppressWarnings("unchecked")
32       private static <T> Deserializer<T> prepareDeserializer(final Deserializer<T> specificDeserializer, final ProcessorContext context, final boolean isKey, final String name) {
33           final Deserializer<T> deserializerToUse;
34   
35           if (specificDeserializer == null) {
36               final Deserializer<?> contextKeyDeserializer = context.keySerde().deserializer();
37               final Deserializer<?> contextValueDeserializer = context.valueSerde().deserializer();
38               deserializerToUse = (Deserializer<T>) (isKey ? contextKeyDeserializer : contextValueDeserializer);
39           } else {
40               deserializerToUse = specificDeserializer;
41               initNullableDeserializer(deserializerToUse, new SerdeGetter(context));
42           }
43           return deserializerToUse;
44       }
45       @SuppressWarnings("unchecked")
46       private static <T> Serializer<T> prepareSerializer(final Serializer<T> specificSerializer, final ProcessorContext context, final boolean isKey, final String name) {
47           final Serializer<T> serializerToUse;
48           if (specificSerializer == null) {
49               final Serializer<?> contextKeySerializer = context.keySerde().serializer();
50               final Serializer<?> contextValueSerializer = context.valueSerde().serializer();
51               serializerToUse = (Serializer<T>) (isKey ? contextKeySerializer : contextValueSerializer);
52           } else {
53               serializerToUse = specificSerializer;
54               initNullableSerializer(serializerToUse, new SerdeGetter(context));
55           }
56           return serializerToUse;
57       }
58   
59       @SuppressWarnings({"rawtypes", "unchecked"})
60       private static <T> Serde<T> prepareSerde(final Serde<T> specificSerde, final SerdeGetter getter, final boolean isKey) {
61           final Serde<T> serdeToUse;
62           if (specificSerde == null) {
63               serdeToUse = (Serde<T>) (isKey ?  getter.keySerde() : getter.valueSerde());
64           } else {
65               serdeToUse = specificSerde;
66           }
67           if (serdeToUse instanceof WrappingNullableSerde) {
68               ((WrappingNullableSerde) serdeToUse).setIfUnset(getter);
69           }
70           return serdeToUse;
71       }
72   
73       public static <K> Deserializer<K> prepareKeyDeserializer(final Deserializer<K> specificDeserializer, final ProcessorContext context, final String name) {
74           return prepareDeserializer(specificDeserializer, context, true, name);
75       }
76   
77       public static <V> Deserializer<V> prepareValueDeserializer(final Deserializer<V> specificDeserializer, final ProcessorContext context, final String name) {
78           return prepareDeserializer(specificDeserializer, context, false, name);
79       }
80   
81       public static <K> Serializer<K> prepareKeySerializer(final Serializer<K> specificSerializer, final ProcessorContext context, final String name) {
82           return prepareSerializer(specificSerializer, context, true, name);
83       }
84   
85       public static <V> Serializer<V> prepareValueSerializer(final Serializer<V> specificSerializer, final ProcessorContext context, final String name) {
86           return prepareSerializer(specificSerializer, context, false, name);
87       }
88   
89       public static <K> Serde<K> prepareKeySerde(final Serde<K> specificSerde, final SerdeGetter getter) {
90           return prepareSerde(specificSerde, getter, true);
91       }
92   
93       public static <V> Serde<V> prepareValueSerde(final Serde<V> specificSerde, final SerdeGetter getter) {
94           return prepareSerde(specificSerde, getter, false);
95       }
96       @SuppressWarnings({"rawtypes", "unchecked"})
97       public static <T> void initNullableSerializer(final Serializer<T> specificSerializer, final SerdeGetter getter) {
98           if (specificSerializer instanceof WrappingNullableSerializer) {
99               ((WrappingNullableSerializer) specificSerializer).setIfUnset(getter);
100           }
101       }
102       @SuppressWarnings({"rawtypes", "unchecked"})
103       public static <T> void initNullableDeserializer(final Deserializer<T> specificDeserializer, final SerdeGetter getter) {
104           if (specificDeserializer instanceof WrappingNullableDeserializer) {
105               ((WrappingNullableDeserializer) specificDeserializer).setIfUnset(getter);
106           }
107       }
108   
109   }
