102       /**
103        * Ensures that the state base directory as well as the application's sub-directory are created.
104        *
105        * @param config              streams application configuration to read the root state directory path
106        * @param time                system timer used to execute periodic cleanup procedure
107        * @param hasPersistentStores only when the application's topology does have stores persisted on local file
108        *                            system, we would go ahead and auto-create the corresponding application / task / store
109        *                            directories whenever necessary; otherwise no directories would be created.
110        * @param hasNamedTopologies  whether this application is composed of independent named topologies
111        *
112        * @throws ProcessorStateException if the base state directory or application state directory does not exist
113        *                                 and could not be created when hasPersistentStores is enabled.
114        */
115       public StateDirectory(final StreamsConfig config, final Time time, final boolean hasPersistentStores, final boolean hasNamedTopologies) {
116           this.time = time;
117           this.hasPersistentStores = hasPersistentStores;
118           this.hasNamedTopologies = hasNamedTopologies;
119           this.appId = config.getString(StreamsConfig.APPLICATION_ID_CONFIG);
120           final String stateDirName = config.getString(StreamsConfig.STATE_DIR_CONFIG);
121           final File baseDir = new File(stateDirName);
122           stateDir = new File(baseDir, appId);
123   
124           if (this.hasPersistentStores) {
125               if (!baseDir.exists() && !baseDir.mkdirs()) {
126                   throw new ProcessorStateException(
127                       String.format("base state directory [%s] doesn't exist and couldn't be created", stateDirName));
128               }
129               if (!stateDir.exists() && !stateDir.mkdir()) {
130                   throw new ProcessorStateException(
131                       String.format("state directory [%s] doesn't exist and couldn't be created", stateDir.getPath()));
132               } else if (stateDir.exists() && !stateDir.isDirectory()) {
133                   throw new ProcessorStateException(
134                       String.format("state directory [%s] can't be created as there is an existing file with the same name", stateDir.getPath()));
135               }
136   
137               if (stateDirName.startsWith(System.getProperty("java.io.tmpdir"))) {
138                   log.warn("Using an OS temp directory in the state.dir property can cause failures with writing" +
139                       " the checkpoint file due to the fact that this directory can be cleared by the OS." +
140                       " Resolved state.dir: [" + stateDirName + "]");
141   
142               }
143               // change the dir permission to "rwxr-x---" to avoid world readable
144               configurePermissions(baseDir);
145               configurePermissions(stateDir);
146           }

149       private void configurePermissions(final File file) {
150           final Path path = file.toPath();
151           if (path.getFileSystem().supportedFileAttributeViews().contains("posix")) {
152               final Set<PosixFilePermission> perms = PosixFilePermissions.fromString("rwxr-x---");
153               try {
154                   Files.setPosixFilePermissions(path, perms);
155               } catch (final IOException e) {
156                   log.error("Error changing permissions for the directory {} ", path, e);
157               }
158           } else {
159               boolean set = file.setReadable(true, true);
160               set &= file.setWritable(true, true);
161               set &= file.setExecutable(true, true);
162               if (!set) {
163                   log.error("Failed to change permissions for the directory {}", file);
164               }
165           }
166       }

168       /**
169        * @return true if the state directory was successfully locked
170        */
171       private boolean lockStateDirectory() {
172           final File lockFile = new File(stateDir, LOCK_FILE_NAME);
173           try {
174               stateDirLockChannel = FileChannel.open(lockFile.toPath(), StandardOpenOption.CREATE, StandardOpenOption.WRITE);
175               stateDirLock = tryLock(stateDirLockChannel);
176           } catch (final IOException e) {
177               log.error("Unable to lock the state directory due to unexpected exception", e);
178               throw new ProcessorStateException(String.format("Failed to lock the state directory [%s] during startup",
179                   stateDir.getAbsolutePath()), e);
180           }
181           return stateDirLock != null;
182       }

184       public UUID initializeProcessId() {
185           if (!hasPersistentStores) {
186               return UUID.randomUUID();
187           }
188   
189           if (!lockStateDirectory()) {
190               log.error("Unable to obtain lock as state directory is already locked by another process");
191               throw new StreamsException(String.format("Unable to initialize state, this can happen if multiple instances of " +
192                                              "Kafka Streams are running in the same state directory " +
193                                              "(current state directory is [%s]", stateDir.getAbsolutePath()));
194           }
195   
196           final File processFile = new File(stateDir, PROCESS_FILE_NAME);
197           final ObjectMapper mapper = new ObjectMapper();
198   
199           try {
200               if (processFile.exists()) {
201                   try {
202                       final StateDirectoryProcessFile processFileData = mapper.readValue(processFile, StateDirectoryProcessFile.class);
203                       log.info("Reading UUID from process file: {}", processFileData.processId);
204                       if (processFileData.processId != null) {
205                           return processFileData.processId;
206                       }
207                   } catch (final Exception e) {
208                       log.warn("Failed to read json process file", e);
209                   }
210               }
211   
212               final StateDirectoryProcessFile processFileData = new StateDirectoryProcessFile(UUID.randomUUID());
213               log.info("No process id found on disk, got fresh process id {}", processFileData.processId);
214   
215               mapper.writeValue(processFile, processFileData);
216               return processFileData.processId;
217           } catch (final IOException e) {
218               log.error("Unable to read/write process file due to unexpected exception", e);
219               throw new ProcessorStateException(e);
220           }
221       }

223       /**
224        * Get or create the directory for the provided {@link TaskId}.
225        * @return directory for the {@link TaskId}
226        * @throws ProcessorStateException if the task directory does not exist and could not be created
227        */
228       public File getOrCreateDirectoryForTask(final TaskId taskId) {
229           final File taskParentDir = getTaskDirectoryParentName(taskId);
230           final File taskDir = new File(taskParentDir, StateManagerUtil.toTaskDirString(taskId));
231           if (hasPersistentStores) {
232               if (!taskDir.exists()) {
233                   synchronized (taskDirCreationLock) {
234                       // to avoid a race condition, we need to check again if the directory does not exist:
235                       // otherwise, two threads might pass the outer `if` (and enter the `then` block),
236                       // one blocks on `synchronized` while the other creates the directory,
237                       // and the blocking one fails when trying to create it after it's unblocked
238                       if (!taskParentDir.exists() && !taskParentDir.mkdir()) {
239                           throw new ProcessorStateException(
240                               String.format("Parent [%s] of task directory [%s] doesn't exist and couldn't be created",
241                                   taskParentDir.getPath(), taskDir.getPath()));
242                       }
243                       if (!taskDir.exists() && !taskDir.mkdir()) {
244                           throw new ProcessorStateException(
245                               String.format("task directory [%s] doesn't exist and couldn't be created", taskDir.getPath()));
246                       }
247                   }
248               } else if (!taskDir.isDirectory()) {
249                   throw new ProcessorStateException(
250                       String.format("state directory [%s] can't be created as there is an existing file with the same name", taskDir.getPath()));
251               }
252           }
253           return taskDir;
254       }

288       private boolean taskDirIsEmpty(final File taskDir) {
289           final File[] storeDirs = taskDir.listFiles(pathname ->
290                   !pathname.getName().equals(CHECKPOINT_FILE_NAME));
291   
292           boolean taskDirEmpty = true;
293   
294           // if the task is stateless, storeDirs would be null
295           if (storeDirs != null && storeDirs.length > 0) {
296               for (final File file : storeDirs) {
297                   // We removed the task directory locking but some upgrading applications may still have old lock files on disk,
298                   // we just lazily delete those in this method since it's the only thing that would be affected by these
299                   if (file.getName().endsWith(LOCK_FILE_NAME)) {
300                       if (!file.delete()) {
301                           // If we hit an error deleting this just ignore it and move on, we'll retry again at some point
302                           log.warn("Error encountered deleting lock file in {}", taskDir);
303                       }
304                   } else {
305                       // If it's not a lock file then the directory is not empty,
306                       // but finish up the loop in case there's a lock file left to delete
307                       log.trace("TaskDir {} was not empty, found {}", taskDir, file);
308                       taskDirEmpty = false;
309                   }
310               }
311           }
312           return taskDirEmpty;
313       }

334       private String logPrefix() {
335           return String.format("stream-thread [%s]", Thread.currentThread().getName());
336       }
337   
338       /**
339        * Get the lock for the {@link TaskId}s directory if it is available
340        * @param taskId task id
341        * @return true if successful
342        */
343       synchronized boolean lock(final TaskId taskId) {
344           if (!hasPersistentStores) {
345               return true;
346           }
347   
348           final Thread lockOwner = lockedTasksToOwner.get(taskId);
349           if (lockOwner != null) {
350               if (lockOwner.equals(Thread.currentThread())) {
351                   log.trace("{} Found cached state dir lock for task {}", logPrefix(), taskId);
352                   // we already own the lock
353                   return true;
354               } else {
355                   // another thread owns the lock
356                   return false;
357               }
358           } else if (!stateDir.exists()) {
359               log.error("Tried to lock task directory for {} but the state directory does not exist", taskId);
360               throw new IllegalStateException("The state directory has been deleted");
361           } else {
362               lockedTasksToOwner.put(taskId, Thread.currentThread());
363               // make sure the task directory actually exists, and create it if not
364               getOrCreateDirectoryForTask(taskId);
365               return true;
366           }
367       }
368   
369       /**
370        * Unlock the state directory for the given {@link TaskId}.
371        */
372       synchronized void unlock(final TaskId taskId) {
373           final Thread lockOwner = lockedTasksToOwner.get(taskId);
374           if (lockOwner != null && lockOwner.equals(Thread.currentThread())) {
375               lockedTasksToOwner.remove(taskId);
376               log.debug("{} Released state dir lock for task {}", logPrefix(), taskId);
377           }
378       }

380       public void close() {
381           if (hasPersistentStores) {
382               try {
383                   stateDirLock.release();
384                   stateDirLockChannel.close();
385   
386                   stateDirLock = null;
387                   stateDirLockChannel = null;
388               } catch (final IOException e) {
389                   log.error("Unexpected exception while unlocking the state dir", e);
390                   throw new StreamsException(String.format("Failed to release the lock on the state directory [%s]", stateDir.getAbsolutePath()), e);
391               }
392   
393               // all threads should be stopped and cleaned up by now, so none should remain holding a lock
394               if (!lockedTasksToOwner.isEmpty()) {
395                   log.error("Some task directories still locked while closing state, this indicates unclean shutdown: {}", lockedTasksToOwner);
396               }
397           }
398       }

400       public synchronized void clean() {
401           try {
402               cleanStateAndTaskDirectoriesCalledByUser();
403           } catch (final Exception e) {
404               throw new StreamsException(e);
405           }
406   
407           try {
408               if (stateDir.exists()) {
409                   Utils.delete(globalStateDir().getAbsoluteFile());
410               }
411           } catch (final IOException exception) {
412               log.error(
413                   String.format("%s Failed to delete global state directory of %s due to an unexpected exception",
414                       logPrefix(), appId),
415                   exception
416               );
417               throw new StreamsException(exception);
418           }
419   
420           try {
421               if (hasPersistentStores && stateDir.exists() && !stateDir.delete()) {
422                   log.warn(
423                       String.format("%s Failed to delete state store directory of %s for it is not empty",
424                           logPrefix(), stateDir.getAbsolutePath())
425                   );
426               }
427           } catch (final SecurityException exception) {
428               log.error(
429                   String.format("%s Failed to delete state store directory of %s due to an unexpected exception",
430                       logPrefix(), stateDir.getAbsolutePath()),
431                   exception
432               );
433               throw new StreamsException(exception);
434           }
435       }

437       /**
438        * Remove the directories for any {@link TaskId}s that are no-longer
439        * owned by this {@link StreamThread} and aren't locked by either
440        * another process or another {@link StreamThread}
441        * @param cleanupDelayMs only remove directories if they haven't been modified for at least
442        *                       this amount of time (milliseconds)
443        */
444       public synchronized void cleanRemovedTasks(final long cleanupDelayMs) {
445           try {
446               cleanRemovedTasksCalledByCleanerThread(cleanupDelayMs);
447           } catch (final Exception cannotHappen) {
448               throw new IllegalStateException("Should have swallowed exception.", cannotHappen);
449           }
450       }

452       private void cleanRemovedTasksCalledByCleanerThread(final long cleanupDelayMs) {
453           for (final TaskDirectory taskDir : listAllTaskDirectories()) {
454               final String dirName = taskDir.file().getName();
455               final TaskId id = parseTaskDirectoryName(dirName, taskDir.namedTopology());
456               if (!lockedTasksToOwner.containsKey(id)) {
457                   try {
458                       if (lock(id)) {
459                           final long now = time.milliseconds();
460                           final long lastModifiedMs = taskDir.file().lastModified();
461                           if (now - cleanupDelayMs > lastModifiedMs) {
462                               log.info("{} Deleting obsolete state directory {} for task {} as {}ms has elapsed (cleanup delay is {}ms).",
463                                   logPrefix(), dirName, id, now - lastModifiedMs, cleanupDelayMs);
464                               Utils.delete(taskDir.file());
465                           }
466                       }
467                   } catch (final IOException exception) {
468                       log.warn(
469                           String.format("%s Swallowed the following exception during deletion of obsolete state directory %s for task %s:",
470                               logPrefix(), dirName, id),
471                           exception
472                       );
473                   } finally {
474                       unlock(id);
475                   }
476               }
477           }
478           // Ok to ignore returned exception as it should be swallowed
479           maybeCleanEmptyNamedTopologyDirs(true);
480       }

482       /**
483        * Cleans up any leftover named topology directories that are empty, if any exist
484        * @param logExceptionAsWarn if true, an exception will be logged as a warning
485        *                       if false, an exception will be logged as error
486        * @return the first IOException to be encountered
487        */
488       private IOException maybeCleanEmptyNamedTopologyDirs(final boolean logExceptionAsWarn) {
489           if (!hasNamedTopologies) {
490               return null;
491           }
492   
493           final AtomicReference<IOException> firstException = new AtomicReference<>(null);
494           final File[] namedTopologyDirs = stateDir.listFiles(pathname ->
495                   pathname.isDirectory() && NAMED_TOPOLOGY_DIR_PATH_NAME.matcher(pathname.getName()).matches()
496           );
497           if (namedTopologyDirs != null) {
498               for (final File namedTopologyDir : namedTopologyDirs) {
499                   final File[] contents = namedTopologyDir.listFiles();
500                   if (contents != null && contents.length == 0) {
501                       try {
502                           Utils.delete(namedTopologyDir);
503                       } catch (final IOException exception) {
504                           if (logExceptionAsWarn) {
505                               log.warn(
506                                   String.format("%sSwallowed the following exception during deletion of named topology directory %s",
507                                       logPrefix(), namedTopologyDir.getName()),
508                                   exception
509                               );
510                           } else {
511                               log.error(
512                                   String.format("%s Failed to delete named topology directory %s with exception:",
513                                       logPrefix(), namedTopologyDir.getName()),
514                                   exception
515                               );
516                           }
517                           firstException.compareAndSet(null, exception);
518                       }
519                   }
520               }
521           }
522           return firstException.get();
523       }

526        * Clears out any local state found for the given NamedTopology after it was removed
527        *
528        * @throws StreamsException if cleanup failed
529        */
530       public void clearLocalStateForNamedTopology(final String topologyName) {

537           } catch (final IOException e) {

539               throw new StreamsException("Unable to delete state for the named topology " + topologyName,
540                                          e, new TaskId(-1, -1, topologyName)); // use dummy taskid to report source topology for this error
541           }
542       }

544       private void cleanStateAndTaskDirectoriesCalledByUser() throws Exception {
545           if (!lockedTasksToOwner.isEmpty()) {
546               log.warn("Found some still-locked task directories when user requested to cleaning up the state, "
547                   + "since Streams is not running any more these will be ignored to complete the cleanup");
548           }
549           final AtomicReference<Exception> firstException = new AtomicReference<>();
550           for (final TaskDirectory taskDir : listAllTaskDirectories()) {
551               final String dirName = taskDir.file().getName();
552               final TaskId id = parseTaskDirectoryName(dirName, taskDir.namedTopology());
553               try {
554                   log.info("{} Deleting task directory {} for {} as user calling cleanup.",
555                       logPrefix(), dirName, id);
556   
557                   if (lockedTasksToOwner.containsKey(id)) {
558                       log.warn("{} Task {} in state directory {} was still locked by {}",
559                           logPrefix(), dirName, id, lockedTasksToOwner.get(id));
560                   }
561                   Utils.delete(taskDir.file());
562               } catch (final IOException exception) {
563                   log.error(
564                       String.format("%s Failed to delete task directory %s for %s with exception:",
565                           logPrefix(), dirName, id),
566                       exception
567                   );
568                   firstException.compareAndSet(null, exception);
569               }
570           }
571   
572           firstException.compareAndSet(null, maybeCleanEmptyNamedTopologyDirs(false));
573   
574           final Exception exception = firstException.get();
575           if (exception != null) {
576               throw exception;
577           }
578       }

602       private List<TaskDirectory> listTaskDirectories(final FileFilter filter) {
603           final List<TaskDirectory> taskDirectories = new ArrayList<>();
604           if (hasPersistentStores && stateDir.exists()) {
605               if (hasNamedTopologies) {
606                   for (final File namedTopologyDir : listNamedTopologyDirs()) {
607                       final String namedTopology = parseNamedTopologyFromDirectory(namedTopologyDir.getName());
608                       final File[] taskDirs = namedTopologyDir.listFiles(filter);
609                       if (taskDirs != null) {
610                           taskDirectories.addAll(Arrays.stream(taskDirs)
611                               .map(f -> new TaskDirectory(f, namedTopology)).collect(Collectors.toList()));
612                       }
613                   }
614               } else {
615                   final File[] taskDirs =
616                       stateDir.listFiles(filter);
617                   if (taskDirs != null) {
618                       taskDirectories.addAll(Arrays.stream(taskDirs)
619                                                  .map(f -> new TaskDirectory(f, null)).collect(Collectors.toList()));
620                   }
621               }
622           }
623   
624           return taskDirectories;
625       }

629           return namedTopologyDirectories != null ? Arrays.asList(namedTopologyDirectories) : Collections.emptyList();
