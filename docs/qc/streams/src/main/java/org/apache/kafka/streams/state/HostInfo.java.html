20   import static org.apache.kafka.common.utils.Utils.getPort;
21   
22   import org.apache.kafka.common.config.ConfigException;
23   import org.apache.kafka.common.utils.Utils;
24   import org.apache.kafka.streams.KafkaStreams;

27   /**
28    * Represents a user defined endpoint in a {@link org.apache.kafka.streams.KafkaStreams} application.
29    * Instances of this class can be obtained by calling one of:
30    *  {@link KafkaStreams#metadataForAllStreamsClients()}
31    *  {@link KafkaStreams#streamsMetadataForStore(String)}
32    *
33    *  The HostInfo is constructed during Partition Assignment
34    *  see {@link StreamsPartitionAssignor}
35    *  It is extracted from the config {@link org.apache.kafka.streams.StreamsConfig#APPLICATION_SERVER_CONFIG}
36    *
37    *  If developers wish to expose an endpoint in their KafkaStreams applications they should provide the above
38    *  config.
39    */
40   public class HostInfo {
41       private final String host;
42       private final int port;
43   
44       public HostInfo(final String host,
45                       final int port) {
46           this.host = host;
47           this.port = port;
48       }
49   
50       /**
51        * @throws ConfigException if the host or port cannot be parsed from the given endpoint string
52        * @return a new HostInfo or null if endPoint is null or has no characters
53        */
54       public static HostInfo buildFromEndpoint(final String endPoint) {
55           if (Utils.isBlank(endPoint)) {
56               return null;
57           }
58   
59           final String host = getHost(endPoint);
60           final Integer port = getPort(endPoint);
61   
62           if (host == null || port == null) {
63               throw new ConfigException(
64                   String.format("Error parsing host address %s. Expected format host:port.", endPoint)
65               );
66           }
67           return new HostInfo(host, port);
68       }
69   
70       /**
71        * @return a sentinel for cases where the host metadata is currently unavailable, eg during rebalance operations.
72        */
73       public static HostInfo unavailable() {
74           return new HostInfo("unavailable", -1);
75       }
76   
77       @Override
78       public boolean equals(final Object o) {
79           if (this == o) {
80               return true;
81           }
82           if (o == null || getClass() != o.getClass()) {
83               return false;
84           }
85   
86           final HostInfo hostInfo = (HostInfo) o;
87           return port == hostInfo.port && host.equals(hostInfo.host);
88       }
89   
90       @Override
91       public int hashCode() {
92           int result = host.hashCode();
93           result = 31 * result + port;
94           return result;
95       }
96   
97       public String host() {
98           return host;
99       }
100   
101       public int port() {
102           return port;
103       }
104   
105       @Override
106       public String toString() {
107           return "HostInfo{" +
108                   "host=\'" + host + '\'' +
109                   ", port=" + port +
110                   '}';
111       }
112   }
