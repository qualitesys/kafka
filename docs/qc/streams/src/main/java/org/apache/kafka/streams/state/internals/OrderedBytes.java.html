18   package org.apache.kafka.streams.state.internals;
19   
20   import org.apache.kafka.common.utils.Bytes;
21   
22   import java.nio.ByteBuffer;
23   
24   class OrderedBytes {
25   
26       private static final int MIN_KEY_LENGTH = 1;
27       /**
28        * Returns the upper byte range for a key with a given fixed size maximum suffix
29        *
30        * Assumes the minimum key length is one byte
31        */
32       static Bytes upperRange(final Bytes key, final byte[] maxSuffix) {
33           final byte[] bytes = key.get();
34           final ByteBuffer rangeEnd = ByteBuffer.allocate(bytes.length + maxSuffix.length);
35           final int firstTimestampByte = maxSuffix[0] & 0xFF;
36   
37           // if firstTimestampByte is 0, we'll put all key bytes into range result because `(bytes[i] & 0xFF) >= firstTimestampByte`
38           // will always be true (this is a byte to unsigned int conversion comparison)
39           if (firstTimestampByte == 0) {
40               return Bytes.wrap(
41                   rangeEnd
42                       .put(bytes)
43                       .put(maxSuffix)
44                       .array()
45               );
46           } else {
47               int i = 0;
48               while (i < bytes.length && (
49                   i < MIN_KEY_LENGTH // assumes keys are at least one byte long
50                   || (bytes[i] & 0xFF) >= firstTimestampByte
51                   )) {
52                   rangeEnd.put(bytes[i++]);
53               }
54   
55               rangeEnd.put(maxSuffix);
56               rangeEnd.flip();
57   
58               final byte[] res = new byte[rangeEnd.remaining()];
59               ByteBuffer.wrap(res).put(rangeEnd);
60               return Bytes.wrap(res);
61           }
62       }
63   
64       static Bytes lowerRange(final Bytes key, final byte[] minSuffix) {
65           final byte[] bytes = key.get();
66           final ByteBuffer rangeStart = ByteBuffer.allocate(bytes.length + minSuffix.length);
67           // any key in the range would start at least with the given prefix to be
68           // in the range, and have at least SUFFIX_SIZE number of trailing zero bytes.
69   
70           // unless there is a maximum key length, you can keep appending more zero bytes
71           // to keyFrom to create a key that will match the range, yet that would precede
72           // KeySchema.toBinaryKey(keyFrom, from, 0) in byte order
73           return Bytes.wrap(
74               rangeStart
75                   .put(bytes)
76                   .put(minSuffix)
77                   .array()
78           );
79       }
80   }
