37   /**
38    * PartitionGroup is used to buffer all co-partitioned records for processing.
39    *
40    * In other words, it represents the "same" partition over multiple co-partitioned topics, and it is used
41    * to buffer records from that partition in each of the contained topic-partitions.
42    * Each StreamTask has exactly one PartitionGroup.
43    *
44    * PartitionGroup implements the algorithm that determines in what order buffered records are selected for processing.
45    *
46    * Specifically, when polled, it returns the record from the topic-partition with the lowest stream-time.
47    * Stream-time for a topic-partition is defined as the highest timestamp
48    * yet observed at the head of that topic-partition.
49    *
50    * PartitionGroup also maintains a stream-time for the group as a whole.
51    * This is defined as the highest timestamp of any record yet polled from the PartitionGroup.
52    * Note however that any computation that depends on stream-time should track it on a per-operator basis to obtain an
53    * accurate view of the local time as seen by that processor.
54    *
55    * The PartitionGroups's stream-time is initially UNKNOWN (-1), and it set to a known value upon first poll.
56    * As a consequence of the definition, the PartitionGroup's stream-time is non-decreasing
57    * (i.e., it increases or stays the same over time).
58    */
59   public class PartitionGroup {

112       public boolean readyToProcess(final long wallClockTime) {
113           if (maxTaskIdleMs == StreamsConfig.MAX_TASK_IDLE_MS_DISABLED) {
114               if (logger.isTraceEnabled() && !allBuffered && totalBuffered > 0) {
115                   final Set<TopicPartition> bufferedPartitions = new HashSet<>();
116                   final Set<TopicPartition> emptyPartitions = new HashSet<>();
117                   for (final Map.Entry<TopicPartition, RecordQueue> entry : partitionQueues.entrySet()) {
118                       if (entry.getValue().isEmpty()) {
119                           emptyPartitions.add(entry.getKey());
120                       } else {
121                           bufferedPartitions.add(entry.getKey());
122                       }
123                   }
124                   logger.trace("Ready for processing because max.task.idle.ms is disabled." +
125                                   "\n\tThere may be out-of-order processing for this task as a result." +
126                                   "\n\tBuffered partitions: {}" +
127                                   "\n\tNon-buffered partitions: {}",
128                           bufferedPartitions,
129                           emptyPartitions);
130               }
131               return true;
132           }
133   
134           final Set<TopicPartition> queued = new HashSet<>();
135           Map<TopicPartition, Long> enforced = null;
136   
137           for (final Map.Entry<TopicPartition, RecordQueue> entry : partitionQueues.entrySet()) {
138               final TopicPartition partition = entry.getKey();
139               final RecordQueue queue = entry.getValue();
140   
141   
142               if (!queue.isEmpty()) {
143                   // this partition is ready for processing
144                   idlePartitionDeadlines.remove(partition);
145                   queued.add(partition);
146               } else {
147                   final OptionalLong fetchedLag = lagProvider.apply(partition);
148   
149                   if (!fetchedLag.isPresent()) {
150                       // must wait to fetch metadata for the partition
151                       idlePartitionDeadlines.remove(partition);
152                       logger.trace("Waiting to fetch data for {}", partition);
153                       return false;
154                   } else if (fetchedLag.getAsLong() > 0L) {
155                       // must wait to poll the data we know to be on the broker
156                       idlePartitionDeadlines.remove(partition);
157                       logger.trace(
158                               "Lag for {} is currently {}, but no data is buffered locally. Waiting to buffer some records.",
159                               partition,
160                               fetchedLag.getAsLong()
161                       );
162                       return false;
163                   } else {
164                       // p is known to have zero lag. wait for maxTaskIdleMs to see if more data shows up.
165                       // One alternative would be to set the deadline to nullableMetadata.receivedTimestamp + maxTaskIdleMs
166                       // instead. That way, we would start the idling timer as of the freshness of our knowledge about the zero
167                       // lag instead of when we happen to run this method, but realistically it's probably a small difference
168                       // and using wall clock time seems more intuitive for users,
169                       // since the log message will be as of wallClockTime.
170                       idlePartitionDeadlines.putIfAbsent(partition, wallClockTime + maxTaskIdleMs);
171                       final long deadline = idlePartitionDeadlines.get(partition);
172                       if (wallClockTime < deadline) {
173                           logger.trace(
174                                   "Lag for {} is currently 0 and current time is {}. Waiting for new data to be produced for configured idle time {} (deadline is {}).",
175                                   partition,
176                                   wallClockTime,
177                                   maxTaskIdleMs,
178                                   deadline
179                           );
180                           return false;
181                       } else {
182                           // this partition is ready for processing due to the task idling deadline passing
183                           if (enforced == null) {
184                               enforced = new HashMap<>();
185                           }
186                           enforced.put(partition, deadline);
187                       }
188                   }
189               }
190           }
191           if (enforced == null) {
192               logger.trace("All partitions were buffered locally, so this task is ready for processing.");
193               return true;
194           } else if (queued.isEmpty()) {
195               logger.trace("No partitions were buffered locally, so this task is not ready for processing.");
196               return false;
197           } else {
198               enforcedProcessingSensor.record(1.0d, wallClockTime);
199               logger.trace("Continuing to process although some partitions are empty on the broker." +
200                               "\n\tThere may be out-of-order processing for this task as a result." +
201                               "\n\tPartitions with local data: {}." +
202                               "\n\tPartitions we gave up waiting for, with their corresponding deadlines: {}." +
203                               "\n\tConfigured max.task.idle.ms: {}." +
204                               "\n\tCurrent wall-clock time: {}.",
205                       queued,
206                       enforced,
207                       maxTaskIdleMs,
208                       wallClockTime);
209               return true;
210           }
211       }

222       // creates queues for new partitions, removes old queues, saves cached records for previously assigned partitions
223       void updatePartitions(final Set<TopicPartition> newInputPartitions, final Function<TopicPartition, RecordQueue> recordQueueCreator) {
224           final Set<TopicPartition> removedPartitions = new HashSet<>();
225           final Iterator<Map.Entry<TopicPartition, RecordQueue>> queuesIterator = partitionQueues.entrySet().iterator();
226           while (queuesIterator.hasNext()) {
227               final Map.Entry<TopicPartition, RecordQueue> queueEntry = queuesIterator.next();
228               final TopicPartition topicPartition = queueEntry.getKey();
229               if (!newInputPartitions.contains(topicPartition)) {
230                   // if partition is removed should delete its queue
231                   totalBuffered -= queueEntry.getValue().size();
232                   totalBytesBuffered -= queueEntry.getValue().getTotalBytesBuffered();
233                   queuesIterator.remove();
234                   removedPartitions.add(topicPartition);
235               }
236               newInputPartitions.remove(topicPartition);
237           }
238           for (final TopicPartition newInputPartition : newInputPartitions) {
239               partitionQueues.put(newInputPartition, recordQueueCreator.apply(newInputPartition));
240           }
241           nonEmptyQueuesByTime.removeIf(q -> removedPartitions.contains(q.partition()));
242           allBuffered = allBuffered && newInputPartitions.isEmpty();
243       }

245       void setPartitionTime(final TopicPartition partition, final long partitionTime) {
246           final RecordQueue queue = partitionQueues.get(partition);
247           if (queue == null) {
248               throw new IllegalStateException("Partition " + partition + " not found.");
249           }
250           if (streamTime < partitionTime) {
251               streamTime = partitionTime;
252           }
253           queue.setPartitionTime(partitionTime);
254       }

256       /**
257        * Get the next record and queue
258        *
259        * @return StampedRecord
260        */
261       StampedRecord nextRecord(final RecordInfo info, final long wallClockTime) {
262           StampedRecord record = null;
263   
264           final RecordQueue queue = nonEmptyQueuesByTime.poll();
265           info.queue = queue;
266   
267           if (queue != null) {
268               // get the buffer size of queue before poll
269               final long oldBufferSize = queue.getTotalBytesBuffered();
270               // get the first record from this queue.
271               record = queue.poll(wallClockTime);
272               // After polling, the buffer size would have reduced.
273               final long newBufferSize = queue.getTotalBytesBuffered();
274   
275               if (record != null) {
276                   --totalBuffered;
277                   totalBytesBuffered -= oldBufferSize - newBufferSize;
278                   totalBytesSensor.record(totalBytesBuffered);
279                   if (queue.isEmpty()) {
280                       // if a certain queue has been drained, reset the flag
281                       allBuffered = false;
282                   } else {
283                       nonEmptyQueuesByTime.offer(queue);
284                   }
285   
286                   // always update the stream-time to the record's timestamp yet to be processed if it is larger
287                   if (record.timestamp > streamTime) {
288                       streamTime = record.timestamp;
289                       recordLatenessSensor.record(0, wallClockTime);
290                   } else {
291                       recordLatenessSensor.record(streamTime - record.timestamp, wallClockTime);
292                   }
293               }
294           }
295   
296           return record;
297       }

299       /**
300        * Adds raw records to this partition group
301        *
302        * @param partition the partition
303        * @param rawRecords  the raw records
304        * @return the queue size for the partition
305        */
306       int addRawRecords(final TopicPartition partition, final Iterable<ConsumerRecord<byte[], byte[]>> rawRecords) {
307           final RecordQueue recordQueue = partitionQueues.get(partition);
308   
309           if (recordQueue == null) {
310               throw new IllegalStateException("Partition " + partition + " not found.");
311           }
312   
313           final int oldSize = recordQueue.size();
314           final long oldBufferSize = recordQueue.getTotalBytesBuffered();
315           final int newSize = recordQueue.addRawRecords(rawRecords);
316           final long newBufferSize = recordQueue.getTotalBytesBuffered();
317   
318           // add this record queue to be considered for processing in the future if it was empty before
319           if (oldSize == 0 && newSize > 0) {
320               nonEmptyQueuesByTime.offer(recordQueue);
321   
322               // if all partitions now are non-empty, set the flag
323               // we do not need to update the stream-time here since this task will definitely be
324               // processed next, and hence the stream-time will be updated when we retrieved records by then
325               if (nonEmptyQueuesByTime.size() == this.partitionQueues.size()) {
326                   allBuffered = true;
327               }
328           }
329   
330           totalBuffered += newSize - oldSize;
331           totalBytesBuffered += newBufferSize - oldBufferSize;
332           totalBytesSensor.record(totalBytesBuffered);
333           return newSize;
334       }
