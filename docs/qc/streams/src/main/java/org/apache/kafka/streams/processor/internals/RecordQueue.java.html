20   import org.apache.kafka.common.TopicPartition;
21   import org.apache.kafka.common.metrics.Sensor;
22   import org.apache.kafka.common.utils.LogContext;
23   import org.apache.kafka.streams.errors.DeserializationExceptionHandler;
24   import org.apache.kafka.streams.errors.StreamsException;
25   import org.apache.kafka.streams.processor.internals.metrics.TaskMetrics;
26   import org.apache.kafka.streams.processor.api.ProcessorContext;
27   import org.apache.kafka.streams.processor.TimestampExtractor;
28   import org.apache.kafka.streams.processor.internals.metrics.TopicMetrics;

30   import org.slf4j.Logger;

32   import java.util.ArrayDeque;
33   
34   import static org.apache.kafka.streams.processor.internals.ClientUtils.consumerRecordSizeInBytes;

36   /**
37    * RecordQueue is a FIFO queue of {@link StampedRecord} (ConsumerRecord + timestamp). It also keeps track of the
38    * partition timestamp defined as the largest timestamp seen on the partition so far; this is passed to the
39    * timestamp extractor.
40    */
41   public class RecordQueue {
42   
43       public static final long UNKNOWN = ConsumerRecord.NO_TIMESTAMP;
44   
45       private final Logger log;
46       private final SourceNode<?, ?> source;
47       private final TopicPartition partition;
48       private final ProcessorContext<?, ?> processorContext;
49       private final TimestampExtractor timestampExtractor;
50       private final RecordDeserializer recordDeserializer;
51       private final ArrayDeque<ConsumerRecord<byte[], byte[]>> fifoQueue;
52   
53       private StampedRecord headRecord = null;
54       private long partitionTime = UNKNOWN;
55   
56       private final Sensor droppedRecordsSensor;
57       private final Sensor consumedSensor;
58       private long headRecordSizeInBytes;
59   
60       RecordQueue(final TopicPartition partition,
61                   final SourceNode<?, ?> source,
62                   final TimestampExtractor timestampExtractor,
63                   final DeserializationExceptionHandler deserializationExceptionHandler,
64                   final InternalProcessorContext<?, ?> processorContext,
65                   final LogContext logContext) {
66           this.source = source;
67           this.partition = partition;
68           this.fifoQueue = new ArrayDeque<>();
69           this.timestampExtractor = timestampExtractor;
70           this.processorContext = processorContext;
71   
72           final String threadName = Thread.currentThread().getName();
73           droppedRecordsSensor = TaskMetrics.droppedRecordsSensor(
74               threadName,
75               processorContext.taskId().toString(),
76               processorContext.metrics()
77           );
78           consumedSensor = TopicMetrics.consumedSensor(
79               threadName,
80               processorContext.taskId().toString(),
81               source.name(),
82               partition.topic(),
83               processorContext.metrics()
84           );
85           recordDeserializer = new RecordDeserializer(
86               source,
87               deserializationExceptionHandler,
88               logContext,
89               droppedRecordsSensor
90           );
91           this.log = logContext.logger(RecordQueue.class);
92           this.headRecordSizeInBytes = 0L;
93       }
94   
95       void setPartitionTime(final long partitionTime) {
96           this.partitionTime = partitionTime;
97       }
98   
99       /**
100        * Returns the corresponding source node in the topology
101        *
102        * @return SourceNode
103        */
104       public SourceNode<?, ?> source() {
105           return source;
106       }
107   
108       /**
109        * Returns the partition with which this queue is associated
110        *
111        * @return TopicPartition
112        */
113       public TopicPartition partition() {
114           return partition;
115       }
116   
117       /**
118        * Add a batch of {@link ConsumerRecord} into the queue
119        *
120        * @param rawRecords the raw records
121        * @return the size of this queue
122        */
123       int addRawRecords(final Iterable<ConsumerRecord<byte[], byte[]>> rawRecords) {
124           for (final ConsumerRecord<byte[], byte[]> rawRecord : rawRecords) {
125               fifoQueue.addLast(rawRecord);
126           }
127   
128           updateHead();
129   
130           return size();
131       }
132   
133       /**
134        * Get the next {@link StampedRecord} from the queue
135        *
136        * @return StampedRecord
137        */
138       public StampedRecord poll(final long wallClockTime) {
139           final StampedRecord recordToReturn = headRecord;
140   
141           consumedSensor.record(headRecordSizeInBytes, wallClockTime);
142   
143           headRecord = null;
144           headRecordSizeInBytes = 0L;
145           partitionTime = Math.max(partitionTime, recordToReturn.timestamp);
146   
147           updateHead();
148   
149           return recordToReturn;
150       }
151   
152       /**
153        * Returns the number of records in the queue
154        *
155        * @return the number of records
156        */
157       public int size() {
158           // plus one deserialized head record for timestamp tracking
159           return fifoQueue.size() + (headRecord == null ? 0 : 1);
160       }
161   
162       /**
163        * Tests if the queue is empty
164        *
165        * @return true if the queue is empty, otherwise false
166        */
167       public boolean isEmpty() {
168           return fifoQueue.isEmpty() && headRecord == null;
169       }
170   
171       /**
172        * Returns the head record's timestamp
173        *
174        * @return timestamp
175        */
176       public long headRecordTimestamp() {
177           return headRecord == null ? UNKNOWN : headRecord.timestamp;
178       }
179   
180       public Long headRecordOffset() {
181           return headRecord == null ? null : headRecord.offset();
182       }
183   
184       /**
185        * Clear the fifo queue of its elements
186        */
187       public void clear() {
188           fifoQueue.clear();
189           headRecord = null;
190           headRecordSizeInBytes = 0L;
191           partitionTime = UNKNOWN;
192       }
193   
194       public void close() {
195           processorContext.metrics().removeSensor(consumedSensor);
196       }
197   
198       private void updateHead() {
199           ConsumerRecord<byte[], byte[]> lastCorruptedRecord = null;
200   
201           while (headRecord == null && !fifoQueue.isEmpty()) {
202               final ConsumerRecord<byte[], byte[]> raw = fifoQueue.pollFirst();
203               final ConsumerRecord<Object, Object> deserialized =
204                   recordDeserializer.deserialize(processorContext, raw);
205   
206               if (deserialized == null) {
207                   // this only happens if the deserializer decides to skip. It has already logged the reason.
208                   lastCorruptedRecord = raw;
209                   continue;
210               }
211   
212               final long timestamp;
213               try {
214                   timestamp = timestampExtractor.extract(deserialized, partitionTime);
215               } catch (final StreamsException internalFatalExtractorException) {
216                   throw internalFatalExtractorException;
217               } catch (final Exception fatalUserException) {
218                   throw new StreamsException(
219                           String.format("Fatal user code error in TimestampExtractor callback for record %s.", deserialized),
220                           fatalUserException);
221               }
222               log.trace("Source node {} extracted timestamp {} for record {}", source.name(), timestamp, deserialized);
223   
224               // drop message if TS is invalid, i.e., negative
225               if (timestamp < 0) {
226                   log.warn(
227                           "Skipping record due to negative extracted timestamp. topic=[{}] partition=[{}] offset=[{}] extractedTimestamp=[{}] extractor=[{}]",
228                           deserialized.topic(), deserialized.partition(), deserialized.offset(), timestamp, timestampExtractor.getClass().getCanonicalName()
229                   );
230                   droppedRecordsSensor.record();
231                   continue;
232               }
233               headRecord = new StampedRecord(deserialized, timestamp);
234               headRecordSizeInBytes = consumerRecordSizeInBytes(raw);
235           }
236   
237           // if all records in the FIFO queue are corrupted, make the last one the headRecord
238           // This record is used to update the offsets. See KAFKA-6502 for more details.
239           if (headRecord == null && lastCorruptedRecord != null) {
240               headRecord = new CorruptedRecord(lastCorruptedRecord);
241           }
242       }
243   
244       /**
245        * @return the local partitionTime for this particular RecordQueue
246        */
247       long partitionTime() {
248           return partitionTime;
249       }
250   }
