17   package org.apache.kafka.streams.kstream.internals.foreignkeyjoin;
18   
19   import org.apache.kafka.common.serialization.Deserializer;
20   import org.apache.kafka.common.serialization.Serde;
21   import org.apache.kafka.common.serialization.Serializer;
22   import org.apache.kafka.common.utils.Bytes;
23   import org.apache.kafka.streams.processor.api.ProcessorContext;
24   
25   import java.nio.ByteBuffer;
26   import java.util.function.Supplier;
27   
28   /**
29    * Factory for creating CombinedKey serializers / deserializers.
30    */
31   public class CombinedKeySchema<KO, K> {
32       private final Supplier<String> undecoratedPrimaryKeySerdeTopicSupplier;
33       private final Supplier<String> undecoratedForeignKeySerdeTopicSupplier;
34       private String primaryKeySerdeTopic;
35       private String foreignKeySerdeTopic;
36       private Serializer<K> primaryKeySerializer;
37       private Deserializer<K> primaryKeyDeserializer;
38       private Serializer<KO> foreignKeySerializer;
39       private Deserializer<KO> foreignKeyDeserializer;
40   
41       public CombinedKeySchema(final Supplier<String> foreignKeySerdeTopicSupplier,
42                                final Serde<KO> foreignKeySerde,
43                                final Supplier<String> primaryKeySerdeTopicSupplier,
44                                final Serde<K> primaryKeySerde) {
45           undecoratedPrimaryKeySerdeTopicSupplier = primaryKeySerdeTopicSupplier;
46           undecoratedForeignKeySerdeTopicSupplier = foreignKeySerdeTopicSupplier;
47           primaryKeySerializer = primaryKeySerde == null ? null : primaryKeySerde.serializer();
48           primaryKeyDeserializer = primaryKeySerde == null ? null : primaryKeySerde.deserializer();
49           foreignKeyDeserializer = foreignKeySerde == null ? null : foreignKeySerde.deserializer();
50           foreignKeySerializer = foreignKeySerde == null ? null : foreignKeySerde.serializer();
51       }
52   
53       @SuppressWarnings("unchecked")
54       public void init(final ProcessorContext<?, ?> context) {
55           primaryKeySerdeTopic = undecoratedPrimaryKeySerdeTopicSupplier.get();
56           foreignKeySerdeTopic = undecoratedForeignKeySerdeTopicSupplier.get();
57           primaryKeySerializer = primaryKeySerializer == null ? (Serializer<K>) context.keySerde().serializer() : primaryKeySerializer;
58           primaryKeyDeserializer = primaryKeyDeserializer == null ? (Deserializer<K>) context.keySerde().deserializer() : primaryKeyDeserializer;
59           foreignKeySerializer = foreignKeySerializer == null ? (Serializer<KO>) context.keySerde().serializer() : foreignKeySerializer;
60           foreignKeyDeserializer = foreignKeyDeserializer == null ? (Deserializer<KO>) context.keySerde().deserializer() : foreignKeyDeserializer;
61       }
62   
63       Bytes toBytes(final KO foreignKey, final K primaryKey) {
64           //The serialization format - note that primaryKeySerialized may be null, such as when a prefixScan
65           //key is being created.
66           //{Integer.BYTES foreignKeyLength}{foreignKeySerialized}{Optional-primaryKeySerialized}
67           final byte[] foreignKeySerializedData = foreignKeySerializer.serialize(foreignKeySerdeTopic,
68                                                                                  foreignKey);
69   
70           //? bytes
71           final byte[] primaryKeySerializedData = primaryKeySerializer.serialize(primaryKeySerdeTopic,
72                                                                                  primaryKey);
73   
74           final ByteBuffer buf = ByteBuffer.allocate(Integer.BYTES + foreignKeySerializedData.length + primaryKeySerializedData.length);
75           buf.putInt(foreignKeySerializedData.length);
76           buf.put(foreignKeySerializedData);
77           buf.put(primaryKeySerializedData);
78           return Bytes.wrap(buf.array());
79       }
80   
81   
82       public CombinedKey<KO, K> fromBytes(final Bytes data) {
83           //{Integer.BYTES foreignKeyLength}{foreignKeySerialized}{Optional-primaryKeySerialized}
84           final byte[] dataArray = data.get();
85           final ByteBuffer dataBuffer = ByteBuffer.wrap(dataArray);
86           final int foreignKeyLength = dataBuffer.getInt();
87           final byte[] foreignKeyRaw = new byte[foreignKeyLength];
88           dataBuffer.get(foreignKeyRaw, 0, foreignKeyLength);
89           final KO foreignKey = foreignKeyDeserializer.deserialize(foreignKeySerdeTopic, foreignKeyRaw);
90   
91           final byte[] primaryKeyRaw = new byte[dataArray.length - foreignKeyLength - Integer.BYTES];
92           dataBuffer.get(primaryKeyRaw, 0, primaryKeyRaw.length);
93           final K primaryKey = primaryKeyDeserializer.deserialize(primaryKeySerdeTopic, primaryKeyRaw);
94           return new CombinedKey<>(foreignKey, primaryKey);
95       }
96   
97       Bytes prefixBytes(final KO key) {
98           //The serialization format. Note that primaryKeySerialized is not required/used in this function.
99           //{Integer.BYTES foreignKeyLength}{foreignKeySerialized}{Optional-primaryKeySerialized}
100   
101           final byte[] foreignKeySerializedData = foreignKeySerializer.serialize(foreignKeySerdeTopic, key);
102   
103           final ByteBuffer buf = ByteBuffer.allocate(Integer.BYTES + foreignKeySerializedData.length);
104           buf.putInt(foreignKeySerializedData.length);
105           buf.put(foreignKeySerializedData);
106           return Bytes.wrap(buf.array());
107       }
