79       private class KStreamSlidingWindowAggregateProcessor extends AbstractKStreamTimeWindowAggregateProcessor<KIn, VIn, VAgg> {
80           private Boolean reverseIteratorPossible = null;
81   
82           protected KStreamSlidingWindowAggregateProcessor(final String storeName,
83                                                            final EmitStrategy emitStrategy,
84                                                            final boolean sendOldValues) {
85               super(storeName, emitStrategy, sendOldValues);
86           }

88           @Override
89           public void process(final Record<KIn, VIn> record) {
90               if (record.key() == null || record.value() == null) {
91                   if (context().recordMetadata().isPresent()) {
92                       final RecordMetadata recordMetadata = context().recordMetadata().get();
93                       log.warn(
94                           "Skipping record due to null key or value. "
95                               + "topic=[{}] partition=[{}] offset=[{}]",
96                           recordMetadata.topic(), recordMetadata.partition(), recordMetadata.offset()
97                       );
98                   } else {
99                       log.warn(
100                           "Skipping record due to null key or value. Topic, partition, and offset not known."
101                       );
102                   }
103                   droppedRecordsSensor.record();
104                   return;
105               }
106   
107               updateObservedStreamTime(record.timestamp());
108               final long windowCloseTime = observedStreamTime - windows.gracePeriodMs();
109   
110               final long windowStart = record.timestamp();
111               final long windowEnd = record.timestamp() + windows.timeDifferenceMs();
112               if (windowEnd < windowCloseTime) {
113                   final String window = "[" + windowStart + "," + windowEnd + "]";
114                   logSkippedRecordForExpiredWindow(log, record.timestamp(), windowCloseTime, window);
115                   return;
116               }
117   
118               if (record.timestamp() < windows.timeDifferenceMs()) {
119                   processEarly(record, windowCloseTime);
120                   return;
121               }
122   
123               if (reverseIteratorPossible == null) {
124                   try {
125                       windowStore.backwardFetch(record.key(), 0L, 0L);
126                       reverseIteratorPossible = true;
127                       log.debug("Sliding Windows aggregate using a reverse iterator");
128                   } catch (final UnsupportedOperationException e)  {
129                       reverseIteratorPossible = false;
130                       log.debug("Sliding Windows aggregate using a forward iterator");
131                   }
132               }
133   
134               if (reverseIteratorPossible) {
135                   processReverse(record, windowCloseTime);
136               } else {
137                   processInOrder(record, windowCloseTime);
138               }
139   
140               maybeForwardFinalResult(record, windowCloseTime);
141           }

143           public void processInOrder(final Record<KIn, VIn> record, final long windowCloseTime) {
144               final Set<Long> windowStartTimes = new HashSet<>();
145   
146               // aggregate that will go in the current record’s left/right window (if needed)
147               ValueAndTimestamp<VAgg> leftWinAgg = null;
148               ValueAndTimestamp<VAgg> rightWinAgg = null;
149   
150               //if current record's left/right windows already exist
151               boolean leftWinAlreadyCreated = false;
152               boolean rightWinAlreadyCreated = false;
153   
154               Long previousRecordTimestamp = null;
155   
156               try (
157                   final KeyValueIterator<Windowed<KIn>, ValueAndTimestamp<VAgg>> iterator = windowStore.fetch(
158                       record.key(),
159                       record.key(),
160                       Math.max(0, record.timestamp() - 2 * windows.timeDifferenceMs()),
161                       // add 1 to upper bound to catch the current record's right window, if it exists, without more calls to the store
162                       record.timestamp() + 1)
163               ) {
164                   while (iterator.hasNext()) {
165                       final KeyValue<Windowed<KIn>, ValueAndTimestamp<VAgg>> windowBeingProcessed = iterator.next();
166                       final long startTime = windowBeingProcessed.key.window().start();
167                       windowStartTimes.add(startTime);
168                       final long endTime = startTime + windows.timeDifferenceMs();
169                       final long windowMaxRecordTimestamp = windowBeingProcessed.value.timestamp();
170   
171                       if (endTime < record.timestamp()) {
172                           leftWinAgg = windowBeingProcessed.value;
173                           previousRecordTimestamp = windowMaxRecordTimestamp;
174                       } else if (endTime == record.timestamp()) {
175                           leftWinAlreadyCreated = true;
176                           if (windowMaxRecordTimestamp < record.timestamp()) {
177                               previousRecordTimestamp = windowMaxRecordTimestamp;
178                           }
179                           updateWindowAndForward(
180                               windowBeingProcessed.key.window(),
181                               windowBeingProcessed.value,
182                               record,
183                               windowCloseTime);
184                       } else if (endTime > record.timestamp() && startTime <= record.timestamp()) {
185                           rightWinAgg = windowBeingProcessed.value;
186                           updateWindowAndForward(
187                               windowBeingProcessed.key.window(),
188                               windowBeingProcessed.value,
189                               record,
190                               windowCloseTime);
191                       } else if (startTime == record.timestamp() + 1) {
192                           rightWinAlreadyCreated = true;
193                       } else {
194                           log.error(
195                               "Unexpected window with start {} found when processing record at {} in `KStreamSlidingWindowAggregate`.",
196                               startTime, record.timestamp()
197                           );
198                           throw new IllegalStateException("Unexpected window found when processing sliding windows");
199                       }
200                   }
201               }
202               createWindows(record, windowCloseTime, windowStartTimes, rightWinAgg, leftWinAgg, leftWinAlreadyCreated, rightWinAlreadyCreated, previousRecordTimestamp);
203           }

205           public void processReverse(final Record<KIn, VIn> record, final long windowCloseTime) {
206               final Set<Long> windowStartTimes = new HashSet<>();
207   
208               // aggregate that will go in the current record’s left/right window (if needed)
209               ValueAndTimestamp<VAgg> leftWinAgg = null;
210               ValueAndTimestamp<VAgg> rightWinAgg = null;
211   
212               //if current record's left/right windows already exist
213               boolean leftWinAlreadyCreated = false;
214               boolean rightWinAlreadyCreated = false;
215   
216               Long previousRecordTimestamp = null;
217   
218               try (
219                   final KeyValueIterator<Windowed<KIn>, ValueAndTimestamp<VAgg>> iterator = windowStore.backwardFetch(
220                       record.key(),
221                       record.key(),
222                       Math.max(0, record.timestamp() - 2 * windows.timeDifferenceMs()),
223                       // add 1 to upper bound to catch the current record's right window, if it exists, without more calls to the store
224                       record.timestamp() + 1)
225               ) {
226                   while (iterator.hasNext()) {
227                       final KeyValue<Windowed<KIn>, ValueAndTimestamp<VAgg>> windowBeingProcessed = iterator.next();
228                       final long startTime = windowBeingProcessed.key.window().start();
229                       windowStartTimes.add(startTime);
230                       final long endTime = startTime + windows.timeDifferenceMs();
231                       final long windowMaxRecordTimestamp = windowBeingProcessed.value.timestamp();
232                       if (startTime == record.timestamp() + 1) {
233                           rightWinAlreadyCreated = true;
234                       } else if (endTime > record.timestamp()) {
235                           if (rightWinAgg == null) {
236                               rightWinAgg = windowBeingProcessed.value;
237                           }
238                           updateWindowAndForward(windowBeingProcessed.key.window(), windowBeingProcessed.value, record, windowCloseTime);
239                       } else if (endTime == record.timestamp()) {
240                           leftWinAlreadyCreated = true;
241                           updateWindowAndForward(windowBeingProcessed.key.window(), windowBeingProcessed.value, record, windowCloseTime);
242                           if (windowMaxRecordTimestamp < record.timestamp()) {
243                               previousRecordTimestamp = windowMaxRecordTimestamp;
244                           } else {
245                               return;
246                           }
247                       } else if (endTime < record.timestamp()) {
248                           leftWinAgg = windowBeingProcessed.value;
249                           previousRecordTimestamp = windowMaxRecordTimestamp;
250                           break;
251                       } else {
252                           log.error(
253                               "Unexpected window with start {} found when processing record at {} in `KStreamSlidingWindowAggregate`.",
254                               startTime, record.timestamp()
255                           );
256                           throw new IllegalStateException("Unexpected window found when processing sliding windows");
257                       }
258                   }
259               }
260               createWindows(record, windowCloseTime, windowStartTimes, rightWinAgg, leftWinAgg, leftWinAlreadyCreated, rightWinAlreadyCreated, previousRecordTimestamp);
261           }
262   
263           /**
264            * Created to handle records where 0 < inputRecordTimestamp < timeDifferenceMs. These records would create
265            * windows with negative start times, which is not supported. Instead, we will put them into the [0, timeDifferenceMs]
266            * window as a "workaround", and we will update or create their right windows as new records come in later
267            */
268           private void processEarly(final Record<KIn, VIn> record, final long windowCloseTime) {
269               if (record.timestamp() < 0 || record.timestamp() >= windows.timeDifferenceMs()) {
270                   log.error(
271                       "Early record for sliding windows must fall between fall between 0 <= inputRecordTimestamp. Timestamp {} does not fall between 0 <= {}",
272                       record.timestamp(), windows.timeDifferenceMs()
273                   );
274                   throw new IllegalArgumentException("Early record for sliding windows must fall between fall between 0 <= inputRecordTimestamp");
275               }
276   
277               // A window from [0, timeDifferenceMs] that holds all early records
278               KeyValue<Windowed<KIn>, ValueAndTimestamp<VAgg>> combinedWindow = null;
279               ValueAndTimestamp<VAgg> rightWinAgg = null;
280               boolean rightWinAlreadyCreated = false;
281               final Set<Long> windowStartTimes = new HashSet<>();
282   
283               Long previousRecordTimestamp = null;
284   
285               try (
286                   final KeyValueIterator<Windowed<KIn>, ValueAndTimestamp<VAgg>> iterator = windowStore.fetch(
287                       record.key(),
288                       record.key(),
289                       0,
290                       // add 1 to upper bound to catch the current record's right window, if it exists, without more calls to the store
291                       record.timestamp() + 1)
292               ) {
293                   while (iterator.hasNext()) {
294                       final KeyValue<Windowed<KIn>, ValueAndTimestamp<VAgg>> windowBeingProcessed = iterator.next();
295                       final long startTime = windowBeingProcessed.key.window().start();
296                       windowStartTimes.add(startTime);
297                       final long windowMaxRecordTimestamp = windowBeingProcessed.value.timestamp();
298   
299                       if (startTime == 0) {
300                           combinedWindow = windowBeingProcessed;
301                           // We don't need to store previousRecordTimestamp if maxRecordTimestamp >= timestamp
302                           // because the previous record's right window (if there is a previous record)
303                           // would have already been created by maxRecordTimestamp
304                           if (windowMaxRecordTimestamp < record.timestamp()) {
305                               previousRecordTimestamp = windowMaxRecordTimestamp;
306                           }
307   
308                       } else if (startTime <= record.timestamp()) {
309                           rightWinAgg = windowBeingProcessed.value;
310                           updateWindowAndForward(windowBeingProcessed.key.window(), windowBeingProcessed.value, record, windowCloseTime);
311                       } else if (startTime == record.timestamp() + 1) {
312                           rightWinAlreadyCreated = true;
313                       } else {
314                           log.error(
315                               "Unexpected window with start {} found when processing record at {} in `KStreamSlidingWindowAggregate`.",
316                               startTime, record.timestamp()
317                           );
318                           throw new IllegalStateException("Unexpected window found when processing sliding windows");
319                       }
320                   }
321               }
322   
323               // If there wasn't a right window agg found and we need a right window for our new record,
324               // the current aggregate in the combined window will go in the new record's right window. We can be sure that the combined
325               // window only holds records that fall into the current record's right window for two reasons:
326               // 1. If there were records earlier than the current record AND later than the current record, there would be a right window found
327               // when we looked for right window agg.
328               // 2. If there was only a record before the current record, we wouldn't need a right window for the current record and wouldn't update the
329               // rightWinAgg value here, as the combinedWindow.value.timestamp() < inputRecordTimestamp
330               if (rightWinAgg == null && combinedWindow != null && combinedWindow.value.timestamp() > record.timestamp()) {
331                   rightWinAgg = combinedWindow.value;
332               }
333   
334               if (!rightWinAlreadyCreated && rightWindowIsNotEmpty(rightWinAgg, record.timestamp())) {
335                   createCurrentRecordRightWindow(record.timestamp(), rightWinAgg, record);
336               }
337   
338               //create the right window for the previous record if the previous record exists and the window hasn't already been created
339               if (previousRecordTimestamp != null && !windowStartTimes.contains(previousRecordTimestamp + 1)) {
340                   createPreviousRecordRightWindow(previousRecordTimestamp + 1, record, windowCloseTime);
341               }
342   
343               if (combinedWindow == null) {
344                   final TimeWindow window = new TimeWindow(0, windows.timeDifferenceMs());
345                   final ValueAndTimestamp<VAgg> valueAndTime = ValueAndTimestamp.make(initializer.apply(), record.timestamp());
346                   updateWindowAndForward(window, valueAndTime, record, windowCloseTime);
347   
348               } else {
349                   //update the combined window with the new aggregate
350                   updateWindowAndForward(combinedWindow.key.window(), combinedWindow.value, record, windowCloseTime);
351               }
352   
353           }

355           private void createWindows(final Record<KIn, VIn> record,
356                                      final long closeTime,
357                                      final Set<Long> windowStartTimes,
358                                      final ValueAndTimestamp<VAgg> rightWinAgg,
359                                      final ValueAndTimestamp<VAgg> leftWinAgg,
360                                      final boolean leftWinAlreadyCreated,
361                                      final boolean rightWinAlreadyCreated,
362                                      final Long previousRecordTimestamp) {
363               // create right window for previous record
364               if (previousRecordTimestamp != null) {
365                   final long previousRightWinStart = previousRecordTimestamp + 1;
366                   if (previousRecordRightWindowDoesNotExistAndIsNotEmpty(windowStartTimes, previousRightWinStart, record.timestamp())) {
367                       createPreviousRecordRightWindow(previousRightWinStart, record, closeTime);
368                   }
369               }
370   
371               // create left window for new record
372               if (!leftWinAlreadyCreated) {
373                   final ValueAndTimestamp<VAgg> valueAndTime;
374                   if (leftWindowNotEmpty(previousRecordTimestamp, record.timestamp())) {
375                       valueAndTime = ValueAndTimestamp.make(leftWinAgg.value(), record.timestamp());
376                   } else {
377                       valueAndTime = ValueAndTimestamp.make(initializer.apply(), record.timestamp());
378                   }
379                   final TimeWindow window = new TimeWindow(record.timestamp() - windows.timeDifferenceMs(), record.timestamp());
380                   updateWindowAndForward(window, valueAndTime, record, closeTime);
381               }
382   
383               // create right window for new record, if necessary
384               if (!rightWinAlreadyCreated && rightWindowIsNotEmpty(rightWinAgg, record.timestamp())) {
385                   createCurrentRecordRightWindow(record.timestamp(), rightWinAgg, record);
386               }
387           }
388   
389           private void createCurrentRecordRightWindow(final long inputRecordTimestamp,
390                                                       final ValueAndTimestamp<VAgg> rightWinAgg,
391                                                       final Record<KIn, VIn> record) {
392               final TimeWindow window = new TimeWindow(inputRecordTimestamp + 1, inputRecordTimestamp + 1 + windows.timeDifferenceMs());
393               windowStore.put(
394                   record.key(),
395                   rightWinAgg,
396                   window.start());
397               maybeForwardUpdate(record, window, null, rightWinAgg.value(), rightWinAgg.timestamp());
398           }
399   
400           private void createPreviousRecordRightWindow(final long windowStart,
401                                                        final Record<KIn, VIn> record,
402                                                        final long closeTime) {
403               final TimeWindow window = new TimeWindow(windowStart, windowStart + windows.timeDifferenceMs());
404               final ValueAndTimestamp<VAgg> valueAndTime = ValueAndTimestamp.make(initializer.apply(), record.timestamp());
405               updateWindowAndForward(window, valueAndTime, record, closeTime);
406           }

408           // checks if the previous record falls into the current records left window; if yes, the left window is not empty, otherwise it is empty
409           private boolean leftWindowNotEmpty(final Long previousRecordTimestamp, final long inputRecordTimestamp) {
410               return previousRecordTimestamp != null && inputRecordTimestamp - windows.timeDifferenceMs() <= previousRecordTimestamp;
411           }
412   
413           // checks if the previous record's right window does not already exist and the current record falls within previous record's right window
414           private boolean previousRecordRightWindowDoesNotExistAndIsNotEmpty(final Set<Long> windowStartTimes,
415                                                                              final long previousRightWindowStart,
416                                                                              final long inputRecordTimestamp) {

443           private void updateWindowAndForward(final Window window,
444                                               final ValueAndTimestamp<VAgg> valueAndTime,
445                                               final Record<KIn, VIn> record,
446                                               final long windowCloseTime) {
447               final long windowStart = window.start();
448               final long windowEnd = window.end();
449   
450               if (windowEnd >= windowCloseTime) {
451                   // get aggregate from existing window
452                   final VAgg oldAgg = getValueOrNull(valueAndTime);
453                   final VAgg newAgg = aggregator.apply(record.key(), record.value(), oldAgg);
454   
455                   final long newTimestamp = oldAgg == null ? record.timestamp() : Math.max(record.timestamp(), valueAndTime.timestamp());
456                   windowStore.put(
457                       record.key(),
458                       ValueAndTimestamp.make(newAgg, newTimestamp),
459                       windowStart);
460                   maybeForwardUpdate(record, window, oldAgg, newAgg, newTimestamp);
461               } else {
462                   final String windowString = "[" + windowStart + "," + windowEnd + "]";
463                   logSkippedRecordForExpiredWindow(log, record.timestamp(), windowCloseTime, windowString);
464               }
465           }
