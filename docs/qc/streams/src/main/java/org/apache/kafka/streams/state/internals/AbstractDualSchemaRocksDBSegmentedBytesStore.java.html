77       public KeyValueIterator<Bytes, byte[]> all() {
78           final List<S> searchSpace = segments.allSegments(true);
79           final Bytes from = baseKeySchema.lowerRange(null, 0);
80           final Bytes to = baseKeySchema.upperRange(null, Long.MAX_VALUE);
81   
82           return new SegmentIterator<>(
83                   searchSpace.iterator(),
84                   baseKeySchema.hasNextCondition(null, null, 0, Long.MAX_VALUE, true),
85                   from,
86                   to,
87                   true);
88       }

94           final Bytes to = baseKeySchema.upperRange(null, Long.MAX_VALUE);
95

99                   from,

104       @Override
105       public void remove(final Bytes rawBaseKey) {
106           final long timestamp = baseKeySchema.segmentTimestamp(rawBaseKey);
107           observedStreamTime = Math.max(observedStreamTime, timestamp);
108           final S segment = segments.getSegmentForTimestamp(timestamp);
109           if (segment == null) {
110               return;
111           }
112           segment.delete(rawBaseKey);
113   
114           if (hasIndex()) {
115               final KeyValue<Bytes, byte[]> kv = getIndexKeyValue(rawBaseKey, null);
116               segment.delete(kv.key);
117           }
118       }
119   
120       abstract protected KeyValue<Bytes, byte[]> getIndexKeyValue(final Bytes baseKey, final byte[] baseValue);
121   
122       // For testing
123       void putIndex(final Bytes indexKey, final byte[] value) {
124           if (!hasIndex()) {
125               throw new IllegalStateException("Index store doesn't exist");
126           }
127   
128           final long timestamp = indexKeySchema.get().segmentTimestamp(indexKey);
129           final long segmentId = segments.segmentId(timestamp);
130           final S segment = segments.getOrCreateSegmentIfLive(segmentId, context, observedStreamTime);
131   
132           if (segment != null) {
133               segment.put(indexKey, value);
134           }
135       }

137       byte[] getIndex(final Bytes indexKey) {
138           if (!hasIndex()) {
139               throw new IllegalStateException("Index store doesn't exist");
140           }
141   
142           final long timestamp = indexKeySchema.get().segmentTimestamp(indexKey);
143           final long segmentId = segments.segmentId(timestamp);
144           final S segment = segments.getOrCreateSegmentIfLive(segmentId, context, observedStreamTime);
145   
146           if (segment != null) {
147               return segment.get(indexKey);
148           }
149           return null;
150       }

166       @Override
167       public void put(final Bytes rawBaseKey,
168                       final byte[] value) {
169           final long timestamp = baseKeySchema.segmentTimestamp(rawBaseKey);
170           observedStreamTime = Math.max(observedStreamTime, timestamp);
171           final long segmentId = segments.segmentId(timestamp);
172           final S segment = segments.getOrCreateSegmentIfLive(segmentId, context, observedStreamTime);
173   
174           if (segment == null) {
175               expiredRecordSensor.record(1.0d, ProcessorContextUtils.currentSystemTime(context));
176               LOG.warn("Skipping record for expired segment.");
177           } else {
178               StoreQueryUtils.updatePosition(position, stateStoreContext);
179   
180               // Put to index first so that if put to base failed, when we iterate index, we will
181               // find no base value. If put to base first but putting to index fails, when we iterate
182               // index, we can't find the key but if we iterate over base store, we can find the key
183               // which lead to inconsistency.
184               if (hasIndex()) {
185                   final KeyValue<Bytes, byte[]> indexKeyValue = getIndexKeyValue(rawBaseKey, value);
186                   segment.put(indexKeyValue.key, indexKeyValue.value);
187               }
188               segment.put(rawBaseKey, value);
189           }
190       }

192       @Override
193       public byte[] get(final Bytes rawKey) {
194           final S segment = segments.getSegmentForTimestamp(baseKeySchema.segmentTimestamp(rawKey));
195           if (segment == null) {
196               return null;
197           }
198           return segment.get(rawKey);
199       }

223

225           this.positionCheckpoint = new OffsetCheckpoint(positionCheckpointFile);
226           this.position = StoreQueryUtils.readPositionFromCheckpoint(positionCheckpoint);
227

232               () -> StoreQueryUtils.checkpointPosition(positionCheckpoint, position)

236   
237           consistencyEnabled = StreamsConfig.InternalConfig.getBoolean(
238               context.appConfigs(),

240               false
241           );
242       }

276       // Visible for testing
277       void restoreAllInternal(final Collection<ConsumerRecord<byte[], byte[]>> records) {
278           try {
279               final Map<S, WriteBatch> writeBatchMap = getWriteBatches(records);
280               for (final Map.Entry<S, WriteBatch> entry : writeBatchMap.entrySet()) {
281                   final S segment = entry.getKey();
282                   final WriteBatch batch = entry.getValue();
283                   segment.write(batch);
284                   batch.close();
285               }
286           } catch (final RocksDBException e) {
287               throw new ProcessorStateException("Error restoring batch to store " + this.name, e);
288           }
289       }
