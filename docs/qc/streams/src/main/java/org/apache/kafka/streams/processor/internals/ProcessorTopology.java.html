145       public void updateSourceTopics(final Map<String, List<String>> allSourceTopicsByNodeName) {
146           sourceNodesByTopic.clear();
147           for (final Map.Entry<String, SourceNode<?, ?>> sourceNodeEntry : sourceNodesByName.entrySet()) {
148               final String sourceNodeName = sourceNodeEntry.getKey();
149               final SourceNode<?, ?> sourceNode = sourceNodeEntry.getValue();
150   
151               final List<String> updatedSourceTopics = allSourceTopicsByNodeName.get(sourceNodeName);
152               if (updatedSourceTopics == null) {
153                   log.error("Unable to find source node {} in updated topics map {}",
154                             sourceNodeName, allSourceTopicsByNodeName);
155                   throw new IllegalStateException("Node " + sourceNodeName + " not found in full topology");
156               }
157   
158               log.trace("Updating source node {} with new topics {}", sourceNodeName, updatedSourceTopics);
159               for (final String topic : updatedSourceTopics) {
160                   if (sourceNodesByTopic.containsKey(topic)) {
161                       log.error("Tried to subscribe topic {} to two nodes when updating topics from {}",
162                                 topic, allSourceTopicsByNodeName);
163                       throw new IllegalStateException("Topic " + topic + " was already registered to source node "
164                                                           + sourceNodesByTopic.get(topic).name());
165                   }
166                   sourceNodesByTopic.put(topic, sourceNode);
167               }
168           }
169       }

171       private String childrenToString(final String indent, final List<? extends ProcessorNode<?, ?, ?, ?>> children) {
172           if (children == null || children.isEmpty()) {
173               return "";
174           }
175   
176           final StringBuilder sb = new StringBuilder(indent + "\tchildren:\t[");
177           for (final ProcessorNode<?, ?, ?, ?> child : children) {
178               sb.append(child.name());
179               sb.append(", ");
180           }
181           sb.setLength(sb.length() - 2);  // remove the last comma
182           sb.append("]\n");
183   
184           // recursively print children
185           for (final ProcessorNode<?, ?, ?, ?> child : children) {
186               sb.append(child.toString(indent)).append(childrenToString(indent, child.children()));
187           }
188           return sb.toString();
189       }

201       /**
202        * Produces a string representation containing useful information this topology.
203        * This is useful in debugging scenarios.
204        * @return A string representation of this instance.
205        */
206       public String toString(final String indent) {
207           final Map<SourceNode<?, ?>, List<String>> sourceToTopics = new HashMap<>();
208           for (final Map.Entry<String, SourceNode<?, ?>> sourceNodeEntry : sourceNodesByTopic.entrySet()) {
209               final String topic = sourceNodeEntry.getKey();
210               final SourceNode<?, ?> source = sourceNodeEntry.getValue();
211               sourceToTopics.computeIfAbsent(source, s -> new ArrayList<>());
212               sourceToTopics.get(source).add(topic);
213           }
214   
215           final StringBuilder sb = new StringBuilder(indent + "ProcessorTopology:\n");
216   
217           // start from sources
218           for (final Map.Entry<SourceNode<?, ?>, List<String>> sourceNodeEntry : sourceToTopics.entrySet()) {
219               final SourceNode<?, ?> source = sourceNodeEntry.getKey();
220               final List<String> topics = sourceNodeEntry.getValue();
221               sb.append(source.toString(indent + "\t"))
222                   .append(topicsToString(indent + "\t", topics))
223                   .append(childrenToString(indent + "\t", source.children()));
224           }
225           return sb.toString();
226       }

228       private static String topicsToString(final String indent, final List<String> topics) {
229           final StringBuilder sb = new StringBuilder();
230           sb.append(indent).append("\ttopics:\t\t[");
231           for (final String topic : topics) {
232               sb.append(topic);
233               sb.append(", ");
234           }
235           sb.setLength(sb.length() - 2);  // remove the last comma
236           sb.append("]\n");
237           return sb.toString();
238       }
