17   package org.apache.kafka.streams.processor.internals;
18   
19   import java.io.ByteArrayOutputStream;
20   import java.nio.ByteBuffer;
21   import java.nio.charset.StandardCharsets;
22   import java.util.HashMap;
23   import java.util.Map;
24   import java.util.Objects;
25   
26   /**
27    * ProcessorMetadata to be access and populated by processor node. This will be committed along with
28    * offset. This metadata is mainly for windowed aggregation processor to store last emitted timestamp
29    * for now. Therefore, the supported metadata value type is only Long which is timestamp type.
30    */
31   public class ProcessorMetadata {
32   
33       private final Map<String, Long> metadata;
34   
35       // Whether metadata should be committed. We only need to commit if metadata is updated via
36       // put() or set explicitly
37       private boolean needsCommit;
38   
39       public ProcessorMetadata() {
40           this(new HashMap<>());
41       }
42   
43       public ProcessorMetadata(final Map<String, Long> metadata) {
44           this.metadata = metadata;
45           needsCommit = false;
46       }
47   
48       public static ProcessorMetadata deserialize(final byte[] metaDataBytes) {
49           if (metaDataBytes == null || metaDataBytes.length == 0) {
50               return new ProcessorMetadata();
51           }
52   
53           final ByteBuffer buffer = ByteBuffer.wrap(metaDataBytes);
54           final int entrySize = buffer.getInt();
55           final Map<String, Long> metadata = new HashMap<>(entrySize);
56           for (int i = 0; i < entrySize; i++) {
57               final int keySize = buffer.getInt();
58               final byte[] keyBytes = new byte[keySize];
59               buffer.get(keyBytes);
60               final Long value = buffer.getLong();
61               metadata.put(new String(keyBytes, StandardCharsets.UTF_8), value);
62           }
63           return new ProcessorMetadata(metadata);
64       }
65   
66       public byte[] serialize() {
67           if (metadata.isEmpty()) {
68               return new byte[0];
69           }
70   
71           final ByteArrayOutputStream outputStream = new ByteArrayOutputStream();
72           final byte[] mapSizeBytes = ByteBuffer.allocate(Integer.BYTES).putInt(metadata.size()).array();
73           outputStream.write(mapSizeBytes, 0, mapSizeBytes.length);
74   
75           for (final Map.Entry<String, Long> entry : metadata.entrySet()) {
76               final byte[] keyBytes = entry.getKey().getBytes(StandardCharsets.UTF_8);
77               final int keyLen = keyBytes.length;
78               final byte[] buffer = ByteBuffer.allocate(Integer.BYTES + keyBytes.length + Long.BYTES)
79                   .putInt(keyLen)
80                   .put(keyBytes)
81                   .putLong(entry.getValue())
82                   .array();
83               outputStream.write(buffer, 0, buffer.length);
84           }
85           return outputStream.toByteArray();
86       }
87   
88       public void put(final String key, final long value) {
89           metadata.put(key, value);
90           needsCommit = true;
91       }
92   
93       public Long get(final String key) {
94           return metadata.get(key);
95       }
96   
97       /**
98        * Merge with other metadata. Missing keys will be added. Existing key's value will be updated to
99        * max
100        * @param other Other metadata to be merged
101        */
102       public void update(final ProcessorMetadata other) {
103           if (other == null) {
104               return;
105           }
106           for (final Map.Entry<String, Long> kv : other.metadata.entrySet()) {
107               final Long value = metadata.get(kv.getKey());
108               if (value == null || value < kv.getValue()) {
109                   metadata.put(kv.getKey(), kv.getValue());
110               }
111           }
112       }
113   
114       public void setNeedsCommit(final boolean needsCommit) {
115           this.needsCommit = needsCommit;
116       }
117   
118       /**
119        * Whether metadata needs to be committed. It should be committed only if put is or
120        * {@link #setNeedsCommit} is called explicitly
121        *
122        * @return If metadata needs to be committed.
123        */
124       public boolean needsCommit() {
125           return needsCommit;
126       }
127   
128       @Override
129       public int hashCode() {
130           // needsCommit is not considered in hashCode or equals
131           return Objects.hashCode(metadata);
132       }
133   
134       @Override
135       public boolean equals(final Object obj) {
136           if (obj == null || obj.getClass() != getClass()) {
137               return false;
138           }
139           if (this == obj) {
140               return true;
141           }
142   
143           return metadata.equals(((ProcessorMetadata) obj).metadata);
144       }
145   }
