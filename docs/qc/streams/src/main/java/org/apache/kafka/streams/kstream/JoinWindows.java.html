28   /**
29    * The window specifications used for joins.
30    * <p>
31    * A {@code JoinWindows} instance defines a maximum time difference for a {@link KStream#join(KStream, ValueJoiner,
32    * JoinWindows) join over two streams} on the same key.
33    * In SQL-style you would express this join as
34    * <pre>{@code
35    *     SELECT * FROM stream1, stream2
36    *     WHERE
37    *       stream1.key = stream2.key
38    *       AND
39    *       stream1.ts - before <= stream2.ts AND stream2.ts <= stream1.ts + after
40    * }</pre>
41    * There are three different window configuration supported:
42    * <ul>
43    *     <li>before = after = time-difference</li>
44    *     <li>before = 0 and after = time-difference</li>
45    *     <li>before = time-difference and after = 0</li>
46    * </ul>
47    * A join is symmetric in the sense, that a join specification on the first stream returns the same result record as
48    * a join specification on the second stream with flipped before and after values.
49    * <p>
50    * Both values (before and after) must not result in an "inverse" window, i.e., upper-interval bound cannot be smaller
51    * than lower-interval bound.
52    * <p>
53    * {@code JoinWindows} are sliding windows, thus, they are aligned to the actual record timestamps.
54    * This implies, that each input record defines its own window with start and end time being relative to the record's
55    * timestamp.
56    * <p>
57    * For time semantics, see {@link TimestampExtractor}.
58    *
59    * @see TimeWindows
60    * @see UnlimitedWindows
61    * @see SessionWindows
62    * @see KStream#join(KStream, ValueJoiner, JoinWindows)
63    * @see KStream#join(KStream, ValueJoiner, JoinWindows, StreamJoined)
64    * @see KStream#leftJoin(KStream, ValueJoiner, JoinWindows)
65    * @see KStream#leftJoin(KStream, ValueJoiner, JoinWindows, StreamJoined)
66    * @see KStream#outerJoin(KStream, ValueJoiner, JoinWindows)
67    * @see KStream#outerJoin(KStream, ValueJoiner, JoinWindows, StreamJoined)
68    * @see TimestampExtractor
69    */
70   public class JoinWindows extends Windows<Window> {
71   
72       /** Maximum time difference for tuples that are before the join tuple. */
73       public final long beforeMs;
74       /** Maximum time difference for tuples that are after the join tuple. */
75       public final long afterMs;
76   
77       private final long graceMs;
78   
79       /**
80        * Enable left/outer stream-stream join, by not emitting left/outer results eagerly, but only after the grace period passed.
81        * This flag can only be enabled via ofTimeDifferenceAndGrace or ofTimeDifferenceWithNoGrace.
82        */
83       protected final boolean enableSpuriousResultFix;
84   
85       protected JoinWindows(final JoinWindows joinWindows) {
86           this(joinWindows.beforeMs, joinWindows.afterMs, joinWindows.graceMs, joinWindows.enableSpuriousResultFix);
87       }
88   
89       private JoinWindows(final long beforeMs,
90                           final long afterMs,
91                           final long graceMs,
92                           final boolean enableSpuriousResultFix) {
93           if (beforeMs + afterMs < 0) {
94               throw new IllegalArgumentException("Window interval (ie, beforeMs+afterMs) must not be negative.");
95           }
96   
97           if (graceMs < 0) {
98               throw new IllegalArgumentException("Grace period must not be negative.");
99           }
100   
101           this.afterMs = afterMs;
102           this.beforeMs = beforeMs;
103           this.graceMs = graceMs;
104           this.enableSpuriousResultFix = enableSpuriousResultFix;
105       }
106   
107       /**
108        * Specifies that records of the same key are joinable if their timestamps are within {@code timeDifference},
109        * i.e., the timestamp of a record from the secondary stream is max {@code timeDifference} before or after
110        * the timestamp of the record from the primary stream.
111        * <p>
112        * Using this method explicitly sets the grace period to the duration specified by {@code afterWindowEnd}, which
113        * means that only out-of-order records arriving more than the grace period after the window end will be dropped.
114        * The window close, after which any incoming records are considered late and will be rejected, is defined as
115        * {@code windowEnd + afterWindowEnd}
116        *
117        * @param timeDifference join window interval
118        * @param afterWindowEnd The grace period to admit out-of-order events to a window.
119        * @return A new JoinWindows object with the specified window definition and grace period
120        * @throws IllegalArgumentException if {@code timeDifference} is negative or can't be represented as {@code long milliseconds}
121        *                                  if {@code afterWindowEnd} is negative or can't be represented as {@code long milliseconds}
122        */
123       public static JoinWindows ofTimeDifferenceAndGrace(final Duration timeDifference, final Duration afterWindowEnd) {
124           final String timeDifferenceMsgPrefix = prepareMillisCheckFailMsgPrefix(timeDifference, "timeDifference");
125           final long timeDifferenceMs = validateMillisecondDuration(timeDifference, timeDifferenceMsgPrefix);
126   
127           final String afterWindowEndMsgPrefix = prepareMillisCheckFailMsgPrefix(afterWindowEnd, "afterWindowEnd");
128           final long afterWindowEndMs = validateMillisecondDuration(afterWindowEnd, afterWindowEndMsgPrefix);
129   
130           return new JoinWindows(timeDifferenceMs, timeDifferenceMs, afterWindowEndMs, true);
131       }
132   
133       /**
134        * Specifies that records of the same key are joinable if their timestamps are within {@code timeDifference},
135        * i.e., the timestamp of a record from the secondary stream is max {@code timeDifference} before or after
136        * the timestamp of the record from the primary stream.
137        * <p>
138        * CAUTION: Using this method implicitly sets the grace period to zero, which means that any out-of-order
139        * records arriving after the window ends are considered late and will be dropped.
140        *
141        * @param timeDifference join window interval
142        * @return a new JoinWindows object with the window definition and no grace period. Note that this means out-of-order records arriving after the window end will be dropped
143        * @throws IllegalArgumentException if {@code timeDifference} is negative or can't be represented as {@code long milliseconds}
144        */
145       public static JoinWindows ofTimeDifferenceWithNoGrace(final Duration timeDifference) {
146           return ofTimeDifferenceAndGrace(timeDifference, Duration.ofMillis(NO_GRACE_PERIOD));
147       }
148   
149       /**
150        * Specifies that records of the same key are joinable if their timestamps are within {@code timeDifference},
151        * i.e., the timestamp of a record from the secondary stream is max {@code timeDifference} before or after
152        * the timestamp of the record from the primary stream.
153        *
154        * @param timeDifference join window interval
155        * @return a new JoinWindows object with the window definition with and grace period (default to 24 hours minus {@code timeDifference})
156        * @throws IllegalArgumentException if {@code timeDifference} is negative or can't be represented as {@code long milliseconds}
157        * @deprecated since 3.0. Use {@link #ofTimeDifferenceWithNoGrace(Duration)}} instead
158        */
159       @Deprecated
160       public static JoinWindows of(final Duration timeDifference) throws IllegalArgumentException {
161           final String msgPrefix = prepareMillisCheckFailMsgPrefix(timeDifference, "timeDifference");
162           final long timeDifferenceMs = validateMillisecondDuration(timeDifference, msgPrefix);
163           return new JoinWindows(timeDifferenceMs, timeDifferenceMs, Math.max(DEPRECATED_DEFAULT_24_HR_GRACE_PERIOD - timeDifferenceMs * 2, 0), false);
164       }
165   
166       /**
167        * Changes the start window boundary to {@code timeDifference} but keep the end window boundary as is.
168        * Thus, records of the same key are joinable if the timestamp of a record from the secondary stream is at most
169        * {@code timeDifference} earlier than the timestamp of the record from the primary stream.
170        * {@code timeDifference} can be negative but its absolute value must not be larger than current window "after"
171        * value (which would result in a negative window size).
172        *
173        * @param timeDifference relative window start time
174        * @throws IllegalArgumentException if the resulting window size is negative or {@code timeDifference} can't be represented as {@code long milliseconds}
175        */
176       public JoinWindows before(final Duration timeDifference) throws IllegalArgumentException {
177           final String msgPrefix = prepareMillisCheckFailMsgPrefix(timeDifference, "timeDifference");
178           final long timeDifferenceMs = validateMillisecondDuration(timeDifference, msgPrefix);
179           return new JoinWindows(timeDifferenceMs, afterMs, graceMs, enableSpuriousResultFix);
180       }
181   
182       /**
183        * Changes the end window boundary to {@code timeDifference} but keep the start window boundary as is.
184        * Thus, records of the same key are joinable if the timestamp of a record from the secondary stream is at most
185        * {@code timeDifference} later than the timestamp of the record from the primary stream.
186        * {@code timeDifference} can be negative but its absolute value must not be larger than current window "before"
187        * value (which would result in a negative window size).
188        *
189        * @param timeDifference relative window end time
190        * @throws IllegalArgumentException if the resulting window size is negative or {@code timeDifference} can't be represented as {@code long milliseconds}
191        */
192       public JoinWindows after(final Duration timeDifference) throws IllegalArgumentException {
193           final String msgPrefix = prepareMillisCheckFailMsgPrefix(timeDifference, "timeDifference");
194           final long timeDifferenceMs = validateMillisecondDuration(timeDifference, msgPrefix);
195           return new JoinWindows(beforeMs, timeDifferenceMs, graceMs, enableSpuriousResultFix);
196       }

209       @Override
210       public long size() {
211           return beforeMs + afterMs;
212       }
213   
214       /**
215        * Reject out-of-order events that are delayed more than {@code afterWindowEnd}
216        * after the end of its window.
217        * <p>
218        * Delay is defined as (stream_time - record_timestamp).
219        *
220        * @param afterWindowEnd The grace period to admit out-of-order events to a window.
221        * @return this updated builder
222        * @throws IllegalArgumentException if the {@code afterWindowEnd} is negative or can't be represented as {@code long milliseconds}
223        * @throws IllegalStateException if {@link #grace(Duration)} is called after {@link #ofTimeDifferenceAndGrace(Duration, Duration)} or {@link #ofTimeDifferenceWithNoGrace(Duration)}
224        * @deprecated since 3.0. Use {@link #ofTimeDifferenceAndGrace(Duration, Duration)} instead
225        */
226       @Deprecated
227       public JoinWindows grace(final Duration afterWindowEnd) throws IllegalArgumentException {
228           // re-use the enableSpuriousResultFix flag to identify if grace is called after ofTimeDifferenceAndGrace/ofTimeDifferenceWithNoGrace
229           if (this.enableSpuriousResultFix) {
230               throw new IllegalStateException(
231                   "Cannot call grace() after setting grace value via ofTimeDifferenceAndGrace or ofTimeDifferenceWithNoGrace.");
232           }
233   
234           final String msgPrefix = prepareMillisCheckFailMsgPrefix(afterWindowEnd, "afterWindowEnd");
235           final long afterWindowEndMs = validateMillisecondDuration(afterWindowEnd, msgPrefix);
236           return new JoinWindows(beforeMs, afterMs, afterWindowEndMs, false);
237       }

240       public long gracePeriodMs() {
241           return graceMs;
242       }

244       @Override
245       public boolean equals(final Object o) {
246           if (this == o) {
247               return true;
248           }
249           if (o == null || getClass() != o.getClass()) {
250               return false;
251           }
252           final JoinWindows that = (JoinWindows) o;
253           return beforeMs == that.beforeMs &&
254               afterMs == that.afterMs &&
255               graceMs == that.graceMs;
256       }

259       public int hashCode() {

261       }

269               '}';
