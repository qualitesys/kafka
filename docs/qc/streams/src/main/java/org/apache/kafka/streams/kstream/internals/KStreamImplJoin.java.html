90       public <K, V1, V2, VOut> KStream<K, VOut> join(final KStream<K, V1> lhs,
91                                                      final KStream<K, V2> other,
92                                                      final ValueJoinerWithKey<? super K, ? super V1, ? super V2, ? extends VOut> joiner,
93                                                      final JoinWindows windows,
94                                                      final StreamJoined<K, V1, V2> streamJoined) {
95   
96           final StreamJoinedInternal<K, V1, V2> streamJoinedInternal = new StreamJoinedInternal<>(streamJoined);
97           final NamedInternal renamed = new NamedInternal(streamJoinedInternal.name());
98           final String joinThisSuffix = rightOuter ? "-outer-this-join" : "-this-join";
99           final String joinOtherSuffix = leftOuter ? "-outer-other-join" : "-other-join";
100   
101           final String thisWindowStreamProcessorName = renamed.suffixWithOrElseGet(
102               "-this-windowed", builder, KStreamImpl.WINDOWED_NAME);
103           final String otherWindowStreamProcessorName = renamed.suffixWithOrElseGet(
104               "-other-windowed", builder, KStreamImpl.WINDOWED_NAME);
105   
106           final String joinThisGeneratedName = rightOuter ? builder.newProcessorName(KStreamImpl.OUTERTHIS_NAME) : builder.newProcessorName(KStreamImpl.JOINTHIS_NAME);
107           final String joinOtherGeneratedName = leftOuter ? builder.newProcessorName(KStreamImpl.OUTEROTHER_NAME) : builder.newProcessorName(KStreamImpl.JOINOTHER_NAME);
108   
109           final String joinThisName = renamed.suffixWithOrElseGet(joinThisSuffix, joinThisGeneratedName);
110           final String joinOtherName = renamed.suffixWithOrElseGet(joinOtherSuffix, joinOtherGeneratedName);
111   
112           final String joinMergeName = renamed.suffixWithOrElseGet(
113               "-merge", builder, KStreamImpl.MERGE_NAME);
114   
115           final GraphNode thisGraphNode = ((AbstractStream<?, ?>) lhs).graphNode;
116           final GraphNode otherGraphNode = ((AbstractStream<?, ?>) other).graphNode;
117   
118           final StoreBuilder<WindowStore<K, V1>> thisWindowStore;
119           final StoreBuilder<WindowStore<K, V2>> otherWindowStore;
120           final String userProvidedBaseStoreName = streamJoinedInternal.storeName();
121   
122           final WindowBytesStoreSupplier thisStoreSupplier = streamJoinedInternal.thisStoreSupplier();
123           final WindowBytesStoreSupplier otherStoreSupplier = streamJoinedInternal.otherStoreSupplier();
124   
125           assertUniqueStoreNames(thisStoreSupplier, otherStoreSupplier);
126   
127           if (thisStoreSupplier == null) {
128               final String thisJoinStoreName = userProvidedBaseStoreName == null ? joinThisGeneratedName : userProvidedBaseStoreName + joinThisSuffix;
129               thisWindowStore = joinWindowStoreBuilder(thisJoinStoreName, windows, streamJoinedInternal.keySerde(), streamJoinedInternal.valueSerde(), streamJoinedInternal.loggingEnabled(), streamJoinedInternal.logConfig());
130           } else {
131               assertWindowSettings(thisStoreSupplier, windows);
132               thisWindowStore = joinWindowStoreBuilderFromSupplier(thisStoreSupplier, streamJoinedInternal.keySerde(), streamJoinedInternal.valueSerde());
133           }
134   
135           if (otherStoreSupplier == null) {
136               final String otherJoinStoreName = userProvidedBaseStoreName == null ? joinOtherGeneratedName : userProvidedBaseStoreName + joinOtherSuffix;
137               otherWindowStore = joinWindowStoreBuilder(otherJoinStoreName, windows, streamJoinedInternal.keySerde(), streamJoinedInternal.otherValueSerde(), streamJoinedInternal.loggingEnabled(), streamJoinedInternal.logConfig());
138           } else {
139               assertWindowSettings(otherStoreSupplier, windows);
140               otherWindowStore = joinWindowStoreBuilderFromSupplier(otherStoreSupplier, streamJoinedInternal.keySerde(), streamJoinedInternal.otherValueSerde());
141           }
142   
143           final KStreamJoinWindow<K, V1> thisWindowedStream = new KStreamJoinWindow<>(thisWindowStore.name());
144   
145           final ProcessorParameters<K, V1, ?, ?> thisWindowStreamProcessorParams = new ProcessorParameters<>(thisWindowedStream, thisWindowStreamProcessorName);
146           final ProcessorGraphNode<K, V1> thisWindowedStreamsNode = new ProcessorGraphNode<>(thisWindowStreamProcessorName, thisWindowStreamProcessorParams);
147           builder.addGraphNode(thisGraphNode, thisWindowedStreamsNode);
148   
149           final KStreamJoinWindow<K, V2> otherWindowedStream = new KStreamJoinWindow<>(otherWindowStore.name());
150   
151           final ProcessorParameters<K, V2, ?, ?> otherWindowStreamProcessorParams = new ProcessorParameters<>(otherWindowedStream, otherWindowStreamProcessorName);
152           final ProcessorGraphNode<K, V2> otherWindowedStreamsNode = new ProcessorGraphNode<>(otherWindowStreamProcessorName, otherWindowStreamProcessorParams);
153           builder.addGraphNode(otherGraphNode, otherWindowedStreamsNode);
154   
155           Optional<StoreBuilder<KeyValueStore<TimestampedKeyAndJoinSide<K>, LeftOrRightValue<V1, V2>>>> outerJoinWindowStore = Optional.empty();
156           if (leftOuter) {
157               outerJoinWindowStore = Optional.of(sharedOuterJoinWindowStoreBuilder(windows, streamJoinedInternal, joinThisGeneratedName));
158           }
159   
160           // Time-shared between joins to keep track of the maximum stream time
161           final TimeTracker sharedTimeTracker = new TimeTracker();
162   
163           final JoinWindowsInternal internalWindows = new JoinWindowsInternal(windows);
164           final KStreamKStreamJoin<K, V1, V2, VOut> joinThis = new KStreamKStreamJoin<>(
165               true,
166               otherWindowStore.name(),
167               internalWindows,
168               joiner,
169               leftOuter,
170               outerJoinWindowStore.map(StoreBuilder::name),
171               sharedTimeTracker
172           );
173   
174           final KStreamKStreamJoin<K, V2, V1, VOut> joinOther = new KStreamKStreamJoin<>(
175               false,
176               thisWindowStore.name(),
177               internalWindows,
178               AbstractStream.reverseJoinerWithKey(joiner),
179               rightOuter,
180               outerJoinWindowStore.map(StoreBuilder::name),
181               sharedTimeTracker
182           );
183   
184           final PassThrough<K, VOut> joinMerge = new PassThrough<>();
185   
186           final StreamStreamJoinNode.StreamStreamJoinNodeBuilder<K, V1, V2, VOut> joinBuilder = StreamStreamJoinNode.streamStreamJoinNodeBuilder();
187   
188           final ProcessorParameters<K, V1, ?, ?> joinThisProcessorParams = new ProcessorParameters<>(joinThis, joinThisName);
189           final ProcessorParameters<K, V2, ?, ?> joinOtherProcessorParams = new ProcessorParameters<>(joinOther, joinOtherName);
190           final ProcessorParameters<K, VOut, ?, ?> joinMergeProcessorParams = new ProcessorParameters<>(joinMerge, joinMergeName);
191   
192           joinBuilder.withJoinMergeProcessorParameters(joinMergeProcessorParams)
193                      .withJoinThisProcessorParameters(joinThisProcessorParams)
194                      .withJoinOtherProcessorParameters(joinOtherProcessorParams)
195                      .withThisWindowStoreBuilder(thisWindowStore)
196                      .withOtherWindowStoreBuilder(otherWindowStore)
197                      .withThisWindowedStreamProcessorParameters(thisWindowStreamProcessorParams)
198                      .withOtherWindowedStreamProcessorParameters(otherWindowStreamProcessorParams)
199                      .withOuterJoinWindowStoreBuilder(outerJoinWindowStore)
200                      .withValueJoiner(joiner)
201                      .withNodeName(joinMergeName);
202   
203           if (internalWindows.spuriousResultFixEnabled()) {
204               joinBuilder.withSpuriousResultFixEnabled();
205           }
206   
207           final GraphNode joinGraphNode = joinBuilder.build();
208   
209           builder.addGraphNode(Arrays.asList(thisGraphNode, otherGraphNode), joinGraphNode);
210   
211           final Set<String> allSourceNodes = new HashSet<>(((KStreamImpl<K, V1>) lhs).subTopologySourceNodes);
212           allSourceNodes.addAll(((KStreamImpl<K, V2>) other).subTopologySourceNodes);
213   
214           // do not have serde for joined result;
215           // also for key serde we do not inherit from either since we cannot tell if these two serdes are different
216           return new KStreamImpl<>(joinMergeName, streamJoinedInternal.keySerde(), null, allSourceNodes, false, joinGraphNode, builder);
217       }

219       private void assertWindowSettings(final WindowBytesStoreSupplier supplier, final JoinWindows joinWindows) {
220           if (!supplier.retainDuplicates()) {
221               throw new StreamsException("The StoreSupplier must set retainDuplicates=true, found retainDuplicates=false");
222           }
223           final boolean allMatch = supplier.retentionPeriod() == (joinWindows.size() + joinWindows.gracePeriodMs()) &&
224               supplier.windowSize() == joinWindows.size();
225           if (!allMatch) {
226               throw new StreamsException(String.format("Window settings mismatch. WindowBytesStoreSupplier settings %s must match JoinWindows settings %s" +
227                                                            " for the window size and retention period", supplier, joinWindows));
228           }
229       }

231       private void assertUniqueStoreNames(final WindowBytesStoreSupplier supplier,
232                                           final WindowBytesStoreSupplier otherSupplier) {
233   
234           if (supplier != null
235               && otherSupplier != null
236               && supplier.name().equals(otherSupplier.name())) {
237               throw new StreamsException("Both StoreSuppliers have the same name.  StoreSuppliers must provide unique names");
238           }
239       }

242                                                                                    final JoinWindows windows,

246                                                                                    final Map<String, String> logConfig) {
247           final StoreBuilder<WindowStore<K, V>> builder = Stores.windowStoreBuilder(
248               Stores.persistentWindowStore(
249                   storeName + "-store",
250                   Duration.ofMillis(windows.size() + windows.gracePeriodMs()),
251                   Duration.ofMillis(windows.size()),
252                   true
253               ),
254               keySerde,
255               valueSerde
256           );
257           if (loggingEnabled) {
258               builder.withLoggingEnabled(logConfig);
259           } else {
260               builder.withLoggingDisabled();
261           }
262   
263           return builder;
264       }

266       private <K, V1, V2> String buildOuterJoinWindowStoreName(final StreamJoinedInternal<K, V1, V2> streamJoinedInternal, final String joinThisGeneratedName) {
267           final String outerJoinSuffix = rightOuter ? "-outer-shared-join" : "-left-shared-join";
268   
269           if (streamJoinedInternal.thisStoreSupplier() != null && !streamJoinedInternal.thisStoreSupplier().name().isEmpty()) {
270               return streamJoinedInternal.thisStoreSupplier().name() + outerJoinSuffix;
271           } else if (streamJoinedInternal.storeName() != null) {
272               return streamJoinedInternal.storeName() + outerJoinSuffix;
273           } else {
274               return KStreamImpl.OUTERSHARED_NAME
275                   + joinThisGeneratedName.substring(
276                   rightOuter
277                       ? KStreamImpl.OUTERTHIS_NAME.length()
278                       : KStreamImpl.JOINTHIS_NAME.length());
279           }
280       }

282       private <K, V1, V2> StoreBuilder<KeyValueStore<TimestampedKeyAndJoinSide<K>, LeftOrRightValue<V1, V2>>> sharedOuterJoinWindowStoreBuilder(final JoinWindows windows,
283                                                                                                                                                 final StreamJoinedInternal<K, V1, V2> streamJoinedInternal,
284                                                                                                                                                 final String joinThisGeneratedName) {
285           final boolean persistent = streamJoinedInternal.thisStoreSupplier() == null || streamJoinedInternal.thisStoreSupplier().get().persistent();
286           final String storeName = buildOuterJoinWindowStoreName(streamJoinedInternal, joinThisGeneratedName) + "-store";
287   
288           // we are using a key-value store with list-values for the shared store, and have the window retention / grace period
289           // handled totally on the processor node level, and hence here we are only validating these values but not using them at all
290           final Duration retentionPeriod = Duration.ofMillis(windows.size() + windows.gracePeriodMs());
291           final Duration windowSize = Duration.ofMillis(windows.size());
292           final String rpMsgPrefix = prepareMillisCheckFailMsgPrefix(retentionPeriod, "retentionPeriod");
293           final long retentionMs = validateMillisecondDuration(retentionPeriod, rpMsgPrefix);
294           final String wsMsgPrefix = prepareMillisCheckFailMsgPrefix(windowSize, "windowSize");
295           final long windowSizeMs = validateMillisecondDuration(windowSize, wsMsgPrefix);
296   
297           if (retentionMs < 0L) {
298               throw new IllegalArgumentException("retentionPeriod cannot be negative");
299           }
300           if (windowSizeMs < 0L) {
301               throw new IllegalArgumentException("windowSize cannot be negative");
302           }
303           if (windowSizeMs > retentionMs) {
304               throw new IllegalArgumentException("The retention period of the window store "
305                       + storeName + " must be no smaller than its window size. Got size=["
306                       + windowSizeMs + "], retention=[" + retentionMs + "]");
307           }
308   
309           final TimestampedKeyAndJoinSideSerde<K> timestampedKeyAndJoinSideSerde = new TimestampedKeyAndJoinSideSerde<>(streamJoinedInternal.keySerde());
310           final LeftOrRightValueSerde<V1, V2> leftOrRightValueSerde = new LeftOrRightValueSerde<>(streamJoinedInternal.valueSerde(), streamJoinedInternal.otherValueSerde());
311   
312           final StoreBuilder<KeyValueStore<TimestampedKeyAndJoinSide<K>, LeftOrRightValue<V1, V2>>> builder =
313               new ListValueStoreBuilder<>(
314                   persistent ? Stores.persistentKeyValueStore(storeName) : Stores.inMemoryKeyValueStore(storeName),
315                   timestampedKeyAndJoinSideSerde,
316                   leftOrRightValueSerde,
317                   Time.SYSTEM
318               );
319   
320           if (streamJoinedInternal.loggingEnabled()) {
321               builder.withLoggingEnabled(streamJoinedInternal.logConfig());
322           } else {
323               builder.withLoggingDisabled();
324           }
325   
326           return builder;
327       }
