17   package org.apache.kafka.streams;
18   
19   import java.util.Objects;
20   
21   /**
22    * Encapsulates information about lag, at a store partition replica (active or standby). This information is constantly changing as the
23    * tasks process records and thus, they should be treated as simply instantaenous measure of lag.
24    */
25   public class LagInfo {
26   
27       private final long currentOffsetPosition;
28   
29       private final long endOffsetPosition;
30   
31       private final long offsetLag;
32   
33       LagInfo(final long currentOffsetPosition, final long endOffsetPosition) {
34           this.currentOffsetPosition = currentOffsetPosition;
35           this.endOffsetPosition = endOffsetPosition;
36           this.offsetLag = Math.max(0, endOffsetPosition - currentOffsetPosition);
37       }
38   
39       /**
40        * Get the current maximum offset on the store partition's changelog topic, that has been successfully written into
41        * the store partition's state store.
42        *
43        * @return current consume offset for standby/restoring store partitions &amp;   simply endoffset for active store partition replicas
44        */
45       public long currentOffsetPosition() {
46           return this.currentOffsetPosition;
47       }
48   
49       /**
50        * Get the end offset position for this store partition's changelog topic on the Kafka brokers.
51        *
52        * @return last offset written to the changelog topic partition
53        */
54       public long endOffsetPosition() {
55           return this.endOffsetPosition;
56       }
57   
58       /**
59        * Get the measured lag between current and end offset positions, for this store partition replica
60        *
61        * @return lag as measured by message offsets
62        */
63       public long offsetLag() {
64           return this.offsetLag;
65       }
66   
67       @Override
68       public boolean equals(final Object obj) {
69           if (!(obj instanceof LagInfo)) {
70               return false;
71           }
72           final LagInfo other = (LagInfo) obj;
73           return currentOffsetPosition == other.currentOffsetPosition
74               && endOffsetPosition == other.endOffsetPosition
75               && this.offsetLag == other.offsetLag;
76       }
77   
78       @Override
79       public int hashCode() {
80           return Objects.hash(currentOffsetPosition, endOffsetPosition, offsetLag);
81       }
82   
83       @Override
84       public String toString() {
85           return "LagInfo {" +
86               " currentOffsetPosition=" + currentOffsetPosition +
87               ", endOffsetPosition=" + endOffsetPosition +
88               ", offsetLag=" + offsetLag +
89               '}';
90       }
91   }
