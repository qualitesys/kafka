75       @Override
76       public void transitionToActive(final StreamTask streamTask, final RecordCollector recordCollector, final ThreadCache newCache) {
77           if (stateManager.taskType() != TaskType.ACTIVE) {
78               throw new IllegalStateException("Tried to transition processor context to active but the state manager's " +

89           if (stateManager.taskType() != TaskType.STANDBY) {
90               throw new IllegalStateException("Tried to transition processor context to standby but the state manager's " +
91                                                   "type was " + stateManager.taskType());
92           }
93           this.streamTask = null;
94           this.collector = null;
95           this.cache = newCache;
96           addAllFlushListenersToNewCache();

121       @Override
122       public void logChange(final String storeName,
123                             final Bytes key,
124                             final byte[] value,
125                             final long timestamp,
126                             final Position position) {
127           throwUnsupportedOperationExceptionIfStandby("logChange");
128   
129           final TopicPartition changelogPartition = stateManager().registeredChangelogPartitionFor(storeName);
130   
131           final Headers headers;
132           if (!consistencyEnabled) {
133               headers = null;
134           } else {
135               // Add the vector clock to the header part of every record
136               headers = new RecordHeaders();
137               headers.add(ChangelogRecordDeserializationHelper.CHANGELOG_VERSION_HEADER_RECORD_CONSISTENCY);
138               headers.add(new RecordHeader(ChangelogRecordDeserializationHelper.CHANGELOG_POSITION_HEADER_KEY,
139                       PositionSerde.serialize(position).array()));
140           }
141   
142           collector.send(
143               changelogPartition.topic(),
144               key,
145               value,
146               headers,
147               changelogPartition.partition(),
148               timestamp,
149               BYTES_KEY_SERIALIZER,
150               BYTEARRAY_VALUE_SERIALIZER,
151               null,
152               null);
153       }

155       /**
156        * @throws StreamsException if an attempt is made to access this state store from an unknown node
157        * @throws UnsupportedOperationException if the current streamTask type is standby
158        */
159       @SuppressWarnings("unchecked")
160       @Override
161       public <S extends StateStore> S  getStateStore(final String name) {
162           throwUnsupportedOperationExceptionIfStandby("getStateStore");
163           if (currentNode() == null) {
164               throw new StreamsException("Accessing from an unknown node");
165           }
166   
167           final StateStore globalStore = stateManager.getGlobalStore(name);
168           if (globalStore != null) {
169               return (S) getReadOnlyStore(globalStore);
170           }
171   
172           if (!currentNode().stateStores.contains(name)) {
173               throw new StreamsException("Processor " + currentNode().name() + " has no access to StateStore " + name +
174                   " as the store is not connected to the processor. If you add stores manually via '.addStateStore()' " +
175                   "make sure to connect the added store to the processor by providing the processor name to " +
176                   "'.addStateStore()' or connect them via '.connectProcessorAndStateStores()'. " +
177                   "DSL users need to provide the store name to '.process()', '.transform()', or '.transformValues()' " +
178                   "to connect the store to the corresponding operator, or they can provide a StoreBuilder by implementing " +
179                   "the stores() method on the Supplier itself. If you do not add stores manually, " +
180                   "please file a bug report at https://issues.apache.org/jira/projects/KAFKA.");
181           }
182   
183           final StateStore store = stateManager.getStore(name);
184           return (S) getReadWriteStore(store);
185       }

231       @SuppressWarnings("unchecked")
232       @Override
233       public <K, V> void forward(final Record<K, V> record, final String childName) {
234           throwUnsupportedOperationExceptionIfStandby("forward");
235   
236           final ProcessorNode<?, ?, ?, ?> previousNode = currentNode();
237           if (previousNode == null) {
238               throw new StreamsException("Current node is unknown. This can happen if 'forward()' is called " +
239                       "in an illegal scope. The root cause could be that a 'Processor' or 'Transformer' instance" +
240                       " is shared. To avoid this error, make sure that your suppliers return new instances " +
241                       "each time 'get()' of Supplier is called and do not return the same object reference " +
242                       "multiple times.");
243           }
244   
245           final ProcessorRecordContext previousContext = recordContext;
246   
247           try {
248               // we don't want to set the recordContext if it's null, since that means that
249               // the context itself is undefined. this isn't perfect, since downstream
250               // old API processors wouldn't see the timestamps or headers of upstream
251               // new API processors. But then again, from the perspective of those old-API
252               // processors, even consulting the timestamp or headers when the record context
253               // is undefined is itself not well defined. Plus, I don't think we need to worry
254               // too much about heterogeneous applications, in which the upstream processor is
255               // implementing the new API and the downstream one is implementing the old API.
256               // So, this seems like a fine compromise for now.
257               if (recordContext != null && (record.timestamp() != timestamp() || record.headers() != headers())) {
258                   recordContext = new ProcessorRecordContext(
259                       record.timestamp(),
260                       recordContext.offset(),
261                       recordContext.partition(),
262                       recordContext.topic(),
263                       record.headers());
264               }
265   
266               if (childName == null) {
267                   final List<? extends ProcessorNode<?, ?, ?, ?>> children = currentNode().children();
268                   for (final ProcessorNode<?, ?, ?, ?> child : children) {
269                       forwardInternal((ProcessorNode<K, V, ?, ?>) child, record);
270                   }
271               } else {
272                   final ProcessorNode<?, ?, ?, ?> child = currentNode().getChild(childName);
273                   if (child == null) {
274                       throw new StreamsException("Unknown downstream node: " + childName
275                                                      + " either does not exist or is not connected to this processor.");
276                   }
277                   forwardInternal((ProcessorNode<K, V, ?, ?>) child, record);
278               }
279   
280           } finally {
281               recordContext = previousContext;
282               setCurrentNode(previousNode);
283           }
284       }
285   
286       private <K, V> void forwardInternal(final ProcessorNode<K, V, ?, ?> child,
287                                           final Record<K, V> record) {
288           setCurrentNode(child);
289   
290           child.process(record);
291   
292           if (child.isTerminalNode()) {
293               streamTask.maybeRecordE2ELatency(record.timestamp(), currentSystemTimeMs(), child.name());
294           }
295       }

304       public Cancellable schedule(final Duration interval,
