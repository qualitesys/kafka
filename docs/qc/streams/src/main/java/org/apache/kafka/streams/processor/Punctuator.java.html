19   import java.time.Duration;
20   import org.apache.kafka.streams.processor.api.Record;
21   
22   /**
23    * A functional interface used as an argument to {@link ProcessorContext#schedule(Duration, PunctuationType, Punctuator)}.
24    *
25    * @see Cancellable
26    */
27   public interface Punctuator {
28   
29       /**
30        * Perform the scheduled periodic operation.
31        *
32        * <p> If this method accesses {@link ProcessorContext} or
33        * {@link org.apache.kafka.streams.processor.api.ProcessorContext}, record metadata like topic,
34        * partition, and offset or {@link org.apache.kafka.streams.processor.api.RecordMetadata} won't
35        * be available.
36        *
37        * <p> Furthermore, for any record that is sent downstream via {@link ProcessorContext#forward(Object, Object)}
38        * or {@link org.apache.kafka.streams.processor.api.ProcessorContext#forward(Record)}, there
39        * won't be any record metadata. If {@link ProcessorContext#forward(Object, Object)} is used,
40        * it's also not possible to set records headers.
41        *
42        * @param timestamp when the operation is being called, depending on {@link PunctuationType}
43        */
44       void punctuate(long timestamp);
45   }
