28   /**
29    * RocksDB store backed by two SegmentedBytesStores which can optimize scan by time as well as window
30    * lookup for a specific key.
31    *
32    * Schema for first SegmentedBytesStore (base store) is as below:
33    *     Key schema: | timestamp + [timestamp] + recordkey |
34    *     Value schema: | value |. Value here is determined by caller.
35    *
36    * Schema for second SegmentedBytesStore (index store) is as below:
37    *     Key schema: | record + timestamp + [timestamp]|
38    *     Value schema: ||
39    *
40    * Note there could be two timestamps if we store both window end time and window start time.
41    *
42    * Operations:
43    *     Put: 1. Put to index store. 2. Put to base store.
44    *     Delete: 1. Delete from base store. 2. Delete from index store.
45    * Since we need to update two stores, failure can happen in the middle. We put in index store first
46    * to make sure if a failure happens in second step and the view is inconsistent, we can't get the
47    * value for the key. We delete from base store first to make sure if a failure happens in second step
48    * and the view is inconsistent, we can't get the value for the key.
49    *
50    * Note:
51    *     Index store can be optional if we can construct the timestamp in base store instead of looking
52    *     them up from index store.
53    *
54    * @see RocksDBTimeOrderedSessionSegmentedBytesStore
55    * @see RocksDBTimeOrderedWindowSegmentedBytesStore
56    */
57   public abstract class AbstractRocksDBTimeOrderedSegmentedBytesStore extends AbstractDualSchemaRocksDBSegmentedBytesStore<KeyValueSegment> {
58       private static final Logger LOG = LoggerFactory.getLogger(AbstractDualSchemaRocksDBSegmentedBytesStore.class);
59   
60       abstract class IndexToBaseStoreIterator implements KeyValueIterator<Bytes, byte[]> {
61           private final KeyValueIterator<Bytes, byte[]> indexIterator;
62           private byte[] cachedValue;
63   
64   
65           IndexToBaseStoreIterator(final KeyValueIterator<Bytes, byte[]> indexIterator) {
66               this.indexIterator = indexIterator;
67           }
68   
69           @Override
70           public void close() {
71               indexIterator.close();
72           }
73   
74           @Override
75           public Bytes peekNextKey() {
76               if (!hasNext()) {
77                   throw new NoSuchElementException();
78               }
79               return getBaseKey(indexIterator.peekNextKey());
80           }
81   
82           @Override
83           public boolean hasNext() {
84               while (indexIterator.hasNext()) {
85                   final Bytes key = indexIterator.peekNextKey();
86                   final Bytes baseKey = getBaseKey(key);
87   
88                   cachedValue = get(baseKey);
89                   if (cachedValue == null) {
90                       // Key not in base store, inconsistency happened and remove from index.
91                       indexIterator.next();
92                       AbstractRocksDBTimeOrderedSegmentedBytesStore.this.removeIndex(key);
93                   } else {
94                       return true;
95                   }
96               }
97               return false;
98           }
99   
100           @Override
101           public KeyValue<Bytes, byte[]> next() {
102               if (cachedValue == null && !hasNext()) {
103                   throw new NoSuchElementException();
104               }
105               final KeyValue<Bytes, byte[]> ret = indexIterator.next();
106               final byte[] value = cachedValue;
107               cachedValue = null;
108               return KeyValue.pair(getBaseKey(ret.key), value);
109           }
110   
111           abstract protected Bytes getBaseKey(final Bytes indexKey);
112       }

141                                             final long from,
142                                             final long to,
143                                             final boolean forward) {
144           if (indexKeySchema.isPresent()) {
145               final List<KeyValueSegment> searchSpace = indexKeySchema.get().segmentsToSearch(segments, from, to,
146                   forward);
147

152                   searchSpace.iterator(),
153                   indexKeySchema.get().hasNextCondition(key, key, from, to, forward),

163           final Bytes binaryTo = baseKeySchema.upperRangeFixedSize(key, to);
