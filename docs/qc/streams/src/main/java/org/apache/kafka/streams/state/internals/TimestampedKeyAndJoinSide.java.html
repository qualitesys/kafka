17   package org.apache.kafka.streams.state.internals;
18   
19   import org.apache.kafka.streams.KeyValue;
20   
21   import java.util.Objects;
22   
23   /**
24    * Combines a timestamped key from a {@link KeyValue} with a boolean value referencing if the key is
25    * part of the left join (true) or right join (false). This class is only useful when a state
26    * store needs to be shared between left and right processors, and each processor needs to
27    * access the key of the other processor.
28    *
29    * Note that it might be cleaner to have two layers for such usages: first a KeyAndJoinSide, where the Key
30    * is in the form of a <timestamp, key>; but with the nested structure serdes would need extra byte array copies.
31    * Since it is only used in a single place today we decided to combine them into a single type / serde.
32    */
33   public class TimestampedKeyAndJoinSide<K> {
34       private final K key;
35       private final long timestamp;
36       private final boolean leftSide;
37   
38       private TimestampedKeyAndJoinSide(final boolean leftSide, final K key, final long timestamp) {
39           this.key = Objects.requireNonNull(key, "key cannot be null");
40           this.leftSide = leftSide;
41           this.timestamp = timestamp;
42       }
43   
44       /**
45        * Create a new {@link TimestampedKeyAndJoinSide} instance if the provide {@code key} is not {@code null}.
46        *
47        * @param leftSide True if the key is part of the left join side; False if it is from the right join side
48        * @param key      the key
49        * @param <K>      the type of the key
50        * @return a new {@link TimestampedKeyAndJoinSide} instance if the provide {@code key} is not {@code null}
51        */
52       public static <K> TimestampedKeyAndJoinSide<K> make(final boolean leftSide, final K key, final long timestamp) {
53           return new TimestampedKeyAndJoinSide<>(leftSide, key, timestamp);
54       }
55   
56       public boolean isLeftSide() {
57           return leftSide;
58       }
59   
60       public K getKey() {
61           return key;
62       }
63   
64       public long getTimestamp() {
65           return timestamp;
66       }
67   
68       @Override
69       public String toString() {
70           final String joinSide = leftSide ? "left" : "right";
71           return "<" + joinSide + "," + key + ":" + timestamp + ">";
72       }
73   
74       @Override
75       public boolean equals(final Object o) {
76           if (this == o) {
77               return true;
78           }
79           if (o == null || getClass() != o.getClass()) {
80               return false;
81           }
82           final TimestampedKeyAndJoinSide<?> that = (TimestampedKeyAndJoinSide<?>) o;
83           return leftSide == that.leftSide &&
84               Objects.equals(key, that.key) &&
85               timestamp == that.timestamp;
86       }
87   
88       @Override
89       public int hashCode() {
90           return Objects.hash(leftSide, key, timestamp);
91       }
92   }
