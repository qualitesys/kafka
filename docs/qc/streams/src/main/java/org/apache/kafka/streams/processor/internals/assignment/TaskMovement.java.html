45           this.caughtUpClients = caughtUpClients;

56       private static boolean taskIsNotCaughtUpOnClientAndOtherMoreCaughtUpClientsExist(final TaskId task,
57                                                                                        final UUID client,
58                                                                                        final Map<UUID, ClientState> clientStates,
59                                                                                        final Map<TaskId, SortedSet<UUID>> tasksToCaughtUpClients,
60                                                                                        final Map<TaskId, SortedSet<UUID>> tasksToClientByLag) {
61           final SortedSet<UUID> taskClients = requireNonNull(tasksToClientByLag.get(task), "uninitialized set");
62           if (taskIsCaughtUpOnClient(task, client, tasksToCaughtUpClients)) {
63               return false;
64           }
65           final long mostCaughtUpLag = clientStates.get(taskClients.first()).lagFor(task);
66           final long clientLag = clientStates.get(client).lagFor(task);
67           return mostCaughtUpLag < clientLag;
68       }

70       private static boolean taskIsCaughtUpOnClient(final TaskId task,
71                                                     final UUID client,
72                                                     final Map<TaskId, SortedSet<UUID>> tasksToCaughtUpClients) {
73           final Set<UUID> caughtUpClients = requireNonNull(tasksToCaughtUpClients.get(task), "uninitialized set");
74           return caughtUpClients.contains(client);
75       }

77       static int assignActiveTaskMovements(final Map<TaskId, SortedSet<UUID>> tasksToCaughtUpClients,
78                                            final Map<TaskId, SortedSet<UUID>> tasksToClientByLag,
79                                            final Map<UUID, ClientState> clientStates,
80                                            final Map<UUID, Set<TaskId>> warmups,
81                                            final AtomicInteger remainingWarmupReplicas) {
82           final BiFunction<UUID, TaskId, Boolean> caughtUpPredicate =
83               (client, task) -> taskIsCaughtUpOnClient(task, client, tasksToCaughtUpClients);
84   
85           final ConstrainedPrioritySet caughtUpClientsByTaskLoad = new ConstrainedPrioritySet(
86               caughtUpPredicate,
87               client -> clientStates.get(client).assignedTaskLoad()
88           );
89   
90           final Queue<TaskMovement> taskMovements = new PriorityQueue<>(
91               Comparator.comparing(TaskMovement::numCaughtUpClients).thenComparing(TaskMovement::task)
92           );
93   
94           for (final Map.Entry<UUID, ClientState> clientStateEntry : clientStates.entrySet()) {
95               final UUID client = clientStateEntry.getKey();
96               final ClientState state = clientStateEntry.getValue();
97               for (final TaskId task : state.activeTasks()) {
98                   // if the desired client is not caught up, and there is another client that _is_ more caught up, then
99                   // we schedule a movement, so we can move the active task to a more caught-up client. We'll try to
100                   // assign a warm-up to the desired client so that we can move it later on.
101                   if (taskIsNotCaughtUpOnClientAndOtherMoreCaughtUpClientsExist(task, client, clientStates, tasksToCaughtUpClients, tasksToClientByLag)) {
102                       taskMovements.add(new TaskMovement(task, client, tasksToCaughtUpClients.get(task)));
103                   }
104               }
105               caughtUpClientsByTaskLoad.offer(client);
106           }
107   
108           final int movementsNeeded = taskMovements.size();
109   
110           for (final TaskMovement movement : taskMovements) {
111               // Attempt to find a caught up standby, otherwise find any caught up client, failing that use the most
112               // caught up client.
113               final boolean moved = tryToSwapStandbyAndActiveOnCaughtUpClient(clientStates, caughtUpClientsByTaskLoad, movement) ||
114                       tryToMoveActiveToCaughtUpClientAndTryToWarmUp(clientStates, warmups, remainingWarmupReplicas, caughtUpClientsByTaskLoad, movement) ||
115                       tryToMoveActiveToMostCaughtUpClient(tasksToClientByLag, clientStates, warmups, remainingWarmupReplicas, caughtUpClientsByTaskLoad, movement);
116   
117               if (!moved) {
118                   throw new IllegalStateException("Tried to move task to more caught-up client as scheduled before but none exist");
119               }
120           }
121   
122           return movementsNeeded;
123       }

125       static int assignStandbyTaskMovements(final Map<TaskId, SortedSet<UUID>> tasksToCaughtUpClients,
126                                             final Map<TaskId, SortedSet<UUID>> tasksToClientByLag,
127                                             final Map<UUID, ClientState> clientStates,
128                                             final AtomicInteger remainingWarmupReplicas,
129                                             final Map<UUID, Set<TaskId>> warmups) {
130           final BiFunction<UUID, TaskId, Boolean> caughtUpPredicate =
131               (client, task) -> taskIsCaughtUpOnClient(task, client, tasksToCaughtUpClients);
132   
133           final ConstrainedPrioritySet caughtUpClientsByTaskLoad = new ConstrainedPrioritySet(
134               caughtUpPredicate,
135               client -> clientStates.get(client).assignedTaskLoad()
136           );
137   
138           final Queue<TaskMovement> taskMovements = new PriorityQueue<>(
139               Comparator.comparing(TaskMovement::numCaughtUpClients).thenComparing(TaskMovement::task)
140           );
141   
142           for (final Map.Entry<UUID, ClientState> clientStateEntry : clientStates.entrySet()) {
143               final UUID destination = clientStateEntry.getKey();
144               final ClientState state = clientStateEntry.getValue();
145               for (final TaskId task : state.standbyTasks()) {
146                   if (warmups.getOrDefault(destination, Collections.emptySet()).contains(task)) {
147                       // this is a warmup, so we won't move it.
148                   } else if (taskIsNotCaughtUpOnClientAndOtherMoreCaughtUpClientsExist(task, destination, clientStates, tasksToCaughtUpClients, tasksToClientByLag)) {
149                       // if the desired client is not caught up, and there is another client that _is_ more caught up, then
150                       // we schedule a movement, so we can move the active task to the caught-up client. We'll try to
151                       // assign a warm-up to the desired client so that we can move it later on.
152                       taskMovements.add(new TaskMovement(task, destination, tasksToCaughtUpClients.get(task)));
153                   }
154               }
155               caughtUpClientsByTaskLoad.offer(destination);
156           }
157   
158           int movementsNeeded = 0;
159   
160           for (final TaskMovement movement : taskMovements) {
161               final Function<UUID, Boolean> eligibleClientPredicate =
162                       clientId -> !clientStates.get(clientId).hasAssignedTask(movement.task);
163               UUID sourceClient = caughtUpClientsByTaskLoad.poll(
164                   movement.task,
165                   eligibleClientPredicate
166               );
167   
168               if (sourceClient == null) {
169                   sourceClient = mostCaughtUpEligibleClient(tasksToClientByLag, eligibleClientPredicate, movement.task, movement.destination);
170               }
171   
172               if (sourceClient == null) {
173                   // then there's no caught-up client that doesn't already have a copy of this task, so there's
174                   // nowhere to move it.
175               } else {
176                   moveStandbyAndTryToWarmUp(
177                       remainingWarmupReplicas,
178                       movement.task,
179                       clientStates.get(sourceClient),
180                       clientStates.get(movement.destination)
181                   );
182                   caughtUpClientsByTaskLoad.offerAll(asList(sourceClient, movement.destination));
183                   movementsNeeded++;
184               }
185           }
186   
187           return movementsNeeded;
188       }

190       private static boolean tryToSwapStandbyAndActiveOnCaughtUpClient(final Map<UUID, ClientState> clientStates,
191                                                                        final ConstrainedPrioritySet caughtUpClientsByTaskLoad,
192                                                                        final TaskMovement movement) {
193           final UUID caughtUpStandbySourceClient = caughtUpClientsByTaskLoad.poll(
194                   movement.task,
195                   c -> clientStates.get(c).hasStandbyTask(movement.task)
196           );
197           if (caughtUpStandbySourceClient != null) {
198               swapStandbyAndActive(
199                       movement.task,
200                       clientStates.get(caughtUpStandbySourceClient),
201                       clientStates.get(movement.destination)
202               );
203               caughtUpClientsByTaskLoad.offerAll(asList(caughtUpStandbySourceClient, movement.destination));
204               return true;
205           }
206           return false;
207       }

209       private static boolean tryToMoveActiveToCaughtUpClientAndTryToWarmUp(final Map<UUID, ClientState> clientStates,
210                                                                            final Map<UUID, Set<TaskId>> warmups,
211                                                                            final AtomicInteger remainingWarmupReplicas,
212                                                                            final ConstrainedPrioritySet caughtUpClientsByTaskLoad,
213                                                                            final TaskMovement movement) {
214           final UUID caughtUpSourceClient = caughtUpClientsByTaskLoad.poll(movement.task);
215           if (caughtUpSourceClient != null) {
216               moveActiveAndTryToWarmUp(
217                       remainingWarmupReplicas,
218                       movement.task,
219                       clientStates.get(caughtUpSourceClient),
220                       clientStates.get(movement.destination),
221                       warmups.computeIfAbsent(movement.destination, x -> new TreeSet<>())
222               );
223               caughtUpClientsByTaskLoad.offerAll(asList(caughtUpSourceClient, movement.destination));
224               return true;
225           }
226           return false;
227       }

229       private static boolean tryToMoveActiveToMostCaughtUpClient(final Map<TaskId, SortedSet<UUID>> tasksToClientByLag,
230                                                                  final Map<UUID, ClientState> clientStates,
231                                                                  final Map<UUID, Set<TaskId>> warmups,
232                                                                  final AtomicInteger remainingWarmupReplicas,
233                                                                  final ConstrainedPrioritySet caughtUpClientsByTaskLoad,
234                                                                  final TaskMovement movement) {
235           final UUID mostCaughtUpSourceClient = mostCaughtUpEligibleClient(tasksToClientByLag, movement.task, movement.destination);
236           if (mostCaughtUpSourceClient != null) {
237               if (clientStates.get(mostCaughtUpSourceClient).hasStandbyTask(movement.task)) {
238                   swapStandbyAndActive(
239                           movement.task,
240                           clientStates.get(mostCaughtUpSourceClient),
241                           clientStates.get(movement.destination)
242                   );
243               } else {
244                   moveActiveAndTryToWarmUp(
245                           remainingWarmupReplicas,
246                           movement.task,
247                           clientStates.get(mostCaughtUpSourceClient),
248                           clientStates.get(movement.destination),
249                           warmups.computeIfAbsent(movement.destination, x -> new TreeSet<>())
250                   );
251               }
252               caughtUpClientsByTaskLoad.offerAll(asList(mostCaughtUpSourceClient, movement.destination));
253               return true;
254           }
255           return false;
256       }

258       private static void moveActiveAndTryToWarmUp(final AtomicInteger remainingWarmupReplicas,
259                                                    final TaskId task,
260                                                    final ClientState sourceClientState,
261                                                    final ClientState destinationClientState,
262                                                    final Set<TaskId> warmups) {
263           sourceClientState.assignActive(task);
264   
265           if (remainingWarmupReplicas.getAndDecrement() > 0) {
266               destinationClientState.unassignActive(task);
267               destinationClientState.assignStandby(task);
268               warmups.add(task);
269           } else {
270               // we have no more standbys or warmups to hand out, so we have to try and move it
271               // to the destination in a follow-on rebalance
272               destinationClientState.unassignActive(task);
273           }
274       }

276       private static void moveStandbyAndTryToWarmUp(final AtomicInteger remainingWarmupReplicas,
277                                                     final TaskId task,
278                                                     final ClientState sourceClientState,
279                                                     final ClientState destinationClientState) {
280           sourceClientState.assignStandby(task);
281   
282           if (remainingWarmupReplicas.getAndDecrement() > 0) {
283               // Then we can leave it also assigned to the destination as a warmup
284           } else {
285               // we have no more warmups to hand out, so we have to try and move it
286               // to the destination in a follow-on rebalance
287               destinationClientState.unassignStandby(task);
288           }
289       }

291       private static void swapStandbyAndActive(final TaskId task,
292                                                final ClientState sourceClientState,
293                                                final ClientState destinationClientState) {
294           sourceClientState.unassignStandby(task);
295           sourceClientState.assignActive(task);
296           destinationClientState.unassignActive(task);
297           destinationClientState.assignStandby(task);
298       }

300       private static UUID mostCaughtUpEligibleClient(final Map<TaskId, SortedSet<UUID>> tasksToClientByLag,
301                                                      final TaskId task,
302                                                      final UUID destinationClient) {
303           return mostCaughtUpEligibleClient(tasksToClientByLag, client -> true, task, destinationClient);
304       }
305   
306       private static UUID mostCaughtUpEligibleClient(final Map<TaskId, SortedSet<UUID>> tasksToClientByLag,
307                                                      final Function<UUID, Boolean> constraint,
308                                                      final TaskId task,
309                                                      final UUID destinationClient) {
310           for (final UUID client : tasksToClientByLag.get(task)) {
311               if (destinationClient.equals(client)) {
312                   break;
313               } else if (constraint.apply(client)) {
314                   return client;
315               }
316           }
317           return null;
318       }
