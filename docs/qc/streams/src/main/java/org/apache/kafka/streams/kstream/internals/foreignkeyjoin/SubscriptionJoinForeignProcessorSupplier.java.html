18   package org.apache.kafka.streams.kstream.internals.foreignkeyjoin;
19   
20   import org.apache.kafka.common.errors.UnsupportedVersionException;
21   import org.apache.kafka.streams.kstream.internals.Change;
22   import org.apache.kafka.streams.kstream.internals.KTableValueGetter;
23   import org.apache.kafka.streams.kstream.internals.KTableValueGetterSupplier;
24   import org.apache.kafka.streams.processor.api.ContextualProcessor;
25   import org.apache.kafka.streams.processor.api.Processor;
26   import org.apache.kafka.streams.processor.api.ProcessorContext;
27   import org.apache.kafka.streams.processor.api.ProcessorSupplier;
28   import org.apache.kafka.streams.processor.api.Record;
29   import org.apache.kafka.streams.state.ValueAndTimestamp;
30   
31   import java.util.Objects;
32   
33   /**
34    * Receives {@code SubscriptionWrapper<K>} events and processes them according to their Instruction.
35    * Depending on the results, {@code SubscriptionResponseWrapper}s are created, which will be propagated to
36    * the {@code SubscriptionResolverJoinProcessorSupplier} instance.
37    *
38    * @param <K> Type of primary keys
39    * @param <KO> Type of foreign key
40    * @param <VO> Type of foreign value
41    */
42   public class SubscriptionJoinForeignProcessorSupplier<K, KO, VO>
43       implements ProcessorSupplier<CombinedKey<KO, K>, Change<ValueAndTimestamp<SubscriptionWrapper<K>>>, K, SubscriptionResponseWrapper<VO>> {
44   
45       private final KTableValueGetterSupplier<KO, VO> foreignValueGetterSupplier;
46   
47       public SubscriptionJoinForeignProcessorSupplier(final KTableValueGetterSupplier<KO, VO> foreignValueGetterSupplier) {
48           this.foreignValueGetterSupplier = foreignValueGetterSupplier;
49       }
50   
51       @Override
52       public Processor<CombinedKey<KO, K>, Change<ValueAndTimestamp<SubscriptionWrapper<K>>>, K, SubscriptionResponseWrapper<VO>> get() {
53   
54           return new ContextualProcessor<CombinedKey<KO, K>, Change<ValueAndTimestamp<SubscriptionWrapper<K>>>, K, SubscriptionResponseWrapper<VO>>() {
55   
56               private KTableValueGetter<KO, VO> foreignValues;
57   
58               @Override
59               public void init(final ProcessorContext<K, SubscriptionResponseWrapper<VO>> context) {
60                   super.init(context);
61                   foreignValues = foreignValueGetterSupplier.get();
62                   foreignValues.init(context);
63               }
64   
65               @Override
66               public void process(final Record<CombinedKey<KO, K>, Change<ValueAndTimestamp<SubscriptionWrapper<K>>>> record) {
67                   Objects.requireNonNull(record.key(), "This processor should never see a null key.");
68                   Objects.requireNonNull(record.value(), "This processor should never see a null value.");
69                   final ValueAndTimestamp<SubscriptionWrapper<K>> valueAndTimestamp = record.value().newValue;
70                   Objects.requireNonNull(valueAndTimestamp, "This processor should never see a null newValue.");
71                   final SubscriptionWrapper<K> value = valueAndTimestamp.value();
72   
73                   if (value.getVersion() != SubscriptionWrapper.CURRENT_VERSION) {
74                       //Guard against modifications to SubscriptionWrapper. Need to ensure that there is compatibility
75                       //with previous versions to enable rolling upgrades. Must develop a strategy for upgrading
76                       //from older SubscriptionWrapper versions to newer versions.
77                       throw new UnsupportedVersionException("SubscriptionWrapper is of an incompatible version.");
78                   }
79   
80                   final ValueAndTimestamp<VO> foreignValueAndTime = foreignValues.get(record.key().getForeignKey());
81   
82                   final long resultTimestamp =
83                       foreignValueAndTime == null ?
84                           valueAndTimestamp.timestamp() :
85                           Math.max(valueAndTimestamp.timestamp(), foreignValueAndTime.timestamp());
86   
87                   switch (value.getInstruction()) {
88                       case DELETE_KEY_AND_PROPAGATE:
89                           context().forward(
90                               record.withKey(record.key().getPrimaryKey())
91                                   .withValue(new SubscriptionResponseWrapper<VO>(
92                                       value.getHash(),
93                                       null,
94                                       value.getPrimaryPartition()
95                                   ))
96                                   .withTimestamp(resultTimestamp)
97                           );
98                           break;
99                       case PROPAGATE_NULL_IF_NO_FK_VAL_AVAILABLE:
100                           //This one needs to go through regardless of LEFT or INNER join, since the extracted FK was
101                           //changed and there is no match for it. We must propagate the (key, null) to ensure that the
102                           //downstream consumers are alerted to this fact.
103                           final VO valueToSend = foreignValueAndTime == null ? null : foreignValueAndTime.value();
104   
105                           context().forward(
106                               record.withKey(record.key().getPrimaryKey())
107                                   .withValue(new SubscriptionResponseWrapper<>(value.getHash(), valueToSend, value.getPrimaryPartition()))
108                                   .withTimestamp(resultTimestamp)
109                           );
110                           break;
111                       case PROPAGATE_ONLY_IF_FK_VAL_AVAILABLE:
112                           if (foreignValueAndTime != null) {
113                               context().forward(
114                                   record.withKey(record.key().getPrimaryKey())
115                                      .withValue(new SubscriptionResponseWrapper<>(
116                                          value.getHash(),
117                                          foreignValueAndTime.value(),
118                                          value.getPrimaryPartition()
119                                      ))
120                                      .withTimestamp(resultTimestamp)
121                               );
122                           }
123                           break;
124                       case DELETE_KEY_NO_PROPAGATE:
125                           break;
126                       default:
127                           throw new IllegalStateException("Unhandled instruction: " + value.getInstruction());
128                   }
129               }
130           };
131       }
132   }
