17   package org.apache.kafka.streams.kstream;
18   
19   import java.time.Duration;
20   import org.apache.kafka.streams.KeyValue;
21   import org.apache.kafka.streams.errors.StreamsException;
22   import org.apache.kafka.streams.processor.ProcessorContext;
23   import org.apache.kafka.streams.processor.PunctuationType;
24   import org.apache.kafka.streams.processor.Punctuator;
25   import org.apache.kafka.streams.processor.StateStore;
26   import org.apache.kafka.streams.processor.To;
27   
28   /**
29    * The {@code ValueTransformerWithKey} interface for stateful mapping of a value to a new value (with possible new type).
30    * This is a stateful record-by-record operation, i.e, {@link #transform(Object, Object)} is invoked individually for each
31    * record of a stream and can access and modify a state that is available beyond a single call of
32    * {@link #transform(Object, Object)} (cf. {@link ValueMapper} for stateless value transformation).
33    * Additionally, this {@code ValueTransformerWithKey} can
34    * {@link ProcessorContext#schedule(Duration, PunctuationType, Punctuator) schedule} a method to be
35    * {@link Punctuator#punctuate(long) called periodically} with the provided context.
36    * Note that the key is read-only and should not be modified, as this can lead to corrupt partitioning.
37    * If {@code ValueTransformerWithKey} is applied to a {@link KeyValue} pair record the record's key is preserved.
38    * <p>
39    * Use {@link ValueTransformerWithKeySupplier} to provide new instances of {@link ValueTransformerWithKey} to
40    * Kafka Stream's runtime.
41    * <p>
42    * If a record's key and value should be modified {@link Transformer} can be used.
43    *
44    * @param <K>  key type
45    * @param <V>  value type
46    * @param <VR> transformed value type
47    * @see ValueTransformer
48    * @see ValueTransformerWithKeySupplier
49    * @see KStream#transformValues(ValueTransformerSupplier, String...)
50    * @see KStream#transformValues(ValueTransformerWithKeySupplier, String...)
51    * @see Transformer
52    */
53   
54   public interface ValueTransformerWithKey<K, V, VR> {
55   
56       /**
57        * Initialize this transformer.
58        * This is called once per instance when the topology gets initialized.
59        * <p>
60        * The provided {@link ProcessorContext context} can be used to access topology and record meta data, to
61        * {@link ProcessorContext#schedule(Duration, PunctuationType, Punctuator) schedule} a method to be
62        * {@link Punctuator#punctuate(long) called periodically} and to access attached {@link StateStore}s.
63        * <p>
64        * Note that {@link ProcessorContext} is updated in the background with the current record's meta data.
65        * Thus, it only contains valid record meta data when accessed within {@link #transform(Object, Object)}.
66        * <p>
67        * Note that using {@link ProcessorContext#forward(Object, Object)} or
68        * {@link ProcessorContext#forward(Object, Object, To)} is not allowed within any method of
69        * {@code ValueTransformerWithKey} and will result in an {@link StreamsException exception}.
70        *
71        * @param context the context
72        * @throws IllegalStateException If store gets registered after initialization is already finished
73        * @throws StreamsException if the store's change log does not contain the partition
74        */
75       void init(final ProcessorContext context);
76   
77       /**
78        * Transform the given [key and] value to a new value.
79        * Additionally, any {@link StateStore} that is {@link KStream#transformValues(ValueTransformerWithKeySupplier, String...)
80        * attached} to this operator can be accessed and modified arbitrarily (cf.
81        * {@link ProcessorContext#getStateStore(String)}).
82        * <p>
83        * Note that using {@link ProcessorContext#forward(Object, Object)} or
84        * {@link ProcessorContext#forward(Object, Object, To)} is not allowed within {@code transform} and
85        * will result in an {@link StreamsException exception}.
86        * <p>
87        * Note that if a {@code ValueTransformerWithKey} is used in a {@link KTable#transformValues(ValueTransformerWithKeySupplier, String...)}
88        * (or any other overload of {@code KTable#transformValues(...)}) operation,
89        * then the provided {@link ProcessorContext} from {@link #init(ProcessorContext)}
90        * does not guarantee that all context information will be available when {@code transform()}
91        * is executed, as it might be executed "out-of-band" due to some internal optimizations
92        * applied by the Kafka Streams DSL.
93        *
94        * @param readOnlyKey the read-only key
95        * @param value       the value to be transformed
96        * @return the new value
97        */
98       VR transform(final K readOnlyKey, final V value);
99   
100       /**
101        * Close this processor and clean up any resources.
102        * <p>
103        * It is not possible to return any new output records within {@code close()}.
104        * Using {@link ProcessorContext#forward(Object, Object)} or {@link ProcessorContext#forward(Object, Object, To)},
105        * will result in an {@link StreamsException exception}.
106        */
107       void close();
108   }
