44   /**
45    * Single-threaded executor class for the active tasks assigned to this thread.
46    */
47   public class TaskExecutor {
48   
49       private final Logger log;
50   
51       private final boolean hasNamedTopologies;
52       private final ProcessingMode processingMode;
53       private final Tasks tasks;
54       private final TaskExecutionMetadata taskExecutionMetadata;

56       public TaskExecutor(final Tasks tasks,
57                           final TaskExecutionMetadata taskExecutionMetadata,
58                           final ProcessingMode processingMode,
59                           final boolean hasNamedTopologies,
60                           final LogContext logContext) {
61           this.tasks = tasks;
62           this.taskExecutionMetadata = taskExecutionMetadata;
63           this.processingMode = processingMode;
64           this.hasNamedTopologies = hasNamedTopologies;
65           this.log = logContext.logger(getClass());
66       }

68       /**
69        * @throws TaskMigratedException if the task producer got fenced (EOS only)
70        * @throws StreamsException      if any task threw an exception while processing
71        */
72       int process(final int maxNumRecords, final Time time) {
73           int totalProcessed = 0;
74           Task lastProcessed = null;
75   
76           for (final Task task : tasks.activeTasks()) {
77               final long now = time.milliseconds();
78               try {
79                   if (taskExecutionMetadata.canProcessTask(task, now)) {
80                       lastProcessed = task;
81                       totalProcessed += processTask(task, maxNumRecords, now, time);
82                   }
83               } catch (final Throwable t) {
84                   taskExecutionMetadata.registerTaskError(task, t, now);
85                   tasks.removeTaskFromCuccessfullyProcessedBeforeClosing(lastProcessed);
86                   commitSuccessfullyProcessedTasks();
87                   throw t;
88               }
89           }
90   
91           return totalProcessed;
92       }

94       private long processTask(final Task task, final int maxNumRecords, final long begin, final Time time) {
95           int processed = 0;
96           long now = begin;
97   
98           final long then = now;
99           try {
100               while (processed < maxNumRecords && task.process(now)) {
101                   task.clearTaskTimeout();
102                   processed++;
103               }
104               // TODO: enable regardless of whether using named topologies
105               if (processed > 0 && hasNamedTopologies && processingMode != EXACTLY_ONCE_V2) {
106                   log.trace("Successfully processed task {}", task.id());
107                   tasks.addToSuccessfullyProcessed(task);
108               }
109           } catch (final TimeoutException timeoutException) {
110               // TODO consolidate TimeoutException retries with general error handling
111               task.maybeInitTaskTimeoutOrThrow(now, timeoutException);
112               log.error(
113                   String.format(
114                       "Could not complete processing records for %s due to the following exception; will move to next task and retry later",
115                       task.id()),
116                   timeoutException
117               );
118           } catch (final TaskMigratedException e) {
119               log.info("Failed to process stream task {} since it got migrated to another thread already. " +
120                   "Will trigger a new rebalance and close all tasks as zombies together.", task.id());
121               throw e;
122           } catch (final StreamsException e) {
123               log.error(String.format("Failed to process stream task %s due to the following error:", task.id()), e);
124               e.setTaskId(task.id());
125               throw e;
126           } catch (final RuntimeException e) {
127               log.error(String.format("Failed to process stream task %s due to the following error:", task.id()), e);
128               throw new StreamsException(e, task.id());
129           } finally {
130               now = time.milliseconds();
131               task.recordProcessBatchTime(now - then);
132           }
133           return processed;
134       }
135   
136       /**
137        * @throws TaskMigratedException if committing offsets failed (non-EOS)
138        *                               or if the task producer got fenced (EOS)
139        * @throws TimeoutException if committing offsets failed due to TimeoutException (non-EOS)
140        * @throws TaskCorruptedException if committing offsets failed due to TimeoutException (EOS)
141        * @param consumedOffsetsAndMetadata an empty map that will be filled in with the prepared offsets
142        * @return number of committed offsets, or -1 if we are in the middle of a rebalance and cannot commit
143        */
144       int commitTasksAndMaybeUpdateCommittableOffsets(final Collection<Task> tasksToCommit,
145                                                       final Map<Task, Map<TopicPartition, OffsetAndMetadata>> consumedOffsetsAndMetadata) {
146           int committed = 0;
147           for (final Task task : tasksToCommit) {
148               // we need to call commitNeeded first since we need to update committable offsets
149               if (task.commitNeeded()) {
150                   final Map<TopicPartition, OffsetAndMetadata> offsetAndMetadata = task.prepareCommit();
151                   if (!offsetAndMetadata.isEmpty()) {
152                       consumedOffsetsAndMetadata.put(task, offsetAndMetadata);
153                   }
154               }
155           }
156   
157           commitOffsetsOrTransaction(consumedOffsetsAndMetadata);
158   
159           for (final Task task : tasksToCommit) {
160               if (task.commitNeeded()) {
161                   task.clearTaskTimeout();
162                   ++committed;
163                   task.postCommit(false);
164               }
165           }
166           return committed;
167       }
168   
169       /**
170        * Caution: do not invoke this directly if it's possible a rebalance is occurring, as the commit will fail. If
171        * this is a possibility, prefer the {@link #commitTasksAndMaybeUpdateCommittableOffsets} instead.
172        *
173        * @throws TaskMigratedException   if committing offsets failed due to CommitFailedException (non-EOS)
174        * @throws TimeoutException        if committing offsets failed due to TimeoutException (non-EOS)
175        * @throws TaskCorruptedException  if committing offsets failed due to TimeoutException (EOS)
176        */
177       void commitOffsetsOrTransaction(final Map<Task, Map<TopicPartition, OffsetAndMetadata>> offsetsPerTask) {
178           log.debug("Committing task offsets {}", offsetsPerTask.entrySet().stream().collect(Collectors.toMap(t -> t.getKey().id(), Entry::getValue))); // avoid logging actual Task objects
179   
180           final Set<TaskId> corruptedTasks = new HashSet<>();
181   
182           if (!offsetsPerTask.isEmpty()) {
183               if (processingMode == EXACTLY_ONCE_ALPHA) {
184                   for (final Map.Entry<Task, Map<TopicPartition, OffsetAndMetadata>> taskToCommit : offsetsPerTask.entrySet()) {
185                       final Task task = taskToCommit.getKey();
186                       try {
187                           tasks.streamsProducerForTask(task.id())
188                               .commitTransaction(taskToCommit.getValue(), tasks.mainConsumer().groupMetadata());
189                           updateTaskCommitMetadata(taskToCommit.getValue());
190                       } catch (final TimeoutException timeoutException) {
191                           log.error(
192                               String.format("Committing task %s failed.", task.id()),
193                               timeoutException
194                           );
195                           corruptedTasks.add(task.id());
196                       }
197                   }
198               } else {
199                   final Map<TopicPartition, OffsetAndMetadata> allOffsets = offsetsPerTask.values().stream()
200                       .flatMap(e -> e.entrySet().stream()).collect(Collectors.toMap(Map.Entry::getKey, Map.Entry::getValue));
201   
202                   if (processingMode == EXACTLY_ONCE_V2) {
203                       try {
204                           tasks.threadProducer().commitTransaction(allOffsets, tasks.mainConsumer().groupMetadata());
205                           updateTaskCommitMetadata(allOffsets);
206                       } catch (final TimeoutException timeoutException) {
207                           log.error(
208                               String.format("Committing task(s) %s failed.",
209                                   offsetsPerTask
210                                       .keySet()
211                                       .stream()
212                                       .map(t -> t.id().toString())
213                                       .collect(Collectors.joining(", "))),
214                               timeoutException
215                           );
216                           offsetsPerTask
217                               .keySet()
218                               .forEach(task -> corruptedTasks.add(task.id()));
219                       }
220                   } else {
221                       try {
222                           tasks.mainConsumer().commitSync(allOffsets);
223                           updateTaskCommitMetadata(allOffsets);
224                       } catch (final CommitFailedException error) {
225                           throw new TaskMigratedException("Consumer committing offsets failed, " +
226                               "indicating the corresponding thread is no longer part of the group", error);
227                       } catch (final TimeoutException timeoutException) {
228                           log.error(
229                               String.format("Committing task(s) %s failed.",
230                                   offsetsPerTask
231                                       .keySet()
232                                       .stream()
233                                       .map(t -> t.id().toString())
234                                       .collect(Collectors.joining(", "))),
235                               timeoutException
236                           );
237                           throw timeoutException;
238                       } catch (final KafkaException error) {
239                           throw new StreamsException("Error encountered committing offsets via consumer", error);
240                       }
241                   }
242               }
243   
244               if (!corruptedTasks.isEmpty()) {
245                   throw new TaskCorruptedException(corruptedTasks);
246               }
247           }
248       }

250       private void updateTaskCommitMetadata(final Map<TopicPartition, OffsetAndMetadata> allOffsets) {
251           for (final Task task: tasks.activeTasks()) {
252               if (task instanceof StreamTask) {
253                   for (final TopicPartition topicPartition : task.inputPartitions()) {
254                       if (allOffsets.containsKey(topicPartition)) {
255                           ((StreamTask) task).updateCommittedOffsets(topicPartition, allOffsets.get(topicPartition).offset());
256                       }
257                   }
258               }
259           }
260       }

262       private void commitSuccessfullyProcessedTasks() {
263           if (!tasks.successfullyProcessed().isEmpty()) {
264               log.info("Streams encountered an error when processing tasks." +
265                   " Will commit all previously successfully processed tasks {}",
266                   tasks.successfullyProcessed().stream().map(Task::id));
267               commitTasksAndMaybeUpdateCommittableOffsets(tasks.successfullyProcessed(), new HashMap<>());
268           }
269           tasks.clearSuccessfullyProcessed();
270       }

272       /**
273        * @throws TaskMigratedException if the task producer got fenced (EOS only)
274        */
275       int punctuate() {
276           int punctuated = 0;
277   
278           for (final Task task : tasks.notPausedActiveTasks()) {
279               try {
280                   if (task.maybePunctuateStreamTime()) {
281                       punctuated++;
282                   }
283                   if (task.maybePunctuateSystemTime()) {
284                       punctuated++;
285                   }
286               } catch (final TaskMigratedException e) {
287                   log.info("Failed to punctuate stream task {} since it got migrated to another thread already. " +
288                       "Will trigger a new rebalance and close all tasks as zombies together.", task.id());
289                   throw e;
290               } catch (final StreamsException e) {
291                   log.error("Failed to punctuate stream task {} due to the following error:", task.id(), e);
292                   e.setTaskId(task.id());
293                   throw e;
294               } catch (final KafkaException e) {
295                   log.error("Failed to punctuate stream task {} due to the following error:", task.id(), e);
296                   throw new StreamsException(e, task.id());
297               }
298           }
299   
300           return punctuated;
301       }
