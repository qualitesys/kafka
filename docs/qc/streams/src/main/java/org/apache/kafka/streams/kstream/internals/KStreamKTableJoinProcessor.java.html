17   package org.apache.kafka.streams.kstream.internals;
18   
19   import org.apache.kafka.common.metrics.Sensor;
20   import org.apache.kafka.streams.kstream.KeyValueMapper;
21   import org.apache.kafka.streams.kstream.ValueJoinerWithKey;
22   import org.apache.kafka.streams.processor.api.ContextualProcessor;
23   import org.apache.kafka.streams.processor.api.ProcessorContext;
24   import org.apache.kafka.streams.processor.api.Record;
25   import org.apache.kafka.streams.processor.api.RecordMetadata;
26   import org.apache.kafka.streams.processor.internals.metrics.StreamsMetricsImpl;
27   import org.slf4j.Logger;
28   import org.slf4j.LoggerFactory;
29   
30   import static org.apache.kafka.streams.processor.internals.metrics.TaskMetrics.droppedRecordsSensor;
31   import static org.apache.kafka.streams.state.ValueAndTimestamp.getValueOrNull;
32   
33   class KStreamKTableJoinProcessor<K1, K2, V1, V2, VOut> extends ContextualProcessor<K1, V1, K1, VOut> {
34       private static final Logger LOG = LoggerFactory.getLogger(KStreamKTableJoin.class);
35   
36       private final KTableValueGetter<K2, V2> valueGetter;
37       private final KeyValueMapper<? super K1, ? super V1, ? extends K2> keyMapper;
38       private final ValueJoinerWithKey<? super K1, ? super V1, ? super V2, ? extends VOut> joiner;
39       private final boolean leftJoin;
40       private Sensor droppedRecordsSensor;
41   
42       KStreamKTableJoinProcessor(final KTableValueGetter<K2, V2> valueGetter,
43                                  final KeyValueMapper<? super K1, ? super V1, ? extends K2> keyMapper,
44                                  final ValueJoinerWithKey<? super K1, ? super V1, ? super V2, ? extends VOut> joiner,
45                                  final boolean leftJoin) {
46           this.valueGetter = valueGetter;
47           this.keyMapper = keyMapper;
48           this.joiner = joiner;
49           this.leftJoin = leftJoin;
50       }
51   
52       @Override
53       public void init(final ProcessorContext<K1, VOut> context) {
54           super.init(context);
55           final StreamsMetricsImpl metrics = (StreamsMetricsImpl) context.metrics();
56           droppedRecordsSensor = droppedRecordsSensor(Thread.currentThread().getName(), context.taskId().toString(), metrics);
57           valueGetter.init(context);
58       }
59   
60       @Override
61       public void process(final Record<K1, V1> record) {
62           // we do join iff the join keys are equal, thus, if {@code keyMapper} returns {@code null} we
63           // cannot join and just ignore the record. Note for KTables, this is the same as having a null key
64           // since keyMapper just returns the key, but for GlobalKTables we can have other keyMappers
65           //
66           // we also ignore the record if value is null, because in a key-value data model a null-value indicates
67           // an empty message (ie, there is nothing to be joined) -- this contrast SQL NULL semantics
68           // furthermore, on left/outer joins 'null' in ValueJoiner#apply() indicates a missing record --
69           // thus, to be consistent and to avoid ambiguous null semantics, null values are ignored
70           final K2 mappedKey = keyMapper.apply(record.key(), record.value());
71           if (mappedKey == null || record.value() == null) {
72               if (context().recordMetadata().isPresent()) {
73                   final RecordMetadata recordMetadata = context().recordMetadata().get();
74                   LOG.warn(
75                       "Skipping record due to null join key or value. "
76                           + "topic=[{}] partition=[{}] offset=[{}]",
77                       recordMetadata.topic(), recordMetadata.partition(), recordMetadata.offset()
78                   );
79               } else {
80                   LOG.warn(
81                       "Skipping record due to null join key or value. Topic, partition, and offset not known."
82                   );
83               }
84               droppedRecordsSensor.record();
85           } else {
86               final V2 value2 = getValueOrNull(valueGetter.get(mappedKey));
87               if (leftJoin || value2 != null) {
88                   context().forward(record.withValue(joiner.apply(record.key(), record.value(), value2)));
89               }
90           }
91       }
92   
93       @Override
94       public void close() {
95           valueGetter.close();
96       }
97   }
