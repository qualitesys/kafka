98       private KTable<Windowed<K>, Long> doCount(final Named named,
99                                                 final Materialized<K, Long, WindowStore<Bytes, byte[]>> materialized) {
100           final MaterializedInternal<K, Long, WindowStore<Bytes, byte[]>> materializedInternal =
101               new MaterializedInternal<>(materialized, builder, AGGREGATE_NAME);
102   
103           if (materializedInternal.keySerde() == null) {
104               materializedInternal.withKeySerde(keySerde);
105           }
106           if (materializedInternal.valueSerde() == null) {
107               materializedInternal.withValueSerde(Serdes.Long());
108           }
109   
110           final String aggregateName = new NamedInternal(named).orElseGenerateWithPrefix(builder, AGGREGATE_NAME);
111   
112           return aggregateBuilder.build(
113                   new NamedInternal(aggregateName),
114                   materialize(materializedInternal),
115                   new KStreamWindowAggregate<>(windows, materializedInternal.storeName(), emitStrategy, aggregateBuilder.countInitializer, aggregateBuilder.countAggregator),
116                   materializedInternal.queryableStoreName(),
117                   materializedInternal.keySerde() != null ? new FullTimeWindowedSerde<>(materializedInternal.keySerde(), windows.size()) : null,
118                   materializedInternal.valueSerde());
119   
120   
121       }

191           Objects.requireNonNull(reducer, "reducer can't be null");

193           Objects.requireNonNull(materialized, "materialized can't be null");
194

202               materializedInternal.withValueSerde(valueSerde);

204   
205           final String reduceName = new NamedInternal(named).orElseGenerateWithPrefix(builder, REDUCE_NAME);
206   
207           return aggregateBuilder.build(
208                   new NamedInternal(reduceName),
209                   materialize(materializedInternal),

220               throw new IllegalArgumentException("ON_WINDOW_CLOSE emit strategy cannot be used for UnlimitedWindows");

222           this.emitStrategy = emitStrategy;
223           return this;

226       private <VR> StoreBuilder<TimestampedWindowStore<K, VR>> materialize(final MaterializedInternal<K, VR, WindowStore<Bytes, byte[]>> materialized) {
227           WindowBytesStoreSupplier supplier = (WindowBytesStoreSupplier) materialized.storeSupplier();
228           if (supplier == null) {
229               final long retentionPeriod = materialized.retention() != null ?
230                   materialized.retention().toMillis() : windows.size() + windows.gracePeriodMs();
231   
232               if ((windows.size() + windows.gracePeriodMs()) > retentionPeriod) {
233                   throw new IllegalArgumentException("The retention period of the window store "
234                           + name + " must be no smaller than its window size plus the grace period."
235                           + " Got size=[" + windows.size() + "],"
236                           + " grace=[" + windows.gracePeriodMs() + "],"
237                           + " retention=[" + retentionPeriod + "]");
238               }
239   
240               switch (materialized.storeType()) {
241                   case IN_MEMORY:
242                       supplier = Stores.inMemoryWindowStore(
243                           materialized.storeName(),
244                           Duration.ofMillis(retentionPeriod),
245                           Duration.ofMillis(windows.size()),
246                           false
247                       );
248                       break;
249                   case ROCKS_DB:
250                       supplier = emitStrategy.type() == StrategyType.ON_WINDOW_CLOSE ?
251                           RocksDbIndexedTimeOrderedWindowBytesStoreSupplier.create(
252                               materialized.storeName(),
253                               Duration.ofMillis(retentionPeriod),
254                               Duration.ofMillis(windows.size()),
255                               false,
256                               false
257                           ) :
258                           Stores.persistentTimestampedWindowStore(
259                               materialized.storeName(),
260                               Duration.ofMillis(retentionPeriod),
261                               Duration.ofMillis(windows.size()),
262                               false
263                       );
264                       break;
265                   default:
266                       throw new IllegalStateException("Unknown store type: " + materialized.storeType());
267               }
268           }
269   
270           final StoreBuilder<TimestampedWindowStore<K, VR>> builder = Stores.timestampedWindowStoreBuilder(
271               supplier,
272               materialized.keySerde(),
273               materialized.valueSerde()
274           );
275   
276           if (materialized.loggingEnabled()) {
277               builder.withLoggingEnabled(materialized.logConfig());
278           } else {
279               builder.withLoggingDisabled();
280           }
281   
282           if (materialized.cachingEnabled()) {
283               builder.withCachingEnabled();
284           }
285           return builder;
286       }

288       private Aggregator<K, V, V> aggregatorForReducer(final Reducer<V> reducer) {
289           return (aggKey, value, aggregate) -> aggregate == null ? value : reducer.apply(aggregate, value);
290       }
