37   /**
38    * A RocksDB backed time-ordered segmented bytes store for session key schema.
39    */
40   public class RocksDBTimeOrderedSessionSegmentedBytesStore extends AbstractRocksDBTimeOrderedSegmentedBytesStore {
41   
42       private class SessionKeySchemaIndexToBaseStoreIterator extends IndexToBaseStoreIterator {
43           SessionKeySchemaIndexToBaseStoreIterator(final KeyValueIterator<Bytes, byte[]> indexIterator) {
44               super(indexIterator);
45           }
46   
47           @Override
48           protected Bytes getBaseKey(final Bytes indexKey) {
49               final Window window = KeyFirstSessionKeySchema.extractWindow(indexKey.get());
50               final byte[] key = KeyFirstSessionKeySchema.extractKeyBytes(indexKey.get());
51   
52               return TimeFirstSessionKeySchema.toBinary(Bytes.wrap(key), window.start(), window.end());
53           }
54       }
55   
56       RocksDBTimeOrderedSessionSegmentedBytesStore(final String name,
57                                                    final String metricsScope,
58                                                    final long retention,
59                                                    final long segmentInterval,
60                                                    final boolean withIndex) {
61           super(name, metricsScope, retention, segmentInterval, new TimeFirstSessionKeySchema(),
62               Optional.ofNullable(withIndex ? new KeyFirstSessionKeySchema() : null));
63       }

65       public byte[] fetchSession(final Bytes key,
66                                  final long earliestSessionEndTime,
67                                  final long latestSessionStartTime) {
68           return get(TimeFirstSessionKeySchema.toBinary(
69               key,
70               earliestSessionEndTime,
71               latestSessionStartTime
72           ));
73       }
74   
75       public KeyValueIterator<Bytes, byte[]> fetchSessions(final long earliestSessionEndTime,
76                                                            final long latestSessionEndTime) {
77           final List<KeyValueSegment> searchSpace = segments.segments(earliestSessionEndTime, latestSessionEndTime, true);
78   
79           // here we want [0, latestSE, FF] as the upper bound to cover any possible keys,
80           // but since we can only get upper bound based on timestamps, we use a slight larger upper bound as [0, latestSE+1]
81           final Bytes binaryFrom = baseKeySchema.lowerRangeFixedSize(null, earliestSessionEndTime);
82           final Bytes binaryTo = baseKeySchema.lowerRangeFixedSize(null, latestSessionEndTime + 1);
83   
84           return new SegmentIterator<>(
85                   searchSpace.iterator(),
86                   iterator -> {
87                       while (iterator.hasNext()) {
88                           final Bytes bytes = iterator.peekNextKey();
89   
90                           final Windowed<Bytes> windowedKey = TimeFirstSessionKeySchema.from(bytes);
91                           final long endTime = windowedKey.window().end();
92   
93                           if (endTime <= latestSessionEndTime && endTime >= earliestSessionEndTime) {
94                               return true;
95                           }
96                           iterator.next();
97                       }
98                       return false;
99                   },
100                   binaryFrom,
101                   binaryTo,
102                   true);
103       }
104   
105       public void remove(final Windowed<Bytes> key) {
106           remove(TimeFirstSessionKeySchema.toBinary(key));
107       }
108   
109       public void put(final Windowed<Bytes> sessionKey, final byte[] aggregate) {
110           put(TimeFirstSessionKeySchema.toBinary(sessionKey), aggregate);
111       }

113       @Override
114       protected KeyValue<Bytes, byte[]> getIndexKeyValue(final Bytes baseKey, final byte[] baseValue) {
115           final Window window = TimeFirstSessionKeySchema.extractWindow(baseKey.get());
116           final byte[] key = TimeFirstSessionKeySchema.extractKeyBytes(baseKey.get());
117           return KeyValue.pair(KeyFirstSessionKeySchema.toBinary(Bytes.wrap(key), window.start(), window.end()), new byte[0]);
118       }
119   
120       @Override
121       Map<KeyValueSegment, WriteBatch> getWriteBatches(
122           final Collection<ConsumerRecord<byte[], byte[]>> records) {
123           // advance stream time to the max timestamp in the batch
124           for (final ConsumerRecord<byte[], byte[]> record : records) {
125               final long timestamp = SessionKeySchema.extractEndTimestamp(record.key());
126               observedStreamTime = Math.max(observedStreamTime, timestamp);
127           }
128   
129           final Map<KeyValueSegment, WriteBatch> writeBatchMap = new HashMap<>();
130           for (final ConsumerRecord<byte[], byte[]> record : records) {
131               final long timestamp = SessionKeySchema.extractEndTimestamp(record.key());
132               final long segmentId = segments.segmentId(timestamp);
133               final KeyValueSegment segment = segments.getOrCreateSegmentIfLive(segmentId, context, observedStreamTime);
134               if (segment != null) {
135                   ChangelogRecordDeserializationHelper.applyChecksAndUpdatePosition(
136                       record,
137                       consistencyEnabled,
138                       position
139                   );
140                   try {
141                       final WriteBatch batch = writeBatchMap.computeIfAbsent(segment, s -> new WriteBatch());
142   
143                       // Assuming changelog record is serialized using SessionKeySchema
144                       // from ChangeLoggingSessionBytesStore. Reconstruct key/value to restore
145                       if (hasIndex()) {
146                           final byte[] indexKey = KeyFirstSessionKeySchema.prefixNonPrefixSessionKey(record.key());
147                           // Take care of tombstone
148                           final byte[] value = record.value() == null ? null : new byte[0];
149                           segment.addToBatch(new KeyValue<>(indexKey, value), batch);
150                       }
151   
152                       final byte[] baseKey = TimeFirstSessionKeySchema.extractWindowBytesFromNonPrefixSessionKey(record.key());
153                       segment.addToBatch(new KeyValue<>(baseKey, record.value()), batch);
154                   } catch (final RocksDBException e) {
155                       throw new ProcessorStateException("Error restoring batch to store " + name(), e);
156                   }
157               }
158           }
159           return writeBatchMap;
160       }
161   
162       @Override
163       protected IndexToBaseStoreIterator getIndexToBaseStoreIterator(
164           final SegmentIterator<KeyValueSegment> segmentIterator) {
165           return new SessionKeySchemaIndexToBaseStoreIterator(segmentIterator);
166       }
