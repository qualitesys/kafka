102       /**
103        * Schedule a periodic operation for processors. A processor may call this method during
104        * {@link Processor#init(ProcessorContext) initialization} or
105        * {@link Processor#process(Object, Object) processing} to
106        * schedule a periodic callback &mdash; called a punctuation &mdash; to {@link Punctuator#punctuate(long)}.
107        * The type parameter controls what notion of time is used for punctuation:
108        * <ul>
109        *   <li>{@link PunctuationType#STREAM_TIME} &mdash; uses "stream time", which is advanced by the processing of messages
110        *   in accordance with the timestamp as extracted by the {@link TimestampExtractor} in use.
111        *   The first punctuation will be triggered by the first record that is processed.
112        *   <b>NOTE:</b> Only advanced if messages arrive</li>
113        *   <li>{@link PunctuationType#WALL_CLOCK_TIME} &mdash; uses system time (the wall-clock time),
114        *   which is advanced independent of whether new messages arrive.
115        *   The first punctuation will be triggered after interval has elapsed.
116        *   <b>NOTE:</b> This is best effort only as its granularity is limited by how long an iteration of the
117        *   processing loop takes to complete</li>
118        * </ul>
119        *
120        * <b>Skipping punctuations:</b> Punctuations will not be triggered more than once at any given timestamp.
121        * This means that "missed" punctuation will be skipped.
122        * It's possible to "miss" a punctuation if:
123        * <ul>
124        *   <li>with {@link PunctuationType#STREAM_TIME}, when stream time advances more than interval</li>
125        *   <li>with {@link PunctuationType#WALL_CLOCK_TIME}, on GC pause, too short interval, ...</li>
126        * </ul>
127        *
128        * @param interval the time interval between punctuations (supported minimum is 1 millisecond)
129        * @param type one of: {@link PunctuationType#STREAM_TIME}, {@link PunctuationType#WALL_CLOCK_TIME}
130        * @param callback a function consuming timestamps representing the current stream or system time
131        * @return a handle allowing cancellation of the punctuation schedule established by this method
132        * @throws IllegalArgumentException if the interval is not representable in milliseconds
133        */
134       Cancellable schedule(final Duration interval,
135                            final PunctuationType type,
136                            final Punctuator callback);

138       /**
139        * Forward a key/value pair to all downstream processors.
140        * Used the input record's timestamp as timestamp for the output record.
141        *
142        * <p> If this method is called with {@link Punctuator#punctuate(long)} the record that
143        * is sent downstream won't have any associated record metadata like topic, partition, or offset.
144        *
145        * @param key key
146        * @param value value
147        */
148       <K, V> void forward(final K key, final V value);
149

151        * Forward a key/value pair to the specified downstream processors.
152        * Can be used to set the timestamp of the output record.
153        *
154        * <p> If this method is called with {@link Punctuator#punctuate(long)} the record that
155        * is sent downstream won't have any associated record metadata like topic, partition, or offset.
156        *
157        * @param key key
158        * @param value value
159        * @param to the options to use when forwarding
160        */

172        * <p> For example, if this method is invoked within a {@link Punctuator#punctuate(long)

177        * (and siblings), that do not always guarantee to provide a valid topic name, as they might be

232       /**
233        * Return the current timestamp.
234        *
235        * <p> If it is triggered while processing a record streamed from the source processor,
236        * timestamp is defined as the timestamp of the current input record; the timestamp is extracted from
237        * {@link org.apache.kafka.clients.consumer.ConsumerRecord ConsumerRecord} by {@link TimestampExtractor}.
238        * Note, that an upstream {@link Processor} might have set a new timestamp by calling
239        * {@link ProcessorContext#forward(Object, Object, To) forward(..., To.all().withTimestamp(...))}.
240        * In particular, some Kafka Streams DSL operators set result record timestamps explicitly,
241        * to guarantee deterministic results.
242        *
243        * <p> If it is triggered while processing a record generated not from the source processor (for example,
244        * if this method is invoked from the punctuate call), timestamp is defined as the current
245        * task's stream time, which is defined as the largest timestamp of any record processed by the task.
246        *
247        * @return the timestamp
248        */
249       long timestamp();

284        * @return the current system timestamp in milliseconds
285        */
286       long currentSystemTimeMs();
287   
288       /**
289        * Return the current stream-time in milliseconds.
290        *
291        * <p> Stream-time is the maximum observed {@link TimestampExtractor record timestamp} so far
292        * (including the currently processed record), i.e., it can be considered a high-watermark.
293        * Stream-time is tracked on a per-task basis and is preserved across restarts and during task migration.
294        *
295        * <p> Note: this method is not supported for global processors (cf. {@link Topology#addGlobalStore} (...)
296        * and {@link StreamsBuilder#addGlobalStore} (...),
297        * because there is no concept of stream-time for this case.
298        * Calling this method in a global processor will result in an {@link UnsupportedOperationException}.
299        *
300        * @return the current stream-time in milliseconds
301        */
302       long currentStreamTimeMs();
