85       private class KStreamWindowAggregateProcessor extends AbstractKStreamTimeWindowAggregateProcessor<KIn, VIn, VAgg> {
86           protected KStreamWindowAggregateProcessor(final String storeName, final EmitStrategy emitStrategy, final boolean sendOldValues) {
87               super(storeName, emitStrategy, sendOldValues);
88           }
89   
90           @Override
91           public void process(final Record<KIn, VIn> record) {
92               if (record.key() == null) {
93                   if (context().recordMetadata().isPresent()) {
94                       final RecordMetadata recordMetadata = context().recordMetadata().get();
95                       log.warn(
96                           "Skipping record due to null key. "
97                               + "topic=[{}] partition=[{}] offset=[{}]",
98                           recordMetadata.topic(), recordMetadata.partition(), recordMetadata.offset()
99                       );
100                   } else {
101                       log.warn(
102                           "Skipping record due to null key. Topic, partition, and offset not known."
103                       );
104                   }
105                   droppedRecordsSensor.record();
106                   return;
107               }
108   
109               // first get the matching windows
110               final long timestamp = record.timestamp();
111               updateObservedStreamTime(timestamp);
112               final long windowCloseTime = observedStreamTime - windows.gracePeriodMs();
113   
114               final Map<Long, W> matchedWindows = windows.windowsFor(timestamp);
115   
116               // try update the window whose end time is still larger than the window close time,
117               // and create the new window for the rest of unmatched window that do not exist yet;
118               for (final Map.Entry<Long, W> entry : matchedWindows.entrySet()) {
119                   final Long windowStart = entry.getKey();
120                   final long windowEnd = entry.getValue().end();
121                   if (windowEnd > windowCloseTime) {
122                       final ValueAndTimestamp<VAgg> oldAggAndTimestamp = windowStore.fetch(record.key(), windowStart);
123                       VAgg oldAgg = getValueOrNull(oldAggAndTimestamp);
124   
125                       final VAgg newAgg;
126                       final long newTimestamp;
127   
128                       if (oldAgg == null) {
129                           oldAgg = initializer.apply();
130                           newTimestamp = record.timestamp();
131                       } else {
132                           newTimestamp = Math.max(record.timestamp(), oldAggAndTimestamp.timestamp());
133                       }
134   
135                       newAgg = aggregator.apply(record.key(), record.value(), oldAgg);
136   
137                       // update the store with the new value
138                       windowStore.put(record.key(), ValueAndTimestamp.make(newAgg, newTimestamp), windowStart);
139                       maybeForwardUpdate(record, entry.getValue(), oldAgg, newAgg, newTimestamp);
140                   } else {
141                       final String windowString = "[" + windowStart + "," + windowEnd + ")";
142                       logSkippedRecordForExpiredWindow(log, record.timestamp(), windowCloseTime, windowString);
143                   }
144               }
145   
146               maybeForwardFinalResult(record, windowCloseTime);
147           }
148   
149           @Override
150           protected long emitRangeLowerBound(final long windowCloseTime) {
151               // Since time window end timestamp is exclusive, we set the inclusive lower bound plus 1;
152               // Set lower bound to 0L for the first time emit so that when we fetchAll, we fetch from 0L
153               return lastEmitWindowCloseTime == ConsumerRecord.NO_TIMESTAMP ?
154                   0L : Math.max(0L, lastEmitWindowCloseTime - windows.size()) + 1;
155           }
156   
157           @Override
158           protected long emitRangeUpperBound(final long windowCloseTime) {
159               return windowCloseTime - windows.size();
160           }
161   
162           @Override
163           protected boolean shouldRangeFetch(final long emitRangeLowerBound, final long emitRangeUpperBound) {
164               // Don't fetch store if the new emit window close time doesn't
165               // progress enough to cover next window;
166               // Note since window-end time is exclusive we would not count windows whose end time == lower bound, hence
167               // would minus 1 when finding matched windows
168               if (lastEmitWindowCloseTime != ConsumerRecord.NO_TIMESTAMP) {
169                   final Map<Long, W> matchedCloseWindows = windows.windowsFor(emitRangeUpperBound);
170                   final Map<Long, W> matchedEmitWindows = windows.windowsFor(emitRangeLowerBound - 1);
171   
172                   if (matchedCloseWindows.equals(matchedEmitWindows)) {
173                       log.trace("No new windows to emit. LastEmitCloseTime={}, emitRangeLowerBound={}, emitRangeUpperBound={}",
174                               lastEmitWindowCloseTime, emitRangeLowerBound, emitRangeUpperBound);
175                       return false;
176                   }
177               }
178   
179               return true;
180           }
