17   package org.apache.kafka.streams.processor;
18   
19   import org.apache.kafka.clients.consumer.ConsumerRecord;
20   
21   /**
22    * Retrieves current wall clock timestamps as {@link System#currentTimeMillis()}.
23    * <p>
24    * Using this extractor effectively provides <i>processing-time</i> semantics.
25    * <p>
26    * If you need <i>event-time</i> semantics, use {@link FailOnInvalidTimestamp} with
27    * built-in <i>CreateTime</i> or <i>LogAppendTime</i> timestamp (see KIP-32: Add timestamps to Kafka message for details).
28    *
29    * @see FailOnInvalidTimestamp
30    * @see LogAndSkipOnInvalidTimestamp
31    * @see UsePartitionTimeOnInvalidTimestamp
32    */
33   public class WallclockTimestampExtractor implements TimestampExtractor {
34   
35       /**
36        * Return the current wall clock time as timestamp.
37        *
38        * @param record a data record
39        * @param partitionTime the highest extracted valid timestamp of the current record's partition˙ (could be -1 if unknown)
40        * @return the current wall clock time, expressed in milliseconds since midnight, January 1, 1970 UTC
41        */
42       @Override
43       public long extract(final ConsumerRecord<Object, Object> record, final long partitionTime) {
44           return System.currentTimeMillis();
45       }
46   }
