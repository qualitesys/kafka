36   /**
37    * {@code KTable} is an abstraction of a <i>changelog stream</i> from a primary-keyed table.
38    * Each record in this changelog stream is an update on the primary-keyed table with the record key as the primary key.
39    * <p>
40    * A {@code KTable} is either {@link StreamsBuilder#table(String) defined from a single Kafka topic} that is
41    * consumed message by message or the result of a {@code KTable} transformation.
42    * An aggregation of a {@link KStream} also yields a {@code KTable}.
43    * <p>
44    * A {@code KTable} can be transformed record by record, joined with another {@code KTable} or {@link KStream}, or
45    * can be re-partitioned and aggregated into a new {@code KTable}.
46    * <p>
47    * Some {@code KTable}s have an internal state (a {@link ReadOnlyKeyValueStore}) and are therefore queryable via the
48    * interactive queries API.
49    * For example:
50    * <pre>{@code
51    *     final KTable table = ...
52    *     ...
53    *     final KafkaStreams streams = ...;
54    *     streams.start()
55    *     ...
56    *     final String queryableStoreName = table.queryableStoreName(); // returns null if KTable is not queryable
57    *     ReadOnlyKeyValueStore view = streams.store(queryableStoreName, QueryableStoreTypes.timestampedKeyValueStore());
58    *     view.get(key);
59    *}</pre>
60    *<p>
61    * Records from the source topic that have null keys are dropped.
62    *
63    * @param <K> Type of primary keys
64    * @param <V> Type of value changes
65    * @see KStream
66    * @see KGroupedTable
67    * @see GlobalKTable
68    * @see StreamsBuilder#table(String)
69    */

72       /**
73        * Create a new {@code KTable} that consists of all records of this {@code KTable} which satisfy the given
74        * predicate, with default serializers, deserializers, and state store.
75        * All records that do not satisfy the predicate are dropped.
76        * For each {@code KTable} update, the filter is evaluated based on the current update
77        * record and then an update record is produced for the result {@code KTable}.
78        * This is a stateless record-by-record operation.
79        * <p>
80        * Note that {@code filter} for a <i>changelog stream</i> works differently than {@link KStream#filter(Predicate)
81        * record stream filters}, because {@link KeyValue records} with {@code null} values (so-called tombstone records)
82        * have delete semantics.
83        * Thus, for tombstones the provided filter predicate is not evaluated but the tombstone record is forwarded
84        * directly if required (i.e., if there is anything to be deleted).
85        * Furthermore, for each record that gets dropped (i.e., does not satisfy the given predicate) a tombstone record
86        * is forwarded.
87        *
88        * @param predicate a filter {@link Predicate} that is applied to each record
89        * @return a {@code KTable} that contains only those records that satisfy the given predicate
90        * @see #filterNot(Predicate)
91        */
92       KTable<K, V> filter(final Predicate<? super K, ? super V> predicate);
