18   
19   import org.apache.kafka.common.serialization.Serde;
20   import org.apache.kafka.streams.kstream.internals.WindowedSerializer;
21   import org.apache.kafka.streams.kstream.internals.WindowedStreamPartitioner;
22   import org.apache.kafka.streams.processor.StreamPartitioner;
23   
24   /**
25    * This class is used to provide the optional parameters for internal repartition topics.
26    *
27    * @param <K> key type
28    * @param <V> value type
29    * @see KStream#repartition()
30    * @see KStream#repartition(Repartitioned)
31    */
32   public class Repartitioned<K, V> implements NamedOperation<Repartitioned<K, V>> {
33   
34       protected final String name;
35       protected final Serde<K> keySerde;
36       protected final Serde<V> valueSerde;
37       protected final Integer numberOfPartitions;
38       protected final StreamPartitioner<K, V> partitioner;
39   
40       private Repartitioned(final String name,
41                             final Serde<K> keySerde,
42                             final Serde<V> valueSerde,
43                             final Integer numberOfPartitions,
44                             final StreamPartitioner<K, V> partitioner) {
45           this.name = name;
46           this.keySerde = keySerde;
47           this.valueSerde = valueSerde;
48           this.numberOfPartitions = numberOfPartitions;
49           this.partitioner = partitioner;
50       }
51   
52       protected Repartitioned(final Repartitioned<K, V> repartitioned) {
53           this(
54               repartitioned.name,
55               repartitioned.keySerde,
56               repartitioned.valueSerde,
57               repartitioned.numberOfPartitions,
58               repartitioned.partitioner
59           );
60       }
61   
62       /**
63        * Create a {@code Repartitioned} instance with the provided name used as part of the repartition topic.
64        *
65        * @param name the name used as a processor named and part of the repartition topic name.
66        * @param <K>  key type
67        * @param <V>  value type
68        * @return A new {@code Repartitioned} instance configured with processor name and repartition topic name
69        * @see KStream#repartition(Repartitioned)
70        */
71       public static <K, V> Repartitioned<K, V> as(final String name) {
72           return new Repartitioned<>(name, null, null, null, null);
73       }
74   
75       /**
76        * Create a {@code Repartitioned} instance with provided key serde and value serde.
77        *
78        * @param keySerde   Serde to use for serializing the key
79        * @param valueSerde Serde to use for serializing the value
80        * @param <K>        key type
81        * @param <V>        value type
82        * @return A new {@code Repartitioned} instance configured with key serde and value serde
83        * @see KStream#repartition(Repartitioned)
84        */
85       public static <K, V> Repartitioned<K, V> with(final Serde<K> keySerde,
86                                                     final Serde<V> valueSerde) {
87           return new Repartitioned<>(null, keySerde, valueSerde, null, null);
88       }
89   
90       /**
91        * Create a {@code Repartitioned} instance with provided partitioner.
92        *
93        * @param partitioner the function used to determine how records are distributed among partitions of the topic,
94        *                    if not specified and the key serde provides a {@link WindowedSerializer} for the key
95        *                    {@link WindowedStreamPartitioner} will be used—otherwise {@link DefaultPartitioner} will be used
96        * @param <K>         key type
97        * @param <V>         value type
98        * @return A new {@code Repartitioned} instance configured with partitioner
99        * @see KStream#repartition(Repartitioned)
100        */
101       public static <K, V> Repartitioned<K, V> streamPartitioner(final StreamPartitioner<K, V> partitioner) {
102           return new Repartitioned<>(null, null, null, null, partitioner);
103       }
104   
105       /**
106        * Create a {@code Repartitioned} instance with provided number of partitions for repartition topic.
107        *
108        * @param numberOfPartitions number of partitions used when creating repartition topic
109        * @param <K>                key type
110        * @param <V>                value type
111        * @return A new {@code Repartitioned} instance configured number of partitions
112        * @see KStream#repartition(Repartitioned)
113        */
114       public static <K, V> Repartitioned<K, V> numberOfPartitions(final int numberOfPartitions) {
115           return new Repartitioned<>(null, null, null, numberOfPartitions, null);
116       }
117   
118       /**
119        * Create a new instance of {@code Repartitioned} with the provided name used as part of repartition topic and processor name.
120        *
121        * @param name the name used for the processor name and as part of the repartition topic
122        * @return a new {@code Repartitioned} instance configured with the name
123        */
124       @Override
125       public Repartitioned<K, V> withName(final String name) {
126           return new Repartitioned<>(name, keySerde, valueSerde, numberOfPartitions, partitioner);
127       }
128   
129       /**
130        * Create a new instance of {@code Repartitioned} with the provided number of partitions for repartition topic.
131        *
132        * @param numberOfPartitions the name used for the processor name and as part of the repartition topic name
133        * @return a new {@code Repartitioned} instance configured with the number of partitions
134        */
135       public Repartitioned<K, V> withNumberOfPartitions(final int numberOfPartitions) {
136           return new Repartitioned<>(name, keySerde, valueSerde, numberOfPartitions, partitioner);
137       }
138   
139       /**
140        * Create a new instance of {@code Repartitioned} with the provided key serde.
141        *
142        * @param keySerde Serde to use for serializing the key
143        * @return a new {@code Repartitioned} instance configured with the key serde
144        */
145       public Repartitioned<K, V> withKeySerde(final Serde<K> keySerde) {
146           return new Repartitioned<>(name, keySerde, valueSerde, numberOfPartitions, partitioner);
147       }
148   
149       /**
150        * Create a new instance of {@code Repartitioned} with the provided value serde.
151        *
152        * @param valueSerde Serde to use for serializing the value
153        * @return a new {@code Repartitioned} instance configured with the value serde
154        */
155       public Repartitioned<K, V> withValueSerde(final Serde<V> valueSerde) {
156           return new Repartitioned<>(name, keySerde, valueSerde, numberOfPartitions, partitioner);
157       }
158   
159       /**
160        * Create a new instance of {@code Repartitioned} with the provided partitioner.
161        *
162        * @param partitioner the function used to determine how records are distributed among partitions of the topic,
163        *                    if not specified and the key serde provides a {@link WindowedSerializer} for the key
164        *                    {@link WindowedStreamPartitioner} will be used—otherwise {@link DefaultPartitioner} wil be used
165        * @return a new {@code Repartitioned} instance configured with provided partitioner
166        */
167       public Repartitioned<K, V> withStreamPartitioner(final StreamPartitioner<K, V> partitioner) {
168           return new Repartitioned<>(name, keySerde, valueSerde, numberOfPartitions, partitioner);
169       }
