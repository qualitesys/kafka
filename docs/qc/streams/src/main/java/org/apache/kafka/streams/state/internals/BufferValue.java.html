17   package org.apache.kafka.streams.state.internals;
18   
19   import org.apache.kafka.streams.processor.internals.ProcessorRecordContext;
20   
21   import java.nio.ByteBuffer;
22   import java.util.Arrays;
23   import java.util.Objects;
24   
25   import static org.apache.kafka.common.utils.Utils.getNullableArray;
26   import static org.apache.kafka.common.utils.Utils.getNullableSizePrefixedArray;
27   
28   public final class BufferValue {
29       private static final int NULL_VALUE_SENTINEL = -1;
30       private static final int OLD_PREV_DUPLICATE_VALUE_SENTINEL = -2;
31       private final byte[] priorValue;
32       private final byte[] oldValue;
33       private final byte[] newValue;
34       private final ProcessorRecordContext recordContext;
35   
36       BufferValue(final byte[] priorValue,
37                   final byte[] oldValue,
38                   final byte[] newValue,
39                   final ProcessorRecordContext recordContext) {
40           this.oldValue = oldValue;
41           this.newValue = newValue;
42           this.recordContext = recordContext;
43   
44           // This de-duplicates the prior and old references.
45           // If they were already the same reference, the comparison is trivially fast, so we don't specifically check
46           // for that case.
47           if (Arrays.equals(priorValue, oldValue)) {
48               this.priorValue = oldValue;
49           } else {
50               this.priorValue = priorValue;
51           }
52       }
53   
54       byte[] priorValue() {
55           return priorValue;
56       }
57   
58       byte[] oldValue() {
59           return oldValue;
60       }
61   
62       byte[] newValue() {
63           return newValue;
64       }
65   
66       ProcessorRecordContext context() {
67           return recordContext;
68       }
69   
70       static BufferValue deserialize(final ByteBuffer buffer) {
71           final ProcessorRecordContext context = ProcessorRecordContext.deserialize(buffer);
72   
73           final byte[] priorValue = getNullableSizePrefixedArray(buffer);
74   
75           final byte[] oldValue;
76           final int oldValueLength = buffer.getInt();
77           if (oldValueLength == OLD_PREV_DUPLICATE_VALUE_SENTINEL) {
78               oldValue = priorValue;
79           } else {
80               oldValue = getNullableArray(buffer, oldValueLength);
81           }
82   
83           final byte[] newValue = getNullableSizePrefixedArray(buffer);
84   
85           return new BufferValue(priorValue, oldValue, newValue, context);
86       }
87   
88       ByteBuffer serialize(final int endPadding) {
89   
90           final int sizeOfValueLength = Integer.BYTES;
91   
92           final int sizeOfPriorValue = priorValue == null ? 0 : priorValue.length;
93           final int sizeOfOldValue = oldValue == null || priorValue == oldValue ? 0 : oldValue.length;
94           final int sizeOfNewValue = newValue == null ? 0 : newValue.length;
95   
96           final byte[] serializedContext = recordContext.serialize();
97   
98           final ByteBuffer buffer = ByteBuffer.allocate(
99               serializedContext.length
100                   + sizeOfValueLength + sizeOfPriorValue
101                   + sizeOfValueLength + sizeOfOldValue
102                   + sizeOfValueLength + sizeOfNewValue
103                   + endPadding
104           );
105   
106           buffer.put(serializedContext);
107   
108           addValue(buffer, priorValue);
109   
110           if (oldValue == null) {
111               buffer.putInt(NULL_VALUE_SENTINEL);
112           } else if (Arrays.equals(priorValue, oldValue)) {
113               buffer.putInt(OLD_PREV_DUPLICATE_VALUE_SENTINEL);
114           } else {
115               buffer.putInt(sizeOfOldValue);
116               buffer.put(oldValue);
117           }
118   
119           addValue(buffer, newValue);
120   
121           return buffer;
122       }
123   
124       private static void addValue(final ByteBuffer buffer, final byte[] value) {
125           if (value == null) {
126               buffer.putInt(NULL_VALUE_SENTINEL);
127           } else {
128               buffer.putInt(value.length);
129               buffer.put(value);
130           }
131       }
132   
133       long residentMemorySizeEstimate() {
134           return (priorValue == null ? 0 : priorValue.length)
135               + (oldValue == null || priorValue == oldValue ? 0 : oldValue.length)
136               + (newValue == null ? 0 : newValue.length)
137               + recordContext.residentMemorySizeEstimate();
138       }
139   
140       @Override
141       public boolean equals(final Object o) {
142           if (this == o) return true;
143           if (o == null || getClass() != o.getClass()) return false;
144           final BufferValue that = (BufferValue) o;
145           return Arrays.equals(priorValue, that.priorValue) &&
146               Arrays.equals(oldValue, that.oldValue) &&
147               Arrays.equals(newValue, that.newValue) &&
148               Objects.equals(recordContext, that.recordContext);
149       }
150   
151       @Override
152       public int hashCode() {
153           int result = Objects.hash(recordContext);
154           result = 31 * result + Arrays.hashCode(priorValue);
155           result = 31 * result + Arrays.hashCode(oldValue);
156           result = 31 * result + Arrays.hashCode(newValue);
157           return result;
158       }
159   
160       @Override
161       public String toString() {
162           return "BufferValue{" +
163               "priorValue=" + Arrays.toString(priorValue) +
164               ", oldValue=" + Arrays.toString(oldValue) +
165               ", newValue=" + Arrays.toString(newValue) +
166               ", recordContext=" + recordContext +
167               '}';
168       }
169   }
