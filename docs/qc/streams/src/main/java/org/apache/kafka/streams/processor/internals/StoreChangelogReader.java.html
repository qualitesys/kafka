56   /**
57    * ChangelogReader is created and maintained by the stream thread and used for both updating standby tasks and
58    * restoring active tasks. It manages the restore consumer, including its assigned partitions, when to pause / resume
59    * these partitions, etc.
60    * <p>
61    * The reader also maintains the source of truth for restoration state: only active tasks restoring changelog could
62    * be completed, while standby tasks updating changelog would always be in restoring state after being initialized.
63    */
64   public class StoreChangelogReader implements ChangelogReader {
65       private static final long RESTORE_LOG_INTERVAL_MS = 10_000L;
66       private long lastRestoreLogTime = 0L;

73           RESTORING("RESTORING", 0),

240       private boolean hasRestoredToEnd(final ChangelogMetadata metadata) {
241           final Long endOffset = metadata.restoreEndOffset;
242           if (endOffset == null) {
243               // end offset is not initialized meaning that it is from a standby task,
244               // this should never happen since we only call this function for active task in restoring phase
245               throw new IllegalStateException("End offset for changelog " + metadata + " is unknown when deciding " +
246                   "if it has completed restoration, this should never happen.");
247           } else if (endOffset == 0) {
248               // this is a special case, meaning there's nothing to be restored since the changelog has no data
249               // OR the changelog is a source topic and there's no committed offset
250               return true;
251           } else if (metadata.bufferedRecords.isEmpty()) {
252               // NOTE there are several corner cases that we need to consider:
253               //  1) the end / committed offset returned from the consumer is the last offset + 1
254               //  2) there could be txn markers as the last record if EOS is enabled at the producer
255               //
256               // It is possible that: the last record's offset == last txn marker offset - 1 == end / committed offset - 2
257               //
258               // So we make the following decision:
259               //  1) if all the buffered records have been applied, then we compare the end offset with the
260               //     current consumer's position, which is the "next" record to fetch, bypassing the txn marker already
261               //  2) if not all the buffered records have been applied, then it means we are restricted by the end offset,
262               //     and the consumer's position is likely already ahead of that end offset. Then we just need to check
263               //     the first record in the remaining buffer and see if that record is no smaller than the end offset.
264               final TopicPartition partition = metadata.storeMetadata.changelogPartition();
265               try {
266                   return restoreConsumer.position(partition) >= endOffset;
267               } catch (final TimeoutException timeoutException) {
268                   // re-throw to trigger `task.timeout.ms`
269                   throw timeoutException;
270               } catch (final KafkaException e) {
271                   // this also includes InvalidOffsetException, which should not happen under normal
272                   // execution, hence it is also okay to wrap it as fatal StreamsException
273                   throw new StreamsException("Restore consumer get unexpected error trying to get the position " +
274                       " of " + partition, e);
275               }
276           } else {
277               return metadata.bufferedRecords.get(0).offset() >= endOffset;
278           }
279       }

281       // Once some new tasks are created, we transit to restore them and pause on the existing standby tasks. It is
282       // possible that when newly created tasks are created the changelog reader are still restoring existing
283       // active tasks, and hence this function is idempotent and can be called multiple times.
284       //
285       // NOTE: even if the newly created tasks do not need any restoring, we still first transit to this state and then
286       // immediately transit back -- there's no overhead of transiting back and forth but simplifies the logic a lot.
287       @Override
288       public void enforceRestoreActive() {
289           if (state != ChangelogReaderState.ACTIVE_RESTORING) {
290               log.debug("Transiting to restore active tasks: {}", changelogs);
291               lastRestoreLogTime = 0L;
292   
293               // pause all partitions that are for standby tasks from the restore consumer
294               pauseChangelogsFromRestoreConsumer(standbyRestoringChangelogs());
295   
296               state = ChangelogReaderState.ACTIVE_RESTORING;
297           }
298       }
299   
300       // Only after we've completed restoring all active tasks we'll then move back to resume updating standby tasks.
301       // This function is NOT idempotent: if it is already in updating standby tasks mode, we should not call it again.
302       //
303       // NOTE: we do not clear completed active restoring changelogs or remove partitions from restore consumer either
304       // upon completing them but only pause the corresponding partitions; the changelog metadata / partitions would only
305       // be cleared when the corresponding task is being removed from the thread. In other words, the restore consumer
306       // should contain all changelogs that are RESTORING or COMPLETED
307       @Override
308       public void transitToUpdateStandby() {
309           if (state != ChangelogReaderState.ACTIVE_RESTORING) {
310               throw new IllegalStateException(
311                   "The changelog reader is not restoring active tasks (is " + state + ") while trying to " +
312                       "transit to update standby tasks: " + changelogs
313               );
314           }
315   
316           log.debug("Transiting to update standby tasks: {}", changelogs);
317   
318           // resume all standby restoring changelogs from the restore consumer
319           resumeChangelogsFromRestoreConsumer(standbyRestoringChangelogs());
320   
321           state = ChangelogReaderState.STANDBY_UPDATING;
322       }
323   
324       /**
325        * Since it is shared for multiple tasks and hence multiple state managers, the registration would take its
326        * corresponding state manager as well for restoring.
327        */
328       @Override
329       public void register(final TopicPartition partition, final ProcessorStateManager stateManager) {
330           final StateStoreMetadata storeMetadata = stateManager.storeMetadata(partition);
331           if (storeMetadata == null) {
332               throw new IllegalStateException("Cannot find the corresponding state store metadata for changelog " +
333                   partition);
334           }
335   
336           final ChangelogMetadata changelogMetadata = new ChangelogMetadata(storeMetadata, stateManager);
337   
338           // initializing limit offset to 0L for standby changelog to effectively disable any restoration until it is updated
339           if (stateManager.taskType() == Task.TaskType.STANDBY && stateManager.changelogAsSource(partition)) {
340               changelogMetadata.restoreEndOffset = 0L;
341           }
342   
343           if (changelogs.putIfAbsent(partition, changelogMetadata) != null) {
344               throw new IllegalStateException("There is already a changelog registered for " + partition +
345                   ", this should not happen: " + changelogs);
346           }
347       }
348   
349       private ChangelogMetadata restoringChangelogByPartition(final TopicPartition partition) {
350           final ChangelogMetadata changelogMetadata = changelogs.get(partition);
351           if (changelogMetadata == null) {
352               throw new IllegalStateException("The corresponding changelog restorer for " + partition +
353                   " does not exist, this should not happen.");
354           }
355           if (changelogMetadata.changelogState != ChangelogState.RESTORING) {
356               throw new IllegalStateException("The corresponding changelog restorer for " + partition +
357                   " has already transited to completed state, this should not happen.");
358           }
359   
360           return changelogMetadata;
361       }

401               .filter(metadata -> metadata.changelogState == ChangelogState.COMPLETED)

406       // 1. if there are any registered changelogs that needs initialization, try to initialize them first;
407       // 2. if all changelogs have finished, return early;
408       // 3. if there are any restoring changelogs, try to read from the restore consumer and process them.
409       @Override
410       public void restore(final Map<TaskId, Task> tasks) {
411           initializeChangelogs(tasks, registeredChangelogs());
412   
413           if (!activeRestoringChangelogs().isEmpty() && state == ChangelogReaderState.STANDBY_UPDATING) {
414               throw new IllegalStateException("Should not be in standby updating state if there are still un-completed active changelogs");
415           }
416   
417           if (allChangelogsCompleted()) {
418               log.debug("Finished restoring all changelogs {}", changelogs.keySet());
419               return;
420           }
421   
422           final Set<TopicPartition> restoringChangelogs = restoringChangelogs();
423           if (!restoringChangelogs.isEmpty()) {
424               final ConsumerRecords<byte[], byte[]> polledRecords;
425   
426               try {
427                   pauseResumePartitions(tasks, restoringChangelogs);
428   
429                   // for restoring active and updating standby we may prefer different poll time
430                   // in order to make sure we call the main consumer#poll in time.
431                   // TODO: once we move ChangelogReader to a separate thread this may no longer be a concern
432                   polledRecords = restoreConsumer.poll(state == ChangelogReaderState.STANDBY_UPDATING ? Duration.ZERO : pollTime);
433   
434                   // TODO (?) If we cannot fetch records during restore, should we trigger `task.timeout.ms` ?
435                   // TODO (?) If we cannot fetch records for standby task, should we trigger `task.timeout.ms` ?
436               } catch (final InvalidOffsetException e) {
437                   log.warn("Encountered " + e.getClass().getName() +
438                       " fetching records from restore consumer for partitions " + e.partitions() + ", it is likely that " +
439                       "the consumer's position has fallen out of the topic partition offset range because the topic was " +
440                       "truncated or compacted on the broker, marking the corresponding tasks as corrupted and re-initializing" +
441                       " it later.", e);
442   
443                   final Set<TaskId> corruptedTasks = new HashSet<>();
444                   e.partitions().forEach(partition -> corruptedTasks.add(changelogs.get(partition).stateManager.taskId()));
445                   throw new TaskCorruptedException(corruptedTasks, e);
446               } catch (final KafkaException e) {
447                   throw new StreamsException("Restore consumer get unexpected error polling records.", e);
448               }
449   
450               for (final TopicPartition partition : polledRecords.partitions()) {
451                   bufferChangelogRecords(restoringChangelogByPartition(partition), polledRecords.records(partition));
452               }
453   
454               for (final TopicPartition partition : restoringChangelogs) {
455                   // even if some partition do not have any accumulated data, we still trigger
456                   // restoring since some changelog may not need to restore any at all, and the
457                   // restore to end check needs to be executed still.
458                   // TODO: we always try to restore as a batch when some records are accumulated, which may result in
459                   //       small batches; this can be optimized in the future, e.g. wait longer for larger batches.
460                   final TaskId taskId = changelogs.get(partition).stateManager.taskId();
461                   try {
462                       if (restoreChangelog(changelogs.get(partition))) {
463                           final Task task = tasks.get(taskId);
464                           if (task != null) {
465                               task.clearTaskTimeout();
466                           }
467                       }
468                   } catch (final TimeoutException timeoutException) {
469                       tasks.get(taskId).maybeInitTaskTimeoutOrThrow(
470                           time.milliseconds(),
471                           timeoutException
472                       );
473                   }
474               }
475   
476               maybeUpdateLimitOffsetsForStandbyChangelogs(tasks);
477   
478               maybeLogRestorationProgress();
479           }
480       }

560       private void maybeUpdateLimitOffsetsForStandbyChangelogs(final Map<TaskId, Task> tasks) {
561           // we only consider updating the limit offset for standbys if we are not restoring active tasks
562           if (state == ChangelogReaderState.STANDBY_UPDATING &&
563               updateOffsetIntervalMs < time.milliseconds() - lastUpdateOffsetTime) {
564   
565               // when the interval has elapsed we should try to update the limit offset for standbys reading from
566               // a source changelog with the new committed offset, unless there are no buffered records since 
567               // we only need the limit when processing new records
568               // for other changelog partitions we do not need to update limit offset at all since we never need to
569               // check when it completes based on limit offset anyways: the end offset would keep increasing and the
570               // standby never need to stop
571               final Set<TopicPartition> changelogsWithLimitOffsets = changelogs.entrySet().stream()
572                   .filter(entry -> entry.getValue().stateManager.taskType() == Task.TaskType.STANDBY &&
573                       entry.getValue().stateManager.changelogAsSource(entry.getKey()))
574                   .map(Map.Entry::getKey).collect(Collectors.toSet());
575   
576               for (final TopicPartition partition : changelogsWithLimitOffsets) {
577                   if (!changelogs.get(partition).bufferedRecords().isEmpty()) {
578                       updateLimitOffsetsForStandbyChangelogs(committedOffsetForChangelogs(tasks, changelogsWithLimitOffsets));
579                       lastUpdateOffsetTime = time.milliseconds();
580                       break;
581                   }
582               }
583           }
584       }

586       private void bufferChangelogRecords(final ChangelogMetadata changelogMetadata, final List<ConsumerRecord<byte[], byte[]>> records) {
587           // update the buffered records and limit index with the fetched records
588           for (final ConsumerRecord<byte[], byte[]> record : records) {
589               // filter polled records for null-keys and also possibly update buffer limit index
590               if (record.key() == null) {
591                   log.warn("Read changelog record with null key from changelog {} at offset {}, " +
592                       "skipping it for restoration", changelogMetadata.storeMetadata.changelogPartition(), record.offset());
593               } else {
594                   changelogMetadata.bufferedRecords.add(record);
595                   final long offset = record.offset();
596                   if (changelogMetadata.restoreEndOffset == null || offset < changelogMetadata.restoreEndOffset) {
597                       changelogMetadata.bufferedLimitIndex = changelogMetadata.bufferedRecords.size();
598                   }
599               }
600           }
601       }
602   
603       /**
604        * restore a changelog with its buffered records if there's any; for active changelogs also check if
605        * it has completed the restoration and can transit to COMPLETED state and trigger restore callbacks
606        */
607       private boolean restoreChangelog(final ChangelogMetadata changelogMetadata) {
608           final ProcessorStateManager stateManager = changelogMetadata.stateManager;
609           final StateStoreMetadata storeMetadata = changelogMetadata.storeMetadata;
610           final TopicPartition partition = storeMetadata.changelogPartition();
611           final String storeName = storeMetadata.store().name();
612           final int numRecords = changelogMetadata.bufferedLimitIndex;
613   
614           boolean madeProgress = false;
615   
616           if (numRecords != 0) {
617               madeProgress = true;
618   
619               final List<ConsumerRecord<byte[], byte[]>> records = changelogMetadata.bufferedRecords.subList(0, numRecords);
620               stateManager.restore(storeMetadata, records);
621   
622               // NOTE here we use removeRange of ArrayList in order to achieve efficiency with range shifting,
623               // otherwise one-at-a-time removal or addition would be very costly; if all records are restored
624               // then we can further optimize to save the array-shift but just set array elements to null
625               if (numRecords < changelogMetadata.bufferedRecords.size()) {
626                   records.clear();
627               } else {
628                   changelogMetadata.bufferedRecords.clear();
629               }
630   
631               final Long currentOffset = storeMetadata.offset();
632               log.trace("Restored {} records from changelog {} to store {}, end offset is {}, current offset is {}",
633                   partition, storeName, numRecords, recordEndOffset(changelogMetadata.restoreEndOffset), currentOffset);
634   
635               changelogMetadata.bufferedLimitIndex = 0;
636               changelogMetadata.totalRestored += numRecords;
637   
638               // do not trigger restore listener if we are processing standby tasks
639               if (changelogMetadata.stateManager.taskType() == Task.TaskType.ACTIVE) {
640                   try {
641                       stateRestoreListener.onBatchRestored(partition, storeName, currentOffset, numRecords);
642                   } catch (final Exception e) {
643                       throw new StreamsException("State restore listener failed on batch restored", e);
644                   }
645               }
646           }
647   
648           // we should check even if there's nothing restored, but do not check completed if we are processing standby tasks
649           if (changelogMetadata.stateManager.taskType() == Task.TaskType.ACTIVE && hasRestoredToEnd(changelogMetadata)) {
650               madeProgress = true;
651   
652               log.info("Finished restoring changelog {} to store {} with a total number of {} records",
653                   partition, storeName, changelogMetadata.totalRestored);
654   
655               changelogMetadata.transitTo(ChangelogState.COMPLETED);
656               pauseChangelogsFromRestoreConsumer(Collections.singleton(partition));
657   
658               try {
659                   stateRestoreListener.onRestoreEnd(partition, storeName, changelogMetadata.totalRestored);
660               } catch (final Exception e) {
661                   throw new StreamsException("State restore listener failed on restore completed", e);
662               }
663           }
664   
665           return madeProgress;
666       }

691       private Map<TopicPartition, Long> committedOffsetForChangelogs(final Map<TaskId, Task> tasks, final Set<TopicPartition> partitions) {
692           if (partitions.isEmpty()) {
693               return Collections.emptyMap();
694           }
695   
696           try {
697               // those which do not have a committed offset would default to 0
698               final ListConsumerGroupOffsetsOptions options = new ListConsumerGroupOffsetsOptions();
699               options.topicPartitions(new ArrayList<>(partitions));
700               options.requireStable(true);
701               final Map<TopicPartition, Long> committedOffsets = adminClient.listConsumerGroupOffsets(groupId, options)
702                       .partitionsToOffsetAndMetadata().get().entrySet()
703                       .stream()
704                       .collect(Collectors.toMap(Map.Entry::getKey, e -> e.getValue() == null ? 0L : e.getValue().offset()));
705   
706               clearTaskTimeout(getTasksFromPartitions(tasks, partitions));
707               return committedOffsets;
708           } catch (final TimeoutException | InterruptedException | ExecutionException retriableException) {
709               log.debug("Could not retrieve the committed offsets for partitions {} due to {}, will retry in the next run loop",
710                   partitions, retriableException.toString());
711               maybeInitTaskTimeoutOrThrow(getTasksFromPartitions(tasks, partitions), retriableException);
712               return Collections.emptyMap();
713           } catch (final KafkaException e) {
714               throw new StreamsException(String.format("Failed to retrieve committed offsets for %s", partitions), e);
715           }
716       }

718       private Map<TopicPartition, Long> endOffsetForChangelogs(final Map<TaskId, Task> tasks, final Set<TopicPartition> partitions) {
719           if (partitions.isEmpty()) {
720               return Collections.emptyMap();
721           }
722   
723           try {
724               // we always use read_uncommitted to get log end offset since the last committed txn may have not advanced the LSO for EOS,
725               // see KAFKA-10167 for more details
726               final ListOffsetsOptions options = new ListOffsetsOptions(IsolationLevel.READ_UNCOMMITTED);
727               final Map<TopicPartition, OffsetSpec> offsetSpecs =
728                   partitions.stream().collect(Collectors.toMap(Function.identity(), tp -> OffsetSpec.latest()));
729               final Map<TopicPartition, Long> logEndOffsets = adminClient.listOffsets(offsetSpecs, options)
730                   .all().get().entrySet()
731                   .stream().collect(Collectors.toMap(Map.Entry::getKey, entry -> entry.getValue().offset()));
732   
733               clearTaskTimeout(getTasksFromPartitions(tasks, partitions));
734               return logEndOffsets;
735           } catch (final TimeoutException | InterruptedException | ExecutionException retriableException) {
736               log.debug("Could not fetch all end offsets for {} due to {}, will retry in the next run loop",
737                   partitions, retriableException.toString());
738               maybeInitTaskTimeoutOrThrow(getTasksFromPartitions(tasks, partitions), retriableException);
739               return Collections.emptyMap();
740           } catch (final KafkaException e) {
741               throw new StreamsException(String.format("Failed to retrieve end offsets for %s", partitions), e);
742           }
743       }

745       private void updateLimitOffsetsForStandbyChangelogs(final Map<TopicPartition, Long> committedOffsets) {
746           for (final ChangelogMetadata metadata : changelogs.values()) {
747               final TopicPartition partition = metadata.storeMetadata.changelogPartition();
748               if (metadata.stateManager.taskType() == Task.TaskType.STANDBY &&
749                   metadata.stateManager.changelogAsSource(partition) &&
750                   committedOffsets.containsKey(partition)) {
751   
752                   final Long newLimit = committedOffsets.get(partition);
753                   final Long previousLimit = metadata.restoreEndOffset;
754   
755                   if (previousLimit != null && previousLimit > newLimit) {
756                       throw new IllegalStateException("Offset limit should monotonically increase, but was reduced for partition " +
757                           partition + ". New limit: " + newLimit + ". Previous limit: " + previousLimit);
758                   }
759   
760                   metadata.restoreEndOffset = newLimit;
761   
762                   // update the limit index for buffered records
763                   while (metadata.bufferedLimitIndex < metadata.bufferedRecords.size() &&
764                       metadata.bufferedRecords.get(metadata.bufferedLimitIndex).offset() < metadata.restoreEndOffset)
765                       metadata.bufferedLimitIndex++;
766               }
767           }
768       }

770       private void initializeChangelogs(final Map<TaskId, Task> tasks,
771                                         final Set<ChangelogMetadata> newPartitionsToRestore) {
772           if (newPartitionsToRestore.isEmpty()) {
773               return;
774           }
775   
776           // for active changelogs, we need to find their end offset before transit to restoring
777           // if the changelog is on source topic, then its end offset should be the minimum of
778           // its committed offset and its end offset; for standby tasks that use source topics
779           // as changelogs, we want to initialize their limit offsets as committed offsets as well
780           final Set<TopicPartition> newPartitionsToFindEndOffset = new HashSet<>();
781           final Set<TopicPartition> newPartitionsToFindCommittedOffset = new HashSet<>();
782   
783           for (final ChangelogMetadata metadata : newPartitionsToRestore) {
784               final TopicPartition partition = metadata.storeMetadata.changelogPartition();
785   
786               // TODO K9113: when TaskType.GLOBAL is added we need to modify this
787               if (metadata.stateManager.taskType() == Task.TaskType.ACTIVE) {
788                   newPartitionsToFindEndOffset.add(partition);
789               }
790   
791               if (metadata.stateManager.changelogAsSource(partition)) {
792                   newPartitionsToFindCommittedOffset.add(partition);
793               }
794           }
795   
796           // NOTE we assume that all requested partitions will be included in the returned map for both end/committed
797           // offsets, i.e., it would not return partial result and would timeout if some of the results cannot be found
798           final Map<TopicPartition, Long> endOffsets = endOffsetForChangelogs(tasks, newPartitionsToFindEndOffset);
799           final Map<TopicPartition, Long> committedOffsets = committedOffsetForChangelogs(tasks, newPartitionsToFindCommittedOffset);
800   
801           for (final TopicPartition partition : newPartitionsToFindEndOffset) {
802               final ChangelogMetadata changelogMetadata = changelogs.get(partition);
803               final Long endOffset = endOffsets.get(partition);
804               final Long committedOffset = newPartitionsToFindCommittedOffset.contains(partition) ?
805                   committedOffsets.get(partition) : Long.valueOf(Long.MAX_VALUE);
806   
807               if (endOffset != null && committedOffset != null) {
808                   if (changelogMetadata.restoreEndOffset != null) {
809                       throw new IllegalStateException("End offset for " + partition +
810                           " should only be initialized once. Existing value: " + changelogMetadata.restoreEndOffset +
811                           ", new value: (" + endOffset + ", " + committedOffset + ")");
812                   }
813   
814                   changelogMetadata.restoreEndOffset = Math.min(endOffset, committedOffset);
815   
816                   log.debug("End offset for changelog {} initialized as {}.", partition, changelogMetadata.restoreEndOffset);
817               } else {
818                   if (!newPartitionsToRestore.remove(changelogMetadata)) {
819                       throw new IllegalStateException("New changelogs to restore " + newPartitionsToRestore +
820                           " does not contain the one looking for end offset: " + partition + ", this should not happen.");
821                   }
822   
823                   log.info("End offset for changelog {} cannot be found; will retry in the next time.", partition);
824               }
825           }
826   
827           // try initialize limit offsets for standby tasks for the first time
828           if (!committedOffsets.isEmpty()) {
829               updateLimitOffsetsForStandbyChangelogs(committedOffsets);
830           }
831   
832           // add new partitions to the restore consumer and transit them to restoring state
833           addChangelogsToRestoreConsumer(newPartitionsToRestore.stream().map(metadata -> metadata.storeMetadata.changelogPartition())
834               .collect(Collectors.toSet()));
835   
836           newPartitionsToRestore.forEach(metadata -> metadata.transitTo(ChangelogState.RESTORING));
837   
838           // if it is in the active restoring mode, we immediately pause those standby changelogs
839           // here we just blindly pause all (including the existing and newly added)
840           if (state == ChangelogReaderState.ACTIVE_RESTORING) {
841               pauseChangelogsFromRestoreConsumer(standbyRestoringChangelogs());
842           }
843   
844           // prepare newly added partitions of the restore consumer by setting their starting position
845           prepareChangelogs(newPartitionsToRestore);
846       }
847   
848       private void addChangelogsToRestoreConsumer(final Set<TopicPartition> partitions) {
849           final Set<TopicPartition> assignment = new HashSet<>(restoreConsumer.assignment());
850   
851           // the current assignment should not contain any of the new partitions
852           if (assignment.removeAll(partitions)) {
853               throw new IllegalStateException("The current assignment " + restoreConsumer.assignment() + " " +
854                   "already contains some of the new partitions " + partitions);
855           }
856           assignment.addAll(partitions);
857           restoreConsumer.assign(assignment);
858   
859           log.debug("Added partitions {} to the restore consumer, current assignment is {}", partitions, assignment);
860       }
861   
862       private void pauseChangelogsFromRestoreConsumer(final Collection<TopicPartition> partitions) {
863           final Set<TopicPartition> assignment = new HashSet<>(restoreConsumer.assignment());
864   
865           // the current assignment should contain all the partitions to pause
866           if (!assignment.containsAll(partitions)) {
867               throw new IllegalStateException("The current assignment " + assignment + " " +
868                   "does not contain some of the partitions " + partitions + " for pausing.");
869           }
870           restoreConsumer.pause(partitions);
871   
872           log.debug("Paused partitions {} from the restore consumer", partitions);
873       }
874   
875       private void removeChangelogsFromRestoreConsumer(final Collection<TopicPartition> partitions) {
876           final Set<TopicPartition> assignment = new HashSet<>(restoreConsumer.assignment());
877   
878           // the current assignment should contain all the partitions to remove
879           if (!assignment.containsAll(partitions)) {
880               throw new IllegalStateException("The current assignment " + assignment + " " +
881                   "does not contain some of the partitions " + partitions + " for removing.");
882           }
883           assignment.removeAll(partitions);
884           restoreConsumer.assign(assignment);
885       }
886   
887       private void resumeChangelogsFromRestoreConsumer(final Collection<TopicPartition> partitions) {
888           final Set<TopicPartition> assignment = new HashSet<>(restoreConsumer.assignment());
889   
890           // the current assignment should contain all the partitions to resume
891           if (!assignment.containsAll(partitions)) {
892               throw new IllegalStateException("The current assignment " + assignment + " " +
893                   "does not contain some of the partitions " + partitions + " for resuming.");
894           }
895           restoreConsumer.resume(partitions);
896   
897           log.debug("Resumed partitions {} from the restore consumer", partitions);
898       }

900       private void prepareChangelogs(final Set<ChangelogMetadata> newPartitionsToRestore) {
901           // separate those who do not have the current offset loaded from checkpoint
902           final Set<TopicPartition> newPartitionsWithoutStartOffset = new HashSet<>();
903   
904           for (final ChangelogMetadata changelogMetadata : newPartitionsToRestore) {
905               final StateStoreMetadata storeMetadata = changelogMetadata.storeMetadata;
906               final TopicPartition partition = storeMetadata.changelogPartition();
907               final Long currentOffset = storeMetadata.offset();
908               final Long endOffset = changelogs.get(partition).restoreEndOffset;
909   
910               if (currentOffset != null) {
911                   // the current offset is the offset of the last record, so we should set the position
912                   // as that offset + 1 as the "next" record to fetch; seek is not a blocking call so
913                   // there's nothing to capture
914                   restoreConsumer.seek(partition, currentOffset + 1);
915   
916                   log.debug("Start restoring changelog partition {} from current offset {} to end offset {}.",
917                       partition, currentOffset, recordEndOffset(endOffset));
918               } else {
919                   log.debug("Start restoring changelog partition {} from the beginning offset to end offset {} " +
920                       "since we cannot find current offset.", partition, recordEndOffset(endOffset));
921   
922                   newPartitionsWithoutStartOffset.add(partition);
923               }
924           }
925   
926           // optimization: batch all seek-to-beginning offsets in a single request
927           //               seek is not a blocking call so there's nothing to capture
928           if (!newPartitionsWithoutStartOffset.isEmpty()) {
929               restoreConsumer.seekToBeginning(newPartitionsWithoutStartOffset);
930           }
931   
932           // do not trigger restore listener if we are processing standby tasks
933           for (final ChangelogMetadata changelogMetadata : newPartitionsToRestore) {
934               if (changelogMetadata.stateManager.taskType() == Task.TaskType.ACTIVE) {
935                   final StateStoreMetadata storeMetadata = changelogMetadata.storeMetadata;
936                   final TopicPartition partition = storeMetadata.changelogPartition();
937                   final String storeName = storeMetadata.store().name();
938   
939                   long startOffset = 0L;
940                   try {
941                       startOffset = restoreConsumer.position(partition);
942                   } catch (final TimeoutException swallow) {
943                       // if we cannot find the starting position at the beginning, just use the default 0L
944                   } catch (final KafkaException e) {
945                       // this also includes InvalidOffsetException, which should not happen under normal
946                       // execution, hence it is also okay to wrap it as fatal StreamsException
947                       throw new StreamsException("Restore consumer get unexpected error trying to get the position " +
948                           " of " + partition, e);
949                   }
950   
951                   try {
952                       stateRestoreListener.onRestoreStart(partition, storeName, startOffset, changelogMetadata.restoreEndOffset);
953                   } catch (final Exception e) {
954                       throw new StreamsException("State restore listener failed on batch restored", e);
955                   }
956               }
957           }
958       }

961       public void unregister(final Collection<TopicPartition> revokedChangelogs) {
962           // Only changelogs that are initialized have been added to the restore consumer's assignment
963           final List<TopicPartition> revokedInitializedChangelogs = new ArrayList<>();
964   
965           for (final TopicPartition partition : revokedChangelogs) {
966               final ChangelogMetadata changelogMetadata = changelogs.remove(partition);
967               if (changelogMetadata != null) {
968                   if (!changelogMetadata.state().equals(ChangelogState.REGISTERED)) {
969                       revokedInitializedChangelogs.add(partition);
970                   }
971   
972                   changelogMetadata.clear();
973               } else {
974                   log.debug("Changelog partition {} could not be found," +
975                       " it could be already cleaned up during the handling" +
976                       " of task corruption and never restore again", partition);
977               }
978           }
979   
980           removeChangelogsFromRestoreConsumer(revokedInitializedChangelogs);
981       }
