17   package org.apache.kafka.streams.processor.internals;
18   
19   import org.apache.kafka.streams.processor.Cancellable;
20   import org.apache.kafka.streams.processor.Punctuator;
21   
22   public class PunctuationSchedule extends Stamped<ProcessorNode> {
23   
24       private final long interval;
25       private final Punctuator punctuator;
26       private boolean isCancelled = false;
27       // this Cancellable will be re-pointed at the successor schedule in next()
28       private final RepointableCancellable cancellable;
29   
30       PunctuationSchedule(final ProcessorNode node,
31                           final long time,
32                           final long interval,
33                           final Punctuator punctuator) {
34           this(node, time, interval, punctuator, new RepointableCancellable());
35           cancellable.setSchedule(this);
36       }
37   
38       private PunctuationSchedule(final ProcessorNode node,
39                                   final long time,
40                                   final long interval,
41                                   final Punctuator punctuator,
42                                   final RepointableCancellable cancellable) {
43           super(node, time);
44           this.interval = interval;
45           this.punctuator = punctuator;
46           this.cancellable = cancellable;
47       }
48   
49       public ProcessorNode node() {
50           return value;
51       }
52   
53       public Punctuator punctuator() {
54           return punctuator;
55       }
56   
57       public Cancellable cancellable() {
58           return cancellable;
59       }
60   
61       void markCancelled() {
62           isCancelled = true;
63       }
64   
65       boolean isCancelled() {
66           return isCancelled;
67       }
68   
69       public PunctuationSchedule next(final long currTimestamp) {
70           long nextPunctuationTime = timestamp + interval;
71           if (currTimestamp >= nextPunctuationTime) {
72               // we missed one ore more punctuations
73               // avoid scheduling a new punctuations immediately, this can happen:
74               // - when using STREAM_TIME punctuation and there was a gap i.e., no data was
75               //   received for at least 2*interval
76               // - when using WALL_CLOCK_TIME and there was a gap i.e., punctuation was delayed for at least 2*interval (GC pause, overload, ...)
77               final long intervalsMissed = (currTimestamp - timestamp) / interval;
78               nextPunctuationTime = timestamp + (intervalsMissed + 1) * interval;
79           }
80   
81           final PunctuationSchedule nextSchedule = new PunctuationSchedule(value, nextPunctuationTime, interval, punctuator, cancellable);
82   
83           cancellable.setSchedule(nextSchedule);
84   
85           return nextSchedule;
86       }
87   
88       @Override
89       public boolean equals(final Object other) {
90           return super.equals(other);
91       }
92   
93       @Override
94       public int hashCode() {
95           return super.hashCode();
96       }
97   
98       private static class RepointableCancellable implements Cancellable {
99           private PunctuationSchedule schedule;
100   
101           synchronized void setSchedule(final PunctuationSchedule schedule) {
102               this.schedule = schedule;
103           }
104   
105           @Override
106           synchronized public void cancel() {
107               schedule.markCancelled();
108           }
109       }
110   }
