29   /**
30    * A sliding window used for aggregating events.
31    * <p>
32    * Sliding Windows are defined based on a record's timestamp, the window size based on the given maximum time difference (inclusive) between
33    * records in the same window, and the given window grace period. While the window is sliding over the input data stream, a new window is
34    * created each time a record enters the sliding window or a record drops out of the sliding window.
35    * <p>
36    * Records that come after set grace period will be ignored, i.e., a window is closed when
37    * {@code stream-time > window-end + grace-period}.
38    * <p>
39    * For example, if we have a time difference of 5000ms and the following data arrives:
40    * <pre>
41    * +--------------------------------------+
42    * |    key    |    value    |    time    |
43    * +-----------+-------------+------------+
44    * |    A      |     1       |    8000    |
45    * +-----------+-------------+------------+
46    * |    A      |     2       |    9200    |
47    * +-----------+-------------+------------+
48    * |    A      |     3       |    12400   |
49    * +-----------+-------------+------------+
50    * </pre>
51    * We'd have the following 5 windows:
52    * <ul>
53    *     <li>window {@code [3000;8000]} contains [1] (created when first record enters the window)</li>
54    *     <li>window {@code [4200;9200]} contains [1,2] (created when second record enters the window)</li>
55    *     <li>window {@code [7400;12400]} contains [1,2,3] (created when third record enters the window)</li>
56    *     <li>window {@code [8001;13001]} contains [2,3] (created when the first record drops out of the window)</li>
57    *     <li>window {@code [9201;14201]} contains [3] (created when the second record drops out of the window)</li>
58    * </ul>
59    *<p>
60    * Note that while SlidingWindows are of a fixed size, as are {@link TimeWindows}, the start and end points of the window
61    * depend on when events occur in the stream (i.e., event timestamps), similar to {@link SessionWindows}.
62    * <p>
63    * For time semantics, see {@link TimestampExtractor}.
64    *
65    * @see TimeWindows
66    * @see SessionWindows
67    * @see UnlimitedWindows
68    * @see JoinWindows
69    * @see KGroupedStream#windowedBy(SlidingWindows)
70    * @see CogroupedKStream#windowedBy(SlidingWindows)
71    * @see TimestampExtractor
72    */
73   
74   public final class SlidingWindows {
75   
76       /** The size of the windows in milliseconds, defined by the max time difference between records. */
77       private final long timeDifferenceMs;
78   
79       /** The grace period in milliseconds. */
80       private final long graceMs;
81   
82       private SlidingWindows(final long timeDifferenceMs, final long graceMs) {
83           this.timeDifferenceMs = timeDifferenceMs;
84           this.graceMs = graceMs;
85   
86           if (timeDifferenceMs < 0) {
87               throw new IllegalArgumentException("Window time difference must not be negative.");
88           }
89   
90           if (graceMs < 0) {
91               throw new IllegalArgumentException("Window grace period must not be negative.");
92           }
93       }
94   
95       /**
96        * Return a window definition with the window size based on the given maximum time difference (inclusive) between
97        * records in the same window and given window grace period. Reject out-of-order events that arrive after {@code grace}.
98        * A window is closed when {@code stream-time > window-end + grace-period}.
99        * <p>
100        * CAUTION: Using this method implicitly sets the grace period to zero, which means that any out-of-order
101        * records arriving after the window ends are considered late and will be dropped.
102        *
103        * @param timeDifference the max time difference (inclusive) between two records in a window
104        * @return a new window definition with no grace period. Note that this means out-of-order records arriving after the window end will be dropped
105        * @throws IllegalArgumentException if the timeDifference is negative or can't be represented as {@code long milliseconds}
106        */
107       public static SlidingWindows ofTimeDifferenceWithNoGrace(final Duration timeDifference) throws IllegalArgumentException {
108           return ofTimeDifferenceAndGrace(timeDifference, ofMillis(NO_GRACE_PERIOD));
109       }
110   
111       /**
112        * Return a window definition with the window size based on the given maximum time difference (inclusive) between
113        * records in the same window and given window grace period. Reject out-of-order events that arrive after {@code afterWindowEnd}.
114        * A window is closed when {@code stream-time > window-end + grace-period}.
115        *
116        * @param timeDifference the max time difference (inclusive) between two records in a window
117        * @param afterWindowEnd  the grace period to admit out-of-order events to a window
118        * @return a new window definition with the specified grace period
119        * @throws IllegalArgumentException if the timeDifference or afterWindowEnd (grace period) is negative or can't be represented as {@code long milliseconds}
120        */
121       public static SlidingWindows ofTimeDifferenceAndGrace(final Duration timeDifference, final Duration afterWindowEnd) throws IllegalArgumentException {
122           final String timeDifferenceMsgPrefix = prepareMillisCheckFailMsgPrefix(timeDifference, "timeDifference");
123           final long timeDifferenceMs = validateMillisecondDuration(timeDifference, timeDifferenceMsgPrefix);
124           final String afterWindowEndMsgPrefix = prepareMillisCheckFailMsgPrefix(afterWindowEnd, "afterWindowEnd");
125           final long afterWindowEndMs = validateMillisecondDuration(afterWindowEnd, afterWindowEndMsgPrefix);
126   
127           return new SlidingWindows(timeDifferenceMs, afterWindowEndMs);
128       }
129   
130       /**
131        * Return a window definition with the window size based on the given maximum time difference (inclusive) between
132        * records in the same window and given window grace period. Reject out-of-order events that arrive after {@code grace}.
133        * A window is closed when {@code stream-time > window-end + grace-period}.
134        *
135        * @param timeDifference the max time difference (inclusive) between two records in a window
136        * @param grace the grace period to admit out-of-order events to a window
137        * @return a new window definition
138        * @throws IllegalArgumentException if the specified window size is &lt; 0 or grace &lt; 0, or either can't be represented as {@code long milliseconds}
139        * @deprecated since 3.0. Use {@link #ofTimeDifferenceWithNoGrace(Duration)} or {@link #ofTimeDifferenceAndGrace(Duration, Duration)} instead
140        */
141       @Deprecated
142       public static SlidingWindows withTimeDifferenceAndGrace(final Duration timeDifference, final Duration grace) throws IllegalArgumentException {
143           final String msgPrefixSize = prepareMillisCheckFailMsgPrefix(timeDifference, "timeDifference");
144           final long timeDifferenceMs = validateMillisecondDuration(timeDifference, msgPrefixSize);
145   
146           final String msgPrefixGrace = prepareMillisCheckFailMsgPrefix(grace, "grace");
147           final long graceMs = validateMillisecondDuration(grace, msgPrefixGrace);
148   
149           return new SlidingWindows(timeDifferenceMs, graceMs);
150       }
151   
152       public long timeDifferenceMs() {
153           return timeDifferenceMs;
154       }
155   
156       public long gracePeriodMs() {
157           return graceMs;
158       }
159   
160       @Override
161       public boolean equals(final Object o) {
162           if (this == o) {
163               return true;
164           }
165           if (o == null || getClass() != o.getClass()) {
166               return false;
167           }
168           final SlidingWindows that = (SlidingWindows) o;
169           return timeDifferenceMs == that.timeDifferenceMs &&
170               graceMs == that.graceMs;
171       }
172   
173       @Override
174       public int hashCode() {
175           return Objects.hash(timeDifferenceMs, graceMs);
176       }
177   
178       @Override
179       public String toString() {
180           return "SlidingWindows{" +
181               ", sizeMs=" + timeDifferenceMs +
182               ", graceMs=" + graceMs +
183               '}';
184       }
