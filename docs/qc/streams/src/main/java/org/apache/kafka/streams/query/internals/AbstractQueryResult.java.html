17   package org.apache.kafka.streams.query.internals;
18   
19   
20   import org.apache.kafka.streams.query.Position;
21   import org.apache.kafka.streams.query.PositionBound;
22   import org.apache.kafka.streams.query.QueryResult;
23   import org.apache.kafka.streams.query.StateQueryRequest;
24   
25   import java.util.LinkedList;
26   import java.util.List;
27   
28   /**
29    * Container for a single partition's result when executing a {@link StateQueryRequest}.
30    *
31    * @param <R> The result type of the query.
32    */
33   public abstract class AbstractQueryResult<R> implements QueryResult<R> {
34   
35       private List<String> executionInfo = new LinkedList<>();
36       private Position position;
37   
38       public AbstractQueryResult() {
39   
40       }
41   
42       public AbstractQueryResult(final List<String> executionInfo, final Position position) {
43           this.executionInfo = executionInfo;
44           this.position = position;
45       }
46   
47       /**
48        * Used by stores to add detailed execution information (if requested) during query execution.
49        */
50       public void addExecutionInfo(final String message) {
51           executionInfo.add(message);
52       }
53   
54       /**
55        * Used by stores to report what exact position in the store's history it was at when it
56        * executed the query.
57        */
58       public void setPosition(final Position position) {
59           this.position = position;
60       }
61   
62       /**
63        * If detailed execution information was requested in {@link StateQueryRequest#enableExecutionInfo()},
64        * this method returned the execution details for this partition's result.
65        */
66       public List<String> getExecutionInfo() {
67           return executionInfo;
68       }
69   
70       /**
71        * This state partition's exact position in its history when this query was executed. Can be
72        * used in conjunction with subsequent queries via {@link StateQueryRequest#withPositionBound(PositionBound)}.
73        * <p>
74        * Note: stores are encouraged, but not required to set this property.
75        */
76       public Position getPosition() {
77           return position;
78       }
79   
80   }
