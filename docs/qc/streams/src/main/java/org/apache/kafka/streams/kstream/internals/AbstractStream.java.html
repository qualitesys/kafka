39   /*
40    * Any classes (KTable, KStream, etc) extending this class should follow the serde specification precedence ordering as:
41    *
42    * 1) Overridden values via control objects (e.g. Materialized, Serialized, Consumed, etc)
43    * 2) Serdes that can be inferred from the operator itself (e.g. groupBy().count(), where value serde can default to `LongSerde`).
44    * 3) Serde inherited from parent operator if possible (note if the key / value types have been changed, then the corresponding serde cannot be inherited).
45    * 4) Default serde specified in the config.
46    */
47   public abstract class AbstractStream<K, V> {
48   
49       protected final String name;
50       protected final Serde<K> keySerde;
51       protected final Serde<V> valueSerde;
52       protected final Set<String> subTopologySourceNodes;
53       protected final GraphNode graphNode;
54       protected final InternalStreamsBuilder builder;
55   
56       // This copy-constructor will allow to extend KStream
57       // and KTable APIs with new methods without impacting the public interface.
58       public AbstractStream(final AbstractStream<K, V> stream) {
59           this.name = stream.name;
60           this.builder = stream.builder;
61           this.keySerde = stream.keySerde;
62           this.valueSerde = stream.valueSerde;
63           this.subTopologySourceNodes = stream.subTopologySourceNodes;
64           this.graphNode = stream.graphNode;
65       }
66   
67       AbstractStream(final String name,
68                      final Serde<K> keySerde,
69                      final Serde<V> valueSerde,
70                      final Set<String> subTopologySourceNodes,
71                      final GraphNode graphNode,
72                      final InternalStreamsBuilder builder) {
73           if (subTopologySourceNodes == null || subTopologySourceNodes.isEmpty()) {
74               throw new IllegalArgumentException("parameter <sourceNodes> must not be null or empty");
75           }
76   
77           this.name = name;
78           this.builder = builder;
79           this.keySerde = keySerde;
80           this.valueSerde = valueSerde;
81           this.subTopologySourceNodes = subTopologySourceNodes;
82           this.graphNode = graphNode;
83       }
84   
85       // This method allows to expose the InternalTopologyBuilder instance
86       // to subclasses that extend AbstractStream class.
87       protected InternalTopologyBuilder internalTopologyBuilder() {
88           return builder.internalTopologyBuilder;
89       }
90   
91       Set<String> ensureCopartitionWith(final Collection<? extends AbstractStream<K, ?>> otherStreams) {
92           final Set<String> allSourceNodes = new HashSet<>(subTopologySourceNodes);
93           for (final AbstractStream<K, ?> other: otherStreams) {
94               allSourceNodes.addAll(other.subTopologySourceNodes);
95           }
96           builder.internalTopologyBuilder.copartitionSources(allSourceNodes);
97   
98           return allSourceNodes;
99       }

101       static <T2, T1, R> ValueJoiner<T2, T1, R> reverseJoiner(final ValueJoiner<T1, T2, R> joiner) {
102           return (value2, value1) -> joiner.apply(value1, value2);
103       }
104   
105       static <K, T2, T1, R> ValueJoinerWithKey<K, T2, T1, R> reverseJoinerWithKey(final ValueJoinerWithKey<K, T1, T2, R> joiner) {
106           return (key, value2, value1) -> joiner.apply(key, value1, value2);
107       }
108   
109       static <K, V, VR> ValueMapperWithKey<K, V, VR> withKey(final ValueMapper<V, VR> valueMapper) {
110           Objects.requireNonNull(valueMapper, "valueMapper can't be null");
111           return (readOnlyKey, value) -> valueMapper.apply(value);
112       }
113   
114       static <K, V, VR> ValueTransformerWithKeySupplier<K, V, VR> toValueTransformerWithKeySupplier(
115           final ValueTransformerSupplier<V, VR> valueTransformerSupplier) {
116           Objects.requireNonNull(valueTransformerSupplier, "valueTransformerSupplier can't be null");
117           ApiUtils.checkSupplier(valueTransformerSupplier);
118           return new ValueTransformerWithKeySupplier<K, V, VR>() {
119               @Override
120               public ValueTransformerWithKey<K, V, VR> get() {
121                   final ValueTransformer<V, VR> valueTransformer = valueTransformerSupplier.get();
122                   return new ValueTransformerWithKey<K, V, VR>() {
123                       @Override
124                       public void init(final ProcessorContext context) {
125                           valueTransformer.init(context);
126                       }
127   
128                       @Override
129                       public VR transform(final K readOnlyKey, final V value) {
130                           return valueTransformer.transform(value);
131                       }
132   
133                       @Override
134                       public void close() {
135                           valueTransformer.close();
136                       }
137                   };
138               }
139   
140               @Override
141               public Set<StoreBuilder<?>> stores() {
142                   return valueTransformerSupplier.stores();
143               }
144           };
145       }

147       static <K, V1, V2, VR> ValueJoinerWithKey<K, V1, V2, VR> toValueJoinerWithKey(final ValueJoiner<V1, V2, VR> valueJoiner) {
148           Objects.requireNonNull(valueJoiner, "joiner can't be null");
149           return (readOnlyKey, value1, value2) -> valueJoiner.apply(value1, value2);
150       }

154           return keySerde;
155       }

157       public Serde<V> valueSerde() {
158           return valueSerde;
159       }
