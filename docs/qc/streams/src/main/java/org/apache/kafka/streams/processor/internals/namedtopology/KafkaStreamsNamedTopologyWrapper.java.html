157       /**
158        * Add a new NamedTopology to a running Kafka Streams app. If multiple instances of the application are running,
159        * you should inform all of them by calling {@code #addNamedTopology(NamedTopology)} on each client in order for
160        * it to begin processing the new topology.
161        * This method is not purely Async.
162        *
163        * May complete exceptionally (does not actually directly throw) with:
164        *
165        * @throws IllegalArgumentException if this topology name is already in use
166        * @throws IllegalStateException    if streams has not been started or has already shut down
167        * @throws TopologyException        if this topology subscribes to any input topics or pattern already in use
168        */
169       public AddNamedTopologyResult addNamedTopology(final NamedTopology newTopology) {
170           log.info("Adding new NamedTopology: {}", newTopology.name());
171           final KafkaFutureImpl<Void> future = new KafkaFutureImpl<>();
172   
173           if (hasStartedOrFinishedShuttingDown()) {
174               future.completeExceptionally(
175                   new IllegalStateException("Cannot add a NamedTopology while the state is " + super.state)
176               );
177           } else if (getTopologyByName(newTopology.name()).isPresent()) {
178               future.completeExceptionally(
179                   new IllegalArgumentException("Unable to add the new NamedTopology " + newTopology.name() +
180                                                      " as another of the same name already exists")
181               );
182           } else {
183               topologyMetadata.registerAndBuildNewTopology(future, newTopology.internalTopologyBuilder());
184               maybeCompleteFutureIfStillInCREATED(future, "adding topology " + newTopology.name());
185           }
186   
187           return new AddNamedTopologyResult(future);
188       }

190       /**
191        * Remove an existing NamedTopology from a running Kafka Streams app. If multiple instances of the application are
192        * running, you should inform all of them by calling {@code #removeNamedTopology(String)} on each client to ensure
193        * it stops processing the old topology.
194        * This method is not purely Async.
195        *
196        * @param topologyToRemove          name of the topology to be removed
197        * @param resetOffsets              whether to reset the committed offsets for any source topics
198        *
199        * May complete exceptionally (does not actually directly throw) with:
200        *
201        * @throws IllegalArgumentException if this topology name cannot be found
202        * @throws IllegalStateException    if streams has not been started or has already shut down
203        */
204       public RemoveNamedTopologyResult removeNamedTopology(final String topologyToRemove, final boolean resetOffsets) {
205           log.info("Informed to remove topology {} with resetOffsets={} ", topologyToRemove, resetOffsets);
206   
207           final KafkaFutureImpl<Void> removeTopologyFuture = new KafkaFutureImpl<>();
208   
209           if (hasStartedOrFinishedShuttingDown()) {
210               log.error("Attempted to remove topology {} from while the Kafka Streams was in state {}, "
211                             + "topologies cannot be modified if the application has begun or completed shutting down.",
212                         topologyToRemove, state
213               );
214               removeTopologyFuture.completeExceptionally(
215                   new IllegalStateException("Cannot remove a NamedTopology while the state is " + super.state)
216               );
217           } else if (!getTopologyByName(topologyToRemove).isPresent()) {
218               log.error("Attempted to remove unknown topology {}. This application currently contains the"
219                             + "following topologies: {}.", topologyToRemove, topologyMetadata.namedTopologiesView()
220               );
221               removeTopologyFuture.completeExceptionally(
222                   new UnknownTopologyException("Unable to remove topology", topologyToRemove)
223               );
224           }
225   
226           final Set<TopicPartition> partitionsToReset = metadataForLocalThreads()
227               .stream()
228               .flatMap(t -> {
229                   final Set<TaskMetadata> tasks = new HashSet<>(t.activeTasks());
230                   return tasks.stream();
231               })
232               .flatMap(t -> t.topicPartitions().stream())
233               .filter(t -> topologyMetadata.sourceTopicsForTopology(topologyToRemove).contains(t.topic()))
234               .collect(Collectors.toSet());
235   
236           topologyMetadata.unregisterTopology(removeTopologyFuture, topologyToRemove);
237   
238           final boolean skipResetForUnstartedApplication =
239               maybeCompleteFutureIfStillInCREATED(removeTopologyFuture, "removing topology " + topologyToRemove);
240   
241           if (resetOffsets && !skipResetForUnstartedApplication && !partitionsToReset.isEmpty()) {
242               log.info("Resetting offsets for the following partitions of {} removed NamedTopology {}: {}",
243                        removeTopologyFuture.isCompletedExceptionally() ? "unsuccessfully" : "successfully",
244                        topologyToRemove, partitionsToReset
245               );
246               return new RemoveNamedTopologyResult(
247                   removeTopologyFuture,
248                   topologyToRemove,
249                   () -> resetOffsets(partitionsToReset)
250               );
251           } else {
252               return new RemoveNamedTopologyResult(removeTopologyFuture);
253           }
254       }

256       /**
257        * @return  true iff the application is still in CREATED and the future was completed
258        */
259       private boolean maybeCompleteFutureIfStillInCREATED(final KafkaFutureImpl<Void> updateTopologyFuture,
260                                                           final String operation) {
261           if (state == State.CREATED && !updateTopologyFuture.isDone()) {
262               updateTopologyFuture.complete(null);
263               log.info("Completed {} since application has not been started", operation);
264               return true;
265           } else {
266               return false;
267           }

270       private void resetOffsets(final Set<TopicPartition> partitionsToReset) throws StreamsException {
271           // The number of times to retry upon failure
272           int retries = 100;
273           while (true) {
274               try {
275                   final DeleteConsumerGroupOffsetsResult deleteOffsetsResult = adminClient.deleteConsumerGroupOffsets(
276                       applicationConfigs.getString(StreamsConfig.APPLICATION_ID_CONFIG),
277                       partitionsToReset);
278                   deleteOffsetsResult.all().get();
279                   log.info("Successfully completed resetting offsets.");
280                   break;
281               } catch (final InterruptedException ex) {
282                   ex.printStackTrace();
283                   log.error("Offset reset failed.", ex);
284                   throw new StreamsException(ex);
285               } catch (final ExecutionException ex) {
286                   final Throwable error = ex.getCause() != null ? ex.getCause() : ex;
287   
288                   if (error instanceof GroupSubscribedToTopicException &&
289                       error.getMessage()
290                           .equals("Deleting offsets of a topic is forbidden while the consumer group is actively subscribed to it.")) {
291                       log.debug("Offset reset failed, there may be other nodes which have not yet finished removing this topology", error);
292                   } else if (error instanceof GroupIdNotFoundException) {
293                       log.info("The offsets have been reset by another client or the group has been deleted, no need to retry further.");
294                       break;
295                   } else {
296                       if (--retries > 0) {
297                           log.error("Offset reset failed, retries remaining: " + retries, error);
298                       } else {
299                           log.error("Offset reset failed, no retries remaining.", error);
300                           throw new StreamsException(error);
301                       }
302                   }
303               }
304               try {
305                   Thread.sleep(100);
306               } catch (final InterruptedException ex) {
307                   ex.printStackTrace();
308               }
309           }
310       }
