17   package org.apache.kafka.streams.kstream;
18   
19   import org.apache.kafka.streams.KafkaStreams;
20   import org.apache.kafka.streams.KeyValue;
21   import org.apache.kafka.streams.StreamsBuilder;
22   import org.apache.kafka.streams.state.ReadOnlyKeyValueStore;
23   
24   /**
25    * {@code GlobalKTable} is an abstraction of a <i>changelog stream</i> from a primary-keyed table.
26    * Each record in this changelog stream is an update on the primary-keyed table with the record key as the primary key.
27    * <p>
28    * {@code GlobalKTable} can only be used as right-hand side input for {@link KStream stream}-table joins.
29    * <p>
30    * In contrast to a {@link KTable} that is partitioned over all {@link KafkaStreams} instances, a {@code GlobalKTable}
31    * is fully replicated per {@link KafkaStreams} instance.
32    * Every partition of the underlying topic is consumed by each {@code GlobalKTable}, such that the full set of data is
33    * available in every {@link KafkaStreams} instance.
34    * This provides the ability to perform joins with {@link KStream} without having to repartition the input stream.
35    * All joins with the {@code GlobalKTable} require that a {@link KeyValueMapper} is provided that can map from the
36    * {@link KeyValue} of the left hand side {@link KStream} to the key of the right hand side {@code GlobalKTable}.
37    * <p>
38    * A {@code GlobalKTable} is created via a {@link StreamsBuilder}. For example:
39    * <pre>{@code
40    * builder.globalTable("topic-name", "queryable-store-name");
41    * }</pre>
42    * all {@code GlobalKTable}s are backed by a {@link ReadOnlyKeyValueStore} and are therefore queryable via the
43    * interactive queries API.
44    * For example:
45    * <pre>{@code
46    * final GlobalKTable globalOne = builder.globalTable("g1", "g1-store");
47    * final GlobalKTable globalTwo = builder.globalTable("g2", "g2-store");
48    * ...
49    * final KafkaStreams streams = ...;
50    * streams.start()
51    * ...
52    * ReadOnlyKeyValueStore view = streams.store("g1-store", QueryableStoreTypes.timestampedKeyValueStore());
53    * view.get(key); // can be done on any key, as all keys are present
54    *}</pre>
55    * Note that in contrast to {@link KTable} a {@code GlobalKTable}'s state holds a full copy of the underlying topic,
56    * thus all keys can be queried locally.
57    * <p>
58    * Records from the source topic that have null keys are dropped.
59    *
60    * @param <K> Type of primary keys
61    * @param <V> Type of value changes
62    * @see KTable
63    * @see StreamsBuilder#globalTable(String)
64    * @see KStream#join(GlobalKTable, KeyValueMapper, ValueJoiner)
65    * @see KStream#leftJoin(GlobalKTable, KeyValueMapper, ValueJoiner)
66    */
67   public interface GlobalKTable<K, V> {
68       /**
69        * Get the name of the local state store that can be used to query this {@code GlobalKTable}.
70        *
71        * @return the underlying state store name, or {@code null} if this {@code GlobalKTable} cannot be queried.
72        */
73       String queryableStoreName();
74   }
