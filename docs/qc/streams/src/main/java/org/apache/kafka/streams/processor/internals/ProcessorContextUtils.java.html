26   /**
27    * This class bridges the gap for components that _should_ be compatible with
28    * the public ProcessorContext interface, but have come to depend on features
29    * in InternalProcessorContext. In theory, all the features adapted here could
30    * migrate to the public interface, so each method in this class should reference
31    * the ticket that would ultimately obviate it.
32    */
33   public final class ProcessorContextUtils {
34   
35       private ProcessorContextUtils() {}
36   
37       /**
38        * Note that KIP-622 would move currentSystemTimeMs to ProcessorContext,
39        * removing the need for this method.
40        */
41       public static long currentSystemTime(final ProcessorContext context) {
42           return context.currentSystemTimeMs();
43       }

47        */
48       public static StreamsMetricsImpl getMetricsImpl(final ProcessorContext context) {
49           return (StreamsMetricsImpl) context.metrics();
50       }

53        * Should be removed as part of KAFKA-10217
54        */
55       public static StreamsMetricsImpl getMetricsImpl(final StateStoreContext context) {
56           return (StreamsMetricsImpl) context.metrics();
57       }

59       public static String changelogFor(final ProcessorContext context, final String storeName, final Boolean newChangelogTopic) {
60           final String prefix = getPrefix(context.appConfigs(), context.applicationId());
61           if (context instanceof InternalProcessorContext && !newChangelogTopic) {
62               final String changelogTopic = ((InternalProcessorContext) context).changelogFor(storeName);
63               if (changelogTopic != null)
64                   return changelogTopic;
65   
66           }
67           return ProcessorStateManager.storeChangelogTopic(prefix, storeName, context.taskId().topologyName());
68       }
69   
70       public static String changelogFor(final StateStoreContext context, final String storeName, final Boolean newChangelogTopic) {
71           final String prefix = getPrefix(context.appConfigs(), context.applicationId());
72           if (context instanceof InternalProcessorContext && !newChangelogTopic) {
73               final String changelogTopic = ((InternalProcessorContext) context).changelogFor(storeName);
74               if (changelogTopic != null)
75                   return changelogTopic;
76   
77           }
78           return ProcessorStateManager.storeChangelogTopic(prefix, storeName, context.taskId().topologyName());
79       }
80   
81       public static String getPrefix(final Map<String, Object> configs, final String applicationId) {
82           if (configs == null) {
83               return applicationId;
84           } else {
85               return StreamsConfig.InternalConfig.getString(
86                   configs,
87                   StreamsConfig.InternalConfig.TOPIC_PREFIX_ALTERNATIVE,
88                   applicationId
89               );
90           }
91       }
92   
93       public static InternalProcessorContext asInternalProcessorContext(final ProcessorContext context) {
94           if (context instanceof InternalProcessorContext) {
95               return (InternalProcessorContext) context;
96           } else {
97               throw new IllegalArgumentException(
98                   "This component requires internal features of Kafka Streams and must be disabled for unit tests."
99               );
100           }
101       }

106           } else {

110           }
