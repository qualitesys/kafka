17   package org.apache.kafka.streams.state.internals;
18   
19   import org.apache.kafka.common.serialization.LongSerializer;
20   import org.apache.kafka.common.serialization.Serializer;
21   import org.apache.kafka.streams.kstream.internals.WrappingNullableSerializer;
22   import org.apache.kafka.streams.processor.internals.SerdeGetter;
23   import org.apache.kafka.streams.state.ValueAndTimestamp;
24   
25   import java.nio.ByteBuffer;
26   import java.util.Map;
27   import java.util.Objects;
28   
29   import static org.apache.kafka.streams.kstream.internals.WrappingNullableUtils.initNullableSerializer;
30   
31   public class ValueAndTimestampSerializer<V> implements WrappingNullableSerializer<ValueAndTimestamp<V>, Void, V> {
32       public final Serializer<V> valueSerializer;
33       private final Serializer<Long> timestampSerializer;
34   
35       ValueAndTimestampSerializer(final Serializer<V> valueSerializer) {
36           Objects.requireNonNull(valueSerializer);
37           this.valueSerializer = valueSerializer;
38           timestampSerializer = new LongSerializer();
39       }
40   
41       public static boolean valuesAreSameAndTimeIsIncreasing(final byte[] oldRecord, final byte[] newRecord) {
42           if (oldRecord == newRecord) {
43               // same reference, so they are trivially the same (might both be null)
44               return true;
45           } else if (oldRecord == null || newRecord == null) {
46               // only one is null, so they cannot be the same
47               return false;
48           } else if (newRecord.length != oldRecord.length) {
49               // they are different length, so they cannot be the same
50               return false;
51           } else if (timeIsDecreasing(oldRecord, newRecord)) {
52               // the record time represents the beginning of the validity interval, so if the time
53               // moves backwards, we need to do the update regardless of whether the value has changed
54               return false;
55           } else {
56               // all other checks have fallen through, so we actually compare the binary data of the two values
57               return valuesAreSame(oldRecord, newRecord);
58           }
59       }
60   
61       @Override
62       public void configure(final Map<String, ?> configs,
63                             final boolean isKey) {
64           valueSerializer.configure(configs, isKey);
65           timestampSerializer.configure(configs, isKey);
66       }
67   
68       @Override
69       public byte[] serialize(final String topic,
70                               final ValueAndTimestamp<V> data) {
71           if (data == null) {
72               return null;
73           }
74           return serialize(topic, data.value(), data.timestamp());
75       }
76   
77       public byte[] serialize(final String topic,
78                               final V data,
79                               final long timestamp) {
80           if (data == null) {
81               return null;
82           }
83           final byte[] rawValue = valueSerializer.serialize(topic, data);
84   
85           // Since we can't control the result of the internal serializer, we make sure that the result
86           // is not null as well.
87           // Serializing non-null values to null can be useful when working with Optional-like values
88           // where the Optional.empty case is serialized to null.
89           // See the discussion here: https://github.com/apache/kafka/pull/7679
90           if (rawValue == null) {
91               return null;
92           }
93   
94           final byte[] rawTimestamp = timestampSerializer.serialize(topic, timestamp);
95           return ByteBuffer
96               .allocate(rawTimestamp.length + rawValue.length)
97               .put(rawTimestamp)
98               .put(rawValue)
99               .array();
100       }
101   
102       @Override
103       public void close() {
104           valueSerializer.close();
105           timestampSerializer.close();
106       }
107   
108       private static boolean timeIsDecreasing(final byte[] oldRecord, final byte[] newRecord) {
109           return extractTimestamp(newRecord) <= extractTimestamp(oldRecord);
110       }
111   
112       private static long extractTimestamp(final byte[] bytes) {
113           final byte[] timestampBytes = new byte[Long.BYTES];
114           System.arraycopy(bytes, 0, timestampBytes, 0, Long.BYTES);
115           return ByteBuffer.wrap(timestampBytes).getLong();
116       }
117   
118       private static boolean valuesAreSame(final byte[] left, final byte[] right) {
119           for (int i = Long.BYTES; i < left.length; i++) {
120               if (left[i] != right[i]) {
121                   return false;
122               }
123           }
124           return true;
125       }
126   
127       @Override
128       public void setIfUnset(final SerdeGetter getter) {
129           // ValueAndTimestampSerializer never wraps a null serializer (or configure would throw),
130           // but it may wrap a serializer that itself wraps a null serializer.
131           initNullableSerializer(valueSerializer, getter);
132       }
133   }
