57       static DeserializationResult deserializeV0(final ConsumerRecord<byte[], byte[]> record,
58                                                  final Bytes key,
59                                                  final byte[] previousBufferedValue) {
60   
61           final ByteBuffer timeAndValue = ByteBuffer.wrap(record.value());
62           final long time = timeAndValue.getLong();
63           final byte[] changelogValue = new byte[record.value().length - 8];
64           timeAndValue.get(changelogValue);
65   
66           final Change<byte[]> change = requireNonNull(FullChangeSerde.decomposeLegacyFormattedArrayIntoChangeArrays(changelogValue));
67   
68           final ProcessorRecordContext recordContext = new ProcessorRecordContext(
69               record.timestamp(),
70               record.offset(),
71               record.partition(),
72               record.topic(),
73               record.headers()
74           );
75   
76           return new DeserializationResult(
77               time,
78               key,
79               new BufferValue(
80                   previousBufferedValue == null ? change.oldValue : previousBufferedValue,
81                   change.oldValue,
82                   change.newValue,
83                   recordContext
84               )
85           );
86       }

92           final long time = timeAndValue.getLong();
93           final byte[] changelogValue = new byte[record.value().length - 8];

107               )

111       static DeserializationResult duckTypeV2(final ConsumerRecord<byte[], byte[]> record, final Bytes key) {
112           DeserializationResult deserializationResult = null;
113           RuntimeException v2DeserializationException = null;
114           RuntimeException v3DeserializationException = null;
115           try {
116               deserializationResult = deserializeV2(record, key);
117           } catch (final RuntimeException e) {
118               v2DeserializationException = e;
119           }
120           // versions 2.4.0, 2.4.1, and 2.5.0 would have erroneously encoded a V3 record with the
121           // V2 header, so we'll try duck-typing to see if this is decodable as V3
122           if (deserializationResult == null) {
123               try {
124                   deserializationResult = deserializeV3(record, key);
125               } catch (final RuntimeException e) {
126                   v3DeserializationException = e;
127               }
128           }
129   
130           if (deserializationResult == null) {
131               // ok, it wasn't V3 either. Throw both exceptions:
132               final RuntimeException exception =
133                   new RuntimeException("Couldn't deserialize record as v2 or v3: " + record,
134                                        v2DeserializationException);
135               exception.addSuppressed(v3DeserializationException);
136               throw exception;
137           }
138           return deserializationResult;
139       }

141       private static DeserializationResult deserializeV2(final ConsumerRecord<byte[], byte[]> record,
142                                                          final Bytes key) {
143           final ByteBuffer valueAndTime = ByteBuffer.wrap(record.value());
144           final ContextualRecord contextualRecord = ContextualRecord.deserialize(valueAndTime);
145           final Change<byte[]> change = requireNonNull(FullChangeSerde.decomposeLegacyFormattedArrayIntoChangeArrays(contextualRecord.value()));
146           final byte[] priorValue = Utils.getNullableSizePrefixedArray(valueAndTime);
147           final long time = valueAndTime.getLong();
148           final BufferValue bufferValue = new BufferValue(priorValue, change.oldValue, change.newValue, contextualRecord.recordContext());
149           return new DeserializationResult(time, key, bufferValue);
150       }
151   
152       static DeserializationResult deserializeV3(final ConsumerRecord<byte[], byte[]> record, final Bytes key) {
153           final ByteBuffer valueAndTime = ByteBuffer.wrap(record.value());
154           final BufferValue bufferValue = BufferValue.deserialize(valueAndTime);
155           final long time = valueAndTime.getLong();
156           return new DeserializationResult(time, key, bufferValue);
157       }
