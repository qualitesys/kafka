18   package org.apache.kafka.streams.state.internals;
19   
20   import org.apache.kafka.common.serialization.Deserializer;
21   import org.apache.kafka.common.utils.Bytes;
22   import org.apache.kafka.streams.KeyValue;
23   import org.apache.kafka.streams.kstream.Windowed;
24   import org.apache.kafka.streams.state.KeyValueIterator;
25   import org.apache.kafka.streams.state.StateSerdes;
26   
27   class MergedSortedCacheWindowStoreKeyValueIterator
28       extends AbstractMergedSortedCacheStoreIterator<Windowed<Bytes>, Windowed<Bytes>, byte[], byte[]> {
29   
30       private final StateSerdes<Bytes, byte[]> serdes;
31       private final long windowSize;
32       private final SegmentedCacheFunction cacheFunction;
33       private final StoreKeyToWindowKey storeKeyToWindowKey;
34       private final WindowKeyToBytes windowKeyToBytes;
35   
36       MergedSortedCacheWindowStoreKeyValueIterator(
37           final PeekingKeyValueIterator<Bytes, LRUCacheEntry> filteredCacheIterator,
38           final KeyValueIterator<Windowed<Bytes>, byte[]> underlyingIterator,
39           final StateSerdes<Bytes, byte[]> serdes,
40           final long windowSize,
41           final SegmentedCacheFunction cacheFunction,
42           final boolean forward
43       ) {
44           this(filteredCacheIterator, underlyingIterator, serdes,
45               windowSize, cacheFunction, forward, WindowKeySchema::fromStoreKey, WindowKeySchema::toStoreKeyBinary);
46       }
47   
48       MergedSortedCacheWindowStoreKeyValueIterator(
49           final PeekingKeyValueIterator<Bytes, LRUCacheEntry> filteredCacheIterator,
50           final KeyValueIterator<Windowed<Bytes>, byte[]> underlyingIterator,
51           final StateSerdes<Bytes, byte[]> serdes,
52           final long windowSize,
53           final SegmentedCacheFunction cacheFunction,
54           final boolean forward,
55           final StoreKeyToWindowKey storeKeyToWindowKey,
56           final WindowKeyToBytes windowKeyToBytes
57       ) {
58           super(filteredCacheIterator, underlyingIterator, forward);
59           this.serdes = serdes;
60           this.windowSize = windowSize;
61           this.cacheFunction = cacheFunction;
62           this.storeKeyToWindowKey = storeKeyToWindowKey;
63           this.windowKeyToBytes = windowKeyToBytes;
64       }
65   
66       @Override
67       Windowed<Bytes> deserializeStoreKey(final Windowed<Bytes> key) {
68           return key;
69       }
70   
71       @Override
72       KeyValue<Windowed<Bytes>, byte[]> deserializeStorePair(final KeyValue<Windowed<Bytes>, byte[]> pair) {
73           return pair;
74       }
75   
76       @Override
77       Windowed<Bytes> deserializeCacheKey(final Bytes cacheKey) {
78           final byte[] binaryKey = cacheFunction.key(cacheKey).get();
79           return storeKeyToWindowKey.toWindowKey(binaryKey, windowSize, serdes.keyDeserializer(), serdes.topic());
80       }
81   
82       @Override
83       byte[] deserializeCacheValue(final LRUCacheEntry cacheEntry) {
84           return cacheEntry.value();
85       }
86   
87       @Override
88       int compare(final Bytes cacheKey, final Windowed<Bytes> storeKey) {
89           final Bytes storeKeyBytes = windowKeyToBytes.toBytes(storeKey.key(), storeKey.window().start(), 0);
90           return cacheFunction.compareSegmentedKeys(cacheKey, storeKeyBytes);
91       }
92   
93       @FunctionalInterface
94       interface StoreKeyToWindowKey {
95           Windowed<Bytes> toWindowKey(final byte[] binaryKey, final long windowSize, final Deserializer<Bytes> deserializer, final String topic);
96       }
97   
98       @FunctionalInterface
99       interface WindowKeyToBytes {
100           Bytes toBytes(final Bytes key, final long windowStart, final int seqNum);
101       }
102   }
