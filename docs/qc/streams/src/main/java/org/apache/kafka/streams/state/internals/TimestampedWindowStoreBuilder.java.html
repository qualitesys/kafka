43   public class TimestampedWindowStoreBuilder<K, V>
44       extends AbstractStoreBuilder<K, ValueAndTimestamp<V>, TimestampedWindowStore<K, V>> {
45   
46       private static final Logger LOG = LoggerFactory.getLogger(TimestampedWindowStoreBuilder.class);
47   
48       private final WindowBytesStoreSupplier storeSupplier;
49   
50       public TimestampedWindowStoreBuilder(final WindowBytesStoreSupplier storeSupplier,
51                                            final Serde<K> keySerde,
52                                            final Serde<V> valueSerde,
53                                            final Time time) {
54           super(storeSupplier.name(), keySerde, valueSerde == null ? null : new ValueAndTimestampSerde<>(valueSerde), time);
55           Objects.requireNonNull(storeSupplier, "storeSupplier can't be null");
56           Objects.requireNonNull(storeSupplier.metricsScope(), "storeSupplier's metricsScope can't be null");
57           this.storeSupplier = storeSupplier;
58       }
59   
60       @Override
61       public TimestampedWindowStore<K, V> build() {
62           WindowStore<Bytes, byte[]> store = storeSupplier.get();
63           if (!(store instanceof TimestampedBytesStore)) {
64               if (store.persistent()) {
65                   store = new WindowToTimestampedWindowByteStoreAdapter(store);
66               } else {
67                   store = new InMemoryTimestampedWindowStoreMarker(store);
68               }
69           }
70           if (storeSupplier.retainDuplicates() && enableCaching) {
71               LOG.warn("Disabling caching for {} since store was configured to retain duplicates", storeSupplier.name());
72               enableCaching = false;
73           }
74   
75           return new MeteredTimestampedWindowStore<>(
76               maybeWrapCaching(maybeWrapLogging(store)),
77               storeSupplier.windowSize(),
78               storeSupplier.metricsScope(),
79               time,
80               keySerde,
81               valueSerde);
82       }
83   
84       private WindowStore<Bytes, byte[]> maybeWrapCaching(final WindowStore<Bytes, byte[]> inner) {
85           if (!enableCaching) {
86               return inner;
87           }
88   
89           final boolean isTimeOrdered = isTimeOrderedStore(inner);
90           if (isTimeOrdered) {
91               return new TimeOrderedCachingWindowStore(
92                   inner,
93                   storeSupplier.windowSize(),
94                   storeSupplier.segmentIntervalMs());
95           }
96   
97           return new CachingWindowStore(
98               inner,
99               storeSupplier.windowSize(),
100               storeSupplier.segmentIntervalMs());
101       }
102   
103       private boolean isTimeOrderedStore(final StateStore stateStore) {
104           if (stateStore instanceof RocksDBTimeOrderedWindowStore) {
105               return true;
106           }
107           if (stateStore instanceof WrappedStateStore) {
108               return isTimeOrderedStore(((WrappedStateStore) stateStore).wrapped());
109           }
110           return false;
111       }
112   
113       private WindowStore<Bytes, byte[]> maybeWrapLogging(final WindowStore<Bytes, byte[]> inner) {
114           if (!enableLogging) {
115               return inner;
116           }
117           return new ChangeLoggingTimestampedWindowBytesStore(inner, storeSupplier.retainDuplicates());
118       }
119   
120       public long retentionPeriod() {
121           return storeSupplier.retentionPeriod();
122       }

125       private final static class InMemoryTimestampedWindowStoreMarker
126           implements WindowStore<Bytes, byte[]>, TimestampedBytesStore {
127   
128           private final WindowStore<Bytes, byte[]> wrapped;
129   
130           private InMemoryTimestampedWindowStoreMarker(final WindowStore<Bytes, byte[]> wrapped) {
131               if (wrapped.persistent()) {
132                   throw new IllegalArgumentException("Provided store must not be a persistent store, but it is.");
133               }
134               this.wrapped = wrapped;
135           }

229           @Override
230           public <R> QueryResult<R> query(final Query<R> query,
231               final PositionBound positionBound,
232               final QueryConfig config) {
233   
234               final long start = config.isCollectExecutionInfo() ? System.nanoTime() : -1L;
235               final QueryResult<R> result = wrapped.query(query, positionBound, config);
236               if (config.isCollectExecutionInfo()) {
237                   final long end = System.nanoTime();
238                   result.addExecutionInfo("Handled in " + getClass() + " in " + (end - start) + "ns");
239               }
240               return result;
241           }

246           }

250               return wrapped.name();
251           }

254           public boolean persistent() {
255               return false;
256           }
257       }
