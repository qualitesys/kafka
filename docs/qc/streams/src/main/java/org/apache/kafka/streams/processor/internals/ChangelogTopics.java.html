17   package org.apache.kafka.streams.processor.internals;
18   
19   import org.apache.kafka.common.TopicPartition;
20   import org.apache.kafka.common.utils.LogContext;
21   import org.apache.kafka.streams.processor.TaskId;
22   import org.apache.kafka.streams.processor.internals.InternalTopologyBuilder.TopicsInfo;
23   import org.apache.kafka.streams.processor.internals.TopologyMetadata.Subtopology;
24   
25   import org.slf4j.Logger;
26   
27   import java.util.Collections;
28   import java.util.HashMap;
29   import java.util.HashSet;
30   import java.util.Map;
31   import java.util.Set;
32   import java.util.stream.Collectors;
33   
34   import static org.apache.kafka.streams.processor.internals.assignment.StreamsAssignmentProtocolVersions.UNKNOWN;
35   
36   public class ChangelogTopics {
37   
38       private final InternalTopicManager internalTopicManager;
39       private final Map<Subtopology, TopicsInfo> topicGroups;
40       private final Map<Subtopology, Set<TaskId>> tasksForTopicGroup;
41       private final Map<TaskId, Set<TopicPartition>> changelogPartitionsForStatefulTask = new HashMap<>();
42       private final Map<TaskId, Set<TopicPartition>> preExistingChangelogPartitionsForTask = new HashMap<>();
43       private final Set<TopicPartition> preExistingNonSourceTopicBasedChangelogPartitions = new HashSet<>();
44       private final Set<String> sourceTopicBasedChangelogTopics = new HashSet<>();
45       private final Set<TopicPartition> preExistingSourceTopicBasedChangelogPartitions = new HashSet<>();
46       private final Logger log;
47   
48       public ChangelogTopics(final InternalTopicManager internalTopicManager,
49                              final Map<Subtopology, TopicsInfo> topicGroups,
50                              final Map<Subtopology, Set<TaskId>> tasksForTopicGroup,
51                              final String logPrefix) {
52           this.internalTopicManager = internalTopicManager;
53           this.topicGroups = topicGroups;
54           this.tasksForTopicGroup = tasksForTopicGroup;
55           final LogContext logContext = new LogContext(logPrefix);
56           log = logContext.logger(getClass());
57       }
58   
59       public void setup() {
60           // add tasks to state change log topic subscribers
61           final Map<String, InternalTopicConfig> changelogTopicMetadata = new HashMap<>();
62           for (final Map.Entry<Subtopology, TopicsInfo> entry : topicGroups.entrySet()) {
63               final Subtopology subtopology = entry.getKey();
64               final TopicsInfo topicsInfo = entry.getValue();
65   
66               final Set<TaskId> topicGroupTasks = tasksForTopicGroup.get(subtopology);
67               if (topicGroupTasks == null) {
68                   log.debug("No tasks found for subtopology {}", subtopology);
69                   continue;
70               } else if (topicsInfo.stateChangelogTopics.isEmpty()) {
71                   continue;
72               }
73   
74               for (final TaskId task : topicGroupTasks) {
75                   final Set<TopicPartition> changelogTopicPartitions = topicsInfo.stateChangelogTopics
76                       .keySet()
77                       .stream()
78                       .map(topic -> new TopicPartition(topic, task.partition()))
79                       .collect(Collectors.toSet());
80                   changelogPartitionsForStatefulTask.put(task, changelogTopicPartitions);
81               }
82   
83               for (final InternalTopicConfig topicConfig : topicsInfo.nonSourceChangelogTopics()) {
84                   // the expected number of partitions is the max value of TaskId.partition + 1
85                   int numPartitions = UNKNOWN;
86                   for (final TaskId task : topicGroupTasks) {
87                       if (numPartitions < task.partition() + 1) {
88                           numPartitions = task.partition() + 1;
89                       }
90                   }
91                   topicConfig.setNumberOfPartitions(numPartitions);
92                   changelogTopicMetadata.put(topicConfig.name(), topicConfig);
93               }
94               sourceTopicBasedChangelogTopics.addAll(topicsInfo.sourceTopicChangelogs());
95           }
96   
97           final Set<String> newlyCreatedChangelogTopics = internalTopicManager.makeReady(changelogTopicMetadata);
98           log.debug("Created state changelog topics {} from the parsed topology.", changelogTopicMetadata.values());
99   
100           for (final Map.Entry<TaskId, Set<TopicPartition>> entry : changelogPartitionsForStatefulTask.entrySet()) {
101               final TaskId taskId = entry.getKey();
102               final Set<TopicPartition> topicPartitions = entry.getValue();
103               for (final TopicPartition topicPartition : topicPartitions) {
104                   if (!newlyCreatedChangelogTopics.contains(topicPartition.topic())) {
105                       preExistingChangelogPartitionsForTask.computeIfAbsent(taskId, task -> new HashSet<>()).add(topicPartition);
106                       if (!sourceTopicBasedChangelogTopics.contains(topicPartition.topic())) {
107                           preExistingNonSourceTopicBasedChangelogPartitions.add(topicPartition);
108                       } else {
109                           preExistingSourceTopicBasedChangelogPartitions.add(topicPartition);
110                       }
111                   }
112               }
113           }
114       }
115   
116       public Set<TopicPartition> preExistingNonSourceTopicBasedPartitions() {
117           return Collections.unmodifiableSet(preExistingNonSourceTopicBasedChangelogPartitions);
118       }
119   
120       public Set<TopicPartition> preExistingPartitionsFor(final TaskId taskId) {
121           if (preExistingChangelogPartitionsForTask.containsKey(taskId)) {
122               return Collections.unmodifiableSet(preExistingChangelogPartitionsForTask.get(taskId));
123           }
124           return Collections.emptySet();
125       }
126   
127       public Set<TopicPartition> preExistingSourceTopicBasedPartitions() {
128           return Collections.unmodifiableSet(preExistingSourceTopicBasedChangelogPartitions);
129       }
130   
131       public Set<TaskId> statefulTaskIds() {
132           return Collections.unmodifiableSet(changelogPartitionsForStatefulTask.keySet());
133       }
134   }
