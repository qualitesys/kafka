26   import java.util.Objects;
27   
28   import static org.apache.kafka.streams.internals.ApiUtils.prepareMillisCheckFailMsgPrefix;

30   /**
31    * The unlimited window specifications used for aggregations.
32    * <p>
33    * An unlimited time window is also called landmark window.
34    * It has a fixed starting point while its window end is defined as infinite.
35    * With this regard, it is a fixed-size window with infinite window size.
36    * <p>
37    * For time semantics, see {@link TimestampExtractor}.
38    *
39    * @see TimeWindows
40    * @see SessionWindows
41    * @see JoinWindows
42    * @see KGroupedStream#windowedBy(Windows)
43    * @see TimestampExtractor
44    */
45   public final class UnlimitedWindows extends Windows<UnlimitedWindow> {
46   
47       private static final long DEFAULT_START_TIMESTAMP_MS = 0L;
48   
49       /** The start timestamp of the window. */
50       @SuppressWarnings("WeakerAccess")
51       public final long startMs;
52   
53       private UnlimitedWindows(final long startMs) {
54           this.startMs = startMs;
55       }
56   
57       /**
58        * Return an unlimited window starting at timestamp zero.
59        */
60       public static UnlimitedWindows of() {
61           return new UnlimitedWindows(DEFAULT_START_TIMESTAMP_MS);
62       }
63   
64       /**
65        * Return a new unlimited window for the specified start timestamp.
66        *
67        * @param start the window start time
68        * @return a new unlimited window that starts at {@code start}
69        * @throws IllegalArgumentException if the start time is negative or can't be represented as {@code long milliseconds}
70        */
71       public UnlimitedWindows startOn(final Instant start) throws IllegalArgumentException {
72           final String msgPrefix = prepareMillisCheckFailMsgPrefix(start, "start");
73           final long startMs = ApiUtils.validateMillisecondInstant(start, msgPrefix);
74           if (startMs < 0) {
75               throw new IllegalArgumentException("Window start time (startMs) cannot be negative.");
76           }
77           return new UnlimitedWindows(startMs);
78       }
79   
80       @Override
81       public Map<Long, UnlimitedWindow> windowsFor(final long timestamp) {
82           // always return the single unlimited window
83   
84           // we cannot use Collections.singleMap since it does not support remove()
85           final Map<Long, UnlimitedWindow> windows = new HashMap<>();
86           if (timestamp >= startMs) {
87               windows.put(startMs, new UnlimitedWindow(startMs));
88           }
89           return windows;
90       }
91   
92       /**
93        * {@inheritDoc}
94        * As unlimited windows have conceptually infinite size, this methods just returns {@link Long#MAX_VALUE}.
95        *
96        * @return the size of the specified windows which is {@link Long#MAX_VALUE}
97        */
98       @Override
99       public long size() {
100           return Long.MAX_VALUE;
101       }
102   
103       @Override
104       public long gracePeriodMs() {
105           return 0L;
106       }
107   
108       @Override
109       public boolean equals(final Object o) {
110           if (this == o) {
111               return true;
112           }
113           if (o == null || getClass() != o.getClass()) {
114               return false;
115           }
116           final UnlimitedWindows that = (UnlimitedWindows) o;
117           return startMs == that.startMs;
118       }
119   
120       @Override
121       public int hashCode() {
122           return Objects.hash(startMs);
123       }
124   
125       @Override
126       public String toString() {
127           return "UnlimitedWindows{" +
128               "startMs=" + startMs +
129               '}';
130       }
131   }
