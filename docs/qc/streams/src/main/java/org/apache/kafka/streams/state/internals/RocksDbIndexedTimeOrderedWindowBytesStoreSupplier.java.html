28   public class RocksDbIndexedTimeOrderedWindowBytesStoreSupplier implements WindowBytesStoreSupplier {
29       public enum WindowStoreTypes {
30           DEFAULT_WINDOW_STORE,
31           INDEXED_WINDOW_STORE
32       }
33   
34       private final String name;
35       private final long retentionPeriod;
36       private final long segmentInterval;
37       private final long windowSize;
38       private final boolean retainDuplicates;
39       private final WindowStoreTypes windowStoreType;
40   
41       public static RocksDbIndexedTimeOrderedWindowBytesStoreSupplier create(final String name,
42                                                                              final Duration retentionPeriod,
43                                                                              final Duration windowSize,
44                                                                              final boolean retainDuplicates,
45                                                                              final boolean hasIndex) {
46           Objects.requireNonNull(name, "name cannot be null");
47           final String rpMsgPrefix = prepareMillisCheckFailMsgPrefix(retentionPeriod, "retentionPeriod");
48           final long retentionMs = validateMillisecondDuration(retentionPeriod, rpMsgPrefix);
49           final String wsMsgPrefix = prepareMillisCheckFailMsgPrefix(windowSize, "windowSize");
50           final long windowSizeMs = validateMillisecondDuration(windowSize, wsMsgPrefix);
51   
52           final long defaultSegmentInterval = Math.max(retentionMs / 2, 60_000L);
53   
54           if (retentionMs < 0L) {
55               throw new IllegalArgumentException("retentionPeriod cannot be negative");
56           }
57           if (windowSizeMs < 0L) {
58               throw new IllegalArgumentException("windowSize cannot be negative");
59           }
60           if (defaultSegmentInterval < 1L) {
61               throw new IllegalArgumentException("segmentInterval cannot be zero or negative");
62           }
63           if (windowSizeMs > retentionMs) {
64               throw new IllegalArgumentException("The retention period of the window store "
65                   + name + " must be no smaller than its window size. Got size=["
66                   + windowSizeMs + "], retention=[" + retentionMs + "]");
67           }
68   
69           return new RocksDbIndexedTimeOrderedWindowBytesStoreSupplier(name, retentionMs,
70               defaultSegmentInterval, windowSizeMs, retainDuplicates, hasIndex);
71       }
72   
73       public RocksDbIndexedTimeOrderedWindowBytesStoreSupplier(final String name,
74                                              final long retentionPeriod,
75                                              final long segmentInterval,
76                                              final long windowSize,
77                                              final boolean retainDuplicates,
78                                              final boolean withIndex) {
79           this(name, retentionPeriod, segmentInterval, windowSize, retainDuplicates,
80               withIndex
81                   ? WindowStoreTypes.INDEXED_WINDOW_STORE
82                   : WindowStoreTypes.DEFAULT_WINDOW_STORE);
83       }
84   
85       public RocksDbIndexedTimeOrderedWindowBytesStoreSupplier(final String name,
86                                              final long retentionPeriod,
87                                              final long segmentInterval,
88                                              final long windowSize,
89                                              final boolean retainDuplicates,
90                                              final WindowStoreTypes windowStoreType) {
91           this.name = name;
92           this.retentionPeriod = retentionPeriod;
93           this.segmentInterval = segmentInterval;
94           this.windowSize = windowSize;
95           this.retainDuplicates = retainDuplicates;
96           this.windowStoreType = windowStoreType;
97       }

99       @Override
100       public String name() {
101           return name;
102       }
103   
104       @Override
105       public WindowStore<Bytes, byte[]> get() {
106           switch (windowStoreType) {
107               case DEFAULT_WINDOW_STORE:
108                   return new RocksDBTimeOrderedWindowStore(
109                       new RocksDBTimeOrderedWindowSegmentedBytesStore(
110                           name,
111                           metricsScope(),
112                           retentionPeriod,
113                           segmentInterval,
114                           false),
115                       retainDuplicates,
116                       windowSize);
117               case INDEXED_WINDOW_STORE:
118                   return new RocksDBTimeOrderedWindowStore(
119                       new RocksDBTimeOrderedWindowSegmentedBytesStore(
120                           name,
121                           metricsScope(),
122                           retentionPeriod,
123                           segmentInterval,
124                           true),
125                       retainDuplicates,
126                       windowSize);
127               default:
128                   throw new IllegalArgumentException("invalid window store type: " + windowStoreType);
129           }
130       }

160                      "name='" + name + '\'' +
161                      ", retentionPeriod=" + retentionPeriod +
162                      ", segmentInterval=" + segmentInterval +
163                      ", windowSize=" + windowSize +
