17   package org.apache.kafka.streams.kstream.internals;
18   
19   import org.apache.kafka.streams.kstream.Window;
20   
21   /**
22    * {@link UnlimitedWindow} is an "infinite" large window with a fixed (inclusive) start time.
23    * All windows of the same {@link org.apache.kafka.streams.kstream.UnlimitedWindows window specification} will have the
24    * same start time.
25    * To make the window size "infinite" end time is set to {@link Long#MAX_VALUE}.
26    * <p>
27    * For time semantics, see {@link org.apache.kafka.streams.processor.TimestampExtractor TimestampExtractor}.
28    *
29    * @see TimeWindow
30    * @see SessionWindow
31    * @see org.apache.kafka.streams.kstream.UnlimitedWindows
32    * @see org.apache.kafka.streams.processor.TimestampExtractor
33    */
34   public class UnlimitedWindow extends Window {
35   
36       /**
37        * Create a new window for the given start time (inclusive).
38        *
39        * @param startMs the start timestamp of the window (inclusive)
40        * @throws IllegalArgumentException if {@code start} is negative
41        */
42       public UnlimitedWindow(final long startMs) {
43           super(startMs, Long.MAX_VALUE);
44       }
45   
46       /**
47        * Returns {@code true} if the given window is of the same type, because all unlimited windows overlap with each
48        * other due to their infinite size.
49        *
50        * @param other another window
51        * @return {@code true}
52        * @throws IllegalArgumentException if the {@code other} window has a different type than {@code this} window
53        */
54       @Override
55       public boolean overlap(final Window other) {
56           if (getClass() != other.getClass()) {
57               throw new IllegalArgumentException("Cannot compare windows of different type. Other window has type "
58                   + other.getClass() + ".");
59           }
60           return true;
61       }
62   
63   }
