103       void createTasks(final Map<TaskId, Set<TopicPartition>> activeTasksToCreate,
104                        final Map<TaskId, Set<TopicPartition>> standbyTasksToCreate) {
105           for (final Map.Entry<TaskId, Set<TopicPartition>> taskToBeCreated : activeTasksToCreate.entrySet()) {
106               final TaskId taskId = taskToBeCreated.getKey();
107   
108               if (activeTasksPerId.containsKey(taskId)) {
109                   throw new IllegalStateException("Attempted to create an active task that we already own: " + taskId);
110               }
111           }
112   
113           for (final Map.Entry<TaskId, Set<TopicPartition>> taskToBeCreated : standbyTasksToCreate.entrySet()) {
114               final TaskId taskId = taskToBeCreated.getKey();
115   
116               if (standbyTasksPerId.containsKey(taskId)) {
117                   throw new IllegalStateException("Attempted to create a standby task that we already own: " + taskId);
118               }
119           }
120   
121           // keep this check to simplify testing (ie, no need to mock `activeTaskCreator`)
122           if (!activeTasksToCreate.isEmpty()) {
123               // TODO: change type to `StreamTask`
124               for (final Task activeTask : activeTaskCreator.createTasks(mainConsumer, activeTasksToCreate)) {
125                   activeTasksPerId.put(activeTask.id(), activeTask);
126                   allTasksPerId.put(activeTask.id(), activeTask);
127                   for (final TopicPartition topicPartition : activeTask.inputPartitions()) {
128                       activeTasksPerPartition.put(topicPartition, activeTask);
129                   }
130               }
131           }
132   
133           // keep this check to simplify testing (ie, no need to mock `standbyTaskCreator`)
134           if (!standbyTasksToCreate.isEmpty()) {
135               // TODO: change type to `StandbyTask`
136               for (final Task standbyTask : standbyTaskCreator.createTasks(standbyTasksToCreate)) {
137                   standbyTasksPerId.put(standbyTask.id(), standbyTask);
138                   allTasksPerId.put(standbyTask.id(), standbyTask);
139               }
140           }
141       }

143       void convertActiveToStandby(final StreamTask activeTask,
144                                   final Set<TopicPartition> partitions,
145                                   final Map<TaskId, RuntimeException> taskCloseExceptions) {
146           if (activeTasksPerId.remove(activeTask.id()) == null) {
147               throw new IllegalStateException("Attempted to convert unknown active task to standby task: " + activeTask.id());
148           }
149           final Set<TopicPartition> toBeRemoved = activeTasksPerPartition.entrySet().stream()
150               .filter(e -> e.getValue().id().equals(activeTask.id()))
151               .map(Map.Entry::getKey)
152               .collect(Collectors.toSet());
153           toBeRemoved.forEach(activeTasksPerPartition::remove);
154   
155           cleanUpTaskProducerAndRemoveTask(activeTask.id(), taskCloseExceptions);
156   
157           final StandbyTask standbyTask = standbyTaskCreator.createStandbyTaskFromActive(activeTask, partitions);
158           standbyTasksPerId.put(standbyTask.id(), standbyTask);
159           allTasksPerId.put(standbyTask.id(), standbyTask);
160       }

162       void convertStandbyToActive(final StandbyTask standbyTask, final Set<TopicPartition> partitions) {
163           if (standbyTasksPerId.remove(standbyTask.id()) == null) {
164               throw new IllegalStateException("Attempted to convert unknown standby task to stream task: " + standbyTask.id());
165           }
166   
167           final StreamTask activeTask = activeTaskCreator.createActiveTaskFromStandby(standbyTask, partitions, mainConsumer);
168           activeTasksPerId.put(activeTask.id(), activeTask);
169           for (final TopicPartition topicPartition : activeTask.inputPartitions()) {
170               activeTasksPerPartition.put(topicPartition, activeTask);
171           }
172           allTasksPerId.put(activeTask.id(), activeTask);
173       }

175       void updateInputPartitionsAndResume(final Task task, final Set<TopicPartition> topicPartitions) {
176           final boolean requiresUpdate = !task.inputPartitions().equals(topicPartitions);
177           if (requiresUpdate) {
178               log.debug("Update task {} inputPartitions: current {}, new {}", task, task.inputPartitions(), topicPartitions);
179               for (final TopicPartition inputPartition : task.inputPartitions()) {
180                   activeTasksPerPartition.remove(inputPartition);
181               }
182               if (task.isActive()) {
183                   for (final TopicPartition topicPartition : topicPartitions) {
184                       activeTasksPerPartition.put(topicPartition, task);
185                   }
186               }
187               task.updateInputPartitions(topicPartitions, topologyMetadata.nodeToSourceTopics(task.id()));
188           }
189           task.resume();
190       }

192       void cleanUpTaskProducerAndRemoveTask(final TaskId taskId,
193                                             final Map<TaskId, RuntimeException> taskCloseExceptions) {
194           try {
195               activeTaskCreator.closeAndRemoveTaskProducerIfNeeded(taskId);
196           } catch (final RuntimeException e) {
197               final String uncleanMessage = String.format("Failed to close task %s cleanly. Attempting to close remaining tasks before re-throwing:", taskId);
198               log.error(uncleanMessage, e);
199               taskCloseExceptions.putIfAbsent(taskId, e);
200           }
201           removeTaskBeforeClosing(taskId);
202       }

217       void removeTaskBeforeClosing(final TaskId taskId) {
218           activeTasksPerId.remove(taskId);
219           final Set<TopicPartition> toBeRemoved = activeTasksPerPartition.entrySet().stream()
220               .filter(e -> e.getValue().id().equals(taskId))
221               .map(Map.Entry::getKey)
222               .collect(Collectors.toSet());
223           toBeRemoved.forEach(activeTasksPerPartition::remove);
224           standbyTasksPerId.remove(taskId);
225           allTasksPerId.remove(taskId);
226       }
