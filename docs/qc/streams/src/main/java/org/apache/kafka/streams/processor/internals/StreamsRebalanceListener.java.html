17   package org.apache.kafka.streams.processor.internals;
18   
19   import java.util.concurrent.atomic.AtomicInteger;
20   import org.apache.kafka.clients.consumer.ConsumerRebalanceListener;
21   import org.apache.kafka.common.TopicPartition;
22   import org.apache.kafka.common.utils.Time;
23   import org.apache.kafka.streams.errors.MissingSourceTopicException;
24   import org.apache.kafka.streams.errors.TaskAssignmentException;
25   import org.apache.kafka.streams.processor.internals.StreamThread.State;
26   import org.apache.kafka.streams.processor.internals.assignment.AssignorError;
27   import org.slf4j.Logger;
28   
29   import java.util.Collection;
30   
31   public class StreamsRebalanceListener implements ConsumerRebalanceListener {
32   
33       private final Time time;
34       private final TaskManager taskManager;
35       private final StreamThread streamThread;
36       private final Logger log;
37       private final AtomicInteger assignmentErrorCode;
38   
39       StreamsRebalanceListener(final Time time,
40                                final TaskManager taskManager,
41                                final StreamThread streamThread,
42                                final Logger log,
43                                final AtomicInteger assignmentErrorCode) {
44           this.time = time;
45           this.taskManager = taskManager;
46           this.streamThread = streamThread;
47           this.log = log;
48           this.assignmentErrorCode = assignmentErrorCode;
49       }
50   
51       @Override
52       public void onPartitionsAssigned(final Collection<TopicPartition> partitions) {
53           // NB: all task management is already handled by:
54           // org.apache.kafka.streams.processor.internals.StreamsPartitionAssignor.onAssignment
55           if (assignmentErrorCode.get() == AssignorError.INCOMPLETE_SOURCE_TOPIC_METADATA.code()) {
56               log.error("Received error code {}", AssignorError.INCOMPLETE_SOURCE_TOPIC_METADATA);
57               taskManager.handleRebalanceComplete();
58               throw new MissingSourceTopicException("One or more source topics were missing during rebalance");
59           } else if (assignmentErrorCode.get() == AssignorError.VERSION_PROBING.code()) {
60               log.info("Received version probing code {}", AssignorError.VERSION_PROBING);
61           }  else if (assignmentErrorCode.get() == AssignorError.ASSIGNMENT_ERROR.code()) {
62               log.error("Received error code {}", AssignorError.ASSIGNMENT_ERROR);
63               taskManager.handleRebalanceComplete();
64               throw new TaskAssignmentException("Hit an unexpected exception during task assignment phase of rebalance");
65           } else if (assignmentErrorCode.get() == AssignorError.SHUTDOWN_REQUESTED.code()) {
66               log.error("A Kafka Streams client in this Kafka Streams application is requesting to shutdown the application");
67               taskManager.handleRebalanceComplete();
68               streamThread.shutdownToError();
69               return;
70           } else if (assignmentErrorCode.get() != AssignorError.NONE.code()) {
71               log.error("Received unknown error code {}", assignmentErrorCode.get());
72               throw new TaskAssignmentException("Hit an unrecognized exception during rebalance");
73           }
74   
75           streamThread.setState(State.PARTITIONS_ASSIGNED);
76           streamThread.setPartitionAssignedTime(time.milliseconds());
77           taskManager.handleRebalanceComplete();
78       }
79   
80       @Override
81       public void onPartitionsRevoked(final Collection<TopicPartition> partitions) {
82           log.debug("Current state {}: revoked partitions {} because of consumer rebalance.\n" +
83                         "\tcurrently assigned active tasks: {}\n" +
84                         "\tcurrently assigned standby tasks: {}\n",
85                     streamThread.state(),
86                     partitions,
87                     taskManager.activeTaskIds(),
88                     taskManager.standbyTaskIds());
89   
90           // We need to still invoke handleRevocation if the thread has been told to shut down, but we shouldn't ever
91           // transition away from PENDING_SHUTDOWN once it's been initiated (to anything other than DEAD)
92           if ((streamThread.setState(State.PARTITIONS_REVOKED) != null || streamThread.state() == State.PENDING_SHUTDOWN) && !partitions.isEmpty()) {
93               final long start = time.milliseconds();
94               try {
95                   taskManager.handleRevocation(partitions);
96               } finally {
97                   log.info("partition revocation took {} ms.", time.milliseconds() - start);
98               }
99           }
100       }
101   
102       @Override
103       public void onPartitionsLost(final Collection<TopicPartition> partitions) {
104           log.info("at state {}: partitions {} lost due to missed rebalance.\n" +
105                        "\tlost active tasks: {}\n" +
106                        "\tlost assigned standby tasks: {}\n",
107                    streamThread.state(),
108                    partitions,
109                    taskManager.activeTaskIds(),
110                    taskManager.standbyTaskIds());
111   
112           final long start = time.milliseconds();
113           try {
114               // close all active tasks as lost but don't try to commit offsets as we no longer own them
115               taskManager.handleLostAll();
116           } finally {
117               log.info("partitions lost took {} ms.", time.milliseconds() - start);
118           }
119       }
120   }
