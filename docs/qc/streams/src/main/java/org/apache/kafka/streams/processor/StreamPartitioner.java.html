19   import org.apache.kafka.streams.Topology;

21   /**
22    * Determine how records are distributed among the partitions in a Kafka topic. If not specified, the underlying producer's
23    * {@link org.apache.kafka.clients.producer.internals.DefaultPartitioner} will be used to determine the partition.
24    * <p>
25    * Kafka topics are divided into one or more <i>partitions</i>. Since each partition must fit on the servers that host it, so
26    * using multiple partitions allows the topic to scale beyond a size that will fit on a single machine. Partitions also enable you
27    * to use multiple instances of your topology to process in parallel all of the records on the topology's source topics.
28    * <p>
29    * When a topology is instantiated, each of its sources are assigned a subset of that topic's partitions. That means that only
30    * those processors in that topology instance will consume the records from those partitions. In many cases, Kafka Streams will
31    * automatically manage these instances, and adjust when new topology instances are added or removed.
32    * <p>
33    * Some topologies, though, need more control over which records appear in each partition. For example, some topologies that have
34    * stateful processors may want all records within a range of keys to always be delivered to and handled by the same topology instance.
35    * An upstream topology producing records to that topic can use a custom <i>stream partitioner</i> to precisely and consistently
36    * determine to which partition each record should be written.
37    * <p>
38    * To do this, create a <code>StreamPartitioner</code> implementation, and when you build your topology specify that custom partitioner
39    * when {@link Topology#addSink(String, String, org.apache.kafka.common.serialization.Serializer, org.apache.kafka.common.serialization.Serializer, StreamPartitioner, String...) adding a sink}
40    * for that topic.
41    * <p>
42    * All StreamPartitioner implementations should be stateless and a pure function so they can be shared across topic and sink nodes.
43    * 
44    * @param <K> the type of keys
45    * @param <V> the type of values
46    * @see Topology#addSink(String, String, org.apache.kafka.common.serialization.Serializer,
47    *      org.apache.kafka.common.serialization.Serializer, StreamPartitioner, String...)
48    * @see Topology#addSink(String, String, StreamPartitioner, String...)
49    */
50   public interface StreamPartitioner<K, V> {
51   
52       /**
53        * Determine the partition number for a record with the given key and value and the current number of partitions.
54        *
55        * @param topic the topic name this record is sent to
56        * @param key the key of the record
57        * @param value the value of the record
58        * @param numPartitions the total number of partitions
59        * @return an integer between 0 and {@code numPartitions-1}, or {@code null} if the default partitioning logic should be used
60        */
61       Integer partition(String topic, K key, V value, int numPartitions);
62   }
