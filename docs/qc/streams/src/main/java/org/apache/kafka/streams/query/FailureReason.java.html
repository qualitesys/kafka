20   import org.apache.kafka.common.annotation.InterfaceStability.Evolving;
21   
22   /**
23    * This enumeration type captures the various top-level reasons that a particular
24    * partition of a store would fail to execute a query. Stores should generally
25    * respond with a failure message instead of throwing an exception.
26    * <p>
27    * Intended to be used in {@link QueryResult#forFailure(FailureReason, String)}.
28    */
29   @Evolving
30   public enum FailureReason {
31       /**
32        * Failure indicating that the store doesn't know how to handle the given query.
33        */
34       UNKNOWN_QUERY_TYPE,
35   
36       /**
37        * The query required to execute on an active task (via {@link StateQueryRequest#requireActive()}),
38        * but while executing the query, the task was either a Standby task, or it was an Active task
39        * not in the RUNNING state. The failure message will contain the reason for the failure.
40        * <p>
41        * The caller should either try again later or try a different replica.
42        */
43       NOT_ACTIVE,
44   
45       /**
46        * Failure indicating that the store partition is not (yet) up to the desired bound. The caller
47        * should either try again later or try a different replica.
48        */
49       NOT_UP_TO_BOUND,
50   
51       /**
52        * Failure indicating that the requested store partition is not present on the local
53        * KafkaStreams instance. It may have been migrated to another instance during a rebalance. The
54        * caller is recommended to try a different replica.
55        */
56       NOT_PRESENT,
57   
58       /**
59        * The requested store partition does not exist at all. For example, partition 4 was requested,
60        * but the store in question only has 4 partitions (0 through 3).
61        */
62       DOES_NOT_EXIST,
63   
64       /**
65        * The store that handled the query got an exception during query execution. The message
66        * will contain the exception details. Depending on the nature of the exception, the caller
67        * may be able to retry this instance or may need to try a different instance.
68        */
69       STORE_EXCEPTION;
70   }
