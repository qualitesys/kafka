68   /**
69    * Configuration for a {@link KafkaStreams} instance.
70    * Can also be used to configure the Kafka Streams internal {@link KafkaConsumer}, {@link KafkaProducer} and {@link Admin}.
71    * To avoid consumer/producer/admin property conflicts, you should prefix those properties using
72    * {@link #consumerPrefix(String)}, {@link #producerPrefix(String)} and {@link #adminClientPrefix(String)}, respectively.
73    * <p>
74    * Example:
75    * <pre>{@code
76    * // potentially wrong: sets "metadata.max.age.ms" to 1 minute for producer AND consumer
77    * Properties streamsProperties = new Properties();
78    * streamsProperties.put(ConsumerConfig.METADATA_MAX_AGE_CONFIG, 60000);
79    * // or
80    * streamsProperties.put(ProducerConfig.METADATA_MAX_AGE_CONFIG, 60000);
81    *
82    * // suggested:
83    * Properties streamsProperties = new Properties();
84    * // sets "metadata.max.age.ms" to 1 minute for consumer only
85    * streamsProperties.put(StreamsConfig.consumerPrefix(ConsumerConfig.METADATA_MAX_AGE_CONFIG), 60000);
86    * // sets "metadata.max.age.ms" to 1 minute for producer only
87    * streamsProperties.put(StreamsConfig.producerPrefix(ProducerConfig.METADATA_MAX_AGE_CONFIG), 60000);
88    *
89    * StreamsConfig streamsConfig = new StreamsConfig(streamsProperties);
90    * }</pre>
91    *
92    * This instance can also be used to pass in custom configurations to different modules (e.g. passing a special config in your customized serde class).
93    * The consumer/producer/admin prefix can also be used to distinguish these custom config values passed to different clients with the same config name.
94    * * Example:
95    * <pre>{@code
96    * Properties streamsProperties = new Properties();
97    * // sets "my.custom.config" to "foo" for consumer only
98    * streamsProperties.put(StreamsConfig.consumerPrefix("my.custom.config"), "foo");
99    * // sets "my.custom.config" to "bar" for producer only
100    * streamsProperties.put(StreamsConfig.producerPrefix("my.custom.config"), "bar");
101    * // sets "my.custom.config2" to "boom" for all clients universally
102    * streamsProperties.put("my.custom.config2", "boom");
103    *
104    * // as a result, inside producer's serde class configure(..) function,
105    * // users can now read both key-value pairs "my.custom.config" -> "foo"
106    * // and "my.custom.config2" -> "boom" from the config map
107    * StreamsConfig streamsConfig = new StreamsConfig(streamsProperties);
108    * }</pre>
109    *
110    *
111    * When increasing {@link ProducerConfig#MAX_BLOCK_MS_CONFIG} to be more resilient to non-available brokers you should also
112    * increase {@link ConsumerConfig#MAX_POLL_INTERVAL_MS_CONFIG} using the following guidance:
113    * <pre>
114    *     max.poll.interval.ms &gt; max.block.ms
115    * </pre>
116    *
117    *
118    * Kafka Streams requires at least the following properties to be set:
119    * <ul>
120    *  <li>{@link #APPLICATION_ID_CONFIG "application.id"}</li>
121    *  <li>{@link #BOOTSTRAP_SERVERS_CONFIG "bootstrap.servers"}</li>
122    * </ul>
123    *
124    * By default, Kafka Streams does not allow users to overwrite the following properties (Streams setting shown in parentheses):
125    * <ul>
126    *   <li>{@link ConsumerConfig#GROUP_ID_CONFIG "group.id"} (&lt;application.id&gt;) - Streams client will always use the application ID a consumer group ID</li>
127    *   <li>{@link ConsumerConfig#ENABLE_AUTO_COMMIT_CONFIG "enable.auto.commit"} (false) - Streams client will always disable/turn off auto committing</li>
128    *   <li>{@link ConsumerConfig#PARTITION_ASSIGNMENT_STRATEGY_CONFIG "partition.assignment.strategy"} (<code>StreamsPartitionAssignor</code>) - Streams client will always use its own partition assignor</li>
129    * </ul>
130    *
131    * If {@link #PROCESSING_GUARANTEE_CONFIG "processing.guarantee"} is set to {@link #EXACTLY_ONCE_V2 "exactly_once_v2"},
132    * {@link #EXACTLY_ONCE "exactly_once"} (deprecated), or {@link #EXACTLY_ONCE_BETA "exactly_once_beta"} (deprecated), Kafka Streams does not
133    * allow users to overwrite the following properties (Streams setting shown in parentheses):
134    * <ul>
135    *   <li>{@link ConsumerConfig#ISOLATION_LEVEL_CONFIG "isolation.level"} (read_committed) - Consumers will always read committed data only</li>
136    *   <li>{@link ProducerConfig#ENABLE_IDEMPOTENCE_CONFIG "enable.idempotence"} (true) - Producer will always have idempotency enabled</li>
137    * </ul>
138    *
139    * @see KafkaStreams#KafkaStreams(org.apache.kafka.streams.Topology, Properties)
140    * @see ConsumerConfig
141    * @see ProducerConfig
142    */
