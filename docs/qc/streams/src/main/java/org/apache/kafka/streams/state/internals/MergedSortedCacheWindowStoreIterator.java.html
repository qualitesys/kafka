17   package org.apache.kafka.streams.state.internals;
18   
19   import java.util.function.Function;
20   import org.apache.kafka.common.utils.Bytes;
21   import org.apache.kafka.streams.KeyValue;
22   import org.apache.kafka.streams.state.KeyValueIterator;
23   import org.apache.kafka.streams.state.WindowStoreIterator;
24   
25   import static org.apache.kafka.streams.state.internals.SegmentedCacheFunction.bytesFromCacheKey;
26   
27   /**
28    * Merges two iterators. Assumes each of them is sorted by key
29    *
30    */
31   class MergedSortedCacheWindowStoreIterator extends AbstractMergedSortedCacheStoreIterator<Long, Long, byte[], byte[]> implements WindowStoreIterator<byte[]> {
32   
33       private final Function<byte[], Long> timestampExtractor;
34   
35       MergedSortedCacheWindowStoreIterator(final PeekingKeyValueIterator<Bytes, LRUCacheEntry> cacheIterator,
36                                            final KeyValueIterator<Long, byte[]> storeIterator,
37                                            final boolean forward) {
38           this(cacheIterator, storeIterator, forward, WindowKeySchema::extractStoreTimestamp);
39       }
40   
41       MergedSortedCacheWindowStoreIterator(final PeekingKeyValueIterator<Bytes, LRUCacheEntry> cacheIterator,
42                                            final KeyValueIterator<Long, byte[]> storeIterator,
43                                            final boolean forward,
44                                            final Function<byte[], Long> tsExtractor) {
45           super(cacheIterator, storeIterator, forward);
46           this.timestampExtractor = tsExtractor;
47       }
48   
49       @Override
50       public KeyValue<Long, byte[]> deserializeStorePair(final KeyValue<Long, byte[]> pair) {
51           return pair;
52       }
53   
54       @Override
55       Long deserializeCacheKey(final Bytes cacheKey) {
56           final byte[] binaryKey = bytesFromCacheKey(cacheKey);
57           return timestampExtractor.apply(binaryKey);
58       }
59   
60       @Override
61       byte[] deserializeCacheValue(final LRUCacheEntry cacheEntry) {
62           return cacheEntry.value();
63       }
64   
65       @Override
66       public Long deserializeStoreKey(final Long key) {
67           return key;
68       }
69   
70       @Override
71       public int compare(final Bytes cacheKey, final Long storeKey) {
72           final byte[] binaryKey = bytesFromCacheKey(cacheKey);
73   
74           final Long cacheTimestamp = timestampExtractor.apply(binaryKey);
75           return cacheTimestamp.compareTo(storeKey);
76       }
77   }
