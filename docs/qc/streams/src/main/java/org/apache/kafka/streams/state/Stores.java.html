41   /**
42    * Factory for creating state stores in Kafka Streams.
43    * <p>
44    * When using the high-level DSL, i.e., {@link org.apache.kafka.streams.StreamsBuilder StreamsBuilder}, users create
45    * {@link StoreSupplier}s that can be further customized via
46    * {@link org.apache.kafka.streams.kstream.Materialized Materialized}.
47    * For example, a topic read as {@link org.apache.kafka.streams.kstream.KTable KTable} can be materialized into an
48    * in-memory store with custom key/value serdes and caching disabled:
49    * <pre>{@code
50    * StreamsBuilder builder = new StreamsBuilder();
51    * KeyValueBytesStoreSupplier storeSupplier = Stores.inMemoryKeyValueStore("queryable-store-name");
52    * KTable<Long,String> table = builder.table(
53    *   "topicName",
54    *   Materialized.<Long,String>as(storeSupplier)
55    *               .withKeySerde(Serdes.Long())
56    *               .withValueSerde(Serdes.String())
57    *               .withCachingDisabled());
58    * }</pre>
59    * When using the Processor API, i.e., {@link org.apache.kafka.streams.Topology Topology}, users create
60    * {@link StoreBuilder}s that can be attached to {@link org.apache.kafka.streams.processor.api.Processor Processor}s.
61    * For example, you can create a {@link org.apache.kafka.streams.kstream.Windowed windowed} RocksDB store with custom
62    * changelog topic configuration like:
63    * <pre>{@code
64    * Topology topology = new Topology();
65    * topology.addProcessor("processorName", ...);
66    *
67    * Map<String,String> topicConfig = new HashMap<>();
68    * StoreBuilder<WindowStore<Integer, Long>> storeBuilder = Stores
69    *   .windowStoreBuilder(
70    *     Stores.persistentWindowStore("queryable-store-name", ...),
71    *     Serdes.Integer(),
72    *     Serdes.Long())
73    *   .withLoggingEnabled(topicConfig);
74    *
75    * topology.addStateStore(storeBuilder, "processorName");
76    * }</pre>
77    */
78   public final class Stores {
