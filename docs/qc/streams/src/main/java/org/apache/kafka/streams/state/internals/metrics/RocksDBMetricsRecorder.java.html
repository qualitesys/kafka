137       /**
138        * The initialisation of the metrics recorder is idempotent.
139        */
140       public void init(final StreamsMetricsImpl streamsMetrics,
141                        final TaskId taskId) {
142           Objects.requireNonNull(streamsMetrics, "Streams metrics must not be null");
143           Objects.requireNonNull(streamsMetrics, "task ID must not be null");
144           if (this.taskId != null && !this.taskId.equals(taskId)) {
145               throw new IllegalStateException("Metrics recorder is re-initialised with different task: previous task is " +
146                   this.taskId + " whereas current task is " + taskId + ". This is a bug in Kafka Streams. " +
147                   "Please open a bug report under https://issues.apache.org/jira/projects/KAFKA/issues");
148           }
149           if (this.streamsMetrics != null && this.streamsMetrics != streamsMetrics) {
150               throw new IllegalStateException("Metrics recorder is re-initialised with different Streams metrics. "
151                   + "This is a bug in Kafka Streams. " +
152                   "Please open a bug report under https://issues.apache.org/jira/projects/KAFKA/issues");
153           }
154           final RocksDBMetricContext metricContext = new RocksDBMetricContext(taskId.toString(), metricsScope, storeName);
155           initSensors(streamsMetrics, metricContext);
156           initGauges(streamsMetrics, metricContext);
157           this.taskId = taskId;
158           this.streamsMetrics = streamsMetrics;
159       }

161       public void addValueProviders(final String segmentName,
162                                     final RocksDB db,
163                                     final Cache cache,
164                                     final Statistics statistics) {
165           if (storeToValueProviders.isEmpty()) {
166               logger.debug("Adding metrics recorder of task {} to metrics recording trigger", taskId);
167               streamsMetrics.rocksDBMetricsRecordingTrigger().addMetricsRecorder(this);
168           } else if (storeToValueProviders.containsKey(segmentName)) {
169               throw new IllegalStateException("Value providers for store " + segmentName + " of task " + taskId +
170                   " has been already added. This is a bug in Kafka Streams. " +
171                   "Please open a bug report under https://issues.apache.org/jira/projects/KAFKA/issues");
172           }
173           verifyDbAndCacheAndStatistics(segmentName, db, cache, statistics);
174           logger.debug("Adding value providers for store {} of task {}", segmentName, taskId);
175           storeToValueProviders.put(segmentName, new DbAndCacheAndStatistics(db, cache, statistics));
176       }

178       private void verifyDbAndCacheAndStatistics(final String segmentName,
179                                                  final RocksDB db,
180                                                  final Cache cache,
181                                                  final Statistics statistics) {
182           for (final DbAndCacheAndStatistics valueProviders : storeToValueProviders.values()) {
183               verifyConsistencyOfValueProvidersAcrossSegments(segmentName, statistics, valueProviders.statistics, "statistics");
184               verifyConsistencyOfValueProvidersAcrossSegments(segmentName, cache, valueProviders.cache, "cache");
185               if (db == valueProviders.db) {
186                   throw new IllegalStateException("DB instance for store " + segmentName + " of task " + taskId +
187                       " was already added for another segment as a value provider. This is a bug in Kafka Streams. " +
188                       "Please open a bug report under https://issues.apache.org/jira/projects/KAFKA/issues");
189               }
190               if (storeToValueProviders.size() == 1 && cache != valueProviders.cache) {
191                   singleCache = false;
192               } else if (singleCache && cache != valueProviders.cache || !singleCache && cache == valueProviders.cache) {
193                   throw new IllegalStateException("Caches for store " + storeName + " of task " + taskId +
194                       " are either not all distinct or do not all refer to the same cache. This is a bug in Kafka Streams. " +
195                       "Please open a bug report under https://issues.apache.org/jira/projects/KAFKA/issues");
196               }
197           }
198       }

200       private void verifyConsistencyOfValueProvidersAcrossSegments(final String segmentName,
201                                                                    final Object newValueProvider,
202                                                                    final Object oldValueProvider,
203                                                                    final String valueProviderName) {
204           if (newValueProvider == null && oldValueProvider != null ||
205               newValueProvider != null && oldValueProvider == null) {
206   
207               final char capitalizedFirstChar = valueProviderName.toUpperCase(Locale.US).charAt(0);
208               final StringBuilder capitalizedValueProviderName = new StringBuilder(valueProviderName);
209               capitalizedValueProviderName.setCharAt(0, capitalizedFirstChar);
210               throw new IllegalStateException(capitalizedValueProviderName +
211                   " for segment " + segmentName + " of task " + taskId +
212                   " is" + (newValueProvider == null ? " " : " not ") + "null although the " + valueProviderName +
213                   " of another segment in this metrics recorder is" + (newValueProvider != null ? " " : " not ") + "null. " +
214                   "This is a bug in Kafka Streams. " +
215                   "Please open a bug report under https://issues.apache.org/jira/projects/KAFKA/issues");
216           }
217       }

355       private Gauge<BigInteger> gaugeToComputeSumOfProperties(final String propertyName) {
356           return (metricsConfig, now) -> {
357               BigInteger result = BigInteger.valueOf(0);
358               for (final DbAndCacheAndStatistics valueProvider : storeToValueProviders.values()) {
359                   try {
360                       // values of RocksDB properties are of type unsigned long in C++, i.e., in Java we need to use
361                       // BigInteger and construct the object from the byte representation of the value
362                       result = result.add(new BigInteger(1, longToBytes(
363                           valueProvider.db.getAggregatedLongProperty(ROCKSDB_PROPERTIES_PREFIX + propertyName)
364                       )));
365                   } catch (final RocksDBException e) {
366                       throw new ProcessorStateException("Error recording RocksDB metric " + propertyName, e);
367                   }
368               }
369               return result;
370           };
371       }

373       private Gauge<BigInteger> gaugeToComputeBlockCacheMetrics(final String propertyName) {
374           return (metricsConfig, now) -> {
375               BigInteger result = BigInteger.valueOf(0);
376               for (final DbAndCacheAndStatistics valueProvider : storeToValueProviders.values()) {
377                   try {
378                       if (singleCache) {
379                           // values of RocksDB properties are of type unsigned long in C++, i.e., in Java we need to use
380                           // BigInteger and construct the object from the byte representation of the value
381                           result = new BigInteger(1, longToBytes(
382                               valueProvider.db.getAggregatedLongProperty(ROCKSDB_PROPERTIES_PREFIX + propertyName)
383                           ));
384                           break;
385                       } else {
386                           // values of RocksDB properties are of type unsigned long in C++, i.e., in Java we need to use
387                           // BigInteger and construct the object from the byte representation of the value
388                           result = result.add(new BigInteger(1, longToBytes(
389                               valueProvider.db.getAggregatedLongProperty(ROCKSDB_PROPERTIES_PREFIX + propertyName)
390                           )));
391                       }
392                   } catch (final RocksDBException e) {
393                       throw new ProcessorStateException("Error recording RocksDB metric " + propertyName, e);
394                   }
395               }
396               return result;
397           };
398       }

400       private static byte[] longToBytes(final long data) {
401           final ByteBuffer conversionBuffer = ByteBuffer.allocate(Long.BYTES);
402           conversionBuffer.putLong(0, data);
403           return conversionBuffer.array();
404       }

425       public void record(final long now) {
426           logger.debug("Recording metrics for store {}", storeName);
427           long bytesWrittenToDatabase = 0;
428           long bytesReadFromDatabase = 0;
429           long memtableBytesFlushed = 0;
430           long memtableHits = 0;
431           long memtableMisses = 0;
432           long blockCacheDataHits = 0;
433           long blockCacheDataMisses = 0;
434           long blockCacheIndexHits = 0;
435           long blockCacheIndexMisses = 0;
436           long blockCacheFilterHits = 0;
437           long blockCacheFilterMisses = 0;
438           long writeStallDuration = 0;
439           long bytesWrittenDuringCompaction = 0;
440           long bytesReadDuringCompaction = 0;
441           long numberOfOpenFiles = 0;
442           long numberOfFileErrors = 0;
443           long memtableFlushTimeSum = 0;
444           long memtableFlushTimeCount = 0;
445           double memtableFlushTimeMin = Double.MAX_VALUE;
446           double memtableFlushTimeMax = 0.0;
447           long compactionTimeSum = 0;
448           long compactionTimeCount = 0;
449           double compactionTimeMin = Double.MAX_VALUE;
450           double compactionTimeMax = 0.0;
451           boolean shouldRecord = true;
452           for (final DbAndCacheAndStatistics valueProviders : storeToValueProviders.values()) {
453               if (valueProviders.statistics == null) {
454                   shouldRecord = false;
455                   break;
456               }
457               bytesWrittenToDatabase += valueProviders.statistics.getAndResetTickerCount(TickerType.BYTES_WRITTEN);
458               bytesReadFromDatabase += valueProviders.statistics.getAndResetTickerCount(TickerType.BYTES_READ);
459               memtableBytesFlushed += valueProviders.statistics.getAndResetTickerCount(TickerType.FLUSH_WRITE_BYTES);
460               memtableHits += valueProviders.statistics.getAndResetTickerCount(TickerType.MEMTABLE_HIT);
461               memtableMisses += valueProviders.statistics.getAndResetTickerCount(TickerType.MEMTABLE_MISS);
462               blockCacheDataHits += valueProviders.statistics.getAndResetTickerCount(TickerType.BLOCK_CACHE_DATA_HIT);
463               blockCacheDataMisses += valueProviders.statistics.getAndResetTickerCount(TickerType.BLOCK_CACHE_DATA_MISS);
464               blockCacheIndexHits += valueProviders.statistics.getAndResetTickerCount(TickerType.BLOCK_CACHE_INDEX_HIT);
465               blockCacheIndexMisses += valueProviders.statistics.getAndResetTickerCount(TickerType.BLOCK_CACHE_INDEX_MISS);
466               blockCacheFilterHits += valueProviders.statistics.getAndResetTickerCount(TickerType.BLOCK_CACHE_FILTER_HIT);
467               blockCacheFilterMisses += valueProviders.statistics.getAndResetTickerCount(TickerType.BLOCK_CACHE_FILTER_MISS);
468               writeStallDuration += valueProviders.statistics.getAndResetTickerCount(TickerType.STALL_MICROS);
469               bytesWrittenDuringCompaction += valueProviders.statistics.getAndResetTickerCount(TickerType.COMPACT_WRITE_BYTES);
470               bytesReadDuringCompaction += valueProviders.statistics.getAndResetTickerCount(TickerType.COMPACT_READ_BYTES);
471               numberOfOpenFiles += valueProviders.statistics.getAndResetTickerCount(TickerType.NO_FILE_OPENS)
472                   - valueProviders.statistics.getAndResetTickerCount(TickerType.NO_FILE_CLOSES);
473               numberOfFileErrors += valueProviders.statistics.getAndResetTickerCount(TickerType.NO_FILE_ERRORS);
474               final HistogramData memtableFlushTimeData = valueProviders.statistics.getHistogramData(HistogramType.FLUSH_TIME);
475               memtableFlushTimeSum += memtableFlushTimeData.getSum();
476               memtableFlushTimeCount += memtableFlushTimeData.getCount();
477               memtableFlushTimeMin = Double.min(memtableFlushTimeMin, memtableFlushTimeData.getMin());
478               memtableFlushTimeMax = Double.max(memtableFlushTimeMax, memtableFlushTimeData.getMax());
479               final HistogramData compactionTimeData = valueProviders.statistics.getHistogramData(HistogramType.COMPACTION_TIME);
480               compactionTimeSum += compactionTimeData.getSum();
481               compactionTimeCount += compactionTimeData.getCount();
482               compactionTimeMin = Double.min(compactionTimeMin, compactionTimeData.getMin());
483               compactionTimeMax = Double.max(compactionTimeMax, compactionTimeData.getMax());
484           }
485           if (shouldRecord) {
486               bytesWrittenToDatabaseSensor.record(bytesWrittenToDatabase, now);
487               bytesReadFromDatabaseSensor.record(bytesReadFromDatabase, now);
488               memtableBytesFlushedSensor.record(memtableBytesFlushed, now);
489               memtableHitRatioSensor.record(computeHitRatio(memtableHits, memtableMisses), now);
490               memtableAvgFlushTimeSensor.record(computeAvg(memtableFlushTimeSum, memtableFlushTimeCount), now);
491               memtableMinFlushTimeSensor.record(memtableFlushTimeMin, now);
492               memtableMaxFlushTimeSensor.record(memtableFlushTimeMax, now);
493               blockCacheDataHitRatioSensor.record(computeHitRatio(blockCacheDataHits, blockCacheDataMisses), now);
494               blockCacheIndexHitRatioSensor.record(computeHitRatio(blockCacheIndexHits, blockCacheIndexMisses), now);
495               blockCacheFilterHitRatioSensor.record(computeHitRatio(blockCacheFilterHits, blockCacheFilterMisses), now);
496               writeStallDurationSensor.record(writeStallDuration, now);
497               bytesWrittenDuringCompactionSensor.record(bytesWrittenDuringCompaction, now);
498               bytesReadDuringCompactionSensor.record(bytesReadDuringCompaction, now);
499               compactionTimeAvgSensor.record(computeAvg(compactionTimeSum, compactionTimeCount), now);
500               compactionTimeMinSensor.record(compactionTimeMin, now);
501               compactionTimeMaxSensor.record(compactionTimeMax, now);
502               numberOfOpenFilesSensor.record(numberOfOpenFiles, now);
503               numberOfFileErrorsSensor.record(numberOfFileErrors, now);
504           }
505       }

507       private double computeHitRatio(final long hits, final long misses) {
508           if (hits == 0) {
509               return 0;
510           }
511           return (double) hits / (hits + misses);
512       }
513   
514       private double computeAvg(final long sum, final long count) {
515           if (count == 0) {
516               return 0;
517           }
518           return (double) sum / count;
519       }
