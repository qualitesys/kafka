174       @SuppressWarnings("unchecked")
175       void openDB(final Map<String, Object> configs, final File stateDir) {
176           // initialize the default rocksdb options
177   
178           final DBOptions dbOptions = new DBOptions();
179           final ColumnFamilyOptions columnFamilyOptions = new ColumnFamilyOptions();
180           userSpecifiedOptions = new RocksDBGenericOptionsToDbOptionsColumnFamilyOptionsAdapter(dbOptions, columnFamilyOptions);
181   
182           final BlockBasedTableConfigWithAccessibleCache tableConfig = new BlockBasedTableConfigWithAccessibleCache();
183           cache = new LRUCache(BLOCK_CACHE_SIZE);
184           tableConfig.setBlockCache(cache);
185           tableConfig.setBlockSize(BLOCK_SIZE);
186   
187           filter = new BloomFilter();
188           tableConfig.setFilterPolicy(filter);
189   
190           userSpecifiedOptions.optimizeFiltersForHits();
191           userSpecifiedOptions.setTableFormatConfig(tableConfig);
192           userSpecifiedOptions.setWriteBufferSize(WRITE_BUFFER_SIZE);
193           userSpecifiedOptions.setCompressionType(COMPRESSION_TYPE);
194           userSpecifiedOptions.setCompactionStyle(COMPACTION_STYLE);
195           userSpecifiedOptions.setMaxWriteBufferNumber(MAX_WRITE_BUFFERS);
196           userSpecifiedOptions.setCreateIfMissing(true);
197           userSpecifiedOptions.setErrorIfExists(false);
198           userSpecifiedOptions.setInfoLogLevel(InfoLogLevel.ERROR_LEVEL);
199           // this is the recommended way to increase parallelism in RocksDb
200           // note that the current implementation of setIncreaseParallelism affects the number
201           // of compaction threads but not flush threads (the latter remains one). Also,
202           // the parallelism value needs to be at least two because of the code in
203           // https://github.com/facebook/rocksdb/blob/62ad0a9b19f0be4cefa70b6b32876e764b7f3c11/util/options.cc#L580
204           // subtracts one from the value passed to determine the number of compaction threads
205           // (this could be a bug in the RocksDB code and their devs have been contacted).
206           userSpecifiedOptions.setIncreaseParallelism(Math.max(Runtime.getRuntime().availableProcessors(), 2));
207   
208           wOptions = new WriteOptions();
209           wOptions.setDisableWAL(true);
210   
211           fOptions = new FlushOptions();
212           fOptions.setWaitForFlush(true);
213   
214           final Class<RocksDBConfigSetter> configSetterClass =
215                   (Class<RocksDBConfigSetter>) configs.get(StreamsConfig.ROCKSDB_CONFIG_SETTER_CLASS_CONFIG);
216   
217           if (configSetterClass != null) {
218               configSetter = Utils.newInstance(configSetterClass);
219               configSetter.setConfig(name, userSpecifiedOptions, configs);
220           }
221   
222           dbDir = new File(new File(stateDir, parentDir), name);
223           try {
224               Files.createDirectories(dbDir.getParentFile().toPath());
225               Files.createDirectories(dbDir.getAbsoluteFile().toPath());
226           } catch (final IOException fatal) {
227               throw new ProcessorStateException(fatal);
228           }
229   
230           // Setup statistics before the database is opened, otherwise the statistics are not updated
231           // with the measurements from Rocks DB
232           maybeSetUpStatistics(configs);
233   
234           openRocksDB(dbOptions, columnFamilyOptions);
235           open = true;
236   
237           addValueProvidersToMetricsRecorder();
238       }

240       private void maybeSetUpStatistics(final Map<String, Object> configs) {
241           if (userSpecifiedOptions.statistics() != null) {
242               userSpecifiedStatistics = true;
243           }
244           if (!userSpecifiedStatistics &&
245                   RecordingLevel.forName((String) configs.get(METRICS_RECORDING_LEVEL_CONFIG)) == RecordingLevel.DEBUG) {
246   
247               // metrics recorder will clean up statistics object
248               final Statistics statistics = new Statistics();
249               userSpecifiedOptions.setStatistics(statistics);
250           }
251       }
252   
253       private void addValueProvidersToMetricsRecorder() {
254           final TableFormatConfig tableFormatConfig = userSpecifiedOptions.tableFormatConfig();
255           final Statistics statistics = userSpecifiedStatistics ? null : userSpecifiedOptions.statistics();
256           if (tableFormatConfig instanceof BlockBasedTableConfigWithAccessibleCache) {
257               final Cache cache = ((BlockBasedTableConfigWithAccessibleCache) tableFormatConfig).blockCache();
258               metricsRecorder.addValueProviders(name, db, cache, statistics);
259           } else if (tableFormatConfig instanceof BlockBasedTableConfig) {
260               throw new ProcessorStateException("The used block-based table format configuration does not expose the " +
261                       "block cache. Use the BlockBasedTableConfig instance provided by Options#tableFormatConfig() to configure " +
262                       "the block-based table format of RocksDB. Do not provide a new instance of BlockBasedTableConfig to " +
263                       "the RocksDB options.");
264           } else {
265               metricsRecorder.addValueProviders(name, db, null, statistics);
266           }
267       }

269       void openRocksDB(final DBOptions dbOptions,
270                        final ColumnFamilyOptions columnFamilyOptions) {
271           final List<ColumnFamilyDescriptor> columnFamilyDescriptors

280           }

449       /**
450        * Return an approximate count of key-value mappings in this store.
451        *
452        * <code>RocksDB</code> cannot return an exact entry count without doing a
453        * full scan, so this method relies on the <code>rocksdb.estimate-num-keys</code>
454        * property to get an approximate count. The returned size also includes
455        * a count of dirty keys in the store's in-memory cache, which may lead to some
456        * double-counting of entries and inflate the estimate.
457        *
458        * @return an approximate count of key-value mappings in the store.
459        */
460       @Override
461       public long approximateNumEntries() {
462           validateStoreOpen();
463           final long numEntries;
464           try {
465               numEntries = dbAccessor.approximateNumEntries();
466           } catch (final RocksDBException e) {
467               throw new ProcessorStateException("Error fetching property from store " + name, e);
468           }
469           if (isOverflowing(numEntries)) {
470               return Long.MAX_VALUE;
471           }
472           return numEntries;
473       }
