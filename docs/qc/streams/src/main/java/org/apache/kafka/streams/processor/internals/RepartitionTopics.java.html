50       private final TopologyMetadata topologyMetadata;
51       private final Cluster clusterMetadata;
52       private final CopartitionedTopicsEnforcer copartitionedTopicsEnforcer;

57       public RepartitionTopics(final TopologyMetadata topologyMetadata,
58                                final InternalTopicManager internalTopicManager,
59                                final CopartitionedTopicsEnforcer copartitionedTopicsEnforcer,
60                                final Cluster clusterMetadata,
61                                final String logPrefix) {
62           this.topologyMetadata = topologyMetadata;
63           this.internalTopicManager = internalTopicManager;
64           this.clusterMetadata = clusterMetadata;
65           this.copartitionedTopicsEnforcer = copartitionedTopicsEnforcer;
66           final LogContext logContext = new LogContext(logPrefix);
67           log = logContext.logger(getClass());
68       }

70       public void setup() {
71           final Map<String, InternalTopicConfig> repartitionTopicMetadata = computeRepartitionTopicConfig(clusterMetadata);
72   
73           if (repartitionTopicMetadata.isEmpty()) {
74               if (missingInputTopicsBySubtopology.isEmpty()) {
75                   log.info("Skipping the repartition topic validation since there are no repartition topics.");
76               } else {
77                   log.info("Skipping the repartition topic validation since all topologies containing repartition"
78                                + "topics are missing external user source topics and cannot be processed.");
79               }
80           } else {
81               // ensure the co-partitioning topics within the group have the same number of partitions,
82               // and enforce the number of partitions for those repartition topics to be the same if they
83               // are co-partitioned as well.
84               ensureCopartitioning(topologyMetadata.copartitionGroups(), repartitionTopicMetadata, clusterMetadata);
85   
86               // make sure the repartition source topics exist with the right number of partitions,
87               // create these topics if necessary
88               internalTopicManager.makeReady(repartitionTopicMetadata);
89   
90               // augment the metadata with the newly computed number of partitions for all the
91               // repartition source topics
92               for (final Map.Entry<String, InternalTopicConfig> entry : repartitionTopicMetadata.entrySet()) {
93                   final String topic = entry.getKey();
94                   final int numPartitions = entry.getValue().numberOfPartitions().orElse(-1);
95   
96                   for (int partition = 0; partition < numPartitions; partition++) {
97                       topicPartitionInfos.put(
98                           new TopicPartition(topic, partition),
99                           new PartitionInfo(topic, partition, null, new Node[0], new Node[0])
100                       );
101                   }
102               }
103           }
104       }

106       public Set<String> topologiesWithMissingInputTopics() {
107           return missingInputTopicsBySubtopology.keySet()
108               .stream()
109               .map(s -> getTopologyNameOrElseUnnamed(s.namedTopology))
110               .collect(Collectors.toSet());
111       }

113       public Queue<StreamsException> missingSourceTopicExceptions() {
114           return missingInputTopicsBySubtopology.entrySet().stream().map(entry -> {
115               final Set<String> missingSourceTopics = entry.getValue();
116               final int subtopologyId = entry.getKey().nodeGroupId;
117               final String topologyName = entry.getKey().namedTopology;
118   
119               return new StreamsException(
120                   new MissingSourceTopicException(String.format(
121                       "Missing source topics %s for subtopology %d of topology %s",
122                       missingSourceTopics, subtopologyId, topologyName)),
123                   new TaskId(subtopologyId, 0, topologyName));
124           }).collect(Collectors.toCollection(LinkedList::new));
125       }
126   
127       public Map<TopicPartition, PartitionInfo> topicPartitionsInfo() {
128           return Collections.unmodifiableMap(topicPartitionInfos);
129       }

131       /**
132        * @param clusterMetadata  cluster metadata, eg which topics exist on the brokers
133        */
134       private Map<String, InternalTopicConfig> computeRepartitionTopicConfig(final Cluster clusterMetadata) {
135           final Set<TopicsInfo> allTopicsInfo = new HashSet<>();
136           final Map<String, InternalTopicConfig> allRepartitionTopicConfigs = new HashMap<>();
137           for (final Map.Entry<String, Map<Subtopology, TopicsInfo>> topologyEntry : topologyMetadata.topologyToSubtopologyTopicsInfoMap().entrySet()) {
138               final String topologyName = topologyMetadata.hasNamedTopologies() ? topologyEntry.getKey() : null;
139   
140               final Set<TopicsInfo> topicsInfoForTopology = new HashSet<>();
141               final Set<String> missingSourceTopicsForTopology = new HashSet<>();
142               final Map<String, InternalTopicConfig> repartitionTopicConfigsForTopology = new HashMap<>();
143   
144               for (final Map.Entry<Subtopology, TopicsInfo> subtopologyEntry : topologyEntry.getValue().entrySet()) {
145                   final TopicsInfo topicsInfo = subtopologyEntry.getValue();
146   
147                   topicsInfoForTopology.add(topicsInfo);
148                   repartitionTopicConfigsForTopology.putAll(
149                       topicsInfo.repartitionSourceTopics
150                           .values()
151                           .stream()
152                           .collect(Collectors.toMap(InternalTopicConfig::name, topicConfig -> topicConfig)));
153   
154                   final Set<String> missingSourceTopicsForSubtopology = computeMissingExternalSourceTopics(topicsInfo, clusterMetadata);
155                   missingSourceTopicsForTopology.addAll(missingSourceTopicsForSubtopology);
156                   if (!missingSourceTopicsForSubtopology.isEmpty()) {
157                       final Subtopology subtopology = subtopologyEntry.getKey();
158                       missingInputTopicsBySubtopology.put(subtopology, missingSourceTopicsForSubtopology);
159                       log.error("Subtopology {} was missing source topics {} and will be excluded from the current assignment, "
160                           + "this can be due to the consumer client's metadata being stale or because they have "
161                           + "not been created yet. Please verify that you have created all input topics; if they "
162                           + "do exist, you just need to wait for the metadata to be updated, at which time a new "
163                           + "rebalance will be kicked off automatically and the topology will be retried at that time.",
164                           subtopology.nodeGroupId, missingSourceTopicsForSubtopology);
165                   }
166               }
167   
168               if (missingSourceTopicsForTopology.isEmpty()) {
169                   allRepartitionTopicConfigs.putAll(repartitionTopicConfigsForTopology);
170                   allTopicsInfo.addAll(topicsInfoForTopology);
171               } else {
172                   log.debug("Skipping repartition topic validation for entire topology {} due to missing source topics {}",
173                       topologyName, missingSourceTopicsForTopology);
174               }
175           }
176           setRepartitionSourceTopicPartitionCount(allRepartitionTopicConfigs, allTopicsInfo, clusterMetadata);
177   
178           return allRepartitionTopicConfigs;
179       }

181       private void ensureCopartitioning(final Collection<Set<String>> copartitionGroups,
182                                         final Map<String, InternalTopicConfig> repartitionTopicMetadata,
183                                         final Cluster clusterMetadata) {
184           for (final Set<String> copartitionGroup : copartitionGroups) {
185               copartitionedTopicsEnforcer.enforce(copartitionGroup, repartitionTopicMetadata, clusterMetadata);
186           }
187       }

189       private Set<String> computeMissingExternalSourceTopics(final TopicsInfo topicsInfo,
190                                                              final Cluster clusterMetadata) {
191           final Set<String> missingExternalSourceTopics = new HashSet<>(topicsInfo.sourceTopics);
192           missingExternalSourceTopics.removeAll(topicsInfo.repartitionSourceTopics.keySet());
193           missingExternalSourceTopics.removeAll(clusterMetadata.topics());
194           return missingExternalSourceTopics;
195       }

197       /**
198        * Computes the number of partitions and sets it for each repartition topic in repartitionTopicMetadata
199        */
200       private void setRepartitionSourceTopicPartitionCount(final Map<String, InternalTopicConfig> repartitionTopicMetadata,
201                                                            final Collection<TopicsInfo> topicGroups,
202                                                            final Cluster clusterMetadata) {
203           boolean partitionCountNeeded;
204           do {
205               partitionCountNeeded = false;
206               boolean progressMadeThisIteration = false;  // avoid infinitely looping without making any progress on unknown repartitions
207   
208               for (final TopicsInfo topicsInfo : topicGroups) {
209                   for (final String repartitionSourceTopic : topicsInfo.repartitionSourceTopics.keySet()) {
210                       final Optional<Integer> repartitionSourceTopicPartitionCount =
211                           repartitionTopicMetadata.get(repartitionSourceTopic).numberOfPartitions();
212   
213                       if (!repartitionSourceTopicPartitionCount.isPresent()) {
214                           final Integer numPartitions = computePartitionCount(
215                               repartitionTopicMetadata,
216                               topicGroups,
217                               clusterMetadata,
218                               repartitionSourceTopic
219                           );
220   
221                           if (numPartitions == null) {
222                               partitionCountNeeded = true;
223                               log.trace("Unable to determine number of partitions for {}, another iteration is needed",
224                                   repartitionSourceTopic);
225                           } else {
226                               log.trace("Determined number of partitions for {} to be {}", repartitionSourceTopic, numPartitions);
227                               repartitionTopicMetadata.get(repartitionSourceTopic).setNumberOfPartitions(numPartitions);
228                               progressMadeThisIteration = true;
229                           }
230                       }
231                   }
232               }
233               if (!progressMadeThisIteration && partitionCountNeeded) {
234                   log.error("Unable to determine the number of partitions of all repartition topics, most likely a source topic is missing or pattern doesn't match any topics\n" +
235                       "topic groups: {}\n" +
236                       "cluster topics: {}.", topicGroups, clusterMetadata.topics());
237                   throw new TaskAssignmentException("Failed to compute number of partitions for all repartition topics, " +
238                       "make sure all user input topics are created and all Pattern subscriptions match at least one topic in the cluster");
239               }
240           } while (partitionCountNeeded);
241       }

243       private Integer computePartitionCount(final Map<String, InternalTopicConfig> repartitionTopicMetadata,
244                                             final Collection<TopicsInfo> topicGroups,
245                                             final Cluster clusterMetadata,
246                                             final String repartitionSourceTopic) {
247           Integer partitionCount = null;
248           // try set the number of partitions for this repartition topic if it is not set yet
249           for (final TopicsInfo topicsInfo : topicGroups) {
250               final Set<String> sinkTopics = topicsInfo.sinkTopics;
251   
252               if (sinkTopics.contains(repartitionSourceTopic)) {
253                   // if this topic is one of the sink topics of this topology,
254                   // use the maximum of all its source topic partitions as the number of partitions
255                   for (final String upstreamSourceTopic : topicsInfo.sourceTopics) {
256                       Integer numPartitionsCandidate = null;
257                       // It is possible the sourceTopic is another internal topic, i.e,
258                       // map().join().join(map())
259                       if (repartitionTopicMetadata.containsKey(upstreamSourceTopic)) {
260                           if (repartitionTopicMetadata.get(upstreamSourceTopic).numberOfPartitions().isPresent()) {
261                               numPartitionsCandidate =
262                                   repartitionTopicMetadata.get(upstreamSourceTopic).numberOfPartitions().get();
263                           }
264                       } else {
265                           final Integer count = clusterMetadata.partitionCountForTopic(upstreamSourceTopic);
266                           if (count == null) {
267                               throw new TaskAssignmentException(
268                                   "No partition count found for source topic "
269                                       + upstreamSourceTopic
270                                       + ", but it should have been."
271                               );
272                           }
273                           numPartitionsCandidate = count;
274                       }
275   
276                       if (numPartitionsCandidate != null) {
277                           if (partitionCount == null || numPartitionsCandidate > partitionCount) {
278                               partitionCount = numPartitionsCandidate;
279                           }
280                       }
281                   }
282               }
283           }
284           return partitionCount;
285       }
