143       private void putAndMaybeForward(final List<DirtyEntry> entries,
144                                       final InternalProcessorContext<?, ?> context) {
145   
146           // Track what base key or index key we already processed so don't reprocess
147           final Set<Bytes> processedBasedKey = new HashSet<>();
148   
149           for (final ThreadCache.DirtyEntry entry : entries) {
150               final byte[] binaryWindowKey = baseKeyCacheFunction.key(entry.key()).get();
151               final boolean isBaseKey = PrefixedWindowKeySchemas.isTimeFirstSchemaKey(
152                   binaryWindowKey);
153   
154               final DirtyEntry finalEntry;
155               if (!isBaseKey) {
156                   final Bytes baseKey = indexKeyToBaseKey(Bytes.wrap(binaryWindowKey));
157                   if (hasIndex && processedBasedKey.contains(baseKey)) {
158                       // Processed in base
159                       continue;
160                   }
161   
162                   final Bytes cachedBaseKey = baseKeyCacheFunction.cacheKey(baseKey);
163                   final LRUCacheEntry value = context.cache().get(cacheName, cachedBaseKey);
164                   // Base key value is already evicted, which should be handled already
165                   if (value == null) {
166                       continue;
167                   }
168   
169                   finalEntry = new DirtyEntry(entry.key(), value.value(), value);
170   
171                   if (hasIndex) {
172                       processedBasedKey.add(baseKey);
173                   }
174               } else {
175                   final Bytes baseKey = Bytes.wrap(binaryWindowKey);
176                   if (hasIndex && processedBasedKey.contains(baseKey)) {
177                       // Processed in index
178                       continue;
179                   }
180                   finalEntry = entry;
181                   if (hasIndex) {
182                       processedBasedKey.add(Bytes.wrap(binaryWindowKey));
183                   }
184               }
185   
186               final Windowed<Bytes> windowedKeyBytes;
187               if (isBaseKey) {
188                   windowedKeyBytes = TimeFirstWindowKeySchema.fromStoreBytesKey(binaryWindowKey,
189                       windowSize);
190               } else {
191                   windowedKeyBytes = KeyFirstWindowKeySchema.fromStoreBytesKey(binaryWindowKey,
192                       windowSize);
193               }
194   
195               final long windowStartTimestamp = windowedKeyBytes.window().start();
196               final Bytes binaryKey = windowedKeyBytes.key();
197   
198               putAndMaybeForward(context, finalEntry, binaryKey, windowStartTimestamp);
199           }
200       }

202       private void putAndMaybeForward(final InternalProcessorContext<?, ?> context,
203                                       final DirtyEntry finalEntry,
204                                       final Bytes binaryKey,
205                                       final long windowStartTimestamp) {
206           if (flushListener != null) {
207               final byte[] rawNewValue = finalEntry.newValue();
208               final byte[] rawOldValue = rawNewValue == null || sendOldValues ?
209                   wrapped().fetch(binaryKey, windowStartTimestamp) : null;
210   
211               // this is an optimization: if this key did not exist in underlying store and also not in the cache,
212               // we can skip flushing to downstream as well as writing to underlying store
213               if (rawNewValue != null || rawOldValue != null) {
214                   // we need to get the old values if needed, and then put to store, and then flush
215                   final ProcessorRecordContext current = context.recordContext();
216                   try {
217                       context.setRecordContext(finalEntry.entry().context());
218                       wrapped().put(binaryKey, finalEntry.newValue(), windowStartTimestamp);
219   
220                       flushListener.apply(
221                           new Record<>(
222                               WindowKeySchema.toStoreKeyBinary(binaryKey,
223                                       windowStartTimestamp, 0)
224                                   .get(),
225                               new Change<>(rawNewValue, sendOldValues ? rawOldValue : null),
226                               finalEntry.entry().context().timestamp(),
227                               finalEntry.entry().context().headers()));
228                   } finally {
229                       context.setRecordContext(current);
230                   }
231               }
232           } else {
233               final ProcessorRecordContext current = context.recordContext();
234               try {
235                   context.setRecordContext(finalEntry.entry().context());
236                   wrapped().put(binaryKey, finalEntry.newValue(), windowStartTimestamp);
237               } finally {
238                   context.setRecordContext(current);
239               }
240           }
241       }

259       @Override
260       public synchronized void put(final Bytes key,
261                                    final byte[] value,
262                                    final long windowStartTimestamp) {
263           // since this function may not access the underlying inner store, we need to validate
264           // if store is open outside as well.
265           validateStoreOpen();
266   
267           final Bytes baseKeyBytes = TimeFirstWindowKeySchema.toStoreKeyBinary(key, windowStartTimestamp, 0);
268           final LRUCacheEntry entry =
269               new LRUCacheEntry(
270                   value,
271                   context.headers(),
272                   true,
273                   context.offset(),
274                   context.timestamp(),
275                   context.partition(),
276                   context.topic());
277   
278           // Put to index first so that base can be evicted later
279           if (hasIndex) {
280               // Important: put base key first to avoid the situation that if we put index first,
281               // it could be evicted when we are putting base key. In that case, base key is not yet
282               // in cache so we can't store key/value to store when index is evicted. Then if we fetch
283               // using index, we can't find it in either store or cache
284               context.cache().put(cacheName, baseKeyCacheFunction.cacheKey(baseKeyBytes), entry);
285               final LRUCacheEntry emptyEntry =
286                   new LRUCacheEntry(
287                       new byte[0],
288                       new RecordHeaders(),
289                       true,
290                       context.offset(),
291                       context.timestamp(),
292                       context.partition(),
293                       "");
294               final Bytes indexKey = KeyFirstWindowKeySchema.toStoreKeyBinary(key, windowStartTimestamp, 0);
295               context.cache().put(cacheName, indexKeyCacheFunction.cacheKey(indexKey), emptyEntry);
296           } else {
297               context.cache().put(cacheName, baseKeyCacheFunction.cacheKey(baseKeyBytes), entry);
298           }
299           maxObservedTimestamp.set(Math.max(windowStartTimestamp, maxObservedTimestamp.get()));
300       }

423       private KeyValueIterator<Windowed<Bytes>, byte[]> fetchKeyRange(final KeyValueIterator<Windowed<Bytes>, byte[]> underlyingIterator,

537       private class CacheIteratorWrapper implements PeekingKeyValueIterator<Bytes, LRUCacheEntry> {
538   
539           private final long segmentInterval;
540           private final Bytes keyFrom;
541           private final Bytes keyTo;
542           private final long timeTo;
543           private final boolean forward;
544           private final boolean useIndex; // If we are iterating from index
545   
546           private long lastSegmentId;
547           private long currentSegmentId;
548           private Bytes cacheKeyFrom;
549           private Bytes cacheKeyTo;
550           private LRUCacheEntry cachedBaseValue;
551           private final SegmentedCacheFunction cacheFunction;
552   
553           private ThreadCache.MemoryLRUCacheBytesIterator current;
554   
555           private CacheIteratorWrapper(final Bytes key,
556                                        final long timeFrom,
557                                        final long timeTo,
558                                        final boolean forward,
559                                        final boolean index) {
560               this(key, key, timeFrom, timeTo, forward, index);
561           }
562   
563           private CacheIteratorWrapper(final Bytes keyFrom,
564                                        final Bytes keyTo,
565                                        final long timeFrom,
566                                        final long timeTo,
567                                        final boolean forward,
568                                        final boolean index) {
569               this.keyFrom = keyFrom;
570               this.keyTo = keyTo;
571               this.timeTo = timeTo;
572               this.forward = forward;
573               this.useIndex = index;
574   
575               cacheFunction = index ? indexKeyCacheFunction : baseKeyCacheFunction;
576   
577               this.segmentInterval = cacheFunction.getSegmentInterval();
578   
579               if (forward) {
580                   this.lastSegmentId = cacheFunction.segmentId(Math.min(timeTo, maxObservedTimestamp.get()));
581                   this.currentSegmentId = cacheFunction.segmentId(timeFrom);
582   
583                   setCacheKeyRange(timeFrom, currentSegmentLastTime());
584                   this.current = context.cache().range(cacheName, cacheKeyFrom, cacheKeyTo);
585               } else {
586                   this.currentSegmentId = cacheFunction.segmentId(Math.min(timeTo, maxObservedTimestamp.get()));
587                   this.lastSegmentId = cacheFunction.segmentId(timeFrom);
588   
589                   setCacheKeyRange(currentSegmentBeginTime(), Math.min(timeTo, maxObservedTimestamp.get()));
590                   this.current = context.cache().reverseRange(cacheName, cacheKeyFrom, cacheKeyTo);
591               }
592           }

594           @Override
595           public boolean hasNext() {
596               if (current == null) {
597                   return false;
598               }
599   
600               if (useIndex) {
601                   do {
602                       // If iterating from index, need to make sure base key/value exist in cache
603                       while (current.hasNext()) {
604                           final Bytes cacheIndexKey = current.peekNextKey();
605                           final Bytes indexKey = indexKeyCacheFunction.key(cacheIndexKey);
606                           final Bytes baseKey = indexKeyToBaseKey(indexKey);
607                           final Bytes cachedBaseKey = baseKeyCacheFunction.cacheKey(baseKey);
608                           cachedBaseValue = context.cache().get(cacheName, cachedBaseKey);
609                           if (cachedBaseValue != null) {
610                               return true;
611                           }
612                           current.next();
613                       }
614                       getNextSegmentIterator();
615                   } while (current != null);
616                   return false;
617               }
618   
619               if (current.hasNext()) {
620                   return true;
621               }
622   
623               while (!current.hasNext()) {
624                   getNextSegmentIterator();
625                   if (current == null) {
626                       return false;
627                   }
628               }
629               return true;
630           }
631   
632           @Override
633           public Bytes peekNextKey() {
634               if (!hasNext()) {
635                   throw new NoSuchElementException();
636               }
637               return current.peekNextKey();
638           }

646                   final KeyValue<Bytes, LRUCacheEntry> kv = current.peekNext();
647                   return KeyValue.pair(kv.key, cachedBaseValue);
648               }

650           }

655                   throw new NoSuchElementException();
656               }
657               if (useIndex) {
658                   final KeyValue<Bytes, LRUCacheEntry> kv = current.next();

669           private long currentSegmentBeginTime() {
670               return currentSegmentId * segmentInterval;
671           }
672   
673           private long currentSegmentLastTime() {
674               return Math.min(timeTo, currentSegmentBeginTime() + segmentInterval - 1);
675           }
676   
677           private void getNextSegmentIterator() {
678               if (forward) {
679                   ++currentSegmentId;
680                   // updating as maxObservedTimestamp can change while iterating
681                   lastSegmentId = cacheFunction.segmentId(Math.min(timeTo, maxObservedTimestamp.get()));
682   
683                   if (currentSegmentId > lastSegmentId) {
684                       current = null;
685                       return;
686                   }
687   
688                   setCacheKeyRange(currentSegmentBeginTime(), currentSegmentLastTime());
689   
690                   current.close();
691   
692                   current = context.cache().range(cacheName, cacheKeyFrom, cacheKeyTo);
693               } else {
694                   --currentSegmentId;
695   
696                   // last segment id is stable when iterating backward, therefore no need to update
697                   if (currentSegmentId < lastSegmentId) {
698                       current = null;
699                       return;
700                   }
701   
702                   setCacheKeyRange(currentSegmentBeginTime(), currentSegmentLastTime());
703   
704                   current.close();
705   
706                   current = context.cache().reverseRange(cacheName, cacheKeyFrom, cacheKeyTo);
707               }
708           }
709   
710           private void setCacheKeyRange(final long lowerRangeEndTime, final long upperRangeEndTime) {
711               if (cacheFunction.segmentId(lowerRangeEndTime) != cacheFunction.segmentId(upperRangeEndTime)) {
712                   throw new IllegalStateException("Error iterating over segments: segment interval has changed");
713               }
714   
715               final KeySchema schema = useIndex ? indexKeySchema : baseKeySchema;
716   
717               if (keyFrom != null && keyFrom.equals(keyTo)) {
718                   cacheKeyFrom = cacheFunction.cacheKey(schema.lowerRangeFixedSize(keyFrom, lowerRangeEndTime), currentSegmentId);
719                   cacheKeyTo = cacheFunction.cacheKey(schema.upperRangeFixedSize(keyTo, upperRangeEndTime), currentSegmentId);
720               } else {
721                   cacheKeyFrom = cacheFunction.cacheKey(schema.lowerRange(keyFrom, lowerRangeEndTime), currentSegmentId);
722                   cacheKeyTo = cacheFunction.cacheKey(schema.upperRange(keyTo, timeTo), currentSegmentId);
723               }
724           }
725       }
