45   public abstract class AbstractKStreamTimeWindowAggregateProcessor<KIn, VIn, VAgg> extends ContextualProcessor<KIn, VIn, Windowed<KIn>, Change<VAgg>> {
46   
47       private final Time time = Time.SYSTEM;
48       private final String storeName;
49       private final EmitStrategy emitStrategy;
50       private final boolean sendOldValues;
51       protected final TimeTracker timeTracker = new TimeTracker();
52   
53       private TimestampedTupleForwarder<Windowed<KIn>, VAgg> tupleForwarder;
54       protected TimestampedWindowStore<KIn, VAgg> windowStore;
55       protected Sensor droppedRecordsSensor;
56       protected Sensor emittedRecordsSensor;
57       protected Sensor emitFinalLatencySensor;
58       protected long lastEmitWindowCloseTime = ConsumerRecord.NO_TIMESTAMP;
59       protected long observedStreamTime = ConsumerRecord.NO_TIMESTAMP;
60       protected InternalProcessorContext<Windowed<KIn>, Change<VAgg>> internalProcessorContext;
61   
62       protected AbstractKStreamTimeWindowAggregateProcessor(final String storeName,
63                                                             final EmitStrategy emitStrategy,
64                                                             final boolean sendOldValues) {
65           this.storeName = storeName;
66           this.emitStrategy = emitStrategy;
67           this.sendOldValues = sendOldValues;
68       }
69   
70       @Override
71       public void init(final ProcessorContext<Windowed<KIn>, Change<VAgg>> context) {
72           super.init(context);
73           internalProcessorContext = (InternalProcessorContext<Windowed<KIn>, Change<VAgg>>) context;
74           final StreamsMetricsImpl metrics = internalProcessorContext.metrics();
75           final String threadId = Thread.currentThread().getName();
76           droppedRecordsSensor = droppedRecordsSensor(threadId, context.taskId().toString(), metrics);
77           emittedRecordsSensor = emittedRecordsSensor(threadId, context.taskId().toString(),
78               internalProcessorContext.currentNode().name(), metrics);
79           emitFinalLatencySensor = emitFinalLatencySensor(threadId, context.taskId().toString(),
80               internalProcessorContext.currentNode().name(), metrics);
81           windowStore = context.getStateStore(storeName);
82   
83           if (emitStrategy.type() == StrategyType.ON_WINDOW_CLOSE) {
84               // Restore last emit close time for ON_WINDOW_CLOSE strategy
85               final Long lastEmitWindowCloseTime = internalProcessorContext.processorMetadataForKey(storeName);
86               if (lastEmitWindowCloseTime != null) {
87                   this.lastEmitWindowCloseTime = lastEmitWindowCloseTime;
88               }
89               final long emitInterval = StreamsConfig.InternalConfig.getLong(
90                   context.appConfigs(),
91                   EMIT_INTERVAL_MS_KSTREAMS_WINDOWED_AGGREGATION,
92                   1000L
93               );
94               timeTracker.setEmitInterval(emitInterval);
95   
96               tupleForwarder = new TimestampedTupleForwarder<>(context, sendOldValues);
97           } else {
98               tupleForwarder = new TimestampedTupleForwarder<>(
99                   windowStore,
100                   context,
101                   new TimestampedCacheFlushListener<>(context),
102                   sendOldValues);
103           }
104       }
105   
106       protected void maybeForwardUpdate(final Record<KIn, VIn> record,
107                                         final Window window,
108                                         final VAgg oldAgg,
109                                         final VAgg newAgg,
110                                         final long newTimestamp) {
111           if (emitStrategy.type() == StrategyType.ON_WINDOW_CLOSE) {
112               return;
113           }
114   
115           tupleForwarder.maybeForward(
116               record.withKey(new Windowed<>(record.key(), window))
117                   .withValue(new Change<>(newAgg, sendOldValues ? oldAgg : null))
118                   .withTimestamp(newTimestamp));
119       }
120   
121       protected void maybeForwardFinalResult(final Record<KIn, VIn> record, final long windowCloseTime) {
122           if (shouldEmitFinal(windowCloseTime)) {
123               final long emitRangeUpperBound = emitRangeUpperBound(windowCloseTime);
124   
125               // if the upper bound is smaller than 0, then there's no window closed ever;
126               // and we can skip range fetching
127               if (emitRangeUpperBound >= 0) {
128                   final long emitRangeLowerBound = emitRangeLowerBound(windowCloseTime);
129   
130                   if (shouldRangeFetch(emitRangeLowerBound, emitRangeUpperBound)) {
131                       fetchAndEmit(record, windowCloseTime, emitRangeLowerBound, emitRangeUpperBound);
132                   }
133               }
134           }
135       }
136   
137       protected void logSkippedRecordForExpiredWindow(final Logger log,
138                                                       final long timestamp,
139                                                       final long windowExpire,
140                                                       final String window) {
141           if (context().recordMetadata().isPresent()) {
142               final RecordMetadata recordMetadata = context().recordMetadata().get();
143               log.warn("Skipping record for expired window. " +
144                   "topic=[{}] " +
145                   "partition=[{}] " +
146                   "offset=[{}] " +
147                   "timestamp=[{}] " +
148                   "window={} " +
149                   "expiration=[{}] " +
150                   "streamTime=[{}]",
151                   recordMetadata.topic(),
152                   recordMetadata.partition(),
153                   recordMetadata.offset(),
154                   timestamp,
155                   window,
156                   windowExpire,
157                   observedStreamTime
158               );
159           } else {
160               log.warn("Skipping record for expired window. Topic, partition, and offset not known. " +
161                   "timestamp=[{}] " +
162                   "window={} " +
163                   "expiration=[{}] " +
164                   "streamTime=[{}]",
165                   timestamp,
166                   window,
167                   windowExpire,
168                   observedStreamTime
169               );
170           }
171           droppedRecordsSensor.record();
172       }
173   
174       protected void updateObservedStreamTime(final long timestamp) {
175           observedStreamTime = Math.max(observedStreamTime, timestamp);
176       }
177   
178       private boolean shouldEmitFinal(final long closeTime) {
179           if (emitStrategy.type() != StrategyType.ON_WINDOW_CLOSE) {
180               return false;
181           }
182   
183           final long now = internalProcessorContext.currentSystemTimeMs();
184           // Throttle emit frequency
185           if (now < timeTracker.nextTimeToEmit) {
186               return false;
187           }
188   
189           // Schedule next emit time based on now to avoid the case that if system time jumps a lot,
190           // this can be triggered every time
191           timeTracker.nextTimeToEmit = now;
192           timeTracker.advanceNextTimeToEmit();
193   
194           // Only EMIT if the window close time does progress
195           return lastEmitWindowCloseTime == ConsumerRecord.NO_TIMESTAMP || lastEmitWindowCloseTime < closeTime;
196       }
197   
198       private void fetchAndEmit(final Record<KIn, VIn> record,
199                                 final long windowCloseTime,
200                                 final long emitRangeLowerBound,
201                                 final long emitRangeUpperBound) {
202           final long startMs = time.milliseconds();
203   
204           final KeyValueIterator<Windowed<KIn>, ValueAndTimestamp<VAgg>> windowToEmit = windowStore
205               .fetchAll(emitRangeLowerBound, emitRangeUpperBound);
206   
207           int emittedCount = 0;
208           while (windowToEmit.hasNext()) {
209               emittedCount++;
210               final KeyValue<Windowed<KIn>, ValueAndTimestamp<VAgg>> kv = windowToEmit.next();
211   
212               tupleForwarder.maybeForward(
213                   record.withKey(kv.key)
214                       .withValue(new Change<>(kv.value.value(), null))
215                       .withTimestamp(kv.value.timestamp())
216                       .withHeaders(record.headers()));
217           }
218           emittedRecordsSensor.record(emittedCount);
219           emitFinalLatencySensor.record(time.milliseconds() - startMs);
220   
221           lastEmitWindowCloseTime = windowCloseTime;
222           internalProcessorContext.addProcessorMetadataKeyValue(storeName, windowCloseTime);
223       }
224   
225       // upper and lower bound are inclusive; the bounds could be negative in which case we would skip range fetching and emitting
226       abstract protected long emitRangeLowerBound(final long windowCloseTime);
227   
228       abstract protected long emitRangeUpperBound(final long windowCloseTime);
229   
230       abstract protected boolean shouldRangeFetch(final long emitRangeLowerBound, final long emitRangeUpperBound);
231   }
