139       /**
140        * @throws TaskAssignmentException if method fails to encode the data, e.g., if there is an
141        * IO exception during encoding
142        */
143       public ByteBuffer encode() {
144           final ByteArrayOutputStream baos = new ByteArrayOutputStream();
145   
146           try (final DataOutputStream out = new DataOutputStream(baos)) {
147               switch (usedVersion) {
148                   case 1:
149                       out.writeInt(usedVersion); // version
150                       encodeActiveAndStandbyTaskAssignment(out);
151                       break;
152                   case 2:
153                       out.writeInt(usedVersion); // version
154                       encodeActiveAndStandbyTaskAssignment(out);
155                       encodePartitionsByHost(out);
156                       break;
157                   case 3:
158                       out.writeInt(usedVersion);
159                       out.writeInt(commonlySupportedVersion);
160                       encodeActiveAndStandbyTaskAssignment(out);
161                       encodePartitionsByHost(out);
162                       break;
163                   case 4:
164                       out.writeInt(usedVersion);
165                       out.writeInt(commonlySupportedVersion);
166                       encodeActiveAndStandbyTaskAssignment(out);
167                       encodePartitionsByHost(out);
168                       out.writeInt(errCode);
169                       break;
170                   case 5:
171                       out.writeInt(usedVersion);
172                       out.writeInt(commonlySupportedVersion);
173                       encodeActiveAndStandbyTaskAssignment(out);
174                       encodePartitionsByHostAsDictionary(out);
175                       out.writeInt(errCode);
176                       break;
177                   case 6:
178                       out.writeInt(usedVersion);
179                       out.writeInt(commonlySupportedVersion);
180                       encodeActiveAndStandbyTaskAssignment(out);
181                       encodeActiveAndStandbyHostPartitions(out);
182                       out.writeInt(errCode);
183                       break;
184                   case 7:
185                   case 8:
186                   case 9:
187                   case 10:
188                   case 11:
189                       out.writeInt(usedVersion);
190                       out.writeInt(commonlySupportedVersion);
191                       encodeActiveAndStandbyTaskAssignment(out);
192                       encodeActiveAndStandbyHostPartitions(out);
193                       out.writeInt(errCode);
194                       out.writeLong(nextRebalanceMs);
195                       break;
196                   default:
197                       throw new IllegalStateException("Unknown metadata version: " + usedVersion
198                               + "; latest commonly supported version: " + commonlySupportedVersion);
199               }
200   
201               out.flush();
202               out.close();
203   
204               return ByteBuffer.wrap(baos.toByteArray());
205           } catch (final IOException ex) {
206               throw new TaskAssignmentException("Failed to encode AssignmentInfo", ex);
207           }
208       }

210       private void encodeActiveAndStandbyTaskAssignment(final DataOutputStream out) throws IOException {
211           // encode active tasks
212           out.writeInt(activeTasks.size());
213           for (final TaskId id : activeTasks) {
214               writeTaskIdTo(id, out, usedVersion);
215           }
216   
217           // encode standby tasks
218           out.writeInt(standbyTasks.size());
219           for (final Map.Entry<TaskId, Set<TopicPartition>> entry : standbyTasks.entrySet()) {
220               final TaskId id = entry.getKey();
221               writeTaskIdTo(id, out, usedVersion);
222   
223               final Set<TopicPartition> partitions = entry.getValue();
224               writeTopicPartitions(out, partitions);
225           }
226       }

228       private void encodePartitionsByHost(final DataOutputStream out) throws IOException {
229           // encode partitions by host
230           out.writeInt(partitionsByHost.size());
231           for (final Map.Entry<HostInfo, Set<TopicPartition>> entry : partitionsByHost.entrySet()) {
232               writeHostInfo(out, entry.getKey());
233               writeTopicPartitions(out, entry.getValue());
234           }
235       }

237       private void encodeHostPartitionMapUsingDictionary(final DataOutputStream out,
238                                                          final Map<String, Integer> topicNameDict,
239                                                          final Map<HostInfo, Set<TopicPartition>> hostPartitionMap) throws IOException {
240           // encode partitions by host
241           out.writeInt(hostPartitionMap.size());
242   
243           // Write the topic index, partition
244           for (final Map.Entry<HostInfo, Set<TopicPartition>> entry : hostPartitionMap.entrySet()) {
245               writeHostInfo(out, entry.getKey());
246               out.writeInt(entry.getValue().size());
247               for (final TopicPartition partition : entry.getValue()) {
248                   out.writeInt(topicNameDict.get(partition.topic()));
249                   out.writeInt(partition.partition());
250               }
251           }
252       }

254       private Map<String, Integer> encodeTopicDictionaryAndGet(final DataOutputStream out,
255                                                                final Set<TopicPartition> topicPartitions) throws IOException {
256           // Build a dictionary to encode topicNames
257           int topicIndex = 0;
258           final Map<String, Integer> topicNameDict = new HashMap<>();
259           for (final TopicPartition topicPartition : topicPartitions) {
260               if (!topicNameDict.containsKey(topicPartition.topic())) {
261                   topicNameDict.put(topicPartition.topic(), topicIndex++);
262               }
263           }
264   
265           // write the topic name dictionary out
266           out.writeInt(topicNameDict.size());
267           for (final Map.Entry<String, Integer> entry : topicNameDict.entrySet()) {
268               out.writeInt(entry.getValue());
269               out.writeUTF(entry.getKey());
270           }
271   
272           return topicNameDict;
273       }

280       }

286           final Map<String, Integer> topicNameDict = encodeTopicDictionaryAndGet(out, allTopicPartitions);
287           encodeHostPartitionMapUsingDictionary(out, topicNameDict, partitionsByHost);
288           encodeHostPartitionMapUsingDictionary(out, topicNameDict, standbyPartitionsByHost);
289       }

305       /**
306        * @throws TaskAssignmentException if method fails to decode the data or if the data version is unknown
307        */
308       public static AssignmentInfo decode(final ByteBuffer data) {
309           // ensure we are at the beginning of the ByteBuffer
310           data.rewind();
311   
312           try (final DataInputStream in = new DataInputStream(new ByteBufferInputStream(data))) {
313               final AssignmentInfo assignmentInfo;
314   
315               final int usedVersion = in.readInt();
316               final int commonlySupportedVersion;
317               switch (usedVersion) {
318                   case 1:
319                       assignmentInfo = new AssignmentInfo(usedVersion, UNKNOWN);
320                       decodeActiveTasks(assignmentInfo, in);
321                       decodeStandbyTasks(assignmentInfo, in);
322                       assignmentInfo.partitionsByHost = new HashMap<>();
323                       break;
324                   case 2:
325                       assignmentInfo = new AssignmentInfo(usedVersion, UNKNOWN);
326                       decodeActiveTasks(assignmentInfo, in);
327                       decodeStandbyTasks(assignmentInfo, in);
328                       decodePartitionsByHost(assignmentInfo, in);
329                       break;
330                   case 3:
331                       commonlySupportedVersion = in.readInt();
332                       assignmentInfo = new AssignmentInfo(usedVersion, commonlySupportedVersion);
333                       decodeActiveTasks(assignmentInfo, in);
334                       decodeStandbyTasks(assignmentInfo, in);
335                       decodePartitionsByHost(assignmentInfo, in);
336                       break;
337                   case 4:
338                       commonlySupportedVersion = in.readInt();
339                       assignmentInfo = new AssignmentInfo(usedVersion, commonlySupportedVersion);
340                       decodeActiveTasks(assignmentInfo, in);
341                       decodeStandbyTasks(assignmentInfo, in);
342                       decodePartitionsByHost(assignmentInfo, in);
343                       assignmentInfo.errCode = in.readInt();
344                       break;
345                   case 5:
346                       commonlySupportedVersion = in.readInt();
347                       assignmentInfo = new AssignmentInfo(usedVersion, commonlySupportedVersion);
348                       decodeActiveTasks(assignmentInfo, in);
349                       decodeStandbyTasks(assignmentInfo, in);
350                       decodePartitionsByHostUsingDictionary(assignmentInfo, in);
351                       assignmentInfo.errCode = in.readInt();
352                       break;
353                   case 6:
354                       commonlySupportedVersion = in.readInt();
355                       assignmentInfo = new AssignmentInfo(usedVersion, commonlySupportedVersion);
356                       decodeActiveTasks(assignmentInfo, in);
357                       decodeStandbyTasks(assignmentInfo, in);
358                       decodeActiveAndStandbyHostPartitions(assignmentInfo, in);
359                       assignmentInfo.errCode = in.readInt();
360                       break;
361                   case 7:
362                   case 8:
363                   case 9:
364                   case 10:
365                   case 11:
366                       commonlySupportedVersion = in.readInt();
367                       assignmentInfo = new AssignmentInfo(usedVersion, commonlySupportedVersion);
368                       decodeActiveTasks(assignmentInfo, in);
369                       decodeStandbyTasks(assignmentInfo, in);
370                       decodeActiveAndStandbyHostPartitions(assignmentInfo, in);
371                       assignmentInfo.errCode = in.readInt();
372                       assignmentInfo.nextRebalanceMs = in.readLong();
373                       break;
374                   default:
375                       final TaskAssignmentException fatalException = new TaskAssignmentException("Unable to decode assignment data: " +
376                           "used version: " + usedVersion + "; latest supported version: " + LATEST_SUPPORTED_VERSION);
377                       log.error(fatalException.getMessage(), fatalException);
378                       throw fatalException;
379               }
380   
381               return assignmentInfo;
382           } catch (final IOException ex) {
383               throw new TaskAssignmentException("Failed to decode AssignmentInfo", ex);
384           }
385       }

387       private static void decodeActiveTasks(final AssignmentInfo assignmentInfo,
388                                             final DataInputStream in) throws IOException {
389           final int count = in.readInt();
390           assignmentInfo.activeTasks = new ArrayList<>(count);
391           for (int i = 0; i < count; i++) {
392               assignmentInfo.activeTasks.add(readTaskIdFrom(in, assignmentInfo.usedVersion));
393           }
394       }

397                                              final DataInputStream in) throws IOException {
398           final int count = in.readInt();
399           assignmentInfo.standbyTasks = new HashMap<>(count);
400           for (int i = 0; i < count; i++) {
401               final TaskId id = readTaskIdFrom(in, assignmentInfo.usedVersion);
402               assignmentInfo.standbyTasks.put(id, readTopicPartitions(in));
403           }

459                                                              final Map<Integer, String> topicIndexDict) throws IOException {

464           }
