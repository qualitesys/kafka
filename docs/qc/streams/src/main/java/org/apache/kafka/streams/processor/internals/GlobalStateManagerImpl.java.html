83       public GlobalStateManagerImpl(final LogContext logContext,
84                                     final Time time,

98           for (final StateStore store : topology.globalStateStores()) {
99               globalStoreNames.add(store.name());
100               if (!store.persistent()) {
101                   globalNonPersistentStoresTopics.add(changelogFor(store.name()));
102               }
103           }
104   
105           log = logContext.logger(GlobalStateManagerImpl.class);

116               config.getLong(StreamsConfig.POLL_MS_CONFIG) + requestTimeoutMs

126       @Override
127       public Set<String> initialize() {
128           try {
129               checkpointFileCache.putAll(checkpointFile.read());
130           } catch (final IOException e) {
131               throw new StreamsException("Failed to read checkpoints for global state globalStores", e);
132           }
133   
134           final Set<String> changelogTopics = new HashSet<>();
135           for (final StateStore stateStore : topology.globalStateStores()) {
136               final String sourceTopic = storeToChangelogTopic.get(stateStore.name());
137               changelogTopics.add(sourceTopic);
138               stateStore.init((StateStoreContext) globalProcessorContext, stateStore);
139           }
140   
141           // make sure each topic-partition from checkpointFileCache is associated with a global state store
142           checkpointFileCache.keySet().forEach(tp -> {
143               if (!changelogTopics.contains(tp.topic())) {
144                   log.error(
145                       "Encountered a topic-partition in the global checkpoint file not associated with any global" +
146                           " state store, topic-partition: {}, checkpoint file: {}. If this topic-partition is no longer valid," +
147                           " an application reset and state store directory cleanup will be required.",
148                       tp.topic(),
149                       checkpointFile.toString()
150                   );
151                   throw new StreamsException("Encountered a topic-partition not associated with any global state store");
152               }
153           });
154           return Collections.unmodifiableSet(globalStoreNames);
155       }

170       @Override
171       public void registerStore(final StateStore store,
172                                 final StateRestoreCallback stateRestoreCallback,
173                                 final CommitCallback ignored) {
174           log.info("Restoring state for global store {}", store.name());
175   
176           // TODO (KAFKA-12887): we should not trigger user's exception handler for illegal-argument but always
177           // fail-crash; in this case we would not need to immediately close the state store before throwing
178           if (globalStores.containsKey(store.name())) {
179               store.close();
180               throw new IllegalArgumentException(String.format("Global Store %s has already been registered", store.name()));
181           }
182   
183           if (!globalStoreNames.contains(store.name())) {
184               store.close();
185               throw new IllegalArgumentException(String.format("Trying to register store %s that is not a known global store", store.name()));
186           }
187   
188           // register the store first, so that if later an exception is thrown then eventually while we call `close`
189           // on the state manager this state store would be closed as well
190           globalStores.put(store.name(), Optional.of(store));
191   
192           if (stateRestoreCallback == null) {
193               throw new IllegalArgumentException(String.format("The stateRestoreCallback provided for store %s was null", store.name()));
194           }
195   
196           final List<TopicPartition> topicPartitions = topicPartitionsForStore(store);
197           final Map<TopicPartition, Long> highWatermarks = retryUntilSuccessOrThrowOnTaskTimeout(
198               () -> globalConsumer.endOffsets(topicPartitions),
199               String.format(
200                   "Failed to get offsets for partitions %s. The broker may be transiently unavailable at the moment.",
201                   topicPartitions
202               )
203           );
204   
205           try {
206               restoreState(
207                   stateRestoreCallback,
208                   topicPartitions,
209                   highWatermarks,
210                   store.name(),
211                   converterForStore(store)
212               );
213           } finally {
214               globalConsumer.unsubscribe();
215           }
216       }

218       private List<TopicPartition> topicPartitionsForStore(final StateStore store) {
219           final String sourceTopic = storeToChangelogTopic.get(store.name());
220   
221           final List<PartitionInfo> partitionInfos = retryUntilSuccessOrThrowOnTaskTimeout(
222               () -> globalConsumer.partitionsFor(sourceTopic),
223               String.format(
224                   "Failed to get partitions for topic %s. The broker may be transiently unavailable at the moment.",
225                   sourceTopic
226               )
227           );
228   
229           if (partitionInfos == null || partitionInfos.isEmpty()) {
230               throw new StreamsException(String.format("There are no partitions available for topic %s when initializing global store %s", sourceTopic, store.name()));
231           }
232   
233           final List<TopicPartition> topicPartitions = new ArrayList<>();
234           for (final PartitionInfo partition : partitionInfos) {
235               topicPartitions.add(new TopicPartition(partition.topic(), partition.partition()));
236           }
237           return topicPartitions;
238       }

240       private void restoreState(final StateRestoreCallback stateRestoreCallback,
241                                 final List<TopicPartition> topicPartitions,
242                                 final Map<TopicPartition, Long> highWatermarks,
243                                 final String storeName,
244                                 final RecordConverter recordConverter) {
245           for (final TopicPartition topicPartition : topicPartitions) {
246               long currentDeadline = NO_DEADLINE;
247   
248               globalConsumer.assign(Collections.singletonList(topicPartition));
249               long offset;
250               final Long checkpoint = checkpointFileCache.get(topicPartition);
251               if (checkpoint != null) {
252                   globalConsumer.seek(topicPartition, checkpoint);
253                   offset = checkpoint;
254               } else {
255                   globalConsumer.seekToBeginning(Collections.singletonList(topicPartition));
256                   offset = getGlobalConsumerOffset(topicPartition);
257               }
258   
259               final Long highWatermark = highWatermarks.get(topicPartition);
260               final RecordBatchingStateRestoreCallback stateRestoreAdapter =
261                   StateRestoreCallbackAdapter.adapt(stateRestoreCallback);
262   
263               stateRestoreListener.onRestoreStart(topicPartition, storeName, offset, highWatermark);
264               long restoreCount = 0L;
265   
266               while (offset < highWatermark) {
267                   // we add `request.timeout.ms` to `poll.ms` because `poll.ms` might be too short
268                   // to give a fetch request a fair chance to actually complete and we don't want to
269                   // start `task.timeout.ms` too early
270                   //
271                   // TODO with https://issues.apache.org/jira/browse/KAFKA-10315 we can just call
272                   //      `poll(pollMS)` without adding the request timeout and do a more precise
273                   //      timeout handling
274                   final ConsumerRecords<byte[], byte[]> records = globalConsumer.poll(pollMsPlusRequestTimeout);
275                   if (records.isEmpty()) {
276                       currentDeadline = maybeUpdateDeadlineOrThrow(currentDeadline);
277                   } else {
278                       currentDeadline = NO_DEADLINE;
279                   }
280   
281                   final List<ConsumerRecord<byte[], byte[]>> restoreRecords = new ArrayList<>();
282                   for (final ConsumerRecord<byte[], byte[]> record : records.records(topicPartition)) {
283                       if (record.key() != null) {
284                           restoreRecords.add(recordConverter.convert(record));
285                       }
286                   }
287   
288                   offset = getGlobalConsumerOffset(topicPartition);
289   
290                   stateRestoreAdapter.restoreBatch(restoreRecords);
291                   stateRestoreListener.onBatchRestored(topicPartition, storeName, offset, restoreRecords.size());
292                   restoreCount += restoreRecords.size();
293               }
294               stateRestoreListener.onRestoreEnd(topicPartition, storeName, restoreCount);
295               checkpointFileCache.put(topicPartition, offset);
296           }
297       }

299       private long getGlobalConsumerOffset(final TopicPartition topicPartition) {
300           return retryUntilSuccessOrThrowOnTaskTimeout(
301               () -> globalConsumer.position(topicPartition),
302               String.format(
303                   "Failed to get position for partition %s. The broker may be transiently unavailable at the moment.",
304                   topicPartition
305               )
306           );
307       }

309       private <R> R retryUntilSuccessOrThrowOnTaskTimeout(final Supplier<R> supplier,
310                                                           final String errorMessage) {
311           long deadlineMs = NO_DEADLINE;
312   
313           do {
314               try {
315                   return supplier.get();
316               } catch (final TimeoutException retriableException) {
317                   if (taskTimeoutMs == 0L) {
318                       throw new StreamsException(
319                           String.format(
320                               "Retrying is disabled. You can enable it by setting `%s` to a value larger than zero.",
321                               StreamsConfig.TASK_TIMEOUT_MS_CONFIG
322                           ),
323                           retriableException
324                       );
325                   }
326   
327                   deadlineMs = maybeUpdateDeadlineOrThrow(deadlineMs);
328   
329                   log.warn(errorMessage, retriableException);
330               }
331           } while (true);
332       }
333   
334       private long maybeUpdateDeadlineOrThrow(final long currentDeadlineMs) {
335           final long currentWallClockMs = time.milliseconds();
336   
337           if (currentDeadlineMs == NO_DEADLINE) {
338               final long newDeadlineMs = currentWallClockMs + taskTimeoutMs;
339               return newDeadlineMs < 0L ? Long.MAX_VALUE : newDeadlineMs;
340           } else if (currentWallClockMs >= currentDeadlineMs) {
341               throw new TimeoutException(String.format(
342                   "Global task did not make progress to restore state within %d ms. Adjust `%s` if needed.",
343                   currentWallClockMs - currentDeadlineMs + taskTimeoutMs,
344                   StreamsConfig.TASK_TIMEOUT_MS_CONFIG
345               ));
346           }
347   
348           return currentDeadlineMs;
349       }

351       @Override
352       public void flush() {
353           log.debug("Flushing all global globalStores registered in the state manager");
354           for (final Map.Entry<String, Optional<StateStore>> entry : globalStores.entrySet()) {
355               if (entry.getValue().isPresent()) {
356                   final StateStore store = entry.getValue().get();
357                   try {
358                       log.trace("Flushing global store={}", store.name());
359                       store.flush();
360                   } catch (final RuntimeException e) {
361                       throw new ProcessorStateException(
362                           String.format("Failed to flush global state store %s", store.name()),
363                           e
364                       );
365                   }
366               } else {
367                   throw new IllegalStateException("Expected " + entry.getKey() + " to have been initialized");
368               }
369           }
370       }

372       @Override
373       public void close() {
374           if (globalStores.isEmpty()) {
375               return;
376           }
377           final StringBuilder closeFailed = new StringBuilder();
378           for (final Map.Entry<String, Optional<StateStore>> entry : globalStores.entrySet()) {
379               if (entry.getValue().isPresent()) {
380                   log.debug("Closing global storage engine {}", entry.getKey());
381                   try {
382                       entry.getValue().get().close();
383                   } catch (final RuntimeException e) {
384                       log.error("Failed to close global state store {}", entry.getKey(), e);
385                       closeFailed.append("Failed to close global state store:")
386                           .append(entry.getKey())
387                           .append(". Reason: ")
388                           .append(e)
389                           .append("\n");
390                   }
391                   globalStores.put(entry.getKey(), Optional.empty());
392               } else {
393                   log.info("Skipping to close non-initialized store {}", entry.getKey());
394               }
395           }
396   
397           if (closeFailed.length() > 0) {
398               throw new ProcessorStateException("Exceptions caught during close of 1 or more global state globalStores\n" + closeFailed);
399           }
400       }

407       @Override
408       public void checkpoint() {
409           final Map<TopicPartition, Long> filteredOffsets = new HashMap<>();
410   
411           // Skip non persistent store
412           for (final Map.Entry<TopicPartition, Long> topicPartitionOffset : checkpointFileCache.entrySet()) {
413               final String topic = topicPartitionOffset.getKey().topic();
414               if (!globalNonPersistentStoresTopics.contains(topic)) {
415                   filteredOffsets.put(topicPartitionOffset.getKey(), topicPartitionOffset.getValue());
416               }
417           }
418   
419           try {
420               checkpointFile.write(filteredOffsets);
421           } catch (final IOException e) {
422               log.warn("Failed to write offset checkpoint file to {} for global stores: {}." +
423                   " This may occur if OS cleaned the state.dir in case when it is located in the (default) ${java.io.tmpdir}/kafka-streams directory." +
424                   " Changing the location of state.dir may resolve the problem", checkpointFile, e);
425           }
426       }
