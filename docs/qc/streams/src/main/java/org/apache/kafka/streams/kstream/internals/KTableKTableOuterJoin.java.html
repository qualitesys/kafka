39                             final KTableImpl<K, ?, V2> table2,
40                             final ValueJoiner<? super V1, ? super V2, ? extends VOut> joiner) {

42       }

66       private class KTableKTableOuterJoinProcessor extends ContextualProcessor<K, Change<V1>, K, Change<VOut>> {
67   
68           private final KTableValueGetter<K, V2> valueGetter;
69           private Sensor droppedRecordsSensor;
70   
71           KTableKTableOuterJoinProcessor(final KTableValueGetter<K, V2> valueGetter) {
72               this.valueGetter = valueGetter;
73           }
74   
75           @Override
76           public void init(final ProcessorContext<K, Change<VOut>> context) {
77               super.init(context);
78               droppedRecordsSensor = droppedRecordsSensor(
79                   Thread.currentThread().getName(),
80                   context.taskId().toString(),
81                   (StreamsMetricsImpl) context.metrics()
82               );
83               valueGetter.init(context);
84           }
85   
86           @Override
87           public void process(final Record<K, Change<V1>> record) {
88               // we do join iff keys are equal, thus, if key is null we cannot join and just ignore the record
89               if (record.key() == null) {
90                   if (context().recordMetadata().isPresent()) {
91                       final RecordMetadata recordMetadata = context().recordMetadata().get();
92                       LOG.warn(
93                           "Skipping record due to null key. "
94                               + "topic=[{}] partition=[{}] offset=[{}]",
95                           recordMetadata.topic(), recordMetadata.partition(), recordMetadata.offset()
96                       );
97                   } else {
98                       LOG.warn(
99                           "Skipping record due to null key. Topic, partition, and offset not known."
100                       );
101                   }
102                   droppedRecordsSensor.record();
103                   return;
104               }
105   
106               VOut newValue = null;
107               final long resultTimestamp;
108               VOut oldValue = null;
109   
110               final ValueAndTimestamp<V2> valueAndTimestamp2 = valueGetter.get(record.key());
111               final V2 value2 = getValueOrNull(valueAndTimestamp2);
112               if (value2 == null) {
113                   if (record.value().newValue == null && record.value().oldValue == null) {
114                       return;
115                   }
116                   resultTimestamp = record.timestamp();
117               } else {
118                   resultTimestamp = Math.max(record.timestamp(), valueAndTimestamp2.timestamp());
119               }
120   
121               if (value2 != null || record.value().newValue != null) {
122                   newValue = joiner.apply(record.value().newValue, value2);
123               }
124   
125               if (sendOldValues && (value2 != null || record.value().oldValue != null)) {
126                   oldValue = joiner.apply(record.value().oldValue, value2);
127               }
128   
129               context().forward(record.withValue(new Change<>(newValue, oldValue)).withTimestamp(resultTimestamp));
130           }
131   
132           @Override
133           public void close() {
134               valueGetter.close();
135           }
136       }

138       private class KTableKTableOuterJoinValueGetter implements KTableValueGetter<K, VOut> {
139   
140           private final KTableValueGetter<K, V1> valueGetter1;
141           private final KTableValueGetter<K, V2> valueGetter2;
142   
143           KTableKTableOuterJoinValueGetter(final KTableValueGetter<K, V1> valueGetter1,
144                                            final KTableValueGetter<K, V2> valueGetter2) {
145               this.valueGetter1 = valueGetter1;
146               this.valueGetter2 = valueGetter2;
147           }
148   
149           @Override
150           public void init(final ProcessorContext<?, ?> context) {
151               valueGetter1.init(context);
152               valueGetter2.init(context);
153           }
154   
155           @Override
156           public ValueAndTimestamp<VOut> get(final K key) {
157               VOut newValue = null;
158   
159               final ValueAndTimestamp<V1> valueAndTimestamp1 = valueGetter1.get(key);
160               final V1 value1;
161               final long timestamp1;
162               if (valueAndTimestamp1 == null) {
163                   value1 = null;
164                   timestamp1 = UNKNOWN;
165               } else {
166                   value1 = valueAndTimestamp1.value();
167                   timestamp1 = valueAndTimestamp1.timestamp();
168               }
169   
170               final ValueAndTimestamp<V2> valueAndTimestamp2 = valueGetter2.get(key);
171               final V2 value2;
172               final long timestamp2;
173               if (valueAndTimestamp2 == null) {
174                   value2 = null;
175                   timestamp2 = UNKNOWN;
176               } else {
177                   value2 = valueAndTimestamp2.value();
178                   timestamp2 = valueAndTimestamp2.timestamp();
179               }
180   
181               if (value1 != null || value2 != null) {
182                   newValue = joiner.apply(value1, value2);
183               }
184   
185               return ValueAndTimestamp.make(newValue, Math.max(timestamp1, timestamp2));
186           }
187   
188           @Override
189           public void close() {
190               valueGetter1.close();
191               valueGetter2.close();
192           }
193       }
