17   package org.apache.kafka.streams;
18   
19   import org.apache.kafka.streams.state.HostInfo;
20   
21   import java.util.Collections;
22   import java.util.Objects;
23   import java.util.Set;
24   
25   /**
26    * Represents all the metadata related to a key, where a particular key resides in a {@link KafkaStreams} application.
27    * It contains the active {@link HostInfo} and a set of standby {@link HostInfo}s, denoting the instances where the key resides.
28    * It also contains the partition number where the key belongs, which could be useful when used in conjunction with other APIs.
29    * e.g: Relating with lags for that store partition.
30    * NOTE: This is a point in time view. It may change as rebalances happen.
31    */
32   public class KeyQueryMetadata {
33       /**
34        * Sentinel to indicate that the KeyQueryMetadata is currently unavailable. This can occur during rebalance
35        * operations.
36        */
37       public static final KeyQueryMetadata NOT_AVAILABLE =
38           new KeyQueryMetadata(HostInfo.unavailable(), Collections.emptySet(), -1);
39   
40       private final HostInfo activeHost;
41   
42       private final Set<HostInfo> standbyHosts;
43   
44       private final int partition;
45   
46       public KeyQueryMetadata(final HostInfo activeHost, final Set<HostInfo> standbyHosts, final int partition) {
47           this.activeHost = activeHost;
48           this.standbyHosts = standbyHosts;
49           this.partition = partition;
50       }
51   
52       /**
53        * Get the active Kafka Streams instance for given key.
54        *
55        * @return active instance's {@link HostInfo}
56        * @deprecated Use {@link #activeHost()} instead.
57        */
58       @Deprecated
59       public HostInfo getActiveHost() {
60           return activeHost;
61       }
62   
63       /**
64        * Get the Kafka Streams instances that host the key as standbys.
65        *
66        * @return set of standby {@link HostInfo} or a empty set, if no standbys are configured
67        * @deprecated Use {@link #standbyHosts()} instead.
68        */
69       @Deprecated
70       public Set<HostInfo> getStandbyHosts() {
71           return standbyHosts;
72       }
73   
74       /**
75        * Get the store partition corresponding to the key.
76        *
77        * @return store partition number
78        * @deprecated Use {@link #partition()} instead.
79        */
80       @Deprecated
81       public int getPartition() {
82           return partition;
83       }
84   
85       /**
86        * Get the active Kafka Streams instance for given key.
87        *
88        * @return active instance's {@link HostInfo}
89        */
90       public HostInfo activeHost() {
91           return activeHost;
92       }
93   
94       /**
95        * Get the Kafka Streams instances that host the key as standbys.
96        *
97        * @return set of standby {@link HostInfo} or a empty set, if no standbys are configured
98        */
99       public Set<HostInfo> standbyHosts() {
100           return standbyHosts;
101       }
102   
103       /**
104        * Get the store partition corresponding to the key.
105        *
106        * @return store partition number
107        */
108       public int partition() {
109           return partition;
110       }
111   
112       @Override
113       public boolean equals(final Object obj) {
114           if (!(obj instanceof KeyQueryMetadata)) {
115               return false;
116           }
117           final KeyQueryMetadata keyQueryMetadata = (KeyQueryMetadata) obj;
118           return Objects.equals(keyQueryMetadata.activeHost, activeHost)
119               && Objects.equals(keyQueryMetadata.standbyHosts, standbyHosts)
120               && Objects.equals(keyQueryMetadata.partition, partition);
121       }
122   
123       @Override
124       public String toString() {
125           return "KeyQueryMetadata {" +
126                   "activeHost=" + activeHost +
127                   ", standbyHosts=" + standbyHosts +
128                   ", partition=" + partition +
129                   '}';
130       }
131   
132       @Override
133       public int hashCode() {
134           return Objects.hash(activeHost, standbyHosts, partition);
135       }
136   }
