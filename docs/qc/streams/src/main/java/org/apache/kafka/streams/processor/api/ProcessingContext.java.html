35   /**
36    * Processor context interface.
37    */
38   public interface ProcessingContext {
39   
40       /**
41        * Return the application id.
42        *
43        * @return the application id
44        */
45       String applicationId();
46   
47       /**
48        * Return the task id.
49        *
50        * @return the task id
51        */
52       TaskId taskId();
53   
54       /**
55        * Return the metadata of the current record if available. Processors may be invoked to
56        * process a source record from an input topic, to run a scheduled punctuation
57        * (see {@link ProcessingContext#schedule(Duration, PunctuationType, Punctuator)}),
58        * or because a parent processor called {@code forward(Record)}.
59        * <p>
60        * In the case of a punctuation, there is no source record, so this metadata would be
61        * undefined. Note that when a punctuator invokes {@code forward(Record)},
62        * downstream processors will receive the forwarded record as a regular
63        * {@link Processor#process(Record)} or {@link FixedKeyProcessor#process(FixedKeyRecord)} invocation.
64        * In other words, it wouldn't be apparent to
65        * downstream processors whether the record being processed came from an input topic
66        * or punctuation and therefore whether this metadata is defined. This is why
67        * the return type of this method is {@link Optional}.
68        * <p>
69        * If there is any possibility of punctuators upstream, any access
70        * to this field should consider the case of
71        * "<code>recordMetadata().isPresent() == false</code>".
72        * Of course, it would be safest to always guard this condition.
73        */
74       Optional<RecordMetadata> recordMetadata();
75   
76       /**
77        * Return the default key serde.
78        *
79        * @return the key serializer
80        */
81       Serde<?> keySerde();
82   
83       /**
84        * Return the default value serde.
85        *
86        * @return the value serializer
87        */
88       Serde<?> valueSerde();
89   
90       /**
91        * Return the state directory for the partition.
92        *
93        * @return the state directory
94        */
95       File stateDir();
96   
97       /**
98        * Return Metrics instance.
99        *
100        * @return StreamsMetrics
101        */
102       StreamsMetrics metrics();
103   
104       /**
105        * Get the state store given the store name.
106        *
107        * @param name The store name
108        * @param <S> The type or interface of the store to return
109        * @return The state store instance
110        *
111        * @throws ClassCastException if the return type isn't a type or interface of the actual returned store.
112        */
113       <S extends StateStore> S getStateStore(final String name);
114   
115       /**
116        * Schedule a periodic operation for processors. A processor may call this method during
117        * {@link Processor#init(ProcessorContext) initialization},
118        * {@link Processor#process(Record) processing},
119        * {@link FixedKeyProcessor#init(FixedKeyProcessorContext) initialization}, or
120        * {@link FixedKeyProcessor#process(FixedKeyRecord) processing} to
121        * schedule a periodic callback &mdash; called a punctuation &mdash; to {@link Punctuator#punctuate(long)}.
122        * The type parameter controls what notion of time is used for punctuation:
123        * <ul>
124        *   <li>{@link PunctuationType#STREAM_TIME} &mdash; uses "stream time", which is advanced by the processing of messages
125        *   in accordance with the timestamp as extracted by the {@link TimestampExtractor} in use.
126        *   The first punctuation will be triggered by the first record that is processed.
127        *   <b>NOTE:</b> Only advanced if messages arrive</li>
128        *   <li>{@link PunctuationType#WALL_CLOCK_TIME} &mdash; uses system time (the wall-clock time),
129        *   which is advanced independent of whether new messages arrive.
130        *   The first punctuation will be triggered after interval has elapsed.
131        *   <b>NOTE:</b> This is best effort only as its granularity is limited by how long an iteration of the
132        *   processing loop takes to complete</li>
133        * </ul>
134        *
135        * <b>Skipping punctuations:</b> Punctuations will not be triggered more than once at any given timestamp.
136        * This means that "missed" punctuation will be skipped.
137        * It's possible to "miss" a punctuation if:
138        * <ul>
139        *   <li>with {@link PunctuationType#STREAM_TIME}, when stream time advances more than interval</li>
140        *   <li>with {@link PunctuationType#WALL_CLOCK_TIME}, on GC pause, too short interval, ...</li>
141        * </ul>
142        *
143        * @param interval the time interval between punctuations (supported minimum is 1 millisecond)
144        * @param type one of: {@link PunctuationType#STREAM_TIME}, {@link PunctuationType#WALL_CLOCK_TIME}
145        * @param callback a function consuming timestamps representing the current stream or system time
146        * @return a handle allowing cancellation of the punctuation schedule established by this method
147        * @throws IllegalArgumentException if the interval is not representable in milliseconds
148        */
149       Cancellable schedule(final Duration interval,
150                            final PunctuationType type,
151                            final Punctuator callback);
152   
153       /**
154        * Request a commit.
155        */
156       void commit();
157   
158       /**
159        * Returns all the application config properties as key/value pairs.
160        *
161        * <p> The config properties are defined in the {@link org.apache.kafka.streams.StreamsConfig}
162        * object and associated to the ProcessorContext.
163        *
164        * <p> The type of the values is dependent on the {@link org.apache.kafka.common.config.ConfigDef.Type type} of the property
165        * (e.g. the value of {@link org.apache.kafka.streams.StreamsConfig#DEFAULT_KEY_SERDE_CLASS_CONFIG DEFAULT_KEY_SERDE_CLASS_CONFIG}
166        * will be of type {@link Class}, even if it was specified as a String to
167        * {@link org.apache.kafka.streams.StreamsConfig#StreamsConfig(Map) StreamsConfig(Map)}).
168        *
169        * @return all the key/values from the StreamsConfig properties
170        */
171       Map<String, Object> appConfigs();
172   
173       /**
174        * Return all the application config properties with the given key prefix, as key/value pairs
175        * stripping the prefix.
176        *
177        * <p> The config properties are defined in the {@link org.apache.kafka.streams.StreamsConfig}
178        * object and associated to the ProcessorContext.
179        *
180        * @param prefix the properties prefix
181        * @return the key/values matching the given prefix from the StreamsConfig properties.
182        */
183       Map<String, Object> appConfigsWithPrefix(final String prefix);
184   
185       /**
186        * Return the current system timestamp (also called wall-clock time) in milliseconds.
187        *
188        * <p> Note: this method returns the internally cached system timestamp from the Kafka Stream runtime.
189        * Thus, it may return a different value compared to {@code System.currentTimeMillis()}.
190        *
191        * @return the current system timestamp in milliseconds
192        */
193       long currentSystemTimeMs();
194   
195       /**
196        * Return the current stream-time in milliseconds.
197        *
198        * <p> Stream-time is the maximum observed {@link TimestampExtractor record timestamp} so far
199        * (including the currently processed record), i.e., it can be considered a high-watermark.
200        * Stream-time is tracked on a per-task basis and is preserved across restarts and during task migration.
201        *
202        * <p> Note: this method is not supported for global processors (cf. {@link Topology#addGlobalStore} (...)
203        * and {@link StreamsBuilder#addGlobalStore} (...),
204        * because there is no concept of stream-time for this case.
205        * Calling this method in a global processor will result in an {@link UnsupportedOperationException}.
206        *
207        * @return the current stream-time in milliseconds
208        */
209       long currentStreamTimeMs();
210   }
