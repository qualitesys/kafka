17   package org.apache.kafka.streams.processor.internals.assignment;

19   import org.apache.kafka.common.Cluster;
20   import org.apache.kafka.common.utils.LogContext;
21   import org.apache.kafka.streams.errors.TopologyException;
22   import org.apache.kafka.streams.processor.internals.InternalTopicConfig;
23   import org.slf4j.Logger;
24   
25   import java.util.Collection;
26   import java.util.Map;
27   import java.util.Map.Entry;
28   import java.util.Optional;
29   import java.util.Set;
30   import java.util.TreeMap;
31   import java.util.function.Supplier;
32   import java.util.stream.Collectors;
33   
34   public class CopartitionedTopicsEnforcer {
35       private final String logPrefix;
36       private final Logger log;
37   
38       public CopartitionedTopicsEnforcer(final String logPrefix) {
39           this.logPrefix = logPrefix;
40           final LogContext logContext = new LogContext(logPrefix);
41           log = logContext.logger(getClass());
42       }
43   
44       public void enforce(final Set<String> copartitionGroup,
45                           final Map<String, InternalTopicConfig> allRepartitionTopicsNumPartitions,
46                           final Cluster metadata) {
47           if (copartitionGroup.isEmpty()) {
48               return;
49           }
50   
51           final Map<Object, InternalTopicConfig> repartitionTopicConfigs =
52               copartitionGroup.stream()
53                               .filter(allRepartitionTopicsNumPartitions::containsKey)
54                               .collect(Collectors.toMap(topic -> topic, allRepartitionTopicsNumPartitions::get));
55   
56           final Map<String, Integer> nonRepartitionTopicPartitions =
57               copartitionGroup.stream().filter(topic -> !allRepartitionTopicsNumPartitions.containsKey(topic))
58                               .collect(Collectors.toMap(topic -> topic, topic -> {
59                                   final Integer partitions = metadata.partitionCountForTopic(topic);
60                                   if (partitions == null) {
61                                       final String str = String.format("%sTopic not found: %s", logPrefix, topic);
62                                       log.error(str);
63                                       throw new IllegalStateException(str);
64                                   } else {
65                                       return partitions;
66                                   }
67                               }));
68   
69           final int numPartitionsToUseForRepartitionTopics;
70           final Collection<InternalTopicConfig> internalTopicConfigs = repartitionTopicConfigs.values();
71   
72           if (copartitionGroup.equals(repartitionTopicConfigs.keySet())) {
73               final Collection<InternalTopicConfig> internalTopicConfigsWithEnforcedNumberOfPartitions = internalTopicConfigs
74                   .stream()
75                   .filter(InternalTopicConfig::hasEnforcedNumberOfPartitions)
76                   .collect(Collectors.toList());
77   
78               // if there's at least one repartition topic with enforced number of partitions
79               // validate that they all have same number of partitions
80               if (!internalTopicConfigsWithEnforcedNumberOfPartitions.isEmpty()) {
81                   numPartitionsToUseForRepartitionTopics = validateAndGetNumOfPartitions(
82                       repartitionTopicConfigs,
83                       internalTopicConfigsWithEnforcedNumberOfPartitions
84                   );
85               } else {
86                   // If all topics for this co-partition group are repartition topics,
87                   // then set the number of partitions to be the maximum of the number of partitions.
88                   numPartitionsToUseForRepartitionTopics = getMaxPartitions(repartitionTopicConfigs);
89               }
90           } else {
91               // Otherwise, use the number of partitions from external topics (which must all be the same)
92               numPartitionsToUseForRepartitionTopics = getSamePartitions(nonRepartitionTopicPartitions);
93           }
94   
95           // coerce all the repartition topics to use the decided number of partitions.
96           for (final InternalTopicConfig config : internalTopicConfigs) {
97               maybeSetNumberOfPartitionsForInternalTopic(numPartitionsToUseForRepartitionTopics, config);
98   
99               final int numberOfPartitionsOfInternalTopic = config
100                   .numberOfPartitions()
101                   .orElseThrow(emptyNumberOfPartitionsExceptionSupplier(config.name()));
102   
103               if (numberOfPartitionsOfInternalTopic != numPartitionsToUseForRepartitionTopics) {
104                   final String msg = String.format("%sNumber of partitions [%s] of repartition topic [%s] " +
105                                                    "doesn't match number of partitions [%s] of the source topic.",
106                                                    logPrefix,
107                                                    numberOfPartitionsOfInternalTopic,
108                                                    config.name(),
109                                                    numPartitionsToUseForRepartitionTopics);
110                   throw new TopologyException(msg);
111               }
112           }
113       }
114   
115       private static void maybeSetNumberOfPartitionsForInternalTopic(final int numPartitionsToUseForRepartitionTopics,
116                                                                      final InternalTopicConfig config) {
117           if (!config.hasEnforcedNumberOfPartitions()) {
118               config.setNumberOfPartitions(numPartitionsToUseForRepartitionTopics);
119           }
120       }
121   
122       private int validateAndGetNumOfPartitions(final Map<Object, InternalTopicConfig> repartitionTopicConfigs,
123                                                 final Collection<InternalTopicConfig> internalTopicConfigs) {
124           final InternalTopicConfig firstInternalTopicConfig = internalTopicConfigs.iterator().next();
125   
126           final int firstNumberOfPartitionsOfInternalTopic = firstInternalTopicConfig
127               .numberOfPartitions()
128               .orElseThrow(emptyNumberOfPartitionsExceptionSupplier(firstInternalTopicConfig.name()));
129   
130           for (final InternalTopicConfig internalTopicConfig : internalTopicConfigs) {
131               final Integer numberOfPartitions = internalTopicConfig
132                   .numberOfPartitions()
133                   .orElseThrow(emptyNumberOfPartitionsExceptionSupplier(internalTopicConfig.name()));
134   
135               if (numberOfPartitions != firstNumberOfPartitionsOfInternalTopic) {
136                   final Map<Object, Integer> repartitionTopics = repartitionTopicConfigs
137                       .entrySet()
138                       .stream()
139                       .collect(Collectors.toMap(Entry::getKey, entry -> entry.getValue().numberOfPartitions().get()));
140   
141                   final String msg = String.format("%sFollowing topics do not have the same number of partitions: [%s]",
142                                                    logPrefix,
143                                                    new TreeMap<>(repartitionTopics));
144                   throw new TopologyException(msg);
145               }
146           }
147   
148           return firstNumberOfPartitionsOfInternalTopic;
149       }
150   
151       private static Supplier<TopologyException> emptyNumberOfPartitionsExceptionSupplier(final String topic) {
152           return () -> new TopologyException("Number of partitions is not set for topic: " + topic);
153       }
154   
155       private int getSamePartitions(final Map<String, Integer> nonRepartitionTopicsInCopartitionGroup) {
156           final int partitions = nonRepartitionTopicsInCopartitionGroup.values().iterator().next();
157           for (final Entry<String, Integer> entry : nonRepartitionTopicsInCopartitionGroup.entrySet()) {
158               if (entry.getValue() != partitions) {
159                   final TreeMap<String, Integer> sorted = new TreeMap<>(nonRepartitionTopicsInCopartitionGroup);
160                   throw new TopologyException(
161                       String.format("%sTopics not co-partitioned: [%s]",
162                                     logPrefix, sorted)
163                   );
164               }
165           }
166           return partitions;
167       }
168   
169       private int getMaxPartitions(final Map<Object, InternalTopicConfig> repartitionTopicsInCopartitionGroup) {
170           int maxPartitions = 0;
171   
172           for (final InternalTopicConfig config : repartitionTopicsInCopartitionGroup.values()) {
173               final Optional<Integer> partitions = config.numberOfPartitions();
174               maxPartitions = Integer.max(maxPartitions, partitions.orElse(maxPartitions));
175           }
176           if (maxPartitions <= 0) {
177               throw new IllegalStateException(logPrefix + "Could not validate the copartitioning of topics: " + repartitionTopicsInCopartitionGroup.keySet());
178           }
179           return maxPartitions;
180       }
