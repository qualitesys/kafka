72           /**
73            * Create a buffer unconstrained by size (either keys or bytes).
74            *
75            * As a result, the buffer will consume as much memory as it needs, dictated by the time bound.
76            *
77            * If there isn't enough heap available to meet the demand, the application will encounter an
78            * {@link OutOfMemoryError} and shut down (not guaranteed to be a graceful exit). Also, note that
79            * JVM processes under extreme memory pressure may exhibit poor GC behavior.
80            *
81            * This is a convenient option if you doubt that your buffer will be that large, but also don't
82            * wish to pick particular constraints, such as in testing.
83            *
84            * This buffer is "strict" in the sense that it will enforce the time bound or crash.
85            * It will never emit early.
86            */
87           static StrictBufferConfig unbounded() {
88               return new StrictBufferConfigImpl();
89           }

92            * Set the buffer to be unconstrained by size (either keys or bytes).
93            *
94            * As a result, the buffer will consume as much memory as it needs, dictated by the time bound.

144       /**
145        * Configure the suppression to emit only the "final results" from the window.
146        *
147        * By default all Streams operators emit results whenever new results are available.
148        * This includes windowed operations.
149        *
150        * This configuration will instead emit just one result per key for each window, guaranteeing
151        * to deliver only the final result. This option is suitable for use cases in which the business logic
152        * requires a hard guarantee that only the final result is propagated. For example, sending alerts.
153        *
154        * To accomplish this, the operator will buffer events from the window until the window close (that is,
155        * until the end-time passes, and additionally until the grace period expires). Since windowed operators
156        * are required to reject out-of-order events for a window whose grace period is expired, there is an additional
157        * guarantee that the final results emitted from this suppression will match any queryable state upstream.
158        *
159        * @param bufferConfig A configuration specifying how much space to use for buffering intermediate results.
160        *                     This is required to be a "strict" config, since it would violate the "final results"
161        *                     property to emit early and then issue an update later.
162        * @return a "final results" mode suppression configuration
163        */
164       static Suppressed<Windowed> untilWindowCloses(final StrictBufferConfig bufferConfig) {
165           return new FinalResultsSuppressionBuilder<>(null, bufferConfig);
166       }
167   
168       /**
169        * Configure the suppression to wait {@code timeToWaitForMoreEvents} amount of time after receiving a record
170        * before emitting it further downstream. If another record for the same key arrives in the mean time, it replaces
171        * the first record in the buffer but does <em>not</em> re-start the timer.
172        *
173        * @param timeToWaitForMoreEvents The amount of time to wait, per record, for new events.
174        * @param bufferConfig A configuration specifying how much space to use for buffering intermediate results.
175        * @param <K> The key type for the KTable to apply this suppression to.
176        * @return a suppression configuration
177        */
178       static <K> Suppressed<K> untilTimeLimit(final Duration timeToWaitForMoreEvents, final BufferConfig bufferConfig) {
179           return new SuppressedInternal<>(null, timeToWaitForMoreEvents, bufferConfig, null, false);
180       }

182       /**
183        * Use the specified name for the suppression node in the topology.
184        * <p>
185        * This can be used to insert a suppression without changing the rest of the topology names
186        * (and therefore not requiring an application reset).
187        * <p>
188        * Note however, that once a suppression has buffered some records, removing it from the topology would cause
189        * the loss of those records.
190        * <p>
191        * A suppression can be "disabled" with the configuration {@code untilTimeLimit(Duration.ZERO, ...}.
192        *
193        * @param name The name to be used for the suppression node and changelog topic
194        * @return The same configuration with the addition of the given {@code name}.
195        */
196       @Override
197       Suppressed<K> withName(final String name);
