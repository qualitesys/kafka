18   package org.apache.kafka.streams.kstream.internals.foreignkeyjoin;
19   
20   import org.apache.kafka.common.metrics.Sensor;
21   import org.apache.kafka.common.serialization.Serde;
22   import org.apache.kafka.common.serialization.Serializer;
23   import org.apache.kafka.streams.kstream.internals.Change;
24   import org.apache.kafka.streams.processor.api.ContextualProcessor;
25   import org.apache.kafka.streams.processor.api.Processor;
26   import org.apache.kafka.streams.processor.api.ProcessorContext;
27   import org.apache.kafka.streams.processor.api.ProcessorSupplier;
28   import org.apache.kafka.streams.processor.api.Record;
29   import org.apache.kafka.streams.processor.api.RecordMetadata;
30   import org.apache.kafka.streams.processor.internals.metrics.StreamsMetricsImpl;
31   import org.apache.kafka.streams.processor.internals.metrics.TaskMetrics;
32   import org.apache.kafka.streams.state.internals.Murmur3;
33   import org.slf4j.Logger;
34   import org.slf4j.LoggerFactory;
35   
36   import java.util.Arrays;
37   import java.util.function.Function;
38   import java.util.function.Supplier;
39   
40   import static org.apache.kafka.streams.kstream.internals.foreignkeyjoin.SubscriptionWrapper.Instruction.DELETE_KEY_AND_PROPAGATE;
41   import static org.apache.kafka.streams.kstream.internals.foreignkeyjoin.SubscriptionWrapper.Instruction.DELETE_KEY_NO_PROPAGATE;
42   import static org.apache.kafka.streams.kstream.internals.foreignkeyjoin.SubscriptionWrapper.Instruction.PROPAGATE_NULL_IF_NO_FK_VAL_AVAILABLE;
43   import static org.apache.kafka.streams.kstream.internals.foreignkeyjoin.SubscriptionWrapper.Instruction.PROPAGATE_ONLY_IF_FK_VAL_AVAILABLE;
44   
45   public class ForeignJoinSubscriptionSendProcessorSupplier<K, KO, V> implements ProcessorSupplier<K, Change<V>, KO, SubscriptionWrapper<K>> {
46       private static final Logger LOG = LoggerFactory.getLogger(ForeignJoinSubscriptionSendProcessorSupplier.class);
47   
48       private final Function<V, KO> foreignKeyExtractor;
49       private final Supplier<String> foreignKeySerdeTopicSupplier;
50       private final Supplier<String> valueSerdeTopicSupplier;
51       private final boolean leftJoin;
52       private Serializer<KO> foreignKeySerializer;
53       private Serializer<V> valueSerializer;
54   
55       public ForeignJoinSubscriptionSendProcessorSupplier(final Function<V, KO> foreignKeyExtractor,
56                                                           final Supplier<String> foreignKeySerdeTopicSupplier,
57                                                           final Supplier<String> valueSerdeTopicSupplier,
58                                                           final Serde<KO> foreignKeySerde,
59                                                           final Serializer<V> valueSerializer,
60                                                           final boolean leftJoin) {
61           this.foreignKeyExtractor = foreignKeyExtractor;
62           this.foreignKeySerdeTopicSupplier = foreignKeySerdeTopicSupplier;
63           this.valueSerdeTopicSupplier = valueSerdeTopicSupplier;
64           this.valueSerializer = valueSerializer;
65           this.leftJoin = leftJoin;
66           foreignKeySerializer = foreignKeySerde == null ? null : foreignKeySerde.serializer();
67       }
68   
69       @Override
70       public Processor<K, Change<V>, KO, SubscriptionWrapper<K>> get() {
71           return new UnbindChangeProcessor();
72       }
73   
74       private class UnbindChangeProcessor extends ContextualProcessor<K, Change<V>, KO, SubscriptionWrapper<K>> {
75   
76           private Sensor droppedRecordsSensor;
77           private String foreignKeySerdeTopic;
78           private String valueSerdeTopic;
79   
80           @SuppressWarnings("unchecked")
81           @Override
82           public void init(final ProcessorContext<KO, SubscriptionWrapper<K>> context) {
83               super.init(context);
84               foreignKeySerdeTopic = foreignKeySerdeTopicSupplier.get();
85               valueSerdeTopic = valueSerdeTopicSupplier.get();
86               // get default key serde if it wasn't supplied directly at construction
87               if (foreignKeySerializer == null) {
88                   foreignKeySerializer = (Serializer<KO>) context.keySerde().serializer();
89               }
90               if (valueSerializer == null) {
91                   valueSerializer = (Serializer<V>) context.valueSerde().serializer();
92               }
93               droppedRecordsSensor = TaskMetrics.droppedRecordsSensor(
94                   Thread.currentThread().getName(),
95                   context.taskId().toString(),
96                   (StreamsMetricsImpl) context.metrics()
97               );
98           }
99   
100           @Override
101           public void process(final Record<K, Change<V>> record) {
102               final long[] currentHash = record.value().newValue == null ?
103                   null :
104                   Murmur3.hash128(valueSerializer.serialize(valueSerdeTopic, record.value().newValue));
105   
106               final int partition = context().recordMetadata().get().partition();
107               if (record.value().oldValue != null) {
108                   final KO oldForeignKey = foreignKeyExtractor.apply(record.value().oldValue);
109                   if (oldForeignKey == null) {
110                       if (context().recordMetadata().isPresent()) {
111                           final RecordMetadata recordMetadata = context().recordMetadata().get();
112                           LOG.warn(
113                               "Skipping record due to null foreign key. "
114                                   + "topic=[{}] partition=[{}] offset=[{}]",
115                               recordMetadata.topic(), recordMetadata.partition(), recordMetadata.offset()
116                           );
117                       } else {
118                           LOG.warn(
119                               "Skipping record due to null foreign key. Topic, partition, and offset not known."
120                           );
121                       }
122                       droppedRecordsSensor.record();
123                       return;
124                   }
125                   if (record.value().newValue != null) {
126                       final KO newForeignKey = foreignKeyExtractor.apply(record.value().newValue);
127                       if (newForeignKey == null) {
128                           if (context().recordMetadata().isPresent()) {
129                               final RecordMetadata recordMetadata = context().recordMetadata().get();
130                               LOG.warn(
131                                   "Skipping record due to null foreign key. "
132                                       + "topic=[{}] partition=[{}] offset=[{}]",
133                                   recordMetadata.topic(), recordMetadata.partition(), recordMetadata.offset()
134                               );
135                           } else {
136                               LOG.warn(
137                                   "Skipping record due to null foreign key. Topic, partition, and offset not known."
138                               );
139                           }
140                           droppedRecordsSensor.record();
141                           return;
142                       }
143   
144                       final byte[] serialOldForeignKey =
145                           foreignKeySerializer.serialize(foreignKeySerdeTopic, oldForeignKey);
146                       final byte[] serialNewForeignKey =
147                           foreignKeySerializer.serialize(foreignKeySerdeTopic, newForeignKey);
148                       if (!Arrays.equals(serialNewForeignKey, serialOldForeignKey)) {
149                           //Different Foreign Key - delete the old key value and propagate the new one.
150                           //Delete it from the oldKey's state store
151                           context().forward(
152                               record.withKey(oldForeignKey)
153                                   .withValue(new SubscriptionWrapper<>(
154                                       currentHash,
155                                       DELETE_KEY_NO_PROPAGATE,
156                                       record.key(),
157                                       partition
158                                   )));
159                           //Add to the newKey's state store. Additionally, propagate null if no FK is found there,
160                           //since we must "unset" any output set by the previous FK-join. This is true for both INNER
161                           //and LEFT join.
162                       }
163                       context().forward(
164                           record.withKey(newForeignKey)
165                               .withValue(new SubscriptionWrapper<>(
166                                   currentHash,
167                                   PROPAGATE_NULL_IF_NO_FK_VAL_AVAILABLE,
168                                   record.key(),
169                                   partition
170                               )));
171                   } else {
172                       //A simple propagatable delete. Delete from the state store and propagate the delete onwards.
173                       context().forward(
174                           record.withKey(oldForeignKey)
175                              .withValue(new SubscriptionWrapper<>(
176                                  currentHash,
177                                  DELETE_KEY_AND_PROPAGATE,
178                                  record.key(),
179                                  partition
180                              )));
181                   }
182               } else if (record.value().newValue != null) {
183                   //change.oldValue is null, which means it was deleted at least once before, or it is brand new.
184                   //In either case, we only need to propagate if the FK_VAL is available, as the null from the delete would
185                   //have been propagated otherwise.
186   
187                   final SubscriptionWrapper.Instruction instruction;
188                   if (leftJoin) {
189                       //Want to send info even if RHS is null.
190                       instruction = PROPAGATE_NULL_IF_NO_FK_VAL_AVAILABLE;
191                   } else {
192                       instruction = PROPAGATE_ONLY_IF_FK_VAL_AVAILABLE;
193                   }
194                   final KO newForeignKey = foreignKeyExtractor.apply(record.value().newValue);
195                   if (newForeignKey == null) {
196                       if (context().recordMetadata().isPresent()) {
197                           final RecordMetadata recordMetadata = context().recordMetadata().get();
198                           LOG.warn(
199                               "Skipping record due to null foreign key. "
200                                   + "topic=[{}] partition=[{}] offset=[{}]",
201                               recordMetadata.topic(), recordMetadata.partition(), recordMetadata.offset()
202                           );
203                       } else {
204                           LOG.warn(
205                               "Skipping record due to null foreign key. Topic, partition, and offset not known."
206                           );
207                       }
208                       droppedRecordsSensor.record();
209                   } else {
210                       context().forward(
211                           record.withKey(newForeignKey)
212                               .withValue(new SubscriptionWrapper<>(
213                                   currentHash,
214                                   instruction,
215                                   record.key(),
216                                   partition)));
217                   }
218               }
219           }
220       }
221   }
