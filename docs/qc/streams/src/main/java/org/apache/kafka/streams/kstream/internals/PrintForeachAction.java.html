17   package org.apache.kafka.streams.kstream.internals;
18   
19   import org.apache.kafka.streams.kstream.ForeachAction;
20   import org.apache.kafka.streams.kstream.KeyValueMapper;
21   
22   import java.io.OutputStream;
23   import java.io.OutputStreamWriter;
24   import java.io.PrintWriter;
25   import java.nio.charset.StandardCharsets;
26   
27   public class PrintForeachAction<K, V> implements ForeachAction<K, V> {
28   
29       private final String label;
30       private final PrintWriter printWriter;
31       private final boolean closable;
32       private final KeyValueMapper<? super K, ? super V, String> mapper;
33   
34       /**
35        * Print customized output with given writer. The {@link OutputStream} can be {@link System#out} or the others.
36        *
37        * @param outputStream The output stream to write to.
38        * @param mapper The mapper which can allow user to customize output will be printed.
39        * @param label The given name will be printed.
40        */
41       PrintForeachAction(final OutputStream outputStream,
42                          final KeyValueMapper<? super K, ? super V, String> mapper,
43                          final String label) {
44           this.printWriter = new PrintWriter(new OutputStreamWriter(outputStream, StandardCharsets.UTF_8));
45           this.closable = outputStream != System.out && outputStream != System.err;
46           this.mapper = mapper;
47           this.label = label;
48       }
49   
50       @Override
51       public void apply(final K key, final V value) {
52           final String data = String.format("[%s]: %s", label, mapper.apply(key, value));
53           printWriter.println(data);
54           if (!closable) {
55               printWriter.flush();
56           }
57       }
58   
59       public void close() {
60           if (closable) {
61               printWriter.close();
62           } else {
63               printWriter.flush();
64           }
65       }
66   
67   }
