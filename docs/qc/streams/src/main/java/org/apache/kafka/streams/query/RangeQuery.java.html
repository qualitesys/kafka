19   package org.apache.kafka.streams.query;
20   
21   
22   import org.apache.kafka.common.annotation.InterfaceStability.Evolving;
23   import org.apache.kafka.streams.state.KeyValueIterator;
24   
25   
26   import java.util.Optional;
27   
28   /**
29    * Interactive query for issuing range queries and scans over KeyValue stores.
30    * <p>
31    *  A range query retrieves a set of records, specified using an upper and/or lower bound on the keys.
32    * <p>
33    * A scan query retrieves all records contained in the store.
34    * <p>
35    */
36   @Evolving
37   public final class RangeQuery<K, V> implements Query<KeyValueIterator<K, V>> {
38   
39   
40       private final Optional<K> lower;
41       private final Optional<K> upper;
42   
43   
44       private RangeQuery(final Optional<K> lower, final Optional<K> upper) {
45           this.lower = lower;
46           this.upper = upper;
47       }
48   
49       /**
50        * Interactive range query using a lower and upper bound to filter the keys returned.
51        * @param lower The key that specifies the lower bound of the range
52        * @param upper The key that specifies the upper bound of the range
53        * @param <K> The key type
54        * @param <V> The value type
55        */
56       public static <K, V> RangeQuery<K, V> withRange(final K lower, final K upper) {
57           return new RangeQuery<>(Optional.of(lower), Optional.of(upper));
58       }
59   
60       /**
61        * Interactive range query using an upper bound to filter the keys returned.
62        * @param upper The key that specifies the upper bound of the range
63        * @param <K> The key type
64        * @param <V> The value type
65        */
66       public static <K, V> RangeQuery<K, V> withUpperBound(final K upper) {
67           return new RangeQuery<>(Optional.empty(), Optional.of(upper));
68       }
69   
70       /**
71        * Interactive range query using a lower bound to filter the keys returned.
72        * @param lower The key that specifies the lower bound of the range
73        * @param <K> The key type
74        * @param <V> The value type
75        */
76       public static <K, V> RangeQuery<K, V> withLowerBound(final K lower) {
77           return new RangeQuery<>(Optional.of(lower), Optional.empty());
78       }
79   
80       /**
81        * Interactive scan query that returns all records in the store.
82        * @param <K> The key type
83        * @param <V> The value type
84        */
85       public static <K, V> RangeQuery<K, V> withNoBounds() {
86           return new RangeQuery<>(Optional.empty(), Optional.empty());
87       }
88   
89       /**
90        * The lower bound of the query, if specified.
91        */
92       public Optional<K> getLowerBound() {
93           return lower;
94       }
95   
96       /**
97        * The upper bound of the query, if specified
98        */
99       public Optional<K> getUpperBound() {
100           return upper;
101       }
102   }
