17   package org.apache.kafka.streams.kstream;
18   
19   import org.apache.kafka.streams.KeyValue;
20   
21   /**
22    * The {@code Reducer} interface for combining two values of the same type into a new value.
23    * In contrast to {@link Aggregator} the result type must be the same as the input type.
24    * <p>
25    * The provided values can be either original values from input {@link KeyValue} pair records or be a previously
26    * computed result from {@link Reducer#apply(Object, Object)}.
27    * <p>
28    * {@code Reducer} can be used to implement aggregation functions like sum, min, or max.
29    *
30    * @param <V> value type
31    * @see KGroupedStream#reduce(Reducer)
32    * @see KGroupedStream#reduce(Reducer, Materialized)
33    * @see TimeWindowedKStream#reduce(Reducer)
34    * @see TimeWindowedKStream#reduce(Reducer, Materialized)
35    * @see SessionWindowedKStream#reduce(Reducer)
36    * @see SessionWindowedKStream#reduce(Reducer, Materialized)
37    * @see Aggregator
38    */
39   public interface Reducer<V> {
40   
41       /**
42        * Aggregate the two given values into a single one.
43        *
44        * @param value1 the first value for the aggregation
45        * @param value2 the second value for the aggregation
46        * @return the aggregated value
47        */
48       V apply(final V value1, final V value2);
49   }
