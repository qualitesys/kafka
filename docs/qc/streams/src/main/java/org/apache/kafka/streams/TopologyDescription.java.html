26   /**
27    * A meta representation of a {@link Topology topology}.
28    * <p>
29    * The nodes of a topology are grouped into {@link Subtopology sub-topologies} if they are connected.
30    * In contrast, two sub-topologies are not connected but can be linked to each other via topics, i.e., if one
31    * sub-topology {@link Topology#addSink(String, String, String...) writes} into a topic and another sub-topology
32    * {@link Topology#addSource(String, String...) reads} from the same topic.
33    * Message {@link ProcessorContext#forward(Object, Object) forwards} using custom Processors and Transformers are not considered in the topology graph.
34    * <p>
35    * When {@link KafkaStreams#start()} is called, different sub-topologies will be constructed and executed as independent
36    * {@link StreamTask tasks}.
37    */
38   public interface TopologyDescription {
39       /**
40        * A connected sub-graph of a {@link Topology}.
41        * <p>
42        * Nodes of a {@code Subtopology} are connected
43        * {@link Topology#addProcessor(String, ProcessorSupplier, String...) directly} or indirectly via
44        * {@link Topology#connectProcessorAndStateStores(String, String...) state stores}
45        * (i.e., if multiple processors share the same state).
46        */
47       interface Subtopology {
48           /**
49            * Internally assigned unique ID.
50            * @return the ID of the sub-topology
51            */
52           int id();
53   
54           /**
55            * All nodes of this sub-topology.
56            * @return set of all nodes within the sub-topology
57            */
58           Set<Node> nodes();
59       }
60   
61       /**
62        * Represents a {@link Topology#addGlobalStore(org.apache.kafka.streams.state.StoreBuilder, String,
63        * org.apache.kafka.common.serialization.Deserializer, org.apache.kafka.common.serialization.Deserializer, String,
64        * String, org.apache.kafka.streams.processor.api.ProcessorSupplier) global store}.
65        * Adding a global store results in adding a source node and one stateful processor node.
66        * Note, that all added global stores form a single unit (similar to a {@link Subtopology}) even if different
67        * global stores are not connected to each other.
68        * Furthermore, global stores are available to all processors without connecting them explicitly, and thus global
69        * stores will never be part of any {@link Subtopology}.
70        */
71       interface GlobalStore {
72           /**
73            * The source node reading from a "global" topic.
74            * @return the "global" source node
75            */
76           Source source();
77   
78           /**
79            * The processor node maintaining the global store.
80            * @return the "global" processor node
81            */
82           Processor processor();
83   
84           int id();
85       }
86   
87       /**
88        * A node of a topology. Can be a source, sink, or processor node.
89        */
90       interface Node {
91           /**
92            * The name of the node. Will never be {@code null}.
93            * @return the name of the node
94            */
95           String name();
96           /**
97            * The predecessors of this node within a sub-topology.
98            * Note, sources do not have any predecessors.
99            * Will never be {@code null}.
100            * @return set of all predecessors
101            */
102           Set<Node> predecessors();
103           /**
104            * The successor of this node within a sub-topology.
105            * Note, sinks do not have any successors.
106            * Will never be {@code null}.
107            * @return set of all successor
108            */
109           Set<Node> successors();
110       }
111   
112   
113       /**
114        * A source node of a topology.
115        */
116       interface Source extends Node {
117   
118           /**
119            * The topic names this source node is reading from.
120            * @return a set of topic names
121            */
122           Set<String> topicSet();
123   
124           /**
125            * The pattern used to match topic names that is reading from.
126            * @return the pattern used to match topic names
127            */
128           Pattern topicPattern();
129       }
130   
131       /**
132        * A processor node of a topology.
133        */
134       interface Processor extends Node {
135           /**
136            * The names of all connected stores.
137            * @return set of store names
138            */
139           Set<String> stores();
140       }
141   
142       /**
143        * A sink node of a topology.
144        */
145       interface Sink extends Node {
146           /**
147            * The topic name this sink node is writing to.
148            * Could be {@code null} if the topic name can only be dynamically determined based on {@link TopicNameExtractor}
149            * @return a topic name
150            */
151           String topic();
152   
153           /**
154            * The {@link TopicNameExtractor} class that this sink node uses to dynamically extract the topic name to write to.
155            * Could be {@code null} if the topic name is not dynamically determined.
156            * @return the {@link TopicNameExtractor} class used get the topic name
157            */
158           TopicNameExtractor topicNameExtractor();
159       }
160   
161       /**
162        * All sub-topologies of the represented topology.
163        * @return set of all sub-topologies
164        */
165       Set<Subtopology> subtopologies();
166   
167       /**
168        * All global stores of the represented topology.
169        * @return set of all global stores
170        */
171       Set<GlobalStore> globalStores();
172   
173   }
