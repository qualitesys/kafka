20   /**
21    * The result key type of a windowed stream aggregation.
22    * <p>
23    * If a {@link KStream} gets grouped and aggregated using a window-aggregation the resulting {@link KTable} is a
24    * so-called "windowed {@link KTable}" with a combined key type that encodes the corresponding aggregation window and
25    * the original record key.
26    * Thus, a windowed {@link KTable} has type {@code <Windowed<K>,V>}.
27    *
28    * @param <K> type of the key
29    * @see KGroupedStream#windowedBy(Windows)
30    * @see KGroupedStream#windowedBy(SessionWindows)
31    */
32   public class Windowed<K> {
33   
34       private final K key;
35   
36       private final Window window;
37   
38       public Windowed(final K key, final Window window) {
39           this.key = key;
40           this.window = window;
41       }
42   
43       /**
44        * Return the key of the window.
45        *
46        * @return the key of the window
47        */
48       public K key() {
49           return key;
50       }
51   
52       /**
53        * Return the window containing the values associated with this key.
54        *
55        * @return the window containing the values
56        */
57       public Window window() {
58           return window;
59       }
60   
61       @Override
62       public String toString() {
63           return "[" + key + "@" + window.start() + "/" + window.end() + "]";
64       }
65   
66       @Override
67       public boolean equals(final Object obj) {
68           if (obj == this) {
69               return true;
70           }
71           if (!(obj instanceof Windowed)) {
72               return false;
73           }
74           final Windowed<?> that = (Windowed) obj;
75           return window.equals(that.window) && key.equals(that.key);
76       }
77   
78       @Override
79       public int hashCode() {
80           final long n = ((long) window.hashCode() << 32) | key.hashCode();
81           return (int) (n % 0xFFFFFFFFL);
82       }
83   }
