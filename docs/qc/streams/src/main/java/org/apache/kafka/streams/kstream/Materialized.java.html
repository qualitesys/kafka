38   /**
39    * Used to describe how a {@link StateStore} should be materialized.
40    * You can either provide a custom {@link StateStore} backend through one of the provided methods accepting a supplier
41    * or use the default RocksDB backends by providing just a store name.
42    * <p>
43    * For example, you can read a topic as {@link KTable} and force a state store materialization to access the content
44    * via Interactive Queries API:
45    * <pre>{@code
46    * StreamsBuilder builder = new StreamsBuilder();
47    * KTable<Integer, Integer> table = builder.table(
48    *   "topicName",
49    *   Materialized.as("queryable-store-name"));
50    * }</pre>
51    *
52    * @param <K> type of record key
53    * @param <V> type of record value
54    * @param <S> type of state store (note: state stores always have key/value types {@code <Bytes,byte[]>}
55    *
56    * @see org.apache.kafka.streams.state.Stores
57    */
58   public class Materialized<K, V, S extends StateStore> {
59       protected StoreSupplier<S> storeSupplier;
60       protected String storeName;

64       protected boolean cachingEnabled = true;
65       protected Map<String, String> topicConfig = new HashMap<>();
66       protected Duration retention;
67       public StoreType storeType;

264       /**
265        * Configure retention period for window and session stores. Ignored for key/value stores.
266        *
267        * Overridden by pre-configured store suppliers
268        * ({@link Materialized#as(SessionBytesStoreSupplier)} or {@link Materialized#as(WindowBytesStoreSupplier)}).
269        *
270        * Note that the retention period must be at least long enough to contain the windowed data's entire life cycle,
271        * from window-start through window-end, and for the entire grace period. If not specified, the retention
272        * period would be set as the window length (from window-start through window-end) plus the grace period.
273        *
274        * @param retention the retention time
275        * @return itself
276        * @throws IllegalArgumentException if retention is negative or can't be represented as {@code long milliseconds}
277        */
278       public Materialized<K, V, S> withRetention(final Duration retention) throws IllegalArgumentException {
279           final String msgPrefix = prepareMillisCheckFailMsgPrefix(retention, "retention");
280           final long retentionMs = validateMillisecondDuration(retention, msgPrefix);
281   
282           if (retentionMs < 0) {
283               throw new IllegalArgumentException("Retention must not be negative.");
284           }
285           this.retention = retention;
286           return this;
287       }
