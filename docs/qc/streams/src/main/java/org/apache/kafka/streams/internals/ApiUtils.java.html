21   import java.time.Duration;
22   import java.time.Instant;
23   import java.util.function.Supplier;
24   
25   import static java.lang.String.format;
26   
27   public final class ApiUtils {
28   
29       private static final String MILLISECOND_VALIDATION_FAIL_MSG_FRMT = "Invalid value for parameter \"%s\" (value was: %s). ";
30       private static final String VALIDATE_MILLISECOND_NULL_SUFFIX = "It shouldn't be null.";
31       private static final String VALIDATE_MILLISECOND_OVERFLOW_SUFFIX = "It can't be converted to milliseconds.";
32   
33       private ApiUtils() {
34       }
35   
36       /**
37        * Validates that milliseconds from {@code duration} can be retrieved.
38        * @param duration Duration to check.
39        * @param messagePrefix Prefix text for an error message.
40        * @return Milliseconds from {@code duration}.
41        */
42       public static long validateMillisecondDuration(final Duration duration, final String messagePrefix) {
43           try {
44               if (duration == null) {
45                   throw new IllegalArgumentException(messagePrefix + VALIDATE_MILLISECOND_NULL_SUFFIX);
46               }
47   
48               return duration.toMillis();
49           } catch (final ArithmeticException e) {
50               throw new IllegalArgumentException(messagePrefix + VALIDATE_MILLISECOND_OVERFLOW_SUFFIX, e);
51           }
52       }
53   
54       /**
55        * Validates that milliseconds from {@code instant} can be retrieved.
56        * @param instant Instant to check.
57        * @param messagePrefix Prefix text for an error message.
58        * @return Milliseconds from {@code instant}.
59        */
60       public static long validateMillisecondInstant(final Instant instant, final String messagePrefix) {
61           try {
62               if (instant == null) {
63                   throw new IllegalArgumentException(messagePrefix + VALIDATE_MILLISECOND_NULL_SUFFIX);
64               }
65   
66               return instant.toEpochMilli();
67           } catch (final ArithmeticException e) {
68               throw new IllegalArgumentException(messagePrefix + VALIDATE_MILLISECOND_OVERFLOW_SUFFIX, e);
69           }
70       }
71   
72       /**
73        * Generates the prefix message for validateMillisecondXXXXXX() utility
74        * @param value Object to be converted to milliseconds
75        * @param name Object name
76        * @return Error message prefix to use in exception
77        */
78       public static String prepareMillisCheckFailMsgPrefix(final Object value, final String name) {
79           return format(MILLISECOND_VALIDATION_FAIL_MSG_FRMT, name, value);
80       }
81   
82       /**
83        * @throws IllegalArgumentException if the same instance is obtained each time
84        */
85       public static void checkSupplier(final Supplier<?> supplier) {
86           if (supplier.get() == supplier.get()) {
87               final String supplierClass = supplier.getClass().getName();
88               throw new IllegalArgumentException(String.format("%s generates single reference." +
89                       " %s#get() must return a new object each time it is called.", supplierClass, supplierClass));
90           }
91       }
92   
93       /**
94        * @throws IllegalArgumentException if the same instance is obtained each time
95        */
96       public static <VR, V> void checkSupplier(final ValueTransformerSupplier<V, VR> supplier) {
97           if (supplier.get() == supplier.get()) {
98               final String supplierClass = supplier.getClass().getName();
99               throw new IllegalArgumentException(String.format("%s generates single reference." +
100                       " %s#get() must return a new object each time it is called.", supplierClass, supplierClass));
101           }
102       }
103   }
