78       /**
79        * Stream thread states are the possible states that a stream thread can be in.
80        * A thread must only be in one state at a time
81        * The expected state transitions with the following defined states is:
82        *
83        * <pre>
84        *                 +-------------+
85        *          +<---- | Created (0) |
86        *          |      +-----+-------+
87        *          |            |
88        *          |            v
89        *          |      +-----+-------+
90        *          +<---- | Starting (1)|----->+
91        *          |      +-----+-------+      |
92        *          |                           |
93        *          |            +<----------+  |
94        *          |            |           |  |
95        *          |            v           |  |
96        *          |      +-----+-------+   |  |
97        *          +<---- | Partitions  | --+  |
98        *          |      | Revoked (2) | <----+
99        *          |      +-----+-------+      |
100        *          |           |  ^            |
101        *          |           v  |            |
102        *          |      +-----+-------+      |
103        *          +<---- | Partitions  |      |
104        *          |      | Assigned (3)| <----+
105        *          |      +-----+-------+      |
106        *          |            |              |
107        *          |            +<----------+  |
108        *          |            |           |  |
109        *          |            v           |  |
110        *          |      +-----+-------+   |  |
111        *          |      |             | --+  |
112        *          |      | Running (4) | ---->+
113        *          |      +-----+-------+
114        *          |            |
115        *          |            v
116        *          |      +-----+-------+
117        *          +----> | Pending     |
118        *                 | Shutdown (5)|
119        *                 +-----+-------+
120        *                       |
121        *                       v
122        *                 +-----+-------+
123        *                 | Dead (6)    |
124        *                 +-------------+
125        * </pre>
126        *
127        * Note the following:
128        * <ul>
129        *     <li>Any state can go to PENDING_SHUTDOWN. That is because streams can be closed at any time.</li>
130        *     <li>
131        *         State PENDING_SHUTDOWN may want to transit to some other states other than DEAD,
132        *         in the corner case when the shutdown is triggered while the thread is still in the rebalance loop.
133        *         In this case we will forbid the transition but will not treat as an error.
134        *     </li>
135        *     <li>
136        *         State PARTITIONS_REVOKED may want transit to itself indefinitely, in the corner case when
137        *         the coordinator repeatedly fails in-between revoking partitions and assigning new partitions.
138        *         Also during streams instance start up PARTITIONS_REVOKED may want to transit to itself as well.
139        *         In this case we will allow the transition but it will be a no-op as the set of revoked partitions
140        *         should be empty.
141        *     </li>
142        * </ul>
143        */
144       public enum State implements ThreadStateTransitionValidator {
145   
146           CREATED(1, 5),                    // 0
147           STARTING(2, 3, 5),                // 1
148           PARTITIONS_REVOKED(2, 3, 5),      // 2
149           PARTITIONS_ASSIGNED(2, 3, 4, 5),  // 3
150           RUNNING(2, 3, 4, 5),              // 4
151           PENDING_SHUTDOWN(6),              // 5
152           DEAD;                             // 6
153   
154           private final Set<Integer> validTransitions = new HashSet<>();
155   
156           State(final Integer... validTransitions) {
157               this.validTransitions.addAll(Arrays.asList(validTransitions));
158           }
159   
160           public boolean isAlive() {
161               return equals(RUNNING) || equals(STARTING) || equals(PARTITIONS_REVOKED) || equals(PARTITIONS_ASSIGNED);
162           }
163   
164           @Override
165           public boolean isValidTransition(final ThreadStateTransitionValidator newState) {
166               final State tmpState = (State) newState;
167               return validTransitions.contains(tmpState.ordinal());
168           }
169       }

206       /**
207        * Sets the state
208        *
209        * @param newState New state
210        * @return The state prior to the call to setState, or null if the transition is invalid
211        */
212       State setState(final State newState) {
213           final State oldState;
214   
215           synchronized (stateLock) {
216               oldState = state;
217   
218               if (state == State.PENDING_SHUTDOWN && newState != State.DEAD) {
219                   log.debug("Ignoring request to transit from PENDING_SHUTDOWN to {}: " +
220                                 "only DEAD state is a valid next state", newState);
221                   // when the state is already in PENDING_SHUTDOWN, all other transitions will be
222                   // refused but we do not throw exception here
223                   return null;
224               } else if (state == State.DEAD) {
225                   log.debug("Ignoring request to transit from DEAD to {}: " +
226                                 "no valid next state after DEAD", newState);
227                   // when the state is already in NOT_RUNNING, all its transitions
228                   // will be refused but we do not throw exception here
229                   return null;
230               } else if (!state.isValidTransition(newState)) {
231                   log.error("Unexpected state transition from {} to {}", oldState, newState);
232                   throw new StreamsException(logPrefix + "Unexpected state transition from " + oldState + " to " + newState);
233               } else {
234                   log.info("State transition from {} to {}", oldState, newState);
235               }
236   
237               state = newState;
238               if (newState == State.RUNNING) {
239                   updateThreadMetadata(taskManager.activeTaskMap(), taskManager.standbyTaskMap());
240               }
241   
242               stateLock.notifyAll();
243           }
244   
245           if (stateListener != null) {
246               stateListener.onChange(this, state, oldState);
247           }
248   
249           return oldState;
250       }
251   
252       public boolean isRunning() {
253           synchronized (stateLock) {
254               return state.isAlive();
255           }

568       boolean runLoop() {

571           // if the thread is still in the middle of a rebalance, we should keep polling
572           // until the rebalance is completed before we close and commit the tasks

601                       if (enforceRebalance && eosEnabled) {
602                           log.info("Active task(s) got corrupted. Triggering a rebalance.");
603                           mainConsumer.enforceRebalance("Active tasks corrupted");

605                   } catch (final TaskMigratedException taskMigrated) {
606                       handleTaskMigrated(taskMigrated);
607                   }
608               } catch (final TaskMigratedException e) {

613                       errorMessage.startsWith("Broker unexpectedly doesn't support requireStable flag on version ")) {
614   
615                       log.error("Shutting down because the Kafka cluster seems to be on a too old version. " +
616                                 "Setting {}=\"{}\"/\"{}\" requires broker version 2.5 or higher.",
617                             StreamsConfig.PROCESSING_GUARANTEE_CONFIG,
618                             StreamsConfig.EXACTLY_ONCE_V2, StreamsConfig.EXACTLY_ONCE_BETA);
619                   }
620                   failedStreamThreadSensor.record();
621                   this.streamsUncaughtExceptionHandler.accept(new StreamsException(e), false);
622                   return false;
623               } catch (final StreamsException e) {
624                   throw e;
625               } catch (final Exception e) {
626                   throw new StreamsException(e);
627               }
628           }

649       public boolean waitOnThreadState(final StreamThread.State targetState, final long timeoutMs) {
650           final long begin = time.milliseconds();
651           synchronized (stateLock) {
652               boolean interrupted = false;
653               long elapsedMs = 0L;
654               try {
655                   while (state != targetState) {
656                       if (timeoutMs >= elapsedMs) {
657                           final long remainingMs = timeoutMs - elapsedMs;
658                           try {
659                               stateLock.wait(remainingMs);
660                           } catch (final InterruptedException e) {
661                               interrupted = true;
662                           }
663                       } else {
664                           log.debug("Cannot transit to {} within {}ms", targetState, timeoutMs);
665                           return false;
666                       }
667                       elapsedMs = time.milliseconds() - begin;
668                   }
669                   return true;
670               } finally {
671                   // Make sure to restore the interruption status before returning.
672                   // We do not always own the current thread that executes this method, i.e., we do not know the
673                   // interruption policy of the thread. The least we can do is restore the interruption status before
674                   // the current thread exits this method.
675                   if (interrupted) {
676                       Thread.currentThread().interrupt();
677                   }
678               }
679           }
680       }

712       /**
713        * One iteration of a thread includes the following steps:
714        *
715        * 1. poll records from main consumer and add to buffer;
716        * 2. restore from restore consumer and update standby tasks if necessary;
717        * 3. process active tasks from the buffers;
718        * 4. punctuate active tasks if necessary;
719        * 5. commit all tasks if necessary;
720        *
721        * Among them, step 3/4/5 is done in batches in which we try to process as much as possible while trying to
722        * stop iteration to call the next iteration when it's close to the next main consumer's poll deadline
723        *
724        * @throws IllegalStateException If store gets registered after initialized is already finished
725        * @throws StreamsException      If the store's change log does not contain the partition
726        * @throws TaskMigratedException If another thread wrote to the changelog topic that is currently restored
727        *                               or if committing offsets failed (non-EOS)
728        *                               or if the task producer got fenced (EOS)
729        */
730       // Visible for testing
731       void runOnce() {
732           final long startMs = time.milliseconds();
733           now = startMs;
734   
735           final long pollLatency = pollPhase();
736   
737           // Shutdown hook could potentially be triggered and transit the thread state to PENDING_SHUTDOWN during #pollRequests().
738           // The task manager internal states could be uninitialized if the state transition happens during #onPartitionsAssigned().
739           // Should only proceed when the thread is still running after #pollRequests(), because no external state mutation
740           // could affect the task manager state beyond this point within #runOnce().
741           if (!isRunning()) {
742               log.debug("Thread state is already {}, skipping the run once call after poll request", state);
743               return;
744           }
745   
746           initializeAndRestorePhase();
747   
748           // TODO: we should record the restore latency and its relative time spent ratio after
749           //       we figure out how to move this method out of the stream thread
750           advanceNowAndComputeLatency();
751   
752           int totalProcessed = 0;
753           long totalCommitLatency = 0L;
754           long totalProcessLatency = 0L;
755           long totalPunctuateLatency = 0L;
756           if (state == State.RUNNING) {
757               /*
758                * Within an iteration, after processing up to N (N initialized as 1 upon start up) records for each applicable tasks, check the current time:
759                *  1. If it is time to punctuate, do it;
760                *  2. If it is time to commit, do it, this should be after 1) since punctuate may trigger commit;
761                *  3. If there's no records processed, end the current iteration immediately;
762                *  4. If we are close to consumer's next poll deadline, end the current iteration immediately;
763                *  5. If any of 1), 2) and 4) happens, half N for next iteration;
764                *  6. Otherwise, increment N.
765                */
766               do {
767                   log.debug("Processing tasks with {} iterations.", numIterations);
768                   final int processed = taskManager.process(numIterations, time);
769                   final long processLatency = advanceNowAndComputeLatency();
770                   totalProcessLatency += processLatency;
771                   if (processed > 0) {
772                       // It makes no difference to the outcome of these metrics when we record "0",
773                       // so we can just avoid the method call when we didn't process anything.
774                       processRateSensor.record(processed, now);
775   
776                       // This metric is scaled to represent the _average_ processing time of _each_
777                       // task. Note, it's hard to interpret this as defined; the per-task process-ratio
778                       // as well as total time ratio spent on processing compared with polling / committing etc
779                       // are reported on other metrics.
780                       processLatencySensor.record(processLatency / (double) processed, now);
781   
782                       totalProcessed += processed;
783                       totalRecordsProcessedSinceLastSummary += processed;
784                   }
785   
786                   log.debug("Processed {} records with {} iterations; invoking punctuators if necessary",
787                             processed,
788                             numIterations);
789   
790                   final int punctuated = taskManager.punctuate();
791                   totalPunctuatorsSinceLastSummary += punctuated;
792                   final long punctuateLatency = advanceNowAndComputeLatency();
793                   totalPunctuateLatency += punctuateLatency;
794                   if (punctuated > 0) {
795                       punctuateSensor.record(punctuateLatency / (double) punctuated, now);
796                   }
797   
798                   log.debug("{} punctuators ran.", punctuated);
799   
800                   final long beforeCommitMs = now;
801                   final int committed = maybeCommit();
802                   final long commitLatency = Math.max(now - beforeCommitMs, 0);
803                   totalCommitLatency += commitLatency;
804                   if (committed > 0) {
805                       totalCommittedSinceLastSummary += committed;
806                       commitSensor.record(commitLatency / (double) committed, now);
807   
808                       if (log.isDebugEnabled()) {
809                           log.debug("Committed all active tasks {} and standby tasks {} in {}ms",
810                               taskManager.activeTaskIds(), taskManager.standbyTaskIds(), commitLatency);
811                       }
812                   }
813   
814                   if (processed == 0) {
815                       // if there are no records to be processed, exit after punctuate / commit
816                       break;
817                   } else if (Math.max(now - lastPollMs, 0) > maxPollTimeMs / 2) {
818                       numIterations = numIterations > 1 ? numIterations / 2 : numIterations;
819                       break;
820                   } else if (punctuated > 0 || committed > 0) {
821                       numIterations = numIterations > 1 ? numIterations / 2 : numIterations;
822                   } else {
823                       numIterations++;
824                   }
825               } while (true);
826   
827               // we record the ratio out of the while loop so that the accumulated latency spans over
828               // multiple iterations with reasonably large max.num.records and hence is less vulnerable to outliers
829               taskManager.recordTaskProcessRatio(totalProcessLatency, now);
830           }
831   
832           now = time.milliseconds();
833           final long runOnceLatency = now - startMs;
834           processRecordsSensor.record(totalProcessed, now);
835           processRatioSensor.record((double) totalProcessLatency / runOnceLatency, now);
836           punctuateRatioSensor.record((double) totalPunctuateLatency / runOnceLatency, now);
837           pollRatioSensor.record((double) pollLatency / runOnceLatency, now);
838           commitRatioSensor.record((double) totalCommitLatency / runOnceLatency, now);
839   
840           final boolean logProcessingSummary = now - lastLogSummaryMs > LOG_SUMMARY_INTERVAL_MS;
841           if (logProcessingSummary) {
842               log.info("Processed {} total records, ran {} punctuators, and committed {} total tasks since the last update",
843                    totalRecordsProcessedSinceLastSummary, totalPunctuatorsSinceLastSummary, totalCommittedSinceLastSummary);
844   
845               totalRecordsProcessedSinceLastSummary = 0L;
846               totalPunctuatorsSinceLastSummary = 0L;
847               totalCommittedSinceLastSummary = 0L;
848               lastLogSummaryMs = now;
849           }
850       }

852       private void initializeAndRestorePhase() {
853           // only try to initialize the assigned tasks
854           // if the state is still in PARTITION_ASSIGNED after the poll call
855           final State stateSnapshot = state;
856           if (stateSnapshot == State.PARTITIONS_ASSIGNED
857               || stateSnapshot == State.RUNNING && taskManager.needsInitializationOrRestoration()) {
858   
859               log.debug("State is {}; initializing tasks if necessary", stateSnapshot);
860   
861               // transit to restore active is idempotent so we can call it multiple times
862               changelogReader.enforceRestoreActive();

902       private long pollPhase() {
903           final ConsumerRecords<byte[], byte[]> records;
904           log.debug("Invoking poll on main Consumer");
905   
906           if (state == State.PARTITIONS_ASSIGNED) {
907               // try to fetch some records with zero poll millis
908               // to unblock the restoration as soon as possible
909               records = pollRequests(Duration.ZERO);
910           } else if (state == State.PARTITIONS_REVOKED) {
911               // try to fetch some records with zero poll millis to unblock
912               // other useful work while waiting for the join response
913               records = pollRequests(Duration.ZERO);
914           } else if (state == State.RUNNING || state == State.STARTING) {
915               // try to fetch some records with normal poll time
916               // in order to get long polling
917               records = pollRequests(pollTime);
918           } else if (state == State.PENDING_SHUTDOWN) {
919               // we are only here because there's rebalance in progress,
920               // just poll with zero to complete it
921               records = pollRequests(Duration.ZERO);
922           } else {
923               // any other state should not happen
924               log.error("Unexpected state {} during normal iteration", state);
925               throw new StreamsException(logPrefix + "Unexpected state " + state + " during normal iteration");
926           }
927   
928           final long pollLatency = advanceNowAndComputeLatency();
929   
930           final int numRecords = records.count();
931   
932           for (final TopicPartition topicPartition: records.partitions()) {
933               records
934                   .records(topicPartition)
935                   .stream()
936                   .max(Comparator.comparing(ConsumerRecord::offset))
937                   .ifPresent(t -> taskManager.updateTaskEndMetadata(topicPartition, t.offset()));
938           }
939   
940           log.debug("Main Consumer poll completed in {} ms and fetched {} records from partitions {}",
941               pollLatency, numRecords, records.partitions());
942   
943           pollSensor.record(pollLatency, now);
944   
945           if (!records.isEmpty()) {
946               pollRecordsSensor.record(numRecords, now);
947               taskManager.addRecordsToTasks(records);
948           }
949   
950           while (!nonFatalExceptionsToHandle.isEmpty()) {
951               streamsUncaughtExceptionHandler.accept(nonFatalExceptionsToHandle.poll(), true);
952           }
953           return pollLatency;
954       }

958        *
959        * @param pollTime how long to block in Consumer#poll
960        * @return Next batch of records or null if no records available.
961        * @throws TaskMigratedException if the task producer got fenced (EOS only)

977       private void resetOffsets(final Set<TopicPartition> partitions, final Exception cause) {
978           final Set<String> loggedTopics = new HashSet<>();
979           final Set<TopicPartition> seekToBeginning = new HashSet<>();
980           final Set<TopicPartition> seekToEnd = new HashSet<>();
981           final Set<TopicPartition> notReset = new HashSet<>();
982   
983           for (final TopicPartition partition : partitions) {
984               final OffsetResetStrategy offsetResetStrategy = topologyMetadata.offsetResetStrategy(partition.topic());
985   
986               // This may be null if the task we are currently processing was apart of a named topology that was just removed.
987               // TODO KAFKA-13713: keep the StreamThreads and TopologyMetadata view of named topologies in sync until final thread has acked
988               if (offsetResetStrategy != null) {
989                   switch (offsetResetStrategy) {
990                       case EARLIEST:
991                           addToResetList(partition, seekToBeginning, "Setting topic '{}' to consume from {} offset", "earliest", loggedTopics);
992                           break;
993                       case LATEST:
994                           addToResetList(partition, seekToEnd, "Setting topic '{}' to consume from {} offset", "latest", loggedTopics);
995                           break;
996                       case NONE:
997                           if ("earliest".equals(originalReset)) {
998                               addToResetList(partition, seekToBeginning, "No custom setting defined for topic '{}' using original config '{}' for offset reset", "earliest", loggedTopics);
999                           } else if ("latest".equals(originalReset)) {
1000                               addToResetList(partition, seekToEnd, "No custom setting defined for topic '{}' using original config '{}' for offset reset", "latest", loggedTopics);
1001                           } else {
1002                               notReset.add(partition);
1003                           }
1004                           break;
1005                       default:
1006                           throw new IllegalStateException("Unable to locate topic " + partition.topic() + " in the topology");
1007                   }
1008               }
1009           }
1010   
1011           if (notReset.isEmpty()) {
1012               if (!seekToBeginning.isEmpty()) {
1013                   mainConsumer.seekToBeginning(seekToBeginning);
1014               }
1015   
1016               if (!seekToEnd.isEmpty()) {
1017                   mainConsumer.seekToEnd(seekToEnd);
1018               }
1019           } else {
1020               final String notResetString =
1021                   notReset.stream()
1022                           .map(TopicPartition::topic)
1023                           .distinct()
1024                           .collect(Collectors.joining(","));
1025   
1026               final String format = String.format(
1027                   "No valid committed offset found for input [%s] and no valid reset policy configured." +
1028                       " You need to set configuration parameter \"auto.offset.reset\" or specify a topic specific reset " +
1029                       "policy via StreamsBuilder#stream(..., Consumed.with(Topology.AutoOffsetReset)) or " +
1030                       "StreamsBuilder#table(..., Consumed.with(Topology.AutoOffsetReset))",
1031                   notResetString
1032               );
1033   
1034               if (cause == null) {
1035                   throw new StreamsException(format);
1036               } else {
1037                   throw new StreamsException(format, cause);
1038               }
1039           }
1040       }
