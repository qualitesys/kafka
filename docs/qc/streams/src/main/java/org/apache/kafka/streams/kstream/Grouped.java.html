17   package org.apache.kafka.streams.kstream;
18   
19   import org.apache.kafka.common.serialization.Serde;
20   
21   /**
22    * The class that is used to capture the key and value {@link Serde}s and set the part of name used for
23    * repartition topics when performing {@link KStream#groupBy(KeyValueMapper, Grouped)}, {@link
24    * KStream#groupByKey(Grouped)}, or {@link KTable#groupBy(KeyValueMapper, Grouped)} operations.  Note
25    * that Kafka Streams does not always create repartition topics for grouping operations.
26    *
27    * @param <K> the key type
28    * @param <V> the value type
29    */
30   public class Grouped<K, V> implements NamedOperation<Grouped<K, V>> {
31   
32       protected final Serde<K> keySerde;
33       protected final Serde<V> valueSerde;
34       protected final String name;
35   
36       private Grouped(final String name,
37                       final Serde<K> keySerde,
38                       final Serde<V> valueSerde) {
39           this.name = name;
40           this.keySerde = keySerde;
41           this.valueSerde = valueSerde;
42       }
43   
44       protected Grouped(final Grouped<K, V> grouped) {
45           this(grouped.name, grouped.keySerde, grouped.valueSerde);
46       }
47   
48       /**
49        * Create a {@link Grouped} instance with the provided name used as part of the repartition topic if required.
50        *
51        * @param name the name used for a repartition topic if required
52        * @return a new {@link Grouped} configured with the name
53        * @see KStream#groupByKey(Grouped)
54        * @see KStream#groupBy(KeyValueMapper, Grouped)
55        * @see KTable#groupBy(KeyValueMapper, Grouped)
56        */
57       public static <K, V> Grouped<K, V> as(final String name) {
58           return new Grouped<>(name, null, null);
59       }
60   
61   
62       /**
63        * Create a {@link Grouped} instance with the provided keySerde. If {@code null} the default key serde from config will be used.
64        *
65        * @param keySerde the Serde used for serializing the key. If {@code null} the default key serde from config will be used
66        * @return a new {@link Grouped} configured with the keySerde
67        * @see KStream#groupByKey(Grouped)
68        * @see KStream#groupBy(KeyValueMapper, Grouped)
69        * @see KTable#groupBy(KeyValueMapper, Grouped)
70        */
71       public static <K, V> Grouped<K, V> keySerde(final Serde<K> keySerde) {
72           return new Grouped<>(null, keySerde, null);
73       }
74   
75   
76       /**
77        * Create a {@link Grouped} instance with the provided valueSerde.  If {@code null} the default value serde from config will be used.
78        *
79        * @param valueSerde the {@link Serde} used for serializing the value. If {@code null} the default value serde from config will be used
80        * @return a new {@link Grouped} configured with the valueSerde
81        * @see KStream#groupByKey(Grouped)
82        * @see KStream#groupBy(KeyValueMapper, Grouped)
83        * @see KTable#groupBy(KeyValueMapper, Grouped)
84        */
85       public static <K, V> Grouped<K, V> valueSerde(final Serde<V> valueSerde) {
86           return new Grouped<>(null, null, valueSerde);
87       }
88   
89       /**
90        * Create a {@link Grouped} instance with the provided  name, keySerde, and valueSerde. If the keySerde and/or the valueSerde is
91        * {@code null} the default value for the respective serde from config will be used.
92        *
93        * @param name       the name used as part of the repartition topic name if required
94        * @param keySerde   the {@link Serde} used for serializing the key. If {@code null} the default key serde from config will be used
95        * @param valueSerde the {@link Serde} used for serializing the value. If {@code null} the default value serde from config will be used
96        * @return a new {@link Grouped} configured with the name, keySerde, and valueSerde
97        * @see KStream#groupByKey(Grouped)
98        * @see KStream#groupBy(KeyValueMapper, Grouped)
99        * @see KTable#groupBy(KeyValueMapper, Grouped)
100        */
101       public static <K, V> Grouped<K, V> with(final String name,
102                                               final Serde<K> keySerde,
103                                               final Serde<V> valueSerde) {
104           return new Grouped<>(name, keySerde, valueSerde);
105       }
106   
107   
108       /**
109        * Create a {@link Grouped} instance with the provided keySerde and valueSerde.  If the keySerde and/or the valueSerde is
110        * {@code null} the default value for the respective serde from config will be used.
111        *
112        * @param keySerde   the {@link Serde} used for serializing the key. If {@code null} the default key serde from config will be used
113        * @param valueSerde the {@link Serde} used for serializing the value. If {@code null} the default value serde from config will be used
114        * @return a new {@link Grouped} configured with the keySerde, and valueSerde
115        * @see KStream#groupByKey(Grouped)
116        * @see KStream#groupBy(KeyValueMapper, Grouped)
117        * @see KTable#groupBy(KeyValueMapper, Grouped)
118        */
119       public static <K, V> Grouped<K, V> with(final Serde<K> keySerde,
120                                               final Serde<V> valueSerde) {
121           return new Grouped<>(null, keySerde, valueSerde);
122       }
123   
124       /**
125        * Perform the grouping operation with the name for a repartition topic if required.  Note
126        * that Kafka Streams does not always create repartition topics for grouping operations.
127        *
128        * @param name the name used for the processor name and as part of the repartition topic name if required
129        * @return a new {@link Grouped} instance configured with the name
130        * */
131       @Override
132       public Grouped<K, V> withName(final String name) {
133           return new Grouped<>(name, keySerde, valueSerde);
134       }
135   
136       /**
137        * Perform the grouping operation using the provided keySerde for serializing the key.
138        *
139        * @param keySerde {@link Serde} to use for serializing the key. If {@code null} the default key serde from config will be used
140        * @return a new {@link Grouped} instance configured with the keySerde
141        */
142       public Grouped<K, V> withKeySerde(final Serde<K> keySerde) {
143           return new Grouped<>(name, keySerde, valueSerde);
144       }
145   
146       /**
147        * Perform the grouping operation using the provided valueSerde for serializing the value.
148        *
149        * @param valueSerde {@link Serde} to use for serializing the value. If {@code null} the default value serde from config will be used
150        * @return a new {@link Grouped} instance configured with the valueSerde
151        */
152       public Grouped<K, V> withValueSerde(final Serde<V> valueSerde) {
153           return new Grouped<>(name, keySerde, valueSerde);
154       }
155   
156   }
