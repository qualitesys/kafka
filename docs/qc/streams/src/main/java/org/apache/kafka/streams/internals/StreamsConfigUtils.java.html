23   import static org.apache.kafka.streams.StreamsConfig.CACHE_MAX_BYTES_BUFFERING_CONFIG;
24   import static org.apache.kafka.streams.StreamsConfig.STATESTORE_CACHE_MAX_BYTES_CONFIG;

26   public class StreamsConfigUtils {
27   
28       private static final Logger LOG = LoggerFactory.getLogger(StreamsConfigUtils.class);
29   
30       public enum ProcessingMode {
31           AT_LEAST_ONCE("AT_LEAST_ONCE"),
32   
33           EXACTLY_ONCE_ALPHA("EXACTLY_ONCE_ALPHA"),
34   
35           EXACTLY_ONCE_V2("EXACTLY_ONCE_V2");
36   
37           public final String name;
38   
39           ProcessingMode(final String name) {
40               this.name = name;
41           }
42       }
43       
44       @SuppressWarnings("deprecation")
45       public static ProcessingMode processingMode(final StreamsConfig config) {
46           if (StreamsConfig.EXACTLY_ONCE.equals(config.getString(StreamsConfig.PROCESSING_GUARANTEE_CONFIG))) {
47               return ProcessingMode.EXACTLY_ONCE_ALPHA;
48           } else if (StreamsConfig.EXACTLY_ONCE_BETA.equals(config.getString(StreamsConfig.PROCESSING_GUARANTEE_CONFIG))) {
49               return ProcessingMode.EXACTLY_ONCE_V2;
50           } else if (StreamsConfig.EXACTLY_ONCE_V2.equals(config.getString(StreamsConfig.PROCESSING_GUARANTEE_CONFIG))) {
51               return ProcessingMode.EXACTLY_ONCE_V2;
52           } else {
53               return ProcessingMode.AT_LEAST_ONCE;
54           }
55       }
56   
57       @SuppressWarnings("deprecation")
58       public static String processingModeString(final ProcessingMode processingMode) {
59           if (processingMode == ProcessingMode.EXACTLY_ONCE_V2) {
60               return StreamsConfig.EXACTLY_ONCE_V2;
61           } else if (processingMode == ProcessingMode.EXACTLY_ONCE_ALPHA) {
62               return StreamsConfig.EXACTLY_ONCE;
63           } else {
64               return StreamsConfig.AT_LEAST_ONCE;
65           }
66       }
67   
68       public static boolean eosEnabled(final StreamsConfig config) {
69           return eosEnabled(processingMode(config));
70       }
71   
72       public static boolean eosEnabled(final ProcessingMode processingMode) {
73           return processingMode == ProcessingMode.EXACTLY_ONCE_ALPHA ||
74               processingMode == ProcessingMode.EXACTLY_ONCE_V2;
75       }
76   
77       @SuppressWarnings("deprecation")
78       public static long getTotalCacheSize(final StreamsConfig config) {
79           // both deprecated and new config set. Warn and use the new one.
80           if (config.originals().containsKey(CACHE_MAX_BYTES_BUFFERING_CONFIG) && config.originals().containsKey(STATESTORE_CACHE_MAX_BYTES_CONFIG)) {
81               if (!config.getLong(CACHE_MAX_BYTES_BUFFERING_CONFIG).equals(config.getLong(STATESTORE_CACHE_MAX_BYTES_CONFIG))) {
82                   LOG.warn("Both deprecated config {} and the new config {} are set, hence {} is ignored and {} is used instead.",
83                            CACHE_MAX_BYTES_BUFFERING_CONFIG,
84                            STATESTORE_CACHE_MAX_BYTES_CONFIG,
85                            CACHE_MAX_BYTES_BUFFERING_CONFIG,
86                            STATESTORE_CACHE_MAX_BYTES_CONFIG);
87               }
88               return config.getLong(STATESTORE_CACHE_MAX_BYTES_CONFIG);
89           } else if (config.originals().containsKey(CACHE_MAX_BYTES_BUFFERING_CONFIG)) {
90               // only deprecated config set.
91               LOG.warn("Deprecated config {} is set, and will be used; we suggest setting the new config {} instead as deprecated {} would be removed in the future.",
92                        CACHE_MAX_BYTES_BUFFERING_CONFIG,
93                        STATESTORE_CACHE_MAX_BYTES_CONFIG,
94                        CACHE_MAX_BYTES_BUFFERING_CONFIG);
95               return config.getLong(CACHE_MAX_BYTES_BUFFERING_CONFIG);
96           }
97           // only new or no config set. Use default or user specified value.
98           return config.getLong(STATESTORE_CACHE_MAX_BYTES_CONFIG);
99       }
