44   public class HighAvailabilityTaskAssignor implements TaskAssignor {
45       private static final Logger log = LoggerFactory.getLogger(HighAvailabilityTaskAssignor.class);
46   
47       @Override
48       public boolean assign(final Map<UUID, ClientState> clients,
49                             final Set<TaskId> allTaskIds,
50                             final Set<TaskId> statefulTaskIds,
51                             final AssignmentConfigs configs) {
52           final SortedSet<TaskId> statefulTasks = new TreeSet<>(statefulTaskIds);
53           final TreeMap<UUID, ClientState> clientStates = new TreeMap<>(clients);
54   
55           assignActiveStatefulTasks(clientStates, statefulTasks);
56   
57           assignStandbyReplicaTasks(
58               clientStates,
59               allTaskIds,
60               statefulTasks,
61               configs
62           );
63   
64           final AtomicInteger remainingWarmupReplicas = new AtomicInteger(configs.maxWarmupReplicas);
65   
66           final Map<TaskId, SortedSet<UUID>> tasksToCaughtUpClients = tasksToCaughtUpClients(
67               statefulTasks,
68               clientStates,
69               configs.acceptableRecoveryLag
70           );
71   
72           final Map<TaskId, SortedSet<UUID>> tasksToClientByLag = tasksToClientByLag(statefulTasks, clientStates);
73   
74           // We temporarily need to know which standby tasks were intended as warmups
75           // for active tasks, so that we don't move them (again) when we plan standby
76           // task movements. We can then immediately treat warmups exactly the same as
77           // hot-standby replicas, so we just track it right here as metadata, rather
78           // than add "warmup" assignments to ClientState, for example.
79           final Map<UUID, Set<TaskId>> warmups = new TreeMap<>();
80   
81           final int neededActiveTaskMovements = assignActiveTaskMovements(
82               tasksToCaughtUpClients,
83               tasksToClientByLag,
84               clientStates,
85               warmups,
86               remainingWarmupReplicas
87           );
88   
89           final int neededStandbyTaskMovements = assignStandbyTaskMovements(
90               tasksToCaughtUpClients,
91               tasksToClientByLag,
92               clientStates,
93               remainingWarmupReplicas,
94               warmups
95           );
96   
97           assignStatelessActiveTasks(clientStates, diff(TreeSet::new, allTaskIds, statefulTasks));
98   
99           final boolean probingRebalanceNeeded = neededActiveTaskMovements + neededStandbyTaskMovements > 0;
100   
101           log.info("Decided on assignment: " +
102                    clientStates +
103                    " with" +
104                    (probingRebalanceNeeded ? "" : " no") +
105                    " followup probing rebalance.");
106   
107           return probingRebalanceNeeded;
108       }

110       private static void assignActiveStatefulTasks(final SortedMap<UUID, ClientState> clientStates,
111                                                     final SortedSet<TaskId> statefulTasks) {
112           Iterator<ClientState> clientStateIterator = null;
113           for (final TaskId task : statefulTasks) {
114               if (clientStateIterator == null || !clientStateIterator.hasNext()) {
115                   clientStateIterator = clientStates.values().iterator();
116               }
117               clientStateIterator.next().assignActive(task);
118           }
119   
120           balanceTasksOverThreads(
121               clientStates,
122               ClientState::activeTasks,
123               ClientState::unassignActive,
124               ClientState::assignActive,
125               (source, destination) -> true
126           );
127       }

129       private void assignStandbyReplicaTasks(final TreeMap<UUID, ClientState> clientStates,
130                                              final Set<TaskId> allTaskIds,
131                                              final Set<TaskId> statefulTasks,
132                                              final AssignmentConfigs configs) {
133           if (configs.numStandbyReplicas == 0) {
134               return;
135           }
136   
137           final StandbyTaskAssignor standbyTaskAssignor = StandbyTaskAssignorFactory.create(configs);
138   
139           standbyTaskAssignor.assign(clientStates, allTaskIds, statefulTasks, configs);
140   
141           balanceTasksOverThreads(
142               clientStates,
143               ClientState::standbyTasks,
144               ClientState::unassignStandby,
145               ClientState::assignStandby,
146               standbyTaskAssignor::isAllowedTaskMovement
147           );
148       }

150       private static void balanceTasksOverThreads(final SortedMap<UUID, ClientState> clientStates,
151                                                   final Function<ClientState, Set<TaskId>> currentAssignmentAccessor,
152                                                   final BiConsumer<ClientState, TaskId> taskUnassignor,
153                                                   final BiConsumer<ClientState, TaskId> taskAssignor,
154                                                   final BiPredicate<ClientState, ClientState> taskMovementAttemptPredicate) {
155           boolean keepBalancing = true;
156           while (keepBalancing) {
157               keepBalancing = false;
158               for (final Map.Entry<UUID, ClientState> sourceEntry : clientStates.entrySet()) {
159                   final UUID sourceClient = sourceEntry.getKey();
160                   final ClientState sourceClientState = sourceEntry.getValue();
161   
162                   for (final Map.Entry<UUID, ClientState> destinationEntry : clientStates.entrySet()) {
163                       final UUID destinationClient = destinationEntry.getKey();
164                       final ClientState destinationClientState = destinationEntry.getValue();
165                       if (sourceClient.equals(destinationClient)) {
166                           continue;
167                       }
168   
169                       final Set<TaskId> sourceTasks = new TreeSet<>(currentAssignmentAccessor.apply(sourceClientState));
170                       final Iterator<TaskId> sourceIterator = sourceTasks.iterator();
171                       while (shouldMoveATask(sourceClientState, destinationClientState) && sourceIterator.hasNext()) {
172                           final TaskId taskToMove = sourceIterator.next();
173                           final boolean canMove = !destinationClientState.hasAssignedTask(taskToMove)
174                                                   // When ClientTagAwareStandbyTaskAssignor is used, we need to make sure that
175                                                   // sourceClient tags matches destinationClient tags.
176                                                   && taskMovementAttemptPredicate.test(sourceClientState, destinationClientState);
177                           if (canMove) {
178                               taskUnassignor.accept(sourceClientState, taskToMove);
179                               taskAssignor.accept(destinationClientState, taskToMove);
180                               keepBalancing = true;
181                           }
182                       }
183                   }
184               }
185           }
186       }

188       private static boolean shouldMoveATask(final ClientState sourceClientState,
189                                              final ClientState destinationClientState) {
190           final double skew = sourceClientState.assignedTaskLoad() - destinationClientState.assignedTaskLoad();
191   
192           if (skew <= 0) {
193               return false;
194           }
195   
196           final double proposedAssignedTasksPerStreamThreadAtDestination =
197               (destinationClientState.assignedTaskCount() + 1.0) / destinationClientState.capacity();
198           final double proposedAssignedTasksPerStreamThreadAtSource =
199               (sourceClientState.assignedTaskCount() - 1.0) / sourceClientState.capacity();
200           final double proposedSkew = proposedAssignedTasksPerStreamThreadAtSource - proposedAssignedTasksPerStreamThreadAtDestination;
201   
202           if (proposedSkew < 0) {
203               // then the move would only create an imbalance in the other direction.
204               return false;
205           }
206           // we should only move a task if doing so would actually improve the skew.
207           return proposedSkew < skew;
208       }

210       private static void assignStatelessActiveTasks(final TreeMap<UUID, ClientState> clientStates,
211                                                      final Iterable<TaskId> statelessTasks) {
212           final ConstrainedPrioritySet statelessActiveTaskClientsByTaskLoad = new ConstrainedPrioritySet(
213               (client, task) -> true,
214               client -> clientStates.get(client).activeTaskLoad()
215           );
216           statelessActiveTaskClientsByTaskLoad.offerAll(clientStates.keySet());
217   
218           for (final TaskId task : statelessTasks) {
219               final UUID client = statelessActiveTaskClientsByTaskLoad.poll(task);
220               final ClientState state = clientStates.get(client);
221               state.assignActive(task);
222               statelessActiveTaskClientsByTaskLoad.offer(client);
223           }
224       }

226       private static Map<TaskId, SortedSet<UUID>> tasksToCaughtUpClients(final Set<TaskId> statefulTasks,
227                                                                          final Map<UUID, ClientState> clientStates,
228                                                                          final long acceptableRecoveryLag) {
229           final Map<TaskId, SortedSet<UUID>> taskToCaughtUpClients = new HashMap<>();
230   
231           for (final TaskId task : statefulTasks) {
232               final TreeSet<UUID> caughtUpClients = new TreeSet<>();
233               for (final Map.Entry<UUID, ClientState> clientEntry : clientStates.entrySet()) {
234                   final UUID client = clientEntry.getKey();
235                   final long taskLag = clientEntry.getValue().lagFor(task);
236                   if (activeRunning(taskLag) || unbounded(acceptableRecoveryLag) || acceptable(acceptableRecoveryLag, taskLag)) {
237                       caughtUpClients.add(client);
238                   }
239               }
240               taskToCaughtUpClients.put(task, caughtUpClients);
241           }
242   
243           return taskToCaughtUpClients;
244       }
245   
246       private static Map<TaskId, SortedSet<UUID>> tasksToClientByLag(final Set<TaskId> statefulTasks,
247                                                                 final Map<UUID, ClientState> clientStates) {
248           final Map<TaskId, SortedSet<UUID>> tasksToClientByLag = new HashMap<>();
249           for (final TaskId task : statefulTasks) {
250               final SortedSet<UUID> clientLag = new TreeSet<>(Comparator.<UUID>comparingLong(a ->
251                       clientStates.get(a).lagFor(task)).thenComparing(a -> a));
252               clientLag.addAll(clientStates.keySet());
253               tasksToClientByLag.put(task, clientLag);
254           }
255           return tasksToClientByLag;
256       }
257   
258       private static boolean unbounded(final long acceptableRecoveryLag) {
259           return acceptableRecoveryLag == Long.MAX_VALUE;
260       }
261   
262       private static boolean acceptable(final long acceptableRecoveryLag, final long taskLag) {
263           return taskLag >= 0 && taskLag <= acceptableRecoveryLag;
264       }
265   
266       private static boolean activeRunning(final long taskLag) {
267           return taskLag == Task.LATEST_OFFSET;
268       }
