133       /**
134        * Validates the internal topics passed.
135        *
136        * The validation of the internal topics verifies if the topics:
137        * - are missing on the brokers
138        * - have the expected number of partitions
139        * - have configured a clean-up policy that avoids data loss
140        *
141        * @param topicConfigs internal topics to validate
142        *
143        * @return validation results that contains
144        *         - the set of missing internal topics on the brokers
145        *         - descriptions of misconfigurations per topic
146        */
147       public ValidationResult validate(final Map<String, InternalTopicConfig> topicConfigs) {
148           log.info("Starting to validate internal topics {}.", topicConfigs.keySet());
149   
150           final long now = time.milliseconds();
151           final long deadline = now + retryTimeoutMs;
152   
153           final ValidationResult validationResult = new ValidationResult();
154           final Set<String> topicDescriptionsStillToValidate = new HashSet<>(topicConfigs.keySet());
155           final Set<String> topicConfigsStillToValidate = new HashSet<>(topicConfigs.keySet());
156           while (!topicDescriptionsStillToValidate.isEmpty() || !topicConfigsStillToValidate.isEmpty()) {
157               Map<String, KafkaFuture<TopicDescription>> descriptionsForTopic = Collections.emptyMap();
158               if (!topicDescriptionsStillToValidate.isEmpty()) {
159                   final DescribeTopicsResult describeTopicsResult = adminClient.describeTopics(topicDescriptionsStillToValidate);
160                   descriptionsForTopic = describeTopicsResult.topicNameValues();
161               }
162               Map<String, KafkaFuture<Config>> configsForTopic = Collections.emptyMap();
163               if (!topicConfigsStillToValidate.isEmpty()) {
164                   final DescribeConfigsResult describeConfigsResult = adminClient.describeConfigs(
165                       topicConfigsStillToValidate.stream()
166                           .map(topic -> new ConfigResource(Type.TOPIC, topic))
167                           .collect(Collectors.toSet())
168                   );
169                   configsForTopic = describeConfigsResult.values().entrySet().stream()
170                       .collect(Collectors.toMap(entry -> entry.getKey().name(), Map.Entry::getValue));
171               }
172   
173               while (!descriptionsForTopic.isEmpty() || !configsForTopic.isEmpty()) {
174                   if (!descriptionsForTopic.isEmpty()) {
175                       doValidateTopic(
176                           validationResult,
177                           descriptionsForTopic,
178                           topicConfigs,
179                           topicDescriptionsStillToValidate,
180                           (streamsSide, brokerSide) -> validatePartitionCount(validationResult, streamsSide, brokerSide)
181                       );
182                   }
183                   if (!configsForTopic.isEmpty()) {
184                       doValidateTopic(
185                           validationResult,
186                           configsForTopic,
187                           topicConfigs,
188                           topicConfigsStillToValidate,
189                           (streamsSide, brokerSide) -> validateCleanupPolicy(validationResult, streamsSide, brokerSide)
190                       );
191                   }
192   
193                   maybeThrowTimeoutException(
194                       Arrays.asList(topicDescriptionsStillToValidate, topicConfigsStillToValidate),
195                       deadline,
196                       String.format("Could not validate internal topics within %d milliseconds. " +
197                           "This can happen if the Kafka cluster is temporarily not available.", retryTimeoutMs)
198                   );
199   
200                   if (!descriptionsForTopic.isEmpty() || !configsForTopic.isEmpty()) {
201                       Utils.sleep(100);
202                   }
203               }
204   
205               maybeSleep(
206                   Arrays.asList(topicDescriptionsStillToValidate, topicConfigsStillToValidate),
207                   deadline,
208                   "validated"
209               );
210           }
211   
212           log.info("Completed validation of internal topics {}.", topicConfigs.keySet());
213           return validationResult;
214       }

216       private <V> void doValidateTopic(final ValidationResult validationResult,
217                                        final Map<String, KafkaFuture<V>> futuresForTopic,
218                                        final Map<String, InternalTopicConfig> topicsConfigs,
219                                        final Set<String> topicsStillToValidate,
220                                        final BiConsumer<InternalTopicConfig, V> validator) {
221           for (final String topicName : new HashSet<>(topicsStillToValidate)) {
222               if (!futuresForTopic.containsKey(topicName)) {
223                   throw new IllegalStateException("Description results do not contain topics to validate. " + BUG_ERROR_MESSAGE);
224               }
225               final KafkaFuture<V> future = futuresForTopic.get(topicName);
226               if (future.isDone()) {
227                   try {
228                       final V brokerSideTopicConfig = future.get();
229                       final InternalTopicConfig streamsSideTopicConfig = topicsConfigs.get(topicName);
230                       validator.accept(streamsSideTopicConfig, brokerSideTopicConfig);
231                       topicsStillToValidate.remove(topicName);
232                   } catch (final ExecutionException executionException) {
233                       final Throwable cause = executionException.getCause();
234                       if (cause instanceof UnknownTopicOrPartitionException) {
235                           log.info("Internal topic {} is missing", topicName);
236                           validationResult.addMissingTopic(topicName);
237                           topicsStillToValidate.remove(topicName);
238                       } else if (cause instanceof LeaderNotAvailableException) {
239                           log.info("The leader of internal topic {} is not available.", topicName);
240                       } else if (cause instanceof TimeoutException) {
241                           log.info("Retrieving data for internal topic {} timed out.", topicName);
242                       } else {
243                           log.error("Unexpected error during internal topic validation: ", cause);
244                           throw new StreamsException(
245                               String.format("Could not validate internal topic %s for the following reason: ", topicName),
246                               cause
247                           );
248                       }
249                   } catch (final InterruptedException interruptedException) {
250                       throw new InterruptException(interruptedException);
251                   } finally {
252                       futuresForTopic.remove(topicName);
253                   }
254               }
255           }
256       }

258       private void validatePartitionCount(final ValidationResult validationResult,
259                                           final InternalTopicConfig topicConfig,
260                                           final TopicDescription topicDescription) {
261           final String topicName = topicConfig.name();
262           final int requiredPartitionCount = topicConfig.numberOfPartitions()
263               .orElseThrow(() -> new IllegalStateException("No partition count is specified for internal topic " +
264                   topicName + ". " + BUG_ERROR_MESSAGE));
265           final int actualPartitionCount = topicDescription.partitions().size();
266           if (actualPartitionCount != requiredPartitionCount) {
267               validationResult.addMisconfiguration(
268                   topicName,
269                   "Internal topic " + topicName + " requires " + requiredPartitionCount + " partitions, " +
270                   "but the existing topic on the broker has " + actualPartitionCount + " partitions."
271               );
272           }
273       }

275       private void validateCleanupPolicy(final ValidationResult validationResult,
276                                          final InternalTopicConfig topicConfig,
277                                          final Config brokerSideTopicConfig) {
278           if (topicConfig instanceof UnwindowedChangelogTopicConfig) {
279               validateCleanupPolicyForUnwindowedChangelogs(validationResult, topicConfig, brokerSideTopicConfig);
280           } else if (topicConfig instanceof WindowedChangelogTopicConfig) {
281               validateCleanupPolicyForWindowedChangelogs(validationResult, topicConfig, brokerSideTopicConfig);
282           } else if (topicConfig instanceof RepartitionTopicConfig) {
283               validateCleanupPolicyForRepartitionTopic(validationResult, topicConfig, brokerSideTopicConfig);
284           } else {
285               throw new IllegalStateException("Internal topic " + topicConfig.name() + " has unknown type.");
286           }
287       }

289       private void validateCleanupPolicyForUnwindowedChangelogs(final ValidationResult validationResult,
290                                                                 final InternalTopicConfig topicConfig,
291                                                                 final Config brokerSideTopicConfig) {
292           final String topicName = topicConfig.name();
293           final String cleanupPolicy = getBrokerSideConfigValue(brokerSideTopicConfig, TopicConfig.CLEANUP_POLICY_CONFIG, topicName);
294           if (cleanupPolicy.contains(TopicConfig.CLEANUP_POLICY_DELETE)) {
295               validationResult.addMisconfiguration(
296                   topicName,
297                   "Cleanup policy (" + TopicConfig.CLEANUP_POLICY_CONFIG + ") of existing internal topic "
298                       + topicName + " should not contain \""
299                   + TopicConfig.CLEANUP_POLICY_DELETE + "\"."
300               );
301           }
302       }

304       private void validateCleanupPolicyForWindowedChangelogs(final ValidationResult validationResult,
305                                                               final InternalTopicConfig topicConfig,
306                                                               final Config brokerSideTopicConfig) {
307           final String topicName = topicConfig.name();
308           final String cleanupPolicy = getBrokerSideConfigValue(brokerSideTopicConfig, TopicConfig.CLEANUP_POLICY_CONFIG, topicName);
309           if (cleanupPolicy.contains(TopicConfig.CLEANUP_POLICY_DELETE)) {
310               final long brokerSideRetentionMs =
311                   Long.parseLong(getBrokerSideConfigValue(brokerSideTopicConfig, TopicConfig.RETENTION_MS_CONFIG, topicName));
312               final Map<String, String> streamsSideConfig =
313                   topicConfig.getProperties(defaultTopicConfigs, windowChangeLogAdditionalRetention);
314               final long streamsSideRetentionMs = Long.parseLong(streamsSideConfig.get(TopicConfig.RETENTION_MS_CONFIG));
315               if (brokerSideRetentionMs < streamsSideRetentionMs) {
316                   validationResult.addMisconfiguration(
317                       topicName,
318                       "Retention time (" + TopicConfig.RETENTION_MS_CONFIG + ") of existing internal topic "
319                           + topicName + " is " + brokerSideRetentionMs + " but should be " + streamsSideRetentionMs + " or larger."
320                   );
321               }
322               final String brokerSideRetentionBytes =
323                   getBrokerSideConfigValue(brokerSideTopicConfig, TopicConfig.RETENTION_BYTES_CONFIG, topicName);
324               if (brokerSideRetentionBytes != null) {
325                   validationResult.addMisconfiguration(
326                       topicName,
327                       "Retention byte (" + TopicConfig.RETENTION_BYTES_CONFIG + ") of existing internal topic "
328                           + topicName + " is set but it should be unset."
329                   );
330               }
331           }
332       }

334       private void validateCleanupPolicyForRepartitionTopic(final ValidationResult validationResult,
335                                                             final InternalTopicConfig topicConfig,
336                                                             final Config brokerSideTopicConfig) {
337           final String topicName = topicConfig.name();
338           final String cleanupPolicy = getBrokerSideConfigValue(brokerSideTopicConfig, TopicConfig.CLEANUP_POLICY_CONFIG, topicName);
339           if (cleanupPolicy.contains(TopicConfig.CLEANUP_POLICY_COMPACT)) {
340               validationResult.addMisconfiguration(
341                   topicName,
342                   "Cleanup policy (" + TopicConfig.CLEANUP_POLICY_CONFIG + ") of existing internal topic "
343                       + topicName + " should not contain \"" + TopicConfig.CLEANUP_POLICY_COMPACT + "\"."
344               );
345           } else if (cleanupPolicy.contains(TopicConfig.CLEANUP_POLICY_DELETE)) {
346               final long brokerSideRetentionMs =
347                   Long.parseLong(getBrokerSideConfigValue(brokerSideTopicConfig, TopicConfig.RETENTION_MS_CONFIG, topicName));
348               if (brokerSideRetentionMs != -1) {
349                   validationResult.addMisconfiguration(
350                       topicName,
351                       "Retention time (" + TopicConfig.RETENTION_MS_CONFIG + ") of existing internal topic "
352                           + topicName + " is " + brokerSideRetentionMs + " but should be -1."
353                   );
354               }
355               final String brokerSideRetentionBytes =
356                   getBrokerSideConfigValue(brokerSideTopicConfig, TopicConfig.RETENTION_BYTES_CONFIG, topicName);
357               if (brokerSideRetentionBytes != null) {
358                   validationResult.addMisconfiguration(
359                       topicName,
360                       "Retention byte (" + TopicConfig.RETENTION_BYTES_CONFIG + ") of existing internal topic "
361                           + topicName + " is set but it should be unset."
362                   );
363               }
364           }
365       }

367       private String getBrokerSideConfigValue(final Config brokerSideTopicConfig,
368                                               final String configName,
369                                               final String topicName) {
370           final ConfigEntry brokerSideConfigEntry = brokerSideTopicConfig.get(configName);
371           if (brokerSideConfigEntry == null) {
372               throw new IllegalStateException("The config " + configName + " for topic " +
373                   topicName + " could not be " + "retrieved from the brokers. " + BUG_ERROR_MESSAGE);
374           }
375           return brokerSideConfigEntry.value();
376       }

378       /**
379        * Prepares a set of given internal topics.
380        *
381        * If a topic does not exist creates a new topic.
382        * If a topic with the correct number of partitions exists ignores it.
383        * If a topic exists already but has different number of partitions we fail and throw exception requesting user to reset the app before restarting again.
384        * @return the set of topics which had to be newly created
385        */
386       public Set<String> makeReady(final Map<String, InternalTopicConfig> topics) {
387           // we will do the validation / topic-creation in a loop, until we have confirmed all topics
388           // have existed with the expected number of partitions, or some create topic returns fatal errors.
389           log.debug("Starting to validate internal topics {} in partition assignor.", topics);
390   
391           long currentWallClockMs = time.milliseconds();
392           final long deadlineMs = currentWallClockMs + retryTimeoutMs;
393   
394           Set<String> topicsNotReady = new HashSet<>(topics.keySet());
395           final Set<String> newlyCreatedTopics = new HashSet<>();
396   
397           while (!topicsNotReady.isEmpty()) {
398               final Set<String> tempUnknownTopics = new HashSet<>();
399               topicsNotReady = validateTopics(topicsNotReady, topics, tempUnknownTopics);
400               newlyCreatedTopics.addAll(topicsNotReady);
401   
402               if (!topicsNotReady.isEmpty()) {
403                   final Set<NewTopic> newTopics = new HashSet<>();
404   
405                   for (final String topicName : topicsNotReady) {
406                       if (tempUnknownTopics.contains(topicName)) {
407                           // for the tempUnknownTopics, don't create topic for them
408                           // we'll check again later if remaining retries > 0
409                           continue;
410                       }
411                       final InternalTopicConfig internalTopicConfig = Objects.requireNonNull(topics.get(topicName));
412                       final Map<String, String> topicConfig = internalTopicConfig.getProperties(defaultTopicConfigs, windowChangeLogAdditionalRetention);
413   
414                       log.debug("Going to create topic {} with {} partitions and config {}.",
415                           internalTopicConfig.name(),
416                           internalTopicConfig.numberOfPartitions(),
417                           topicConfig);
418   
419                       newTopics.add(
420                           new NewTopic(
421                               internalTopicConfig.name(),
422                               internalTopicConfig.numberOfPartitions(),
423                               Optional.of(replicationFactor))
424                               .configs(topicConfig));
425                   }
426   
427                   // it's possible that although some topics are not ready yet because they
428                   // are temporarily not available, not that they do not exist; in this case
429                   // the new topics to create may be empty and hence we can skip here
430                   if (!newTopics.isEmpty()) {
431                       final CreateTopicsResult createTopicsResult = adminClient.createTopics(newTopics);
432   
433                       for (final Map.Entry<String, KafkaFuture<Void>> createTopicResult : createTopicsResult.values().entrySet()) {
434                           final String topicName = createTopicResult.getKey();
435                           try {
436                               createTopicResult.getValue().get();
437                               topicsNotReady.remove(topicName);
438                           } catch (final InterruptedException fatalException) {
439                               // this should not happen; if it ever happens it indicate a bug
440                               Thread.currentThread().interrupt();
441                               log.error(INTERRUPTED_ERROR_MESSAGE, fatalException);
442                               throw new IllegalStateException(INTERRUPTED_ERROR_MESSAGE, fatalException);
443                           } catch (final ExecutionException executionException) {
444                               final Throwable cause = executionException.getCause();
445                               if (cause instanceof TopicExistsException) {
446                                   // This topic didn't exist earlier or its leader not known before; just retain it for next round of validation.
447                                   log.info(
448                                           "Could not create topic {}. Topic is probably marked for deletion (number of partitions is unknown).\n"
449                                                   +
450                                                   "Will retry to create this topic in {} ms (to let broker finish async delete operation first).\n"
451                                                   +
452                                                   "Error message was: {}", topicName, retryBackOffMs,
453                                           cause.toString());
454                               } else {
455                                   log.error("Unexpected error during topic creation for {}.\n" +
456                                           "Error message was: {}", topicName, cause.toString());
457   
458                                   if (cause instanceof UnsupportedVersionException) {
459                                       final String errorMessage = cause.getMessage();
460                                       if (errorMessage != null &&
461                                               errorMessage.startsWith("Creating topics with default partitions/replication factor are only supported in CreateTopicRequest version 4+")) {
462   
463                                           throw new StreamsException(String.format(
464                                                   "Could not create topic %s, because brokers don't support configuration replication.factor=-1."
465                                                           + " You can change the replication.factor config or upgrade your brokers to version 2.4 or newer to avoid this error.",
466                                                   topicName)
467                                           );
468                                       }
469                                   } else {
470                                       throw new StreamsException(
471                                               String.format("Could not create topic %s.", topicName),
472                                               cause
473                                       );
474                                   }
475                               }
476                           } catch (final TimeoutException retriableException) {
477                               log.error("Creating topic {} timed out.\n" +
478                                       "Error message was: {}", topicName, retriableException.toString());
479                           }
480                       }
481                   }
482               }
483   
484               if (!topicsNotReady.isEmpty()) {
485                   currentWallClockMs = time.milliseconds();
486   
487                   if (currentWallClockMs >= deadlineMs) {
488                       final String timeoutError = String.format("Could not create topics within %d milliseconds. " +
489                           "This can happen if the Kafka cluster is temporarily not available.", retryTimeoutMs);
490                       log.error(timeoutError);
491                       throw new TimeoutException(timeoutError);
492                   }
493                   log.info(
494                       "Topics {} could not be made ready. Will retry in {} milliseconds. Remaining time in milliseconds: {}",
495                       topicsNotReady,
496                       retryBackOffMs,
497                       deadlineMs - currentWallClockMs
498                   );
499                   Utils.sleep(retryBackOffMs);
500               }
501           }
502           log.debug("Completed validating internal topics and created {}", newlyCreatedTopics);
503   
504           return newlyCreatedTopics;
505       }

507       /**
508        * Try to get the number of partitions for the given topics; return the number of partitions for topics that already exists.
509        *
510        * Topics that were not able to get its description will simply not be returned
511        */
512       // visible for testing
513       protected Map<String, Integer> getNumPartitions(final Set<String> topics,
514                                                       final Set<String> tempUnknownTopics) {
515           log.debug("Trying to check if topics {} have been created with expected number of partitions.", topics);
516   
517           final DescribeTopicsResult describeTopicsResult = adminClient.describeTopics(topics);
518           final Map<String, KafkaFuture<TopicDescription>> futures = describeTopicsResult.topicNameValues();
519   
520           final Map<String, Integer> existedTopicPartition = new HashMap<>();
521           for (final Map.Entry<String, KafkaFuture<TopicDescription>> topicFuture : futures.entrySet()) {
522               final String topicName = topicFuture.getKey();
523               try {
524                   final TopicDescription topicDescription = topicFuture.getValue().get();
525                   existedTopicPartition.put(topicName, topicDescription.partitions().size());
526               } catch (final InterruptedException fatalException) {
527                   // this should not happen; if it ever happens it indicate a bug
528                   Thread.currentThread().interrupt();
529                   log.error(INTERRUPTED_ERROR_MESSAGE, fatalException);
530                   throw new IllegalStateException(INTERRUPTED_ERROR_MESSAGE, fatalException);
531               } catch (final ExecutionException couldNotDescribeTopicException) {
532                   final Throwable cause = couldNotDescribeTopicException.getCause();
533                   if (cause instanceof UnknownTopicOrPartitionException) {
534                       // This topic didn't exist, proceed to try to create it
535                       log.debug("Topic {} is unknown or not found, hence not existed yet.\n" +
536                           "Error message was: {}", topicName, cause.toString());
537                   } else if (cause instanceof LeaderNotAvailableException) {
538                       tempUnknownTopics.add(topicName);
539                       log.debug("The leader of topic {} is not available.\n" +
540                           "Error message was: {}", topicName, cause.toString());
541                   } else {
542                       log.error("Unexpected error during topic description for {}.\n" +
543                           "Error message was: {}", topicName, cause.toString());
544                       throw new StreamsException(String.format("Could not create topic %s.", topicName), cause);
545                   }
546               } catch (final TimeoutException retriableException) {
547                   tempUnknownTopics.add(topicName);
548                   log.debug("Describing topic {} (to get number of partitions) timed out.\n" +
549                       "Error message was: {}", topicName, retriableException.toString());
550               }
551           }
552   
553           return existedTopicPartition;
554       }
555   
556       /**
557        * Check the existing topics to have correct number of partitions; and return the remaining topics that needs to be created
558        */
559       private Set<String> validateTopics(final Set<String> topicsToValidate,
560                                          final Map<String, InternalTopicConfig> topicsMap,
561                                          final Set<String> tempUnknownTopics) {
562           if (!topicsMap.keySet().containsAll(topicsToValidate)) {
563               throw new IllegalStateException("The topics map " + topicsMap.keySet() + " does not contain all the topics " +
564                   topicsToValidate + " trying to validate.");
565           }
566   
567           final Map<String, Integer> existedTopicPartition = getNumPartitions(topicsToValidate, tempUnknownTopics);
568   
569           final Set<String> topicsToCreate = new HashSet<>();
570           for (final String topicName : topicsToValidate) {
571               final Optional<Integer> numberOfPartitions = topicsMap.get(topicName).numberOfPartitions();
572               if (!numberOfPartitions.isPresent()) {
573                   log.error("Found undefined number of partitions for topic {}", topicName);
574                   throw new StreamsException("Topic " + topicName + " number of partitions not defined");
575               }
576               if (existedTopicPartition.containsKey(topicName)) {
577                   if (!existedTopicPartition.get(topicName).equals(numberOfPartitions.get())) {
578                       final String errorMsg = String.format("Existing internal topic %s has invalid partitions: " +
579                               "expected: %d; actual: %d. " +
580                               "Use 'kafka.tools.StreamsResetter' tool to clean up invalid topics before processing.",
581                           topicName, numberOfPartitions.get(), existedTopicPartition.get(topicName));
582                       log.error(errorMsg);
583                       throw new StreamsException(errorMsg);
584                   }
585               } else {
586                   topicsToCreate.add(topicName);
587               }
588           }
589   
590           return topicsToCreate;
591       }

593       /**
594        * Sets up internal topics.
595        *
596        * Either the given topic are all created or the method fails with an exception.
597        *
598        * @param topicConfigs internal topics to setup
599        */
600       public void setup(final Map<String, InternalTopicConfig> topicConfigs) {
601           log.info("Starting to setup internal topics {}.", topicConfigs.keySet());
602   
603           final long now = time.milliseconds();
604           final long deadline = now + retryTimeoutMs;
605   
606           final Map<String, Map<String, String>> streamsSideTopicConfigs = topicConfigs.values().stream()
607               .collect(Collectors.toMap(
608                   InternalTopicConfig::name,
609                   topicConfig -> topicConfig.getProperties(defaultTopicConfigs, windowChangeLogAdditionalRetention)
610               ));
611           final Set<String> createdTopics = new HashSet<>();
612           final Set<String> topicStillToCreate = new HashSet<>(topicConfigs.keySet());
613           while (!topicStillToCreate.isEmpty()) {
614               final Set<NewTopic> newTopics = topicStillToCreate.stream()
615                   .map(topicName -> new NewTopic(
616                           topicName,
617                           topicConfigs.get(topicName).numberOfPartitions(),
618                           Optional.of(replicationFactor)
619                       ).configs(streamsSideTopicConfigs.get(topicName))
620                   ).collect(Collectors.toSet());
621   
622               log.info("Going to create internal topics: " + newTopics);
623               final CreateTopicsResult createTopicsResult = adminClient.createTopics(newTopics);
624   
625               processCreateTopicResults(createTopicsResult, topicStillToCreate, createdTopics, deadline);
626   
627               maybeSleep(Collections.singletonList(topicStillToCreate), deadline, "created");
628           }
629   
630           log.info("Completed setup of internal topics {}.", topicConfigs.keySet());
631       }

633       private void processCreateTopicResults(final CreateTopicsResult createTopicsResult,
634                                              final Set<String> topicStillToCreate,
635                                              final Set<String> createdTopics,
636                                              final long deadline) {
637           final Map<String, Throwable> lastErrorsSeenForTopic = new HashMap<>();
638           final Map<String, KafkaFuture<Void>> createResultForTopic = createTopicsResult.values();
639           while (!createResultForTopic.isEmpty()) {
640               for (final String topicName : new HashSet<>(topicStillToCreate)) {
641                   if (!createResultForTopic.containsKey(topicName)) {
642                       cleanUpCreatedTopics(createdTopics);
643                       throw new IllegalStateException("Create topic results do not contain internal topic " + topicName
644                           + " to setup. " + BUG_ERROR_MESSAGE);
645                   }
646                   final KafkaFuture<Void> createResult = createResultForTopic.get(topicName);
647                   if (createResult.isDone()) {
648                       try {
649                           createResult.get();
650                           createdTopics.add(topicName);
651                           topicStillToCreate.remove(topicName);
652                       } catch (final ExecutionException executionException) {
653                           final Throwable cause = executionException.getCause();
654                           if (cause instanceof TopicExistsException) {
655                               lastErrorsSeenForTopic.put(topicName, cause);
656                               log.info("Internal topic {} already exists. Topic is probably marked for deletion. " +
657                                   "Will retry to create this topic later (to let broker complete async delete operation first)",
658                                   topicName);
659                           } else if (cause instanceof TimeoutException) {
660                               lastErrorsSeenForTopic.put(topicName, cause);
661                               log.info("Creating internal topic {} timed out.", topicName);
662                           } else {
663                               cleanUpCreatedTopics(createdTopics);
664                               log.error("Unexpected error during creation of internal topic: ", cause);
665                               throw new StreamsException(
666                                   String.format("Could not create internal topic %s for the following reason: ", topicName),
667                                   cause
668                               );
669                           }
670                       } catch (final InterruptedException interruptedException) {
671                           throw new InterruptException(interruptedException);
672                       } finally {
673                           createResultForTopic.remove(topicName);
674                       }
675                   }
676               }
677   
678               maybeThrowTimeoutExceptionDuringSetup(
679                   topicStillToCreate,
680                   createdTopics,
681                   lastErrorsSeenForTopic,
682                   deadline
683               );
684   
685               if (!createResultForTopic.isEmpty()) {
686                   Utils.sleep(100);
687               }
688           }
689       }

691       private void cleanUpCreatedTopics(final Set<String> topicsToCleanUp) {
692           log.info("Starting to clean up internal topics {}.", topicsToCleanUp);
693   
694           final long now = time.milliseconds();
695           final long deadline = now + retryTimeoutMs;
696   
697           final Set<String> topicsStillToCleanup = new HashSet<>(topicsToCleanUp);
698           while (!topicsStillToCleanup.isEmpty()) {
699               log.info("Going to cleanup internal topics: " + topicsStillToCleanup);
700               final DeleteTopicsResult deleteTopicsResult = adminClient.deleteTopics(topicsStillToCleanup);
701               final Map<String, KafkaFuture<Void>> deleteResultForTopic = deleteTopicsResult.topicNameValues();
702               while (!deleteResultForTopic.isEmpty()) {
703                   for (final String topicName : new HashSet<>(topicsStillToCleanup)) {
704                       if (!deleteResultForTopic.containsKey(topicName)) {
705                           throw new IllegalStateException("Delete topic results do not contain internal topic " + topicName
706                               + " to clean up. " + BUG_ERROR_MESSAGE);
707                       }
708                       final KafkaFuture<Void> deleteResult = deleteResultForTopic.get(topicName);
709                       if (deleteResult.isDone()) {
710                           try {
711                               deleteResult.get();
712                               topicsStillToCleanup.remove(topicName);
713                           } catch (final ExecutionException executionException) {
714                               final Throwable cause = executionException.getCause();
715                               if (cause instanceof UnknownTopicOrPartitionException) {
716                                   log.info("Internal topic {} to clean up is missing", topicName);
717                               } else if (cause instanceof LeaderNotAvailableException) {
718                                   log.info("The leader of internal topic {} to clean up is not available.", topicName);
719                               } else if (cause instanceof TimeoutException) {
720                                   log.info("Cleaning up internal topic {} timed out.", topicName);
721                               } else {
722                                   log.error("Unexpected error during cleanup of internal topics: ", cause);
723                                   throw new StreamsException(
724                                       String.format("Could not clean up internal topics %s, because during the cleanup " +
725                                               "of topic %s the following error occurred: ",
726                                           topicsStillToCleanup, topicName),
727                                       cause
728                                   );
729                               }
730                           } catch (final InterruptedException interruptedException) {
731                               throw new InterruptException(interruptedException);
732                           } finally {
733                               deleteResultForTopic.remove(topicName);
734                           }
735                       }
736                   }
737   
738                   maybeThrowTimeoutException(
739                       Collections.singletonList(topicsStillToCleanup),
740                       deadline,
741                       String.format("Could not cleanup internal topics within %d milliseconds. This can happen if the " +
742                               "Kafka cluster is temporarily not available or the broker did not complete topic creation " +
743                               "before the cleanup. The following internal topics could not be cleaned up: %s",
744                           retryTimeoutMs, topicsStillToCleanup)
745                   );
746   
747                   if (!deleteResultForTopic.isEmpty()) {
748                       Utils.sleep(100);
749                   }
750               }
751   
752               maybeSleep(
753                   Collections.singletonList(topicsStillToCleanup),
754                   deadline,
755                   "validated"
756               );
757           }
758   
759           log.info("Completed cleanup of internal topics {}.", topicsToCleanUp);
760       }

762       private void maybeThrowTimeoutException(final List<Set<String>> topicStillToProcess,
763                                               final long deadline,
764                                               final String errorMessage) {
765           if (topicStillToProcess.stream().anyMatch(resultSet -> !resultSet.isEmpty())) {
766               final long now = time.milliseconds();
767               if (now >= deadline) {
768                   log.error(errorMessage);
769                   throw new TimeoutException(errorMessage);
770               }
771           }
772       }

774       private void maybeThrowTimeoutExceptionDuringSetup(final Set<String> topicStillToProcess,
775                                                          final Set<String> createdTopics,
776                                                          final Map<String, Throwable> lastErrorsSeenForTopic,
777                                                          final long deadline) {
778           if (topicStillToProcess.stream().anyMatch(resultSet -> !resultSet.isEmpty())) {
779               final long now = time.milliseconds();
780               if (now >= deadline) {
781                   cleanUpCreatedTopics(createdTopics);
782                   final String errorMessage = String.format("Could not create internal topics within %d milliseconds. This can happen if the " +
783                       "Kafka cluster is temporarily not available or a topic is marked for deletion and the broker " +
784                       "did not complete its deletion within the timeout. The last errors seen per topic are: %s",
785                       retryTimeoutMs, lastErrorsSeenForTopic);
786                   log.error(errorMessage);
787                   throw new TimeoutException(errorMessage);
788               }
789           }
790       }

792       private void maybeSleep(final List<Set<String>> resultSetsStillToValidate,
793                               final long deadline,
794                               final String action) {
795           if (resultSetsStillToValidate.stream().anyMatch(resultSet -> !resultSet.isEmpty())) {
796               final long now = time.milliseconds();
797               log.info(
798                   "Internal topics {} could not be {}. Will retry in {} milliseconds. Remaining time in milliseconds: {}",
799                   resultSetsStillToValidate.stream().flatMap(Collection::stream).collect(Collectors.toSet()),
800                   action,
801                   retryBackOffMs,
802                   deadline - now
803               );
804               Utils.sleep(retryBackOffMs);
805           }
806       }
