19   import org.apache.kafka.streams.StoreQueryParameters;
20   import org.apache.kafka.streams.errors.InvalidStateStoreException;
21   import org.apache.kafka.streams.processor.StateStore;
22   import org.apache.kafka.streams.processor.TaskId;
23   import org.apache.kafka.streams.processor.internals.StreamThread;
24   import org.apache.kafka.streams.processor.internals.Task;
25   import org.apache.kafka.streams.processor.internals.namedtopology.NamedTopologyStoreQueryParameters;
26   import org.apache.kafka.streams.state.QueryableStoreType;
27   import org.apache.kafka.streams.state.QueryableStoreTypes;
28   import org.apache.kafka.streams.state.TimestampedKeyValueStore;
29   import org.apache.kafka.streams.state.TimestampedWindowStore;

31   import java.util.ArrayList;
32   import java.util.Collection;
33   import java.util.Collections;
34   import java.util.List;

36   public class StreamThreadStateStoreProvider {
37   
38       private final StreamThread streamThread;
39   
40       public StreamThreadStateStoreProvider(final StreamThread streamThread) {
41           this.streamThread = streamThread;
42       }
43   
44       @SuppressWarnings("unchecked")
45       public <T> List<T> stores(final StoreQueryParameters storeQueryParams) {
46           final StreamThread.State state = streamThread.state();
47           if (state == StreamThread.State.DEAD) {
48               return Collections.emptyList();
49           }
50   
51           final String storeName = storeQueryParams.storeName();
52           final QueryableStoreType<T> queryableStoreType = storeQueryParams.queryableStoreType();
53           final String topologyName = storeQueryParams instanceof NamedTopologyStoreQueryParameters ?
54               ((NamedTopologyStoreQueryParameters) storeQueryParams).topologyName() :
55               null;
56   
57           if (storeQueryParams.staleStoresEnabled() ? state.isAlive() : state == StreamThread.State.RUNNING) {
58               final Collection<Task> tasks = storeQueryParams.staleStoresEnabled() ?
59                       streamThread.allTasks().values() : streamThread.activeTasks();
60   
61               if (storeQueryParams.partition() != null) {
62                   for (final Task task : tasks) {
63                       if (task.id().partition() == storeQueryParams.partition() &&
64                           (topologyName == null || topologyName.equals(task.id().topologyName())) &&
65                           task.getStore(storeName) != null &&
66                           storeName.equals(task.getStore(storeName).name())) {
67                           final T typedStore = validateAndCastStores(task.getStore(storeName), queryableStoreType, storeName, task.id());
68                           return Collections.singletonList(typedStore);
69                       }
70                   }
71                   return Collections.emptyList();
72               } else {
73                   final List<T> list = new ArrayList<>();
74                   for (final Task task : tasks) {
75                       final StateStore store = task.getStore(storeName);
76                       if (store == null) {
77                           // then this task doesn't have that store
78                       } else {
79                           final T typedStore = validateAndCastStores(store, queryableStoreType, storeName, task.id());
80                           list.add(typedStore);
81                       }
82                   }
83                   return list;
84               }
85           } else {
86               throw new InvalidStateStoreException("Cannot get state store " + storeName + " because the stream thread is " +
87                                                       state + ", not RUNNING" +
88                                                       (storeQueryParams.staleStoresEnabled() ? " or REBALANCING" : ""));
89           }
90       }
91   
92       @SuppressWarnings("unchecked")
93       private static <T> T validateAndCastStores(final StateStore store,
94                                                  final QueryableStoreType<T> queryableStoreType,
95                                                  final String storeName,
96                                                  final TaskId taskId) {
97           if (store == null) {
98               throw new NullPointerException("Expected store not to be null at this point.");
99           } else if (queryableStoreType.accepts(store)) {
100               if (!store.isOpen()) {
101                   throw new InvalidStateStoreException(
102                           "Cannot get state store " + storeName + " for task " + taskId +
103                               " because the store is not open. " +
104                               "The state store may have migrated to another instance.");
105               }
106               if (store instanceof TimestampedKeyValueStore && queryableStoreType instanceof QueryableStoreTypes.KeyValueStoreType) {
107                   return (T) new ReadOnlyKeyValueStoreFacade<>((TimestampedKeyValueStore<Object, Object>) store);
108               } else if (store instanceof TimestampedWindowStore && queryableStoreType instanceof QueryableStoreTypes.WindowStoreType) {
109                   return (T) new ReadOnlyWindowStoreFacade<>((TimestampedWindowStore<Object, Object>) store);
110               } else {
111                   return (T) store;
112               }
113           } else {
114               throw new InvalidStateStoreException(
115                   "Cannot get state store " + storeName +
116                       " because the queryable store type [" + queryableStoreType.getClass() +
117                       "] does not accept the actual store type [" + store.getClass() + "]."
118               );
119           }
120       }
121   }
