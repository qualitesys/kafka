36   /**
37    * Groups partitions by the partition id.
38    *
39    * Join operations requires that topics of the joining entities are copartitioned, i.e., being partitioned by the same key and having the same
40    * number of partitions. Copartitioning is ensured by having the same number of partitions on
41    * joined topics, and by using the serialization and Producer's default partitioner.
42    */
43   public class PartitionGrouper {
44   
45       private static final Logger log = LoggerFactory.getLogger(PartitionGrouper.class);
46       /**
47        * Generate tasks with the assigned topic partitions.
48        *
49        * @param topicGroups   group of topics that need to be joined together
50        * @param metadata      metadata of the consuming cluster
51        * @return The map from generated task ids to the assigned partitions
52        */
53       public Map<TaskId, Set<TopicPartition>> partitionGroups(final Map<Subtopology, Set<String>> topicGroups, final Cluster metadata) {
54           final Map<TaskId, Set<TopicPartition>> groups = new HashMap<>();
55   
56           for (final Map.Entry<Subtopology, Set<String>> entry : topicGroups.entrySet()) {
57               final Subtopology subtopology = entry.getKey();
58               final Set<String> topicGroup = entry.getValue();
59   
60               final int maxNumPartitions = maxNumPartitions(metadata, topicGroup);
61   
62               for (int partitionId = 0; partitionId < maxNumPartitions; partitionId++) {
63                   final Set<TopicPartition> group = new HashSet<>(topicGroup.size());
64   
65                   for (final String topic : topicGroup) {
66                       final List<PartitionInfo> partitions = metadata.partitionsForTopic(topic);
67                       if (partitionId < partitions.size()) {
68                           group.add(new TopicPartition(topic, partitionId));
69                       }
70                   }
71                   groups.put(new TaskId(subtopology.nodeGroupId, partitionId, subtopology.namedTopology), Collections.unmodifiableSet(group));
72               }
73           }
74   
75           return Collections.unmodifiableMap(groups);
76       }
77   
78       /**
79        * @throws StreamsException if no metadata can be received for a topic
80        */
81       protected int maxNumPartitions(final Cluster metadata, final Set<String> topics) {
82           int maxNumPartitions = 0;
83           for (final String topic : topics) {
84               final List<PartitionInfo> partitions = metadata.partitionsForTopic(topic);
85               if (partitions.isEmpty()) {
86                   log.error("Empty partitions for topic {}", topic);
87                   throw new RuntimeException("Empty partitions for topic " + topic);
88               }
89   
90               final int numPartitions = partitions.size();
91               if (numPartitions > maxNumPartitions) {
92                   maxNumPartitions = numPartitions;
93               }
94           }
95           return maxNumPartitions;
96       }

98   }
