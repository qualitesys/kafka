17   package org.apache.kafka.streams.kstream;
18   
19   import org.apache.kafka.streams.processor.TimestampExtractor;
20   
21   import java.time.Instant;
22   
23   /**
24    * A single window instance, defined by its start and end timestamp.
25    * {@code Window} is agnostic if start/end boundaries are inclusive or exclusive; this is defined by concrete
26    * window implementations.
27    * <p>
28    * To specify how {@code Window} boundaries are defined use {@link Windows}.
29    * For time semantics, see {@link TimestampExtractor}.
30    *
31    * @see Windows
32    * @see org.apache.kafka.streams.kstream.internals.TimeWindow
33    * @see org.apache.kafka.streams.kstream.internals.SessionWindow
34    * @see org.apache.kafka.streams.kstream.internals.UnlimitedWindow
35    * @see TimestampExtractor
36    */
37   public abstract class Window {
38   
39       protected final long startMs;
40       protected final long endMs;
41       private final Instant startTime;
42       private final Instant endTime;
43   
44   
45       /**
46        * Create a new window for the given start and end time.
47        *
48        * @param startMs the start timestamp of the window
49        * @param endMs   the end timestamp of the window
50        * @throws IllegalArgumentException if {@code startMs} is negative or if {@code endMs} is smaller than {@code startMs}
51        */
52       public Window(final long startMs, final long endMs) throws IllegalArgumentException {
53           if (startMs < 0) {
54               throw new IllegalArgumentException("Window startMs time cannot be negative.");
55           }
56           if (endMs < startMs) {
57               throw new IllegalArgumentException("Window endMs time cannot be smaller than window startMs time.");
58           }
59           this.startMs = startMs;
60           this.endMs = endMs;
61   
62           this.startTime = Instant.ofEpochMilli(startMs);
63           this.endTime = Instant.ofEpochMilli(endMs);
64       }
65   
66       /**
67        * Return the start timestamp of this window.
68        *
69        * @return The start timestamp of this window.
70        */
71       public long start() {
72           return startMs;
73       }
74   
75       /**
76        * Return the end timestamp of this window.
77        *
78        * @return The end timestamp of this window.
79        */
80       public long end() {
81           return endMs;
82       }
83   
84       /**
85        * Return the start time of this window.
86        *
87        * @return The start time of this window.
88        */
89       public Instant startTime() {
90           return startTime;
91       }
92   
93       /**
94        * Return the end time of this window.
95        *
96        * @return The end time of this window.
97        */
98       public Instant endTime() {
99           return endTime;
100       }
101   
102       /**
103        * Check if the given window overlaps with this window.
104        * Should throw an {@link IllegalArgumentException} if the {@code other} window has a different type than {@code
105        * this} window.
106        *
107        * @param other another window of the same type
108        * @return {@code true} if {@code other} overlaps with this window&mdash;{@code false} otherwise
109        */
110       public abstract boolean overlap(final Window other);
111   
112       @Override
113       public boolean equals(final Object obj) {
114           if (obj == this) {
115               return true;
116           }
117           if (obj == null) {
118               return false;
119           }
120           if (getClass() != obj.getClass()) {
121               return false;
122           }
123   
124           final Window other = (Window) obj;
125           return startMs == other.startMs && endMs == other.endMs;
126       }
127   
128       @Override
129       public int hashCode() {
130           return (int) (((startMs << 32) | endMs) % 0xFFFFFFFFL);
131       }
132   
133       @Override
134       public String toString() {
135           return "Window{" +
136               "startMs=" + startMs +
137               ", endMs=" + endMs +
138               '}';
139       }
140   }
