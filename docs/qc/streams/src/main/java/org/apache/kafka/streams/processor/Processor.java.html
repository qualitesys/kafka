17   package org.apache.kafka.streams.processor;
18   
19   import java.time.Duration;
20   
21   /**
22    * A processor of key-value pair records.
23    *
24    * @param <K> the type of keys
25    * @param <V> the type of values
26    * @deprecated Since 3.0. Use {@link org.apache.kafka.streams.processor.api.Processor} instead.
27    */
28   @Deprecated
29   public interface Processor<K, V> {
30   
31       /**
32        * Initialize this processor with the given context. The framework ensures this is called once per processor when the topology
33        * that contains it is initialized. When the framework is done with the processor, {@link #close()} will be called on it; the
34        * framework may later re-use the processor by calling {@code #init()} again.
35        * <p>
36        * The provided {@link ProcessorContext context} can be used to access topology and record meta data, to
37        * {@link ProcessorContext#schedule(Duration, PunctuationType, Punctuator) schedule} a method to be
38        * {@link Punctuator#punctuate(long) called periodically} and to access attached {@link StateStore}s.
39        * 
40        * @param context the context; may not be null
41        */
42       void init(ProcessorContext context);
43   
44       /**
45        * Process the record with the given key and value.
46        *
47        * @param key the key for the record
48        * @param value the value for the record
49        */
50       void process(K key, V value);
51   
52       /**
53        * Close this processor and clean up any resources. Be aware that {@code #close()} is called after an internal cleanup.
54        * Thus, it is not possible to write anything to Kafka as underlying clients are already closed. The framework may
55        * later re-use this processor by calling {@code #init()} on it again.
56        * <p>
57        * Note: Do not close any streams managed resources, like {@link StateStore}s here, as they are managed by the library.
58        */
59       void close();
60   }
