20   import org.apache.kafka.common.annotation.InterfaceStability.Evolving;
21   
22   import java.util.Collections;
23   import java.util.Map;
24   import java.util.Map.Entry;
25   import java.util.Objects;
26   import java.util.Set;
27   import java.util.concurrent.ConcurrentHashMap;

29   /**
30    * A representation of a position vector with respect to a set of topic partitions. For example, in
31    * Interactive Query ({@link org.apache.kafka.streams.KafkaStreams#query(StateQueryRequest)}, a
32    * query result may contain information from multiple store partitions, each of which contains
33    * information from multiple input topics' partitions. This class can be used to summarize all of
34    * that positional information.
35    * <p>
36    * This class is threadsafe, although it is mutable. Readers are recommended to use {@link
37    * Position#copy()} to avoid seeing mutations to the Position after they get the reference. For
38    * examples, when a store executes a {@link org.apache.kafka.streams.processor.StateStore#query(Query,
39    * PositionBound, boolean)} request and returns its current position via {@link
40    * QueryResult#setPosition(Position)}, it should pass a copy of its position instead of the mutable
41    * reference.
42    */
43   @Evolving
44   public class Position {
45   
46       private final ConcurrentHashMap<String, ConcurrentHashMap<Integer, Long>> position;
47   
48       private Position(final ConcurrentHashMap<String, ConcurrentHashMap<Integer, Long>> position) {
49           this.position = position;
50       }
51   
52       /**
53        * Create a new, empty Position.
54        */
55       public static Position emptyPosition() {
56           return new Position(new ConcurrentHashMap<>());
57       }
58   
59       /**
60        * Create a new Position and populate it with a mapping of topic -> partition -> offset.
61        * <p>
62        * Note, the resulting Position does not share any structure with the provided map, so
63        * subsequent changes to the map or Position will not affect the other.
64        */
65       public static Position fromMap(final Map<String, ? extends Map<Integer, Long>> map) {
66           return new Position(deepCopy(map));
67       }
68   
69       /**
70        * Augment an existing Position by setting a new offset for a topic and partition.
71        * <p>
72        * Note: enforces monotonicity on offsets. I.e., if there is already a component for the same
73        * topic and partition with a larger offset, the update will succeed but not overwrite the
74        * offset.
75        * <p>
76        * Returns a self-reference for chained calls. Note: this method mutates the Position.
77        */
78       public Position withComponent(final String topic, final int partition, final long offset) {
79           position
80               .computeIfAbsent(topic, k -> new ConcurrentHashMap<>())
81               .compute(
82                   partition,
83                   (integer, prior) -> prior == null || offset > prior ? offset : prior
84               );
85           return this;
86       }
87   
88       /**
89        * Create a deep copy of the Position.
90        */
91       public Position copy() {
92           return new Position(deepCopy(position));
93       }
94   
95       /**
96        * Merges the provided Position into the current instance.
97        * <p>
98        * If both Positions contain the same topic -> partition -> offset mapping, the resulting
99        * Position will contain a mapping with the larger of the two offsets.
100        */
101       public Position merge(final Position other) {
102           if (other == null) {
103               return this;
104           } else {
105               for (final Entry<String, ConcurrentHashMap<Integer, Long>> entry : other.position.entrySet()) {
106                   final String topic = entry.getKey();
107                   final Map<Integer, Long> partitionMap =
108                       position.computeIfAbsent(topic, k -> new ConcurrentHashMap<>());
109                   for (final Entry<Integer, Long> partitionOffset : entry.getValue().entrySet()) {
110                       final Integer partition = partitionOffset.getKey();
111                       final Long offset = partitionOffset.getValue();
112                       if (!partitionMap.containsKey(partition)
113                           || partitionMap.get(partition) < offset) {
114                           partitionMap.put(partition, offset);
115                       }
116                   }
117               }
118               return this;
119           }
120       }
121   
122       /**
123        * Return the topics that are represented in this Position.
124        */
125       public Set<String> getTopics() {
126           return Collections.unmodifiableSet(position.keySet());
127       }
128   
129       /**
130        * Return the partition -> offset mapping for a specific topic.
131        */
132       public Map<Integer, Long> getPartitionPositions(final String topic) {
133           final ConcurrentHashMap<Integer, Long> bound = position.get(topic);
134           return bound == null ? Collections.emptyMap() : Collections.unmodifiableMap(bound);
135       }
136   
137       private static ConcurrentHashMap<String, ConcurrentHashMap<Integer, Long>> deepCopy(
138           final Map<String, ? extends Map<Integer, Long>> map) {
139           if (map == null) {
140               return new ConcurrentHashMap<>();
141           } else {
142               final ConcurrentHashMap<String, ConcurrentHashMap<Integer, Long>> copy =
143                   new ConcurrentHashMap<>(map.size());
144               for (final Entry<String, ? extends Map<Integer, Long>> entry : map.entrySet()) {
145                   copy.put(entry.getKey(), new ConcurrentHashMap<>(entry.getValue()));
146               }
147               return copy;
148           }
149       }
150   
151       @Override
152       public String toString() {
153           return "Position{" +
154               "position=" + position +
155               '}';
156       }
157   
158       @Override
159       public boolean equals(final Object o) {
160           if (this == o) {
161               return true;
162           }
163           if (o == null || getClass() != o.getClass()) {
164               return false;
165           }
166           final Position position1 = (Position) o;
167           return Objects.equals(position, position1.position);
168       }
169   
170       @Override
171       public int hashCode() {
172           throw new UnsupportedOperationException(
173               "This mutable object is not suitable as a hash key");
174       }
175   
176       public boolean isEmpty() {
177           return position.isEmpty();
178       }
179   }
