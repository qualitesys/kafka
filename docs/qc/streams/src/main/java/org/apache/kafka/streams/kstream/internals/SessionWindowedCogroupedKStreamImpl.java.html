42   public class SessionWindowedCogroupedKStreamImpl<K, V> extends
43       AbstractStream<K, V> implements SessionWindowedCogroupedKStream<K, V> {
44   
45       private final SessionWindows sessionWindows;
46       private final CogroupedStreamAggregateBuilder<K, V> aggregateBuilder;
47       private final Map<KGroupedStreamImpl<K, ?>, Aggregator<? super K, ? super Object, V>> groupPatterns;
48   
49       SessionWindowedCogroupedKStreamImpl(final SessionWindows sessionWindows,
50                                           final InternalStreamsBuilder builder,
51                                           final Set<String> subTopologySourceNodes,
52                                           final String name,
53                                           final CogroupedStreamAggregateBuilder<K, V> aggregateBuilder,
54                                           final GraphNode graphNode,
55                                           final Map<KGroupedStreamImpl<K, ?>, Aggregator<? super K, ? super Object, V>> groupPatterns) {
56           super(name, null, null, subTopologySourceNodes, graphNode, builder);
57           //keySerde and valueSerde are null because there are many different groupStreams that they could be from
58           this.sessionWindows = sessionWindows;
59           this.aggregateBuilder = aggregateBuilder;
60           this.groupPatterns = groupPatterns;
61       }

63       @Override
64       public KTable<Windowed<K>, V> aggregate(final Initializer<V> initializer,
65                                               final Merger<? super K, V> sessionMerger) {
66           return aggregate(initializer, sessionMerger, Materialized.with(null, null));
67       }

69       @Override
70       public KTable<Windowed<K>, V> aggregate(final Initializer<V> initializer,
71                                               final Merger<? super K, V> sessionMerger,
72                                               final Materialized<K, V, SessionStore<Bytes, byte[]>> materialized) {
73           return aggregate(initializer, sessionMerger, NamedInternal.empty(), materialized);
74       }

77       public KTable<Windowed<K>, V> aggregate(final Initializer<V> initializer,

79           return aggregate(initializer, sessionMerger, named, Materialized.with(null, null));
80       }

82       @Override
83       public KTable<Windowed<K>, V> aggregate(final Initializer<V> initializer,
84                                               final Merger<? super K, V> sessionMerger, final Named named,
85                                               final Materialized<K, V, SessionStore<Bytes, byte[]>> materialized) {
86           Objects.requireNonNull(initializer, "initializer can't be null");
87           Objects.requireNonNull(sessionMerger, "sessionMerger can't be null");
88           Objects.requireNonNull(materialized, "materialized can't be null");
89           Objects.requireNonNull(named, "named can't be null");
90           final MaterializedInternal<K, V, SessionStore<Bytes, byte[]>> materializedInternal = new MaterializedInternal<>(
91               materialized,
92               builder,
93               CogroupedKStreamImpl.AGGREGATE_NAME);
94           return aggregateBuilder.build(
95               groupPatterns,
96               initializer,
97               new NamedInternal(named),
98               materialize(materializedInternal),
99               materializedInternal.keySerde() != null ?
100                   new WindowedSerdes.SessionWindowedSerde<>(
101                       materializedInternal.keySerde()) :
102                   null,
103               materializedInternal.valueSerde(),
104               materializedInternal.queryableStoreName(),
105               sessionWindows,
106               sessionMerger);
107       }
108   
109       private  StoreBuilder<SessionStore<K, V>> materialize(final MaterializedInternal<K, V, SessionStore<Bytes, byte[]>> materialized) {
110           SessionBytesStoreSupplier supplier = (SessionBytesStoreSupplier) materialized.storeSupplier();
111           if (supplier == null) {
112               final long retentionPeriod = materialized.retention() != null ?
113                   materialized.retention().toMillis() : sessionWindows.inactivityGap() + sessionWindows.gracePeriodMs();
114   
115               if ((sessionWindows.inactivityGap() + sessionWindows.gracePeriodMs()) > retentionPeriod) {
116                   throw new IllegalArgumentException("The retention period of the session store "
117                       + materialized.storeName()
118                       + " must be no smaller than the session inactivity gap plus the"
119                       + " grace period."
120                       + " Got gap=[" + sessionWindows.inactivityGap() + "],"
121                       + " grace=[" + sessionWindows.gracePeriodMs() + "],"
122                       + " retention=[" + retentionPeriod + "]");
123               }
124   
125               switch (materialized.storeType()) {
126                   case IN_MEMORY:
127                       supplier = Stores.inMemorySessionStore(
128                           materialized.storeName(),
129                           Duration.ofMillis(retentionPeriod)
130                       );
131                       break;
132                   case ROCKS_DB:
133                       supplier = Stores.persistentSessionStore(
134                           materialized.storeName(),
135                           Duration.ofMillis(retentionPeriod)
136                       );
137                       break;
138                   default:
139                       throw new IllegalStateException("Unknown store type: " + materialized.storeType());
140               }
141           }
142   
143           final StoreBuilder<SessionStore<K, V>> builder = Stores.sessionStoreBuilder(
144               supplier,
145               materialized.keySerde(),
146               materialized.valueSerde()
147           );
148   
149           if (materialized.loggingEnabled()) {
150               builder.withLoggingEnabled(materialized.logConfig());
151           } else {
152               builder.withLoggingDisabled();
153           }
154   
155           if (materialized.cachingEnabled()) {
156               builder.withCachingEnabled();
157           }
158           return builder;
159       }
