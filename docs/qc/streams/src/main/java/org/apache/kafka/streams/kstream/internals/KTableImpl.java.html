170       private KTable<K, V> doFilter(final Predicate<? super K, ? super V> predicate,
171                                     final Named named,
172                                     final MaterializedInternal<K, V, KeyValueStore<Bytes, byte[]>> materializedInternal,
173                                     final boolean filterNot) {
174           final Serde<K> keySerde;
175           final Serde<V> valueSerde;
176           final String queryableStoreName;
177           final StoreBuilder<TimestampedKeyValueStore<K, V>> storeBuilder;
178   
179           if (materializedInternal != null) {
180               // we actually do not need to generate store names at all since if it is not specified, we will not
181               // materialize the store; but we still need to burn one index BEFORE generating the processor to keep compatibility.
182               if (materializedInternal.storeName() == null) {
183                   builder.newStoreName(FILTER_NAME);
184               }
185               // we can inherit parent key and value serde if user do not provide specific overrides, more specifically:
186               // we preserve the key following the order of 1) materialized, 2) parent
187               keySerde = materializedInternal.keySerde() != null ? materializedInternal.keySerde() : this.keySerde;
188               // we preserve the value following the order of 1) materialized, 2) parent
189               valueSerde = materializedInternal.valueSerde() != null ? materializedInternal.valueSerde() : this.valueSerde;
190               queryableStoreName = materializedInternal.queryableStoreName();
191               // only materialize if materialized is specified and it has queryable name
192               storeBuilder = queryableStoreName != null ? (new TimestampedKeyValueStoreMaterializer<>(materializedInternal)).materialize() : null;
193           } else {
194               keySerde = this.keySerde;
195               valueSerde = this.valueSerde;
196               queryableStoreName = null;
197               storeBuilder = null;
198           }
199           final String name = new NamedInternal(named).orElseGenerateWithPrefix(builder, FILTER_NAME);
200   
201           final KTableProcessorSupplier<K, V, K, V> processorSupplier =
202               new KTableFilter<>(this, predicate, filterNot, queryableStoreName);
203   
204           final ProcessorParameters<K, V, ?, ?> processorParameters = unsafeCastProcessorParametersToCompletelyDifferentType(
205               new ProcessorParameters<>(processorSupplier, name)
206           );
207   
208           final GraphNode tableNode = new TableProcessorNode<>(
209               name,
210               processorParameters,
211               storeBuilder
212           );
213   
214           builder.addGraphNode(this.graphNode, tableNode);
215   
216           return new KTableImpl<K, V, V>(
217               name,
218               keySerde,
219               valueSerde,
220               subTopologySourceNodes,
221               queryableStoreName,
222               processorSupplier,
223               tableNode,
224               builder);
225       }

708       private <VO, VR> KTable<K, VR> doJoin(final KTable<K, VO> other,
709                                             final ValueJoiner<? super V, ? super VO, ? extends VR> joiner,
710                                             final Named joinName,
711                                             final MaterializedInternal<K, VR, KeyValueStore<Bytes, byte[]>> materializedInternal,
712                                             final boolean leftOuter,
713                                             final boolean rightOuter) {
714           Objects.requireNonNull(other, "other can't be null");
715           Objects.requireNonNull(joiner, "joiner can't be null");
716           Objects.requireNonNull(joinName, "joinName can't be null");
717   
718           final NamedInternal renamed = new NamedInternal(joinName);
719           final String joinMergeName = renamed.orElseGenerateWithPrefix(builder, MERGE_NAME);
720           final Set<String> allSourceNodes = ensureCopartitionWith(Collections.singleton((AbstractStream<K, VO>) other));
721   
722           if (leftOuter) {
723               enableSendingOldValues(true);
724           }
725           if (rightOuter) {
726               ((KTableImpl<?, ?, ?>) other).enableSendingOldValues(true);

730           final KTableKTableAbstractJoin<K, VO, V, VR> joinOther;
731   
732           if (!leftOuter) { // inner
733               joinThis = new KTableKTableInnerJoin<>(this, (KTableImpl<K, ?, VO>) other, joiner);
734               joinOther = new KTableKTableInnerJoin<>((KTableImpl<K, ?, VO>) other, this, reverseJoiner(joiner));
735           } else if (!rightOuter) { // left
736               joinThis = new KTableKTableLeftJoin<>(this, (KTableImpl<K, ?, VO>) other, joiner);
737               joinOther = new KTableKTableRightJoin<>((KTableImpl<K, ?, VO>) other, this, reverseJoiner(joiner));
738           } else { // outer
739               joinThis = new KTableKTableOuterJoin<>(this, (KTableImpl<K, ?, VO>) other, joiner);
740               joinOther = new KTableKTableOuterJoin<>((KTableImpl<K, ?, VO>) other, this, reverseJoiner(joiner));
741           }

743           final String joinThisName = renamed.suffixWithOrElseGet("-join-this", builder, JOINTHIS_NAME);
744           final String joinOtherName = renamed.suffixWithOrElseGet("-join-other", builder, JOINOTHER_NAME);
745

748   
749           final Serde<K> keySerde;
750           final Serde<VR> valueSerde;
751           final String queryableStoreName;
752           final StoreBuilder<TimestampedKeyValueStore<K, VR>> storeBuilder;
753   
754           if (materializedInternal != null) {
755               if (materializedInternal.keySerde() == null) {
756                   materializedInternal.withKeySerde(this.keySerde);
757               }
758               keySerde = materializedInternal.keySerde();
759               valueSerde = materializedInternal.valueSerde();
760               queryableStoreName = materializedInternal.storeName();
761               storeBuilder = new TimestampedKeyValueStoreMaterializer<>(materializedInternal).materialize();
762           } else {
763               keySerde = this.keySerde;
764               valueSerde = null;
765               queryableStoreName = null;
766               storeBuilder = null;
767           }

770               KTableKTableJoinNode.<K, V, VO, VR>kTableKTableJoinNodeBuilder()
771                   .withNodeName(joinMergeName)

774                   .withThisJoinSideNodeName(name)
775                   .withOtherJoinSideNodeName(((KTableImpl<?, ?, ?>) other).name)
776                   .withJoinThisStoreNames(valueGetterSupplier().storeNames())
777                   .withJoinOtherStoreNames(((KTableImpl<?, ?, ?>) other).valueGetterSupplier().storeNames())
778                   .withKeySerde(keySerde)
779                   .withValueSerde(valueSerde)
780                   .withQueryableStoreName(queryableStoreName)
781                   .withStoreBuilder(storeBuilder)
782                   .build();
783           builder.addGraphNode(this.graphNode, kTableKTableJoinNode);
784   
785           // we can inherit parent key serde if user do not provide specific overrides
786           return new KTableImpl<K, Change<VR>, VR>(
