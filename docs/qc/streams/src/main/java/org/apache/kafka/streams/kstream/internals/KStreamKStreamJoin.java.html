63       KStreamKStreamJoin(final boolean isLeftSide,
64                          final String otherWindowName,

69                          final TimeTracker sharedTimeTracker) {

73               this.joinBeforeMs = windows.beforeMs;
74               this.joinAfterMs = windows.afterMs;
75               this.joinSpuriousLookBackTimeMs = windows.beforeMs;
76           } else {
77               this.joinBeforeMs = windows.afterMs;

79               this.joinSpuriousLookBackTimeMs = windows.afterMs;

83           this.joiner = joiner;
84           this.outer = outer;

94       private class KStreamKStreamJoinProcessor extends ContextualProcessor<K, V1, K, VOut> {
95           private WindowStore<K, V2> otherWindowStore;
96           private Sensor droppedRecordsSensor;
97           private Optional<KeyValueStore<TimestampedKeyAndJoinSide<K>, LeftOrRightValue<V1, V2>>> outerJoinStore = Optional.empty();
98           private InternalProcessorContext<K, VOut> internalProcessorContext;
99   
100           @Override
101           public void init(final ProcessorContext<K, VOut> context) {
102               super.init(context);
103               internalProcessorContext = (InternalProcessorContext<K, VOut>) context;
104   
105               final StreamsMetricsImpl metrics = (StreamsMetricsImpl) context.metrics();
106               droppedRecordsSensor = droppedRecordsSensor(Thread.currentThread().getName(), context.taskId().toString(), metrics);
107               otherWindowStore = context.getStateStore(otherWindowName);
108   
109               if (enableSpuriousResultFix) {
110                   outerJoinStore = outerJoinWindowName.map(context::getStateStore);
111   
112                   sharedTimeTracker.setEmitInterval(
113                       StreamsConfig.InternalConfig.getLong(
114                           context.appConfigs(),
115                           EMIT_INTERVAL_MS_KSTREAMS_OUTER_JOIN_SPURIOUS_RESULTS_FIX,
116                           1000L
117                       )
118                   );
119               }
120           }

122           @SuppressWarnings("unchecked")
123           @Override
124           public void process(final Record<K, V1> record) {
125               // we do join iff keys are equal, thus, if key is null we cannot join and just ignore the record
126               //
127               // we also ignore the record if value is null, because in a key-value data model a null-value indicates
128               // an empty message (ie, there is nothing to be joined) -- this contrast SQL NULL semantics
129               // furthermore, on left/outer joins 'null' in ValueJoiner#apply() indicates a missing record --
130               // thus, to be consistent and to avoid ambiguous null semantics, null values are ignored
131               if (record.key() == null || record.value() == null) {
132                   if (context().recordMetadata().isPresent()) {
133                       final RecordMetadata recordMetadata = context().recordMetadata().get();
134                       LOG.warn(
135                           "Skipping record due to null key or value. "
136                               + "topic=[{}] partition=[{}] offset=[{}]",
137                           recordMetadata.topic(), recordMetadata.partition(), recordMetadata.offset()
138                       );
139                   } else {
140                       LOG.warn(
141                           "Skipping record due to null key or value. Topic, partition, and offset not known."
142                       );
143                   }
144                   droppedRecordsSensor.record();
145                   return;
146               }
147   
148               boolean needOuterJoin = outer;
149   
150               final long inputRecordTimestamp = record.timestamp();
151               final long timeFrom = Math.max(0L, inputRecordTimestamp - joinBeforeMs);
152               final long timeTo = Math.max(0L, inputRecordTimestamp + joinAfterMs);
153   
154               sharedTimeTracker.advanceStreamTime(inputRecordTimestamp);
155   
156               // Emit all non-joined records which window has closed
157               if (inputRecordTimestamp == sharedTimeTracker.streamTime) {
158                   outerJoinStore.ifPresent(store -> emitNonJoinedOuterRecords(store, record));
159               }
160   
161               try (final WindowStoreIterator<V2> iter = otherWindowStore.fetch(record.key(), timeFrom, timeTo)) {
162                   while (iter.hasNext()) {
163                       needOuterJoin = false;
164                       final KeyValue<Long, V2> otherRecord = iter.next();
165                       final long otherRecordTimestamp = otherRecord.key;
166   
167                       outerJoinStore.ifPresent(store -> {
168                           // use putIfAbsent to first read and see if there's any values for the key,
169                           // if yes delete the key, otherwise do not issue a put;
170                           // we may delete some values with the same key early but since we are going
171                           // range over all values of the same key even after failure, since the other window-store
172                           // is only cleaned up by stream time, so this is okay for at-least-once.
173                           store.putIfAbsent(TimestampedKeyAndJoinSide.make(!isLeftSide, record.key(), otherRecordTimestamp), null);
174                       });
175   
176                       context().forward(
177                           record.withValue(joiner.apply(record.key(), record.value(), otherRecord.value))
178                                  .withTimestamp(Math.max(inputRecordTimestamp, otherRecordTimestamp)));
179                   }
180   
181                   if (needOuterJoin) {
182                       // The maxStreamTime contains the max time observed in both sides of the join.
183                       // Having access to the time observed in the other join side fixes the following
184                       // problem:
185                       //
186                       // Say we have a window size of 5 seconds
187                       //  1. A non-joined record wth time T10 is seen in the left-topic (maxLeftStreamTime: 10)
188                       //     The record is not processed yet, and is added to the outer-join store
189                       //  2. A non-joined record with time T2 is seen in the right-topic (maxRightStreamTime: 2)
190                       //     The record is not processed yet, and is added to the outer-join store
191                       //  3. A joined record with time T11 is seen in the left-topic (maxLeftStreamTime: 11)
192                       //     It is time to look at the expired records. T10 and T2 should be emitted, but
193                       //     because T2 was late, then it is not fetched by the window store, so it is not processed
194                       //
195                       // See KStreamKStreamLeftJoinTest.testLowerWindowBound() tests
196                       //
197                       // This condition below allows us to process the out-of-order records without the need
198                       // to hold it in the temporary outer store
199                       if (!outerJoinStore.isPresent() || timeTo < sharedTimeTracker.streamTime) {
200                           context().forward(record.withValue(joiner.apply(record.key(), record.value(), null)));
201                       } else {
202                           sharedTimeTracker.updatedMinTime(inputRecordTimestamp);
203                           outerJoinStore.ifPresent(store -> store.put(
204                               TimestampedKeyAndJoinSide.make(isLeftSide, record.key(), inputRecordTimestamp),
205                               LeftOrRightValue.make(isLeftSide, record.value())));
206                       }
207                   }
208               }
209           }
210   
211           @SuppressWarnings("unchecked")
212           private void emitNonJoinedOuterRecords(
213               final KeyValueStore<TimestampedKeyAndJoinSide<K>, LeftOrRightValue<V1, V2>> store,
214               final Record<K, V1> record) {
215   
216               // calling `store.all()` creates an iterator what is an expensive operation on RocksDB;
217               // to reduce runtime cost, we try to avoid paying those cost
218   
219               // only try to emit left/outer join results if there _might_ be any result records
220               if (sharedTimeTracker.minTime >= sharedTimeTracker.streamTime - joinSpuriousLookBackTimeMs - joinGraceMs) {
221                   return;
222               }
223               // throttle the emit frequency to a (configurable) interval;
224               // we use processing time to decouple from data properties,
225               // as throttling is a non-functional performance optimization
226               if (internalProcessorContext.currentSystemTimeMs() < sharedTimeTracker.nextTimeToEmit) {
227                   return;
228               }
229               if (sharedTimeTracker.nextTimeToEmit == 0) {
230                   sharedTimeTracker.nextTimeToEmit = internalProcessorContext.currentSystemTimeMs();
231               }
232               sharedTimeTracker.advanceNextTimeToEmit();
233   
234               // reset to MAX_VALUE in case the store is empty
235               sharedTimeTracker.minTime = Long.MAX_VALUE;
236   
237               try (final KeyValueIterator<TimestampedKeyAndJoinSide<K>, LeftOrRightValue<V1, V2>> it = store.all()) {
238                   TimestampedKeyAndJoinSide<K> prevKey = null;
239   
240                   while (it.hasNext()) {
241                       final KeyValue<TimestampedKeyAndJoinSide<K>, LeftOrRightValue<V1, V2>> next = it.next();
242                       final TimestampedKeyAndJoinSide<K> timestampedKeyAndJoinSide = next.key;
243                       final LeftOrRightValue<V1, V2> value = next.value;
244                       final K key = timestampedKeyAndJoinSide.getKey();
245                       final long timestamp = timestampedKeyAndJoinSide.getTimestamp();
246                       sharedTimeTracker.minTime = timestamp;
247   
248                       // Skip next records if window has not closed
249                       if (timestamp + joinAfterMs + joinGraceMs >= sharedTimeTracker.streamTime) {
250                           break;
251                       }
252   
253                       final VOut nullJoinedValue;
254                       if (isLeftSide) {
255                           nullJoinedValue = joiner.apply(key,
256                                   value.getLeftValue(),
257                                   value.getRightValue());
258                       } else {
259                           nullJoinedValue = joiner.apply(key,
260                                   (V1) value.getRightValue(),
261                                   (V2) value.getLeftValue());
262                       }
263   
264                       context().forward(
265                           record.withKey(key).withValue(nullJoinedValue).withTimestamp(timestamp)
266                       );
267   
268                       if (prevKey != null && !prevKey.equals(timestampedKeyAndJoinSide)) {
269                           // blind-delete the previous key from the outer window store now it is emitted;
270                           // we do this because this delete would remove the whole list of values of the same key,
271                           // and hence if we delete eagerly and then fail, we would miss emitting join results of the later
272                           // values in the list.
273                           // we do not use delete() calls since it would incur extra get()
274                           store.put(prevKey, null);
275                       }
276   
277                       prevKey = timestampedKeyAndJoinSide;
278                   }
279   
280                   // at the end of the iteration, we need to delete the last key
281                   if (prevKey != null) {
282                       store.put(prevKey, null);
283                   }
284               }
285           }
286       }
287   }
