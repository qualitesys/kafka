313       @SuppressWarnings("unchecked")
314       private void closeTopology() {
315           log.trace("Closing processor topology");
316   
317           // close the processors
318           // make sure close() is called for each node even when there is a RuntimeException
319           RuntimeException exception = null;
320           for (final ProcessorNode<?, ?, ?, ?> node : topology.processors()) {
321               processorContext.setCurrentNode(node);
322               try {
323                   node.close();
324               } catch (final RuntimeException e) {
325                   exception = e;
326               } finally {
327                   processorContext.setCurrentNode(null);
328               }
329           }
330   
331           if (exception != null) {
332               throw exception;
333           }
334       }

336       /**
337        * <pre>
338        * - resume the task
339        * </pre>
340        */
341       @Override
342       public void resume() {
343           switch (state()) {
344               case CREATED:
345               case RUNNING:
346               case RESTORING:
347                   // no need to do anything, just let them continue running / restoring / closing
348                   log.trace("Skip resuming since state is {}", state());
349                   break;
350   
351               case SUSPENDED:
352                   // just transit the state without any logical changes: suspended and restoring states
353                   // are not actually any different for inner modules
354   
355                   // Deleting checkpoint file before transition to RESTORING state (KAFKA-10362)
356                   try {
357                       stateMgr.deleteCheckPointFileIfEOSEnabled();
358                       log.debug("Deleted check point file upon resuming with EOS enabled");
359                   } catch (final IOException ioe) {
360                       log.error("Encountered error while deleting the checkpoint file due to this exception", ioe);
361                   }
362   
363                   transitionTo(State.RESTORING);
364                   log.info("Resumed to restoring state");
365   
366                   break;
367   
368               case CLOSED:
369                   throw new IllegalStateException("Illegal state " + state() + " while resuming active task " + id);
370   
371               default:
372                   throw new IllegalStateException("Unknown state " + state() + " while resuming active task " + id);
373           }
374           timeCurrentIdlingStarted = Optional.empty();
375       }

377       /**
378        * @throws StreamsException fatal error that should cause the thread to die
379        * @throws TaskMigratedException recoverable error that would cause the task to be removed
380        * @return offsets that should be committed for this task
381        */
382       @Override
383       public Map<TopicPartition, OffsetAndMetadata> prepareCommit() {
384           switch (state()) {
385               case CREATED:
386               case RESTORING:
387               case RUNNING:
388               case SUSPENDED:
389                   // the commitNeeded flag just indicates whether we have reached RUNNING and processed any new data,
390                   // so it only indicates whether the record collector should be flushed or not, whereas the state
391                   // manager should always be flushed; either there is newly restored data or the flush will be a no-op
392                   if (commitNeeded) {
393                       // we need to flush the store caches before flushing the record collector since it may cause some
394                       // cached records to be processed and hence generate more records to be sent out
395                       //
396                       // TODO: this should be removed after we decouple caching with emitting
397                       stateMgr.flushCache();
398                       recordCollector.flush();
399                       hasPendingTxCommit = eosEnabled;
400   
401                       log.debug("Prepared {} task for committing", state());
402                       return committableOffsetsAndMetadata();
403                   } else {
404                       log.debug("Skipped preparing {} task for commit since there is nothing to commit", state());
405                       return Collections.emptyMap();
406                   }
407   
408               case CLOSED:
409                   throw new IllegalStateException("Illegal state " + state() + " while preparing active task " + id + " for committing");
410   
411               default:
412                   throw new IllegalStateException("Unknown state " + state() + " while preparing active task " + id + " for committing");
413           }
414       }

416       private Long findOffset(final TopicPartition partition) {
417           Long offset = partitionGroup.headRecordOffset(partition);
418           if (offset == null) {
419               try {
420                   offset = mainConsumer.position(partition);
421               } catch (final TimeoutException error) {
422                   // the `consumer.position()` call should never block, because we know that we did process data
423                   // for the requested partition and thus the consumer should have a valid local position
424                   // that it can return immediately
425   
426                   // hence, a `TimeoutException` indicates a bug and thus we rethrow it as fatal `IllegalStateException`
427                   throw new IllegalStateException(error);
428               } catch (final KafkaException fatal) {
429                   throw new StreamsException(fatal);
430               }
431           }
432           return offset;
433       }

435       private Map<TopicPartition, OffsetAndMetadata> committableOffsetsAndMetadata() {
436           final Map<TopicPartition, OffsetAndMetadata> committableOffsets;
437   
438           switch (state()) {
439               case CREATED:
440               case RESTORING:
441                   committableOffsets = Collections.emptyMap();
442   
443                   break;
444   
445               case RUNNING:
446               case SUSPENDED:
447                   final Map<TopicPartition, Long> partitionTimes = extractPartitionTimes();
448   
449                   // If there's processor metadata to be committed. We need to commit them to all
450                   // input partitions
451                   final Set<TopicPartition> partitionsNeedCommit = processorContext.getProcessorMetadata().needsCommit() ?
452                       inputPartitions() : consumedOffsets.keySet();
453                   committableOffsets = new HashMap<>(partitionsNeedCommit.size());
454   
455                   for (final TopicPartition partition : partitionsNeedCommit) {
456                       final Long offset = findOffset(partition);
457                       final long partitionTime = partitionTimes.get(partition);
458                       committableOffsets.put(partition, new OffsetAndMetadata(offset,
459                           new TopicPartitionMetadata(partitionTime, processorContext.getProcessorMetadata()).encode()));
460                   }
461                   break;
462   
463               case CLOSED:
464                   throw new IllegalStateException("Illegal state " + state() + " while getting committable offsets for active task " + id);
465   
466               default:
467                   throw new IllegalStateException("Unknown state " + state() + " while post committing active task " + id);
468           }
469   
470           return committableOffsets;
471       }

473       @Override
474       public void postCommit(final boolean enforceCheckpoint) {
475           switch (state()) {
476               case CREATED:
477                   // We should never write a checkpoint for a CREATED task as we may overwrite an existing checkpoint
478                   // with empty uninitialized offsets
479                   log.debug("Skipped writing checkpoint for {} task", state());
480   
481                   break;
482

668       /**
669        * An active task is processable if its buffer contains data for all of its input
670        * source topic partitions, or if it is enforced to be processable.
671        */
672       public boolean isProcessable(final long wallClockTime) {
673           if (state() == State.CLOSED) {
674               // a task is only closing / closed when 1) task manager is closing, 2) a rebalance is undergoing;
675               // in either case we can just log it and move on without notifying the thread since the consumer
676               // would soon be updated to not return any records for this task anymore.
677               log.info("Stream task {} is already in {} state, skip processing it.", id(), state());
678   
679               return false;
680           }
681   
682           if (hasPendingTxCommit) {
683               // if the task has a pending TX commit, we should just retry the commit but not process any records
684               // thus, the task is not processable, even if there is available data in the record queue
685               return false;
686           }
687           final boolean readyToProcess = partitionGroup.readyToProcess(wallClockTime);
688           if (!readyToProcess) {
689               if (!timeCurrentIdlingStarted.isPresent()) {
690                   timeCurrentIdlingStarted = Optional.of(wallClockTime);
691               }
692           } else {
693               timeCurrentIdlingStarted = Optional.empty();
694           }
695           return readyToProcess;
696       }
697   
698       /**
699        * Process one record.
700        *
701        * @return true if this method processes a record, false if it does not process a record.
702        * @throws TaskMigratedException if the task producer got fenced (EOS only)
703        */
704       @SuppressWarnings("unchecked")
705       public boolean process(final long wallClockTime) {
706           if (record == null) {
707               if (!isProcessable(wallClockTime)) {
708                   return false;
709               }
710   
711               // get the next record to process
712               record = partitionGroup.nextRecord(recordInfo, wallClockTime);
713   
714               // if there is no record to process, return immediately
715               if (record == null) {
716                   return false;
717               }
718           }
719   
720           try {
721               final TopicPartition partition = recordInfo.partition();
722   
723               if (!(record instanceof CorruptedRecord)) {
724                   doProcess(wallClockTime);
725               }
726   
727               // update the consumed offset map after processing is done
728               consumedOffsets.put(partition, record.offset());
729               commitNeeded = true;
730   
731               // after processing this record, if its partition queue's buffered size has been
732               // decreased to the threshold, we can then resume the consumption on this partition
733               if (recordInfo.queue().size() == maxBufferedSize) {
734                   mainConsumer.resume(singleton(partition));
735               }
736   
737               record = null;
738           } catch (final TimeoutException timeoutException) {
739               if (!eosEnabled) {
740                   throw timeoutException;
741               } else {
742                   record = null;
743                   throw new TaskCorruptedException(Collections.singleton(id));
744               }
745           } catch (final StreamsException exception) {
746               record = null;
747               throw exception;
748           } catch (final RuntimeException e) {
749               final StreamsException error = new StreamsException(
750                   String.format(
751                       "Exception caught in process. taskId=%s, processor=%s, topic=%s, partition=%d, offset=%d, stacktrace=%s",
752                       id(),
753                       processorContext.currentNode().name(),
754                       record.topic(),
755                       record.partition(),
756                       record.offset(),
757                       getStacktraceString(e)
758                   ),
759                   e
760               );
761               record = null;
762   
763               throw error;
764           } finally {
765               processorContext.setCurrentNode(null);
766           }
767   
768           return true;
769       }

774           final ProcessorNode<Object, Object, Object, Object> currNode = (ProcessorNode<Object, Object, Object, Object>) recordInfo.node();
