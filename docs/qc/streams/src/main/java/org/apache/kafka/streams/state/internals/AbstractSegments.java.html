38   abstract class AbstractSegments<S extends Segment> implements Segments<S> {
39       private static final Logger log = LoggerFactory.getLogger(AbstractSegments.class);
40   
41       final TreeMap<Long, S> segments = new TreeMap<>();
42       final String name;
43       private final long retentionPeriod;
44       private final long segmentInterval;
45       private final SimpleDateFormat formatter;
46   
47       AbstractSegments(final String name, final long retentionPeriod, final long segmentInterval) {
48           this.name = name;
49           this.segmentInterval = segmentInterval;
50           this.retentionPeriod = retentionPeriod;
51           // Create a date formatter. Formatted timestamps are used as segment name suffixes
52           this.formatter = new SimpleDateFormat("yyyyMMddHHmm");
53           this.formatter.setTimeZone(new SimpleTimeZone(0, "UTC"));
54       }

56       @Override
57       public long segmentId(final long timestamp) {
58           return timestamp / segmentInterval;
59       }
60   
61       @Override
62       public String segmentName(final long segmentId) {
63           // (1) previous format used - as a separator so if this changes in the future
64           // then we should use something different.
65           // (2) previous format used : as a separator (which did break KafkaStreams on Windows OS)
66           // so if this changes in the future then we should use something different.
67           return name + "." + segmentId * segmentInterval;
68       }
69   
70       @Override
71       public S getSegmentForTimestamp(final long timestamp) {
72           return segments.get(segmentId(timestamp));
73       }
74   
75       @Override
76       public S getOrCreateSegmentIfLive(final long segmentId,
77                                         final ProcessorContext context,
78                                         final long streamTime) {
79           final long minLiveTimestamp = streamTime - retentionPeriod;
80           final long minLiveSegment = segmentId(minLiveTimestamp);
81   
82           final S toReturn;
83           if (segmentId >= minLiveSegment) {
84               // The segment is live. get it, ensure it's open, and return it.
85               toReturn = getOrCreateSegment(segmentId, context);
86           } else {
87               toReturn = null;
88           }
89   
90           cleanupEarlierThan(minLiveSegment);
91           return toReturn;
92       }
93   
94       @Override
95       public void openExisting(final ProcessorContext context, final long streamTime) {
96           try {
97               final File dir = new File(context.stateDir(), name);
98               if (dir.exists()) {
99                   final String[] list = dir.list();
100                   if (list != null) {
101                       Arrays.stream(list)
102                               .map(segment -> segmentIdFromSegmentName(segment, dir))
103                               .sorted() // open segments in the id order
104                               .filter(segmentId -> segmentId >= 0)
105                               .forEach(segmentId -> getOrCreateSegment(segmentId, context));
106                   }
107               } else {
108                   if (!dir.mkdir()) {
109                       throw new ProcessorStateException(String.format("dir %s doesn't exist and cannot be created for segments %s", dir, name));
110                   }
111               }
112           } catch (final Exception ex) {
113               // ignore
114           }
115   
116           final long minLiveSegment = segmentId(streamTime - retentionPeriod);
117           cleanupEarlierThan(minLiveSegment);
118       }
119   
120       @Override
121       public List<S> segments(final long timeFrom, final long timeTo, final boolean forward) {
122           final List<S> result = new ArrayList<>();
123           final NavigableMap<Long, S> segmentsInRange;
124           if (forward) {
125               segmentsInRange = segments.subMap(
126                   segmentId(timeFrom), true,
127                   segmentId(timeTo), true
128               );
129           } else {
130               segmentsInRange = segments.subMap(
131                   segmentId(timeFrom), true,
132                   segmentId(timeTo), true
133               ).descendingMap();
134           }
135           for (final S segment : segmentsInRange.values()) {
136               if (segment.isOpen()) {
137                   result.add(segment);
138               }
139           }
140           return result;
141       }
142   
143       @Override
144       public List<S> allSegments(final boolean forward) {
145           final List<S> result = new ArrayList<>();
146           final Collection<S> values;
147           if (forward) {
148               values = segments.values();
149           } else {
150               values = segments.descendingMap().values();
151           }
152           for (final S segment : values) {
153               if (segment.isOpen()) {
154                   result.add(segment);
155               }
156           }
157           return result;
158       }
159   
160       @Override
161       public void flush() {
162           for (final S segment : segments.values()) {
163               segment.flush();
164           }
165       }
166   
167       @Override
168       public void close() {
169           for (final S segment : segments.values()) {
170               segment.close();
171           }
172           segments.clear();
173       }
174   
175       private void cleanupEarlierThan(final long minLiveSegment) {
176           final Iterator<Map.Entry<Long, S>> toRemove =
177               segments.headMap(minLiveSegment, false).entrySet().iterator();
178   
179           while (toRemove.hasNext()) {
180               final Map.Entry<Long, S> next = toRemove.next();
181               toRemove.remove();
182               final S segment = next.getValue();
183               segment.close();
184               try {
185                   segment.destroy();
186               } catch (final IOException e) {
187                   log.error("Error destroying {}", segment, e);
188               }
189           }
190       }

192       private long segmentIdFromSegmentName(final String segmentName,
193                                             final File parent) {
194           final int segmentSeparatorIndex = name.length();
195           final char segmentSeparator = segmentName.charAt(segmentSeparatorIndex);
196           final String segmentIdString = segmentName.substring(segmentSeparatorIndex + 1);
197           final long segmentId;
198   
199           // old style segment name with date
200           if (segmentSeparator == '-') {
201               try {
202                   segmentId = formatter.parse(segmentIdString).getTime() / segmentInterval;
203               } catch (final ParseException e) {
204                   log.warn("Unable to parse segmentName {} to a date. This segment will be skipped", segmentName);
205                   return -1L;
206               }
207               renameSegmentFile(parent, segmentName, segmentId);
208           } else {
209               // for both new formats (with : or .) parse segment ID identically
210               try {
211                   segmentId = Long.parseLong(segmentIdString) / segmentInterval;
212               } catch (final NumberFormatException e) {
213                   throw new ProcessorStateException("Unable to parse segment id as long from segmentName: " + segmentName);
214               }
215   
216               // intermediate segment name with : breaks KafkaStreams on Windows OS -> rename segment file to new name with .
217               if (segmentSeparator == ':') {
218                   renameSegmentFile(parent, segmentName, segmentId);
219               }
220           }
221   
222           return segmentId;
223   
224       }
225   
226       private void renameSegmentFile(final File parent,
227                                      final String segmentName,
228                                      final long segmentId) {
229           final File newName = new File(parent, segmentName(segmentId));
230           final File oldName = new File(parent, segmentName);
231           if (!oldName.renameTo(newName)) {
232               throw new ProcessorStateException("Unable to rename old style segment from: "
233                   + oldName
234                   + " to new name: "
235                   + newName);
236           }
237       }
