17   package org.apache.kafka.streams.errors;
18   
19   
20   import org.apache.kafka.clients.consumer.ConsumerRecord;
21   import org.apache.kafka.common.Configurable;
22   import org.apache.kafka.streams.processor.ProcessorContext;
23   
24   /**
25    * Interface that specifies how an exception from source node deserialization
26    * (e.g., reading from Kafka) should be handled.
27    */
28   public interface DeserializationExceptionHandler extends Configurable {
29   
30       /**
31        * Inspect a record and the exception received.
32        * <p>
33        * Note, that the passed in {@link ProcessorContext} only allows to access metadata like the task ID.
34        * However, it cannot be used to emit records via {@link ProcessorContext#forward(Object, Object)};
35        * calling {@code forward()} (and some other methods) would result in a runtime exception.
36        *
37        * @param context processor context
38        * @param record record that failed deserialization
39        * @param exception the actual exception
40        */
41       @SuppressWarnings("deprecation") // Old PAPI. Needs to be migrated.
42       DeserializationHandlerResponse handle(final ProcessorContext context,
43                                             final ConsumerRecord<byte[], byte[]> record,
44                                             final Exception exception);
45   
46       /**
47        * Enumeration that describes the response from the exception handler.
48        */
49       enum DeserializationHandlerResponse {
50           /* continue with processing */
51           CONTINUE(0, "CONTINUE"),
52           /* fail the processing and stop */
53           FAIL(1, "FAIL");
54   
55           /** an english description of the api--this is for debugging and can change */
56           public final String name;
57   
58           /** the permanent and immutable id of an API--this can't change ever */
59           public final int id;
60   
61           DeserializationHandlerResponse(final int id, final String name) {
62               this.id = id;
63               this.name = name;
64           }
65       }
66   
67   }
