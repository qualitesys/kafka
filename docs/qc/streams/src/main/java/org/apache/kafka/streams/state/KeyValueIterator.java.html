17   package org.apache.kafka.streams.state;
18   
19   import org.apache.kafka.streams.KeyValue;
20   
21   import java.io.Closeable;
22   import java.util.Iterator;
23   
24   /**
25    * Iterator interface of {@link KeyValue}.
26    * <p>
27    * Users must call its {@code close} method explicitly upon completeness to release resources,
28    * or use try-with-resources statement (available since JDK7) for this {@link Closeable} class.
29    * Note that {@code remove()} is not supported.
30    *
31    * @param <K> Type of keys
32    * @param <V> Type of values
33    */
34   public interface KeyValueIterator<K, V> extends Iterator<KeyValue<K, V>>, Closeable {
35   
36       @Override
37       void close();
38   
39       /**
40        * Peek at the next key without advancing the iterator
41        * @return the key of the next value that would be returned from the next call to next
42        */
43       K peekNextKey();
44   }
