167       /**
168        * @throws TaskMigratedException
169        */
170       boolean handleCorruption(final Set<TaskId> corruptedTasks) {
171           final Set<Task> corruptedActiveTasks = new HashSet<>();
172           final Set<Task> corruptedStandbyTasks = new HashSet<>();
173   
174           for (final TaskId taskId : corruptedTasks) {
175               final Task task = tasks.task(taskId);
176               if (task.isActive()) {
177                   corruptedActiveTasks.add(task);
178               } else {
179                   corruptedStandbyTasks.add(task);
180               }
181           }
182   
183           // Make sure to clean up any corrupted standby tasks in their entirety before committing
184           // since TaskMigrated can be thrown and the resulting handleLostAll will only clean up active tasks
185           closeDirtyAndRevive(corruptedStandbyTasks, true);
186   
187           // We need to commit before closing the corrupted active tasks since this will force the ongoing txn to abort
188           try {
189               final Collection<Task> tasksToCommit = tasks()
190                   .values()
191                   .stream()
192                   .filter(t -> t.state() == Task.State.RUNNING || t.state() == Task.State.RESTORING)
193                   .filter(t -> !corruptedTasks.contains(t.id()))
194                   .collect(Collectors.toSet());
195               commitTasksAndMaybeUpdateCommittableOffsets(tasksToCommit, new HashMap<>());
196           } catch (final TaskCorruptedException e) {
197               log.info("Some additional tasks were found corrupted while trying to commit, these will be added to the " +
198                            "tasks to clean and revive: {}", e.corruptedTasks());
199               corruptedActiveTasks.addAll(tasks.tasks(e.corruptedTasks()));
200           } catch (final TimeoutException e) {
201               log.info("Hit TimeoutException when committing all non-corrupted tasks, these will be closed and revived");
202               final Collection<Task> uncorruptedTasks = new HashSet<>(tasks.activeTasks());
203               uncorruptedTasks.removeAll(corruptedActiveTasks);
204               // Those tasks which just timed out can just be closed dirty without marking changelogs as corrupted
205               closeDirtyAndRevive(uncorruptedTasks, false);
206           }
207   
208           closeDirtyAndRevive(corruptedActiveTasks, true);
209           return !corruptedActiveTasks.isEmpty();
210       }

212       private void closeDirtyAndRevive(final Collection<Task> taskWithChangelogs, final boolean markAsCorrupted) {
213           for (final Task task : taskWithChangelogs) {
214               final Collection<TopicPartition> corruptedPartitions = task.changelogPartitions();
215   
216               // mark corrupted partitions to not be checkpointed, and then close the task as dirty
217               if (markAsCorrupted) {
218                   task.markChangelogAsCorrupted(corruptedPartitions);
219               }
220   
221               try {
222                   // we do not need to take the returned offsets since we are not going to commit anyways;
223                   // this call is only used for active tasks to flush the cache before suspending and
224                   // closing the topology
225                   task.prepareCommit();
226               } catch (final RuntimeException swallow) {
227                   log.error("Error flushing cache for corrupted task {} ", task.id(), swallow);
228               }
229   
230               try {
231                   task.suspend();
232   
233                   // we need to enforce a checkpoint that removes the corrupted partitions
234                   if (markAsCorrupted) {
235                       task.postCommit(true);
236                   }
237               } catch (final RuntimeException swallow) {
238                   log.error("Error suspending corrupted task {} ", task.id(), swallow);
239               }
240               task.closeDirty();
241   
242               // For active tasks pause their input partitions so we won't poll any more records
243               // for this task until it has been re-initialized;
244               // Note, closeDirty already clears the partition-group for the task.
245               if (task.isActive()) {
246                   final Set<TopicPartition> currentAssignment = mainConsumer.assignment();
247                   final Set<TopicPartition> taskInputPartitions = task.inputPartitions();
248                   final Set<TopicPartition> assignedToPauseAndReset =
249                       intersection(HashSet::new, currentAssignment, taskInputPartitions);
250                   if (!assignedToPauseAndReset.equals(taskInputPartitions)) {
251                       log.warn(
252                           "Expected the current consumer assignment {} to contain the input partitions {}. " +
253                               "Will proceed to recover.",
254                           currentAssignment,
255                           taskInputPartitions
256                       );
257                   }
258   
259                   task.addPartitionsForOffsetReset(assignedToPauseAndReset);
260               }
261               task.revive();
262           }
263       }

265       /**
266        * @throws TaskMigratedException if the task producer got fenced (EOS only)
267        * @throws StreamsException fatal error while creating / initializing the task
268        *
269        * public for upgrade testing only
270        */
271       public void handleAssignment(final Map<TaskId, Set<TopicPartition>> activeTasks,
272                                    final Map<TaskId, Set<TopicPartition>> standbyTasks) {
273           log.info("Handle new assignment with:\n" +
274                        "\tNew active tasks: {}\n" +
275                        "\tNew standby tasks: {}\n" +
276                        "\tExisting active tasks: {}\n" +
277                        "\tExisting standby tasks: {}",
278                    activeTasks.keySet(), standbyTasks.keySet(), activeTaskIds(), standbyTaskIds());
279   
280           topologyMetadata.addSubscribedTopicsFromAssignment(
281               activeTasks.values().stream().flatMap(Collection::stream).collect(Collectors.toList()),
282               logPrefix
283           );
284   
285           final LinkedHashMap<TaskId, RuntimeException> taskCloseExceptions = new LinkedHashMap<>();
286           final Map<TaskId, Set<TopicPartition>> activeTasksToCreate = new HashMap<>(activeTasks);
287           final Map<TaskId, Set<TopicPartition>> standbyTasksToCreate = new HashMap<>(standbyTasks);
288           final Comparator<Task> byId = Comparator.comparing(Task::id);
289           final Set<Task> tasksToRecycle = new TreeSet<>(byId);
290           final Set<Task> tasksToCloseClean = new TreeSet<>(byId);
291           final Set<Task> tasksToCloseDirty = new TreeSet<>(byId);
292   
293           // first rectify all existing tasks
294           for (final Task task : tasks.allTasks()) {
295               if (activeTasks.containsKey(task.id()) && task.isActive()) {
296                   tasks.updateInputPartitionsAndResume(task, activeTasks.get(task.id()));
297                   activeTasksToCreate.remove(task.id());
298               } else if (standbyTasks.containsKey(task.id()) && !task.isActive()) {
299                   tasks.updateInputPartitionsAndResume(task, standbyTasks.get(task.id()));
300                   standbyTasksToCreate.remove(task.id());
301               } else if (activeTasks.containsKey(task.id()) || standbyTasks.containsKey(task.id())) {
302                   // check for tasks that were owned previously but have changed active/standby status
303                   tasksToRecycle.add(task);
304               } else {
305                   tasksToCloseClean.add(task);
306               }
307           }
308   
309           // close and recycle those tasks
310           handleCloseAndRecycle(
311               tasksToRecycle,
312               tasksToCloseClean,
313               tasksToCloseDirty,
314               activeTasksToCreate,
315               standbyTasksToCreate,
316               taskCloseExceptions
317           );
318   
319           if (!taskCloseExceptions.isEmpty()) {
320               log.error("Hit exceptions while closing / recycling tasks: {}", taskCloseExceptions);
321   
322               for (final Map.Entry<TaskId, RuntimeException> entry : taskCloseExceptions.entrySet()) {
323                   if (!(entry.getValue() instanceof TaskMigratedException)) {
324                       final TaskId taskId = entry.getKey();
325                       final RuntimeException exception = entry.getValue();
326                       if (exception instanceof StreamsException) {
327                           ((StreamsException) exception).setTaskId(taskId);
328                           throw exception;
329                       } else if (exception instanceof KafkaException) {
330                           throw new StreamsException(exception, taskId);
331                       } else {
332                           throw new StreamsException(
333                               "Unexpected failure to close " + taskCloseExceptions.size() +
334                                   " task(s) [" + taskCloseExceptions.keySet() + "]. " +
335                                   "First unexpected exception (for task " + taskId + ") follows.",
336                               exception,
337                               taskId
338                           );
339                       }
340                   }
341               }
342   
343               // If all exceptions are task-migrated, we would just throw the first one. No need to wrap with a
344               // StreamsException since TaskMigrated is handled explicitly by the StreamThread
345               final Map.Entry<TaskId, RuntimeException> first = taskCloseExceptions.entrySet().iterator().next();
346               throw first.getValue();
347           }
348   
349           tasks.handleNewAssignmentAndCreateTasks(activeTasksToCreate, standbyTasksToCreate, activeTasks.keySet(), standbyTasks.keySet());
350       }

352       private void handleCloseAndRecycle(final Set<Task> tasksToRecycle,
353                                          final Set<Task> tasksToCloseClean,
354                                          final Set<Task> tasksToCloseDirty,
355                                          final Map<TaskId, Set<TopicPartition>> activeTasksToCreate,
356                                          final Map<TaskId, Set<TopicPartition>> standbyTasksToCreate,
357                                          final LinkedHashMap<TaskId, RuntimeException> taskCloseExceptions) {
358           if (!tasksToCloseDirty.isEmpty()) {
359               throw new IllegalArgumentException("Tasks to close-dirty should be empty");
360           }
361   
362           // for all tasks to close or recycle, we should first write a checkpoint as in post-commit
363           final List<Task> tasksToCheckpoint = new ArrayList<>(tasksToCloseClean);
364           tasksToCheckpoint.addAll(tasksToRecycle);
365           for (final Task task : tasksToCheckpoint) {
366               try {
367                   // Note that we are not actually committing here but just check if we need to write checkpoint file:
368                   // 1) for active tasks prepareCommit should return empty if it has committed during suspension successfully,
369                   //    and their changelog positions should not change at all postCommit would not write the checkpoint again.
370                   // 2) for standby tasks prepareCommit should always return empty, and then in postCommit we would probably
371                   //    write the checkpoint file.
372                   final Map<TopicPartition, OffsetAndMetadata> offsets = task.prepareCommit();
373                   if (!offsets.isEmpty()) {
374                       log.error("Task {} should have been committed when it was suspended, but it reports non-empty " +
375                                       "offsets {} to commit; this means it failed during last commit and hence should be closed dirty",
376                               task.id(), offsets);
377   
378                       tasksToCloseDirty.add(task);
379                   } else if (!task.isActive()) {
380                       // For standby tasks, always try to first suspend before committing (checkpointing) it;
381                       // Since standby tasks do not actually need to commit offsets but only need to
382                       // flush / checkpoint state stores, so we only need to call postCommit here.
383                       task.suspend();
384   
385                       task.postCommit(true);
386                   }
387               } catch (final RuntimeException e) {
388                   final String uncleanMessage = String.format(
389                           "Failed to checkpoint task %s. Attempting to close remaining tasks before re-throwing:",
390                           task.id());
391                   log.error(uncleanMessage, e);
392                   taskCloseExceptions.putIfAbsent(task.id(), e);
393                   // We've already recorded the exception (which is the point of clean).
394                   // Now, we should go ahead and complete the close because a half-closed task is no good to anyone.
395                   tasksToCloseDirty.add(task);
396               }
397           }
398   
399           tasksToCloseClean.removeAll(tasksToCloseDirty);
400           for (final Task task : tasksToCloseClean) {
401               try {
402                   completeTaskCloseClean(task);
403                   if (task.isActive()) {
404                       tasks.cleanUpTaskProducerAndRemoveTask(task.id(), taskCloseExceptions);
405                   }
406               } catch (final RuntimeException e) {
407                   final String uncleanMessage = String.format(
408                           "Failed to close task %s cleanly. Attempting to close remaining tasks before re-throwing:",
409                           task.id());
410                   log.error(uncleanMessage, e);
411                   taskCloseExceptions.putIfAbsent(task.id(), e);
412                   tasksToCloseDirty.add(task);
413               }
414           }
415   
416           tasksToRecycle.removeAll(tasksToCloseDirty);
417           for (final Task oldTask : tasksToRecycle) {
418               try {
419                   if (oldTask.isActive()) {
420                       final Set<TopicPartition> partitions = standbyTasksToCreate.remove(oldTask.id());
421                       tasks.convertActiveToStandby((StreamTask) oldTask, partitions, taskCloseExceptions);
422                   } else {
423                       final Set<TopicPartition> partitions = activeTasksToCreate.remove(oldTask.id());
424                       tasks.convertStandbyToActive((StandbyTask) oldTask, partitions);
425                   }
426               } catch (final RuntimeException e) {
427                   final String uncleanMessage = String.format("Failed to recycle task %s cleanly. Attempting to close remaining tasks before re-throwing:", oldTask.id());
428                   log.error(uncleanMessage, e);
429                   taskCloseExceptions.putIfAbsent(oldTask.id(), e);
430                   tasksToCloseDirty.add(oldTask);
431               }
432           }
433   
434           // for tasks that cannot be cleanly closed or recycled, close them dirty
435           for (final Task task : tasksToCloseDirty) {
436               closeTaskDirty(task);
437               tasks.cleanUpTaskProducerAndRemoveTask(task.id(), taskCloseExceptions);
438           }
439       }

441       /**
442        * Tries to initialize any new or still-uninitialized tasks, then checks if they can/have completed restoration.
443        *
444        * @throws IllegalStateException If store gets registered after initialized is already finished
445        * @throws StreamsException if the store's change log does not contain the partition
446        * @return {@code true} if all tasks are fully restored
447        */
448       boolean tryToCompleteRestoration(final long now, final java.util.function.Consumer<Set<TopicPartition>> offsetResetter) {
449           boolean allRunning = true;
450   
451           final List<Task> activeTasks = new LinkedList<>();
452           for (final Task task : tasks.allTasks()) {
453               try {
454                   task.initializeIfNeeded();
455                   task.clearTaskTimeout();
456               } catch (final LockException lockException) {
457                   // it is possible that if there are multiple threads within the instance that one thread
458                   // trying to grab the task from the other, while the other has not released the lock since
459                   // it did not participate in the rebalance. In this case we can just retry in the next iteration
460                   log.debug("Could not initialize task {} since: {}; will retry", task.id(), lockException.getMessage());
461                   allRunning = false;
462               } catch (final TimeoutException timeoutException) {
463                   task.maybeInitTaskTimeoutOrThrow(now, timeoutException);
464                   allRunning = false;
465               }
466   
467               if (task.isActive()) {
468                   activeTasks.add(task);
469               }
470           }
471   
472           if (allRunning && !activeTasks.isEmpty()) {
473   
474               final Set<TopicPartition> restored = changelogReader.completedChangelogs();
475   
476               for (final Task task : activeTasks) {
477                   if (restored.containsAll(task.changelogPartitions())) {
478                       try {
479                           task.completeRestoration(offsetResetter);
480                           task.clearTaskTimeout();
481                       } catch (final TimeoutException timeoutException) {
482                           task.maybeInitTaskTimeoutOrThrow(now, timeoutException);
483                           log.debug(
484                               String.format(
485                                   "Could not complete restoration for %s due to the following exception; will retry",
486                                   task.id()),
487                               timeoutException
488                           );
489   
490                           allRunning = false;
491                       }
492                   } else {
493                       // we found a restoring task that isn't done restoring, which is evidence that
494                       // not all tasks are running
495                       allRunning = false;
496                   }
497               }
498           }
499   
500           if (allRunning) {
501               // we can call resume multiple times since it is idempotent.
502               mainConsumer.resume(mainConsumer.assignment());
503           }
504   
505           return allRunning;
506       }
507   
508       /**
509        * Handle the revoked partitions and prepare for closing the associated tasks in {@link #handleAssignment(Map, Map)}
510        * We should commit the revoking tasks first before suspending them as we will not officially own them anymore when
511        * {@link #handleAssignment(Map, Map)} is called. Note that only active task partitions are passed in from the
512        * rebalance listener, so we only need to consider/commit active tasks here
513        *
514        * If eos-v2 is used, we must commit ALL tasks. Otherwise, we can just commit those (active) tasks which are revoked
515        *
516        * @throws TaskMigratedException if the task producer got fenced (EOS only)
517        */
518       void handleRevocation(final Collection<TopicPartition> revokedPartitions) {
519           final Set<TopicPartition> remainingRevokedPartitions = new HashSet<>(revokedPartitions);
520   
521           final Set<Task> revokedActiveTasks = new HashSet<>();
522           final Set<Task> commitNeededActiveTasks = new HashSet<>();
523           final Map<Task, Map<TopicPartition, OffsetAndMetadata>> consumedOffsetsPerTask = new HashMap<>();
524           final AtomicReference<RuntimeException> firstException = new AtomicReference<>(null);
525   
526           for (final Task task : activeTaskIterable()) {
527               if (remainingRevokedPartitions.containsAll(task.inputPartitions())) {
528                   // when the task input partitions are included in the revoked list,
529                   // this is an active task and should be revoked
530                   revokedActiveTasks.add(task);
531                   remainingRevokedPartitions.removeAll(task.inputPartitions());
532               } else if (task.commitNeeded()) {
533                   commitNeededActiveTasks.add(task);
534               }
535           }
536   
537           if (!remainingRevokedPartitions.isEmpty()) {
538               log.debug("The following revoked partitions {} are missing from the current task partitions. It could "
539                             + "potentially be due to race condition of consumer detecting the heartbeat failure, or the tasks " +
540                            "have been cleaned up by the handleAssignment callback.", remainingRevokedPartitions);
541           }
542   
543           prepareCommitAndAddOffsetsToMap(revokedActiveTasks, consumedOffsetsPerTask);
544   
545           // if we need to commit any revoking task then we just commit all of those needed committing together
546           final boolean shouldCommitAdditionalTasks = !consumedOffsetsPerTask.isEmpty();
547           if (shouldCommitAdditionalTasks) {
548               prepareCommitAndAddOffsetsToMap(commitNeededActiveTasks, consumedOffsetsPerTask);
549           }
550   
551           // even if commit failed, we should still continue and complete suspending those tasks, so we would capture
552           // any exception and rethrow it at the end. some exceptions may be handled immediately and then swallowed,
553           // as such we just need to skip those dirty tasks in the checkpoint
554           final Set<Task> dirtyTasks = new HashSet<>();
555           try {
556               // in handleRevocation we must call commitOffsetsOrTransaction() directly rather than
557               // commitAndFillInConsumedOffsetsAndMetadataPerTaskMap() to make sure we don't skip the
558               // offset commit because we are in a rebalance
559               taskExecutor.commitOffsetsOrTransaction(consumedOffsetsPerTask);
560           } catch (final TaskCorruptedException e) {
561               log.warn("Some tasks were corrupted when trying to commit offsets, these will be cleaned and revived: {}",
562                        e.corruptedTasks());
563   
564               // If we hit a TaskCorruptedException it must be EOS, just handle the cleanup for those corrupted tasks right here
565               dirtyTasks.addAll(tasks.tasks(e.corruptedTasks()));
566               closeDirtyAndRevive(dirtyTasks, true);
567           } catch (final TimeoutException e) {
568               log.warn("Timed out while trying to commit all tasks during revocation, these will be cleaned and revived");
569   
570               // If we hit a TimeoutException it must be ALOS, just close dirty and revive without wiping the state
571               dirtyTasks.addAll(consumedOffsetsPerTask.keySet());
572               closeDirtyAndRevive(dirtyTasks, false);
573           } catch (final RuntimeException e) {
574               log.error("Exception caught while committing those revoked tasks " + revokedActiveTasks, e);
575               firstException.compareAndSet(null, e);
576               dirtyTasks.addAll(consumedOffsetsPerTask.keySet());
577           }
578   
579           // we enforce checkpointing upon suspending a task: if it is resumed later we just proceed normally, if it is
580           // going to be closed we would checkpoint by then
581           for (final Task task : revokedActiveTasks) {
582               if (!dirtyTasks.contains(task)) {
583                   try {
584                       task.postCommit(true);
585                   } catch (final RuntimeException e) {
586                       log.error("Exception caught while post-committing task " + task.id(), e);
587                       maybeWrapAndSetFirstException(firstException, e, task.id());
588                   }
589               }
590           }
591   
592           if (shouldCommitAdditionalTasks) {
593               for (final Task task : commitNeededActiveTasks) {
594                   if (!dirtyTasks.contains(task)) {
595                       try {
596                           // for non-revoking active tasks, we should not enforce checkpoint
597                           // since if it is EOS enabled, no checkpoint should be written while
598                           // the task is in RUNNING tate
599                           task.postCommit(false);
600                       } catch (final RuntimeException e) {
601                           log.error("Exception caught while post-committing task " + task.id(), e);
602                           maybeWrapAndSetFirstException(firstException, e, task.id());
603                       }
604                   }
605               }
606           }
607   
608           for (final Task task : revokedActiveTasks) {
609               try {
610                   task.suspend();
611               } catch (final RuntimeException e) {
612                   log.error("Caught the following exception while trying to suspend revoked task " + task.id(), e);
613                   maybeWrapAndSetFirstException(firstException, e, task.id());
614               }
615           }
616   
617           if (firstException.get() != null) {
618               throw firstException.get();
619           }
620       }

624           for (final Task task : tasksToPrepare) {
625               try {
626                   final Map<TopicPartition, OffsetAndMetadata> committableOffsets = task.prepareCommit();
627                   if (!committableOffsets.isEmpty()) {
628                       consumedOffsetsPerTask.put(task, committableOffsets);
629                   }
630               } catch (final StreamsException e) {
631                   e.setTaskId(task.id());
632                   throw e;
633               } catch (final Exception e) {
634                   throw new StreamsException(e, task.id());
635               }
636           }
637       }

639       /**
640        * Closes active tasks as zombies, as these partitions have been lost and are no longer owned.
641        * NOTE this method assumes that when it is called, EVERY task/partition has been lost and must
642        * be closed as a zombie.
643        *
644        * @throws TaskMigratedException if the task producer got fenced (EOS only)
645        */
646       void handleLostAll() {
647           log.debug("Closing lost active tasks as zombies.");
648   
649           final Set<Task> allTask = new HashSet<>(tasks.allTasks());
650           for (final Task task : allTask) {
651               // Even though we've apparently dropped out of the group, we can continue safely to maintain our
652               // standby tasks while we rejoin.
653               if (task.isActive()) {
654                   closeTaskDirty(task);
655   
656                   tasks.cleanUpTaskProducerAndRemoveTask(task.id(), new HashMap<>());
657               }
658           }
659

661               tasks.reInitializeThreadProducer();
662           }

665       /**
666        * Compute the offset total summed across all stores in a task. Includes offset sum for any tasks we own the
667        * lock for, which includes assigned and unassigned tasks we locked in {@link #tryToLockAllNonEmptyTaskDirectories()}.
668        * Does not include stateless or non-logged tasks.
669        */
670       public Map<TaskId, Long> getTaskOffsetSums() {
671           final Map<TaskId, Long> taskOffsetSums = new HashMap<>();
672   
673           // Not all tasks will create directories, and there may be directories for tasks we don't currently own,
674           // so we consider all tasks that are either owned or on disk. This includes stateless tasks, which should
675           // just have an empty changelogOffsets map.
676           for (final TaskId id : union(HashSet::new, lockedTaskDirectories, tasks.tasksPerId().keySet())) {
677               final Task task = tasks.owned(id) ? tasks.task(id) : null;
678               // Closed and uninitialized tasks don't have any offsets so we should read directly from the checkpoint
679               if (task != null && task.state() != State.CREATED && task.state() != State.CLOSED) {
680                   final Map<TopicPartition, Long> changelogOffsets = task.changelogOffsets();
681                   if (changelogOffsets.isEmpty()) {
682                       log.debug("Skipping to encode apparently stateless (or non-logged) offset sum for task {}", id);
683                   } else {
684                       taskOffsetSums.put(id, sumOfChangelogOffsets(id, changelogOffsets));
685                   }
686               } else {
687                   final File checkpointFile = stateDirectory.checkpointFileFor(id);
688                   try {
689                       if (checkpointFile.exists()) {
690                           taskOffsetSums.put(id, sumOfChangelogOffsets(id, new OffsetCheckpoint(checkpointFile).read()));
691                       }
692                   } catch (final IOException e) {
693                       log.warn(String.format("Exception caught while trying to read checkpoint for task %s:", id), e);
694                   }
695               }
696           }
697   
698           return taskOffsetSums;
699       }

701       /**
702        * Makes a weak attempt to lock all non-empty task directories in the state dir. We are responsible for computing and
703        * reporting the offset sum for any unassigned tasks we obtain the lock for in the upcoming rebalance. Tasks
704        * that we locked but didn't own will be released at the end of the rebalance (unless of course we were
705        * assigned the task as a result of the rebalance). This method should be idempotent.
706        */
707       private void tryToLockAllNonEmptyTaskDirectories() {
708           // Always clear the set at the beginning as we're always dealing with the
709           // current set of actually-locked tasks.
710           lockedTaskDirectories.clear();
711   
712           for (final TaskDirectory taskDir : stateDirectory.listNonEmptyTaskDirectories()) {
713               final File dir = taskDir.file();
714               final String namedTopology = taskDir.namedTopology();
715               try {
716                   final TaskId id = parseTaskDirectoryName(dir.getName(), namedTopology);
717                   if (stateDirectory.lock(id)) {
718                       lockedTaskDirectories.add(id);
719                       if (!tasks.owned(id)) {
720                           log.debug("Temporarily locked unassigned task {} for the upcoming rebalance", id);
721                       }
722                   }
723               } catch (final TaskIdFormatException e) {
724                   // ignore any unknown files that sit in the same directory
725               }
726           }
727       }

729       /**
730        * Clean up after closed or removed tasks by making sure to unlock any remaining locked directories for them, for
731        * example unassigned tasks or those in the CREATED state when closed, since Task#close will not unlock them
732        */
733       private void releaseLockedDirectoriesForTasks(final Set<TaskId> tasksToUnlock) {
734           final Iterator<TaskId> taskIdIterator = lockedTaskDirectories.iterator();
735           while (taskIdIterator.hasNext()) {
736               final TaskId id = taskIdIterator.next();
737               if (tasksToUnlock.contains(id)) {
738                   stateDirectory.unlock(id);
739                   taskIdIterator.remove();
740               }
741           }
742       }
743   
744       /**
745        * We must release the lock for any unassigned tasks that we temporarily locked in preparation for a
746        * rebalance in {@link #tryToLockAllNonEmptyTaskDirectories()}.
747        */
748       private void releaseLockedUnassignedTaskDirectories() {
749           final Iterator<TaskId> taskIdIterator = lockedTaskDirectories.iterator();
750           while (taskIdIterator.hasNext()) {
751               final TaskId id = taskIdIterator.next();
752               if (!tasks.owned(id)) {
753                   stateDirectory.unlock(id);
754                   taskIdIterator.remove();
755               }
756           }
757       }

759       private long sumOfChangelogOffsets(final TaskId id, final Map<TopicPartition, Long> changelogOffsets) {
760           long offsetSum = 0L;
761           for (final Map.Entry<TopicPartition, Long> changelogEntry : changelogOffsets.entrySet()) {
762               final long offset = changelogEntry.getValue();
763   
764   
765               if (offset == Task.LATEST_OFFSET) {
766                   // this condition can only be true for active tasks; never for standby
767                   // for this case, the offset of all partitions is set to `LATEST_OFFSET`
768                   // and we "forward" the sentinel value directly
769                   return Task.LATEST_OFFSET;
770               } else if (offset != OffsetCheckpoint.OFFSET_UNKNOWN) {
771                   if (offset < 0) {
772                       throw new StreamsException(
773                           new IllegalStateException("Expected not to get a sentinel offset, but got: " + changelogEntry),
774                           id);
775                   }
776                   offsetSum += offset;
777                   if (offsetSum < 0) {
778                       log.warn("Sum of changelog offsets for task {} overflowed, pinning to Long.MAX_VALUE", id);
779                       return Long.MAX_VALUE;
780                   }
781               }
782           }
783   
784           return offsetSum;
785       }

810       void shutdown(final boolean clean) {
811           final AtomicReference<RuntimeException> firstException = new AtomicReference<>(null);
812   
813           // TODO: change type to `StreamTask`
814           final Set<Task> activeTasks = new TreeSet<>(Comparator.comparing(Task::id));
815           activeTasks.addAll(tasks.activeTasks());
816   
817           executeAndMaybeSwallow(
818               clean,
819               () -> closeAndCleanUpTasks(activeTasks, standbyTaskIterable(), clean),
820               e -> firstException.compareAndSet(null, e),
821               e -> log.warn("Ignoring an exception while unlocking remaining task directories.", e)
822           );
823   
824           executeAndMaybeSwallow(
825               clean,
826               tasks::closeThreadProducerIfNeeded,
827               e -> firstException.compareAndSet(null, e),
828               e -> log.warn("Ignoring an exception while closing thread producer.", e)
829           );
830   
831           tasks.clear();
832   
833           // this should be called after closing all tasks and clearing them from `tasks` to make sure we unlock the dir
834           // for any tasks that may have still been in CREATED at the time of shutdown, since Task#close will not do so
835           executeAndMaybeSwallow(
836               clean,
837               this::releaseLockedUnassignedTaskDirectories,
838               e -> firstException.compareAndSet(null, e),
839               e -> log.warn("Ignoring an exception while unlocking remaining task directories.", e)
840           );
841   
842           final RuntimeException fatalException = firstException.get();
843           if (fatalException != null) {
844               throw new RuntimeException("Unexpected exception while closing task", fatalException);
845           }
846       }
