17   package org.apache.kafka.streams.processor.internals;
18   
19   import org.apache.kafka.common.config.TopicConfig;
20   import org.apache.kafka.common.internals.Topic;
21   
22   import java.util.HashMap;
23   import java.util.Map;
24   import java.util.Objects;
25   import java.util.Optional;
26   
27   /**
28    * InternalTopicConfig captures the properties required for configuring
29    * the internal topics we create for change-logs and repartitioning etc.
30    */
31   public abstract class InternalTopicConfig {
32       final String name;
33       final Map<String, String> topicConfigs;
34       final boolean enforceNumberOfPartitions;
35   
36       private Optional<Integer> numberOfPartitions = Optional.empty();
37   
38       static final Map<String, String> INTERNAL_TOPIC_DEFAULT_OVERRIDES = new HashMap<>();
39       static {
40           INTERNAL_TOPIC_DEFAULT_OVERRIDES.put(TopicConfig.MESSAGE_TIMESTAMP_TYPE_CONFIG, "CreateTime");
41       }
42   
43       InternalTopicConfig(final String name, final Map<String, String> topicConfigs) {
44           this.name = Objects.requireNonNull(name, "name can't be null");
45           Topic.validate(name);
46           this.topicConfigs = Objects.requireNonNull(topicConfigs, "topicConfigs can't be null");
47           this.enforceNumberOfPartitions = false;
48       }
49   
50       InternalTopicConfig(final String name,
51                           final Map<String, String> topicConfigs,
52                           final int numberOfPartitions,
53                           final boolean enforceNumberOfPartitions) {
54           this.name = Objects.requireNonNull(name, "name can't be null");
55           Topic.validate(name);
56           validateNumberOfPartitions(numberOfPartitions);
57           this.topicConfigs = Objects.requireNonNull(topicConfigs, "topicConfigs can't be null");
58           this.numberOfPartitions = Optional.of(numberOfPartitions);
59           this.enforceNumberOfPartitions = enforceNumberOfPartitions;
60       }
61   
62       /**
63        * Get the configured properties for this topic. If retentionMs is set then
64        * we add additionalRetentionMs to work out the desired retention when cleanup.policy=compact,delete
65        *
66        * @param additionalRetentionMs - added to retention to allow for clock drift etc
67        * @return Properties to be used when creating the topic
68        */
69       public abstract Map<String, String> getProperties(final Map<String, String> defaultProperties, final long additionalRetentionMs);
70   
71       public boolean hasEnforcedNumberOfPartitions() {
72           return enforceNumberOfPartitions;
73       }
74   
75       public String name() {
76           return name;
77       }
78   
79       public Optional<Integer> numberOfPartitions() {
80           return numberOfPartitions;
81       }
82   
83       public void setNumberOfPartitions(final int numberOfPartitions) {
84           if (hasEnforcedNumberOfPartitions()) {
85               throw new UnsupportedOperationException("number of partitions are enforced on topic " +
86                                                       "" + name() + " and can't be altered.");
87           }
88   
89           validateNumberOfPartitions(numberOfPartitions);
90   
91           this.numberOfPartitions = Optional.of(numberOfPartitions);
92       }
93   
94       private static void validateNumberOfPartitions(final int numberOfPartitions) {
95           if (numberOfPartitions < 1) {
96               throw new IllegalArgumentException("Number of partitions must be at least 1.");
97           }
98       }
99   
100       @Override
101       public String toString() {
102           return "InternalTopicConfig(" +
103                   "name=" + name +
104                   ", topicConfigs=" + topicConfigs +
105                   ", enforceNumberOfPartitions=" + enforceNumberOfPartitions +
106                   ")";
107       }
108   }
