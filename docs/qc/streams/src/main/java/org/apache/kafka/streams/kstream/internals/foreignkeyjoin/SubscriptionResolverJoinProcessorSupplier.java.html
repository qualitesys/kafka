18   package org.apache.kafka.streams.kstream.internals.foreignkeyjoin;
19   
20   import org.apache.kafka.common.errors.UnsupportedVersionException;
21   import org.apache.kafka.common.serialization.Serializer;
22   import org.apache.kafka.streams.kstream.ValueJoiner;
23   import org.apache.kafka.streams.kstream.internals.KTableValueGetter;
24   import org.apache.kafka.streams.kstream.internals.KTableValueGetterSupplier;
25   import org.apache.kafka.streams.processor.api.ContextualProcessor;
26   import org.apache.kafka.streams.processor.api.Processor;
27   import org.apache.kafka.streams.processor.api.ProcessorContext;
28   import org.apache.kafka.streams.processor.api.ProcessorSupplier;
29   import org.apache.kafka.streams.processor.api.Record;
30   import org.apache.kafka.streams.state.ValueAndTimestamp;
31   import org.apache.kafka.streams.state.internals.Murmur3;
32   
33   import java.util.function.Supplier;
34   
35   /**
36    * Receives {@code SubscriptionResponseWrapper<VO>} events and filters out events which do not match the current hash
37    * of the primary key. This eliminates race-condition results for rapidly-changing foreign-keys for a given primary key.
38    * Applies the join and emits nulls according to LEFT/INNER rules.
39    *
40    * @param <K> Type of primary keys
41    * @param <V> Type of primary values
42    * @param <VO> Type of foreign values
43    * @param <VR> Type of joined result of primary and foreign values
44    */
45   public class SubscriptionResolverJoinProcessorSupplier<K, V, VO, VR> implements ProcessorSupplier<K, SubscriptionResponseWrapper<VO>, K, VR> {
46       private final KTableValueGetterSupplier<K, V> valueGetterSupplier;
47       private final Serializer<V> constructionTimeValueSerializer;
48       private final Supplier<String> valueHashSerdePseudoTopicSupplier;
49       private final ValueJoiner<V, VO, VR> joiner;
50       private final boolean leftJoin;
51   
52       public SubscriptionResolverJoinProcessorSupplier(final KTableValueGetterSupplier<K, V> valueGetterSupplier,
53                                                        final Serializer<V> valueSerializer,
54                                                        final Supplier<String> valueHashSerdePseudoTopicSupplier,
55                                                        final ValueJoiner<V, VO, VR> joiner,
56                                                        final boolean leftJoin) {
57           this.valueGetterSupplier = valueGetterSupplier;
58           constructionTimeValueSerializer = valueSerializer;
59           this.valueHashSerdePseudoTopicSupplier = valueHashSerdePseudoTopicSupplier;
60           this.joiner = joiner;
61           this.leftJoin = leftJoin;
62       }
63   
64       @Override
65       public Processor<K, SubscriptionResponseWrapper<VO>, K, VR> get() {
66           return new ContextualProcessor<K, SubscriptionResponseWrapper<VO>, K, VR>() {
67               private String valueHashSerdePseudoTopic;
68               private Serializer<V> runtimeValueSerializer = constructionTimeValueSerializer;
69   
70               private KTableValueGetter<K, V> valueGetter;
71   
72               @SuppressWarnings("unchecked")
73               @Override
74               public void init(final ProcessorContext<K, VR> context) {
75                   super.init(context);
76                   valueHashSerdePseudoTopic = valueHashSerdePseudoTopicSupplier.get();
77                   valueGetter = valueGetterSupplier.get();
78                   valueGetter.init(context);
79                   if (runtimeValueSerializer == null) {
80                       runtimeValueSerializer = (Serializer<V>) context.valueSerde().serializer();
81                   }
82               }
83   
84               @Override
85               public void process(final Record<K, SubscriptionResponseWrapper<VO>> record) {
86                   if (record.value().getVersion() != SubscriptionResponseWrapper.CURRENT_VERSION) {
87                       //Guard against modifications to SubscriptionResponseWrapper. Need to ensure that there is
88                       //compatibility with previous versions to enable rolling upgrades. Must develop a strategy for
89                       //upgrading from older SubscriptionWrapper versions to newer versions.
90                       throw new UnsupportedVersionException("SubscriptionResponseWrapper is of an incompatible version.");
91                   }
92                   final ValueAndTimestamp<V> currentValueWithTimestamp = valueGetter.get(record.key());
93   
94                   final long[] currentHash = currentValueWithTimestamp == null ?
95                       null :
96                       Murmur3.hash128(runtimeValueSerializer.serialize(valueHashSerdePseudoTopic, currentValueWithTimestamp.value()));
97   
98                   final long[] messageHash = record.value().getOriginalValueHash();
99   
100                   //If this value doesn't match the current value from the original table, it is stale and should be discarded.
101                   if (java.util.Arrays.equals(messageHash, currentHash)) {
102                       final VR result;
103   
104                       if (record.value().getForeignValue() == null && (!leftJoin || currentValueWithTimestamp == null)) {
105                           result = null; //Emit tombstone
106                       } else {
107                           result = joiner.apply(currentValueWithTimestamp == null ? null : currentValueWithTimestamp.value(), record.value().getForeignValue());
108                       }
109                       context().forward(record.withValue(result));
110                   }
111               }
112           };
113       }
114   }
