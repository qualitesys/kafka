22   import java.time.Instant;

24   /**
25    * A window store that only supports read operations.
26    * Implementations should be thread-safe as concurrent reads and writes are expected.
27    *
28    * <p>Note: The current implementation of either forward or backward fetches on range-key-range-time does not
29    * obey the ordering when there are multiple local stores hosted on that instance. For example,
30    * if there are two stores from two tasks hosting keys {1,3} and {2,4}, then a range query of key [1,4]
31    * would return in the order of [1,3,2,4] but not [1,2,3,4] since it is just looping over the stores only.
32    *
33    * @param <K> Type of keys
34    * @param <V> Type of values
35    */
36   public interface ReadOnlyWindowStore<K, V> {
37   
38       /**
39        * Get the value of key from a window.
40        *
41        * @param key  the key to fetch
42        * @param time start timestamp (inclusive) of the window
43        * @return The value or {@code null} if no value is found in the window
44        * @throws InvalidStateStoreException if the store is not initialized
45        * @throws NullPointerException       if {@code null} is used for any key.
46        */
47       V fetch(K key, long time);
48   
49       /**
50        * Get all the key-value pairs with the given key and the time range from all the existing windows.
51        * <p>
52        * This iterator must be closed after use.
53        * <p>
54        * The time range is inclusive and applies to the starting timestamp of the window.
55        * For example, if we have the following windows:
56        * <pre>
57        * +-------------------------------+
58        * |  key  | start time | end time |
59        * +-------+------------+----------+
60        * |   A   |     10     |    20    |
61        * +-------+------------+----------+
62        * |   A   |     15     |    25    |
63        * +-------+------------+----------+
64        * |   A   |     20     |    30    |
65        * +-------+------------+----------+
66        * |   A   |     25     |    35    |
67        * +--------------------------------
68        * </pre>
69        * And we call {@code store.fetch("A", Instant.ofEpochMilli(10), Instant.ofEpochMilli(20))} then the results will contain the first
70        * three windows from the table above, i.e., all those where 10 &lt;= start time &lt;= 20.
71        * <p>
72        * For each key, the iterator guarantees ordering of windows, starting from the oldest/earliest
73        * available window to the newest/latest window.
74        *
75        * @param key  the key to fetch
76        * @param timeFrom time range start (inclusive), where iteration starts.
77        * @param timeTo   time range end (inclusive), where iteration ends.
78        * @return an iterator over key-value pairs {@code <timestamp, value>}, from beginning to end of time.
79        * @throws InvalidStateStoreException if the store is not initialized
80        * @throws NullPointerException       if {@code null} is used for key.
81        * @throws IllegalArgumentException   if duration is negative or can't be represented as {@code long milliseconds}
82        */
83       WindowStoreIterator<V> fetch(K key, Instant timeFrom, Instant timeTo) throws IllegalArgumentException;
84   
85       /**
86        * Get all the key-value pairs with the given key and the time range from all the existing windows
87        * in backward order with respect to time (from end to beginning of time).
88        * <p>
89        * This iterator must be closed after use.
90        * <p>
91        * The time range is inclusive and applies to the starting timestamp of the window.
92        * For example, if we have the following windows:
93        * <pre>
94        * +-------------------------------+
95        * |  key  | start time | end time |
96        * +-------+------------+----------+
97        * |   A   |     10     |    20    |
98        * +-------+------------+----------+
99        * |   A   |     15     |    25    |
100        * +-------+------------+----------+
101        * |   A   |     20     |    30    |
102        * +-------+------------+----------+
103        * |   A   |     25     |    35    |
104        * +--------------------------------
105        * </pre>
106        * And we call {@code store.backwardFetch("A", Instant.ofEpochMilli(10), Instant.ofEpochMilli(20))} then the
107        * results will contain the first three windows from the table above in backward order,
108        * i.e., all those where 10 &lt;= start time &lt;= 20.
109        * <p>
110        * For each key, the iterator guarantees ordering of windows, starting from the newest/latest
111        * available window to the oldest/earliest window.
112        *
113        * @param key  the key to fetch
114        * @param timeFrom time range start (inclusive), where iteration ends.
115        * @param timeTo   time range end (inclusive), where iteration starts.
116        * @return an iterator over key-value pairs {@code <timestamp, value>}, from end to beginning of time.
117        * @throws InvalidStateStoreException if the store is not initialized
118        * @throws NullPointerException       if {@code null} is used for key.
119        * @throws IllegalArgumentException   if duration is negative or can't be represented as {@code long milliseconds}
120        */
121       default WindowStoreIterator<V> backwardFetch(K key, Instant timeFrom, Instant timeTo) throws IllegalArgumentException  {
122           throw new UnsupportedOperationException();
123       }
124   
125       /**
126        * Get all the key-value pairs in the given key range and time range from all the existing windows.
127        * <p>
128        * This iterator must be closed after use.
129        *
130        * @param keyFrom     the first key in the range
131        *                    A null value indicates a starting position from the first element in the store.
132        * @param keyTo       the last key in the range
133        *                    A null value indicates that the range ends with the last element in the store.
134        * @param timeFrom time range start (inclusive), where iteration starts.
135        * @param timeTo   time range end (inclusive), where iteration ends.
136        * @return an iterator over windowed key-value pairs {@code <Windowed<K>, value>}, from beginning to end of time.
137        * @throws InvalidStateStoreException if the store is not initialized
138        * @throws IllegalArgumentException   if duration is negative or can't be represented as {@code long milliseconds}
139        */
140       KeyValueIterator<Windowed<K>, V> fetch(K keyFrom, K keyTo, Instant timeFrom, Instant timeTo)
141           throws IllegalArgumentException;
142   
143       /**
144        * Get all the key-value pairs in the given key range and time range from all the existing windows
145        * in backward order with respect to time (from end to beginning of time).
146        * <p>
147        * This iterator must be closed after use.
148        *
149        * @param keyFrom     the first key in the range
150        *                    A null value indicates a starting position from the first element in the store.
151        * @param keyTo       the last key in the range
152        *                    A null value indicates that the range ends with the last element in the store.
153        * @param timeFrom time range start (inclusive), where iteration ends.
154        * @param timeTo   time range end (inclusive), where iteration starts.
155        * @return an iterator over windowed key-value pairs {@code <Windowed<K>, value>}, from end to beginning of time.
156        * @throws InvalidStateStoreException if the store is not initialized
157        * @throws IllegalArgumentException   if duration is negative or can't be represented as {@code long milliseconds}
158        */
159       default KeyValueIterator<Windowed<K>, V> backwardFetch(K keyFrom, K keyTo, Instant timeFrom, Instant timeTo)
160           throws IllegalArgumentException  {
161           throw new UnsupportedOperationException();
162       }
163   
164   
165       /**
166        * Gets all the key-value pairs in the existing windows.
167        *
168        * @return an iterator over windowed key-value pairs {@code <Windowed<K>, value>}, from beginning to end of time.
169        * @throws InvalidStateStoreException if the store is not initialized
170        */
171       KeyValueIterator<Windowed<K>, V> all();
172   
173       /**
174        * Gets all the key-value pairs in the existing windows in backward order
175        * with respect to time (from end to beginning of time).
176        *
177        * @return an backward iterator over windowed key-value pairs {@code <Windowed<K>, value>}, from the end to beginning of time.
178        * @throws InvalidStateStoreException if the store is not initialized
179        */
180       default KeyValueIterator<Windowed<K>, V> backwardAll() {
181           throw new UnsupportedOperationException();
182       }
183   
184       /**
185        * Gets all the key-value pairs that belong to the windows within in the given time range.
186        *
187        * @param timeFrom the beginning of the time slot from which to search (inclusive), where iteration starts.
188        * @param timeTo   the end of the time slot from which to search (inclusive), where iteration ends.
189        * @return an iterator over windowed key-value pairs {@code <Windowed<K>, value>}, from beginning to end of time.
190        * @throws InvalidStateStoreException if the store is not initialized
191        * @throws NullPointerException       if {@code null} is used for any key
192        * @throws IllegalArgumentException   if duration is negative or can't be represented as {@code long milliseconds}
193        */
194       KeyValueIterator<Windowed<K>, V> fetchAll(Instant timeFrom, Instant timeTo) throws IllegalArgumentException;
195   
196       /**
197        * Gets all the key-value pairs that belong to the windows within in the given time range in backward order
198        * with respect to time (from end to beginning of time).
199        *
200        * @param timeFrom the beginning of the time slot from which to search (inclusive), where iteration ends.
201        * @param timeTo   the end of the time slot from which to search (inclusive), where iteration starts.
202        * @return an backward iterator over windowed key-value pairs {@code <Windowed<K>, value>}, from end to beginning of time.
203        * @throws InvalidStateStoreException if the store is not initialized
204        * @throws NullPointerException       if {@code null} is used for any key
205        * @throws IllegalArgumentException   if duration is negative or can't be represented as {@code long milliseconds}
206        */
207       default KeyValueIterator<Windowed<K>, V> backwardFetchAll(Instant timeFrom, Instant timeTo) throws IllegalArgumentException  {
208           throw new UnsupportedOperationException();
209       }
