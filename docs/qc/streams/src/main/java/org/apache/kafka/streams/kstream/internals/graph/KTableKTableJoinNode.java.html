94           // this incorrect cast should be corrected by the end of the KIP-478 implementation

96       }

98       @Override
99       public void writeToTopology(final InternalTopologyBuilder topologyBuilder) {
100           final String thisProcessorName = thisProcessorParameters().processorName();
101           final String otherProcessorName = otherProcessorParameters().processorName();
102           final String mergeProcessorName = mergeProcessorParameters().processorName();
103   
104           topologyBuilder.addProcessor(
105               thisProcessorName,
106               thisProcessorParameters().processorSupplier(),
107               thisJoinSideNodeName());
108   
109           topologyBuilder.addProcessor(
110               otherProcessorName,
111               otherProcessorParameters().processorSupplier(),
112               otherJoinSideNodeName());
113   
114           topologyBuilder.addProcessor(
115               mergeProcessorName,
116               mergeProcessorParameters().processorSupplier(),
117               thisProcessorName,
118               otherProcessorName);
119   
120           topologyBuilder.connectProcessorAndStateStores(thisProcessorName, joinOtherStoreNames);
121           topologyBuilder.connectProcessorAndStateStores(otherProcessorName, joinThisStoreNames);
122   
123           if (storeBuilder != null) {
124               topologyBuilder.addStateStore(storeBuilder, mergeProcessorName);
125           }
126       }
