20   import java.util.Collections;
21   import java.util.List;
22   import java.util.NoSuchElementException;
23   
24   import org.apache.kafka.common.utils.AbstractIterator;
25   import org.apache.kafka.common.utils.Bytes;
26   import org.apache.kafka.streams.KeyValue;
27   import org.apache.kafka.streams.state.KeyValueIterator;
28   import org.apache.kafka.streams.state.KeyValueStore;
29   import org.apache.kafka.common.serialization.Serdes;
30   import org.apache.kafka.common.serialization.Serde;

32   /**
33    * A wrapper key-value store that serializes the record values bytes as a list.
34    * As a result put calls would be interpreted as a get-append-put to the underlying RocksDB store.
35    * A put(k,null) will still delete the key, ie, the full list of all values of this key.
36    * Range iterators would also flatten the value lists and return the values one-by-one.
37    *
38    * This store is used for cases where we do not want to de-duplicate values of the same keys but want to retain all such values.
39    */
40   @SuppressWarnings("unchecked")
41   public class ListValueStore
42       extends WrappedStateStore<KeyValueStore<Bytes, byte[]>, Bytes, byte[]>
43       implements KeyValueStore<Bytes, byte[]> {
44   
45       static private final Serde<List<byte[]>> LIST_SERDE = Serdes.ListSerde(ArrayList.class, Serdes.ByteArray());
46   
47       ListValueStore(final KeyValueStore<Bytes, byte[]> bytesStore) {
48           super(bytesStore);
49       }
50   
51       @Override
52       public void put(final Bytes key, final byte[] addedValue) {
53           // if the value is null we can skip the get and blind delete
54           if (addedValue == null) {
55               wrapped().put(key, null);
56           } else {
57               final byte[] oldValue = wrapped().get(key);
58               putInternal(key, addedValue, oldValue);
59           }
60       }
61   
62       @Override
63       public byte[] putIfAbsent(final Bytes key, final byte[] addedValue) {
64           final byte[] oldValue = wrapped().get(key);
65   
66           if (oldValue != null) {
67               // if the value is null we can skip the get and blind delete
68               if (addedValue == null) {
69                   wrapped().put(key, null);
70               } else {
71                   putInternal(key, addedValue, oldValue);
72               }
73           }
74   
75           // TODO: here we always return null so that deser would not fail.
76           //       we only do this since we know the only caller (stream-stream join processor)
77           //       would not need the actual value at all; the changelogging wrapper would not call this function
78           return null;
79       }
80   
81       // this function assumes the addedValue is not null; callers should check null themselves
82       private void putInternal(final Bytes key, final byte[] addedValue, final byte[] oldValue) {
83           if (oldValue == null) {
84               wrapped().put(key, LIST_SERDE.serializer().serialize(null, Collections.singletonList(addedValue)));
85           } else {
86               final List<byte[]> list = LIST_SERDE.deserializer().deserialize(null, oldValue);
87               list.add(addedValue);
88   
89               wrapped().put(key, LIST_SERDE.serializer().serialize(null, list));
90           }
91       }
92   
93       @Override
94       public void putAll(final List<KeyValue<Bytes, byte[]>> entries) {
95           throw new UnsupportedOperationException("putAll not supported");
96       }
97   
98       @Override
99       public byte[] delete(final Bytes key) {
100           // we intentionally disable delete calls since the returned bytes would
101           // represent a list, not a single value; we need to have a new API for delete if we do need it
102           throw new UnsupportedOperationException("delete not supported");
103       }
104   
105       @Override
106       public byte[] get(final Bytes key) {
107           return wrapped().get(key);
108       }
109   
110       @Override
111       public KeyValueIterator<Bytes, byte[]> range(final Bytes from, final Bytes to) {
112           throw new UnsupportedOperationException("range not supported");
113       }
114   
115       @Override
116       public KeyValueIterator<Bytes, byte[]> all() {
117           return new ValueListIterator(wrapped().all());
118       }
119   
120       @Override
121       public long approximateNumEntries() {
122           return wrapped().approximateNumEntries();
123       }
124   
125       private static class ValueListIterator extends AbstractIterator<KeyValue<Bytes, byte[]>>
126           implements KeyValueIterator<Bytes, byte[]> {
127   
128           private final KeyValueIterator<Bytes, byte[]> bytesIterator;
129           private final List<byte[]> currList = new ArrayList<>();
130           private KeyValue<Bytes, byte[]> next;
131           private Bytes nextKey;
132   
133           ValueListIterator(final KeyValueIterator<Bytes, byte[]> bytesIterator) {
134               this.bytesIterator = bytesIterator;
135           }
136   
137           @Override
138           public Bytes peekNextKey() {
139               if (!hasNext()) {
140                   throw new NoSuchElementException();
141               }
142               return next.key;
143           }
144   
145           @Override
146           public KeyValue<Bytes, byte[]> makeNext() {
147               while (currList.isEmpty() && bytesIterator.hasNext()) {
148                   final KeyValue<Bytes, byte[]> next = bytesIterator.next();
149                   nextKey = next.key;
150                   currList.addAll(LIST_SERDE.deserializer().deserialize(null, next.value));
151               }
152   
153               if (currList.isEmpty()) {
154                   return allDone();
155               } else {
156                   next = KeyValue.pair(nextKey, currList.remove(0));
157                   return next;
158               }
159           }
160   
161           @Override
162           public void close() {
163               bytesIterator.close();
164               // also need to clear the current list buffer since
165               // otherwise even after close the iter can still return data
166               currList.clear();
167           }
168       }
169   }
