31   /**
32    * The fixed-size time-based window specifications used for aggregations.
33    * <p>
34    * The semantics of time-based aggregation windows are: Every T1 (advance) milliseconds, compute the aggregate total for
35    * T2 (size) milliseconds.
36    * <ul>
37    *     <li> If {@code advance < size} a hopping windows is defined:<br />
38    *          it discretize a stream into overlapping windows, which implies that a record maybe contained in one and or
39    *          more "adjacent" windows.</li>
40    *     <li> If {@code advance == size} a tumbling window is defined:<br />
41    *          it discretize a stream into non-overlapping windows, which implies that a record is only ever contained in
42    *          one and only one tumbling window.</li>
43    * </ul>
44    * Thus, the specified {@link TimeWindow}s are aligned to the epoch.
45    * Aligned to the epoch means, that the first window starts at timestamp zero.
46    * For example, hopping windows with size of 5000ms and advance of 3000ms, have window boundaries
47    * [0;5000),[3000;8000),... and not [1000;6000),[4000;9000),... or even something "random" like [1452;6452),[4452;9452),...
48    * <p>
49    * For time semantics, see {@link TimestampExtractor}.
50    *
51    * @see SessionWindows
52    * @see UnlimitedWindows
53    * @see JoinWindows
54    * @see KGroupedStream#windowedBy(Windows)
55    * @see TimestampExtractor
56    */
57   public final class TimeWindows extends Windows<TimeWindow> {
58   
59       /** The size of the windows in milliseconds. */
60       @SuppressWarnings("WeakerAccess")
61       public final long sizeMs;
62   
63       /**
64        * The size of the window's advance interval in milliseconds, i.e., by how much a window moves forward relative to
65        * the previous one.
66        */
67       @SuppressWarnings("WeakerAccess")
68       public final long advanceMs;
69   
70       private final long graceMs;
71   
72       // flag to check if the grace is already set via ofSizeAndGrace or ofSizeWithNoGrace
73       private final boolean hasSetGrace;
74   
75       private TimeWindows(final long sizeMs, final long advanceMs, final long graceMs, final boolean hasSetGrace) {
76           this.sizeMs = sizeMs;
77           this.advanceMs = advanceMs;
78           this.graceMs = graceMs;
79           this.hasSetGrace = hasSetGrace;
80   
81           if (sizeMs <= 0) {
82               throw new IllegalArgumentException("Window size (sizeMs) must be larger than zero.");
83           }
84   
85           if (advanceMs <= 0 || advanceMs > sizeMs) {
86               throw new IllegalArgumentException(String.format("Window advancement interval should be more than zero " +
87                   "and less than window duration which is %d ms, but given advancement interval is: %d ms", sizeMs, advanceMs));
88           }
89   
90           if (graceMs < 0) {
91               throw new IllegalArgumentException("Grace period must not be negative.");
92           }
93       }
94   
95       /**
96        * Return a window definition with the given window size, and with the advance interval being equal to the window
97        * size.
98        * The time interval represented by the N-th window is: {@code [N * size, N * size + size)}.
99        * <p>
100        * This provides the semantics of tumbling windows, which are fixed-sized, gap-less, non-overlapping windows.
101        * Tumbling windows are a special case of hopping windows with {@code advance == size}.
102        * <p>
103        * CAUTION: Using this method implicitly sets the grace period to zero, which means that any out-of-order
104        * records arriving after the window ends are considered late and will be dropped.
105        *
106        * @param size The size of the window
107        * @return a new window definition with default no grace period. Note that this means out-of-order records arriving after the window end will be dropped
108        * @throws IllegalArgumentException if the specified window size is zero or negative or can't be represented as {@code long milliseconds}
109        */
110       public static TimeWindows ofSizeWithNoGrace(final Duration size) throws IllegalArgumentException {
111           return ofSizeAndGrace(size, ofMillis(NO_GRACE_PERIOD));
112       }
113   
114       /**
115        * Return a window definition with the given window size, and with the advance interval being equal to the window
116        * size.
117        * The time interval represented by the N-th window is: {@code [N * size, N * size + size)}.
118        * <p>
119        * This provides the semantics of tumbling windows, which are fixed-sized, gap-less, non-overlapping windows.
120        * Tumbling windows are a special case of hopping windows with {@code advance == size}.
121        * <p>
122        * Using this method explicitly sets the grace period to the duration specified by {@code afterWindowEnd}, which
123        * means that only out-of-order records arriving more than the grace period after the window end will be dropped.
124        * The window close, after which any incoming records are considered late and will be rejected, is defined as
125        * {@code windowEnd + afterWindowEnd}
126        *
127        * @param size The size of the window. Must be larger than zero
128        * @param afterWindowEnd The grace period to admit out-of-order events to a window. Must be non-negative.
129        * @return a TimeWindows object with the specified size and the specified grace period
130        * @throws IllegalArgumentException if {@code afterWindowEnd} is negative or can't be represented as {@code long milliseconds}
131        */
132       public static TimeWindows ofSizeAndGrace(final Duration size, final Duration afterWindowEnd) throws IllegalArgumentException {
133           final String sizeMsgPrefix = prepareMillisCheckFailMsgPrefix(size, "size");
134           final long sizeMs = validateMillisecondDuration(size, sizeMsgPrefix);
135   
136           final String afterWindowEndMsgPrefix = prepareMillisCheckFailMsgPrefix(afterWindowEnd, "afterWindowEnd");
137           final long afterWindowEndMs = validateMillisecondDuration(afterWindowEnd, afterWindowEndMsgPrefix);
138   
139           return new TimeWindows(sizeMs, sizeMs, afterWindowEndMs, true);
140       }
141   
142       /**
143        * Return a window definition with the given window size, and with the advance interval being equal to the window
144        * size.
145        * The time interval represented by the N-th window is: {@code [N * size, N * size + size)}.
146        * <p>
147        * This provides the semantics of tumbling windows, which are fixed-sized, gap-less, non-overlapping windows.
148        * Tumbling windows are a special case of hopping windows with {@code advance == size}.
149        *
150        * @param size The size of the window
151        * @return a new window definition without specifying the grace period (default to 24 hours minus window {@code size})
152        * @throws IllegalArgumentException if the specified window size is zero or negative or can't be represented as {@code long milliseconds}
153        * @deprecated since 3.0. Use {@link #ofSizeWithNoGrace(Duration)} } instead
154        */
155       @Deprecated
156       public static TimeWindows of(final Duration size) throws IllegalArgumentException {
157           final String msgPrefix = prepareMillisCheckFailMsgPrefix(size, "size");
158           final long sizeMs = validateMillisecondDuration(size, msgPrefix);
159   
160           return new TimeWindows(sizeMs, sizeMs, Math.max(DEPRECATED_DEFAULT_24_HR_GRACE_PERIOD - sizeMs, 0), false);
161       }
162   
163       /**
164        * Return a window definition with the original size, but advance ("hop") the window by the given interval, which
165        * specifies by how much a window moves forward relative to the previous one.
166        * The time interval represented by the N-th window is: {@code [N * advance, N * advance + size)}.
167        * <p>
168        * This provides the semantics of hopping windows, which are fixed-sized, overlapping windows.
169        *
170        * @param advance The advance interval ("hop") of the window, with the requirement that {@code 0 < advance.toMillis() <= sizeMs}.
171        * @return a new window definition with default maintain duration of 1 day
172        * @throws IllegalArgumentException if the advance interval is negative, zero, or larger than the window size
173        */
174       public TimeWindows advanceBy(final Duration advance) {
175           final String msgPrefix = prepareMillisCheckFailMsgPrefix(advance, "advance");
176           final long advanceMs = validateMillisecondDuration(advance, msgPrefix);
177           return new TimeWindows(sizeMs, advanceMs, graceMs, false);
178       }
179   
180       @Override
181       public Map<Long, TimeWindow> windowsFor(final long timestamp) {
182           long windowStart = (Math.max(0, timestamp - sizeMs + advanceMs) / advanceMs) * advanceMs;
183           final Map<Long, TimeWindow> windows = new LinkedHashMap<>();
184           while (windowStart <= timestamp) {
185               final TimeWindow window = new TimeWindow(windowStart, windowStart + sizeMs);
186               windows.put(windowStart, window);
187               windowStart += advanceMs;
188           }
189           return windows;
190       }
191   
192       @Override
193       public long size() {
194           return sizeMs;
195       }
196   
197       /**
198        * Reject out-of-order events that arrive more than {@code millisAfterWindowEnd}
199        * after the end of its window.
200        * <p>
201        * Delay is defined as (stream_time - record_timestamp).
202        *
203        * @param afterWindowEnd The grace period to admit out-of-order events to a window.
204        * @return this updated builder
205        * @throws IllegalArgumentException if {@code afterWindowEnd} is negative or can't be represented as {@code long milliseconds}
206        * @throws IllegalStateException if {@link #grace(Duration)} is called after {@link #ofSizeAndGrace(Duration, Duration)} or {@link #ofSizeWithNoGrace(Duration)}
207        * @deprecated since 3.0. Use {@link #ofSizeAndGrace(Duration, Duration)} instead
208        */
209       @Deprecated
210       public TimeWindows grace(final Duration afterWindowEnd) throws IllegalArgumentException {
211           if (this.hasSetGrace) {
212               throw new IllegalStateException(
213                   "Cannot call grace() after setting grace value via ofSizeAndGrace or ofSizeWithNoGrace.");
214           }
215   
216           final String msgPrefix = prepareMillisCheckFailMsgPrefix(afterWindowEnd, "afterWindowEnd");
217           final long afterWindowEndMs = validateMillisecondDuration(afterWindowEnd, msgPrefix);
218   
219           return new TimeWindows(sizeMs, advanceMs, afterWindowEndMs, false);
220       }

222       @Override
223       public long gracePeriodMs() {
224           return graceMs;
225       }

227       @Override
228       public boolean equals(final Object o) {
229           if (this == o) {
230               return true;
231           }
232           if (o == null || getClass() != o.getClass()) {
233               return false;
234           }
235           final TimeWindows that = (TimeWindows) o;
236           return sizeMs == that.sizeMs &&
237               advanceMs == that.advanceMs &&
238               graceMs == that.graceMs;
239       }

242       public int hashCode() {
243           return Objects.hash(sizeMs, advanceMs, graceMs);
244       }

247       public String toString() {
248           return "TimeWindows{" +
249               ", sizeMs=" + sizeMs +
250               ", advanceMs=" + advanceMs +
251               ", graceMs=" + graceMs +
252               '}';
253       }
