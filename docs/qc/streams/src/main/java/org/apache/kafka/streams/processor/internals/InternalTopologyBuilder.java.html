302           List<String> getTopics(final Collection<String> subscribedTopics) {
303               // if it is subscribed via patterns, it is possible that the topic metadata has not been updated
304               // yet and hence the map from source node to topics is stale, in this case we put the pattern as a place holder;
305               // this should only happen for debugging since during runtime this function should always be called after the metadata has updated.
306               if (subscribedTopics.isEmpty()) {
307                   return Collections.singletonList(String.valueOf(pattern));
308               }
309   
310               final List<String> matchedTopics = new ArrayList<>();
311               for (final String update : subscribedTopics) {
312                   if (pattern == topicToPatterns.get(update)) {
313                       matchedTopics.add(update);
314                   } else if (topicToPatterns.containsKey(update) && isMatch(update)) {
315                       // the same topic cannot be matched to more than one pattern
316                       // TODO: we should lift this requirement in the future
317                       throw new TopologyException("Topic " + update +
318                           " is already matched for another regex pattern " + topicToPatterns.get(update) +
319                           " and hence cannot be matched to this regex pattern " + pattern + " any more.");
320                   } else if (isMatch(update)) {
321                       topicToPatterns.put(update, pattern);
322                       matchedTopics.add(update);
323                   }
324               }
325               return matchedTopics;
326           }
