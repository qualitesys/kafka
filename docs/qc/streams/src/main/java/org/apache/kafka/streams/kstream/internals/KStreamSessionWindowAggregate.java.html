82       private class KStreamSessionWindowAggregateProcessor extends
83           ContextualProcessor<KIn, VIn, Windowed<KIn>, Change<VAgg>> {
84   
85           private SessionStore<KIn, VAgg> store;
86           private SessionTupleForwarder<KIn, VAgg> tupleForwarder;
87           private Sensor droppedRecordsSensor;
88           private long observedStreamTime = ConsumerRecord.NO_TIMESTAMP;
89   
90           @Override
91           public void init(final ProcessorContext<Windowed<KIn>, Change<VAgg>> context) {
92               super.init(context);
93               final StreamsMetricsImpl metrics = (StreamsMetricsImpl) context.metrics();
94               final String threadId = Thread.currentThread().getName();
95               droppedRecordsSensor = droppedRecordsSensor(threadId, context.taskId().toString(),
96                   metrics);
97               store = context.getStateStore(storeName);
98               tupleForwarder = new SessionTupleForwarder<>(
99                   store,
100                   context,
101                   new SessionCacheFlushListener<>(context),
102                   sendOldValues
103               );
104           }
105   
106           @Override
107           public void process(final Record<KIn, VIn> record) {
108               // if the key is null, we do not need proceed aggregating
109               // the record with the table
110               if (record.key() == null) {
111                   if (context().recordMetadata().isPresent()) {
112                       final RecordMetadata recordMetadata = context().recordMetadata().get();
113                       LOG.warn(
114                           "Skipping record due to null key. "
115                               + "topic=[{}] partition=[{}] offset=[{}]",
116                           recordMetadata.topic(), recordMetadata.partition(), recordMetadata.offset()
117                       );
118                   } else {
119                       LOG.warn(
120                           "Skipping record due to null key. Topic, partition, and offset not known."
121                       );
122                   }
123                   droppedRecordsSensor.record();
124                   return;
125               }
126   
127               final long timestamp = record.timestamp();
128               observedStreamTime = Math.max(observedStreamTime, timestamp);
129               final long closeTime = observedStreamTime - windows.gracePeriodMs() - windows.inactivityGap();
130   
131               final List<KeyValue<Windowed<KIn>, VAgg>> merged = new ArrayList<>();
132               final SessionWindow newSessionWindow = new SessionWindow(timestamp, timestamp);
133               SessionWindow mergedWindow = newSessionWindow;
134               VAgg agg = initializer.apply();
135   
136               try (
137                   final KeyValueIterator<Windowed<KIn>, VAgg> iterator = store.findSessions(
138                       record.key(),
139                       timestamp - windows.inactivityGap(),
140                       timestamp + windows.inactivityGap()
141                   )
142               ) {
143                   while (iterator.hasNext()) {
144                       final KeyValue<Windowed<KIn>, VAgg> next = iterator.next();
145                       merged.add(next);
146                       agg = sessionMerger.apply(record.key(), agg, next.value);
147                       mergedWindow = mergeSessionWindow(mergedWindow, (SessionWindow) next.key.window());
148                   }
149               }
150   
151               if (mergedWindow.end() < closeTime) {
152                   if (context().recordMetadata().isPresent()) {
153                       final RecordMetadata recordMetadata = context().recordMetadata().get();
154                       LOG.warn(
155                           "Skipping record for expired window. " +
156                               "topic=[{}] " +
157                               "partition=[{}] " +
158                               "offset=[{}] " +
159                               "timestamp=[{}] " +
160                               "window=[{},{}] " +
161                               "expiration=[{}] " +
162                               "streamTime=[{}]",
163                           recordMetadata.topic(), recordMetadata.partition(), recordMetadata.offset(),
164                           timestamp,
165                           mergedWindow.start(), mergedWindow.end(),
166                           closeTime,
167                           observedStreamTime
168                       );
169                   } else {
170                       LOG.warn(
171                           "Skipping record for expired window. Topic, partition, and offset not known. " +
172                               "timestamp=[{}] " +
173                               "window=[{},{}] " +
174                               "expiration=[{}] " +
175                               "streamTime=[{}]",
176                           timestamp,
177                           mergedWindow.start(), mergedWindow.end(),
178                           closeTime,
179                           observedStreamTime
180                       );
181                   }
182                   droppedRecordsSensor.record();
183               } else {
184                   if (!mergedWindow.equals(newSessionWindow)) {
185                       for (final KeyValue<Windowed<KIn>, VAgg> session : merged) {
186                           store.remove(session.key);
187                           tupleForwarder.maybeForward(
188                               record.withKey(session.key)
189                                   .withValue(new Change<>(null, session.value)));
190                       }
191                   }
192   
193                   agg = aggregator.apply(record.key(), record.value(), agg);
194                   final Windowed<KIn> sessionKey = new Windowed<>(record.key(), mergedWindow);
195                   store.put(sessionKey, agg);
196                   tupleForwarder.maybeForward(
197                       record.withKey(sessionKey)
198                           .withValue(new Change<>(agg, null)));
199               }
200           }
201       }
202   
203       private SessionWindow mergeSessionWindow(final SessionWindow one, final SessionWindow two) {
204           final long start = Math.min(one.start(), two.start());
205           final long end = Math.max(one.end(), two.end());
206           return new SessionWindow(start, end);
207       }
