34   /**
35    * A RocksDB backed time-ordered segmented bytes store for window key schema.
36    */
37   public class RocksDBTimeOrderedWindowSegmentedBytesStore extends AbstractRocksDBTimeOrderedSegmentedBytesStore {
38   
39       private class WindowKeySchemaIndexToBaseStoreIterator  extends IndexToBaseStoreIterator {
40           WindowKeySchemaIndexToBaseStoreIterator(final KeyValueIterator<Bytes, byte[]> indexIterator) {
41               super(indexIterator);
42           }
43   
44           @Override
45           protected Bytes getBaseKey(final Bytes indexKey) {
46               final byte[] keyBytes = KeyFirstWindowKeySchema.extractStoreKeyBytes(indexKey.get());
47               final long timestamp = KeyFirstWindowKeySchema.extractStoreTimestamp(indexKey.get());
48               final int seqnum = KeyFirstWindowKeySchema.extractStoreSequence(indexKey.get());
49               return TimeFirstWindowKeySchema.toStoreKeyBinary(keyBytes, timestamp, seqnum);
50           }
51       }
52   
53       RocksDBTimeOrderedWindowSegmentedBytesStore(final String name,
54                                                   final String metricsScope,
55                                                   final long retention,
56                                                   final long segmentInterval,
57                                                   final boolean withIndex) {
58           super(name, metricsScope, retention, segmentInterval, new TimeFirstWindowKeySchema(),
59               Optional.ofNullable(withIndex ? new KeyFirstWindowKeySchema() : null));
60       }
61   
62       public void put(final Bytes key, final long timestamp, final int seqnum, final byte[] value) {
63           final Bytes baseKey = TimeFirstWindowKeySchema.toStoreKeyBinary(key, timestamp, seqnum);
64           put(baseKey, value);
65       }
66   
67       byte[] fetch(final Bytes key, final long timestamp, final int seqnum) {
68           return get(TimeFirstWindowKeySchema.toStoreKeyBinary(key, timestamp, seqnum));
69       }
70   
71       @Override
72       protected KeyValue<Bytes, byte[]> getIndexKeyValue(final Bytes baseKey, final byte[] baseValue) {
73           final byte[] key = TimeFirstWindowKeySchema.extractStoreKeyBytes(baseKey.get());
74           final long timestamp = TimeFirstWindowKeySchema.extractStoreTimestamp(baseKey.get());
75           final int seqnum = TimeFirstWindowKeySchema.extractStoreSequence(baseKey.get());
76   
77           return KeyValue.pair(KeyFirstWindowKeySchema.toStoreKeyBinary(key, timestamp, seqnum), new byte[0]);
78       }
79   
80       @Override
81       Map<KeyValueSegment, WriteBatch> getWriteBatches(
82           final Collection<ConsumerRecord<byte[], byte[]>> records) {
83           // advance stream time to the max timestamp in the batch
84           for (final ConsumerRecord<byte[], byte[]> record : records) {
85               final long timestamp = WindowKeySchema.extractStoreTimestamp(record.key());
86               observedStreamTime = Math.max(observedStreamTime, timestamp);
87           }
88   
89           final Map<KeyValueSegment, WriteBatch> writeBatchMap = new HashMap<>();
90           for (final ConsumerRecord<byte[], byte[]> record : records) {
91               final long timestamp = WindowKeySchema.extractStoreTimestamp(record.key());
92               final long segmentId = segments.segmentId(timestamp);
93               final KeyValueSegment segment = segments.getOrCreateSegmentIfLive(segmentId, context, observedStreamTime);
94               if (segment != null) {
95                   ChangelogRecordDeserializationHelper.applyChecksAndUpdatePosition(
96                       record,
97                       consistencyEnabled,
98                       position
99                   );
100                   try {
101                       final WriteBatch batch = writeBatchMap.computeIfAbsent(segment, s -> new WriteBatch());
102   
103                       // Assuming changelog record is serialized using WindowKeySchema
104                       // from ChangeLoggingTimestampedWindowBytesStore. Reconstruct key/value to restore
105                       if (hasIndex()) {
106                           final byte[] indexKey = KeyFirstWindowKeySchema.fromNonPrefixWindowKey(record.key());
107                           // Take care of tombstone
108                           final byte[] value = record.value() == null ? null : new byte[0];
109                           segment.addToBatch(new KeyValue<>(indexKey, value), batch);
110                       }
111   
112                       final byte[] baseKey = TimeFirstWindowKeySchema.fromNonPrefixWindowKey(record.key());
113                       segment.addToBatch(new KeyValue<>(baseKey, record.value()), batch);
114                   } catch (final RocksDBException e) {
115                       throw new ProcessorStateException("Error restoring batch to store " + name(), e);
116                   }
117               }
118           }
119           return writeBatchMap;
120       }
121   
122       @Override
123       protected IndexToBaseStoreIterator getIndexToBaseStoreIterator(
124           final SegmentIterator<KeyValueSegment> segmentIterator) {
125           return new WindowKeySchemaIndexToBaseStoreIterator(segmentIterator);
126       }
