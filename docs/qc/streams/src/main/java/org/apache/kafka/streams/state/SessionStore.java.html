28   /**
29    * Interface for storing the aggregated values of sessions.
30    * <p>
31    * The key is internally represented as {@link Windowed Windowed&lt;K&gt;} that comprises the plain
32    * key and the {@link Window} that represents window start- and end-timestamp.
33    * <p>
34    * If two sessions are merged, a new session with new start- and end-timestamp must be inserted into
35    * the store while the two old sessions must be deleted.
36    *
37    * @param <K>   type of the record keys
38    * @param <AGG> type of the aggregated values
39    */
40   public interface SessionStore<K, AGG> extends StateStore, ReadOnlySessionStore<K, AGG> {
41   
42       /**
43        * Return all the session window entries that ends between the specified range (both ends are inclusive).
44        * This function would be used to retrieve all closed and immutable windows.
45        *
46        * @param earliestSessionEndTime earliest session end time to search from, inclusive
47        * @param latestSessionEndTime latest session end time to search to, inclusive
48        */
49       default KeyValueIterator<Windowed<K>, AGG> findSessions(final long earliestSessionEndTime,
50                                                               final long latestSessionEndTime) {
51           throw new UnsupportedOperationException(
52                   "This API is not supported by this implementation of SessionStore.");
53       }
54   
55       @Override
56       default KeyValueIterator<Windowed<K>, AGG> findSessions(final K key,
57                                                               final Instant earliestSessionEndTime,
58                                                               final Instant latestSessionStartTime) {
59           return findSessions(
60               key,
61               ApiUtils.validateMillisecondInstant(earliestSessionEndTime,
62                   prepareMillisCheckFailMsgPrefix(earliestSessionEndTime, "earliestSessionEndTime")),
63               ApiUtils.validateMillisecondInstant(latestSessionStartTime,
64                   prepareMillisCheckFailMsgPrefix(latestSessionStartTime, "latestSessionStartTime")));
65       }
66   
67       @Override
68       default KeyValueIterator<Windowed<K>, AGG> backwardFindSessions(final K key,
69                                                                       final Instant earliestSessionEndTime,
70                                                                       final Instant latestSessionStartTime) {
71           return backwardFindSessions(
72               key,
73               ApiUtils.validateMillisecondInstant(earliestSessionEndTime,
74                   prepareMillisCheckFailMsgPrefix(earliestSessionEndTime, "earliestSessionEndTime")),
75               ApiUtils.validateMillisecondInstant(latestSessionStartTime,
76                   prepareMillisCheckFailMsgPrefix(latestSessionStartTime, "latestSessionStartTime")));
77       }
78   
79       default KeyValueIterator<Windowed<K>, AGG> findSessions(final K keyFrom,
80                                                               final K keyTo,
81                                                               final Instant earliestSessionEndTime,
82                                                               final Instant latestSessionStartTime) {
83           return findSessions(
84               keyFrom,
85               keyTo,
86               ApiUtils.validateMillisecondInstant(earliestSessionEndTime,
87                   prepareMillisCheckFailMsgPrefix(earliestSessionEndTime, "earliestSessionEndTime")),
88               ApiUtils.validateMillisecondInstant(latestSessionStartTime,
89                   prepareMillisCheckFailMsgPrefix(latestSessionStartTime, "latestSessionStartTime")));
90       }
91   
92       default KeyValueIterator<Windowed<K>, AGG> backwardFindSessions(final K keyFrom,
93                                                                       final K keyTo,
94                                                                       final Instant earliestSessionEndTime,
95                                                                       final Instant latestSessionStartTime) {
96           return backwardFindSessions(
97               keyFrom,
98               keyTo,
99               ApiUtils.validateMillisecondInstant(earliestSessionEndTime,
100                   prepareMillisCheckFailMsgPrefix(earliestSessionEndTime, "earliestSessionEndTime")),
101               ApiUtils.validateMillisecondInstant(latestSessionStartTime,
102                   prepareMillisCheckFailMsgPrefix(latestSessionStartTime, "latestSessionStartTime")));
103       }
104   
105       default AGG fetchSession(final K key,
106                                final Instant sessionStartTime,
107                                final Instant sessionEndTime) {
108           return fetchSession(key,
109               ApiUtils.validateMillisecondInstant(sessionStartTime,
110                   prepareMillisCheckFailMsgPrefix(sessionStartTime, "sessionStartTime")),
111               ApiUtils.validateMillisecondInstant(sessionEndTime,
112                   prepareMillisCheckFailMsgPrefix(sessionEndTime, "sessionEndTime")));
113       }
114   
115       /**
116        * Remove the session aggregated with provided {@link Windowed} key from the store
117        *
118        * @param sessionKey key of the session to remove
119        * @throws NullPointerException If null is used for sessionKey.
120        */
121       void remove(final Windowed<K> sessionKey);
122   
123       /**
124        * Write the aggregated value for the provided key to the store
125        *
126        * @param sessionKey key of the session to write
127        * @param aggregate  the aggregated value for the session, it can be null; if the serialized
128        *                   bytes are also null it is interpreted as deletes
129        * @throws NullPointerException If null is used for sessionKey.
130        */
131       void put(final Windowed<K> sessionKey, final AGG aggregate);
132   }
