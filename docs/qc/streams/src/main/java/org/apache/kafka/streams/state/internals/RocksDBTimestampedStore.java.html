61                               final String parentDir,
62                               final RocksDBMetricsRecorder metricsRecorder) {
63           super(name, parentDir, metricsRecorder);

66       @Override
67       void openRocksDB(final DBOptions dbOptions,
68                        final ColumnFamilyOptions columnFamilyOptions) {
69           final List<ColumnFamilyDescriptor> columnFamilyDescriptors = asList(
70               new ColumnFamilyDescriptor(RocksDB.DEFAULT_COLUMN_FAMILY, columnFamilyOptions),
71               new ColumnFamilyDescriptor("keyValueWithTimestamp".getBytes(StandardCharsets.UTF_8), columnFamilyOptions));
72           final List<ColumnFamilyHandle> columnFamilies = new ArrayList<>(columnFamilyDescriptors.size());
73   
74           try {
75               db = RocksDB.open(dbOptions, dbDir.getAbsolutePath(), columnFamilyDescriptors, columnFamilies);
76               setDbAccessor(columnFamilies.get(0), columnFamilies.get(1));
77           } catch (final RocksDBException e) {
78               if ("Column family not found: keyValueWithTimestamp".equals(e.getMessage())) {
79                   try {
80                       db = RocksDB.open(dbOptions, dbDir.getAbsolutePath(), columnFamilyDescriptors.subList(0, 1), columnFamilies);
81                       columnFamilies.add(db.createColumnFamily(columnFamilyDescriptors.get(1)));
82                   } catch (final RocksDBException fatal) {
83                       throw new ProcessorStateException("Error opening store " + name + " at location " + dbDir.toString(), fatal);
84                   }
85                   setDbAccessor(columnFamilies.get(0), columnFamilies.get(1));
86               } else {
87                   throw new ProcessorStateException("Error opening store " + name + " at location " + dbDir.toString(), e);
88               }
89           }
90       }

92       private void setDbAccessor(final ColumnFamilyHandle noTimestampColumnFamily,
93                                  final ColumnFamilyHandle withTimestampColumnFamily) {
94           final RocksIterator noTimestampsIter = db.newIterator(noTimestampColumnFamily);
95           noTimestampsIter.seekToFirst();
96           if (noTimestampsIter.isValid()) {
97               log.info("Opening store {} in upgrade mode", name);
98               dbAccessor = new DualColumnFamilyAccessor(noTimestampColumnFamily, withTimestampColumnFamily);
99           } else {
100               log.info("Opening store {} in regular mode", name);
101               dbAccessor = new SingleColumnFamilyAccessor(withTimestampColumnFamily);
102               noTimestampColumnFamily.close();
103           }
104           noTimestampsIter.close();
105       }

109           private final ColumnFamilyHandle oldColumnFamily;

118           @Override
119           public void put(final byte[] key,
120                           final byte[] valueWithTimestamp) {
121               if (valueWithTimestamp == null) {
122                   try {
123                       db.delete(oldColumnFamily, wOptions, key);
124                   } catch (final RocksDBException e) {
125                       // String format is happening in wrapping stores. So formatted message is thrown from wrapping stores.
126                       throw new ProcessorStateException("Error while removing key from store " + name, e);
127                   }
128                   try {
129                       db.delete(newColumnFamily, wOptions, key);
130                   } catch (final RocksDBException e) {
131                       // String format is happening in wrapping stores. So formatted message is thrown from wrapping stores.
132                       throw new ProcessorStateException("Error while removing key from store " + name, e);
133                   }
134               } else {
135                   try {
136                       db.delete(oldColumnFamily, wOptions, key);
137                   } catch (final RocksDBException e) {
138                       // String format is happening in wrapping stores. So formatted message is thrown from wrapping stores.
139                       throw new ProcessorStateException("Error while removing key from store " + name, e);
140                   }
141                   try {
142                       db.put(newColumnFamily, wOptions, key, valueWithTimestamp);
143                       StoreQueryUtils.updatePosition(position, context);
144                   } catch (final RocksDBException e) {
145                       // String format is happening in wrapping stores. So formatted message is thrown from wrapping stores.
146                       throw new ProcessorStateException("Error while putting key/value into store " + name, e);
147                   }
148               }
149           }

151           @Override
152           public void prepareBatch(final List<KeyValue<Bytes, byte[]>> entries,
153                                    final WriteBatch batch) throws RocksDBException {
154               for (final KeyValue<Bytes, byte[]> entry : entries) {
155                   Objects.requireNonNull(entry.key, "key cannot be null");
156                   addToBatch(entry.key.get(), entry.value, batch);
157               }

160           @Override
161           public byte[] get(final byte[] key) throws RocksDBException {
162               final byte[] valueWithTimestamp = db.get(newColumnFamily, key);
163               if (valueWithTimestamp != null) {
164                   return valueWithTimestamp;
165               }
166   
167               final byte[] plainValue = db.get(oldColumnFamily, key);
168               if (plainValue != null) {
169                   final byte[] valueWithUnknownTimestamp = convertToTimestampedFormat(plainValue);
170                   // this does only work, because the changelog topic contains correct data already
171                   // for other format changes, we cannot take this short cut and can only migrate data
172                   // from old to new store on put()
173                   put(key, valueWithUnknownTimestamp);
174                   return valueWithUnknownTimestamp;
175               }
176   
177               return null;
178           }
179   
180           @Override
181           public byte[] getOnly(final byte[] key) throws RocksDBException {
182               final byte[] valueWithTimestamp = db.get(newColumnFamily, key);
183               if (valueWithTimestamp != null) {
184                   return valueWithTimestamp;
185               }
186   
187               final byte[] plainValue = db.get(oldColumnFamily, key);
188               if (plainValue != null) {
189                   return convertToTimestampedFormat(plainValue);
190               }
191   
192               return null;
193           }

225           @Override
226           public KeyValueIterator<Bytes, byte[]> all(final boolean forward) {

229               if (forward) {

234                   innerIterNoTimestamp.seekToLast();

236               return new RocksDBDualCFIterator(name, innerIterWithTimestamp, innerIterNoTimestamp, forward);
237           }

293       private class RocksDBDualCFIterator extends AbstractIterator<KeyValue<Bytes, byte[]>>
294           implements KeyValueIterator<Bytes, byte[]> {
295   
296           // RocksDB's JNI interface does not expose getters/setters that allow the
297           // comparator to be pluggable, and the default is lexicographic, so it's
298           // safe to just force lexicographic comparator here for now.
299           private final Comparator<byte[]> comparator = Bytes.BYTES_LEXICO_COMPARATOR;
300   
301           private final String storeName;
302           private final RocksIterator iterWithTimestamp;
303           private final RocksIterator iterNoTimestamp;
304           private final boolean forward;
305   
306           private volatile boolean open = true;
307   
308           private byte[] nextWithTimestamp;
309           private byte[] nextNoTimestamp;
310           private KeyValue<Bytes, byte[]> next;
311   
312           RocksDBDualCFIterator(final String storeName,
313                                 final RocksIterator iterWithTimestamp,
314                                 final RocksIterator iterNoTimestamp,
315                                 final boolean forward) {
316               this.iterWithTimestamp = iterWithTimestamp;
317               this.iterNoTimestamp = iterNoTimestamp;
318               this.storeName = storeName;
319               this.forward = forward;
320           }
321   
322           @Override
323           public synchronized boolean hasNext() {
324               if (!open) {
325                   throw new InvalidStateStoreException(String.format("RocksDB iterator for store %s has closed", storeName));
326               }
327               return super.hasNext();
328           }
329   
330           @Override
331           public synchronized KeyValue<Bytes, byte[]> next() {
332               return super.next();
333           }
334   
335           @Override
336           public KeyValue<Bytes, byte[]> makeNext() {
337               if (nextNoTimestamp == null && iterNoTimestamp.isValid()) {
338                   nextNoTimestamp = iterNoTimestamp.key();
339               }
340   
341               if (nextWithTimestamp == null && iterWithTimestamp.isValid()) {
342                   nextWithTimestamp = iterWithTimestamp.key();
343               }
344   
345               if (nextNoTimestamp == null && !iterNoTimestamp.isValid()) {
346                   if (nextWithTimestamp == null && !iterWithTimestamp.isValid()) {
347                       return allDone();
348                   } else {
349                       next = KeyValue.pair(new Bytes(nextWithTimestamp), iterWithTimestamp.value());
350                       nextWithTimestamp = null;
351                       if (forward) {
352                           iterWithTimestamp.next();
353                       } else {
354                           iterWithTimestamp.prev();
355                       }
356                   }
357               } else {
358                   if (nextWithTimestamp == null) {
359                       next = KeyValue.pair(new Bytes(nextNoTimestamp), convertToTimestampedFormat(iterNoTimestamp.value()));
360                       nextNoTimestamp = null;
361                       if (forward) {
362                           iterNoTimestamp.next();
363                       } else {
364                           iterNoTimestamp.prev();
365                       }
366                   } else {
367                       if (forward) {
368                           if (comparator.compare(nextNoTimestamp, nextWithTimestamp) <= 0) {
369                               next = KeyValue.pair(new Bytes(nextNoTimestamp), convertToTimestampedFormat(iterNoTimestamp.value()));
370                               nextNoTimestamp = null;
371                               iterNoTimestamp.next();
372                           } else {
373                               next = KeyValue.pair(new Bytes(nextWithTimestamp), iterWithTimestamp.value());
374                               nextWithTimestamp = null;
375                               iterWithTimestamp.next();
376                           }
377                       } else {
378                           if (comparator.compare(nextNoTimestamp, nextWithTimestamp) >= 0) {
379                               next = KeyValue.pair(new Bytes(nextNoTimestamp), convertToTimestampedFormat(iterNoTimestamp.value()));
380                               nextNoTimestamp = null;
381                               iterNoTimestamp.prev();
382                           } else {
383                               next = KeyValue.pair(new Bytes(nextWithTimestamp), iterWithTimestamp.value());
384                               nextWithTimestamp = null;
385                               iterWithTimestamp.prev();
386                           }
387                       }
388                   }
389               }
390               return next;
391           }
392   
393           @Override
394           public synchronized void close() {
395               openIterators.remove(this);
396               iterNoTimestamp.close();
397               iterWithTimestamp.close();
398               open = false;
399           }

402           public Bytes peekNextKey() {
403               if (!hasNext()) {
404                   throw new NoSuchElementException();
405               }
406               return next.key;
407           }

410       private class RocksDBDualCFRangeIterator extends RocksDBDualCFIterator {
411           // RocksDB's JNI interface does not expose getters/setters that allow the
412           // comparator to be pluggable, and the default is lexicographic, so it's
413           // safe to just force lexicographic comparator here for now.
414           private final Comparator<byte[]> comparator = Bytes.BYTES_LEXICO_COMPARATOR;
415           private final byte[] rawLastKey;
416           private final boolean forward;
417           private final boolean toInclusive;
418   
419           RocksDBDualCFRangeIterator(final String storeName,
420                                      final RocksIterator iterWithTimestamp,
421                                      final RocksIterator iterNoTimestamp,
422                                      final Bytes from,
423                                      final Bytes to,
424                                      final boolean forward,
425                                      final boolean toInclusive) {
426               super(storeName, iterWithTimestamp, iterNoTimestamp, forward);
427               this.forward = forward;
428               this.toInclusive = toInclusive;
429               if (forward) {
430                   if (from == null) {
431                       iterWithTimestamp.seekToFirst();
432                       iterNoTimestamp.seekToFirst();
433                   } else {
434                       iterWithTimestamp.seek(from.get());
435                       iterNoTimestamp.seek(from.get());
436                   }
437                   rawLastKey = to == null ? null : to.get();
438               } else {
439                   if (to == null) {
440                       iterWithTimestamp.seekToLast();
441                       iterNoTimestamp.seekToLast();
442                   } else {
443                       iterWithTimestamp.seekForPrev(to.get());
444                       iterNoTimestamp.seekForPrev(to.get());
445                   }
446                   rawLastKey = from == null ? null : from.get();
447               }
448           }
449   
450           @Override
451           public KeyValue<Bytes, byte[]> makeNext() {
452               final KeyValue<Bytes, byte[]> next = super.makeNext();
453   
454               if (next == null) {
455                   return allDone();
456               } else if (rawLastKey == null) {
457                   //null means range endpoint is open
458                   return next;
459               } else {
460                   if (forward) {
461                       if (comparator.compare(next.key.get(), rawLastKey) < 0) {
462                           return next;
463                       } else if (comparator.compare(next.key.get(), rawLastKey) == 0) {
464                           return toInclusive ? next : allDone();
465                       } else {
466                           return allDone();
467                       }
468                   } else {
469                       if (comparator.compare(next.key.get(), rawLastKey) >= 0) {
470                           return next;
471                       } else {
472                           return allDone();
473                       }
474                   }
475               }
476           }
477       }
