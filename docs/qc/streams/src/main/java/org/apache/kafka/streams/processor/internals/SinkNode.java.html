17   package org.apache.kafka.streams.processor.internals;
18   
19   import org.apache.kafka.common.serialization.Serializer;
20   import org.apache.kafka.streams.processor.StreamPartitioner;
21   import org.apache.kafka.streams.processor.TopicNameExtractor;
22   import org.apache.kafka.streams.processor.api.Record;
23   
24   import static org.apache.kafka.streams.kstream.internals.WrappingNullableUtils.prepareKeySerializer;
25   import static org.apache.kafka.streams.kstream.internals.WrappingNullableUtils.prepareValueSerializer;
26   
27   public class SinkNode<KIn, VIn> extends ProcessorNode<KIn, VIn, Void, Void> {
28   
29       private Serializer<KIn> keySerializer;
30       private Serializer<VIn> valSerializer;
31       private final TopicNameExtractor<KIn, VIn> topicExtractor;
32       private final StreamPartitioner<? super KIn, ? super VIn> partitioner;
33   
34       private InternalProcessorContext<Void, Void> context;
35   
36       SinkNode(final String name,
37                final TopicNameExtractor<KIn, VIn> topicExtractor,
38                final Serializer<KIn> keySerializer,
39                final Serializer<VIn> valSerializer,
40                final StreamPartitioner<? super KIn, ? super VIn> partitioner) {
41           super(name);
42   
43           this.topicExtractor = topicExtractor;
44           this.keySerializer = keySerializer;
45           this.valSerializer = valSerializer;
46           this.partitioner = partitioner;
47       }
48   
49       /**
50        * @throws UnsupportedOperationException if this method adds a child to a sink node
51        */
52       @Override
53       public void addChild(final ProcessorNode<Void, Void, ?, ?> child) {
54           throw new UnsupportedOperationException("sink node does not allow addChild");
55       }
56   
57       @Override
58       public void init(final InternalProcessorContext<Void, Void> context) {
59           super.init(context);
60           this.context = context;
61           keySerializer = prepareKeySerializer(keySerializer, context, this.name());
62           valSerializer = prepareValueSerializer(valSerializer, context, this.name());
63       }
64   
65       @Override
66       public void process(final Record<KIn, VIn> record) {
67           final RecordCollector collector = ((RecordCollector.Supplier) context).recordCollector();
68   
69           final KIn key = record.key();
70           final VIn value = record.value();
71   
72           final long timestamp = record.timestamp();
73   
74           final ProcessorRecordContext contextForExtraction =
75               new ProcessorRecordContext(
76                   timestamp,
77                   context.offset(),
78                   context.partition(),
79                   context.topic(),
80                   record.headers()
81               );
82   
83           final String topic = topicExtractor.extract(key, value, contextForExtraction);
84   
85           collector.send(
86               topic,
87               key,
88               value,
89               record.headers(),
90               timestamp,
91               keySerializer,
92               valSerializer,
93               name(),
94               context,
95               partitioner);
96       }
97   
98       /**
99        * @return a string representation of this node, useful for debugging.
100        */
101       @Override
102       public String toString() {
103           return toString("");
104       }
105   
106       /**
107        * @return a string representation of this node starting with the given indent, useful for debugging.
108        */
109       @Override
110       public String toString(final String indent) {
111           final StringBuilder sb = new StringBuilder(super.toString(indent));
112           sb.append(indent).append("\ttopic:\t\t");
113           sb.append(topicExtractor);
114           sb.append("\n");
115           return sb.toString();
116       }
117   
118   }
