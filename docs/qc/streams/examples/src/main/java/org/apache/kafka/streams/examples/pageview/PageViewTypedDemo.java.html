45   /**
46    * Demonstrates how to perform a join between a KStream and a KTable, i.e. an example of a stateful computation,
47    * using specific data types (here: JSON POJO; but can also be Avro specific bindings, etc.) for serdes
48    * in Kafka Streams.
49    *
50    * In this example, we join a stream of pageviews (aka clickstreams) that reads from a topic named "streams-pageview-input"
51    * with a user profile table that reads from a topic named "streams-userprofile-input", where the data format
52    * is JSON string representing a record in the stream or table, to compute the number of pageviews per user region.
53    *
54    * Before running this example you must create the input topics and the output topic (e.g. via
55    * bin/kafka-topics --create ...), and write some data to the input topics (e.g. via
56    * bin/kafka-console-producer). Otherwise you won't see any data arriving in the output topic.
57    *
58    * The inputs for this example are:
59    * - Topic: streams-pageview-input
60    *   Key Format: (String) USER_ID
61    *   Value Format: (JSON) {"_t": "pv", "user": (String USER_ID), "page": (String PAGE_ID), "timestamp": (long ms TIMESTAMP)}
62    *
63    * - Topic: streams-userprofile-input
64    *   Key Format: (String) USER_ID
65    *   Value Format: (JSON) {"_t": "up", "region": (String REGION), "timestamp": (long ms TIMESTAMP)}
66    *
67    * To observe the results, read the output topic (e.g., via bin/kafka-console-consumer)
68    * - Topic: streams-pageviewstats-typed-output
69    *   Key Format: (JSON) {"_t": "wpvbr", "windowStart": (long ms WINDOW_TIMESTAMP), "region": (String REGION)}
70    *   Value Format: (JSON) {"_t": "rc", "count": (long REGION_COUNT), "region": (String REGION)}
71    *
72    * Note, the "_t" field is necessary to help Jackson identify the correct class for deserialization in the
73    * generic {@link JSONSerde}. If you instead specify a specific serde per class, you won't need the extra "_t" field.
74    */
75   @SuppressWarnings({"WeakerAccess", "unused"})
76   public class PageViewTypedDemo {
77   
78       /**
79        * A serde for any class that implements {@link JSONSerdeCompatible}. Note that the classes also need to
80        * be registered in the {@code @JsonSubTypes} annotation on {@link JSONSerdeCompatible}.
81        *
82        * @param <T> The concrete type of the class that gets de/serialized
83        */
84       public static class JSONSerde<T extends JSONSerdeCompatible> implements Serializer<T>, Deserializer<T>, Serde<T> {
85           private static final ObjectMapper OBJECT_MAPPER = new ObjectMapper();
86   
87           @Override
88           public void configure(final Map<String, ?> configs, final boolean isKey) {}
89   
90           @SuppressWarnings("unchecked")
91           @Override
92           public T deserialize(final String topic, final byte[] data) {
93               if (data == null) {
94                   return null;
95               }
96   
97               try {
98                   return (T) OBJECT_MAPPER.readValue(data, JSONSerdeCompatible.class);
99               } catch (final IOException e) {
100                   throw new SerializationException(e);
101               }
102           }
103   
104           @Override
105           public byte[] serialize(final String topic, final T data) {
106               if (data == null) {
107                   return null;
108               }
109   
110               try {
111                   return OBJECT_MAPPER.writeValueAsBytes(data);
112               } catch (final Exception e) {
113                   throw new SerializationException("Error serializing JSON message", e);
114               }
115           }
116   
117           @Override
118           public void close() {}
119   
120           @Override
121           public Serializer<T> serializer() {
122               return this;
123           }
124   
125           @Override
126           public Deserializer<T> deserializer() {
127               return this;
128           }
129       }
130   
131       /**
132        * An interface for registering types that can be de/serialized with {@link JSONSerde}.
133        */
134       @SuppressWarnings("DefaultAnnotationParam") // being explicit for the example
135       @JsonTypeInfo(use = JsonTypeInfo.Id.NAME, include = JsonTypeInfo.As.PROPERTY, property = "_t")
136       @JsonSubTypes({
137                         @JsonSubTypes.Type(value = PageView.class, name = "pv"),
138                         @JsonSubTypes.Type(value = UserProfile.class, name = "up"),
139                         @JsonSubTypes.Type(value = PageViewByRegion.class, name = "pvbr"),
140                         @JsonSubTypes.Type(value = WindowedPageViewByRegion.class, name = "wpvbr"),
141                         @JsonSubTypes.Type(value = RegionCount.class, name = "rc")
142                     })
143       public interface JSONSerdeCompatible {
144   
145       }
146   
147       // POJO classes
148       static public class PageView implements JSONSerdeCompatible {
149           public String user;
150           public String page;
151           public Long timestamp;
152       }
153   
154       static public class UserProfile implements JSONSerdeCompatible {
155           public String region;
156           public Long timestamp;
157       }
158   
159       static public class PageViewByRegion implements JSONSerdeCompatible {
160           public String user;
161           public String page;
162           public String region;
163       }
164   
165       static public class WindowedPageViewByRegion implements JSONSerdeCompatible {
166           public long windowStart;
167           public String region;
168       }
169   
170       static public class RegionCount implements JSONSerdeCompatible {
171           public long count;
172           public String region;
173       }
174   
175       public static void main(final String[] args) {
176           final Properties props = new Properties();
177           props.put(StreamsConfig.APPLICATION_ID_CONFIG, "streams-pageview-typed");
178           props.put(StreamsConfig.BOOTSTRAP_SERVERS_CONFIG, "localhost:9092");
179           props.put(StreamsConfig.DEFAULT_TIMESTAMP_EXTRACTOR_CLASS_CONFIG, JsonTimestampExtractor.class);
180           props.put(StreamsConfig.DEFAULT_KEY_SERDE_CLASS_CONFIG, JSONSerde.class);
181           props.put(StreamsConfig.DEFAULT_VALUE_SERDE_CLASS_CONFIG, JSONSerde.class);
182           props.put(StreamsConfig.STATESTORE_CACHE_MAX_BYTES_CONFIG, 0);
183           props.put(StreamsConfig.COMMIT_INTERVAL_MS_CONFIG, 1000L);
184   
185           // setting offset reset to earliest so that we can re-run the demo code with the same pre-loaded data
186           props.put(ConsumerConfig.AUTO_OFFSET_RESET_CONFIG, "earliest");
187   
188           final StreamsBuilder builder = new StreamsBuilder();
189   
190           final KStream<String, PageView> views = builder.stream("streams-pageview-input", Consumed.with(Serdes.String(), new JSONSerde<>()));
191   
192           final KTable<String, UserProfile> users = builder.table("streams-userprofile-input", Consumed.with(Serdes.String(), new JSONSerde<>()));
193   
194           final Duration duration24Hours = Duration.ofHours(24);
195   
196           final KStream<WindowedPageViewByRegion, RegionCount> regionCount = views
197               .leftJoin(users, (view, profile) -> {
198                   final PageViewByRegion viewByRegion = new PageViewByRegion();
199                   viewByRegion.user = view.user;
200                   viewByRegion.page = view.page;
201   
202                   if (profile != null) {
203                       viewByRegion.region = profile.region;
204                   } else {
205                       viewByRegion.region = "UNKNOWN";
206                   }
207                   return viewByRegion;
208               })
209               .map((user, viewRegion) -> new KeyValue<>(viewRegion.region, viewRegion))
210               .groupByKey(Grouped.with(Serdes.String(), new JSONSerde<>()))
211               .windowedBy(TimeWindows.ofSizeAndGrace(Duration.ofDays(7), duration24Hours).advanceBy(Duration.ofSeconds(1)))
212               .count()
213               .toStream()
214               .map((key, value) -> {
215                   final WindowedPageViewByRegion wViewByRegion = new WindowedPageViewByRegion();
216                   wViewByRegion.windowStart = key.window().start();
217                   wViewByRegion.region = key.key();
218   
219                   final RegionCount rCount = new RegionCount();
220                   rCount.region = key.key();
221                   rCount.count = value;
222   
223                   return new KeyValue<>(wViewByRegion, rCount);
224               });
225   
226           // write to the result topic
227           regionCount.to("streams-pageviewstats-typed-output", Produced.with(new JSONSerde<>(), new JSONSerde<>()));
228   
229           final KafkaStreams streams = new KafkaStreams(builder.build(), props);
230           final CountDownLatch latch = new CountDownLatch(1);
231   
232           // attach shutdown handler to catch control-c
233           Runtime.getRuntime().addShutdownHook(new Thread("streams-pipe-shutdown-hook") {
234               @Override
235               public void run() {
236                   streams.close();
237                   latch.countDown();
238               }
239           });
240   
241           try {
242               streams.start();
243               latch.await();
244           } catch (final Throwable e) {
245               e.printStackTrace();
246               System.exit(1);
247           }
248           System.exit(0);
249       }
250   }
