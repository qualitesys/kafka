18

20   import com.fasterxml.jackson.databind.node.JsonNodeFactory;
21   import com.fasterxml.jackson.databind.node.ObjectNode;
22   import java.time.Duration;
23   import org.apache.kafka.clients.consumer.ConsumerConfig;
24   import org.apache.kafka.common.serialization.Deserializer;
25   import org.apache.kafka.common.serialization.Serde;
26   import org.apache.kafka.common.serialization.Serdes;
27   import org.apache.kafka.common.serialization.Serializer;
28   import org.apache.kafka.connect.json.JsonDeserializer;
29   import org.apache.kafka.connect.json.JsonSerializer;
30   import org.apache.kafka.streams.kstream.Consumed;
31   import org.apache.kafka.streams.KafkaStreams;
32   import org.apache.kafka.streams.KeyValue;
33   import org.apache.kafka.streams.StreamsBuilder;
34   import org.apache.kafka.streams.StreamsConfig;
35   import org.apache.kafka.streams.kstream.Grouped;

39   import org.apache.kafka.streams.kstream.TimeWindows;

41   import java.util.Properties;

43   /**
44    * Demonstrates how to perform a join between a KStream and a KTable, i.e. an example of a stateful computation,
45    * using general data types (here: JSON; but can also be Avro generic bindings, etc.) for serdes
46    * in Kafka Streams.
47    *
48    * In this example, we join a stream of pageviews (aka clickstreams) that reads from  a topic named "streams-pageview-input"
49    * with a user profile table that reads from a topic named "streams-userprofile-input", where the data format
50    * is JSON string representing a record in the stream or table, to compute the number of pageviews per user region.
51    *
52    * Before running this example you must create the input topics and the output topic (e.g. via
53    * bin/kafka-topics.sh --create ...), and write some data to the input topics (e.g. via
54    * bin/kafka-console-producer.sh). Otherwise you won't see any data arriving in the output topic.
55    */
56   public class PageViewUntypedDemo {
57   
58       public static void main(final String[] args) throws Exception {
59           final Properties props = new Properties();
60           props.put(StreamsConfig.APPLICATION_ID_CONFIG, "streams-pageview-untyped");
61           props.put(StreamsConfig.BOOTSTRAP_SERVERS_CONFIG, "localhost:9092");
62           props.put(StreamsConfig.DEFAULT_TIMESTAMP_EXTRACTOR_CLASS_CONFIG, JsonTimestampExtractor.class);
63           props.put(StreamsConfig.STATESTORE_CACHE_MAX_BYTES_CONFIG, 0);
64   
65           // setting offset reset to earliest so that we can re-run the demo code with the same pre-loaded data
66           props.put(ConsumerConfig.AUTO_OFFSET_RESET_CONFIG, "earliest");
67   
68           final StreamsBuilder builder = new StreamsBuilder();
69   
70           final Serializer<JsonNode> jsonSerializer = new JsonSerializer();
71           final Deserializer<JsonNode> jsonDeserializer = new JsonDeserializer();
72           final Serde<JsonNode> jsonSerde = Serdes.serdeFrom(jsonSerializer, jsonDeserializer);
73   
74           final Consumed<String, JsonNode> consumed = Consumed.with(Serdes.String(), jsonSerde);
75           final KStream<String, JsonNode> views = builder.stream("streams-pageview-input", consumed);
76   
77           final KTable<String, JsonNode> users = builder.table("streams-userprofile-input", consumed);
78   
79           final KTable<String, String> userRegions = users.mapValues(record -> record.get("region").textValue());
80   
81           final Duration duration24Hours = Duration.ofHours(24);
82   
83           final KStream<JsonNode, JsonNode> regionCount = views
84               .leftJoin(userRegions, (view, region) -> {
85                   final ObjectNode jNode = JsonNodeFactory.instance.objectNode();
86                   return (JsonNode) jNode.put("user", view.get("user").textValue())
87                           .put("page", view.get("page").textValue())
88                           .put("region", region == null ? "UNKNOWN" : region);
89   
90               })
91               .map((user, viewRegion) -> new KeyValue<>(viewRegion.get("region").textValue(), viewRegion))
92               .groupByKey(Grouped.with(Serdes.String(), jsonSerde))
93               .windowedBy(TimeWindows.ofSizeAndGrace(Duration.ofDays(7), duration24Hours).advanceBy(Duration.ofSeconds(1)))
94               .count()
95               .toStream()
96               .map((key, value) -> {
97                   final ObjectNode keyNode = JsonNodeFactory.instance.objectNode();
98                   keyNode.put("window-start", key.window().start())
99                           .put("region", key.key());
100   
101                   final ObjectNode valueNode = JsonNodeFactory.instance.objectNode();
102                   valueNode.put("count", value);
103   
104                   return new KeyValue<>((JsonNode) keyNode, (JsonNode) valueNode);
105               });
106   
107           // write to the result topic
108           regionCount.to("streams-pageviewstats-untyped-output", Produced.with(jsonSerde, jsonSerde));
109   
110           final KafkaStreams streams = new KafkaStreams(builder.build(), props);
111           streams.start();
112   
113           // usually the stream application would be running forever,
114           // in this example we just let it run for some time and stop since the input data is finite.
115           Thread.sleep(5000L);
116   
117           streams.close();
118       }
119   }
