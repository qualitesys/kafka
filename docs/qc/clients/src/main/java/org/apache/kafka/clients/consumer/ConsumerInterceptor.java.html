21   import org.apache.kafka.common.TopicPartition;

23   import java.util.Map;

25   /**
26    * A plugin interface that allows you to intercept (and possibly mutate) records received by the consumer. A primary use-case
27    * is for third-party components to hook into the consumer applications for custom monitoring, logging, etc.
28    *
29    * <p>
30    * This class will get consumer config properties via <code>configure()</code> method, including clientId assigned
31    * by KafkaConsumer if not specified in the consumer config. The interceptor implementation needs to be aware that it will be
32    * sharing consumer config namespace with other interceptors and serializers, and ensure that there are no conflicts.
33    * <p>
34    * Exceptions thrown by ConsumerInterceptor methods will be caught, logged, but not propagated further. As a result, if
35    * the user configures the interceptor with the wrong key and value type parameters, the consumer will not throw an exception,
36    * just log the errors.
37    * <p>
38    * ConsumerInterceptor callbacks are called from the same thread that invokes
39    * {@link org.apache.kafka.clients.consumer.KafkaConsumer#poll(java.time.Duration)}.
40    * <p>
41    * Implement {@link org.apache.kafka.common.ClusterResourceListener} to receive cluster metadata once it's available. Please see the class documentation for ClusterResourceListener for more information.
42    */
43   public interface ConsumerInterceptor<K, V> extends Configurable, AutoCloseable {
44   
45       /**
46        * This is called just before the records are returned by
47        * {@link org.apache.kafka.clients.consumer.KafkaConsumer#poll(java.time.Duration)}
48        * <p>
49        * This method is allowed to modify consumer records, in which case the new records will be
50        * returned. There is no limitation on number of records that could be returned from this
51        * method. I.e., the interceptor can filter the records or generate new records.
52        * <p>
53        * Any exception thrown by this method will be caught by the caller, logged, but not propagated to the client.
54        * <p>
55        * Since the consumer may run multiple interceptors, a particular interceptor's onConsume() callback will be called
56        * in the order specified by {@link org.apache.kafka.clients.consumer.ConsumerConfig#INTERCEPTOR_CLASSES_CONFIG}.
57        * The first interceptor in the list gets the consumed records, the following interceptor will be passed the records returned
58        * by the previous interceptor, and so on. Since interceptors are allowed to modify records, interceptors may potentially get
59        * the records already modified by other interceptors. However, building a pipeline of mutable interceptors that depend on the output
60        * of the previous interceptor is discouraged, because of potential side-effects caused by interceptors potentially failing
61        * to modify the record and throwing an exception. If one of the interceptors in the list throws an exception from onConsume(),
62        * the exception is caught, logged, and the next interceptor is called with the records returned by the last successful interceptor
63        * in the list, or otherwise the original consumed records.
64        *
65        * @param records records to be consumed by the client or records returned by the previous interceptors in the list.
66        * @return records that are either modified by the interceptor or same as records passed to this method.
67        */
68       ConsumerRecords<K, V> onConsume(ConsumerRecords<K, V> records);
69   
70       /**
71        * This is called when offsets get committed.
72        * <p>
73        * Any exception thrown by this method will be ignored by the caller.
74        *
75        * @param offsets A map of offsets by partition with associated metadata
76        */
77       void onCommit(Map<TopicPartition, OffsetAndMetadata> offsets);
78   
79       /**
80        * This is called when interceptor is closed
81        */
82       void close();
83   }
