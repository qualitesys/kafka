47   /*
48    * Transport layer for SSL communication
49    *
50    *
51    * TLS v1.3 notes:
52    *   https://tools.ietf.org/html/rfc8446#section-4.6 : Post-Handshake Messages
53    *   "TLS also allows other messages to be sent after the main handshake.
54    *   These messages use a handshake content type and are encrypted under
55    *   the appropriate application traffic key."
56    */
57   public class SslTransportLayer implements TransportLayer {
58       private enum State {
59           // Initial state
60           NOT_INITIALIZED,
61           // SSLEngine is in handshake mode
62           HANDSHAKE,
63           // SSL handshake failed, connection will be terminated
64           HANDSHAKE_FAILED,
65           // SSLEngine has completed handshake, post-handshake messages may be pending for TLSv1.3
66           POST_HANDSHAKE,
67           // SSLEngine has completed handshake, any post-handshake messages have been processed for TLSv1.3
68           // For TLSv1.3, we move the channel to READY state when incoming data is processed after handshake
69           READY,
70           // Channel is being closed
71           CLOSING
72       }

173       /**
174       * Sends an SSL close message and closes socketChannel.
175       */
176       @Override
177       public void close() throws IOException {
178           State prevState = state;
179           if (state == State.CLOSING) return;
180           state = State.CLOSING;
181           sslEngine.closeOutbound();
182           try {
183               if (prevState != State.NOT_INITIALIZED && isConnected()) {
184                   if (!flush(netWriteBuffer)) {
185                       throw new IOException("Remaining data in the network buffer, can't send SSL close message.");
186                   }
187                   //prep the buffer for the close message
188                   netWriteBuffer.clear();
189                   //perform the close, since we called sslEngine.closeOutbound
190                   SSLEngineResult wrapResult = sslEngine.wrap(ByteUtils.EMPTY_BUF, netWriteBuffer);
191                   //we should be in a close state
192                   if (wrapResult.getStatus() != SSLEngineResult.Status.CLOSED) {
193                       throw new IOException("Unexpected status returned by SSLEngine.wrap, expected CLOSED, received " +
194                               wrapResult.getStatus() + ". Will not send close message to peer.");
195                   }
196                   netWriteBuffer.flip();
197                   flush(netWriteBuffer);
198               }
199           } catch (IOException ie) {
200               log.debug("Failed to send SSL Close message", ie);
201           } finally {
202               socketChannel.socket().close();
203               socketChannel.close();
204               netReadBuffer = null;
205               netWriteBuffer = null;
206               appReadBuffer = null;
207               if (fileChannelBuffer != null) {
208                   ByteBufferUnmapper.unmap("fileChannelBuffer", fileChannelBuffer);
209                   fileChannelBuffer = null;
210               }
211           }
212       }

231       /**
232       * Flushes the buffer to the network, non blocking.
233       * Visible for testing.
234       * @param buf ByteBuffer
235       * @return boolean true if the buffer has been emptied out, false otherwise
236       * @throws IOException
237       */
238       protected boolean flush(ByteBuffer buf) throws IOException {
239           int remaining = buf.remaining();
240           if (remaining > 0) {
241               int written = socketChannel.write(buf);
242               return written >= remaining;
243           }
244           return true;
245       }

247       /**
248       * Performs SSL handshake, non blocking.
249       * Before application data (kafka protocols) can be sent client & kafka broker must
250       * perform ssl handshake.
251       * During the handshake SSLEngine generates encrypted data that will be transported over socketChannel.
252       * Each SSLEngine operation generates SSLEngineResult , of which SSLEngineResult.handshakeStatus field is used to
253       * determine what operation needs to occur to move handshake along.
254       * A typical handshake might look like this.
255       * +-------------+----------------------------------+-------------+
256       * |  client     |  SSL/TLS message                 | HSStatus    |
257       * +-------------+----------------------------------+-------------+
258       * | wrap()      | ClientHello                      | NEED_UNWRAP |
259       * | unwrap()    | ServerHello/Cert/ServerHelloDone | NEED_WRAP   |
260       * | wrap()      | ClientKeyExchange                | NEED_WRAP   |
261       * | wrap()      | ChangeCipherSpec                 | NEED_WRAP   |
262       * | wrap()      | Finished                         | NEED_UNWRAP |
263       * | unwrap()    | ChangeCipherSpec                 | NEED_UNWRAP |
264       * | unwrap()    | Finished                         | FINISHED    |
265       * +-------------+----------------------------------+-------------+
266       *
267       * @throws IOException if read/write fails
268       * @throws SslAuthenticationException if handshake fails with an {@link SSLException}
269       */
270       @Override
271       public void handshake() throws IOException {
272           if (state == State.NOT_INITIALIZED) {
273               try {
274                   startHandshake();
275               } catch (SSLException e) {
276                   maybeProcessHandshakeFailure(e, false, null);
277               }
278           }
279           if (ready())
280               throw renegotiationException();
281           if (state == State.CLOSING)
282               throw closingException();
283   
284           int read = 0;
285           boolean readable = key.isReadable();
286           try {
287               // Read any available bytes before attempting any writes to ensure that handshake failures
288               // reported by the peer are processed even if writes fail (since peer closes connection
289               // if handshake fails)
290               if (readable)
291                   read = readFromSocketChannel();
292   
293               doHandshake();
294               if (ready())
295                   updateBytesBuffered(true);
296           } catch (SSLException e) {
297               maybeProcessHandshakeFailure(e, true, null);
298           } catch (IOException e) {
299               maybeThrowSslAuthenticationException();
300   
301               // This exception could be due to a write. If there is data available to unwrap in the buffer, or data available
302               // in the socket channel to read and unwrap, process the data so that any SSL handshake exceptions are reported.
303               try {
304                   do {
305                       log.trace("Process any available bytes from peer, netReadBuffer {} netWriterBuffer {} handshakeStatus {} readable? {}",
306                           netReadBuffer, netWriteBuffer, handshakeStatus, readable);
307                       handshakeWrapAfterFailure(false);
308                       handshakeUnwrap(false, true);
309                   } while (readable && readFromSocketChannel() > 0);
310               } catch (SSLException e1) {
311                   maybeProcessHandshakeFailure(e1, false, e);
312               }
313   
314               // If we get here, this is not a handshake failure, throw the original IOException
315               throw e;
316           }
317   
318           // Read from socket failed, so throw any pending handshake exception or EOF exception.
319           if (read == -1) {
320               maybeThrowSslAuthenticationException();
321               throw new EOFException("EOF during handshake, handshake status is " + handshakeStatus);
322           }
323       }
324   
325       @SuppressWarnings("fallthrough")
326       private void doHandshake() throws IOException {
327           boolean read = key.isReadable();
328           boolean write = key.isWritable();
329           handshakeStatus = sslEngine.getHandshakeStatus();
330           if (!flush(netWriteBuffer)) {
331               key.interestOps(key.interestOps() | SelectionKey.OP_WRITE);
332               return;
333           }
334           // Throw any pending handshake exception since `netWriteBuffer` has been flushed
335           maybeThrowSslAuthenticationException();
336   
337           switch (handshakeStatus) {
338               case NEED_TASK:
339                   log.trace("SSLHandshake NEED_TASK channelId {}, appReadBuffer pos {}, netReadBuffer pos {}, netWriteBuffer pos {}",
340                             channelId, appReadBuffer.position(), netReadBuffer.position(), netWriteBuffer.position());
341                   handshakeStatus = runDelegatedTasks();
342                   break;
343               case NEED_WRAP:
344                   log.trace("SSLHandshake NEED_WRAP channelId {}, appReadBuffer pos {}, netReadBuffer pos {}, netWriteBuffer pos {}",
345                             channelId, appReadBuffer.position(), netReadBuffer.position(), netWriteBuffer.position());
346                   handshakeResult = handshakeWrap(write);
347                   if (handshakeResult.getStatus() == Status.BUFFER_OVERFLOW) {
348                       int currentNetWriteBufferSize = netWriteBufferSize();
349                       netWriteBuffer.compact();
350                       netWriteBuffer = Utils.ensureCapacity(netWriteBuffer, currentNetWriteBufferSize);
351                       netWriteBuffer.flip();
352                       if (netWriteBuffer.limit() >= currentNetWriteBufferSize) {
353                           throw new IllegalStateException("Buffer overflow when available data size (" + netWriteBuffer.limit() +
354                                                           ") >= network buffer size (" + currentNetWriteBufferSize + ")");
355                       }
356                   } else if (handshakeResult.getStatus() == Status.BUFFER_UNDERFLOW) {
357                       throw new IllegalStateException("Should not have received BUFFER_UNDERFLOW during handshake WRAP.");
358                   } else if (handshakeResult.getStatus() == Status.CLOSED) {
359                       throw new EOFException();
360                   }
361                   log.trace("SSLHandshake NEED_WRAP channelId {}, handshakeResult {}, appReadBuffer pos {}, netReadBuffer pos {}, netWriteBuffer pos {}",
362                          channelId, handshakeResult, appReadBuffer.position(), netReadBuffer.position(), netWriteBuffer.position());
363                   //if handshake status is not NEED_UNWRAP or unable to flush netWriteBuffer contents
364                   //we will break here otherwise we can do need_unwrap in the same call.
365                   if (handshakeStatus != HandshakeStatus.NEED_UNWRAP || !flush(netWriteBuffer)) {
366                       key.interestOps(key.interestOps() | SelectionKey.OP_WRITE);
367                       break;
368                   }
369               case NEED_UNWRAP:
370                   log.trace("SSLHandshake NEED_UNWRAP channelId {}, appReadBuffer pos {}, netReadBuffer pos {}, netWriteBuffer pos {}",
371                             channelId, appReadBuffer.position(), netReadBuffer.position(), netWriteBuffer.position());
372                   do {
373                       handshakeResult = handshakeUnwrap(read, false);
374                       if (handshakeResult.getStatus() == Status.BUFFER_OVERFLOW) {
375                           int currentAppBufferSize = applicationBufferSize();
376                           appReadBuffer = Utils.ensureCapacity(appReadBuffer, currentAppBufferSize);
377                           if (appReadBuffer.position() > currentAppBufferSize) {
378                               throw new IllegalStateException("Buffer underflow when available data size (" + appReadBuffer.position() +
379                                                              ") > packet buffer size (" + currentAppBufferSize + ")");
380                           }
381                       }
382                   } while (handshakeResult.getStatus() == Status.BUFFER_OVERFLOW);
383                   if (handshakeResult.getStatus() == Status.BUFFER_UNDERFLOW) {
384                       int currentNetReadBufferSize = netReadBufferSize();
385                       netReadBuffer = Utils.ensureCapacity(netReadBuffer, currentNetReadBufferSize);
386                       if (netReadBuffer.position() >= currentNetReadBufferSize) {
387                           throw new IllegalStateException("Buffer underflow when there is available data");
388                       }
389                   } else if (handshakeResult.getStatus() == Status.CLOSED) {
390                       throw new EOFException("SSL handshake status CLOSED during handshake UNWRAP");
391                   }
392                   log.trace("SSLHandshake NEED_UNWRAP channelId {}, handshakeResult {}, appReadBuffer pos {}, netReadBuffer pos {}, netWriteBuffer pos {}",
393                             channelId, handshakeResult, appReadBuffer.position(), netReadBuffer.position(), netWriteBuffer.position());
394   
395                   //if handshakeStatus completed than fall-through to finished status.
396                   //after handshake is finished there is no data left to read/write in socketChannel.
397                   //so the selector won't invoke this channel if we don't go through the handshakeFinished here.
398                   if (handshakeStatus != HandshakeStatus.FINISHED) {
399                       if (handshakeStatus == HandshakeStatus.NEED_WRAP) {
400                           key.interestOps(key.interestOps() | SelectionKey.OP_WRITE);
401                       } else if (handshakeStatus == HandshakeStatus.NEED_UNWRAP) {
402                           key.interestOps(key.interestOps() & ~SelectionKey.OP_WRITE);
403                       }
404                       break;
405                   }
406               case FINISHED:
407                   handshakeFinished();
408                   break;
409               case NOT_HANDSHAKING:
410                   handshakeFinished();
411                   break;
412               default:
413                   throw new IllegalStateException(String.format("Unexpected status [%s]", handshakeStatus));
414           }
415       }

425       /**
426        * Executes the SSLEngine tasks needed.
427        * @return HandshakeStatus
428        */
429       private HandshakeStatus runDelegatedTasks() {
430           for (;;) {
431               Runnable task = delegatedTask();
432               if (task == null) {
433                   break;
434               }
435               task.run();
436           }
437           return sslEngine.getHandshakeStatus();
438       }

440       /**
441        * Checks if the handshake status is finished
442        * Sets the interestOps for the selectionKey.
443        */
444       private void handshakeFinished() throws IOException {
445           // SSLEngine.getHandshakeStatus is transient and it doesn't record FINISHED status properly.
446           // It can move from FINISHED status to NOT_HANDSHAKING after the handshake is completed.
447           // Hence we also need to check handshakeResult.getHandshakeStatus() if the handshake finished or not
448           if (handshakeResult.getHandshakeStatus() == HandshakeStatus.FINISHED) {
449               //we are complete if we have delivered the last packet
450               //remove OP_WRITE if we are complete, otherwise we still have data to write
451               if (netWriteBuffer.hasRemaining())
452                   key.interestOps(key.interestOps() | SelectionKey.OP_WRITE);
453               else {
454                   state = sslEngine.getSession().getProtocol().equals(TLS13) ? State.POST_HANDSHAKE : State.READY;
455                   key.interestOps(key.interestOps() & ~SelectionKey.OP_WRITE);
456                   SSLSession session = sslEngine.getSession();
457                   log.debug("SSL handshake completed successfully with peerHost '{}' peerPort {} peerPrincipal '{}' cipherSuite '{}'",
458                           session.getPeerHost(), session.getPeerPort(), peerPrincipal(), session.getCipherSuite());
459                   metadataRegistry.registerCipherInformation(
460                       new CipherInformation(session.getCipherSuite(),  session.getProtocol()));
461               }
462   
463               log.trace("SSLHandshake FINISHED channelId {}, appReadBuffer pos {}, netReadBuffer pos {}, netWriteBuffer pos {} ",
464                         channelId, appReadBuffer.position(), netReadBuffer.position(), netWriteBuffer.position());
465           } else {
466               throw new IOException("NOT_HANDSHAKING during handshake");
467           }
468       }

470       /**
471       * Performs the WRAP function
472       * @param doWrite boolean
473       * @return SSLEngineResult
474       * @throws IOException
475       */
476       private SSLEngineResult handshakeWrap(boolean doWrite) throws IOException {
477           log.trace("SSLHandshake handshakeWrap {}", channelId);
478           if (netWriteBuffer.hasRemaining())
479               throw new IllegalStateException("handshakeWrap called with netWriteBuffer not empty");
480           //this should never be called with a network buffer that contains data
481           //so we can clear it here.
482           netWriteBuffer.clear();
483           SSLEngineResult result;
484           try {
485               result = sslEngine.wrap(ByteUtils.EMPTY_BUF, netWriteBuffer);
486           } finally {
487               //prepare the results to be written
488               netWriteBuffer.flip();
489           }
490           handshakeStatus = result.getHandshakeStatus();
491           if (result.getStatus() == SSLEngineResult.Status.OK &&
492               result.getHandshakeStatus() == HandshakeStatus.NEED_TASK) {
493               handshakeStatus = runDelegatedTasks();
494           }
495   
496           if (doWrite) flush(netWriteBuffer);
497           return result;
498       }

500       /**
501        * Perform handshake unwrap
502        * @param doRead boolean If true, read more from the socket channel
503        * @param ignoreHandshakeStatus If true, continue to unwrap if data available regardless of handshake status
504        * @return SSLEngineResult
505        * @throws IOException
506        */
507       private SSLEngineResult handshakeUnwrap(boolean doRead, boolean ignoreHandshakeStatus) throws IOException {
508           log.trace("SSLHandshake handshakeUnwrap {}", channelId);
509           SSLEngineResult result;
510           int read = 0;
511           if (doRead)
512               read = readFromSocketChannel();
513           boolean cont;
514           do {
515               //prepare the buffer with the incoming data
516               int position = netReadBuffer.position();
517               netReadBuffer.flip();
518               result = sslEngine.unwrap(netReadBuffer, appReadBuffer);
519               netReadBuffer.compact();
520               handshakeStatus = result.getHandshakeStatus();
521               if (result.getStatus() == SSLEngineResult.Status.OK &&
522                   result.getHandshakeStatus() == HandshakeStatus.NEED_TASK) {
523                   handshakeStatus = runDelegatedTasks();
524               }
525               cont = (result.getStatus() == SSLEngineResult.Status.OK &&
526                       handshakeStatus == HandshakeStatus.NEED_UNWRAP) ||
527                       (ignoreHandshakeStatus && netReadBuffer.position() != position);
528               log.trace("SSLHandshake handshakeUnwrap: handshakeStatus {} status {}", handshakeStatus, result.getStatus());
529           } while (netReadBuffer.position() != 0 && cont);
530   
531           // Throw EOF exception for failed read after processing already received data
532           // so that handshake failures are reported correctly
533           if (read == -1)
534               throw new EOFException("EOF during handshake, handshake status is " + handshakeStatus);
535   
536           return result;
537       }

540       /**
541       * Reads a sequence of bytes from this channel into the given buffer. Reads as much as possible
542       * until either the dst buffer is full or there is no more data in the socket.
543       *
544       * @param dst The buffer into which bytes are to be transferred
545       * @return The number of bytes read, possible zero or -1 if the channel has reached end-of-stream
546       *         and no more data is available
547       * @throws IOException if some other I/O error occurs
548       */
549       @Override
550       public int read(ByteBuffer dst) throws IOException {
551           if (state == State.CLOSING) return -1;
552           else if (!ready()) return 0;
553   
554           //if we have unread decrypted data in appReadBuffer read that into dst buffer.
555           int read = 0;
556           if (appReadBuffer.position() > 0) {
557               read = readFromAppBuffer(dst);
558           }
559   
560           boolean readFromNetwork = false;
561           boolean isClosed = false;
562           // Each loop reads at most once from the socket.
563           while (dst.remaining() > 0) {
564               int netread = 0;
565               netReadBuffer = Utils.ensureCapacity(netReadBuffer, netReadBufferSize());
566               if (netReadBuffer.remaining() > 0) {
567                   netread = readFromSocketChannel();
568                   if (netread > 0)
569                       readFromNetwork = true;
570               }
571   
572               while (netReadBuffer.position() > 0) {
573                   netReadBuffer.flip();
574                   SSLEngineResult unwrapResult;
575                   try {
576                       unwrapResult = sslEngine.unwrap(netReadBuffer, appReadBuffer);
577                       if (state == State.POST_HANDSHAKE && appReadBuffer.position() != 0) {
578                           // For TLSv1.3, we have finished processing post-handshake messages since we are now processing data
579                           state = State.READY;
580                       }
581                   } catch (SSLException e) {
582                       // For TLSv1.3, handle SSL exceptions while processing post-handshake messages as authentication exceptions
583                       if (state == State.POST_HANDSHAKE) {
584                           state = State.HANDSHAKE_FAILED;
585                           throw new SslAuthenticationException("Failed to process post-handshake messages", e);
586                       } else
587                           throw e;
588                   }
589                   netReadBuffer.compact();
590                   // reject renegotiation if TLS < 1.3, key updates for TLS 1.3 are allowed
591                   if (unwrapResult.getHandshakeStatus() != HandshakeStatus.NOT_HANDSHAKING &&
592                           unwrapResult.getHandshakeStatus() != HandshakeStatus.FINISHED &&
593                           unwrapResult.getStatus() == Status.OK &&
594                           !sslEngine.getSession().getProtocol().equals(TLS13)) {
595                       log.error("Renegotiation requested, but it is not supported, channelId {}, " +
596                           "appReadBuffer pos {}, netReadBuffer pos {}, netWriteBuffer pos {} handshakeStatus {}", channelId,
597                           appReadBuffer.position(), netReadBuffer.position(), netWriteBuffer.position(), unwrapResult.getHandshakeStatus());
598                       throw renegotiationException();
599                   }
600   
601                   if (unwrapResult.getStatus() == Status.OK) {
602                       read += readFromAppBuffer(dst);
603                   } else if (unwrapResult.getStatus() == Status.BUFFER_OVERFLOW) {
604                       int currentApplicationBufferSize = applicationBufferSize();
605                       appReadBuffer = Utils.ensureCapacity(appReadBuffer, currentApplicationBufferSize);
606                       if (appReadBuffer.position() >= currentApplicationBufferSize) {
607                           throw new IllegalStateException("Buffer overflow when available data size (" + appReadBuffer.position() +
608                                                           ") >= application buffer size (" + currentApplicationBufferSize + ")");
609                       }
610   
611                       // appReadBuffer will extended upto currentApplicationBufferSize
612                       // we need to read the existing content into dst before we can do unwrap again. If there are no space in dst
613                       // we can break here.
614                       if (dst.hasRemaining())
615                           read += readFromAppBuffer(dst);
616                       else
617                           break;
618                   } else if (unwrapResult.getStatus() == Status.BUFFER_UNDERFLOW) {
619                       int currentNetReadBufferSize = netReadBufferSize();
620                       netReadBuffer = Utils.ensureCapacity(netReadBuffer, currentNetReadBufferSize);
621                       if (netReadBuffer.position() >= currentNetReadBufferSize) {
622                           throw new IllegalStateException("Buffer underflow when available data size (" + netReadBuffer.position() +
623                                                           ") > packet buffer size (" + currentNetReadBufferSize + ")");
624                       }
625                       break;
626                   } else if (unwrapResult.getStatus() == Status.CLOSED) {
627                       // If data has been read and unwrapped, return the data. Close will be handled on the next poll.
628                       if (appReadBuffer.position() == 0 && read == 0)
629                           throw new EOFException();
630                       else {
631                           isClosed = true;
632                           break;
633                       }
634                   }
635               }
636               if (read == 0 && netread < 0)
637                   throw new EOFException("EOF during read");
638               if (netread <= 0 || isClosed)
639                   break;
640           }
641           updateBytesBuffered(readFromNetwork || read > 0);
642           // If data has been read and unwrapped, return the data even if end-of-stream, channel will be closed
643           // on a subsequent poll.
644           return read;
645       }

664        * @param offset - The offset within the buffer array of the first buffer into which bytes are to be transferred; must be non-negative and no larger than dsts.length.

669       @Override

671           if ((offset < 0) || (length < 0) || (offset > dsts.length - length))

678                   int read = read(dsts[i]);
679                   if (read > 0)
680                       totalRead += read;
681                   else
682                       break;
683               }
684               if (!dsts[i].hasRemaining()) {
685                   i++;
686               }
687           }
688           return totalRead;
689       }

692       /**
693       * Writes a sequence of bytes to this channel from the given buffer.
694       *
695       * @param src The buffer from which bytes are to be retrieved
696       * @return The number of bytes read from src, possibly zero, or -1 if the channel has reached end-of-stream
697       * @throws IOException If some other I/O error occurs
698       */
699       @Override
700       public int write(ByteBuffer src) throws IOException {
701           if (state == State.CLOSING)
702               throw closingException();
703           if (!ready())
704               return 0;
705   
706           int written = 0;
707           while (flush(netWriteBuffer) && src.hasRemaining()) {
708               netWriteBuffer.clear();
709               SSLEngineResult wrapResult = sslEngine.wrap(src, netWriteBuffer);
710               netWriteBuffer.flip();
711   
712               // reject renegotiation if TLS < 1.3, key updates for TLS 1.3 are allowed
713               if (wrapResult.getHandshakeStatus() != HandshakeStatus.NOT_HANDSHAKING &&
714                       wrapResult.getStatus() == Status.OK &&
715                       !sslEngine.getSession().getProtocol().equals(TLS13)) {
716                   throw renegotiationException();
717               }
718   
719               if (wrapResult.getStatus() == Status.OK) {
720                   written += wrapResult.bytesConsumed();
721               } else if (wrapResult.getStatus() == Status.BUFFER_OVERFLOW) {
722                   // BUFFER_OVERFLOW means that the last `wrap` call had no effect, so we expand the buffer and try again
723                   netWriteBuffer = Utils.ensureCapacity(netWriteBuffer, netWriteBufferSize());
724                   netWriteBuffer.position(netWriteBuffer.limit());
725               } else if (wrapResult.getStatus() == Status.BUFFER_UNDERFLOW) {
726                   throw new IllegalStateException("SSL BUFFER_UNDERFLOW during write");
727               } else if (wrapResult.getStatus() == Status.CLOSED) {
728                   throw new EOFException();
729               }
730           }
731           return written;
732       }

736       *

738       * @param offset The offset within the buffer array of the first buffer from which bytes are to be retrieved; must be non-negative and no larger than srcs.length.
739       * @param length - The maximum number of buffers to be accessed; must be non-negative and no larger than srcs.length - offset.
740       * @return returns no.of bytes written , possibly zero.
741       * @throws IOException If some other I/O error occurs
742       */
743       @Override
744       public long write(ByteBuffer[] srcs, int offset, int length) throws IOException {
745           if ((offset < 0) || (length < 0) || (offset > srcs.length - length))

747           int totalWritten = 0;

761                   break;
762               }
763           }
764           return totalWritten;
765       }

875       /**
876        * SSL exceptions are propagated as authentication failures so that clients can avoid
877        * retries and report the failure. If `flush` is true, exceptions are propagated after
878        * any pending outgoing bytes are flushed to ensure that the peer is notified of the failure.
879        */
880       private void handshakeFailure(SSLException sslException, boolean flush) throws IOException {
881           //Release all resources such as internal buffers that SSLEngine is managing
882           log.debug("SSL Handshake failed", sslException);
883           sslEngine.closeOutbound();
884           try {
885               sslEngine.closeInbound();
886           } catch (SSLException e) {
887               log.debug("SSLEngine.closeInBound() raised an exception.", e);
888           }
889   
890           state = State.HANDSHAKE_FAILED;
891           handshakeException = new SslAuthenticationException("SSL handshake failed", sslException);
892   
893           // Attempt to flush any outgoing bytes. If flush doesn't complete, delay exception handling until outgoing bytes
894           // are flushed. If write fails because remote end has closed the channel, log the I/O exception and  continue to
895           // handle the handshake failure as an authentication exception.
896           if (!flush || handshakeWrapAfterFailure(flush))
897               throw handshakeException;
898           else
899               log.debug("Delay propagation of handshake exception till {} bytes remaining are flushed", netWriteBuffer.remaining());
900       }
901   
902       // SSL handshake failures are typically thrown as SSLHandshakeException, SSLProtocolException,
903       // SSLPeerUnverifiedException or SSLKeyException if the cause is known. These exceptions indicate
904       // authentication failures (e.g. configuration errors) which should not be retried. But the SSL engine
905       // may also throw exceptions using the base class SSLException in a few cases:
906       //   a) If there are no matching ciphers or TLS version or the private key is invalid, client will be
907       //      unable to process the server message and an SSLException is thrown:
908       //      javax.net.ssl.SSLException: Unrecognized SSL message, plaintext connection?
909       //   b) If server closes the connection gracefully during handshake, client may receive close_notify
910       //      and and an SSLException is thrown:
911       //      javax.net.ssl.SSLException: Received close_notify during handshake
912       // We want to handle a) as a non-retriable SslAuthenticationException and b) as a retriable IOException.
913       // To do this we need to rely on the exception string. Since it is safer to throw a retriable exception
914       // when we are not sure, we will treat only the first exception string as a handshake exception.
915       private void maybeProcessHandshakeFailure(SSLException sslException, boolean flush, IOException ioException) throws IOException {
916           if (sslException instanceof SSLHandshakeException || sslException instanceof SSLProtocolException ||
917                   sslException instanceof SSLPeerUnverifiedException || sslException instanceof SSLKeyException ||
918                   sslException.getMessage().contains("Unrecognized SSL message") ||
919                   sslException.getMessage().contains("Received fatal alert: "))
920               handshakeFailure(sslException, flush);
921           else if (ioException == null)
922               throw sslException;
923           else {
924               log.debug("SSLException while unwrapping data after IOException, original IOException will be propagated", sslException);
925               throw ioException;
926           }
927       }

930       private void maybeThrowSslAuthenticationException() {
931           if (handshakeException != null)
932               throw handshakeException;
933       }

935       /**
936        * Perform handshake wrap after an SSLException or any IOException.
937        *
938        * If `doWrite=false`, we are processing IOException after peer has disconnected, so we
939        * cannot send any more data. We perform any pending wraps so that we can unwrap any
940        * peer data that is already available.
941        *
942        * If `doWrite=true`, we are processing SSLException, we perform wrap and flush
943        * any data to notify the peer of the handshake failure.
944        *
945        * Returns true if no more wrap is required and any data is flushed or discarded.
946        */
947       private boolean handshakeWrapAfterFailure(boolean doWrite) {
948           try {
949               log.trace("handshakeWrapAfterFailure status {} doWrite {}", handshakeStatus, doWrite);
950               while (handshakeStatus == HandshakeStatus.NEED_WRAP && (!doWrite || flush(netWriteBuffer))) {
951                   if (!doWrite)
952                       clearWriteBuffer();
953                   handshakeWrap(doWrite);
954               }
955           } catch (Exception e) {
956               log.debug("Failed to wrap and flush all bytes before closing channel", e);
957               clearWriteBuffer();
958           }
959           if (!doWrite)
960               clearWriteBuffer();
961           return !netWriteBuffer.hasRemaining();
962       }
963   
964       private void clearWriteBuffer() {
965           if (netWriteBuffer.hasRemaining())
966               log.debug("Discarding write buffer {} since peer has disconnected", netWriteBuffer);
967           netWriteBuffer.position(0);
968           netWriteBuffer.limit(0);
969       }

992       @Override
993       public long transferFrom(FileChannel fileChannel, long position, long count) throws IOException {
994           if (state == State.CLOSING)
995               throw closingException();
996           if (state != State.READY)
997               return 0;
998   
999           if (!flush(netWriteBuffer))
1000               return 0;
1001   
1002           long channelSize = fileChannel.size();
1003           if (position > channelSize)
1004               return 0;
1005           int totalBytesToWrite = (int) Math.min(Math.min(count, channelSize - position), Integer.MAX_VALUE);
1006   
1007           if (fileChannelBuffer == null) {
1008               // Pick a size that allows for reasonably efficient disk reads, keeps the memory overhead per connection
1009               // manageable and can typically be drained in a single `write` call. The `netWriteBuffer` is typically 16k
1010               // and the socket send buffer is 100k by default, so 32k is a good number given the mentioned trade-offs.
1011               int transferSize = 32768;
1012               // Allocate a direct buffer to avoid one heap to heap buffer copy. SSLEngine copies the source
1013               // buffer (fileChannelBuffer) to the destination buffer (netWriteBuffer) and then encrypts in-place.
1014               // FileChannel.read() to a heap buffer requires a copy from a direct buffer to a heap buffer, which is not
1015               // useful here.
1016               fileChannelBuffer = ByteBuffer.allocateDirect(transferSize);
1017               // The loop below drains any remaining bytes from the buffer before reading from disk, so we ensure there
1018               // are no remaining bytes in the empty buffer
1019               fileChannelBuffer.position(fileChannelBuffer.limit());
1020           }
1021   
1022           int totalBytesWritten = 0;
1023           long pos = position;
1024           try {
1025               while (totalBytesWritten < totalBytesToWrite) {
1026                   if (!fileChannelBuffer.hasRemaining()) {
1027                       fileChannelBuffer.clear();
1028                       int bytesRemaining = totalBytesToWrite - totalBytesWritten;
1029                       if (bytesRemaining < fileChannelBuffer.limit())
1030                           fileChannelBuffer.limit(bytesRemaining);
1031                       int bytesRead = fileChannel.read(fileChannelBuffer, pos);
1032                       if (bytesRead <= 0)
1033                           break;
1034                       fileChannelBuffer.flip();
1035                   }
1036                   int networkBytesWritten = write(fileChannelBuffer);
1037                   totalBytesWritten += networkBytesWritten;
1038                   // In the case of a partial write we only return the written bytes to the caller. As a result, the
1039                   // `position` passed in the next `transferFrom` call won't include the bytes remaining in
1040                   // `fileChannelBuffer`. By draining `fileChannelBuffer` first, we ensure we update `pos` before
1041                   // we invoke `fileChannel.read`.
1042                   if (fileChannelBuffer.hasRemaining())
1043                       break;
1044                   pos += networkBytesWritten;
1045               }
1046               return totalBytesWritten;
1047           } catch (IOException e) {
1048               if (totalBytesWritten > 0)
1049                   return totalBytesWritten;
1050               throw e;
1051           }
1052       }
1053   }
