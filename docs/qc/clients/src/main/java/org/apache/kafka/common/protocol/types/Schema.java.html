17   package org.apache.kafka.common.protocol.types;
18   
19   import java.nio.ByteBuffer;
20   import java.util.HashMap;
21   import java.util.Map;
22   import java.util.Objects;
23   
24   /**
25    * The schema for a compound record definition
26    */
27   public class Schema extends Type {
28       private final static Object[] NO_VALUES = new Object[0];
29   
30       private final BoundField[] fields;
31       private final Map<String, BoundField> fieldsByName;
32       private final boolean tolerateMissingFieldsWithDefaults;
33       private final Struct cachedStruct;
34   
35       /**
36        * Construct the schema with a given list of its field values
37        *
38        * @param fs the fields of this schema
39        *
40        * @throws SchemaException If the given list have duplicate fields
41        */
42       public Schema(Field... fs) {
43           this(false, fs);
44       }
45   
46       /**
47        * Construct the schema with a given list of its field values and the ability to tolerate
48        * missing optional fields with defaults at the end of the schema definition.
49        *
50        * @param tolerateMissingFieldsWithDefaults whether to accept records with missing optional
51        * fields the end of the schema
52        * @param fs the fields of this schema
53        *
54        * @throws SchemaException If the given list have duplicate fields
55        */
56       public Schema(boolean tolerateMissingFieldsWithDefaults, Field... fs) {
57           this.fields = new BoundField[fs.length];
58           this.fieldsByName = new HashMap<>();
59           this.tolerateMissingFieldsWithDefaults = tolerateMissingFieldsWithDefaults;
60           for (int i = 0; i < this.fields.length; i++) {
61               Field def = fs[i];
62               if (fieldsByName.containsKey(def.name))
63                   throw new SchemaException("Schema contains a duplicate field: " + def.name);
64               this.fields[i] = new BoundField(def, this, i);
65               this.fieldsByName.put(def.name, this.fields[i]);
66           }
67           //6 schemas have no fields at the time of this writing (3 versions each of list_groups and api_versions)
68           //for such schemas there's no point in even creating a unique Struct object when deserializing.
69           this.cachedStruct = this.fields.length > 0 ? null : new Struct(this, NO_VALUES);
70       }
71   
72       /**
73        * Write a struct to the buffer
74        */
75       @Override
76       public void write(ByteBuffer buffer, Object o) {
77           Struct r = (Struct) o;
78           for (BoundField field : fields) {
79               try {
80                   Object value = field.def.type.validate(r.get(field));
81                   field.def.type.write(buffer, value);
82               } catch (Exception e) {
83                   throw new SchemaException("Error writing field '" + field.def.name + "': " +
84                                             (e.getMessage() == null ? e.getClass().getName() : e.getMessage()));
85               }
86           }
87       }
88   
89       /**
90        * Read a struct from the buffer. If this schema is configured to tolerate missing
91        * optional fields at the end of the buffer, these fields are replaced with their default
92        * values; otherwise, if the schema does not tolerate missing fields, or if missing fields
93        * don't have a default value, a {@code SchemaException} is thrown to signify that mandatory
94        * fields are missing.
95        */
96       @Override
97       public Struct read(ByteBuffer buffer) {
98           if (cachedStruct != null) {
99               return cachedStruct;
100           }
101           Object[] objects = new Object[fields.length];
102           for (int i = 0; i < fields.length; i++) {
103               try {
104                   if (tolerateMissingFieldsWithDefaults) {
105                       if (buffer.hasRemaining()) {
106                           objects[i] = fields[i].def.type.read(buffer);
107                       } else if (fields[i].def.hasDefaultValue) {
108                           objects[i] = fields[i].def.defaultValue;
109                       } else {
110                           throw new SchemaException("Missing value for field '" + fields[i].def.name +
111                                   "' which has no default value.");
112                       }
113                   } else {
114                       objects[i] = fields[i].def.type.read(buffer);
115                   }
116               } catch (Exception e) {
117                   throw new SchemaException("Error reading field '" + fields[i].def.name + "': " +
118                                             (e.getMessage() == null ? e.getClass().getName() : e.getMessage()));
119               }
120           }
121           return new Struct(this, objects);
122       }
123   
124       /**
125        * The size of the given record
126        */
127       @Override
128       public int sizeOf(Object o) {
129           int size = 0;
130           Struct r = (Struct) o;
131           for (BoundField field : fields) {
132               try {
133                   size += field.def.type.sizeOf(r.get(field));
134               } catch (Exception e) {
135                   throw new SchemaException("Error computing size for field '" + field.def.name + "': " +
136                           (e.getMessage() == null ? e.getClass().getName() : e.getMessage()));
137               }
138           }
139           return size;
140       }
141   
142       /**
143        * The number of fields in this schema
144        */
145       public int numFields() {
146           return this.fields.length;
147       }
148   
149       /**
150        * Get a field by its slot in the record array
151        *
152        * @param slot The slot at which this field sits
153        * @return The field
154        */
155       public BoundField get(int slot) {
156           return this.fields[slot];
157       }
158   
159       /**
160        * Get a field by its name
161        *
162        * @param name The name of the field
163        * @return The field
164        */
165       public BoundField get(String name) {
166           return this.fieldsByName.get(name);
167       }
168   
169       /**
170        * Get all the fields in this schema
171        */
172       public BoundField[] fields() {
173           return this.fields;
174       }
175   
176       /**
177        * Display a string representation of the schema
178        */
179       @Override
180       public String toString() {
181           StringBuilder b = new StringBuilder();
182           b.append('{');
183           for (int i = 0; i < this.fields.length; i++) {
184               b.append(this.fields[i].toString());
185               if (i < this.fields.length - 1)
186                   b.append(',');
187           }
188           b.append("}");
189           return b.toString();
190       }
191   
192       @Override
193       public Struct validate(Object item) {
194           try {
195               Struct struct = (Struct) item;
196               for (BoundField field : fields) {
197                   try {
198                       field.def.type.validate(struct.get(field));
199                   } catch (SchemaException e) {
200                       throw new SchemaException("Invalid value for field '" + field.def.name + "': " + e.getMessage());
201                   }
202               }
203               return struct;
204           } catch (ClassCastException e) {
205               throw new SchemaException("Not a Struct.");
206           }
207       }
208   
209       public void walk(Visitor visitor) {
210           Objects.requireNonNull(visitor, "visitor must be non-null");
211           handleNode(this, visitor);
212       }
213   
214       private static void handleNode(Type node, Visitor visitor) {
215           if (node instanceof Schema) {
216               Schema schema = (Schema) node;
217               visitor.visit(schema);
218               for (BoundField f : schema.fields())
219                   handleNode(f.def.type, visitor);
220           } else if (node.isArray()) {
221               visitor.visit(node);
222               handleNode(node.arrayElementType().get(), visitor);
223           } else {
224               visitor.visit(node);
225           }
226       }
227   
228       /**
229        * Override one or more of the visit methods with the desired logic.
230        */
231       public static abstract class Visitor {
232           public void visit(Schema schema) {}
233           public void visit(Type field) {}
234       }
