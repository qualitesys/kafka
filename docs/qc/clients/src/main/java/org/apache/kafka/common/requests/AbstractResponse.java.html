58       /**
59        * The number of each type of error in the response, including {@link Errors#NONE} and top-level errors as well as
60        * more specifically scoped errors (such as topic or partition-level errors).
61        * @return A count of errors.
62        */
63       public abstract Map<Errors, Integer> errorCounts();
64   
65       protected Map<Errors, Integer> errorCounts(Errors error) {
66           return Collections.singletonMap(error, 1);
67       }
68   
69       protected Map<Errors, Integer> errorCounts(Stream<Errors> errors) {
70           return errors.collect(Collectors.groupingBy(e -> e, Collectors.summingInt(e -> 1)));
71       }
72   
73       protected Map<Errors, Integer> errorCounts(Collection<Errors> errors) {
74           Map<Errors, Integer> errorCounts = new HashMap<>();
75           for (Errors error : errors)
76               updateErrorCounts(errorCounts, error);
77           return errorCounts;
78       }
79   
80       protected Map<Errors, Integer> apiErrorCounts(Map<?, ApiError> errors) {
81           Map<Errors, Integer> errorCounts = new HashMap<>();
82           for (ApiError apiError : errors.values())
83               updateErrorCounts(errorCounts, apiError.error());
84           return errorCounts;
85       }
86   
87       protected void updateErrorCounts(Map<Errors, Integer> errorCounts, Errors error) {
88           Integer count = errorCounts.getOrDefault(error, 0);
89           errorCounts.put(error, count + 1);
90       }

92       /**
93        * Parse a response from the provided buffer. The buffer is expected to hold both
94        * the {@link ResponseHeader} as well as the response payload.
95        */
96       public static AbstractResponse parseResponse(ByteBuffer buffer, RequestHeader requestHeader) {
97           ApiKeys apiKey = requestHeader.apiKey();
98           short apiVersion = requestHeader.apiVersion();
99   
100           ResponseHeader responseHeader = ResponseHeader.parse(buffer, apiKey.responseHeaderVersion(apiVersion));
101   
102           if (requestHeader.correlationId() != responseHeader.correlationId()) {
103               throw new CorrelationIdMismatchException("Correlation id for response ("
104                   + responseHeader.correlationId() + ") does not match request ("
105                   + requestHeader.correlationId() + "), request header: " + requestHeader,
106                   requestHeader.correlationId(), responseHeader.correlationId());
107           }
108   
109           return AbstractResponse.parseResponse(apiKey, buffer, apiVersion);
110       }
