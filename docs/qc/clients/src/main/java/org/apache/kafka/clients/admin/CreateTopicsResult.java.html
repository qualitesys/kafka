33   @InterfaceStability.Evolving
34   public class CreateTopicsResult {
35       final static int UNKNOWN = -1;
36   
37       private final Map<String, KafkaFuture<TopicMetadataAndConfig>> futures;
38   
39       protected CreateTopicsResult(Map<String, KafkaFuture<TopicMetadataAndConfig>> futures) {
40           this.futures = futures;
41       }
42   
43       /**
44        * Return a map from topic names to futures, which can be used to check the status of individual
45        * topic creations.
46        */
47       public Map<String, KafkaFuture<Void>> values() {
48           return futures.entrySet().stream()
49                   .collect(Collectors.toMap(Map.Entry::getKey, e -> e.getValue().thenApply(v -> (Void) null)));
50       }
51   
52       /**
53        * Return a future which succeeds if all the topic creations succeed.
54        */
55       public KafkaFuture<Void> all() {
56           return KafkaFuture.allOf(futures.values().toArray(new KafkaFuture[0]));
57       }
58   
59       /**
60        * Returns a future that provides topic configs for the topic when the request completes.
61        * <p>

63        * {@link org.apache.kafka.common.errors.UnsupportedVersionException}.

66        * have permission to describe topic configs.
67        */
68       public KafkaFuture<Config> config(String topic) {
69           return futures.get(topic).thenApply(TopicMetadataAndConfig::config);
70       }
71

111       public static class TopicMetadataAndConfig {
112           private final ApiException exception;
113           private final Uuid topicId;
114           private final int numPartitions;
115           private final int replicationFactor;
116           private final Config config;
117   
118           public TopicMetadataAndConfig(Uuid topicId, int numPartitions, int replicationFactor, Config config) {
119               this.exception = null;
120               this.topicId = topicId;
121               this.numPartitions = numPartitions;
122               this.replicationFactor = replicationFactor;
123               this.config = config;
124           }
125   
126           public TopicMetadataAndConfig(ApiException exception) {
127               this.exception = exception;
128               this.topicId = Uuid.ZERO_UUID;
129               this.numPartitions = UNKNOWN;
130               this.replicationFactor = UNKNOWN;
131               this.config = null;
132           }
133
