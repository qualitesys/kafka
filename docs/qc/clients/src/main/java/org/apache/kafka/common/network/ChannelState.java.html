17   package org.apache.kafka.common.network;
18   
19   import org.apache.kafka.common.errors.AuthenticationException;
20   
21   /**
22    * States for KafkaChannel:
23    * <ul>
24    *   <li>NOT_CONNECTED: Connections are created in NOT_CONNECTED state. State is updated
25    *       on {@link TransportLayer#finishConnect()} when socket connection is established.
26    *       PLAINTEXT channels transition from NOT_CONNECTED to READY, others transition
27    *       to AUTHENTICATE. Failures in NOT_CONNECTED state typically indicate that the
28    *       remote endpoint is unavailable, which may be due to misconfigured endpoints.</li>
29    *   <li>AUTHENTICATE: SSL, SASL_SSL and SASL_PLAINTEXT channels are in AUTHENTICATE state during SSL and
30    *       SASL handshake. Disconnections in AUTHENTICATE state may indicate that authentication failed with
31    *       SSL or SASL (broker version < 1.0.0). Channels transition to READY state when authentication completes
32    *       successfully.</li>
33    *   <li>READY: Connected, authenticated channels are in READY state. Channels may transition from
34    *       READY to EXPIRED, FAILED_SEND or LOCAL_CLOSE.</li>
35    *   <li>EXPIRED: Idle connections are moved to EXPIRED state on idle timeout and the channel is closed.</li>
36    *   <li>FAILED_SEND: Channels transition from READY to FAILED_SEND state if the channel is closed due
37    *       to a send failure.</li>
38    *   <li>AUTHENTICATION_FAILED: Channels are moved to this state if the requested SASL mechanism is not
39    *       enabled in the broker or when brokers with versions 1.0.0 and above provide an error response
40    *       during SASL authentication. {@link #exception()} gives the reason provided by the broker for
41    *       authentication failure.</li>
42    *   <li>LOCAL_CLOSE: Channels are moved to LOCAL_CLOSE state if close() is initiated locally.</li>
43    * </ul>
44    * If the remote endpoint closes a channel, the state of the channel reflects the state the channel
45    * was in at the time of disconnection. This state may be useful to identify the reason for disconnection.
46    * <p>
47    * Typical transitions:
48    * <ul>
49    *   <li>PLAINTEXT Good path: NOT_CONNECTED => READY => LOCAL_CLOSE</li>
50    *   <li>SASL/SSL Good path: NOT_CONNECTED => AUTHENTICATE => READY => LOCAL_CLOSE</li>
51    *   <li>Bootstrap server misconfiguration: NOT_CONNECTED, disconnected in NOT_CONNECTED state</li>
52    *   <li>Security misconfiguration: NOT_CONNECTED => AUTHENTICATE => AUTHENTICATION_FAILED, disconnected in AUTHENTICATION_FAILED state</li>
53    *   <li>Security misconfiguration with older broker: NOT_CONNECTED => AUTHENTICATE, disconnected in AUTHENTICATE state</li>
54    * </ul>
55    */
56   public class ChannelState {
57       public enum State {
58           NOT_CONNECTED,
59           AUTHENTICATE,
60           READY,
61           EXPIRED,
62           FAILED_SEND,
63           AUTHENTICATION_FAILED,
64           LOCAL_CLOSE
65       }
66   
67       // AUTHENTICATION_FAILED has a custom exception. For other states,
68       // create a reusable `ChannelState` instance per-state.
69       public static final ChannelState NOT_CONNECTED = new ChannelState(State.NOT_CONNECTED);
70       public static final ChannelState AUTHENTICATE = new ChannelState(State.AUTHENTICATE);
71       public static final ChannelState READY = new ChannelState(State.READY);
72       public static final ChannelState EXPIRED = new ChannelState(State.EXPIRED);
73       public static final ChannelState FAILED_SEND = new ChannelState(State.FAILED_SEND);
74       public static final ChannelState LOCAL_CLOSE = new ChannelState(State.LOCAL_CLOSE);
75   
76       private final State state;
77       private final AuthenticationException exception;
78       private final String remoteAddress;
79   
80       public ChannelState(State state) {
81           this(state, null, null);
82       }
83   
84       public ChannelState(State state, String remoteAddress) {
85           this(state, null, remoteAddress);
86       }
87       
88       public ChannelState(State state, AuthenticationException exception, String remoteAddress) {
89           this.state = state;
90           this.exception = exception;
91           this.remoteAddress = remoteAddress;
92       }
93   
94       public State state() {
95           return state;
96       }
97   
98       public AuthenticationException exception() {
99           return exception;
100       }
101   
102       public String remoteAddress() {
103           return remoteAddress;
104       }
105   }
