30   public abstract class AbstractRequest implements AbstractRequestResponse {
31   
32       public static abstract class Builder<T extends AbstractRequest> {
33           private final ApiKeys apiKey;
34           private final short oldestAllowedVersion;
35           private final short latestAllowedVersion;
36   
37           /**
38            * Construct a new builder which allows any supported version
39            */
40           public Builder(ApiKeys apiKey) {
41               this(apiKey, apiKey.oldestVersion(), apiKey.latestVersion());
42           }
43   
44           /**
45            * Construct a new builder which allows only a specific version
46            */
47           public Builder(ApiKeys apiKey, short allowedVersion) {
48               this(apiKey, allowedVersion, allowedVersion);
49           }
50   
51           /**
52            * Construct a new builder which allows an inclusive range of versions
53            */
54           public Builder(ApiKeys apiKey, short oldestAllowedVersion, short latestAllowedVersion) {
55               this.apiKey = apiKey;
56               this.oldestAllowedVersion = oldestAllowedVersion;
57               this.latestAllowedVersion = latestAllowedVersion;
58           }
59   
60           public ApiKeys apiKey() {
61               return apiKey;
62           }
63   
64           public short oldestAllowedVersion() {
65               return oldestAllowedVersion;
66           }
67   
68           public short latestAllowedVersion() {
69               return latestAllowedVersion;
70           }
71   
72           public T build() {

79       private final short version;

82       public AbstractRequest(ApiKeys apiKey, short version) {
83           if (!apiKey.isVersionSupported(version))
84               throw new UnsupportedVersionException("The " + apiKey + " protocol does not support version " + version);
85           this.version = version;
86           this.apiKey = apiKey;
87       }
88   
89       /**
90        * Get the version of this AbstractRequest object.
91        */
92       public short version() {
93           return version;
94       }
95   
96       public ApiKeys apiKey() {
97           return apiKey;
98       }
99   
100       public final Send toSend(RequestHeader header) {
101           return SendBuilder.buildRequestSend(header, data());
102       }
103   
104       /**
105        * Serializes header and body without prefixing with size (unlike `toSend`, which does include a size prefix).
106        */
107       public final ByteBuffer serializeWithHeader(RequestHeader header) {
108           if (header.apiKey() != apiKey) {
109               throw new IllegalArgumentException("Could not build request " + apiKey + " with header api key " + header.apiKey());
110           }
111           if (header.apiVersion() != version) {
112               throw new IllegalArgumentException("Could not build request version " + version + " with header version " + header.apiVersion());
113           }
114           return RequestUtils.serialize(header.data(), header.headerVersion(), data(), version);
115       }
116   
117       // Visible for testing
118       public final ByteBuffer serialize() {
119           return MessageUtil.toByteBuffer(data(), version);
120       }
121   
122       // Visible for testing
123       final int sizeInBytes() {
124           return data().size(new ObjectSerializationCache(), version);
125       }

127       public String toString(boolean verbose) {
128           return data().toString();
129       }

139       public AbstractResponse getErrorResponse(Throwable e) {
140           return getErrorResponse(AbstractResponse.DEFAULT_THROTTLE_TIME, e);

142   
143       /**
144        * Get an error response for a request with specified throttle time in the response if applicable
145        */
146       public abstract AbstractResponse getErrorResponse(int throttleTimeMs, Throwable e);
147   
148       /**
149        * Get the error counts corresponding to an error response. This is overridden for requests
150        * where response may be null (e.g produce with acks=0).
151        */
152       public Map<Errors, Integer> errorCounts(Throwable e) {
153           AbstractResponse response = getErrorResponse(0, e);
154           if (response == null)
155               throw new IllegalStateException("Error counts could not be obtained for request " + this);
156           else
157               return response.errorCounts();
158       }

160       /**
161        * Factory method for getting a request object based on ApiKey ID and a version
162        */
163       public static RequestAndSize parseRequest(ApiKeys apiKey, short apiVersion, ByteBuffer buffer) {
164           int bufferSize = buffer.remaining();
165           return new RequestAndSize(doParseRequest(apiKey, apiVersion, buffer), bufferSize);
166       }
