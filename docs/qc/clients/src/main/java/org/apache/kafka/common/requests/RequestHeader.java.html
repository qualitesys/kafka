17   package org.apache.kafka.common.requests;
18   
19   import org.apache.kafka.common.errors.InvalidRequestException;
20   import org.apache.kafka.common.errors.UnsupportedVersionException;
21   import org.apache.kafka.common.message.RequestHeaderData;
22   import org.apache.kafka.common.protocol.ApiKeys;
23   import org.apache.kafka.common.protocol.ByteBufferAccessor;
24   import org.apache.kafka.common.protocol.ObjectSerializationCache;
25   
26   import java.nio.ByteBuffer;
27   
28   /**
29    * The header for a request in the Kafka protocol
30    */
31   public class RequestHeader implements AbstractRequestResponse {
32       private final RequestHeaderData data;
33       private final short headerVersion;
34   
35       public RequestHeader(ApiKeys requestApiKey, short requestVersion, String clientId, int correlationId) {
36           this(new RequestHeaderData().
37                   setRequestApiKey(requestApiKey.id).
38                   setRequestApiVersion(requestVersion).
39                   setClientId(clientId).
40                   setCorrelationId(correlationId),
41               requestApiKey.requestHeaderVersion(requestVersion));
42       }
43   
44       public RequestHeader(RequestHeaderData data, short headerVersion) {
45           this.data = data;
46           this.headerVersion = headerVersion;
47       }
48   
49       public ApiKeys apiKey() {
50           return ApiKeys.forId(data.requestApiKey());
51       }
52   
53       public short apiVersion() {
54           return data.requestApiVersion();
55       }
56   
57       public short headerVersion() {
58           return headerVersion;
59       }
60   
61       public String clientId() {
62           return data.clientId();
63       }
64   
65       public int correlationId() {
66           return data.correlationId();
67       }
68   
69       public RequestHeaderData data() {
70           return data;
71       }
72   
73       public void write(ByteBuffer buffer, ObjectSerializationCache serializationCache) {
74           data.write(new ByteBufferAccessor(buffer), serializationCache, headerVersion);
75       }
76   
77       public int size(ObjectSerializationCache serializationCache) {
78           return data.size(serializationCache, headerVersion);
79       }
80   
81       public ResponseHeader toResponseHeader() {
82           return new ResponseHeader(data.correlationId(), apiKey().responseHeaderVersion(apiVersion()));
83       }
84   
85       public static RequestHeader parse(ByteBuffer buffer) {
86           short apiKey = -1;
87           try {
88               // We derive the header version from the request api version, so we read that first.
89               // The request api version is part of `RequestHeaderData`, so we reset the buffer position after the read.
90               int position = buffer.position();
91               apiKey = buffer.getShort();
92               short apiVersion = buffer.getShort();
93               short headerVersion = ApiKeys.forId(apiKey).requestHeaderVersion(apiVersion);
94               buffer.position(position);
95               RequestHeaderData headerData = new RequestHeaderData(
96                   new ByteBufferAccessor(buffer), headerVersion);
97               // Due to a quirk in the protocol, client ID is marked as nullable.
98               // However, we treat a null client ID as equivalent to an empty client ID.
99               if (headerData.clientId() == null) {
100                   headerData.setClientId("");
101               }
102               return new RequestHeader(headerData, headerVersion);
103           } catch (UnsupportedVersionException e) {
104               throw new InvalidRequestException("Unknown API key " + apiKey, e);
105           } catch (Throwable ex) {
106               throw new InvalidRequestException("Error parsing request header. Our best guess of the apiKey is: " +
107                       apiKey, ex);
108           }
109       }
110   
111       @Override
112       public String toString() {
113           return "RequestHeader(apiKey=" + apiKey() +
114                   ", apiVersion=" + apiVersion() +
115                   ", clientId=" + clientId() +
116                   ", correlationId=" + correlationId() +
117                   ")";
118       }
119   
120       @Override
121       public boolean equals(Object o) {
122           if (this == o) return true;
123           if (o == null || getClass() != o.getClass()) return false;
124           RequestHeader that = (RequestHeader) o;
125           return this.data.equals(that.data);
126       }
127   
128       @Override
129       public int hashCode() {
130           return this.data.hashCode();
131       }
132   }
