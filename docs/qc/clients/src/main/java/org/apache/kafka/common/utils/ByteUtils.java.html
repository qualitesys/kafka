389       /**
390        * Number of bytes needed to encode an integer in unsigned variable-length format.
391        *
392        * @param value The signed value
393        *
394        * @see #writeUnsignedVarint(int, DataOutput)
395        */
396       public static int sizeOfUnsignedVarint(int value) {
397           // Protocol buffers varint encoding is variable length, with a minimum of 1 byte
398           // (for zero). The values themselves are not important. What's important here is
399           // any leading zero bits are dropped from output. We can use this leading zero
400           // count w/ fast intrinsic to calc the output length directly.
401   
402           // Test cases verify this matches the output for loop logic exactly.
403   
404           // return (38 - leadingZeros) / 7 + leadingZeros / 32;
405   
406           // The above formula provides the implementation, but the Java encoding is suboptimal
407           // when we have a narrow range of integers, so we can do better manually
408   
409           int leadingZeros = Integer.numberOfLeadingZeros(value);
410           int leadingZerosBelow38DividedBy7 = ((38 - leadingZeros) * 0b10010010010010011) >>> 19;
411           return leadingZerosBelow38DividedBy7 + (leadingZeros >>> 5);
412       }

419       public static int sizeOfVarint(int value) {

427        * @see #sizeOfUnsignedVarint(int)
428        */
429       public static int sizeOfVarlong(long value) {
430           long v = (value << 1) ^ (value >> 63);
431   
432           // For implementation notes @see #sizeOfUnsignedVarint(int)
433           // Similar logic is applied to allow for 64bit input -> 1-9byte output.
434           // return (70 - leadingZeros) / 7 + leadingZeros / 64;
435   
436           int leadingZeros = Long.numberOfLeadingZeros(v);
437           int leadingZerosBelow70DividedBy7 = ((70 - leadingZeros) * 0b10010010010010011) >>> 19;
438           return leadingZerosBelow70DividedBy7 + (leadingZeros >>> 6);
439       }
