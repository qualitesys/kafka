17   package org.apache.kafka.clients.producer.internals;
18   
19   
20   import org.apache.kafka.clients.producer.ProducerInterceptor;
21   import org.apache.kafka.clients.producer.ProducerRecord;
22   import org.apache.kafka.clients.producer.RecordMetadata;
23   import org.apache.kafka.common.TopicPartition;
24   import org.apache.kafka.common.record.RecordBatch;
25   import org.slf4j.Logger;
26   import org.slf4j.LoggerFactory;
27   
28   import java.io.Closeable;
29   import java.util.List;
30   
31   /**
32    * A container that holds the list {@link org.apache.kafka.clients.producer.ProducerInterceptor}
33    * and wraps calls to the chain of custom interceptors.
34    */
35   public class ProducerInterceptors<K, V> implements Closeable {
36       private static final Logger log = LoggerFactory.getLogger(ProducerInterceptors.class);
37       private final List<ProducerInterceptor<K, V>> interceptors;
38   
39       public ProducerInterceptors(List<ProducerInterceptor<K, V>> interceptors) {
40           this.interceptors = interceptors;
41       }
42   
43       /**
44        * This is called when client sends the record to KafkaProducer, before key and value gets serialized.
45        * The method calls {@link ProducerInterceptor#onSend(ProducerRecord)} method. ProducerRecord
46        * returned from the first interceptor's onSend() is passed to the second interceptor onSend(), and so on in the
47        * interceptor chain. The record returned from the last interceptor is returned from this method.
48        *
49        * This method does not throw exceptions. Exceptions thrown by any of interceptor methods are caught and ignored.
50        * If an interceptor in the middle of the chain, that normally modifies the record, throws an exception,
51        * the next interceptor in the chain will be called with a record returned by the previous interceptor that did not
52        * throw an exception.
53        *
54        * @param record the record from client
55        * @return producer record to send to topic/partition
56        */
57       public ProducerRecord<K, V> onSend(ProducerRecord<K, V> record) {
58           ProducerRecord<K, V> interceptRecord = record;
59           for (ProducerInterceptor<K, V> interceptor : this.interceptors) {
60               try {
61                   interceptRecord = interceptor.onSend(interceptRecord);
62               } catch (Exception e) {
63                   // do not propagate interceptor exception, log and continue calling other interceptors
64                   // be careful not to throw exception from here
65                   if (record != null)
66                       log.warn("Error executing interceptor onSend callback for topic: {}, partition: {}", record.topic(), record.partition(), e);
67                   else
68                       log.warn("Error executing interceptor onSend callback", e);
69               }
70           }
71           return interceptRecord;
72       }
73   
74       /**
75        * This method is called when the record sent to the server has been acknowledged, or when sending the record fails before
76        * it gets sent to the server. This method calls {@link ProducerInterceptor#onAcknowledgement(RecordMetadata, Exception)}
77        * method for each interceptor.
78        *
79        * This method does not throw exceptions. Exceptions thrown by any of interceptor methods are caught and ignored.
80        *
81        * @param metadata The metadata for the record that was sent (i.e. the partition and offset).
82        *                 If an error occurred, metadata will only contain valid topic and maybe partition.
83        * @param exception The exception thrown during processing of this record. Null if no error occurred.
84        */
85       public void onAcknowledgement(RecordMetadata metadata, Exception exception) {
86           for (ProducerInterceptor<K, V> interceptor : this.interceptors) {
87               try {
88                   interceptor.onAcknowledgement(metadata, exception);
89               } catch (Exception e) {
90                   // do not propagate interceptor exceptions, just log
91                   log.warn("Error executing interceptor onAcknowledgement callback", e);
92               }
93           }
94       }
95   
96       /**
97        * This method is called when sending the record fails in {@link ProducerInterceptor#onSend
98        * (ProducerRecord)} method. This method calls {@link ProducerInterceptor#onAcknowledgement(RecordMetadata, Exception)}
99        * method for each interceptor
100        *
101        * @param record The record from client
102        * @param interceptTopicPartition  The topic/partition for the record if an error occurred
103        *        after partition gets assigned; the topic part of interceptTopicPartition is the same as in record.
104        * @param exception The exception thrown during processing of this record.
105        */
106       public void onSendError(ProducerRecord<K, V> record, TopicPartition interceptTopicPartition, Exception exception) {
107           for (ProducerInterceptor<K, V> interceptor : this.interceptors) {
108               try {
109                   if (record == null && interceptTopicPartition == null) {
110                       interceptor.onAcknowledgement(null, exception);
111                   } else {
112                       if (interceptTopicPartition == null) {
113                           interceptTopicPartition = extractTopicPartition(record);
114                       }
115                       interceptor.onAcknowledgement(new RecordMetadata(interceptTopicPartition, -1, -1,
116                                       RecordBatch.NO_TIMESTAMP, -1, -1), exception);
117                   }
118               } catch (Exception e) {
119                   // do not propagate interceptor exceptions, just log
120                   log.warn("Error executing interceptor onAcknowledgement callback", e);
121               }
122           }
123       }
124   
125       public static <K, V> TopicPartition extractTopicPartition(ProducerRecord<K, V> record) {
126           return new TopicPartition(record.topic(), record.partition() == null ? RecordMetadata.UNKNOWN_PARTITION : record.partition());
127       }
128   
129       /**
130        * Closes every interceptor in a container.
131        */
132       @Override
133       public void close() {
134           for (ProducerInterceptor<K, V> interceptor : this.interceptors) {
135               try {
136                   interceptor.close();
137               } catch (Exception e) {
138                   log.error("Failed to close producer interceptor ", e);
139               }
140           }
141       }
142   }
