17   package org.apache.kafka.common.security.oauthbearer;
18   
19   import java.util.Set;
20   
21   import org.apache.kafka.common.annotation.InterfaceStability;
22   
23   /**
24    * The <code>b64token</code> value as defined in
25    * <a href="https://tools.ietf.org/html/rfc6750#section-2.1">RFC 6750 Section
26    * 2.1</a> along with the token's specific scope and lifetime and principal
27    * name.
28    * <p>
29    * A network request would be required to re-hydrate an opaque token, and that
30    * could result in (for example) an {@code IOException}, but retrievers for
31    * various attributes ({@link #scope()}, {@link #lifetimeMs()}, etc.) declare no
32    * exceptions. Therefore, if a network request is required for any of these
33    * retriever methods, that request could be performed at construction time so
34    * that the various attributes can be reliably provided thereafter. For example,
35    * a constructor might declare {@code throws IOException} in such a case.
36    * Alternatively, the retrievers could throw unchecked exceptions.
37    * <p>
38    * This interface was introduced in 2.0.0 and, while it feels stable, it could
39    * evolve. We will try to evolve the API in a compatible manner (easier now that
40    * Java 7 and its lack of default methods doesn't have to be supported), but we
41    * reserve the right to make breaking changes in minor releases, if necessary.
42    * We will update the {@code InterfaceStability} annotation and this notice once
43    * the API is considered stable.
44    * 
45    * @see <a href="https://tools.ietf.org/html/rfc6749#section-1.4">RFC 6749
46    *      Section 1.4</a> and
47    *      <a href="https://tools.ietf.org/html/rfc6750#section-2.1">RFC 6750
48    *      Section 2.1</a>
49    */
50   @InterfaceStability.Evolving
51   public interface OAuthBearerToken {
52       /**
53        * The <code>b64token</code> value as defined in
54        * <a href="https://tools.ietf.org/html/rfc6750#section-2.1">RFC 6750 Section
55        * 2.1</a>
56        * 
57        * @return <code>b64token</code> value as defined in
58        *         <a href="https://tools.ietf.org/html/rfc6750#section-2.1">RFC 6750
59        *         Section 2.1</a>
60        */
61       String value();
62   
63       /**
64        * The token's scope of access, as per
65        * <a href="https://tools.ietf.org/html/rfc6749#section-1.4">RFC 6749 Section
66        * 1.4</a>
67        * 
68        * @return the token's (always non-null but potentially empty) scope of access,
69        *         as per <a href="https://tools.ietf.org/html/rfc6749#section-1.4">RFC
70        *         6749 Section 1.4</a>. Note that all values in the returned set will
71        *         be trimmed of preceding and trailing whitespace, and the result will
72        *         never contain the empty string.
73        */
74       Set<String> scope();
75   
76       /**
77        * The token's lifetime, expressed as the number of milliseconds since the
78        * epoch, as per <a href="https://tools.ietf.org/html/rfc6749#section-1.4">RFC
79        * 6749 Section 1.4</a>
80        * 
81        * @return the token'slifetime, expressed as the number of milliseconds since
82        *         the epoch, as per
83        *         <a href="https://tools.ietf.org/html/rfc6749#section-1.4">RFC 6749
84        *         Section 1.4</a>.
85        */
86       long lifetimeMs();
87   
88       /**
89        * The name of the principal to which this credential applies
90        * 
91        * @return the always non-null/non-empty principal name
92        */
93       String principalName();
94   
95       /**
96        * When the credential became valid, in terms of the number of milliseconds
97        * since the epoch, if known, otherwise null. An expiring credential may not
98        * necessarily indicate when it was created -- just when it expires -- so we
99        * need to support a null return value here.
100        * 
101        * @return the time when the credential became valid, in terms of the number of
102        *         milliseconds since the epoch, if known, otherwise null
103        */
104       Long startTimeMs();
105   }
