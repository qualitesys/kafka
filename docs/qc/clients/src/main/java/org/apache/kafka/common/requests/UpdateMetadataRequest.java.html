53                          Map<String, Uuid> topicIds) {
54               super(ApiKeys.UPDATE_METADATA, version, controllerId, controllerEpoch, brokerEpoch);

56               this.liveBrokers = liveBrokers;

60           @Override
61           public UpdateMetadataRequest build(short version) {
62               if (version < 3) {
63                   for (UpdateMetadataBroker broker : liveBrokers) {
64                       if (version == 0) {
65                           if (broker.endpoints().size() != 1)
66                               throw new UnsupportedVersionException("UpdateMetadataRequest v0 requires a single endpoint");
67                           if (broker.endpoints().get(0).securityProtocol() != SecurityProtocol.PLAINTEXT.id)
68                               throw new UnsupportedVersionException("UpdateMetadataRequest v0 only handles PLAINTEXT endpoints");
69                           // Don't null out `endpoints` since it's ignored by the generated code if version >= 1
70                           UpdateMetadataEndpoint endpoint = broker.endpoints().get(0);
71                           broker.setV0Host(endpoint.host());
72                           broker.setV0Port(endpoint.port());
73                       } else {
74                           if (broker.endpoints().stream().anyMatch(endpoint -> !endpoint.listener().isEmpty() &&
75                                   !endpoint.listener().equals(listenerNameFromSecurityProtocol(endpoint)))) {
76                               throw new UnsupportedVersionException("UpdateMetadataRequest v0-v3 does not support custom " +
77                                   "listeners, request version: " + version + ", endpoints: " + broker.endpoints());
78                           }
79                       }
80                   }
81               }
82   
83               UpdateMetadataRequestData data = new UpdateMetadataRequestData()
84                       .setControllerId(controllerId)
85                       .setControllerEpoch(controllerEpoch)
86                       .setBrokerEpoch(brokerEpoch)
87                       .setLiveBrokers(liveBrokers);
88   
89               if (version >= 5) {
90                   Map<String, UpdateMetadataTopicState> topicStatesMap = groupByTopic(topicIds, partitionStates);
91                   data.setTopicStates(new ArrayList<>(topicStatesMap.values()));
92               } else {
93                   data.setUngroupedPartitionStates(partitionStates);
94               }
95   
96               return new UpdateMetadataRequest(data, version);
97           }
98   
99           private static Map<String, UpdateMetadataTopicState> groupByTopic(Map<String, Uuid> topicIds, List<UpdateMetadataPartitionState> partitionStates) {
100               Map<String, UpdateMetadataTopicState> topicStates = new HashMap<>();
101               for (UpdateMetadataPartitionState partition : partitionStates) {
102                   // We don't null out the topic name in UpdateMetadataPartitionState since it's ignored by the generated
103                   // code if version >= 5
104                   UpdateMetadataTopicState topicState = topicStates.computeIfAbsent(partition.topicName(),
105                       t -> new UpdateMetadataTopicState()
106                               .setTopicName(partition.topicName())
107                               .setTopicId(topicIds.getOrDefault(partition.topicName(), Uuid.ZERO_UUID))
108   
109                   );
110                   topicState.partitionStates().add(partition);
111               }
112               return topicStates;
113           }

117               StringBuilder bld = new StringBuilder();

120                   append(", controllerEpoch=").append(controllerEpoch).

124                   append(")");
125               return bld.toString();

131       UpdateMetadataRequest(UpdateMetadataRequestData data, short version) {
132           super(ApiKeys.UPDATE_METADATA, version);
133           this.data = data;
134           // Do this from the constructor to make it thread-safe (even though it's only needed when some methods are called)
135           normalize();
136       }

138       private void normalize() {
139           // Version 0 only supported a single host and port and the protocol was always plaintext
140           // Version 1 added support for multiple endpoints, each with its own security protocol
141           // Version 2 added support for rack
142           // Version 3 added support for listener name, which we can infer from the security protocol for older versions
143           if (version() < 3) {
144               for (UpdateMetadataBroker liveBroker : data.liveBrokers()) {
145                   // Set endpoints so that callers can rely on it always being present
146                   if (version() == 0 && liveBroker.endpoints().isEmpty()) {
147                       SecurityProtocol securityProtocol = SecurityProtocol.PLAINTEXT;
148                       liveBroker.setEndpoints(singletonList(new UpdateMetadataEndpoint()
149                           .setHost(liveBroker.v0Host())
150                           .setPort(liveBroker.v0Port())
151                           .setSecurityProtocol(securityProtocol.id)
152                           .setListener(ListenerName.forSecurityProtocol(securityProtocol).value())));
153                   } else {
154                       for (UpdateMetadataEndpoint endpoint : liveBroker.endpoints()) {
155                           // Set listener so that callers can rely on it always being present
156                           if (endpoint.listener().isEmpty())
157                               endpoint.setListener(listenerNameFromSecurityProtocol(endpoint));
158                       }
159                   }
160               }
161           }
162   
163           if (version() >= 5) {
164               for (UpdateMetadataTopicState topicState : data.topicStates()) {
165                   for (UpdateMetadataPartitionState partitionState : topicState.partitionStates()) {
166                       // Set the topic name so that we can always present the ungrouped view to callers
167                       partitionState.setTopicName(topicState.topicName());
168                   }
169               }
170           }
171       }

173       private static String listenerNameFromSecurityProtocol(UpdateMetadataEndpoint endpoint) {
174           SecurityProtocol securityProtocol = SecurityProtocol.forId(endpoint.securityProtocol());
175           return ListenerName.forSecurityProtocol(securityProtocol).value();
176       }
