46   /**
47    * A registry of sensors and metrics.
48    * <p>
49    * A metric is a named, numerical measurement. A sensor is a handle to record numerical measurements as they occur. Each
50    * Sensor has zero or more associated metrics. For example a Sensor might represent message sizes and we might associate
51    * with this sensor a metric for the average, maximum, or other statistics computed off the sequence of message sizes
52    * that are recorded by the sensor.
53    * <p>
54    * Usage looks something like this:
55    * 
56    * <pre>
57    * // set up metrics:
58    * Metrics metrics = new Metrics(); // this is the global repository of metrics and sensors
59    * Sensor sensor = metrics.sensor(&quot;message-sizes&quot;);
60    * MetricName metricName = new MetricName(&quot;message-size-avg&quot;, &quot;producer-metrics&quot;);
61    * sensor.add(metricName, new Avg());
62    * metricName = new MetricName(&quot;message-size-max&quot;, &quot;producer-metrics&quot;);
63    * sensor.add(metricName, new Max());
64    * 
65    * // as messages are sent we record the sizes
66    * sensor.record(messageSize);
67    * </pre>
68    */

246       /**
247        * Use the specified domain and metric name templates to generate an HTML table documenting the metrics. A separate table section
248        * will be generated for each of the MBeans and the associated attributes. The MBean names are lexicographically sorted to
249        * determine the order of these sections. This order is therefore dependent upon the order of the
250        * tags in each {@link MetricNameTemplate}.
251        *
252        * @param domain the domain or prefix for the JMX MBean names; may not be null
253        * @param allMetrics the collection of all {@link MetricNameTemplate} instances each describing one metric; may not be null
254        * @return the string containing the HTML table; never null
255        */
256       public static String toHtmlTable(String domain, Iterable<MetricNameTemplate> allMetrics) {
257           Map<String, Map<String, String>> beansAndAttributes = new TreeMap<>();
258       
259           try (Metrics metrics = new Metrics()) {
260               for (MetricNameTemplate template : allMetrics) {
261                   Map<String, String> tags = new LinkedHashMap<>();
262                   for (String s : template.tags()) {
263                       tags.put(s, "{" + s + "}");
264                   }
265       
266                   MetricName metricName = metrics.metricName(template.name(), template.group(), template.description(), tags);
267                   String mBeanName = JmxReporter.getMBeanName(domain, metricName);
268                   if (!beansAndAttributes.containsKey(mBeanName)) {
269                       beansAndAttributes.put(mBeanName, new TreeMap<>());
270                   }
271                   Map<String, String> attrAndDesc = beansAndAttributes.get(mBeanName);
272                   if (!attrAndDesc.containsKey(template.name())) {
273                       attrAndDesc.put(template.name(), template.description());
274                   } else {
275                       throw new IllegalArgumentException("mBean '" + mBeanName + "' attribute '" + template.name() + "' is defined twice.");
276                   }
277               }
278           }
279           
280           StringBuilder b = new StringBuilder();
281           b.append("<table class=\"data-table\"><tbody>\n");
282       
283           for (Entry<String, Map<String, String>> e : beansAndAttributes.entrySet()) {
284               b.append("<tr>\n");
285               b.append("<td colspan=3 class=\"mbeanName\" style=\"background-color:#ccc; font-weight: bold;\">");
286               b.append(e.getKey());
287               b.append("</td>");
288               b.append("</tr>\n");
289               
290               b.append("<tr>\n");
291               b.append("<th style=\"width: 90px\"></th>\n");
292               b.append("<th>Attribute name</th>\n");
293               b.append("<th>Description</th>\n");
294               b.append("</tr>\n");
295               
296               for (Entry<String, String> e2 : e.getValue().entrySet()) {
297                   b.append("<tr>\n");
298                   b.append("<td></td>");
299                   b.append("<td>");
300                   b.append(e2.getKey());
301                   b.append("</td>");
302                   b.append("<td>");
303                   b.append(e2.getValue());
304                   b.append("</td>");
305                   b.append("</tr>\n");
306               }
307       
308           }
309           b.append("</tbody></table>");
310       
311           return b.toString();
312       
313       }

404        * @param inactiveSensorExpirationTimeSeconds If no value if recorded on the Sensor for this duration of time,
405        *                                        it is eligible for removal
406        * @param parents The parent sensors
407        * @param recordingLevel The recording level.
408        * @return The sensor that is created
409        */
410       public synchronized Sensor sensor(String name, MetricConfig config, long inactiveSensorExpirationTimeSeconds, Sensor.RecordingLevel recordingLevel, Sensor... parents) {
411           Sensor s = getSensor(name);

459                       }
460                   }
461               }
462               if (childSensors != null) {
463                   for (Sensor childSensor : childSensors)

466           }

628       /**
629        * This iterates over every Sensor and triggers a removeSensor if it has expired
630        * Package private for testing
631        */
632       class ExpireSensorTask implements Runnable {
633           @Override
634           public void run() {
635               for (Map.Entry<String, Sensor> sensorEntry : sensors.entrySet()) {
636                   // removeSensor also locks the sensor object. This is fine because synchronized is reentrant
637                   // There is however a minor race condition here. Assume we have a parent sensor P and child sensor C.
638                   // Calling record on C would cause a record on P as well.
639                   // So expiration time for P == expiration time for C. If the record on P happens via C just after P is removed,
640                   // that will cause C to also get removed.
641                   // Since the expiration time is typically high it is not expected to be a significant concern
642                   // and thus not necessary to optimize
643                   synchronized (sensorEntry.getValue()) {
644                       if (sensorEntry.getValue().hasExpired()) {
645                           log.debug("Removing expired sensor {}", sensorEntry.getKey());
646                           removeSensor(sensorEntry.getKey());
647                       }
648                   }
649               }
650           }
651       }
