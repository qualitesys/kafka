37   /**
38    * This class implements the inner record format for magic 2 and above. The schema is as follows:
39    *
40    *
41    * Record =>
42    *   Length => Varint
43    *   Attributes => Int8
44    *   TimestampDelta => Varlong
45    *   OffsetDelta => Varint
46    *   Key => Bytes
47    *   Value => Bytes
48    *   Headers => [HeaderKey HeaderValue]
49    *     HeaderKey => String
50    *     HeaderValue => Bytes
51    *
52    * Note that in this schema, the Bytes and String types use a variable length integer to represent
53    * the length of the field. The array type used for the headers also uses a Varint for the number of
54    * headers.
55    *
56    * The current record attributes are depicted below:
57    *
58    *  ----------------
59    *  | Unused (0-7) |
60    *  ----------------
61    *
62    * The offset and timestamp deltas compute the difference relative to the base offset and
63    * base timestamp of the batch that this record is contained in.
64    */
65   public class DefaultRecord implements Record {

161       /**
162        * Write the record to `out` and return its size.
163        */
164       public static int writeTo(DataOutputStream out,
165                                 int offsetDelta,
166                                 long timestampDelta,
167                                 ByteBuffer key,
168                                 ByteBuffer value,
169                                 Header[] headers) throws IOException {
170           int sizeInBytes = sizeOfBodyInBytes(offsetDelta, timestampDelta, key, value, headers);
171           ByteUtils.writeVarint(sizeInBytes, out);
172   
173           byte attributes = 0; // there are no used record attributes at the moment
174           out.write(attributes);
175   
176           ByteUtils.writeVarlong(timestampDelta, out);
177           ByteUtils.writeVarint(offsetDelta, out);
178   
179           if (key == null) {
180               ByteUtils.writeVarint(-1, out);
181           } else {
182               int keySize = key.remaining();
183               ByteUtils.writeVarint(keySize, out);
184               Utils.writeTo(out, key, keySize);
185           }
186   
187           if (value == null) {
188               ByteUtils.writeVarint(-1, out);
189           } else {
190               int valueSize = value.remaining();
191               ByteUtils.writeVarint(valueSize, out);
192               Utils.writeTo(out, value, valueSize);
193           }
194   
195           if (headers == null)
196               throw new IllegalArgumentException("Headers cannot be null");
197   
198           ByteUtils.writeVarint(headers.length, out);
199   
200           for (Header header : headers) {
201               String headerKey = header.key();
202               if (headerKey == null)
203                   throw new IllegalArgumentException("Invalid null header key found in headers");
204   
205               byte[] utf8Bytes = Utils.utf8(headerKey);
206               ByteUtils.writeVarint(utf8Bytes.length, out);
207               out.write(utf8Bytes);
208   
209               byte[] headerValue = header.value();
210               if (headerValue == null) {
211                   ByteUtils.writeVarint(-1, out);
212               } else {
213                   ByteUtils.writeVarint(headerValue.length, out);
214                   out.write(headerValue);
215               }
216           }
217   
218           return ByteUtils.sizeOfVarint(sizeInBytes) + sizeInBytes;
219       }

305       private static DefaultRecord readFrom(ByteBuffer buffer,
306                                             int sizeInBytes,
307                                             int sizeOfBodyInBytes,
308                                             long baseOffset,
309                                             long baseTimestamp,
310                                             int baseSequence,
311                                             Long logAppendTime) {
312           try {
313               int recordStart = buffer.position();
314               byte attributes = buffer.get();
315               long timestampDelta = ByteUtils.readVarlong(buffer);
316               long timestamp = baseTimestamp + timestampDelta;
317               if (logAppendTime != null)
318                   timestamp = logAppendTime;
319   
320               int offsetDelta = ByteUtils.readVarint(buffer);
321               long offset = baseOffset + offsetDelta;
322               int sequence = baseSequence >= 0 ?
323                       DefaultRecordBatch.incrementSequence(baseSequence, offsetDelta) :
324                       RecordBatch.NO_SEQUENCE;
325   
326               ByteBuffer key = null;
327               int keySize = ByteUtils.readVarint(buffer);
328               if (keySize >= 0) {
329                   key = buffer.slice();
330                   key.limit(keySize);
331                   buffer.position(buffer.position() + keySize);
332               }
333   
334               ByteBuffer value = null;
335               int valueSize = ByteUtils.readVarint(buffer);
336               if (valueSize >= 0) {
337                   value = buffer.slice();
338                   value.limit(valueSize);
339                   buffer.position(buffer.position() + valueSize);
340               }
341   
342               int numHeaders = ByteUtils.readVarint(buffer);
343               if (numHeaders < 0)
344                   throw new InvalidRecordException("Found invalid number of record headers " + numHeaders);
345   
346               final Header[] headers;
347               if (numHeaders == 0)
348                   headers = Record.EMPTY_HEADERS;
349               else
350                   headers = readHeaders(buffer, numHeaders);
351   
352               // validate whether we have read all header bytes in the current record
353               if (buffer.position() - recordStart != sizeOfBodyInBytes)
354                   throw new InvalidRecordException("Invalid record size: expected to read " + sizeOfBodyInBytes +
355                           " bytes in record payload, but instead read " + (buffer.position() - recordStart));
356   
357               return new DefaultRecord(sizeInBytes, attributes, offset, timestamp, sequence, key, value, headers);
358           } catch (BufferUnderflowException | IllegalArgumentException e) {
359               throw new InvalidRecordException("Found invalid record structure", e);
360           }
361       }

376       private static PartialDefaultRecord readPartiallyFrom(DataInput input,
377                                                             byte[] skipArray,
378                                                             int sizeInBytes,
379                                                             int sizeOfBodyInBytes,
380                                                             long baseOffset,
381                                                             long baseTimestamp,
382                                                             int baseSequence,
383                                                             Long logAppendTime) throws IOException {
384           ByteBuffer skipBuffer = ByteBuffer.wrap(skipArray);
385           // set its limit to 0 to indicate no bytes readable yet
386           skipBuffer.limit(0);
387   
388           try {
389               // reading the attributes / timestamp / offset and key-size does not require
390               // any byte array allocation and therefore we can just read them straight-forwardly
391               IntRef bytesRemaining = PrimitiveRef.ofInt(sizeOfBodyInBytes);
392   
393               byte attributes = readByte(skipBuffer, input, bytesRemaining);
394               long timestampDelta = readVarLong(skipBuffer, input, bytesRemaining);
395               long timestamp = baseTimestamp + timestampDelta;
396               if (logAppendTime != null)
397                   timestamp = logAppendTime;
398   
399               int offsetDelta = readVarInt(skipBuffer, input, bytesRemaining);
400               long offset = baseOffset + offsetDelta;
401               int sequence = baseSequence >= 0 ?
402                   DefaultRecordBatch.incrementSequence(baseSequence, offsetDelta) :
403                   RecordBatch.NO_SEQUENCE;
404   
405               // first skip key
406               int keySize = skipLengthDelimitedField(skipBuffer, input, bytesRemaining);
407   
408               // then skip value
409               int valueSize = skipLengthDelimitedField(skipBuffer, input, bytesRemaining);
410   
411               // then skip header
412               int numHeaders = readVarInt(skipBuffer, input, bytesRemaining);
413               if (numHeaders < 0)
414                   throw new InvalidRecordException("Found invalid number of record headers " + numHeaders);
415               for (int i = 0; i < numHeaders; i++) {
416                   int headerKeySize = skipLengthDelimitedField(skipBuffer, input, bytesRemaining);
417                   if (headerKeySize < 0)
418                       throw new InvalidRecordException("Invalid negative header key size " + headerKeySize);
419   
420                   // headerValueSize
421                   skipLengthDelimitedField(skipBuffer, input, bytesRemaining);
422               }
423   
424               if (bytesRemaining.value > 0 || skipBuffer.remaining() > 0)
425                   throw new InvalidRecordException("Invalid record size: expected to read " + sizeOfBodyInBytes +
426                       " bytes in record payload, but there are still bytes remaining");
427   
428               return new PartialDefaultRecord(sizeInBytes, attributes, offset, timestamp, sequence, keySize, valueSize);
429           } catch (BufferUnderflowException | IllegalArgumentException e) {
430               throw new InvalidRecordException("Found invalid record structure", e);
431           }
432       }

450       private static int readVarInt(ByteBuffer buffer, DataInput input, IntRef bytesRemaining) throws IOException {
451           if (buffer.remaining() < 5 && bytesRemaining.value > 0) {
452               readMore(buffer, input, bytesRemaining);

458       private static int skipLengthDelimitedField(ByteBuffer buffer, DataInput input, IntRef bytesRemaining) throws IOException {
459           boolean needMore = false;
460           int sizeInBytes = -1;
461           int bytesToSkip = -1;
462   
463           while (true) {
464               if (needMore) {
465                   readMore(buffer, input, bytesRemaining);
466                   needMore = false;
467               }
468   
469               if (bytesToSkip < 0) {
470                   if (buffer.remaining() < 5 && bytesRemaining.value > 0) {
471                       needMore = true;
472                   } else {
473                       sizeInBytes = ByteUtils.readVarint(buffer);
474                       if (sizeInBytes <= 0)
475                           return sizeInBytes;
476                       else
477                           bytesToSkip = sizeInBytes;
478   
479                   }
480               } else {
481                   if (bytesToSkip > buffer.remaining()) {
482                       bytesToSkip -= buffer.remaining();
483                       buffer.position(buffer.limit());
484                       needMore = true;
485                   } else {
486                       buffer.position(buffer.position() + bytesToSkip);
487                       return sizeInBytes;
488                   }
489               }
490           }
491       }

493       private static void readMore(ByteBuffer buffer, DataInput input, IntRef bytesRemaining) throws IOException {
494           if (bytesRemaining.value > 0) {
495               byte[] array = buffer.array();
496   
497               // first copy the remaining bytes to the beginning of the array;
498               // at most 4 bytes would be shifted here
499               int stepsToLeftShift = buffer.position();
500               int bytesToLeftShift = buffer.remaining();
501               for (int i = 0; i < bytesToLeftShift; i++) {
502                   array[i] = array[i + stepsToLeftShift];
503               }
504   
505               // then try to read more bytes to the remaining of the array
506               int bytesRead = Math.min(bytesRemaining.value, array.length - bytesToLeftShift);
507               input.readFully(array, bytesToLeftShift, bytesRead);
508               buffer.rewind();
509               // only those many bytes are readable
510               buffer.limit(bytesToLeftShift + bytesRead);
511   
512               bytesRemaining.value -= bytesRead;
513           } else {
514               throw new InvalidRecordException("Invalid record size: expected to read more bytes in record payload");
515           }
516       }

518       private static Header[] readHeaders(ByteBuffer buffer, int numHeaders) {
519           Header[] headers = new Header[numHeaders];
520           for (int i = 0; i < numHeaders; i++) {
521               int headerKeySize = ByteUtils.readVarint(buffer);
522               if (headerKeySize < 0)
523                   throw new InvalidRecordException("Invalid negative header key size " + headerKeySize);
524   
525               ByteBuffer headerKeyBuffer = buffer.slice();
526               headerKeyBuffer.limit(headerKeySize);
527               buffer.position(buffer.position() + headerKeySize);
528   
529               ByteBuffer headerValue = null;
530               int headerValueSize = ByteUtils.readVarint(buffer);
531               if (headerValueSize >= 0) {
532                   headerValue = buffer.slice();
533                   headerValue.limit(headerValueSize);
534                   buffer.position(buffer.position() + headerValueSize);
535               }
536   
537               headers[i] = new RecordHeader(headerKeyBuffer, headerValue);
538           }
539   
540           return headers;

583       private static int sizeOf(int keySize, int valueSize, Header[] headers) {
584           int size = 0;
585           if (keySize < 0)
586               size += NULL_VARINT_SIZE_BYTES;
587           else
588               size += ByteUtils.sizeOfVarint(keySize) + keySize;
589   
590           if (valueSize < 0)
591               size += NULL_VARINT_SIZE_BYTES;
592           else
593               size += ByteUtils.sizeOfVarint(valueSize) + valueSize;
594   
595           if (headers == null)
596               throw new IllegalArgumentException("Headers cannot be null");
597   
598           size += ByteUtils.sizeOfVarint(headers.length);
599           for (Header header : headers) {
600               String headerKey = header.key();
601               if (headerKey == null)
602                   throw new IllegalArgumentException("Invalid null header key found in headers");
603   
604               int headerKeySize = Utils.utf8Length(headerKey);
605               size += ByteUtils.sizeOfVarint(headerKeySize) + headerKeySize;
606   
607               byte[] headerValue = header.value();
608               if (headerValue == null) {
609                   size += NULL_VARINT_SIZE_BYTES;
610               } else {
611                   size += ByteUtils.sizeOfVarint(headerValue.length) + headerValue.length;
612               }
613           }
614           return size;
615       }
