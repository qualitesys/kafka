18   package org.apache.kafka.common.security.oauthbearer.secured;
19   
20   import java.io.IOException;
21   import java.nio.file.Path;
22   import java.security.Key;
23   import java.util.List;
24   import org.apache.kafka.common.utils.Utils;
25   import org.jose4j.jwk.JsonWebKeySet;
26   import org.jose4j.jws.JsonWebSignature;
27   import org.jose4j.jwx.JsonWebStructure;
28   import org.jose4j.keys.resolvers.JwksVerificationKeyResolver;
29   import org.jose4j.keys.resolvers.VerificationKeyResolver;
30   import org.jose4j.lang.JoseException;
31   import org.jose4j.lang.UnresolvableKeyException;
32   import org.slf4j.Logger;
33   import org.slf4j.LoggerFactory;
34   
35   /**
36    * <code>JwksFileVerificationKeyResolver</code> is a {@link VerificationKeyResolver} implementation
37    * that will load the JWKS from the given file system directory.
38    *
39    * A <a href="https://datatracker.ietf.org/doc/html/rfc7517#section-5">JWKS (JSON Web Key Set)</a>
40    * is a JSON document provided by the OAuth/OIDC provider that lists the keys used to sign the JWTs
41    * it issues.
42    *
43    * Here is a sample JWKS JSON document:
44    *
45    * <pre>
46    * {
47    *   "keys": [
48    *     {
49    *       "kty": "RSA",
50    *       "alg": "RS256",
51    *       "kid": "abc123",
52    *       "use": "sig",
53    *       "e": "AQAB",
54    *       "n": "..."
55    *     },
56    *     {
57    *       "kty": "RSA",
58    *       "alg": "RS256",
59    *       "kid": "def456",
60    *       "use": "sig",
61    *       "e": "AQAB",
62    *       "n": "..."
63    *     }
64    *   ]
65    * }
66    * </pre>
67    *
68    * Without going into too much detail, the array of keys enumerates the key data that the provider
69    * is using to sign the JWT. The key ID (<code>kid</code>) is referenced by the JWT's header in
70    * order to match up the JWT's signing key with the key in the JWKS. During the validation step of
71    * the broker, the jose4j OAuth library will use the contents of the appropriate key in the JWKS
72    * to validate the signature.
73    *
74    * Given that the JWKS is referenced by the JWT, the JWKS must be made available by the
75    * OAuth/OIDC provider so that a JWT can be validated.
76    *
77    * @see org.apache.kafka.common.config.SaslConfigs#SASL_OAUTHBEARER_TOKEN_ENDPOINT_URL
78    * @see VerificationKeyResolver
79    */
80   
81   public class JwksFileVerificationKeyResolver implements CloseableVerificationKeyResolver {
82   
83       private static final Logger log = LoggerFactory.getLogger(JwksFileVerificationKeyResolver.class);
84   
85       private final Path jwksFile;
86   
87       private VerificationKeyResolver delegate;
88   
89       public JwksFileVerificationKeyResolver(Path jwksFile) {
90           this.jwksFile = jwksFile;
91       }
92   
93       @Override
94       public void init() throws IOException {
95           log.debug("Starting creation of new VerificationKeyResolver from {}", jwksFile);
96           String json = Utils.readFileAsString(jwksFile.toFile().getPath());
97   
98           JsonWebKeySet jwks;
99   
100           try {
101               jwks = new JsonWebKeySet(json);
102           } catch (JoseException e) {
103               throw new IOException(e);
104           }
105   
106           delegate = new JwksVerificationKeyResolver(jwks.getJsonWebKeys());
107       }
108   
109       @Override
110       public Key resolveKey(JsonWebSignature jws, List<JsonWebStructure> nestingContext) throws UnresolvableKeyException {
111           if (delegate == null)
112               throw new UnresolvableKeyException("VerificationKeyResolver delegate is null; please call init() first");
113   
114           return delegate.resolveKey(jws, nestingContext);
115       }
116   
117   }
