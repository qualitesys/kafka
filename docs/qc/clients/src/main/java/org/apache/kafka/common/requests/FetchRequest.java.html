203           private void addToForgottenTopicMap(List<TopicIdPartition> toForget, Map<String, FetchRequestData.ForgottenTopic> forgottenTopicMap) {
204               toForget.forEach(topicIdPartition -> {
205                   FetchRequestData.ForgottenTopic forgottenTopic = forgottenTopicMap.get(topicIdPartition.topic());
206                   if (forgottenTopic == null) {
207                       forgottenTopic = new ForgottenTopic()
208                               .setTopic(topicIdPartition.topic())
209                               .setTopicId(topicIdPartition.topicId());
210                       forgottenTopicMap.put(topicIdPartition.topic(), forgottenTopic);
211                   }
212                   forgottenTopic.partitions().add(topicIdPartition.partition());
213               });
214           }

216           @Override
217           public FetchRequest build(short version) {
218               if (version < 3) {
219                   maxBytes = DEFAULT_RESPONSE_MAX_BYTES;
220               }
221   
222               FetchRequestData fetchRequestData = new FetchRequestData();
223               fetchRequestData.setReplicaId(replicaId);
224               fetchRequestData.setMaxWaitMs(maxWait);
225               fetchRequestData.setMinBytes(minBytes);
226               fetchRequestData.setMaxBytes(maxBytes);
227               fetchRequestData.setIsolationLevel(isolationLevel.id());
228               fetchRequestData.setForgottenTopicsData(new ArrayList<>());
229   
230               Map<String, FetchRequestData.ForgottenTopic> forgottenTopicMap = new LinkedHashMap<>();
231               addToForgottenTopicMap(removed, forgottenTopicMap);
232   
233               // If a version older than v13 is used, topic-partition which were replaced
234               // by a topic-partition with the same name but a different topic ID are not
235               // sent out in the "forget" set in order to not remove the newly added
236               // partition in the "fetch" set.
237               if (version >= 13) {
238                   addToForgottenTopicMap(replaced, forgottenTopicMap);
239               }
240   
241               forgottenTopicMap.forEach((topic, forgottenTopic) -> fetchRequestData.forgottenTopicsData().add(forgottenTopic));
242   
243               // We collect the partitions in a single FetchTopic only if they appear sequentially in the fetchData
244               fetchRequestData.setTopics(new ArrayList<>());
245               FetchRequestData.FetchTopic fetchTopic = null;
246               for (Map.Entry<TopicPartition, PartitionData> entry : toFetch.entrySet()) {
247                   TopicPartition topicPartition = entry.getKey();
248                   PartitionData partitionData = entry.getValue();
249   
250                   if (fetchTopic == null || !topicPartition.topic().equals(fetchTopic.topic())) {
251                       fetchTopic = new FetchRequestData.FetchTopic()
252                          .setTopic(topicPartition.topic())
253                          .setTopicId(partitionData.topicId)
254                          .setPartitions(new ArrayList<>());
255                       fetchRequestData.topics().add(fetchTopic);
256                   }
257   
258                   FetchRequestData.FetchPartition fetchPartition = new FetchRequestData.FetchPartition()
259                       .setPartition(topicPartition.partition())
260                       .setCurrentLeaderEpoch(partitionData.currentLeaderEpoch.orElse(RecordBatch.NO_PARTITION_LEADER_EPOCH))
261                       .setLastFetchedEpoch(partitionData.lastFetchedEpoch.orElse(RecordBatch.NO_PARTITION_LEADER_EPOCH))
262                       .setFetchOffset(partitionData.fetchOffset)
263                       .setLogStartOffset(partitionData.logStartOffset)
264                       .setPartitionMaxBytes(partitionData.maxBytes);
265   
266                   fetchTopic.partitions().add(fetchPartition);
267               }
268   
269               if (metadata != null) {
270                   fetchRequestData.setSessionEpoch(metadata.epoch());
271                   fetchRequestData.setSessionId(metadata.sessionId());
272               }
273               fetchRequestData.setRackId(rackId);
274   
275               return new FetchRequest(fetchRequestData, version);
276           }

303       @Override
304       public AbstractResponse getErrorResponse(int throttleTimeMs, Throwable e) {
305           // For versions 13+ the error is indicated by setting the top-level error code, and no partitions will be returned.
306           // For earlier versions, the error is indicated in two ways: by setting the same error code in all partitions,
307           // and by setting the top-level error code.  The form where we set the same error code in all partitions
308           // is needed in order to maintain backwards compatibility with older versions of the protocol
309           // in which there was no top-level error code. Note that for incremental fetch responses, there
310           // may not be any partitions at all in the response.  For this reason, the top-level error code
311           // is essential for them.
312           Errors error = Errors.forException(e);
313           List<FetchResponseData.FetchableTopicResponse> topicResponseList = new ArrayList<>();
314           // For version 13+, we know the client can handle a top level error code, so we don't need to send back partitions too.
315           if (version() < 13) {
316               data.topics().forEach(topic -> {
317                   List<FetchResponseData.PartitionData> partitionResponses = topic.partitions().stream().map(partition ->
318                           FetchResponse.partitionResponse(partition.partition(), error)).collect(Collectors.toList());
319                   topicResponseList.add(new FetchResponseData.FetchableTopicResponse()
320                           .setTopic(topic.topic())
321                           .setTopicId(topic.topicId())
322                           .setPartitions(partitionResponses));
323               });
324           }
325           return new FetchResponse(new FetchResponseData()
326                   .setThrottleTimeMs(throttleTimeMs)
327                   .setErrorCode(error.code())
328                   .setSessionId(data.sessionId())
329                   .setResponses(topicResponseList));
330       }

348       // For versions < 13, builds the partitionData map using only the FetchRequestData.
349       // For versions 13+, builds the partitionData map using both the FetchRequestData and a mapping of topic IDs to names.
350       public Map<TopicIdPartition, PartitionData> fetchData(Map<Uuid, String> topicNames) {
351           if (fetchData == null) {
352               synchronized (this) {
353                   if (fetchData == null) {
354                       // Assigning the lazy-initialized `fetchData` in the last step
355                       // to avoid other threads accessing a half-initialized object.
356                       final LinkedHashMap<TopicIdPartition, PartitionData> fetchDataTmp = new LinkedHashMap<>();
357                       final short version = version();
358                       data.topics().forEach(fetchTopic -> {
359                           String name;
360                           if (version < 13) {
361                               name = fetchTopic.topic(); // can't be null
362                           } else {
363                               name = topicNames.get(fetchTopic.topicId());
364                           }
365                           fetchTopic.partitions().forEach(fetchPartition ->
366                                   // Topic name may be null here if the topic name was unable to be resolved using the topicNames map.
367                                   fetchDataTmp.put(new TopicIdPartition(fetchTopic.topicId(), new TopicPartition(name, fetchPartition.partition())),
368                                           new PartitionData(
369                                                   fetchTopic.topicId(),
370                                                   fetchPartition.fetchOffset(),
371                                                   fetchPartition.logStartOffset(),
372                                                   fetchPartition.partitionMaxBytes(),
373                                                   optionalEpoch(fetchPartition.currentLeaderEpoch()),
374                                                   optionalEpoch(fetchPartition.lastFetchedEpoch())
375                                           )
376                                   )
377                           );
378                       });
379                       fetchData = fetchDataTmp;
380                   }
381               }
382           }
383           return fetchData;
384       }
385   
386       // For versions < 13, builds the forgotten topics list using only the FetchRequestData.
387       // For versions 13+, builds the forgotten topics list using both the FetchRequestData and a mapping of topic IDs to names.
388       public List<TopicIdPartition> forgottenTopics(Map<Uuid, String> topicNames) {
389           if (toForget == null) {
390               synchronized (this) {
391                   if (toForget == null) {
392                       // Assigning the lazy-initialized `toForget` in the last step
393                       // to avoid other threads accessing a half-initialized object.
394                       final List<TopicIdPartition> toForgetTmp = new ArrayList<>();
395                       data.forgottenTopicsData().forEach(forgottenTopic -> {
396                           String name;
397                           if (version() < 13) {
398                               name = forgottenTopic.topic(); // can't be null
399                           } else {
400                               name = topicNames.get(forgottenTopic.topicId());
401                           }
402                           // Topic name may be null here if the topic name was unable to be resolved using the topicNames map.
403                           forgottenTopic.partitions().forEach(partitionId -> toForgetTmp.add(new TopicIdPartition(forgottenTopic.topicId(), new TopicPartition(name, partitionId))));
404                       });
405                       toForget = toForgetTmp;
406                   }
407               }
408           }
409           return toForget;
410       }
