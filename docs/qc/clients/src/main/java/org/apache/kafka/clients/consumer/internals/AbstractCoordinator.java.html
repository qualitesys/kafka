83   /**
84    * AbstractCoordinator implements group management for a single group member by interacting with
85    * a designated Kafka broker (the coordinator). Group semantics are provided by extending this class.
86    * See {@link ConsumerCoordinator} for example usage.
87    *
88    * From a high level, Kafka's group management protocol consists of the following sequence of actions:
89    *
90    * <ol>
91    *     <li>Group Registration: Group members register with the coordinator providing their own metadata
92    *         (such as the set of topics they are interested in).</li>
93    *     <li>Group/Leader Selection: The coordinator select the members of the group and chooses one member
94    *         as the leader.</li>
95    *     <li>State Assignment: The leader collects the metadata from all the members of the group and
96    *         assigns state.</li>
97    *     <li>Group Stabilization: Each member receives the state assigned by the leader and begins
98    *         processing.</li>
99    * </ol>
100    *
101    * To leverage this protocol, an implementation must define the format of metadata provided by each
102    * member for group registration in {@link #metadata()} and the format of the state assignment provided
103    * by the leader in {@link #onLeaderElected(String, String, List, boolean)} and becomes available to members in
104    * {@link #onJoinComplete(int, String, String, ByteBuffer)}.
105    *
106    * Note on locking: this class shares state between the caller and a background thread which is
107    * used for sending heartbeats after the client has joined the group. All mutable state as well as
108    * state transitions are protected with the class's monitor. Generally this means acquiring the lock
109    * before reading or writing the state of the group (e.g. generation, memberId) and holding the lock
110    * when sending a request that affects the state of the group (e.g. JoinGroup, LeaveGroup).
111    */

232       /**
233        * Visible for testing.
234        *
235        * Ensure that the coordinator is ready to receive requests.
236        *
237        * @param timer Timer bounding how long this method can block
238        * @return true If coordinator discovery and initial connection succeeded, false otherwise
239        */
240       protected synchronized boolean ensureCoordinatorReady(final Timer timer) {
241           if (!coordinatorUnknown())
242               return true;
243   
244           do {
245               if (fatalFindCoordinatorException != null) {
246                   final RuntimeException fatalException = fatalFindCoordinatorException;
247                   fatalFindCoordinatorException = null;
248                   throw fatalException;
249               }
250               final RequestFuture<Void> future = lookupCoordinator();
251               client.poll(future, timer);
252   
253               if (!future.isDone()) {
254                   // ran out of time
255                   break;
256               }
257   
258               RuntimeException fatalException = null;
259   
260               if (future.failed()) {
261                   if (future.isRetriable()) {
262                       log.debug("Coordinator discovery failed, refreshing metadata", future.exception());
263                       client.awaitMetadataUpdate(timer);
264                   } else {
265                       fatalException = future.exception();
266                       log.info("FindCoordinator request hit fatal exception", fatalException);
267                   }
268               } else if (coordinator != null && client.isUnavailable(coordinator)) {
269                   // we found the coordinator, but the connection has failed, so mark
270                   // it dead and backoff before retrying discovery
271                   markCoordinatorUnknown("coordinator unavailable");
272                   timer.sleep(rebalanceConfig.retryBackoffMs);
273               }
274   
275               clearFindCoordinatorFuture();
276               if (fatalException != null)
277                   throw fatalException;
278           } while (coordinatorUnknown() && timer.notExpired());
279   
280           return !coordinatorUnknown();
281       }

327                   RuntimeException cause = heartbeatThread.failureCause();
328                   heartbeatThread = null;
329                   throw cause;
330               }
331               // Awake the heartbeat thread if needed
332               if (heartbeat.shouldHeartbeat(now)) {
333                   notify();
334               }
335               heartbeat.poll(now);
336           }
337       }

398       /**
399        * Joins the group without starting the heartbeat thread.
400        *
401        * If this function returns true, the state must always be in STABLE and heartbeat enabled.
402        * If this function returns false, the state can be in one of the following:
403        *  * UNJOINED: got error response but times out before being able to re-join, heartbeat disabled
404        *  * PREPARING_REBALANCE: not yet received join-group response before timeout, heartbeat disabled
405        *  * COMPLETING_REBALANCE: not yet received sync-group response before timeout, heartbeat enabled
406        *
407        * Visible for testing.
408        *
409        * @param timer Timer bounding how long this method can block
410        * @throws KafkaException if the callback throws exception
411        * @return true iff the operation succeeded
412        */
413       boolean joinGroupIfNeeded(final Timer timer) {
414           while (rejoinNeededOrPending()) {
415               if (!ensureCoordinatorReady(timer)) {
416                   return false;
417               }
418   
419               // call onJoinPrepare if needed. We set a flag to make sure that we do not call it a second
420               // time if the client is woken up before a pending rebalance completes. This must be called
421               // on each iteration of the loop because an event requiring a rebalance (such as a metadata
422               // refresh which changes the matched subscription set) can occur while another rebalance is
423               // still in progress.
424               if (needsJoinPrepare) {
425                   // need to set the flag before calling onJoinPrepare since the user callback may throw
426                   // exception, in which case upon retry we should not retry onJoinPrepare either.
427                   needsJoinPrepare = false;
428                   // return false when onJoinPrepare is waiting for committing offset
429                   if (!onJoinPrepare(generation.generationId, generation.memberId)) {
430                       needsJoinPrepare = true;
431                       //should not initiateJoinGroup if needsJoinPrepare still is true
432                       return false;
433                   }
434               }
435   
436               final RequestFuture<ByteBuffer> future = initiateJoinGroup();
437               client.poll(future, timer);
438               if (!future.isDone()) {
439                   // we ran out of time
440                   return false;
441               }
442   
443               if (future.succeeded()) {
444                   Generation generationSnapshot;
445                   MemberState stateSnapshot;
446   
447                   // Generation data maybe concurrently cleared by Heartbeat thread.
448                   // Can't use synchronized for {@code onJoinComplete}, because it can be long enough
449                   // and shouldn't block heartbeat thread.
450                   // See {@link PlaintextConsumerTest#testMaxPollIntervalMsDelayInAssignment}
451                   synchronized (AbstractCoordinator.this) {
452                       generationSnapshot = this.generation;
453                       stateSnapshot = this.state;
454                   }
455   
456                   if (!hasGenerationReset(generationSnapshot) && stateSnapshot == MemberState.STABLE) {
457                       // Duplicate the buffer in case `onJoinComplete` does not complete and needs to be retried.
458                       ByteBuffer memberAssignment = future.value().duplicate();
459   
460                       onJoinComplete(generationSnapshot.generationId, generationSnapshot.memberId, generationSnapshot.protocolName, memberAssignment);
461   
462                       // Generally speaking we should always resetJoinGroupFuture once the future is done, but here
463                       // we can only reset the join group future after the completion callback returns. This ensures
464                       // that if the callback is woken up, we will retry it on the next joinGroupIfNeeded.
465                       // And because of that we should explicitly trigger resetJoinGroupFuture in other conditions below.
466                       resetJoinGroupFuture();
467                       needsJoinPrepare = true;
468                   } else {
469                       final String reason = String.format("rebalance failed since the generation/state was " +
470                               "modified by heartbeat thread to %s/%s before the rebalance callback triggered",
471                               generationSnapshot, stateSnapshot);
472   
473                       resetStateAndRejoin(reason, true);
474                       resetJoinGroupFuture();
475                   }
476               } else {
477                   final RuntimeException exception = future.exception();
478   
479                   resetJoinGroupFuture();
480                   synchronized (AbstractCoordinator.this) {
481                       rejoinReason = String.format("rebalance failed due to '%s' (%s)", exception.getMessage(), exception.getClass().getSimpleName());
482                       rejoinNeeded = true;
483                   }
484   
485                   if (exception instanceof UnknownMemberIdException ||
486                       exception instanceof IllegalGenerationException ||
487                       exception instanceof RebalanceInProgressException ||
488                       exception instanceof MemberIdRequiredException)
489                       continue;
490                   else if (!future.isRetriable())
491                       throw exception;
492   
493                   timer.sleep(rebalanceConfig.retryBackoffMs);
494               }
495           }
496           return true;
497       }

503       private synchronized RequestFuture<ByteBuffer> initiateJoinGroup() {
504           // we store the join future in case we are woken up by the user after beginning the
505           // rebalance in the call to poll below. This ensures that we do not mistakenly attempt
506           // to rejoin before the pending rebalance has completed.
507           if (joinFuture == null) {
508               state = MemberState.PREPARING_REBALANCE;
509               // a rebalance can be triggered consecutively if the previous one failed,
510               // in this case we would not update the start time.
511               if (lastRebalanceStartMs == -1L)
512                   lastRebalanceStartMs = time.milliseconds();
513               joinFuture = sendJoinGroupRequest();
514               joinFuture.addListener(new RequestFutureListener<ByteBuffer>() {
515                   @Override
516                   public void onSuccess(ByteBuffer value) {
517                       // do nothing since all the handler logic are in SyncGroupResponseHandler already
518                   }
519   
520                   @Override
521                   public void onFailure(RuntimeException e) {
522                       // we handle failures below after the request finishes. if the join completes
523                       // after having been woken up, the exception is ignored and we will rejoin;
524                       // this can be triggered when either join or sync request failed
525                       synchronized (AbstractCoordinator.this) {
526                           sensors.failedRebalanceSensor.record();
527                       }
528                   }
529               });
530           }
531           return joinFuture;
532       }

534       /**
535        * Join the group and return the assignment for the next generation. This function handles both
536        * JoinGroup and SyncGroup, delegating to {@link #onLeaderElected(String, String, List, boolean)} if
537        * elected leader by the coordinator.
538        *
539        * NOTE: This is visible only for testing
540        *
541        * @return A request future which wraps the assignment returned from the group leader
542        */
543       RequestFuture<ByteBuffer> sendJoinGroupRequest() {
544           if (coordinatorUnknown())
545               return RequestFuture.coordinatorNotAvailable();
546   
547           // send a join group request to the coordinator
548           log.info("(Re-)joining group");
549           JoinGroupRequest.Builder requestBuilder = new JoinGroupRequest.Builder(
550                   new JoinGroupRequestData()
551                           .setGroupId(rebalanceConfig.groupId)
552                           .setSessionTimeoutMs(this.rebalanceConfig.sessionTimeoutMs)
553                           .setMemberId(this.generation.memberId)
554                           .setGroupInstanceId(this.rebalanceConfig.groupInstanceId.orElse(null))
555                           .setProtocolType(protocolType())
556                           .setProtocols(metadata())
557                           .setRebalanceTimeoutMs(this.rebalanceConfig.rebalanceTimeoutMs)
558                           .setReason(this.rejoinReason)
559           );
560   
561           log.debug("Sending JoinGroup ({}) to coordinator {}", requestBuilder, this.coordinator);
562   
563           // Note that we override the request timeout using the rebalance timeout since that is the
564           // maximum time that it may block on the coordinator. We add an extra 5 seconds for small delays.
565           int joinGroupTimeoutMs = Math.max(
566               client.defaultRequestTimeoutMs(),
567               Math.max(
568                   rebalanceConfig.rebalanceTimeoutMs + JOIN_GROUP_TIMEOUT_LAPSE,
569                   rebalanceConfig.rebalanceTimeoutMs) // guard against overflow since rebalance timeout can be MAX_VALUE
570               );
571           return client.send(coordinator, requestBuilder, joinGroupTimeoutMs)
572                   .compose(new JoinGroupResponseHandler(generation));
573       }

575       private class JoinGroupResponseHandler extends CoordinatorResponseHandler<JoinGroupResponse, ByteBuffer> {
576           private JoinGroupResponseHandler(final Generation generation) {
577               super(generation);
578           }
579   
580           @Override
581           public void handle(JoinGroupResponse joinResponse, RequestFuture<ByteBuffer> future) {
582               Errors error = joinResponse.error();
583               if (error == Errors.NONE) {
584                   if (isProtocolTypeInconsistent(joinResponse.data().protocolType())) {
585                       log.error("JoinGroup failed: Inconsistent Protocol Type, received {} but expected {}",
586                           joinResponse.data().protocolType(), protocolType());
587                       future.raise(Errors.INCONSISTENT_GROUP_PROTOCOL);
588                   } else {
589                       log.debug("Received successful JoinGroup response: {}", joinResponse);
590                       sensors.joinSensor.record(response.requestLatencyMs());
591   
592                       synchronized (AbstractCoordinator.this) {
593                           if (state != MemberState.PREPARING_REBALANCE) {
594                               // if the consumer was woken up before a rebalance completes, we may have already left
595                               // the group. In this case, we do not want to continue with the sync group.
596                               future.raise(new UnjoinedGroupException());
597                           } else {
598                               state = MemberState.COMPLETING_REBALANCE;
599   
600                               // we only need to enable heartbeat thread whenever we transit to
601                               // COMPLETING_REBALANCE state since we always transit from this state to STABLE
602                               if (heartbeatThread != null)
603                                   heartbeatThread.enable();
604   
605                               AbstractCoordinator.this.generation = new Generation(
606                                   joinResponse.data().generationId(),
607                                   joinResponse.data().memberId(), joinResponse.data().protocolName());
608   
609                               log.info("Successfully joined group with generation {}", AbstractCoordinator.this.generation);
610   
611                               if (joinResponse.isLeader()) {
612                                   onLeaderElected(joinResponse).chain(future);
613                               } else {
614                                   onJoinFollower().chain(future);
615                               }
616                           }
617                       }
618                   }
619               } else if (error == Errors.COORDINATOR_LOAD_IN_PROGRESS) {
620                   log.info("JoinGroup failed: Coordinator {} is loading the group.", coordinator());
621                   // backoff and retry
622                   future.raise(error);
623               } else if (error == Errors.UNKNOWN_MEMBER_ID) {
624                   log.info("JoinGroup failed: {} Need to re-join the group. Sent generation was {}",
625                            error.message(), sentGeneration);
626                   // only need to reset the member id if generation has not been changed,
627                   // then retry immediately
628                   if (generationUnchanged())
629                       resetStateOnResponseError(ApiKeys.JOIN_GROUP, error, true);
630   
631                   future.raise(error);
632               } else if (error == Errors.COORDINATOR_NOT_AVAILABLE
633                       || error == Errors.NOT_COORDINATOR) {
634                   // re-discover the coordinator and retry with backoff
635                   markCoordinatorUnknown(error);
636                   log.info("JoinGroup failed: {} Marking coordinator unknown. Sent generation was {}",
637                             error.message(), sentGeneration);
638                   future.raise(error);
639               } else if (error == Errors.FENCED_INSTANCE_ID) {
640                   // for join-group request, even if the generation has changed we would not expect the instance id
641                   // gets fenced, and hence we always treat this as a fatal error
642                   log.error("JoinGroup failed: The group instance id {} has been fenced by another instance. " +
643                                 "Sent generation was {}", rebalanceConfig.groupInstanceId, sentGeneration);
644                   future.raise(error);
645               } else if (error == Errors.INCONSISTENT_GROUP_PROTOCOL
646                       || error == Errors.INVALID_SESSION_TIMEOUT
647                       || error == Errors.INVALID_GROUP_ID
648                       || error == Errors.GROUP_AUTHORIZATION_FAILED
649                       || error == Errors.GROUP_MAX_SIZE_REACHED) {
650                   // log the error and re-throw the exception
651                   log.error("JoinGroup failed due to fatal error: {}", error.message());
652                   if (error == Errors.GROUP_MAX_SIZE_REACHED) {
653                       future.raise(new GroupMaxSizeReachedException("Consumer group " + rebalanceConfig.groupId +
654                               " already has the configured maximum number of members."));
655                   } else if (error == Errors.GROUP_AUTHORIZATION_FAILED) {
656                       future.raise(GroupAuthorizationException.forGroupId(rebalanceConfig.groupId));
657                   } else {
658                       future.raise(error);
659                   }
660               } else if (error == Errors.UNSUPPORTED_VERSION) {
661                   log.error("JoinGroup failed due to unsupported version error. Please unset field group.instance.id " +
662                             "and retry to see if the problem resolves");
663                   future.raise(error);
664               } else if (error == Errors.MEMBER_ID_REQUIRED) {
665                   // Broker requires a concrete member id to be allowed to join the group. Update member id
666                   // and send another join group request in next cycle.
667                   String memberId = joinResponse.data().memberId();
668                   log.debug("JoinGroup failed due to non-fatal error: {}. Will set the member id as {} and then rejoin. " +
669                                 "Sent generation was {}", error, memberId, sentGeneration);
670                   synchronized (AbstractCoordinator.this) {
671                       AbstractCoordinator.this.generation = new Generation(OffsetCommitRequest.DEFAULT_GENERATION_ID, memberId, null);
672                   }
673                   requestRejoin("need to re-join with the given member-id: " + memberId);
674   
675                   future.raise(error);
676               } else if (error == Errors.REBALANCE_IN_PROGRESS) {
677                   log.info("JoinGroup failed due to non-fatal error: REBALANCE_IN_PROGRESS, " +
678                       "which could indicate a replication timeout on the broker. Will retry.");
679                   future.raise(error);
680               } else {
681                   // unexpected error, throw the exception
682                   log.error("JoinGroup failed due to unexpected error: {}", error.message());
683                   future.raise(new KafkaException("Unexpected error in join group response: " + error.message()));
684               }
685           }
686       }

753       private class SyncGroupResponseHandler extends CoordinatorResponseHandler<SyncGroupResponse, ByteBuffer> {
754           private SyncGroupResponseHandler(final Generation generation) {
755               super(generation);
756           }
757   
758           @Override
759           public void handle(SyncGroupResponse syncResponse,
760                              RequestFuture<ByteBuffer> future) {
761               Errors error = syncResponse.error();
762               if (error == Errors.NONE) {
763                   if (isProtocolTypeInconsistent(syncResponse.data().protocolType())) {
764                       log.error("SyncGroup failed due to inconsistent Protocol Type, received {} but expected {}",
765                           syncResponse.data().protocolType(), protocolType());
766                       future.raise(Errors.INCONSISTENT_GROUP_PROTOCOL);
767                   } else {
768                       log.debug("Received successful SyncGroup response: {}", syncResponse);
769                       sensors.syncSensor.record(response.requestLatencyMs());
770   
771                       synchronized (AbstractCoordinator.this) {
772                           if (!hasGenerationReset(generation) && state == MemberState.COMPLETING_REBALANCE) {
773                               // check protocol name only if the generation is not reset
774                               final String protocolName = syncResponse.data().protocolName();
775                               final boolean protocolNameInconsistent = protocolName != null &&
776                                   !protocolName.equals(generation.protocolName);
777   
778                               if (protocolNameInconsistent) {
779                                   log.error("SyncGroup failed due to inconsistent Protocol Name, received {} but expected {}",
780                                       protocolName, generation.protocolName);
781   
782                                   future.raise(Errors.INCONSISTENT_GROUP_PROTOCOL);
783                               } else {
784                                   log.info("Successfully synced group in generation {}", generation);
785                                   state = MemberState.STABLE;
786                                   rejoinReason = "";
787                                   rejoinNeeded = false;
788                                   // record rebalance latency
789                                   lastRebalanceEndMs = time.milliseconds();
790                                   sensors.successfulRebalanceSensor.record(lastRebalanceEndMs - lastRebalanceStartMs);
791                                   lastRebalanceStartMs = -1L;
792   
793                                   future.complete(ByteBuffer.wrap(syncResponse.data().assignment()));
794                               }
795                           } else {
796                               log.info("Generation data was cleared by heartbeat thread to {} and state is now {} before " +
797                                   "receiving SyncGroup response, marking this rebalance as failed and retry",
798                                   generation, state);
799                               // use ILLEGAL_GENERATION error code to let it retry immediately
800                               future.raise(Errors.ILLEGAL_GENERATION);
801                           }
802                       }
803                   }
804               } else {
805                   if (error == Errors.GROUP_AUTHORIZATION_FAILED) {
806                       future.raise(GroupAuthorizationException.forGroupId(rebalanceConfig.groupId));
807                   } else if (error == Errors.REBALANCE_IN_PROGRESS) {
808                       log.info("SyncGroup failed: The group began another rebalance. Need to re-join the group. " +
809                                    "Sent generation was {}", sentGeneration);
810                       // consumer didn't get assignment in this generation, so we need to reset generation
811                       // to avoid joinGroup with out-of-data ownedPartitions in cooperative rebalance
812                       resetStateOnResponseError(ApiKeys.SYNC_GROUP, error, false);
813                       future.raise(error);
814                   } else if (error == Errors.FENCED_INSTANCE_ID) {
815                       // for sync-group request, even if the generation has changed we would not expect the instance id
816                       // gets fenced, and hence we always treat this as a fatal error
817                       log.error("SyncGroup failed: The group instance id {} has been fenced by another instance. " +
818                           "Sent generation was {}", rebalanceConfig.groupInstanceId, sentGeneration);
819                       future.raise(error);
820                   } else if (error == Errors.UNKNOWN_MEMBER_ID
