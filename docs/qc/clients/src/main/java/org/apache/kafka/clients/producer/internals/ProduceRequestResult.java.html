17   package org.apache.kafka.clients.producer.internals;
18   
19   import org.apache.kafka.clients.producer.RecordMetadata;
20   import org.apache.kafka.common.TopicPartition;
21   import org.apache.kafka.common.record.RecordBatch;
22   
23   import java.util.concurrent.CountDownLatch;
24   import java.util.concurrent.TimeUnit;
25   import java.util.function.Function;
26   
27   /**
28    * A class that models the future completion of a produce request for a single partition. There is one of these per
29    * partition in a produce request and it is shared by all the {@link RecordMetadata} instances that are batched together
30    * for the same partition in the request.
31    */
32   public class ProduceRequestResult {
33   
34       private final CountDownLatch latch = new CountDownLatch(1);
35       private final TopicPartition topicPartition;
36   
37       private volatile Long baseOffset = null;
38       private volatile long logAppendTime = RecordBatch.NO_TIMESTAMP;
39       private volatile Function<Integer, RuntimeException> errorsByIndex;
40   
41       /**
42        * Create an instance of this class.
43        *
44        * @param topicPartition The topic and partition to which this record set was sent was sent
45        */
46       public ProduceRequestResult(TopicPartition topicPartition) {
47           this.topicPartition = topicPartition;
48       }
49   
50       /**
51        * Set the result of the produce request.
52        *
53        * @param baseOffset The base offset assigned to the record
54        * @param logAppendTime The log append time or -1 if CreateTime is being used
55        * @param errorsByIndex Function mapping the batch index to the exception, or null if the response was successful
56        */
57       public void set(long baseOffset, long logAppendTime, Function<Integer, RuntimeException> errorsByIndex) {
58           this.baseOffset = baseOffset;
59           this.logAppendTime = logAppendTime;
60           this.errorsByIndex = errorsByIndex;
61       }
62   
63       /**
64        * Mark this request as complete and unblock any threads waiting on its completion.
65        */
66       public void done() {
67           if (baseOffset == null)
68               throw new IllegalStateException("The method `set` must be invoked before this method.");
69           this.latch.countDown();
70       }
71   
72       /**
73        * Await the completion of this request
74        */
75       public void await() throws InterruptedException {
76           latch.await();
77       }
78   
79       /**
80        * Await the completion of this request (up to the given time interval)
81        * @param timeout The maximum time to wait
82        * @param unit The unit for the max time
83        * @return true if the request completed, false if we timed out
84        */
85       public boolean await(long timeout, TimeUnit unit) throws InterruptedException {
86           return latch.await(timeout, unit);
87       }
88   
89       /**
90        * The base offset for the request (the first offset in the record set)
91        */
92       public long baseOffset() {
93           return baseOffset;
94       }
95   
96       /**
97        * Return true if log append time is being used for this topic
98        */
99       public boolean hasLogAppendTime() {
100           return logAppendTime != RecordBatch.NO_TIMESTAMP;
101       }
102   
103       /**
104        * The log append time or -1 if CreateTime is being used
105        */
106       public long logAppendTime() {
107           return logAppendTime;
108       }
109   
110       /**
111        * The error thrown (generally on the server) while processing this request
112        */
113       public RuntimeException error(int batchIndex) {
114           if (errorsByIndex == null) {
115               return null;
116           } else {
117               return errorsByIndex.apply(batchIndex);
118           }
119       }
120   
121       /**
122        * The topic and partition to which the record was appended
123        */
124       public TopicPartition topicPartition() {
125           return topicPartition;
126       }
127   
128       /**
129        * Has the request completed?
130        */
131       public boolean completed() {
132           return this.latch.getCount() == 0L;
133       }
134   }
