36       public static class Builder extends AbstractRequest.Builder<MetadataRequest> {
37           private static final MetadataRequestData ALL_TOPICS_REQUEST_DATA = new MetadataRequestData().
38               setTopics(null).setAllowAutoTopicCreation(true);

51           public Builder(List<String> topics, boolean allowAutoTopicCreation, short minVersion, short maxVersion) {
52               super(ApiKeys.METADATA, minVersion, maxVersion);
53               MetadataRequestData data = new MetadataRequestData();
54               if (topics == null)
55                   data.setTopics(null);
56               else {
57                   topics.forEach(topic -> data.topics().add(new MetadataRequestTopic().setName(topic)));
58               }
59   
60               data.setAllowAutoTopicCreation(allowAutoTopicCreation);
61               this.data = data;
62           }
63   
64           public Builder(List<String> topics, boolean allowAutoTopicCreation) {
65               this(topics, allowAutoTopicCreation, ApiKeys.METADATA.oldestVersion(),  ApiKeys.METADATA.latestVersion());
66           }

68           public Builder(List<Uuid> topicIds) {
69               super(ApiKeys.METADATA, ApiKeys.METADATA.oldestVersion(), ApiKeys.METADATA.latestVersion());
70               MetadataRequestData data = new MetadataRequestData();
71               if (topicIds == null)
72                   data.setTopics(null);
73               else {
74                   topicIds.forEach(topicId -> data.topics().add(new MetadataRequestTopic().setTopicId(topicId)));
75               }
76   
77               // It's impossible to create topic with topicId
78               data.setAllowAutoTopicCreation(false);
79               this.data = data;
80           }

82           public static Builder allTopics() {
83               // This never causes auto-creation, but we set the boolean to true because that is the default value when
84               // deserializing V2 and older. This way, the value is consistent after serialization and deserialization.
85               return new Builder(ALL_TOPICS_REQUEST_DATA);
86           }

88           public boolean emptyTopicList() {

90           }

101           }

103           @Override
104           public MetadataRequest build(short version) {
105               if (version < 1)
106                   throw new UnsupportedVersionException("MetadataRequest versions older than 1 are not supported.");
107               if (!data.allowAutoTopicCreation() && version < 4)
108                   throw new UnsupportedVersionException("MetadataRequest versions older than 4 don't support the " +
109                           "allowAutoTopicCreation field");
110               if (data.topics() != null) {
111                   data.topics().forEach(topic -> {
112                       if (topic.name() == null && version < 12)
113                           throw new UnsupportedVersionException("MetadataRequest version " + version +
114                                   " does not support null topic names.");
115                       if (!Uuid.ZERO_UUID.equals(topic.topicId()) && version < 12)
116                           throw new UnsupportedVersionException("MetadataRequest version " + version +
117                               " does not support non-zero topic IDs.");
118                   });
119               }
120               return new MetadataRequest(data, version);
121           }

141       @Override
142       public AbstractResponse getErrorResponse(int throttleTimeMs, Throwable e) {
143           Errors error = Errors.forException(e);
144           MetadataResponseData responseData = new MetadataResponseData();
145           if (data.topics() != null) {
146               for (MetadataRequestTopic topic : data.topics()) {
147                   // the response does not allow null, so convert to empty string if necessary
148                   String topicName = topic.name() == null ? "" : topic.name();
149                   responseData.topics().add(new MetadataResponseData.MetadataResponseTopic()
150                       .setName(topicName)
151                       .setTopicId(topic.topicId())
152                       .setErrorCode(error.code())
153                       .setIsInternal(false)
154                       .setPartitions(Collections.emptyList()));
155               }
156           }
157   
158           responseData.setThrottleTimeMs(throttleTimeMs);
159           return new MetadataResponse(responseData, true);
160       }
161   
162       public boolean isAllTopics() {
163           return (data.topics() == null) ||
164               (data.topics().isEmpty() && version() == 0); // In version 0, an empty topic list indicates
165                                                            // "request metadata for all topics."
166       }
167   
168       public List<String> topics() {
169           if (isAllTopics()) // In version 0, we return null for empty topic list
170               return null;
171           else
172               return data.topics()
173                   .stream()
174                   .map(MetadataRequestTopic::name)
175                   .collect(Collectors.toList());
176       }
177   
178       public List<Uuid> topicIds() {
179           if (isAllTopics())
180               return Collections.emptyList();
181           else if (version() < 10)
182               return Collections.emptyList();
183           else
184               return data.topics()
185                       .stream()
186                       .map(MetadataRequestTopic::topicId)
187                       .collect(Collectors.toList());
188       }
189   
190       public boolean allowAutoTopicCreation() {
191           return data.allowAutoTopicCreation();
192       }

196       }

198       public static List<MetadataRequestTopic> convertToMetadataRequestTopic(final Collection<String> topics) {
199           return topics.stream().map(topic -> new MetadataRequestTopic()
200               .setName(topic))
201               .collect(Collectors.toList());
202       }

204       public static List<MetadataRequestTopic> convertTopicIdsToMetadataRequestTopic(final Collection<Uuid> topicIds) {
205           return topicIds.stream().map(topicId -> new MetadataRequestTopic()
206                   .setTopicId(topicId))
207                   .collect(Collectors.toList());
208       }
