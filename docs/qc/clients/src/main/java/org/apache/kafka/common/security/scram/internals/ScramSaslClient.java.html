44   /**
45    * SaslClient implementation for SASL/SCRAM.
46    * <p>
47    * This implementation expects a login module that populates username as
48    * the Subject's public credential and password as the private credential.
49    *
50    * @see <a href="https://tools.ietf.org/html/rfc5802">RFC 5802</a>
51    *
52    */
53   public class ScramSaslClient implements SaslClient {
54   
55       private static final Logger log = LoggerFactory.getLogger(ScramSaslClient.class);
56   
57       enum State {
58           SEND_CLIENT_FIRST_MESSAGE,
59           RECEIVE_SERVER_FIRST_MESSAGE,
60           RECEIVE_SERVER_FINAL_MESSAGE,
61           COMPLETE,
62           FAILED
63       }
64   
65       private final ScramMechanism mechanism;
66       private final CallbackHandler callbackHandler;
67       private final ScramFormatter formatter;
68       private String clientNonce;
69       private State state;
70       private byte[] saltedPassword;
71       private ScramMessages.ClientFirstMessage clientFirstMessage;
72       private ScramMessages.ServerFirstMessage serverFirstMessage;
73       private ScramMessages.ClientFinalMessage clientFinalMessage;
74   
75       public ScramSaslClient(ScramMechanism mechanism, CallbackHandler cbh) throws NoSuchAlgorithmException {
76           this.mechanism = mechanism;
77           this.callbackHandler = cbh;
78           this.formatter = new ScramFormatter(mechanism);
79           setState(State.SEND_CLIENT_FIRST_MESSAGE);
80       }
81   
82       @Override
83       public String getMechanismName() {
84           return mechanism.mechanismName();
85       }
86   
87       @Override
88       public boolean hasInitialResponse() {
89           return true;
90       }
91   
92       @Override
93       public byte[] evaluateChallenge(byte[] challenge) throws SaslException {
94           try {
95               switch (state) {
96                   case SEND_CLIENT_FIRST_MESSAGE:
97                       if (challenge != null && challenge.length != 0)
98                           throw new SaslException("Expected empty challenge");
99                       clientNonce = formatter.secureRandomString();
100                       NameCallback nameCallback = new NameCallback("Name:");
101                       ScramExtensionsCallback extensionsCallback = new ScramExtensionsCallback();
102   
103                       try {
104                           callbackHandler.handle(new Callback[]{nameCallback});
105                           try {
106                               callbackHandler.handle(new Callback[]{extensionsCallback});
107                           } catch (UnsupportedCallbackException e) {
108                               log.debug("Extensions callback is not supported by client callback handler {}, no extensions will be added",
109                                       callbackHandler);
110                           }
111                       } catch (Throwable e) {
112                           throw new SaslException("User name or extensions could not be obtained", e);
113                       }
114   
115                       String username = nameCallback.getName();
116                       String saslName = ScramFormatter.saslName(username);
117                       Map<String, String> extensions = extensionsCallback.extensions();
118                       this.clientFirstMessage = new ScramMessages.ClientFirstMessage(saslName, clientNonce, extensions);
119                       setState(State.RECEIVE_SERVER_FIRST_MESSAGE);
120                       return clientFirstMessage.toBytes();
121   
122                   case RECEIVE_SERVER_FIRST_MESSAGE:
123                       this.serverFirstMessage = new ServerFirstMessage(challenge);
124                       if (!serverFirstMessage.nonce().startsWith(clientNonce))
125                           throw new SaslException("Invalid server nonce: does not start with client nonce");
126                       if (serverFirstMessage.iterations() < mechanism.minIterations())
127                           throw new SaslException("Requested iterations " + serverFirstMessage.iterations() +  " is less than the minimum " + mechanism.minIterations() + " for " + mechanism);
128                       PasswordCallback passwordCallback = new PasswordCallback("Password:", false);
129                       try {
130                           callbackHandler.handle(new Callback[]{passwordCallback});
131                       } catch (Throwable e) {
132                           throw new SaslException("User name could not be obtained", e);
133                       }
134                       this.clientFinalMessage = handleServerFirstMessage(passwordCallback.getPassword());
135                       setState(State.RECEIVE_SERVER_FINAL_MESSAGE);
136                       return clientFinalMessage.toBytes();
137   
138                   case RECEIVE_SERVER_FINAL_MESSAGE:
139                       ServerFinalMessage serverFinalMessage = new ServerFinalMessage(challenge);
140                       if (serverFinalMessage.error() != null)
141                           throw new SaslException("Sasl authentication using " + mechanism + " failed with error: " + serverFinalMessage.error());
142                       handleServerFinalMessage(serverFinalMessage.serverSignature());
143                       setState(State.COMPLETE);
144                       return null;
145   
146                   default:
147                       throw new IllegalSaslStateException("Unexpected challenge in Sasl client state " + state);
148               }
149           } catch (SaslException e) {
150               setState(State.FAILED);
151               throw e;
152           }
153       }
154   
155       @Override
156       public boolean isComplete() {
157           return state == State.COMPLETE;
158       }
159   
160       @Override
161       public byte[] unwrap(byte[] incoming, int offset, int len) {
162           if (!isComplete())
163               throw new IllegalStateException("Authentication exchange has not completed");
164           return Arrays.copyOfRange(incoming, offset, offset + len);
165       }
166   
167       @Override
168       public byte[] wrap(byte[] outgoing, int offset, int len) {
169           if (!isComplete())
170               throw new IllegalStateException("Authentication exchange has not completed");
171           return Arrays.copyOfRange(outgoing, offset, offset + len);
172       }
173   
174       @Override
175       public Object getNegotiatedProperty(String propName) {
176           if (!isComplete())
177               throw new IllegalStateException("Authentication exchange has not completed");
178           return null;
179       }
180   
181       @Override
182       public void dispose() {
183       }
184   
185       private void setState(State state) {
186           log.debug("Setting SASL/{} client state to {}", mechanism, state);
187           this.state = state;
188       }
189   
190       private ClientFinalMessage handleServerFirstMessage(char[] password) throws SaslException {
191           try {
192               byte[] passwordBytes = ScramFormatter.normalize(new String(password));
193               this.saltedPassword = formatter.hi(passwordBytes, serverFirstMessage.salt(), serverFirstMessage.iterations());
194   
195               ClientFinalMessage clientFinalMessage = new ClientFinalMessage("n,,".getBytes(StandardCharsets.UTF_8), serverFirstMessage.nonce());
196               byte[] clientProof = formatter.clientProof(saltedPassword, clientFirstMessage, serverFirstMessage, clientFinalMessage);
197               clientFinalMessage.proof(clientProof);
198               return clientFinalMessage;
199           } catch (InvalidKeyException e) {
200               throw new SaslException("Client final message could not be created", e);
201           }
202       }
203   
204       private void handleServerFinalMessage(byte[] signature) throws SaslException {
205           try {
206               byte[] serverKey = formatter.serverKey(saltedPassword);
207               byte[] serverSignature = formatter.serverSignature(serverKey, clientFirstMessage, serverFirstMessage, clientFinalMessage);
208               if (!MessageDigest.isEqual(signature, serverSignature))
209                   throw new SaslException("Invalid server signature in server final message");
210           } catch (InvalidKeyException e) {
211               throw new SaslException("Sasl server signature verification failed", e);
212           }
213       }

215       public static class ScramSaslClientFactory implements SaslClientFactory {
216   
217           @Override
218           public SaslClient createSaslClient(String[] mechanisms,
219                   String authorizationId,
220                   String protocol,
221                   String serverName,
222                   Map<String, ?> props,
223                   CallbackHandler cbh) throws SaslException {
224   
225               ScramMechanism mechanism = null;
226               for (String mech : mechanisms) {
227                   mechanism = ScramMechanism.forMechanismName(mech);
228                   if (mechanism != null)
229                       break;
230               }
231               if (mechanism == null)
232                   throw new SaslException(String.format("Requested mechanisms '%s' not supported. Supported mechanisms are '%s'.",
233                           Arrays.asList(mechanisms), ScramMechanism.mechanismNames()));
234   
235               try {
236                   return new ScramSaslClient(mechanism, cbh);
237               } catch (NoSuchAlgorithmException e) {
238                   throw new SaslException("Hash algorithm not supported for mechanism " + mechanism, e);
239               }
240           }
241   
242           @Override
243           public String[] getMechanismNames(Map<String, ?> props) {
244               Collection<String> mechanisms = ScramMechanism.mechanismNames();
245               return mechanisms.toArray(new String[0]);
246           }
247       }
