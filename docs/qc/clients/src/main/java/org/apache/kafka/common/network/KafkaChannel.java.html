35   /**
36    * A Kafka connection either existing on a client (which could be a broker in an
37    * inter-broker scenario) and representing the channel to a remote broker or the
38    * reverse (existing on a broker and representing the channel to a remote
39    * client, which could be a broker in an inter-broker scenario).
40    * <p>
41    * Each instance has the following:
42    * <ul>
43    * <li>a unique ID identifying it in the {@code KafkaClient} instance via which
44    * the connection was made on the client-side or in the instance where it was
45    * accepted on the server-side</li>
46    * <li>a reference to the underlying {@link TransportLayer} to allow reading and
47    * writing</li>
48    * <li>an {@link Authenticator} that performs the authentication (or
49    * re-authentication, if that feature is enabled and it applies to this
50    * connection) by reading and writing directly from/to the same
51    * {@link TransportLayer}.</li>
52    * <li>a {@link MemoryPool} into which responses are read (typically the JVM
53    * heap for clients, though smaller pools can be used for brokers and for
54    * testing out-of-memory scenarios)</li>
55    * <li>a {@link NetworkReceive} representing the current incomplete/in-progress
56    * request (from the server-side perspective) or response (from the client-side
57    * perspective) being read, if applicable; or a non-null value that has had no
58    * data read into it yet or a null value if there is no in-progress
59    * request/response (either could be the case)</li>
60    * <li>a {@link Send} representing the current request (from the client-side
61    * perspective) or response (from the server-side perspective) that is either
62    * waiting to be sent or partially sent, if applicable, or null</li>
63    * <li>a {@link ChannelMuteState} to document if the channel has been muted due
64    * to memory pressure or other reasons</li>
65    * </ul>
66    */
67   public class KafkaChannel implements AutoCloseable {
68       private static final long MIN_REAUTH_INTERVAL_ONE_SECOND_NANOS = 1000 * 1000 * 1000;

70       /**
71        * Mute States for KafkaChannel:
72        * <ul>
73        *   <li> NOT_MUTED: Channel is not muted. This is the default state. </li>
74        *   <li> MUTED: Channel is muted. Channel must be in this state to be unmuted. </li>
75        *   <li> MUTED_AND_RESPONSE_PENDING: (SocketServer only) Channel is muted and SocketServer has not sent a response
76        *                                    back to the client yet (acks != 0) or is currently waiting to receive a
77        *                                    response from the API layer (acks == 0). </li>
78        *   <li> MUTED_AND_THROTTLED: (SocketServer only) Channel is muted and throttling is in progress due to quota
79        *                             violation. </li>
80        *   <li> MUTED_AND_THROTTLED_AND_RESPONSE_PENDING: (SocketServer only) Channel is muted, throttling is in progress,
81        *                                                  and a response is currently pending. </li>
82        * </ul>
83        */
84       public enum ChannelMuteState {
85           NOT_MUTED,
86           MUTED,
87           MUTED_AND_RESPONSE_PENDING,
88           MUTED_AND_THROTTLED,
89           MUTED_AND_THROTTLED_AND_RESPONSE_PENDING
90       }
91   
92       /** Socket server events that will change the mute state:
93        * <ul>
94        *   <li> REQUEST_RECEIVED: A request has been received from the client. </li>
95        *   <li> RESPONSE_SENT: A response has been sent out to the client (ack != 0) or SocketServer has heard back from
96        *                       the API layer (acks = 0) </li>
97        *   <li> THROTTLE_STARTED: Throttling started due to quota violation. </li>
98        *   <li> THROTTLE_ENDED: Throttling ended. </li>
99        * </ul>
100        *
101        * Valid transitions on each event are:
102        * <ul>
103        *   <li> REQUEST_RECEIVED: MUTED => MUTED_AND_RESPONSE_PENDING </li>
104        *   <li> RESPONSE_SENT:    MUTED_AND_RESPONSE_PENDING => MUTED, MUTED_AND_THROTTLED_AND_RESPONSE_PENDING => MUTED_AND_THROTTLED </li>
105        *   <li> THROTTLE_STARTED: MUTED_AND_RESPONSE_PENDING => MUTED_AND_THROTTLED_AND_RESPONSE_PENDING </li>
106        *   <li> THROTTLE_ENDED:   MUTED_AND_THROTTLED => MUTED, MUTED_AND_THROTTLED_AND_RESPONSE_PENDING => MUTED_AND_RESPONSE_PENDING </li>
107        * </ul>
108        */
109       public enum ChannelMuteEvent {
110           REQUEST_RECEIVED,
111           RESPONSE_SENT,
112           THROTTLE_STARTED,
113           THROTTLE_ENDED
114       }

169       /**
170        * Does handshake of transportLayer and authentication using configured authenticator.
171        * For SSL with client authentication enabled, {@link TransportLayer#handshake()} performs
172        * authentication. For SASL, authentication is performed by {@link Authenticator#authenticate()}.
173        */
174       public void prepare() throws AuthenticationException, IOException {
175           boolean authenticating = false;
176           try {
177               if (!transportLayer.ready())
178                   transportLayer.handshake();
179               if (transportLayer.ready() && !authenticator.complete()) {
180                   authenticating = true;
181                   authenticator.authenticate();
182               }
183           } catch (AuthenticationException e) {
184               // Clients are notified of authentication exceptions to enable operations to be terminated
185               // without retries. Other errors are handled as network exceptions in Selector.
186               String remoteDesc = remoteAddress != null ? remoteAddress.toString() : null;
187               state = new ChannelState(ChannelState.State.AUTHENTICATION_FAILED, e, remoteDesc);
188               if (authenticating) {
189                   delayCloseOnAuthenticationFailure();
190                   throw new DelayedResponseAuthenticationException(e);
191               }
192               throw e;
193           }
194           if (ready()) {
195               ++successfulAuthentications;
196               state = ChannelState.READY;
197           }
198       }

249       /**
250        * externally muting a channel should be done via selector to ensure proper state handling
251        */
252       void mute() {
253           if (muteState == ChannelMuteState.NOT_MUTED) {
254               if (!disconnected) transportLayer.removeInterestOps(SelectionKey.OP_READ);
255               muteState = ChannelMuteState.MUTED;
256           }
257       }

259       /**
260        * Unmute the channel. The channel can be unmuted only if it is in the MUTED state. For other muted states
261        * (MUTED_AND_*), this is a no-op.
262        *
263        * @return Whether or not the channel is in the NOT_MUTED state after the call
264        */
265       boolean maybeUnmute() {
266           if (muteState == ChannelMuteState.MUTED) {
267               if (!disconnected) transportLayer.addInterestOps(SelectionKey.OP_READ);
268               muteState = ChannelMuteState.NOT_MUTED;
269           }
270           return muteState == ChannelMuteState.NOT_MUTED;
271       }

273       // Handle the specified channel mute-related event and transition the mute state according to the state machine.
274       public void handleChannelMuteEvent(ChannelMuteEvent event) {
275           boolean stateChanged = false;
276           switch (event) {
277               case REQUEST_RECEIVED:
278                   if (muteState == ChannelMuteState.MUTED) {
279                       muteState = ChannelMuteState.MUTED_AND_RESPONSE_PENDING;
280                       stateChanged = true;
281                   }
282                   break;
283               case RESPONSE_SENT:
284                   if (muteState == ChannelMuteState.MUTED_AND_RESPONSE_PENDING) {
285                       muteState = ChannelMuteState.MUTED;
286                       stateChanged = true;
287                   }
288                   if (muteState == ChannelMuteState.MUTED_AND_THROTTLED_AND_RESPONSE_PENDING) {
289                       muteState = ChannelMuteState.MUTED_AND_THROTTLED;
290                       stateChanged = true;
291                   }
292                   break;
293               case THROTTLE_STARTED:
294                   if (muteState == ChannelMuteState.MUTED_AND_RESPONSE_PENDING) {
295                       muteState = ChannelMuteState.MUTED_AND_THROTTLED_AND_RESPONSE_PENDING;
296                       stateChanged = true;
297                   }
298                   break;
299               case THROTTLE_ENDED:
300                   if (muteState == ChannelMuteState.MUTED_AND_THROTTLED) {
301                       muteState = ChannelMuteState.MUTED;
302                       stateChanged = true;
303                   }
304                   if (muteState == ChannelMuteState.MUTED_AND_THROTTLED_AND_RESPONSE_PENDING) {
305                       muteState = ChannelMuteState.MUTED_AND_RESPONSE_PENDING;
306                       stateChanged = true;
307                   }
308           }
309           if (!stateChanged) {
310               throw new IllegalStateException("Cannot transition from " + muteState.name() + " for " + event.name());
311           }
312       }

343       public boolean isInMutableState() {
344           //some requests do not require memory, so if we do not know what the current (or future) request is
345           //(receive == null) we dont mute. we also dont mute if whatever memory required has already been
346           //successfully allocated (if none is required for the currently-being-read request
347           //receive.memoryAllocated() is expected to return true)
348           if (receive == null || receive.memoryAllocated())
349               return false;
350           //also cannot mute if underlying transport is not in the ready state
351           return transportLayer.ready();
352       }

397       public long read() throws IOException {
398           if (receive == null) {
399               receive = new NetworkReceive(maxReceiveSize, id, memoryPool);
400           }
401

405               //pool must be out of memory, mute ourselves.
406               mute();
407           }

409       }

502       /**
503        * If this is a server-side connection that has an expiration time and at least
504        * 1 second has passed since the prior re-authentication (if any) started then
505        * begin the process of re-authenticating the connection and return true,
506        * otherwise return false
507        * 
508        * @param saslHandshakeNetworkReceive
509        *            the mandatory {@link NetworkReceive} containing the
510        *            {@code SaslHandshakeRequest} that has been received on the server
511        *            and that initiates re-authentication.
512        * @param nowNanosSupplier
513        *            {@code Supplier} of the current time. The value must be in
514        *            nanoseconds as per {@code System.nanoTime()} and is therefore only
515        *            useful when compared to such a value -- it's absolute value is
516        *            meaningless.
517        * 
518        * @return true if this is a server-side connection that has an expiration time
519        *         and at least 1 second has passed since the prior re-authentication
520        *         (if any) started to indicate that the re-authentication process has
521        *         begun, otherwise false
522        * @throws AuthenticationException
523        *             if re-authentication fails due to invalid credentials or other
524        *             security configuration errors
525        * @throws IOException
526        *             if read/write fails due to an I/O error
527        * @throws IllegalStateException
528        *             if this channel is not "ready"
529        */
530       public boolean maybeBeginServerReauthentication(NetworkReceive saslHandshakeNetworkReceive,
531               Supplier<Long> nowNanosSupplier) throws AuthenticationException, IOException {
532           if (!ready())
533               throw new IllegalStateException(
534                       "KafkaChannel should be \"ready\" when processing SASL Handshake for potential re-authentication");
535           /*
536            * Re-authentication is disabled if there is no session expiration time, in
537            * which case the SASL handshake network receive will be processed normally,
538            * which results in a failure result being sent to the client. Also, no need to
539            * check if we are muted since since we are processing a received packet when we
540            * invoke this.
541            */
542           if (authenticator.serverSessionExpirationTimeNanos() == null)
543               return false;
544           /*
545            * We've delayed getting the time as long as possible in case we don't need it,
546            * but at this point we need it -- so get it now.
547            */
548           long nowNanos = nowNanosSupplier.get();
549           /*
550            * Cannot re-authenticate more than once every second; an attempt to do so will
551            * result in the SASL handshake network receive being processed normally, which
552            * results in a failure result being sent to the client.
553            */
554           if (lastReauthenticationStartNanos != 0
555                   && nowNanos - lastReauthenticationStartNanos < MIN_REAUTH_INTERVAL_ONE_SECOND_NANOS)
556               return false;
557           lastReauthenticationStartNanos = nowNanos;
558           swapAuthenticatorsAndBeginReauthentication(
559                   new ReauthenticationContext(authenticator, saslHandshakeNetworkReceive, nowNanos));
560           return true;
561       }

563       /**
564        * If this is a client-side connection that is not muted, there is no
565        * in-progress write, and there is a session expiration time defined that has
566        * past then begin the process of re-authenticating the connection and return
567        * true, otherwise return false
568        * 
569        * @param nowNanosSupplier
570        *            {@code Supplier} of the current time. The value must be in
571        *            nanoseconds as per {@code System.nanoTime()} and is therefore only
572        *            useful when compared to such a value -- it's absolute value is
573        *            meaningless.
574        * 
575        * @return true if this is a client-side connection that is not muted, there is
576        *         no in-progress write, and there is a session expiration time defined
577        *         that has past to indicate that the re-authentication process has
578        *         begun, otherwise false
579        * @throws AuthenticationException
580        *             if re-authentication fails due to invalid credentials or other
581        *             security configuration errors
582        * @throws IOException
583        *             if read/write fails due to an I/O error
584        * @throws IllegalStateException
585        *             if this channel is not "ready"
586        */
587       public boolean maybeBeginClientReauthentication(Supplier<Long> nowNanosSupplier)
588               throws AuthenticationException, IOException {
589           if (!ready())
590               throw new IllegalStateException(
591                       "KafkaChannel should always be \"ready\" when it is checked for possible re-authentication");
592           if (muteState != ChannelMuteState.NOT_MUTED || midWrite
593                   || authenticator.clientSessionReauthenticationTimeNanos() == null)
594               return false;
595           /*
596            * We've delayed getting the time as long as possible in case we don't need it,
597            * but at this point we need it -- so get it now.
598            */
599           long nowNanos = nowNanosSupplier.get();
600           if (nowNanos < authenticator.clientSessionReauthenticationTimeNanos())
601               return false;
602           swapAuthenticatorsAndBeginReauthentication(new ReauthenticationContext(authenticator, receive, nowNanos));
603           receive = null;
604           return true;
605       }
