43   /**
44    * This wrapper supports all versions of the Fetch API
45    *
46    * Possible error codes:
47    *
48    * - {@link Errors#OFFSET_OUT_OF_RANGE} If the fetch offset is out of range for a requested partition
49    * - {@link Errors#TOPIC_AUTHORIZATION_FAILED} If the user does not have READ access to a requested topic
50    * - {@link Errors#REPLICA_NOT_AVAILABLE} If the request is received by a broker with version < 2.6 which is not a replica
51    * - {@link Errors#NOT_LEADER_OR_FOLLOWER} If the broker is not a leader or follower and either the provided leader epoch
52    *     matches the known leader epoch on the broker or is empty
53    * - {@link Errors#FENCED_LEADER_EPOCH} If the epoch is lower than the broker's epoch
54    * - {@link Errors#UNKNOWN_LEADER_EPOCH} If the epoch is larger than the broker's epoch
55    * - {@link Errors#UNKNOWN_TOPIC_OR_PARTITION} If the broker does not have metadata for a topic or partition
56    * - {@link Errors#KAFKA_STORAGE_ERROR} If the log directory for one of the requested partitions is offline
57    * - {@link Errors#UNSUPPORTED_COMPRESSION_TYPE} If a fetched topic is using a compression type which is
58    *     not supported by the fetch request version
59    * - {@link Errors#CORRUPT_MESSAGE} If corrupt message encountered, e.g. when the broker scans the log to find
60    *     the fetch offset after the index lookup
61    * - {@link Errors#UNKNOWN_TOPIC_ID} If the request contains a topic ID unknown to the broker
62    * - {@link Errors#FETCH_SESSION_TOPIC_ID_ERROR} If the request version supports topic IDs but the session does not or vice versa,
63    *     or a topic ID in the request is inconsistent with a topic ID in the session
64    * - {@link Errors#INCONSISTENT_TOPIC_ID} If a topic ID in the session does not match the topic ID in the log
65    * - {@link Errors#UNKNOWN_SERVER_ERROR} For any unexpected errors
66    */
67   public class FetchResponse extends AbstractResponse {
68       public static final long INVALID_HIGH_WATERMARK = -1L;
69       public static final long INVALID_LAST_STABLE_OFFSET = -1L;
70       public static final long INVALID_LOG_START_OFFSET = -1L;
71       public static final int INVALID_PREFERRED_REPLICA_ID = -1;

82       /**
83        * From version 3 or later, the authorized and existing entries in `FetchRequest.fetchData` should be in the same order in `responseData`.
84        * Version 13 introduces topic IDs which can lead to a few new errors. If there is any unknown topic ID in the request, the
85        * response will contain a partition-level UNKNOWN_TOPIC_ID error for that partition.
86        * If a request's topic ID usage is inconsistent with the session, we will return a top level FETCH_SESSION_TOPIC_ID_ERROR error.
87        * We may also return INCONSISTENT_TOPIC_ID error as a partition-level error when a partition in the session has a topic ID
88        * inconsistent with the log.
89        */
90       public FetchResponse(FetchResponseData fetchResponseData) {
91           super(ApiKeys.FETCH);
92           this.data = fetchResponseData;
93       }

95       public Errors error() {
96           return Errors.forCode(data.errorCode());
97       }

99       public LinkedHashMap<TopicPartition, FetchResponseData.PartitionData> responseData(Map<Uuid, String> topicNames, short version) {
100           if (responseData == null) {
101               synchronized (this) {
102                   if (responseData == null) {
103                       // Assigning the lazy-initialized `responseData` in the last step
104                       // to avoid other threads accessing a half-initialized object.
105                       final LinkedHashMap<TopicPartition, FetchResponseData.PartitionData> responseDataTmp =
106                               new LinkedHashMap<>();
107                       data.responses().forEach(topicResponse -> {
108                           String name;
109                           if (version < 13) {
110                               name = topicResponse.topic();
111                           } else {
112                               name = topicNames.get(topicResponse.topicId());
113                           }
114                           if (name != null) {
115                               topicResponse.partitions().forEach(partition ->
116                                   responseDataTmp.put(new TopicPartition(name, partition.partitionIndex()), partition));
117                           }
118                       });
119                       responseData = responseDataTmp;
120                   }
121               }
122           }
123           return responseData;
124       }

158        *

165                                FetchResponseData.PartitionData>> partIterator) {
166           // Since the throttleTimeMs and metadata field sizes are constant and fixed, we can

207       /**
208        * Returns `partition.records` as `Records` (instead of `BaseRecords`). If `records` is `null`, returns `MemoryRecords.EMPTY`.
209        *
210        * If this response was deserialized after a fetch, this method should never fail. An example where this would
211        * fail is a down-converted response (e.g. LazyDownConversionRecords) on the broker (before it's serialized and
212        * sent on the wire).
213        *
214        * @param partition partition data
215        * @return Records or empty record if the records in PartitionData is null.
216        */
217       public static Records recordsOrFail(FetchResponseData.PartitionData partition) {
218           if (partition.records() == null) return MemoryRecords.EMPTY;
219           if (partition.records() instanceof Records) return (Records) partition.records();
220           throw new ClassCastException("The record type is " + partition.records().getClass().getSimpleName() + ", which is not a subtype of " +
221               Records.class.getSimpleName() + ". This method is only safe to call if the `FetchResponse` was deserialized from bytes.");
222       }

239       private static boolean matchingTopic(FetchResponseData.FetchableTopicResponse previousTopic, TopicIdPartition currentTopic) {
240           if (previousTopic == null)
241               return false;
242           if (!previousTopic.topicId().equals(Uuid.ZERO_UUID))
243               return previousTopic.topicId().equals(currentTopic.topicId());
244           else
245               return previousTopic.topic().equals(currentTopic.topicPartition().topic());
246   
247       }

249       private static FetchResponseData toMessage(Errors error,
250                                                  int throttleTimeMs,
251                                                  int sessionId,
252                                                  Iterator<Map.Entry<TopicIdPartition, FetchResponseData.PartitionData>> partIterator) {
253           List<FetchResponseData.FetchableTopicResponse> topicResponseList = new ArrayList<>();
254           while (partIterator.hasNext()) {
255               Map.Entry<TopicIdPartition, FetchResponseData.PartitionData> entry = partIterator.next();
256               FetchResponseData.PartitionData partitionData = entry.getValue();
257               // Since PartitionData alone doesn't know the partition ID, we set it here
258               partitionData.setPartitionIndex(entry.getKey().topicPartition().partition());
259               // We have to keep the order of input topic-partition. Hence, we batch the partitions only if the last
260               // batch is in the same topic group.
261               FetchResponseData.FetchableTopicResponse previousTopic = topicResponseList.isEmpty() ? null
262                   : topicResponseList.get(topicResponseList.size() - 1);
263               if (matchingTopic(previousTopic, entry.getKey()))
264                   previousTopic.partitions().add(partitionData);
265               else {
266                   List<FetchResponseData.PartitionData> partitionResponses = new ArrayList<>();
267                   partitionResponses.add(partitionData);
268                   topicResponseList.add(new FetchResponseData.FetchableTopicResponse()
269                       .setTopic(entry.getKey().topicPartition().topic())
270                       .setTopicId(entry.getKey().topicId())
271                       .setPartitions(partitionResponses));
272               }
273           }
274   
275           return new FetchResponseData()
276               .setThrottleTimeMs(throttleTimeMs)
277               .setErrorCode(error.code())
278               .setSessionId(sessionId)
279               .setResponses(topicResponseList);
280       }
