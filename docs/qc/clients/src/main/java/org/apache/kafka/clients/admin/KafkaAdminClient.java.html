633       @Override
634       public void close(Duration timeout) {
635           long waitTimeMs = timeout.toMillis();
636           if (waitTimeMs < 0)
637               throw new IllegalArgumentException("The timeout cannot be negative.");
638           waitTimeMs = Math.min(TimeUnit.DAYS.toMillis(365), waitTimeMs); // Limit the timeout to a year.
639           long now = time.milliseconds();
640           long newHardShutdownTimeMs = now + waitTimeMs;
641           long prev = INVALID_SHUTDOWN_TIME;
642           while (true) {
643               if (hardShutdownTimeMs.compareAndSet(prev, newHardShutdownTimeMs)) {
644                   if (prev == INVALID_SHUTDOWN_TIME) {
645                       log.debug("Initiating close operation.");
646                   } else {
647                       log.debug("Moving hard shutdown time forward.");
648                   }
649                   client.wakeup(); // Wake the thread, if it is blocked inside poll().
650                   break;
651               }
652               prev = hardShutdownTimeMs.get();
653               if (prev < newHardShutdownTimeMs) {
654                   log.debug("Hard shutdown time is already earlier than requested.");
655                   newHardShutdownTimeMs = prev;
656                   break;
657               }
658           }
659           if (log.isDebugEnabled()) {
660               long deltaMs = Math.max(0, newHardShutdownTimeMs - time.milliseconds());
661               log.debug("Waiting for the I/O thread to exit. Hard shutdown in {} ms.", deltaMs);
662           }
663           try {
664               // close() can be called by AdminClient thread when it invokes callback. That will
665               // cause deadlock, so check for that condition.
666               if (Thread.currentThread() != thread) {
667                   // Wait for the thread to be joined.
668                   thread.join(waitTimeMs);
669               }
670               log.debug("Kafka admin client closed.");
671           } catch (InterruptedException e) {
672               log.debug("Interrupted while joining I/O thread", e);
673               Thread.currentThread().interrupt();
674           }

784           /**
785            * Handle a failure.
786            *
787            * Depending on what the exception is and how many times we have already tried, we may choose to
788            * fail the Call, or retry it. It is important to print the stack traces here in some cases,
789            * since they are not necessarily preserved in ApiVersionException objects.
790            *
791            * @param now           The current time in milliseconds.
792            * @param throwable     The failure exception.
793            */
794           final void fail(long now, Throwable throwable) {
795               if (curNode != null) {
796                   runnable.nodeReadyDeadlines.remove(curNode);
797                   curNode = null;
798               }
799               // If the admin client is closing, we can't retry.
800               if (runnable.closing) {
801                   handleFailure(throwable);
802                   return;
803               }
804               // If this is an UnsupportedVersionException that we can retry, do so. Note that a
805               // protocol downgrade will not count against the total number of retries we get for
806               // this RPC. That is why 'tries' is not incremented.
807               if ((throwable instanceof UnsupportedVersionException) &&
808                        handleUnsupportedVersionException((UnsupportedVersionException) throwable)) {
809                   log.debug("{} attempting protocol downgrade and then retry.", this);
810                   runnable.pendingCalls.add(this);
811                   return;
812               }
813               tries++;
814               nextAllowedTryMs = now + retryBackoffMs;
815   
816               // If the call has timed out, fail.
817               if (calcTimeoutMsRemainingAsInt(now, deadlineMs) <= 0) {
818                   handleTimeoutFailure(now, throwable);
819                   return;
820               }
821               // If the exception is not retriable, fail.
822               if (!(throwable instanceof RetriableException)) {
823                   if (log.isDebugEnabled()) {
824                       log.debug("{} failed with non-retriable exception after {} attempt(s)", this, tries,
825                           new Exception(prettyPrintException(throwable)));
826                   }
827                   handleFailure(throwable);
828                   return;
829               }
830               // If we are out of retries, fail.
831               if (tries > maxRetries) {
832                   handleTimeoutFailure(now, throwable);
833                   return;
834               }
835               if (log.isDebugEnabled()) {
836                   log.debug("{} failed: {}. Beginning retry #{}",
837                       this, prettyPrintException(throwable), tries);
838               }
839               maybeRetry(now, throwable);
