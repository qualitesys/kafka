38   /**
39    * This class is responsible for refreshing logins for both Kafka client and
40    * server when the credential is an OAuth 2 bearer token communicated over
41    * SASL/OAUTHBEARER. An OAuth 2 bearer token has a limited lifetime, and an
42    * instance of this class periodically refreshes it so that the client can
43    * create new connections to brokers on an ongoing basis.
44    * <p>
45    * This class does not need to be explicitly set via the
46    * {@code sasl.login.class} client configuration property or the
47    * {@code listener.name.sasl_[plaintext|ssl].oauthbearer.sasl.login.class}
48    * broker configuration property when the SASL mechanism is OAUTHBEARER; it is
49    * automatically set by default in that case.
50    * <p>
51    * The parameters that impact how the refresh algorithm operates are specified
52    * as part of the producer/consumer/broker configuration and are as follows. See
53    * the documentation for these properties elsewhere for details.
54    * <table>
55    * <tr>
56    * <th>Producer/Consumer/Broker Configuration Property</th>
57    * </tr>
58    * <tr>
59    * <td>{@code sasl.login.refresh.window.factor}</td>
60    * </tr>
61    * <tr>
62    * <td>{@code sasl.login.refresh.window.jitter}</td>
63    * </tr>
64    * <tr>
65    * <td>{@code sasl.login.refresh.min.period.seconds}</td>
66    * </tr>
67    * <tr>
68    * <td>{@code sasl.login.refresh.min.buffer.seconds}</td>
69    * </tr>
70    * </table>
71    * 
72    * @see OAuthBearerLoginModule
73    * @see SaslConfigs#SASL_LOGIN_REFRESH_WINDOW_FACTOR_DOC
74    * @see SaslConfigs#SASL_LOGIN_REFRESH_WINDOW_JITTER_DOC
75    * @see SaslConfigs#SASL_LOGIN_REFRESH_MIN_PERIOD_SECONDS_DOC
76    * @see SaslConfigs#SASL_LOGIN_REFRESH_BUFFER_SECONDS_DOC
77    */
78   public class OAuthBearerRefreshingLogin implements Login {
79       private static final Logger log = LoggerFactory.getLogger(OAuthBearerRefreshingLogin.class);
80       private ExpiringCredentialRefreshingLogin expiringCredentialRefreshingLogin = null;
81   
82       @Override
83       public void configure(Map<String, ?> configs, String contextName, Configuration configuration,
84               AuthenticateCallbackHandler loginCallbackHandler) {
85           /*
86            * Specify this class as the one to synchronize on so that only one OAuth 2
87            * Bearer Token is refreshed at a given time. Specify null if we don't mind
88            * multiple simultaneously refreshes. Refreshes happen on the order of minutes
89            * rather than seconds or milliseconds, and there are typically minutes of
90            * lifetime remaining when the refresh occurs, so serializing them seems
91            * reasonable.
92            */
93           Class<OAuthBearerRefreshingLogin> classToSynchronizeOnPriorToRefresh = OAuthBearerRefreshingLogin.class;
94           expiringCredentialRefreshingLogin = new ExpiringCredentialRefreshingLogin(contextName, configuration,
95                   new ExpiringCredentialRefreshConfig(configs, true), loginCallbackHandler,
96                   classToSynchronizeOnPriorToRefresh) {
97               @Override
98               public ExpiringCredential expiringCredential() {
99                   Set<OAuthBearerToken> privateCredentialTokens = expiringCredentialRefreshingLogin.subject()
100                           .getPrivateCredentials(OAuthBearerToken.class);
101                   if (privateCredentialTokens.isEmpty())
102                       return null;
103                   final OAuthBearerToken token = privateCredentialTokens.iterator().next();
104                   if (log.isDebugEnabled())
105                       log.debug("Found expiring credential with principal '{}'.", token.principalName());
106                   return new ExpiringCredential() {
107                       @Override
108                       public String principalName() {
109                           return token.principalName();
110                       }
111   
112                       @Override
113                       public Long startTimeMs() {
114                           return token.startTimeMs();
115                       }
116   
117                       @Override
118                       public long expireTimeMs() {
119                           return token.lifetimeMs();
120                       }
121   
122                       @Override
123                       public Long absoluteLastRefreshTimeMs() {
124                           return null;
125                       }
126                   };
127               }
128           };
129       }
130   
131       @Override
132       public void close() {
133           if (expiringCredentialRefreshingLogin != null)
134               expiringCredentialRefreshingLogin.close();
135       }
136   
137       @Override
138       public Subject subject() {
139           return expiringCredentialRefreshingLogin != null ? expiringCredentialRefreshingLogin.subject() : null;
140       }
141   
142       @Override
143       public String serviceName() {
144           return expiringCredentialRefreshingLogin != null ? expiringCredentialRefreshingLogin.serviceName() : null;
145       }
146   
147       @Override
148       public synchronized LoginContext login() throws LoginException {
149           if (expiringCredentialRefreshingLogin != null)
150               return expiringCredentialRefreshingLogin.login();
151           throw new LoginException("Login was not configured properly");
152       }
153   }
