100       /**
101        * Append the record to the current record set and return the relative offset within that record set
102        *
103        * @return The RecordSend corresponding to this record or null if there isn't sufficient room.
104        */
105       public FutureRecordMetadata tryAppend(long timestamp, byte[] key, byte[] value, Header[] headers, Callback callback, long now) {
106           if (!recordsBuilder.hasRoomFor(timestamp, key, value, headers)) {
107               return null;
108           } else {
109               this.recordsBuilder.append(timestamp, key, value, headers);
110               this.maxRecordSize = Math.max(this.maxRecordSize, AbstractRecords.estimateSizeInBytesUpperBound(magic(),
111                       recordsBuilder.compressionType(), key, value, headers));
112               this.lastAppendTime = now;
113               FutureRecordMetadata future = new FutureRecordMetadata(this.produceFuture, this.recordCount,
114                                                                      timestamp,
115                                                                      key == null ? -1 : key.length,
116                                                                      value == null ? -1 : value.length,
117                                                                      Time.SYSTEM);
118               // we have to keep every future returned to the users in case the batch needs to be
119               // split to several new batches and resent.
120               thunks.add(new Thunk(callback, future));
121               this.recordCount++;
122               return future;
123           }
124       }
125   
126       /**
127        * This method is only used by {@link #split(int)} when splitting a large batch to smaller ones.
128        * @return true if the record has been successfully appended, false otherwise.
129        */
130       private boolean tryAppendForSplit(long timestamp, ByteBuffer key, ByteBuffer value, Header[] headers, Thunk thunk) {
131           if (!recordsBuilder.hasRoomFor(timestamp, key, value, headers)) {
132               return false;
133           } else {
134               // No need to get the CRC.
135               this.recordsBuilder.append(timestamp, key, value, headers);
136               this.maxRecordSize = Math.max(this.maxRecordSize, AbstractRecords.estimateSizeInBytesUpperBound(magic(),
137                       recordsBuilder.compressionType(), key, value, headers));
138               FutureRecordMetadata future = new FutureRecordMetadata(this.produceFuture, this.recordCount,
139                                                                      timestamp,
140                                                                      key == null ? -1 : key.remaining(),
141                                                                      value == null ? -1 : value.remaining(),
142                                                                      Time.SYSTEM);
143               // Chain the future to the original thunk.
144               thunk.future.chain(future);
145               this.thunks.add(thunk);
146               this.recordCount++;
147               return true;
148           }
149       }

156       public void abort(RuntimeException exception) {
157           if (!finalState.compareAndSet(null, FinalState.ABORTED))
158               throw new IllegalStateException("Batch has already been completed in final state " + finalState.get());

161           completeFutureAndFireCallbacks(ProduceResponse.INVALID_OFFSET, RecordBatch.NO_TIMESTAMP, index -> exception);
162       }

183       /**
184        * Complete the batch exceptionally. The provided top-level exception will be used
185        * for each record future contained in the batch.
186        *
187        * @param topLevelException top-level partition error
188        * @param recordExceptions Record exception function mapping batchIndex to the respective record exception
189        * @return true if the batch was completed as a result of this call, and false

197           Objects.requireNonNull(recordExceptions);

199       }

201       /**
202        * Finalize the state of a batch. Final state, once set, is immutable. This function may be called
203        * once or twice on a batch. It may be called twice if
204        * 1. An inflight batch expires before a response from the broker is received. The batch's final
205        * state is set to FAILED. But it could succeed on the broker and second time around batch.done() may
206        * try to set SUCCEEDED final state.
207        * 2. If a transaction abortion happens or if the producer is closed forcefully, the final state is
208        * ABORTED but again it could succeed if broker responds with a success.
209        *
210        * Attempted transitions from [FAILED | ABORTED] --> SUCCEEDED are logged.
211        * Attempted transitions from one failure state to the same or a different failed state are ignored.
212        * Attempted transitions from SUCCEEDED to the same or a failed state throw an exception.
213        *
214        * @param baseOffset The base offset of the messages assigned by the server
215        * @param logAppendTime The log append time or -1 if CreateTime is being used
216        * @param topLevelException The exception that occurred (or null if the request was successful)
217        * @param recordExceptions Record exception function mapping batchIndex to the respective record exception
218        * @return true if the batch was completed successfully and false if the batch was previously aborted
219        */
220       private boolean done(
221           long baseOffset,
222           long logAppendTime,
223           RuntimeException topLevelException,
224           Function<Integer, RuntimeException> recordExceptions
225       ) {
226           final FinalState tryFinalState = (topLevelException == null) ? FinalState.SUCCEEDED : FinalState.FAILED;
227           if (tryFinalState == FinalState.SUCCEEDED) {
228               log.trace("Successfully produced messages to {} with base offset {}.", topicPartition, baseOffset);
229           } else {
230               log.trace("Failed to produce messages to {} with base offset {}.", topicPartition, baseOffset, topLevelException);
231           }
232   
233           if (this.finalState.compareAndSet(null, tryFinalState)) {
234               completeFutureAndFireCallbacks(baseOffset, logAppendTime, recordExceptions);
235               return true;
236           }
237   
238           if (this.finalState.get() != FinalState.SUCCEEDED) {
239               if (tryFinalState == FinalState.SUCCEEDED) {
240                   // Log if a previously unsuccessful batch succeeded later on.
241                   log.debug("ProduceResponse returned {} for {} after batch with base offset {} had already been {}.",
242                       tryFinalState, topicPartition, baseOffset, this.finalState.get());
243               } else {
244                   // FAILED --> FAILED and ABORTED --> FAILED transitions are ignored.
245                   log.debug("Ignored state transition {} -> {} for {} batch with base offset {}",
246                       this.finalState.get(), tryFinalState, topicPartition, baseOffset);
247               }
248           } else {
249               // A SUCCESSFUL batch must not attempt another state change.
250               throw new IllegalStateException("A " + this.finalState.get() + " batch must not attempt another state change to " + tryFinalState);
251           }
252           return false;
253       }

255       private void completeFutureAndFireCallbacks(
256           long baseOffset,
257           long logAppendTime,
258           Function<Integer, RuntimeException> recordExceptions
259       ) {
260           // Set the future before invoking the callbacks as we rely on its state for the `onCompletion` call
261           produceFuture.set(baseOffset, logAppendTime, recordExceptions);
262   
263           // execute callbacks
264           for (int i = 0; i < thunks.size(); i++) {
265               try {
266                   Thunk thunk = thunks.get(i);
267                   if (thunk.callback != null) {
268                       if (recordExceptions == null) {
269                           RecordMetadata metadata = thunk.future.value();
270                           thunk.callback.onCompletion(metadata, null);
271                       } else {
272                           RuntimeException exception = recordExceptions.apply(i);
273                           thunk.callback.onCompletion(null, exception);
274                       }
275                   }
276               } catch (Exception e) {
277                   log.error("Error executing user-provided callback on message for topic-partition '{}'", topicPartition, e);
278               }
279           }
280   
281           produceFuture.done();
282       }

284       public Deque<ProducerBatch> split(int splitBatchSize) {
285           Deque<ProducerBatch> batches = new ArrayDeque<>();
286           MemoryRecords memoryRecords = recordsBuilder.build();
287   
288           Iterator<MutableRecordBatch> recordBatchIter = memoryRecords.batches().iterator();
289           if (!recordBatchIter.hasNext())
290               throw new IllegalStateException("Cannot split an empty producer batch.");
291   
292           RecordBatch recordBatch = recordBatchIter.next();
293           if (recordBatch.magic() < MAGIC_VALUE_V2 && !recordBatch.isCompressed())
294               throw new IllegalArgumentException("Batch splitting cannot be used with non-compressed messages " +
295                       "with version v0 and v1");
296   
297           if (recordBatchIter.hasNext())
298               throw new IllegalArgumentException("A producer batch should only have one record batch.");
299   
300           Iterator<Thunk> thunkIter = thunks.iterator();
301           // We always allocate batch size because we are already splitting a big batch.
302           // And we also Retain the create time of the original batch.
303           ProducerBatch batch = null;
304   
305           for (Record record : recordBatch) {
306               assert thunkIter.hasNext();
307               Thunk thunk = thunkIter.next();
308               if (batch == null)
309                   batch = createBatchOffAccumulatorForRecord(record, splitBatchSize);
310   
311               // A newly created batch can always host the first message.
312               if (!batch.tryAppendForSplit(record.timestamp(), record.key(), record.value(), record.headers(), thunk)) {
313                   batches.add(batch);
314                   batch.closeForRecordAppends();
315                   batch = createBatchOffAccumulatorForRecord(record, splitBatchSize);
316                   batch.tryAppendForSplit(record.timestamp(), record.key(), record.value(), record.headers(), thunk);
317               }
318           }
319   
320           // Close the last batch and add it to the batch list after split.
321           if (batch != null) {
322               batches.add(batch);
323               batch.closeForRecordAppends();
324           }
325   
326           produceFuture.set(ProduceResponse.INVALID_OFFSET, NO_TIMESTAMP, index -> new RecordBatchTooLargeException());
327           produceFuture.done();
328   
329           if (hasSequence()) {
330               int sequence = baseSequence();
331               ProducerIdAndEpoch producerIdAndEpoch = new ProducerIdAndEpoch(producerId(), producerEpoch());
332               for (ProducerBatch newBatch : batches) {
333                   newBatch.setProducerState(producerIdAndEpoch, sequence, isTransactional());
334                   sequence += newBatch.recordCount;
335               }
336           }
337           return batches;
338       }

340       private ProducerBatch createBatchOffAccumulatorForRecord(Record record, int batchSize) {
341           int initialSize = Math.max(AbstractRecords.estimateSizeInBytesUpperBound(magic(),
342                   recordsBuilder.compressionType(), record.key(), record.value(), record.headers()), batchSize);
343           ByteBuffer buffer = ByteBuffer.allocate(initialSize);
344   
345           // Note that we intentionally do not set producer state (producerId, epoch, sequence, and isTransactional)
346           // for the newly created batch. This will be set when the batch is dequeued for sending (which is consistent
347           // with how normal batches are handled).
348           MemoryRecordsBuilder builder = MemoryRecords.builder(buffer, magic(), recordsBuilder.compressionType(),
349                   TimestampType.CREATE_TIME, 0L);
350           return new ProducerBatch(topicPartition, builder, this.createdMs, true);
351       }

437       public void resetProducerState(ProducerIdAndEpoch producerIdAndEpoch, int baseSequence, boolean isTransactional) {
438           log.info("Resetting sequence number of batch with current sequence {} for partition {} to {}",

440           reopened = true;

442       }
