31       static public class WrapperSerde<T> implements Serde<T> {
32           final private Serializer<T> serializer;
33           final private Deserializer<T> deserializer;
34   
35           public WrapperSerde(Serializer<T> serializer, Deserializer<T> deserializer) {
36               this.serializer = serializer;
37               this.deserializer = deserializer;
38           }
39   
40           @Override
41           public void configure(Map<String, ?> configs, boolean isKey) {
42               serializer.configure(configs, isKey);
43               deserializer.configure(configs, isKey);
44           }
45   
46           @Override
47           public void close() {
48               serializer.close();
49               deserializer.close();
50           }
51   
52           @Override
53           public Serializer<T> serializer() {
54               return serializer;
55           }
56   
57           @Override
58           public Deserializer<T> deserializer() {
59               return deserializer;
60           }
61       }

150       @SuppressWarnings("unchecked")
151       static public <T> Serde<T> serdeFrom(Class<T> type) {
152           if (String.class.isAssignableFrom(type)) {
153               return (Serde<T>) String();
154           }
155   
156           if (Short.class.isAssignableFrom(type)) {
157               return (Serde<T>) Short();
158           }
159   
160           if (Integer.class.isAssignableFrom(type)) {
161               return (Serde<T>) Integer();
162           }
163   
164           if (Long.class.isAssignableFrom(type)) {
165               return (Serde<T>) Long();
166           }
167   
168           if (Float.class.isAssignableFrom(type)) {
169               return (Serde<T>) Float();
170           }
171   
172           if (Double.class.isAssignableFrom(type)) {
173               return (Serde<T>) Double();
174           }
175   
176           if (byte[].class.isAssignableFrom(type)) {
177               return (Serde<T>) ByteArray();
178           }
179   
180           if (ByteBuffer.class.isAssignableFrom(type)) {
181               return (Serde<T>) ByteBuffer();
182           }
183   
184           if (Bytes.class.isAssignableFrom(type)) {
185               return (Serde<T>) Bytes();
186           }
187   
188           if (UUID.class.isAssignableFrom(type)) {
189               return (Serde<T>) UUID();
190           }
191   
192           // TODO: we can also serializes objects of type T using generic Java serialization by default
193           throw new IllegalArgumentException("Unknown class for built-in serializer. Supported types are: " +
194               "String, Short, Integer, Long, Float, Double, ByteArray, ByteBuffer, Bytes, UUID");
195       }

197       /**
198        * Construct a serde object from separate serializer and deserializer
199        *
200        * @param serializer    must not be null.
201        * @param deserializer  must not be null.
202        */
203       static public <T> Serde<T> serdeFrom(final Serializer<T> serializer, final Deserializer<T> deserializer) {
204           if (serializer == null) {
205               throw new IllegalArgumentException("serializer must not be null");
206           }
207           if (deserializer == null) {
208               throw new IllegalArgumentException("deserializer must not be null");
209           }
210   
211           return new WrapperSerde<>(serializer, deserializer);
212       }
