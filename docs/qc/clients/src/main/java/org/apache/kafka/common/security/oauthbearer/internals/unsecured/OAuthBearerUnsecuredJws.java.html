40   /**
41    * A simple unsecured JWS implementation. The '{@code nbf}' claim is ignored if
42    * it is given because the related logic is not required for Kafka testing and
43    * development purposes.
44    *
45    * @see <a href="https://tools.ietf.org/html/rfc7515">RFC 7515</a>
46    */
47   public class OAuthBearerUnsecuredJws implements OAuthBearerToken {

59       /**
60        * Constructor with the given principal and scope claim names
61        *
62        * @param compactSerialization
63        *            the compact serialization to parse as an unsecured JWS
64        * @param principalClaimName
65        *            the required principal claim name
66        * @param scopeClaimName
67        *            the required scope claim name
68        * @throws OAuthBearerIllegalTokenException
69        *             if the compact serialization is not a valid unsecured JWS
70        *             (meaning it did not have 3 dot-separated Base64URL sections
71        *             without an empty digital signature; or the header or claims
72        *             either are not valid Base 64 URL encoded values or are not JSON
73        *             after decoding; or the mandatory '{@code alg}' header value is
74        *             not "{@code none}")
75        */
76       public OAuthBearerUnsecuredJws(String compactSerialization, String principalClaimName, String scopeClaimName)
77               throws OAuthBearerIllegalTokenException {
78           this.compactSerialization = Objects.requireNonNull(compactSerialization);
79           if (compactSerialization.contains(".."))
80               throw new OAuthBearerIllegalTokenException(
81                       OAuthBearerValidationResult.newFailure("Malformed compact serialization contains '..'"));
82           this.splits = extractCompactSerializationSplits();
83           this.header = toMap(splits().get(0));
84           String claimsSplit = splits.get(1);
85           this.claims = toMap(claimsSplit);
86           String alg = Objects.requireNonNull(header().get("alg"), "JWS header must have an Algorithm value").toString();
87           if (!"none".equals(alg))
88               throw new OAuthBearerIllegalTokenException(
89                       OAuthBearerValidationResult.newFailure("Unsecured JWS must have 'none' for an algorithm"));
90           String digitalSignatureSplit = splits.get(2);
91           if (!digitalSignatureSplit.isEmpty())
92               throw new OAuthBearerIllegalTokenException(
93                       OAuthBearerValidationResult.newFailure("Unsecured JWS must not contain a digital signature"));
94           this.principalClaimName = Objects.requireNonNull(principalClaimName).trim();
95           if (this.principalClaimName.isEmpty())
96               throw new IllegalArgumentException("Must specify a non-blank principal claim name");
97           this.scopeClaimName = Objects.requireNonNull(scopeClaimName).trim();
98           if (this.scopeClaimName.isEmpty())
99               throw new IllegalArgumentException("Must specify a non-blank scope claim name");
100           this.scope = calculateScope();
101           Number expirationTimeSeconds = expirationTime();
102           if (expirationTimeSeconds == null)
103               throw new OAuthBearerIllegalTokenException(
104                       OAuthBearerValidationResult.newFailure("No expiration time in JWT"));
105           lifetime = convertClaimTimeInSecondsToMs(expirationTimeSeconds);
106           String principalName = claim(this.principalClaimName, String.class);
107           if (Utils.isBlank(principalName))
108               throw new OAuthBearerIllegalTokenException(OAuthBearerValidationResult
109                       .newFailure("No principal name in JWT claim: " + this.principalClaimName));
110           this.principalName = principalName;
111           this.startTimeMs = calculateStartTimeMs();
112       }

184       /**
185        * Indicate if the claim exists and is the given type
186        *
187        * @param claimName
188        *            the mandatory JWT claim name
189        * @param type
190        *            the mandatory type, which should either be String.class,
191        *            Number.class, or List.class
192        * @return true if the claim exists and is the given type, otherwise false
193        */
194       public boolean isClaimType(String claimName, Class<?> type) {
195           Object value = rawClaim(claimName);
196           Objects.requireNonNull(type);
197           if (value == null)
198               return false;
199           if (type == String.class && value instanceof String)
200               return true;
201           if (type == Number.class && value instanceof Number)
202               return true;
203           return type == List.class && value instanceof List;
204       }

206       /**
207        * Extract a claim of the given type
208        *
209        * @param claimName
210        *            the mandatory JWT claim name
211        * @param type
212        *            the mandatory type, which must either be String.class,
213        *            Number.class, or List.class
214        * @return the claim if it exists, otherwise null
215        * @throws OAuthBearerIllegalTokenException
216        *             if the claim exists but is not the given type
217        */
218       public <T> T claim(String claimName, Class<T> type) throws OAuthBearerIllegalTokenException {
219           Object value = rawClaim(claimName);
220           try {
221               return Objects.requireNonNull(type).cast(value);
222           } catch (ClassCastException e) {
223               throw new OAuthBearerIllegalTokenException(
224                       OAuthBearerValidationResult.newFailure(String.format("The '%s' claim was not of type %s: %s",
225                               claimName, type.getSimpleName(), value.getClass().getSimpleName())));
226           }
227       }

283       /**
284        * Decode the given Base64URL-encoded value, parse the resulting JSON as a JSON
285        * object, and return the map of member names to their values (each value being
286        * represented as either a String, a Number, or a List of Strings).
287        *
288        * @param split
289        *            the value to decode and parse
290        * @return the map of JSON member names to their String, Number, or String List
291        *         value
292        * @throws OAuthBearerIllegalTokenException
293        *             if the given Base64URL-encoded value cannot be decoded or parsed
294        */
295       public static Map<String, Object> toMap(String split) throws OAuthBearerIllegalTokenException {
296           Map<String, Object> retval = new HashMap<>();
297           try {
298               byte[] decode = Base64.getDecoder().decode(split);
299               JsonNode jsonNode = new ObjectMapper().readTree(decode);
300               if (jsonNode == null)
301                   throw new OAuthBearerIllegalTokenException(OAuthBearerValidationResult.newFailure("malformed JSON"));
302               for (Iterator<Entry<String, JsonNode>> iterator = jsonNode.fields(); iterator.hasNext();) {
303                   Entry<String, JsonNode> entry = iterator.next();
304                   retval.put(entry.getKey(), convert(entry.getValue()));
305               }
306               return Collections.unmodifiableMap(retval);
307           } catch (IllegalArgumentException e) {
308               // potentially thrown by java.util.Base64.Decoder implementations
309               throw new OAuthBearerIllegalTokenException(
310                       OAuthBearerValidationResult.newFailure("malformed Base64 URL encoded value"));
311           } catch (IOException e) {
312               throw new OAuthBearerIllegalTokenException(OAuthBearerValidationResult.newFailure("malformed JSON"));
313           }
314       }

316       private List<String> extractCompactSerializationSplits() {
317           List<String> tmpSplits = new ArrayList<>(Arrays.asList(compactSerialization.split("\\.")));
318           if (compactSerialization.endsWith("."))
319               tmpSplits.add("");
320           if (tmpSplits.size() != 3)
321               throw new OAuthBearerIllegalTokenException(OAuthBearerValidationResult.newFailure(
322                       "Unsecured JWS compact serializations must have 3 dot-separated Base64URL-encoded values"));
323           return Collections.unmodifiableList(tmpSplits);
324       }

326       private static Object convert(JsonNode value) {
327           if (value.isArray()) {
328               List<String> retvalList = new ArrayList<>();
329               for (JsonNode arrayElement : value)
330                   retvalList.add(arrayElement.asText());
331               return retvalList;
332           }
333           return value.getNodeType() == JsonNodeType.NUMBER ? value.numberValue() : value.asText();
334       }

336       private Long calculateStartTimeMs() throws OAuthBearerIllegalTokenException {
337           Number issuedAtSeconds = claim("iat", Number.class);
338           return issuedAtSeconds == null ? null : convertClaimTimeInSecondsToMs(issuedAtSeconds);
339       }

341       private static long convertClaimTimeInSecondsToMs(Number claimValue) {
342           return Math.round(claimValue.doubleValue() * 1000);
343       }

345       private Set<String> calculateScope() {
346           String scopeClaimName = scopeClaimName();
347           if (isClaimType(scopeClaimName, String.class)) {
348               String scopeClaimValue = claim(scopeClaimName, String.class);
349               if (Utils.isBlank(scopeClaimValue))
350                   return Collections.emptySet();
351               else {
352                   Set<String> retval = new HashSet<>();
353                   retval.add(scopeClaimValue.trim());
354                   return Collections.unmodifiableSet(retval);
355               }
356           }
357           List<?> scopeClaimValue = claim(scopeClaimName, List.class);
358           if (scopeClaimValue == null || scopeClaimValue.isEmpty())
359               return Collections.emptySet();
360           @SuppressWarnings("unchecked")
361           List<String> stringList = (List<String>) scopeClaimValue;
362           Set<String> retval = new HashSet<>();
363           for (String scope : stringList) {
364               if (!Utils.isBlank(scope)) {
365                   retval.add(scope.trim());
366               }
367           }
368           return Collections.unmodifiableSet(retval);
369       }
