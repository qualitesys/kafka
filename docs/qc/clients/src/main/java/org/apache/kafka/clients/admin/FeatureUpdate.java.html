17   package org.apache.kafka.clients.admin;
18   
19   import java.util.Objects;
20   
21   /**
22    * Encapsulates details about an update to a finalized feature.
23    */
24   public class FeatureUpdate {
25       private final short maxVersionLevel;
26       private final UpgradeType upgradeType;
27   
28       public enum UpgradeType {
29           UNKNOWN(0),
30           UPGRADE(1),
31           SAFE_DOWNGRADE(2),
32           UNSAFE_DOWNGRADE(3);
33   
34           private final byte code;
35   
36           UpgradeType(int code) {
37               this.code = (byte) code;
38           }
39   
40           public byte code() {
41               return code;
42           }
43   
44           public static UpgradeType fromCode(int code) {
45               if (code == 1) {
46                   return UPGRADE;
47               } else if (code == 2) {
48                   return SAFE_DOWNGRADE;
49               } else if (code == 3) {
50                   return UNSAFE_DOWNGRADE;
51               } else {
52                   return UNKNOWN;
53               }
54           }
55       }
56   
57       /**
58        * @param maxVersionLevel   the new maximum version level for the finalized feature.
59        *                          a value of zero is special and indicates that the update is intended to
60        *                          delete the finalized feature, and should be accompanied by setting
61        *                          the allowDowngrade flag to true.
62        * @param allowDowngrade    - true, if this feature update was meant to downgrade the existing
63        *                            maximum version level of the finalized feature. Only "safe" downgrades are
64        *                            enabled with this boolean. See {@link FeatureUpdate#FeatureUpdate(short, UpgradeType)}
65        *                          - false, otherwise.
66        */
67       @Deprecated
68       public FeatureUpdate(final short maxVersionLevel, final boolean allowDowngrade) {
69           this(maxVersionLevel, allowDowngrade ? UpgradeType.SAFE_DOWNGRADE : UpgradeType.UPGRADE);
70       }
71   
72       /**
73        * @param maxVersionLevel   The new maximum version level for the finalized feature.
74        *                          a value of zero is special and indicates that the update is intended to
75        *                          delete the finalized feature, and should be accompanied by setting
76        *                          the upgradeType to safe or unsafe.
77        * @param upgradeType     Indicate what kind of upgrade should be performed in this operation.
78        *                          - UPGRADE: upgrading the feature level
79        *                          - SAFE_DOWNGRADE: only downgrades which do not result in metadata loss are permitted
80        *                          - UNSAFE_DOWNGRADE: any downgrade, including those which may result in metadata loss, are permitted
81        */
82       public FeatureUpdate(final short maxVersionLevel, final UpgradeType upgradeType) {
83           if (maxVersionLevel == 0 && upgradeType.equals(UpgradeType.UPGRADE)) {
84               throw new IllegalArgumentException(String.format(
85                       "The downgradeType flag should be set to SAFE or UNSAFE when the provided maxVersionLevel:%d is < 1.",
86                       maxVersionLevel));
87           }
88           if (maxVersionLevel < 0) {
89               throw new IllegalArgumentException("Cannot specify a negative version level.");
90           }
91           this.maxVersionLevel = maxVersionLevel;
92           this.upgradeType = upgradeType;
93       }
94   
95       public short maxVersionLevel() {
96           return maxVersionLevel;
97       }
98   
99       @Deprecated
100       public boolean allowDowngrade() {
101           return upgradeType != UpgradeType.UPGRADE;
102       }
103   
104       public UpgradeType upgradeType() {
105           return upgradeType;
106       }
107   
108       @Override
109       public boolean equals(Object other) {
110           if (this == other) {
111               return true;
112           }
113   
114           if (!(other instanceof FeatureUpdate)) {
115               return false;
116           }
117   
118           final FeatureUpdate that = (FeatureUpdate) other;
119           return this.maxVersionLevel == that.maxVersionLevel && this.upgradeType.equals(that.upgradeType);
120       }
121   
122       @Override
123       public int hashCode() {
124           return Objects.hash(maxVersionLevel, upgradeType);
125       }
126   
127       @Override
128       public String toString() {
129           return String.format("FeatureUpdate{maxVersionLevel:%d, downgradeType:%s}", maxVersionLevel, upgradeType);
130       }
131   }
