33   /**

37    */

39   
40       /**
41        * See {@link KafkaProducer#initTransactions()}
42        */
43       void initTransactions();
44   
45       /**
46        * See {@link KafkaProducer#beginTransaction()}
47        */
48       void beginTransaction() throws ProducerFencedException;
49   
50       /**
51        * See {@link KafkaProducer#sendOffsetsToTransaction(Map, String)}
52        */

54       void sendOffsetsToTransaction(Map<TopicPartition, OffsetAndMetadata> offsets,
55                                     String consumerGroupId) throws ProducerFencedException;
56   
57       /**
58        * See {@link KafkaProducer#sendOffsetsToTransaction(Map, ConsumerGroupMetadata)}
59        */
60       void sendOffsetsToTransaction(Map<TopicPartition, OffsetAndMetadata> offsets,
61                                     ConsumerGroupMetadata groupMetadata) throws ProducerFencedException;
62   
63       /**
64        * See {@link KafkaProducer#commitTransaction()}
65        */
66       void commitTransaction() throws ProducerFencedException;
67   
68       /**
69        * See {@link KafkaProducer#abortTransaction()}
70        */
71       void abortTransaction() throws ProducerFencedException;
72   
73       /**
74        * See {@link KafkaProducer#send(ProducerRecord)}
75        */
76       Future<RecordMetadata> send(ProducerRecord<K, V> record);
77   
78       /**
79        * See {@link KafkaProducer#send(ProducerRecord, Callback)}
80        */
81       Future<RecordMetadata> send(ProducerRecord<K, V> record, Callback callback);
82   
83       /**
84        * See {@link KafkaProducer#flush()}
85        */
86       void flush();
87   
88       /**
89        * See {@link KafkaProducer#partitionsFor(String)}
90        */
91       List<PartitionInfo> partitionsFor(String topic);
92   
93       /**
94        * See {@link KafkaProducer#metrics()}
95        */
96       Map<MetricName, ? extends Metric> metrics();
97   
98       /**
99        * See {@link KafkaProducer#close()}
100        */
101       void close();
102   
103       /**
104        * See {@link KafkaProducer#close(Duration)}
105        */
106       void close(Duration timeout);
107   }
