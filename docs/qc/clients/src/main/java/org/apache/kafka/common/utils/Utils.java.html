336       /**
337        * Compares two character arrays for equality using a constant-time algorithm, which is needed
338        * for comparing passwords. Two arrays are equal if they have the same length and all
339        * characters at corresponding positions are equal.
340        *
341        * All characters in the first array are examined to determine equality.
342        * The calculation time depends only on the length of this first character array; it does not
343        * depend on the length of the second character array or the contents of either array.
344        *
345        * @param first the first array to compare
346        * @param second the second array to compare
347        * @return true if the arrays are equal, or false otherwise
348        */
349       public static boolean isEqualConstantTime(char[] first, char[] second) {
350           if (first == second) {
351               return true;
352           }
353           if (first == null || second == null) {
354               return false;
355           }
356   
357           if (second.length == 0) {
358               return first.length == 0;
359           }
360   
361           // time-constant comparison that always compares all characters in first array
362           boolean matches = first.length == second.length;
363           for (int i = 0; i < first.length; ++i) {
364               int j = i < second.length ? i : 0;
365               if (first[i] != second[j]) {
366                   matches = false;
367               }
368           }
369           return matches;
370       }

434       /**
435        * Construct a new object using a class name and parameters.
436        *
437        * @param className                 The full name of the class to construct.
438        * @param params                    A sequence of (type, object) elements.
439        * @param <T>                       The type of object to construct.
440        * @return                          The new object.
441        * @throws ClassNotFoundException   If there was a problem constructing the object.
442        */
443       public static <T> T newParameterizedInstance(String className, Object... params)
444               throws ClassNotFoundException {
445           Class<?>[] argTypes = new Class<?>[params.length / 2];
446           Object[] args = new Object[params.length / 2];
447           try {
448               Class<?> c = Class.forName(className, true, Utils.getContextOrKafkaClassLoader());
449               for (int i = 0; i < params.length / 2; i++) {
450                   argTypes[i] = (Class<?>) params[2 * i];
451                   args[i] = params[(2 * i) + 1];
452               }
453               @SuppressWarnings("unchecked")
454               Constructor<T> constructor = (Constructor<T>) c.getConstructor(argTypes);
455               return constructor.newInstance(args);
456           } catch (NoSuchMethodException e) {
457               throw new ClassNotFoundException(String.format("Failed to find " +
458                   "constructor with %s for %s", Utils.join(argTypes, ", "), className), e);
459           } catch (InstantiationException e) {
460               throw new ClassNotFoundException(String.format("Failed to instantiate " +
461                   "%s", className), e);
462           } catch (IllegalAccessException e) {

471       /**
472        * Generates 32 bit murmur2 hash from byte array
473        * @param data byte array to hash
474        * @return 32 bit hash of the given array
475        */
476       @SuppressWarnings("fallthrough")
477       public static int murmur2(final byte[] data) {
478           int length = data.length;
479           int seed = 0x9747b28c;
480           // 'm' and 'r' are mixing constants generated offline.
481           // They're not really 'magic', they just happen to work well.
482           final int m = 0x5bd1e995;
483           final int r = 24;
484   
485           // Initialize the hash to a random value
486           int h = seed ^ length;
487           int length4 = length / 4;
488   
489           for (int i = 0; i < length4; i++) {
490               final int i4 = i * 4;
491               int k = (data[i4 + 0] & 0xff) + ((data[i4 + 1] & 0xff) << 8) + ((data[i4 + 2] & 0xff) << 16) + ((data[i4 + 3] & 0xff) << 24);
492               k *= m;
493               k ^= k >>> r;
494               k *= m;
495               h *= m;
496               h ^= k;
497           }
498   
499           // Handle the last few bytes of the input array
500           switch (length % 4) {
501               case 3:
502                   h ^= (data[(length & ~3) + 2] & 0xff) << 16;
503               case 2:
504                   h ^= (data[(length & ~3) + 1] & 0xff) << 8;
505               case 1:
506                   h ^= data[length & ~3] & 0xff;
507                   h *= m;
508           }
509   
510           h ^= h >>> 13;
511           h *= m;
512           h ^= h >>> 15;
513   
514           return h;
515       }
