17   package org.apache.kafka.common.network;
18   
19   import java.util.Objects;
20   
21   /**
22    * Defines the context in which an {@link Authenticator} is to be created during
23    * a re-authentication.
24    */
25   public class ReauthenticationContext {
26       private final NetworkReceive networkReceive;
27       private final Authenticator previousAuthenticator;
28       private final long reauthenticationBeginNanos;
29   
30       /**
31        * Constructor
32        * 
33        * @param previousAuthenticator
34        *            the mandatory {@link Authenticator} that was previously used to
35        *            authenticate the channel
36        * @param networkReceive
37        *            the applicable {@link NetworkReceive} instance, if any. For the
38        *            client side this may be a response that has been partially read, a
39        *            non-null instance that has had no data read into it yet, or null;
40        *            if it is non-null then this is the instance that data should
41        *            initially be read into during re-authentication. For the server
42        *            side this is mandatory and it must contain the
43        *            {@code SaslHandshakeRequest} that has been received on the server
44        *            and that initiates re-authentication.
45        * 
46        * @param nowNanos
47        *            the current time. The value is in nanoseconds as per
48        *            {@code System.nanoTime()} and is therefore only useful when
49        *            compared to such a value -- it's absolute value is meaningless.
50        *            This defines the moment when re-authentication begins.
51        */
52       public ReauthenticationContext(Authenticator previousAuthenticator, NetworkReceive networkReceive, long nowNanos) {
53           this.previousAuthenticator = Objects.requireNonNull(previousAuthenticator);
54           this.networkReceive = networkReceive;
55           this.reauthenticationBeginNanos = nowNanos;
56       }
57   
58       /**
59        * Return the applicable {@link NetworkReceive} instance, if any. For the client
60        * side this may be a response that has been partially read, a non-null instance
61        * that has had no data read into it yet, or null; if it is non-null then this
62        * is the instance that data should initially be read into during
63        * re-authentication. For the server side this is mandatory and it must contain
64        * the {@code SaslHandshakeRequest} that has been received on the server and
65        * that initiates re-authentication.
66        * 
67        * @return the applicable {@link NetworkReceive} instance, if any
68        */
69       public NetworkReceive networkReceive() {
70           return networkReceive;
71       }
72   
73       /**
74        * Return the always non-null {@link Authenticator} that was previously used to
75        * authenticate the channel
76        * 
77        * @return the always non-null {@link Authenticator} that was previously used to
78        *         authenticate the channel
79        */
80       public Authenticator previousAuthenticator() {
81           return previousAuthenticator;
82       }
83   
84       /**
85        * Return the time when re-authentication began. The value is in nanoseconds as
86        * per {@code System.nanoTime()} and is therefore only useful when compared to
87        * such a value -- it's absolute value is meaningless.
88        * 
89        * @return the time when re-authentication began
90        */
91       public long reauthenticationBeginNanos() {
92           return reauthenticationBeginNanos;
93       }
94   }
