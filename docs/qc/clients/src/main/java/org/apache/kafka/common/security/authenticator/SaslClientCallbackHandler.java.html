17   package org.apache.kafka.common.security.authenticator;
18   
19   import java.security.AccessController;
20   import java.util.List;
21   import java.util.Map;
22   
23   import javax.security.auth.Subject;
24   import javax.security.auth.callback.Callback;
25   import javax.security.auth.callback.NameCallback;
26   import javax.security.auth.callback.PasswordCallback;
27   import javax.security.auth.callback.UnsupportedCallbackException;
28   import javax.security.auth.login.AppConfigurationEntry;
29   import javax.security.sasl.AuthorizeCallback;
30   import javax.security.sasl.RealmCallback;
31   
32   import org.apache.kafka.common.config.SaslConfigs;
33   import org.apache.kafka.common.security.auth.SaslExtensionsCallback;
34   import org.apache.kafka.common.security.auth.AuthenticateCallbackHandler;
35   import org.apache.kafka.common.security.auth.SaslExtensions;
36   import org.apache.kafka.common.security.scram.ScramExtensionsCallback;
37   import org.apache.kafka.common.security.scram.internals.ScramMechanism;
38   
39   /**
40    * Default callback handler for Sasl clients. The callbacks required for the SASL mechanism
41    * configured for the client should be supported by this callback handler. See
42    * <a href="https://docs.oracle.com/javase/8/docs/technotes/guides/security/sasl/sasl-refguide.html">Java SASL API</a>
43    * for the list of SASL callback handlers required for each SASL mechanism.
44    *
45    * For adding custom SASL extensions, a {@link SaslExtensions} may be added to the subject's public credentials
46    */
47   public class SaslClientCallbackHandler implements AuthenticateCallbackHandler {
48   
49       private String mechanism;
50   
51       @Override
52       public void configure(Map<String, ?> configs, String saslMechanism, List<AppConfigurationEntry> jaasConfigEntries) {
53           this.mechanism  = saslMechanism;
54       }
55   
56       @Override
57       public void handle(Callback[] callbacks) throws UnsupportedCallbackException {
58           Subject subject = Subject.getSubject(AccessController.getContext());
59           for (Callback callback : callbacks) {
60               if (callback instanceof NameCallback) {
61                   NameCallback nc = (NameCallback) callback;
62                   if (subject != null && !subject.getPublicCredentials(String.class).isEmpty()) {
63                       nc.setName(subject.getPublicCredentials(String.class).iterator().next());
64                   } else
65                       nc.setName(nc.getDefaultName());
66               } else if (callback instanceof PasswordCallback) {
67                   if (subject != null && !subject.getPrivateCredentials(String.class).isEmpty()) {
68                       char[] password = subject.getPrivateCredentials(String.class).iterator().next().toCharArray();
69                       ((PasswordCallback) callback).setPassword(password);
70                   } else {
71                       String errorMessage = "Could not login: the client is being asked for a password, but the Kafka" +
72                                " client code does not currently support obtaining a password from the user.";
73                       throw new UnsupportedCallbackException(callback, errorMessage);
74                   }
75               } else if (callback instanceof RealmCallback) {
76                   RealmCallback rc = (RealmCallback) callback;
77                   rc.setText(rc.getDefaultText());
78               } else if (callback instanceof AuthorizeCallback) {
79                   AuthorizeCallback ac = (AuthorizeCallback) callback;
80                   String authId = ac.getAuthenticationID();
81                   String authzId = ac.getAuthorizationID();
82                   ac.setAuthorized(authId.equals(authzId));
83                   if (ac.isAuthorized())
84                       ac.setAuthorizedID(authzId);
85               } else if (callback instanceof ScramExtensionsCallback) {
86                   if (ScramMechanism.isScram(mechanism) && subject != null && !subject.getPublicCredentials(Map.class).isEmpty()) {
87                       @SuppressWarnings("unchecked")
88                       Map<String, String> extensions = (Map<String, String>) subject.getPublicCredentials(Map.class).iterator().next();
89                       ((ScramExtensionsCallback) callback).extensions(extensions);
90                   }
91               } else if (callback instanceof SaslExtensionsCallback) {
92                   if (!SaslConfigs.GSSAPI_MECHANISM.equals(mechanism) &&
93                           subject != null && !subject.getPublicCredentials(SaslExtensions.class).isEmpty()) {
94                       SaslExtensions extensions = subject.getPublicCredentials(SaslExtensions.class).iterator().next();
95                       ((SaslExtensionsCallback) callback).extensions(extensions);
96                   }
97               }  else {
98                   throw new UnsupportedCallbackException(callback, "Unrecognized SASL ClientCallback");
99               }
100           }
101       }
102   
103       @Override
104       public void close() {
105       }
106   }
