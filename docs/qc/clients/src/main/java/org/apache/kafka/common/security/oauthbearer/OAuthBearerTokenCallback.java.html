17   package org.apache.kafka.common.security.oauthbearer;
18   
19   import java.util.Objects;
20   
21   import javax.security.auth.callback.Callback;
22   
23   import org.apache.kafka.common.annotation.InterfaceStability;
24   
25   /**
26    * A {@code Callback} for use by the {@code SaslClient} and {@code Login}
27    * implementations when they require an OAuth 2 bearer token. Callback handlers
28    * should use the {@link #error(String, String, String)} method to communicate
29    * errors returned by the authorization server as per
30    * <a href="https://tools.ietf.org/html/rfc6749#section-5.2">RFC 6749: The OAuth
31    * 2.0 Authorization Framework</a>. Callback handlers should communicate other
32    * problems by raising an {@code IOException}.
33    * <p>
34    * This class was introduced in 2.0.0 and, while it feels stable, it could
35    * evolve. We will try to evolve the API in a compatible manner, but we reserve
36    * the right to make breaking changes in minor releases, if necessary. We will
37    * update the {@code InterfaceStability} annotation and this notice once the API
38    * is considered stable.
39    */
40   @InterfaceStability.Evolving
41   public class OAuthBearerTokenCallback implements Callback {
42       private OAuthBearerToken token = null;
43       private String errorCode = null;
44       private String errorDescription = null;
45       private String errorUri = null;
46   
47       /**
48        * Return the (potentially null) token
49        * 
50        * @return the (potentially null) token
51        */
52       public OAuthBearerToken token() {
53           return token;
54       }
55   
56       /**
57        * Return the optional (but always non-empty if not null) error code as per
58        * <a href="https://tools.ietf.org/html/rfc6749#section-5.2">RFC 6749: The OAuth
59        * 2.0 Authorization Framework</a>.
60        * 
61        * @return the optional (but always non-empty if not null) error code
62        */
63       public String errorCode() {
64           return errorCode;
65       }
66   
67       /**
68        * Return the (potentially null) error description as per
69        * <a href="https://tools.ietf.org/html/rfc6749#section-5.2">RFC 6749: The OAuth
70        * 2.0 Authorization Framework</a>.
71        * 
72        * @return the (potentially null) error description
73        */
74       public String errorDescription() {
75           return errorDescription;
76       }
77   
78       /**
79        * Return the (potentially null) error URI as per
80        * <a href="https://tools.ietf.org/html/rfc6749#section-5.2">RFC 6749: The OAuth
81        * 2.0 Authorization Framework</a>.
82        * 
83        * @return the (potentially null) error URI
84        */
85       public String errorUri() {
86           return errorUri;
87       }
88   
89       /**
90        * Set the token. All error-related values are cleared.
91        * 
92        * @param token
93        *            the optional token to set
94        */
95       public void token(OAuthBearerToken token) {
96           this.token = token;
97           this.errorCode = null;
98           this.errorDescription = null;
99           this.errorUri = null;
100       }
101   
102       /**
103        * Set the error values as per
104        * <a href="https://tools.ietf.org/html/rfc6749#section-5.2">RFC 6749: The OAuth
105        * 2.0 Authorization Framework</a>. Any token is cleared.
106        * 
107        * @param errorCode
108        *            the mandatory error code to set
109        * @param errorDescription
110        *            the optional error description to set
111        * @param errorUri
112        *            the optional error URI to set
113        */
114       public void error(String errorCode, String errorDescription, String errorUri) {
115           if (Objects.requireNonNull(errorCode).isEmpty())
116               throw new IllegalArgumentException("error code must not be empty");
117           this.errorCode = errorCode;
118           this.errorDescription = errorDescription;
119           this.errorUri = errorUri;
120           this.token = null;
121       }
122   }
