17   package org.apache.kafka.common.metrics.stats;
18   
19   import java.util.Locale;
20   import java.util.concurrent.TimeUnit;
21   
22   import org.apache.kafka.common.metrics.MeasurableStat;
23   import org.apache.kafka.common.metrics.MetricConfig;
24   
25   import static org.apache.kafka.common.metrics.internals.MetricsUtils.convert;
26   
27   /**
28    * The rate of the given quantity. By default this is the total observed over a set of samples from a sampled statistic
29    * divided by the elapsed time over the sample windows. Alternative {@link SampledStat} implementations can be provided,
30    * however, to record the rate of occurrences (e.g. the count of values measured over the time interval) or other such
31    * values.
32    */
33   public class Rate implements MeasurableStat {
34   
35       protected final TimeUnit unit;
36       protected final SampledStat stat;
37   
38       public Rate() {
39           this(TimeUnit.SECONDS);
40       }
41   
42       public Rate(TimeUnit unit) {
43           this(unit, new WindowedSum());
44       }
45   
46       public Rate(SampledStat stat) {
47           this(TimeUnit.SECONDS, stat);
48       }
49   
50       public Rate(TimeUnit unit, SampledStat stat) {
51           this.stat = stat;
52           this.unit = unit;
53       }
54   
55       public String unitName() {
56           return unit.name().substring(0, unit.name().length() - 2).toLowerCase(Locale.ROOT);
57       }
58   
59       @Override
60       public void record(MetricConfig config, double value, long timeMs) {
61           this.stat.record(config, value, timeMs);
62       }
63   
64       @Override
65       public double measure(MetricConfig config, long now) {
66           double value = stat.measure(config, now);
67           return value / convert(windowSize(config, now), unit);
68       }
69   
70       public long windowSize(MetricConfig config, long now) {
71           // purge old samples before we compute the window size
72           stat.purgeObsoleteSamples(config, now);
73   
74           /*
75            * Here we check the total amount of time elapsed since the oldest non-obsolete window.
76            * This give the total windowSize of the batch which is the time used for Rate computation.
77            * However, there is an issue if we do not have sufficient data for e.g. if only 1 second has elapsed in a 30 second
78            * window, the measured rate will be very high.
79            * Hence we assume that the elapsed time is always N-1 complete windows plus whatever fraction of the final window is complete.
80            *
81            * Note that we could simply count the amount of time elapsed in the current window and add n-1 windows to get the total time,
82            * but this approach does not account for sleeps. SampledStat only creates samples whenever record is called,
83            * if no record is called for a period of time that time is not accounted for in windowSize and produces incorrect results.
84            */
85           long totalElapsedTimeMs = now - stat.oldest(now).lastWindowMs;
86           // Check how many full windows of data we have currently retained
87           int numFullWindows = (int) (totalElapsedTimeMs / config.timeWindowMs());
88           int minFullWindows = config.samples() - 1;
89   
90           // If the available windows are less than the minimum required, add the difference to the totalElapsedTime
91           if (numFullWindows < minFullWindows)
92               totalElapsedTimeMs += (minFullWindows - numFullWindows) * config.timeWindowMs();
93   
94           return totalElapsedTimeMs;
95       }
96   
97       @Override
98       public String toString() {
99           return "Rate(" +
100               "unit=" + unit +
101               ", stat=" + stat +
102               ')';
103       }
104   }
