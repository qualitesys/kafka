42        * The current offset and size for all the fixed-length fields

47       public static final int MAGIC_LENGTH = 1;
48       public static final int ATTRIBUTES_OFFSET = MAGIC_OFFSET + MAGIC_LENGTH;

51       public static final int TIMESTAMP_LENGTH = 8;
52       public static final int KEY_SIZE_OFFSET_V0 = ATTRIBUTES_OFFSET + ATTRIBUTES_LENGTH;
53       public static final int KEY_SIZE_OFFSET_V1 = TIMESTAMP_OFFSET + TIMESTAMP_LENGTH;
54       public static final int KEY_SIZE_LENGTH = 4;
55       public static final int KEY_OFFSET_V0 = KEY_SIZE_OFFSET_V0 + KEY_SIZE_LENGTH;

217       /**
218        * When magic value is greater than 0, the timestamp of a record is determined in the following way:
219        * 1. wrapperRecordTimestampType = null and wrapperRecordTimestamp is null - Uncompressed message, timestamp is in the message.
220        * 2. wrapperRecordTimestampType = LOG_APPEND_TIME and WrapperRecordTimestamp is not null - Compressed message using LOG_APPEND_TIME
221        * 3. wrapperRecordTimestampType = CREATE_TIME and wrapperRecordTimestamp is not null - Compressed message using CREATE_TIME
222        *
223        * @return the timestamp as determined above
224        */

226           if (magic() == RecordBatch.MAGIC_VALUE_V0)
227               return RecordBatch.NO_TIMESTAMP;
228           else {
229               // case 2
230               if (wrapperRecordTimestampType == TimestampType.LOG_APPEND_TIME && wrapperRecordTimestamp != null)
231                   return wrapperRecordTimestamp;
232               // Case 1, 3
233               else
234                   return buffer.getLong(TIMESTAMP_OFFSET);
235           }
236       }

322        *
323        * @param magic The magic value to use
324        * @param timestamp The timestamp of the record
325        * @param key The key of the record (null, if none)
326        * @param value The record value
327        * @param compressionType The compression type used on the contents of the record (if any)
328        * @param timestampType The timestamp type to be used for this record
329        */
330       public static LegacyRecord create(byte magic,
331                                         long timestamp,
332                                         byte[] key,
333                                         byte[] value,
334                                         CompressionType compressionType,
335                                         TimestampType timestampType) {
336           int keySize = key == null ? 0 : key.length;
337           int valueSize = value == null ? 0 : value.length;
338           ByteBuffer buffer = ByteBuffer.allocate(recordSize(magic, keySize, valueSize));
339           write(buffer, magic, timestamp, wrapNullable(key), wrapNullable(value), compressionType, timestampType);
340           buffer.rewind();
341           return new LegacyRecord(buffer);
342       }

348       /**
349        * Write the header for a compressed record set in-place (i.e. assuming the compressed record data has already
350        * been written at the value offset in a wrapped record). This lets you dynamically create a compressed message
351        * set, and then go back later and fill in its size and CRC, which saves the need for copying to another buffer.
352        *
353        * @param buffer The buffer containing the compressed record data positioned at the first offset of the
354        * @param magic The magic value of the record set
355        * @param recordSize The size of the record (including record overhead)
356        * @param timestamp The timestamp of the wrapper record
357        * @param compressionType The compression type used
358        * @param timestampType The timestamp type of the wrapper record
359        */
360       public static void writeCompressedRecordHeader(ByteBuffer buffer,
361                                                      byte magic,
362                                                      int recordSize,
363                                                      long timestamp,
364                                                      CompressionType compressionType,
365                                                      TimestampType timestampType) {
366           int recordPosition = buffer.position();
367           int valueSize = recordSize - recordOverhead(magic);
368   
369           // write the record header with a null value (the key is always null for the wrapper)
370           write(buffer, magic, timestamp, null, null, compressionType, timestampType);
371           buffer.position(recordPosition);
372   
373           // now fill in the value size
374           buffer.putInt(recordPosition + keyOffset(magic), valueSize);
375   
376           // compute and fill the crc from the beginning of the message
377           long crc = crc32(buffer, MAGIC_OFFSET, recordSize - MAGIC_OFFSET);
378           ByteUtils.writeUnsignedInt(buffer, recordPosition + CRC_OFFSET, crc);
379       }

445       // Write a record to the buffer, if the record's compression type is none, then
446       // its value payload should be already compressed with the specified type
447       private static void write(DataOutputStream out,
448                                 byte magic,
449                                 long crc,
450                                 byte attributes,
451                                 long timestamp,
452                                 ByteBuffer key,
453                                 ByteBuffer value) throws IOException {
454           if (magic != RecordBatch.MAGIC_VALUE_V0 && magic != RecordBatch.MAGIC_VALUE_V1)
455               throw new IllegalArgumentException("Invalid magic value " + magic);
456           if (timestamp < 0 && timestamp != RecordBatch.NO_TIMESTAMP)
457               throw new IllegalArgumentException("Invalid message timestamp " + timestamp);
458   
459           // write crc
460           out.writeInt((int) (crc & 0xffffffffL));
461           // write magic value
462           out.writeByte(magic);
463           // write attributes
464           out.writeByte(attributes);
465   
466           // maybe write timestamp
467           if (magic > RecordBatch.MAGIC_VALUE_V0)
468               out.writeLong(timestamp);
469   
470           // write the key
471           if (key == null) {
472               out.writeInt(-1);
473           } else {
474               int size = key.remaining();
475               out.writeInt(size);

522       /**
523        * Compute the checksum of the record from the attributes, key and value payloads
524        */
525       private static long computeChecksum(byte magic, byte attributes, long timestamp, ByteBuffer key, ByteBuffer value) {
526           CRC32 crc = new CRC32();
527           crc.update(magic);
528           crc.update(attributes);
529           if (magic > RecordBatch.MAGIC_VALUE_V0)
530               Checksums.updateLong(crc, timestamp);
531           // update for the key
532           if (key == null) {
533               Checksums.updateInt(crc, -1);
534           } else {
535               int size = key.remaining();
536               Checksums.updateInt(crc, size);
537               Checksums.update(crc, key, size);
538           }
539           // update for the value
540           if (value == null) {
541               Checksums.updateInt(crc, -1);
542           } else {
543               int size = value.remaining();
544               Checksums.updateInt(crc, size);
545               Checksums.update(crc, value, size);
546           }
547           return crc.getValue();
548       }
