17   package org.apache.kafka.clients.admin;
18   
19   import org.apache.kafka.clients.admin.internals.CoordinatorKey;
20   import org.apache.kafka.common.KafkaFuture;
21   import org.apache.kafka.common.annotation.InterfaceStability;
22   
23   import java.util.Collection;
24   import java.util.HashMap;
25   import java.util.Map;
26   import java.util.concurrent.ExecutionException;
27   
28   @InterfaceStability.Evolving
29   public class DescribeTransactionsResult {
30       private final Map<CoordinatorKey, KafkaFuture<TransactionDescription>> futures;
31   
32       DescribeTransactionsResult(Map<CoordinatorKey, KafkaFuture<TransactionDescription>> futures) {
33           this.futures = futures;
34       }
35   
36       /**
37        * Get the description of a specific transactional ID.
38        *
39        * @param transactionalId the transactional ID to describe
40        * @return a future which completes when the transaction description of a particular
41        *         transactional ID is available.
42        * @throws IllegalArgumentException if the `transactionalId` was not included in the
43        *         respective call to {@link Admin#describeTransactions(Collection, DescribeTransactionsOptions)}.
44        */
45       public KafkaFuture<TransactionDescription> description(String transactionalId) {
46           CoordinatorKey key = CoordinatorKey.byTransactionalId(transactionalId);
47           KafkaFuture<TransactionDescription> future = futures.get(key);
48           if (future == null) {
49               throw new IllegalArgumentException("TransactionalId " +
50                   "`" + transactionalId + "` was not included in the request");
51           }
52           return future;
53       }
54       /**
55        * Get a future which returns a map of the transaction descriptions requested in the respective
56        * call to {@link Admin#describeTransactions(Collection, DescribeTransactionsOptions)}.
57        *
58        * If the description fails on any of the transactional IDs in the request, then this future
59        * will also fail.
60        *
61        * @return a future which either completes when all transaction descriptions complete or fails
62        *         if any of the descriptions cannot be obtained
63        */
64       public KafkaFuture<Map<String, TransactionDescription>> all() {
65           return KafkaFuture.allOf(futures.values().toArray(new KafkaFuture[0]))
66               .thenApply(nil -> {
67                   Map<String, TransactionDescription> results = new HashMap<>(futures.size());
68                   for (Map.Entry<CoordinatorKey, KafkaFuture<TransactionDescription>> entry : futures.entrySet()) {
69                       try {
70                           results.put(entry.getKey().idValue, entry.getValue().get());
71                       } catch (InterruptedException | ExecutionException e) {
72                           // This should be unreachable, because allOf ensured that all the futures completed successfully.
73                           throw new RuntimeException(e);
74                       }
75                   }
76                   return results;
77               });
78       }
79   
80   }
