19   
20   import org.apache.kafka.common.KafkaFuture;
21   import org.apache.kafka.common.annotation.InterfaceStability;
22   
23   import java.util.HashMap;
24   import java.util.Map;
25   import java.util.concurrent.ExecutionException;
26   
27   
28   /**
29    * The result of the {@link KafkaAdminClient#describeConsumerGroups(Collection, DescribeConsumerGroupsOptions)}} call.
30    *
31    * The API of this class is evolving, see {@link Admin} for details.
32    */
33   @InterfaceStability.Evolving
34   public class DescribeConsumerGroupsResult {
35   
36       private final Map<String, KafkaFuture<ConsumerGroupDescription>> futures;
37   
38       public DescribeConsumerGroupsResult(final Map<String, KafkaFuture<ConsumerGroupDescription>> futures) {
39           this.futures = futures;
40       }
41   
42       /**
43        * Return a map from group id to futures which yield group descriptions.
44        */
45       public Map<String, KafkaFuture<ConsumerGroupDescription>> describedGroups() {
46           Map<String, KafkaFuture<ConsumerGroupDescription>> describedGroups = new HashMap<>();
47           futures.forEach((key, future) -> describedGroups.put(key, future));
48           return describedGroups;
49       }
50   
51       /**
52        * Return a future which yields all ConsumerGroupDescription objects, if all the describes succeed.
53        */
54       public KafkaFuture<Map<String, ConsumerGroupDescription>> all() {
55           return KafkaFuture.allOf(futures.values().toArray(new KafkaFuture[0])).thenApply(
56               nil -> {
57                   Map<String, ConsumerGroupDescription> descriptions = new HashMap<>(futures.size());
58                   futures.forEach((key, future) -> {
59                       try {
60                           descriptions.put(key, future.get());
61                       } catch (InterruptedException | ExecutionException e) {
62                           // This should be unreachable, since the KafkaFuture#allOf already ensured
63                           // that all of the futures completed successfully.
64                           throw new RuntimeException(e);
65                       }
66                   });
67                   return descriptions;
68               });
69       }
70   }
