17   package org.apache.kafka.common.security.oauthbearer.internals.unsecured;
18   
19   import java.util.Objects;
20   
21   import org.apache.kafka.common.KafkaException;
22   
23   /**
24    * Exception thrown when token validation fails due to a problem with the token
25    * itself (as opposed to a missing remote resource or a configuration problem)
26    */
27   public class OAuthBearerIllegalTokenException extends KafkaException {
28       private static final long serialVersionUID = -5275276640051316350L;
29       private final OAuthBearerValidationResult reason;
30   
31       /**
32        * Constructor
33        * 
34        * @param reason
35        *            the mandatory reason for the validation failure; it must indicate
36        *            failure
37        */
38       public OAuthBearerIllegalTokenException(OAuthBearerValidationResult reason) {
39           super(Objects.requireNonNull(reason).failureDescription());
40           if (reason.success())
41               throw new IllegalArgumentException("The reason indicates success; it must instead indicate failure");
42           this.reason = reason;
43       }
44   
45       /**
46        * Return the (always non-null) reason for the validation failure
47        * 
48        * @return the reason for the validation failure
49        */
50       public OAuthBearerValidationResult reason() {
51           return reason;
52       }
53   }
