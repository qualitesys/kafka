45       private static final List<OffsetFetchRequestTopic> ALL_TOPIC_PARTITIONS = null;
46       private static final List<OffsetFetchRequestTopics> ALL_TOPIC_PARTITIONS_BATCH = null;
47       private final OffsetFetchRequestData data;

54           public Builder(String groupId,
55                          boolean requireStable,
56                          List<TopicPartition> partitions,
57                          boolean throwOnFetchStableOffsetsUnsupported) {
58               super(ApiKeys.OFFSET_FETCH);
59   
60               final List<OffsetFetchRequestTopic> topics;
61               if (partitions != null) {
62                   Map<String, OffsetFetchRequestTopic> offsetFetchRequestTopicMap = new HashMap<>();
63                   for (TopicPartition topicPartition : partitions) {
64                       String topicName = topicPartition.topic();
65                       OffsetFetchRequestTopic topic = offsetFetchRequestTopicMap.getOrDefault(
66                           topicName, new OffsetFetchRequestTopic().setName(topicName));
67                       topic.partitionIndexes().add(topicPartition.partition());
68                       offsetFetchRequestTopicMap.put(topicName, topic);
69                   }
70                   topics = new ArrayList<>(offsetFetchRequestTopicMap.values());
71               } else {
72                   // If passed in partition list is null, it is requesting offsets for all topic partitions.
73                   topics = ALL_TOPIC_PARTITIONS;
74               }
75   
76               this.data = new OffsetFetchRequestData()
77                               .setGroupId(groupId)
78                               .setRequireStable(requireStable)
79                               .setTopics(topics);
80               this.throwOnFetchStableOffsetsUnsupported = throwOnFetchStableOffsetsUnsupported;
81           }
82   
83           boolean isAllTopicPartitions() {
84               return this.data.topics() == ALL_TOPIC_PARTITIONS;
85           }

87           public Builder(Map<String, List<TopicPartition>> groupIdToTopicPartitionMap,
88                          boolean requireStable,
89                          boolean throwOnFetchStableOffsetsUnsupported) {
90               super(ApiKeys.OFFSET_FETCH);
91   
92               List<OffsetFetchRequestGroup> groups = new ArrayList<>();
93               for (Entry<String, List<TopicPartition>> entry : groupIdToTopicPartitionMap.entrySet()) {
94                   String groupName = entry.getKey();
95                   List<TopicPartition> tpList = entry.getValue();
96                   final List<OffsetFetchRequestTopics> topics;
97                   if (tpList != null) {
98                       Map<String, OffsetFetchRequestTopics> offsetFetchRequestTopicMap =
99                           new HashMap<>();
100                       for (TopicPartition topicPartition : tpList) {
101                           String topicName = topicPartition.topic();
102                           OffsetFetchRequestTopics topic = offsetFetchRequestTopicMap.getOrDefault(
103                               topicName, new OffsetFetchRequestTopics().setName(topicName));
104                           topic.partitionIndexes().add(topicPartition.partition());
105                           offsetFetchRequestTopicMap.put(topicName, topic);
106                       }
107                       topics = new ArrayList<>(offsetFetchRequestTopicMap.values());
108                   } else {
109                       topics = ALL_TOPIC_PARTITIONS_BATCH;
110                   }
111                   groups.add(new OffsetFetchRequestGroup()
112                       .setGroupId(groupName)
113                       .setTopics(topics));
114               }
115               this.data = new OffsetFetchRequestData()
116                   .setGroups(groups)
117                   .setRequireStable(requireStable);
118               this.throwOnFetchStableOffsetsUnsupported = throwOnFetchStableOffsetsUnsupported;
119           }

121           @Override
122           public OffsetFetchRequest build(short version) {
123               if (isAllTopicPartitions() && version < 2) {
124                   throw new UnsupportedVersionException("The broker only supports OffsetFetchRequest " +
125                       "v" + version + ", but we need v2 or newer to request all topic partitions.");
126               }
127               if (data.groups().size() > 1 && version < 8) {
128                   throw new NoBatchedOffsetFetchRequestException("Broker does not support"
129                       + " batching groups for fetch offset request on version " + version);
130               }
131               if (data.requireStable() && version < 7) {
132                   if (throwOnFetchStableOffsetsUnsupported) {
133                       throw new UnsupportedVersionException("Broker unexpectedly " +
134                           "doesn't support requireStable flag on version " + version);
135                   } else {
136                       log.trace("Fallback the requireStable flag to false as broker " +
137                           "only supports OffsetFetchRequest version {}. Need " +
138                           "v7 or newer to enable this feature", version);
139                       data.setRequireStable(false);
140                   }
141               }
142               // convert data to use the appropriate version since version 8 uses different format
143               if (version < 8) {
144                   OffsetFetchRequestData oldDataFormat = null;
145                   if (!data.groups().isEmpty()) {
146                       OffsetFetchRequestGroup group = data.groups().get(0);
147                       String groupName = group.groupId();
148                       List<OffsetFetchRequestTopics> topics = group.topics();
149                       List<OffsetFetchRequestTopic> oldFormatTopics = null;
150                       if (topics != null) {
151                           oldFormatTopics = topics
152                               .stream()
153                               .map(t ->
154                                   new OffsetFetchRequestTopic()
155                                       .setName(t.name())
156                                       .setPartitionIndexes(t.partitionIndexes()))
157                               .collect(Collectors.toList());
158                       }
159                       oldDataFormat = new OffsetFetchRequestData()
160                           .setGroupId(groupName)
161                           .setTopics(oldFormatTopics)
162                           .setRequireStable(data.requireStable());
163                   }
164                   return new OffsetFetchRequest(oldDataFormat == null ? data : oldDataFormat, version);
165               } else {
166                   if (data.groups().isEmpty()) {
167                       String groupName = data.groupId();
168                       List<OffsetFetchRequestTopic> oldFormatTopics = data.topics();
169                       List<OffsetFetchRequestTopics> topics = null;
170                       if (oldFormatTopics != null) {
171                           topics = oldFormatTopics
172                               .stream()
173                               .map(t -> new OffsetFetchRequestTopics()
174                                   .setName(t.name())
175                                   .setPartitionIndexes(t.partitionIndexes()))
176                               .collect(Collectors.toList());
177                       }
178                       OffsetFetchRequestData convertedDataFormat =
179                           new OffsetFetchRequestData()
180                               .setGroups(Collections.singletonList(
181                                   new OffsetFetchRequestGroup()
182                                       .setGroupId(groupName)
183                                       .setTopics(topics)))
184                               .setRequireStable(data.requireStable());
185                       return new OffsetFetchRequest(convertedDataFormat, version);
186                   }
187               }
188               return new OffsetFetchRequest(data, version);
189           }

197       /**
198        * Indicates that it is not possible to fetch consumer groups in batches with FetchOffset.

200        */

205               super(message);
206           }

209       public List<TopicPartition> partitions() {
210           if (isAllPartitions()) {
211               return null;
212           }
213           List<TopicPartition> partitions = new ArrayList<>();
214           for (OffsetFetchRequestTopic topic : data.topics()) {
215               for (Integer partitionIndex : topic.partitionIndexes()) {
216                   partitions.add(new TopicPartition(topic.name(), partitionIndex));
217               }
218           }
219           return partitions;
220       }

230       public Map<String, List<TopicPartition>> groupIdsToPartitions() {
231           Map<String, List<TopicPartition>> groupIdsToPartitions = new HashMap<>();
232           for (OffsetFetchRequestGroup group : data.groups()) {
233               List<TopicPartition> tpList = null;
234               if (group.topics() != ALL_TOPIC_PARTITIONS_BATCH) {
235                   tpList = new ArrayList<>();
236                   for (OffsetFetchRequestTopics topic : group.topics()) {
237                       for (Integer partitionIndex : topic.partitionIndexes()) {
238                           tpList.add(new TopicPartition(topic.name(), partitionIndex));
239                       }
240                   }
241               }
242               groupIdsToPartitions.put(group.groupId(), tpList);
243           }
244           return groupIdsToPartitions;
245       }

248           Map<String, List<OffsetFetchRequestTopics>> groupIdsToTopics =
249               new HashMap<>(data.groups().size());
250           data.groups().forEach(g -> groupIdsToTopics.put(g.groupId(), g.topics()));
251           return groupIdsToTopics;
252       }

270       public OffsetFetchResponse getErrorResponse(int throttleTimeMs, Errors error) {
271           Map<TopicPartition, OffsetFetchResponse.PartitionData> responsePartitions = new HashMap<>();
272           if (version() < 2) {
273               OffsetFetchResponse.PartitionData partitionError = new OffsetFetchResponse.PartitionData(
274                       OffsetFetchResponse.INVALID_OFFSET,
275                       Optional.empty(),
276                       OffsetFetchResponse.NO_METADATA,
277                       error);
278   
279               for (OffsetFetchRequestTopic topic : this.data.topics()) {
280                   for (int partitionIndex : topic.partitionIndexes()) {
281                       responsePartitions.put(
282                           new TopicPartition(topic.name(), partitionIndex), partitionError);
283                   }
284               }
285               return new OffsetFetchResponse(error, responsePartitions);
286           }
287           if (version() == 2) {
288               return new OffsetFetchResponse(error, responsePartitions);
289           }
290           if (version() >= 3 && version() < 8) {
291               return new OffsetFetchResponse(throttleTimeMs, error, responsePartitions);
292           }
293           List<String> groupIds = groupIds();
294           Map<String, Errors> errorsMap = new HashMap<>(groupIds.size());
295           Map<String, Map<TopicPartition, OffsetFetchResponse.PartitionData>> partitionMap =
296               new HashMap<>(groupIds.size());
297           for (String g : groupIds) {
298               errorsMap.put(g, error);
299               partitionMap.put(g, responsePartitions);
300           }
301           return new OffsetFetchResponse(throttleTimeMs, errorsMap, partitionMap);
302       }
