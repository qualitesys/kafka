18   package org.apache.kafka.common.acl;
19   
20   import java.util.Objects;
21   
22   /**
23    * An internal, private class which contains the data stored in AccessControlEntry and
24    * AccessControlEntryFilter objects.
25    */
26   class AccessControlEntryData {
27       private final String principal;
28       private final String host;
29       private final AclOperation operation;
30       private final AclPermissionType permissionType;
31   
32       AccessControlEntryData(String principal, String host, AclOperation operation, AclPermissionType permissionType) {
33           this.principal = principal;
34           this.host = host;
35           this.operation = operation;
36           this.permissionType = permissionType;
37       }
38   
39       String principal() {
40           return principal;
41       }
42   
43       String host() {
44           return host;
45       }
46   
47       AclOperation operation() {
48           return operation;
49       }
50   
51       AclPermissionType permissionType() {
52           return permissionType;
53       }
54   
55       /**
56        * Returns a string describing an ANY or UNKNOWN field, or null if there is
57        * no such field.
58        */
59       public String findIndefiniteField() {
60           if (principal() == null)
61               return "Principal is NULL";
62           if (host() == null)
63               return "Host is NULL";
64           if (operation() == AclOperation.ANY)
65               return "Operation is ANY";
66           if (operation() == AclOperation.UNKNOWN)
67               return "Operation is UNKNOWN";
68           if (permissionType() == AclPermissionType.ANY)
69               return "Permission type is ANY";
70           if (permissionType() == AclPermissionType.UNKNOWN)
71               return "Permission type is UNKNOWN";
72           return null;
73       }
74   
75       @Override
76       public String toString() {
77           return "(principal=" + (principal == null ? "<any>" : principal) +
78                  ", host=" + (host == null ? "<any>" : host) +
79                  ", operation=" + operation +
80                  ", permissionType=" + permissionType + ")";
81       }
82   
83       /**
84        * Return true if there are any UNKNOWN components.
85        */
86       boolean isUnknown() {
87           return operation.isUnknown() || permissionType.isUnknown();
88       }
89   
90       @Override
91       public boolean equals(Object o) {
92           if (!(o instanceof AccessControlEntryData))
93               return false;
94           AccessControlEntryData other = (AccessControlEntryData) o;
95           return Objects.equals(principal, other.principal) &&
96               Objects.equals(host, other.host) &&
97               Objects.equals(operation, other.operation) &&
98               Objects.equals(permissionType, other.permissionType);
99       }
100   
101       @Override
102       public int hashCode() {
103           return Objects.hash(principal, host, operation, permissionType);
104       }
105   }
