167       @Override
168       public synchronized ConsumerRecords<K, V> poll(final Duration timeout) {
169           ensureNotClosed();
170   
171           lastPollTimeout = timeout;
172   
173           // Synchronize around the entire execution so new tasks to be triggered on subsequent poll calls can be added in
174           // the callback
175           synchronized (pollTasks) {
176               Runnable task = pollTasks.poll();
177               if (task != null)
178                   task.run();
179           }
180   
181           if (wakeup.get()) {
182               wakeup.set(false);
183               throw new WakeupException();
184           }
185   
186           if (pollException != null) {
187               RuntimeException exception = this.pollException;
188               this.pollException = null;
189               throw exception;
190           }
191   
192           // Handle seeks that need to wait for a poll() call to be processed
193           for (TopicPartition tp : subscriptions.assignedPartitions())
194               if (!subscriptions.hasValidPosition(tp))
195                   updateFetchPosition(tp);
196   
197           // update the consumed offset
198           final Map<TopicPartition, List<ConsumerRecord<K, V>>> results = new HashMap<>();
199           final List<TopicPartition> toClear = new ArrayList<>();
200   
201           for (Map.Entry<TopicPartition, List<ConsumerRecord<K, V>>> entry : this.records.entrySet()) {
202               if (!subscriptions.isPaused(entry.getKey())) {
203                   final List<ConsumerRecord<K, V>> recs = entry.getValue();
204                   for (final ConsumerRecord<K, V> rec : recs) {
205                       long position = subscriptions.position(entry.getKey()).offset;
206   
207                       if (beginningOffsets.get(entry.getKey()) != null && beginningOffsets.get(entry.getKey()) > position) {
208                           throw new OffsetOutOfRangeException(Collections.singletonMap(entry.getKey(), position));
209                       }
210   
211                       if (assignment().contains(entry.getKey()) && rec.offset() >= position) {
212                           results.computeIfAbsent(entry.getKey(), partition -> new ArrayList<>()).add(rec);
213                           Metadata.LeaderAndEpoch leaderAndEpoch = new Metadata.LeaderAndEpoch(Optional.empty(), rec.leaderEpoch());
214                           SubscriptionState.FetchPosition newPosition = new SubscriptionState.FetchPosition(
215                                   rec.offset() + 1, rec.leaderEpoch(), leaderAndEpoch);
216                           subscriptions.position(entry.getKey(), newPosition);
217                       }
218                   }
219                   toClear.add(entry.getKey());
220               }
221           }
222   
223           toClear.forEach(p -> this.records.remove(p));
224           return new ConsumerRecords<>(results);
225       }

492       private void updateFetchPosition(TopicPartition tp) {
493           if (subscriptions.isOffsetResetNeeded(tp)) {
494               resetOffsetPosition(tp);
495           } else if (!committed.containsKey(tp)) {
496               subscriptions.requestOffsetReset(tp);
497               resetOffsetPosition(tp);
498           } else {
499               subscriptions.seek(tp, committed.get(tp).offset());
500           }
501       }
502   
503       private void resetOffsetPosition(TopicPartition tp) {
504           OffsetResetStrategy strategy = subscriptions.resetStrategy(tp);
505           Long offset;
506           if (strategy == OffsetResetStrategy.EARLIEST) {
507               offset = beginningOffsets.get(tp);
508               if (offset == null)
509                   throw new IllegalStateException("MockConsumer didn't have beginning offset specified, but tried to seek to beginning");
510           } else if (strategy == OffsetResetStrategy.LATEST) {
511               offset = endOffsets.get(tp);
512               if (offset == null)
513                   throw new IllegalStateException("MockConsumer didn't have end offset specified, but tried to seek to end");
514           } else {
515               throw new NoOffsetForPartitionException(tp);
516           }
517           seek(tp, offset);
518       }
