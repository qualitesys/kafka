37   /**
38    * Base driver implementation for APIs which target partition leaders.
39    */
40   public class PartitionLeaderStrategy implements AdminApiLookupStrategy<TopicPartition> {
41       private static final ApiRequestScope SINGLE_REQUEST_SCOPE = new ApiRequestScope() {
42       };
43   
44       private final Logger log;
45   
46       public PartitionLeaderStrategy(LogContext logContext) {
47           this.log = logContext.logger(PartitionLeaderStrategy.class);
48       }
49   
50       @Override
51       public ApiRequestScope lookupScope(TopicPartition key) {
52           // Metadata requests can group topic partitions arbitrarily, so they can all share
53           // the same request context
54           return SINGLE_REQUEST_SCOPE;
55       }
56   
57       @Override
58       public MetadataRequest.Builder buildRequest(Set<TopicPartition> partitions) {
59           MetadataRequestData request = new MetadataRequestData();
60           request.setAllowAutoTopicCreation(false);
61           partitions.stream().map(TopicPartition::topic).distinct().forEach(topic ->
62               request.topics().add(new MetadataRequestData.MetadataRequestTopic().setName(topic))
63           );
64           return new MetadataRequest.Builder(request);
65       }
66   
67       private void handleTopicError(
68           String topic,
69           Errors topicError,
70           Set<TopicPartition> requestPartitions,
71           Map<TopicPartition, Throwable> failed
72       ) {
73           switch (topicError) {
74               case UNKNOWN_TOPIC_OR_PARTITION:
75               case LEADER_NOT_AVAILABLE:
76               case BROKER_NOT_AVAILABLE:
77                   log.debug("Metadata request for topic {} returned topic-level error {}. Will retry",
78                       topic, topicError);
79                   break;
80   
81               case TOPIC_AUTHORIZATION_FAILED:
82                   log.error("Received authorization failure for topic {} in `Metadata` response", topic,
83                       topicError.exception());
84                   failAllPartitionsForTopic(topic, requestPartitions, failed, tp -> new TopicAuthorizationException(
85                       "Failed to fetch metadata for partition " + tp + " due to topic authorization failure",
86                       Collections.singleton(topic)));
87                   break;
88   
89               case INVALID_TOPIC_EXCEPTION:
90                   log.error("Received invalid topic error for topic {} in `Metadata` response", topic,
91                       topicError.exception());
92                   failAllPartitionsForTopic(topic, requestPartitions, failed, tp -> new InvalidTopicException(
93                       "Failed to fetch metadata for partition " + tp + " due to invalid topic `" + topic + "`",
94                       Collections.singleton(topic)));
95                   break;
96   
97               default:
98                   log.error("Received unexpected error for topic {} in `Metadata` response", topic,
99                       topicError.exception());
100                   failAllPartitionsForTopic(topic, requestPartitions, failed, tp -> topicError.exception(
101                       "Failed to fetch metadata for partition " + tp + " due to unexpected error for topic `" + topic + "`"));
102           }
103       }
104   
105       private void failAllPartitionsForTopic(
106           String topic,
107           Set<TopicPartition> partitions,
108           Map<TopicPartition, Throwable> failed,
109           Function<TopicPartition, Throwable> exceptionGenerator
110       ) {
111           partitions.stream().filter(tp -> tp.topic().equals(topic)).forEach(tp -> {
112               failed.put(tp, exceptionGenerator.apply(tp));
113           });
114       }
115   
116       private void handlePartitionError(
117           TopicPartition topicPartition,
118           Errors partitionError,
119           Map<TopicPartition, Throwable> failed
120       ) {
121           switch (partitionError) {
122               case NOT_LEADER_OR_FOLLOWER:
123               case REPLICA_NOT_AVAILABLE:
124               case LEADER_NOT_AVAILABLE:
125               case BROKER_NOT_AVAILABLE:
126               case KAFKA_STORAGE_ERROR:
127                   log.debug("Metadata request for partition {} returned partition-level error {}. Will retry",
128                       topicPartition, partitionError);
129                   break;
130   
131               default:
132                   log.error("Received unexpected error for partition {} in `Metadata` response",
133                       topicPartition, partitionError.exception());
134                   failed.put(topicPartition, partitionError.exception(
135                       "Unexpected error during metadata lookup for " + topicPartition));
136           }
137       }
138   
139       @Override
140       public LookupResult<TopicPartition> handleResponse(
141           Set<TopicPartition> requestPartitions,
142           AbstractResponse abstractResponse
143       ) {
144           MetadataResponse response = (MetadataResponse) abstractResponse;
145           Map<TopicPartition, Throwable> failed = new HashMap<>();
146           Map<TopicPartition, Integer> mapped = new HashMap<>();
147   
148           for (MetadataResponseData.MetadataResponseTopic topicMetadata : response.data().topics()) {
149               String topic = topicMetadata.name();
150               Errors topicError = Errors.forCode(topicMetadata.errorCode());
151               if (topicError != Errors.NONE) {
152                   handleTopicError(topic, topicError, requestPartitions, failed);
153                   continue;
154               }
155   
156               for (MetadataResponseData.MetadataResponsePartition partitionMetadata : topicMetadata.partitions()) {
157                   TopicPartition topicPartition = new TopicPartition(topic, partitionMetadata.partitionIndex());
158                   Errors partitionError = Errors.forCode(partitionMetadata.errorCode());
159   
160                   if (!requestPartitions.contains(topicPartition)) {
161                       // The `Metadata` response always returns all partitions for requested
162                       // topics, so we have to filter any that we are not interested in.
163                       continue;
164                   }
165   
166                   if (partitionError != Errors.NONE) {
167                       handlePartitionError(topicPartition, partitionError, failed);
168                       continue;
169                   }
170   
171                   int leaderId = partitionMetadata.leaderId();
172                   if (leaderId >= 0) {
173                       mapped.put(topicPartition, leaderId);
174                   } else {
175                       log.debug("Metadata request for {} returned no error, but the leader is unknown. Will retry",
176                           topicPartition);
177                   }
178               }
179           }
180           return new LookupResult<>(failed, mapped);
181       }
