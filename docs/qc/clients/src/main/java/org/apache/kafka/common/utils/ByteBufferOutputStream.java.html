17   package org.apache.kafka.common.utils;
18   
19   import java.io.OutputStream;
20   import java.nio.ByteBuffer;
21   
22   /**
23    * A ByteBuffer-backed OutputStream that expands the internal ByteBuffer as required. Given this, the caller should
24    * always access the underlying ByteBuffer via the {@link #buffer()} method until all writes are completed.
25    *
26    * This class is typically used for 2 purposes:
27    *
28    * 1. Write to a ByteBuffer when there is a chance that we may need to expand it in order to fit all the desired data
29    * 2. Write to a ByteBuffer via methods that expect an OutputStream interface
30    *
31    * Hard to track bugs can happen when this class is used for the second reason and unexpected buffer expansion happens.
32    * So, it's best to assume that buffer expansion can always happen. An improvement would be to create a separate class
33    * that throws an error if buffer expansion is required to avoid the issue altogether.
34    */
35   public class ByteBufferOutputStream extends OutputStream {
36   
37       private static final float REALLOCATION_FACTOR = 1.1f;
38   
39       private final int initialCapacity;
40       private final int initialPosition;
41       private ByteBuffer buffer;
42   
43       /**
44        * Creates an instance of this class that will write to the received `buffer` up to its `limit`. If necessary to
45        * satisfy `write` or `position` calls, larger buffers will be allocated so the {@link #buffer()} method may return
46        * a different buffer than the received `buffer` parameter.
47        *
48        * Prefer one of the constructors that allocate the internal buffer for clearer semantics.
49        */
50       public ByteBufferOutputStream(ByteBuffer buffer) {
51           this.buffer = buffer;
52           this.initialPosition = buffer.position();
53           this.initialCapacity = buffer.capacity();
54       }
55   
56       public ByteBufferOutputStream(int initialCapacity) {
57           this(initialCapacity, false);
58       }
59   
60       public ByteBufferOutputStream(int initialCapacity, boolean directBuffer) {
61           this(directBuffer ? ByteBuffer.allocateDirect(initialCapacity) : ByteBuffer.allocate(initialCapacity));
62       }
63   
64       public void write(int b) {
65           ensureRemaining(1);
66           buffer.put((byte) b);
67       }
68   
69       public void write(byte[] bytes, int off, int len) {
70           ensureRemaining(len);
71           buffer.put(bytes, off, len);
72       }
73   
74       public void write(ByteBuffer sourceBuffer) {
75           ensureRemaining(sourceBuffer.remaining());
76           buffer.put(sourceBuffer);
77       }
78   
79       public ByteBuffer buffer() {
80           return buffer;
81       }
82   
83       public int position() {
84           return buffer.position();
85       }
86   
87       public int remaining() {
88           return buffer.remaining();
89       }
90   
91       public int limit() {
92           return buffer.limit();
93       }
94   
95       public void position(int position) {
96           ensureRemaining(position - buffer.position());
97           buffer.position(position);
98       }
99   
100       /**
101        * The capacity of the first internal ByteBuffer used by this class. This is useful in cases where a pooled
102        * ByteBuffer was passed via the constructor and it needs to be returned to the pool.
103        */
104       public int initialCapacity() {
105           return initialCapacity;
106       }
107   
108       /**
109        * Ensure there is enough space to write some number of bytes, expanding the underlying buffer if necessary.
110        * This can be used to avoid incremental expansions through calls to {@link #write(int)} when you know how
111        * many total bytes are needed.
112        *
113        * @param remainingBytesRequired The number of bytes required
114        */
115       public void ensureRemaining(int remainingBytesRequired) {
116           if (remainingBytesRequired > buffer.remaining())
117               expandBuffer(remainingBytesRequired);
118       }
119   
120       private void expandBuffer(int remainingRequired) {
121           int expandSize = Math.max((int) (buffer.limit() * REALLOCATION_FACTOR), buffer.position() + remainingRequired);
122           ByteBuffer temp = ByteBuffer.allocate(expandSize);
123           int limit = limit();
124           buffer.flip();
125           temp.put(buffer);
126           buffer.limit(limit);
127           // reset the old buffer's position so that the partial data in the new buffer cannot be mistakenly consumed
128           // we should ideally only do this for the original buffer, but the additional complexity doesn't seem worth it
129           buffer.position(initialPosition);
130           buffer = temp;
131       }
132   
133   }
