17   package org.apache.kafka.common.internals;
18   
19   import org.apache.kafka.common.utils.Exit;
20   
21   /**
22    * An error that indicates the need to exit the JVM process. This should only be used by the server or command-line
23    * tools. Clients should never shutdown the JVM process.
24    *
25    * This exception is expected to be caught at the highest level of the thread so that no shared lock is held by
26    * the thread when it calls {@link Exit#exit(int)}.
27    */
28   public class FatalExitError extends Error {
29   
30       private final static long serialVersionUID = 1L;
31   
32       private final int statusCode;
33   
34       public FatalExitError(int statusCode) {
35           if (statusCode == 0)
36               throw new IllegalArgumentException("statusCode must not be 0");
37           this.statusCode = statusCode;
38       }
39   
40       public FatalExitError() {
41           this(1);
42       }
43   
44       public int statusCode() {
45           return statusCode;
46       }
47   }
