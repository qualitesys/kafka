22   import java.util.ArrayList;
23   import java.util.HashMap;
24   import java.util.HashSet;
25   import java.util.List;
26   import java.util.Map;

31   /**
32    * This class wraps a set of {@link ConfigProvider} instances and uses them to perform
33    * transformations.
34    *
35    * <p>The default variable pattern is of the form <code>${provider:[path:]key}</code>,
36    * where the <code>provider</code> corresponds to a {@link ConfigProvider} instance, as passed to
37    * {@link ConfigTransformer#ConfigTransformer(Map)}.  The pattern will extract a set
38    * of paths (which are optional) and keys and then pass them to {@link ConfigProvider#get(String, Set)} to obtain the
39    * values with which to replace the variables.
40    *
41    * <p>For example, if a Map consisting of an entry with a provider name "file" and provider instance
42    * {@link FileConfigProvider} is passed to the {@link ConfigTransformer#ConfigTransformer(Map)}, and a Properties
43    * file with contents
44    * <pre>
45    * fileKey=someValue
46    * </pre>
47    * resides at the path "/tmp/properties.txt", then when a configuration Map which has an entry with a key "someKey" and
48    * a value "${file:/tmp/properties.txt:fileKey}" is passed to the {@link #transform(Map)} method, then the transformed
49    * Map will have an entry with key "someKey" and a value "someValue".
50    *
51    * <p>This class only depends on {@link ConfigProvider#get(String, Set)} and does not depend on subscription support
52    * in a {@link ConfigProvider}, such as the {@link ConfigProvider#subscribe(String, Set, ConfigChangeCallback)} and
53    * {@link ConfigProvider#unsubscribe(String, Set, ConfigChangeCallback)} methods.
54    */
55   public class ConfigTransformer {
56       public static final Pattern DEFAULT_PATTERN = Pattern.compile("\\$\\{([^}]*?):(([^}]*?):)?([^}]*?)\\}");
57       private static final String EMPTY_PATH = "";
58   
59       private final Map<String, ConfigProvider> configProviders;
60   
61       /**
62        * Creates a ConfigTransformer with the default pattern, of the form <code>${provider:[path:]key}</code>.
63        *
64        * @param configProviders a Map of provider names and {@link ConfigProvider} instances.
65        */
66       public ConfigTransformer(Map<String, ConfigProvider> configProviders) {
67           this.configProviders = configProviders;
68       }
69   
70       /**
71        * Transforms the given configuration data by using the {@link ConfigProvider} instances to
72        * look up values to replace the variables in the pattern.
73        *
74        * @param configs the configuration values to be transformed
75        * @return an instance of {@link ConfigTransformerResult}
76        */
77       public ConfigTransformerResult transform(Map<String, String> configs) {
78           Map<String, Map<String, Set<String>>> keysByProvider = new HashMap<>();
79           Map<String, Map<String, Map<String, String>>> lookupsByProvider = new HashMap<>();
80   
81           // Collect the variables from the given configs that need transformation
82           for (Map.Entry<String, String> config : configs.entrySet()) {
83               if (config.getValue() != null) {
84                   List<ConfigVariable> vars = getVars(config.getValue(), DEFAULT_PATTERN);
85                   for (ConfigVariable var : vars) {
86                       Map<String, Set<String>> keysByPath = keysByProvider.computeIfAbsent(var.providerName, k -> new HashMap<>());
87                       Set<String> keys = keysByPath.computeIfAbsent(var.path, k -> new HashSet<>());
88                       keys.add(var.variable);
89                   }
90               }
91           }
92   
93           // Retrieve requested variables from the ConfigProviders
94           Map<String, Long> ttls = new HashMap<>();
95           for (Map.Entry<String, Map<String, Set<String>>> entry : keysByProvider.entrySet()) {
96               String providerName = entry.getKey();
97               ConfigProvider provider = configProviders.get(providerName);
98               Map<String, Set<String>> keysByPath = entry.getValue();
99               if (provider != null && keysByPath != null) {
100                   for (Map.Entry<String, Set<String>> pathWithKeys : keysByPath.entrySet()) {
101                       String path = pathWithKeys.getKey();
102                       Set<String> keys = new HashSet<>(pathWithKeys.getValue());
103                       ConfigData configData = provider.get(path, keys);
104                       Map<String, String> data = configData.data();
105                       Long ttl = configData.ttl();
106                       if (ttl != null && ttl >= 0) {
107                           ttls.put(path, ttl);
108                       }
109                       Map<String, Map<String, String>> keyValuesByPath =
110                               lookupsByProvider.computeIfAbsent(providerName, k -> new HashMap<>());
111                       keyValuesByPath.put(path, data);
112                   }
113               }
114           }
115   
116           // Perform the transformations by performing variable replacements
117           Map<String, String> data = new HashMap<>(configs);
118           for (Map.Entry<String, String> config : configs.entrySet()) {
119               data.put(config.getKey(), replace(lookupsByProvider, config.getValue(), DEFAULT_PATTERN));
120           }
121           return new ConfigTransformerResult(data, ttls);
122       }
123   
124       private static List<ConfigVariable> getVars(String value, Pattern pattern) {
125           List<ConfigVariable> configVars = new ArrayList<>();
126           Matcher matcher = pattern.matcher(value);
127           while (matcher.find()) {
128               configVars.add(new ConfigVariable(matcher));
129           }
130           return configVars;
131       }
132   
133       private static String replace(Map<String, Map<String, Map<String, String>>> lookupsByProvider,
134                                     String value,
135                                     Pattern pattern) {
136           if (value == null) {
137               return null;
138           }
139           Matcher matcher = pattern.matcher(value);
140           StringBuilder builder = new StringBuilder();
141           int i = 0;
142           while (matcher.find()) {
143               ConfigVariable configVar = new ConfigVariable(matcher);
144               Map<String, Map<String, String>> lookupsByPath = lookupsByProvider.get(configVar.providerName);
145               if (lookupsByPath != null) {
146                   Map<String, String> keyValues = lookupsByPath.get(configVar.path);
147                   String replacement = keyValues.get(configVar.variable);
148                   builder.append(value, i, matcher.start());
149                   if (replacement == null) {
150                       // No replacements will be performed; just return the original value
151                       builder.append(matcher.group(0));
152                   } else {
153                       builder.append(replacement);
154                   }
155                   i = matcher.end();
156               }
157           }
158           builder.append(value, i, value.length());
159           return builder.toString();
160       }
161   
162       private static class ConfigVariable {
163           final String providerName;
164           final String path;
165           final String variable;
166   
167           ConfigVariable(Matcher matcher) {
168               this.providerName = matcher.group(1);
169               this.path = matcher.group(3) != null ? matcher.group(3) : EMPTY_PATH;
170               this.variable = matcher.group(4);
171           }
172   
173           public String toString() {
174               return "(" + providerName + ":" + (path != null ? path + ":" : "") + variable + ")";
175           }
176       }
177   }
