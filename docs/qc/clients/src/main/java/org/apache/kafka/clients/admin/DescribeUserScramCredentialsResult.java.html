34   /**
35    * The result of the {@link Admin#describeUserScramCredentials()} call.
36    *
37    * The API of this class is evolving, see {@link Admin} for details.
38    */
39   @InterfaceStability.Evolving
40   public class DescribeUserScramCredentialsResult {
41       private final KafkaFuture<DescribeUserScramCredentialsResponseData> dataFuture;
42   
43       /**
44        * Package-private constructor
45        *
46        * @param dataFuture the future indicating response data from the call
47        */
48       DescribeUserScramCredentialsResult(KafkaFuture<DescribeUserScramCredentialsResponseData> dataFuture) {
49           this.dataFuture = Objects.requireNonNull(dataFuture);
50       }
51   
52       /**
53        *
54        * @return a future for the results of all described users with map keys (one per user) being consistent with the
55        * contents of the list returned by {@link #users()}. The future will complete successfully only if all such user
56        * descriptions complete successfully.
57        */
58       public KafkaFuture<Map<String, UserScramCredentialsDescription>> all() {
59           final KafkaFutureImpl<Map<String, UserScramCredentialsDescription>> retval = new KafkaFutureImpl<>();
60           dataFuture.whenComplete((data, throwable) -> {
61               if (throwable != null) {
62                   retval.completeExceptionally(throwable);
63               } else {
64                   /* Check to make sure every individual described user succeeded.  Note that a successfully described user
65                    * is one that appears with *either* a NONE error code or a RESOURCE_NOT_FOUND error code. The
66                    * RESOURCE_NOT_FOUND means the client explicitly requested a describe of that particular user but it could
67                    * not be described because it does not exist; such a user will not appear as a key in the returned map.
68                    */
69                   Optional<DescribeUserScramCredentialsResponseData.DescribeUserScramCredentialsResult> optionalFirstFailedDescribe =
70                           data.results().stream().filter(result ->
71                                   result.errorCode() != Errors.NONE.code() && result.errorCode() != Errors.RESOURCE_NOT_FOUND.code()).findFirst();
72                   if (optionalFirstFailedDescribe.isPresent()) {
73                       retval.completeExceptionally(Errors.forCode(optionalFirstFailedDescribe.get().errorCode()).exception(optionalFirstFailedDescribe.get().errorMessage()));
74                   } else {
75                       Map<String, UserScramCredentialsDescription> retvalMap = new HashMap<>();
76                       data.results().stream().forEach(userResult ->
77                               retvalMap.put(userResult.user(), new UserScramCredentialsDescription(userResult.user(),
78                                       getScramCredentialInfosFor(userResult))));
79                       retval.complete(retvalMap);
80                   }
81               }
82           });
83           return retval;
84       }
85   
86       /**
87        *
88        * @return a future indicating the distinct users that meet the request criteria and that have at least one
89        * credential.  The future will not complete successfully if the user is not authorized to perform the describe
90        * operation; otherwise, it will complete successfully as long as the list of users with credentials can be
91        * successfully determined within some hard-coded timeout period. Note that the returned list will not include users
92        * that do not exist/have no credentials: a request to describe an explicit list of users, none of which existed/had
93        * a credential, will result in a future that returns an empty list being returned here. A returned list will
94        * include users that have a credential but that could not be described.
95        */
96       public KafkaFuture<List<String>> users() {
97           final KafkaFutureImpl<List<String>> retval = new KafkaFutureImpl<>();
98           dataFuture.whenComplete((data, throwable) -> {
99               if (throwable != null) {
100                   retval.completeExceptionally(throwable);
101               } else {
102                   retval.complete(data.results().stream()
103                           .filter(result -> result.errorCode() != Errors.RESOURCE_NOT_FOUND.code())
104                           .map(result -> result.user()).collect(Collectors.toList()));
105               }
106           });
107           return retval;
108       }
109   
110       /**
111        *
112        * @param userName the name of the user description being requested
113        * @return a future indicating the description results for the given user. The future will complete exceptionally if
114        * the future returned by {@link #users()} completes exceptionally.  Note that if the given user does not exist in
115        * the list of described users then the returned future will complete exceptionally with
116        * {@link org.apache.kafka.common.errors.ResourceNotFoundException}.
117        */
118       public KafkaFuture<UserScramCredentialsDescription> description(String userName) {
119           final KafkaFutureImpl<UserScramCredentialsDescription> retval = new KafkaFutureImpl<>();
120           dataFuture.whenComplete((data, throwable) -> {
121               if (throwable != null) {
122                   retval.completeExceptionally(throwable);
123               } else {
124                   // it is possible that there is no future for this user (for example, the original describe request was
125                   // for users 1, 2, and 3 but this is looking for user 4), so explicitly take care of that case
126                   Optional<DescribeUserScramCredentialsResponseData.DescribeUserScramCredentialsResult> optionalUserResult =
127                           data.results().stream().filter(result -> result.user().equals(userName)).findFirst();
128                   if (!optionalUserResult.isPresent()) {
129                       retval.completeExceptionally(new ResourceNotFoundException("No such user: " + userName));
130                   } else {
131                       DescribeUserScramCredentialsResponseData.DescribeUserScramCredentialsResult userResult = optionalUserResult.get();
132                       if (userResult.errorCode() != Errors.NONE.code()) {
133                           // RESOURCE_NOT_FOUND is included here
134                           retval.completeExceptionally(Errors.forCode(userResult.errorCode()).exception(userResult.errorMessage()));
135                       } else {
136                           retval.complete(new UserScramCredentialsDescription(userResult.user(), getScramCredentialInfosFor(userResult)));
137                       }
138                   }
139               }
140           });
141           return retval;
142       }
143   
144       private static List<ScramCredentialInfo> getScramCredentialInfosFor(
145               DescribeUserScramCredentialsResponseData.DescribeUserScramCredentialsResult userResult) {
146           return userResult.credentialInfos().stream().map(c ->
147                   new ScramCredentialInfo(ScramMechanism.fromType(c.mechanism()), c.iterations()))
148                   .collect(Collectors.toList());
149       }
