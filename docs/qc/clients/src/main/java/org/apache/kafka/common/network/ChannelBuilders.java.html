51       /**
52        * @param securityProtocol the securityProtocol
53        * @param contextType the contextType, it must be non-null if `securityProtocol` is SASL_*; it is ignored otherwise
54        * @param config client config
55        * @param listenerName the listenerName if contextType is SERVER or null otherwise
56        * @param clientSaslMechanism SASL mechanism if mode is CLIENT, ignored otherwise
57        * @param time the time instance
58        * @param saslHandshakeRequestEnable flag to enable Sasl handshake requests; disabled only for SASL
59        *             inter-broker connections with inter-broker protocol version < 0.10
60        * @param logContext the log context instance
61        *
62        * @return the configured `ChannelBuilder`
63        * @throws IllegalArgumentException if `mode` invariants described above is not maintained
64        */
65       public static ChannelBuilder clientChannelBuilder(
66               SecurityProtocol securityProtocol,
67               JaasContext.Type contextType,
68               AbstractConfig config,
69               ListenerName listenerName,
70               String clientSaslMechanism,
71               Time time,
72               boolean saslHandshakeRequestEnable,
73               LogContext logContext) {
74   
75           if (securityProtocol == SecurityProtocol.SASL_PLAINTEXT || securityProtocol == SecurityProtocol.SASL_SSL) {
76               if (contextType == null)
77                   throw new IllegalArgumentException("`contextType` must be non-null if `securityProtocol` is `" + securityProtocol + "`");
78               if (clientSaslMechanism == null)
79                   throw new IllegalArgumentException("`clientSaslMechanism` must be non-null in client mode if `securityProtocol` is `" + securityProtocol + "`");
80           }
81           return create(securityProtocol, Mode.CLIENT, contextType, config, listenerName, false, clientSaslMechanism,
82                   saslHandshakeRequestEnable, null, null, time, logContext, null);
83       }

86        * @param listenerName the listenerName

88        * @param securityProtocol the securityProtocol
89        * @param config server config
90        * @param credentialCache Credential cache for SASL/SCRAM if SCRAM is enabled
91        * @param tokenCache Delegation token cache
92        * @param time the time instance
93        * @param logContext the log context instance
94        * @param apiVersionSupplier supplier for ApiVersions responses sent prior to authentication
95        *

99                                                         boolean isInterBrokerListener,
100                                                         SecurityProtocol securityProtocol,
101                                                         AbstractConfig config,
102                                                         CredentialCache credentialCache,
103                                                         DelegationTokenCache tokenCache,
104                                                         Time time,

107           return create(securityProtocol, Mode.SERVER, JaasContext.Type.SERVER, config, listenerName,
108                   isInterBrokerListener, null, true, credentialCache,
109                   tokenCache, time, logContext, apiVersionSupplier);
110       }

112       private static ChannelBuilder create(SecurityProtocol securityProtocol,
113                                            Mode mode,
114                                            JaasContext.Type contextType,
115                                            AbstractConfig config,
116                                            ListenerName listenerName,
117                                            boolean isInterBrokerListener,
118                                            String clientSaslMechanism,
119                                            boolean saslHandshakeRequestEnable,
120                                            CredentialCache credentialCache,
121                                            DelegationTokenCache tokenCache,
122                                            Time time,
123                                            LogContext logContext,
124                                            Supplier<ApiVersionsResponse> apiVersionSupplier) {
125           Map<String, Object> configs = channelBuilderConfigs(config, listenerName);
126   
127           ChannelBuilder channelBuilder;
128           switch (securityProtocol) {
129               case SSL:
130                   requireNonNullMode(mode, securityProtocol);
131                   channelBuilder = new SslChannelBuilder(mode, listenerName, isInterBrokerListener, logContext);
132                   break;
133               case SASL_SSL:
134               case SASL_PLAINTEXT:
135                   requireNonNullMode(mode, securityProtocol);
136                   Map<String, JaasContext> jaasContexts;
137                   String sslClientAuthOverride = null;
138                   if (mode == Mode.SERVER) {
139                       @SuppressWarnings("unchecked")
140                       List<String> enabledMechanisms = (List<String>) configs.get(BrokerSecurityConfigs.SASL_ENABLED_MECHANISMS_CONFIG);
141                       jaasContexts = new HashMap<>(enabledMechanisms.size());
142                       for (String mechanism : enabledMechanisms)
143                           jaasContexts.put(mechanism, JaasContext.loadServerContext(listenerName, mechanism, configs));
144   
145                       // SSL client authentication is enabled in brokers for SASL_SSL only if listener-prefixed config is specified.
146                       if (listenerName != null && securityProtocol == SecurityProtocol.SASL_SSL) {
147                           String configuredClientAuth = (String) configs.get(BrokerSecurityConfigs.SSL_CLIENT_AUTH_CONFIG);
148                           String listenerClientAuth = (String) config.originalsWithPrefix(listenerName.configPrefix(), true)
149                                   .get(BrokerSecurityConfigs.SSL_CLIENT_AUTH_CONFIG);
150   
151                           // If `ssl.client.auth` is configured at the listener-level, we don't set an override and SslFactory
152                           // uses the value from `configs`. If not, we propagate `sslClientAuthOverride=NONE` to SslFactory and
153                           // it applies the override to the latest configs when it is configured or reconfigured. `Note that
154                           // ssl.client.auth` cannot be dynamically altered.
155                           if (listenerClientAuth == null) {
156                               sslClientAuthOverride = SslClientAuth.NONE.name().toLowerCase(Locale.ROOT);
157                               if (configuredClientAuth != null && !configuredClientAuth.equalsIgnoreCase(SslClientAuth.NONE.name())) {
158                                   log.warn("Broker configuration '{}' is applied only to SSL listeners. Listener-prefixed configuration can be used" +
159                                           " to enable SSL client authentication for SASL_SSL listeners. In future releases, broker-wide option without" +
160                                           " listener prefix may be applied to SASL_SSL listeners as well. All configuration options intended for specific" +
161                                           " listeners should be listener-prefixed.", BrokerSecurityConfigs.SSL_CLIENT_AUTH_CONFIG);
162                               }
163                           }
164                       }
165                   } else {
166                       // Use server context for inter-broker client connections and client context for other clients
167                       JaasContext jaasContext = contextType == JaasContext.Type.CLIENT ? JaasContext.loadClientContext(configs) :
168                               JaasContext.loadServerContext(listenerName, clientSaslMechanism, configs);
169                       jaasContexts = Collections.singletonMap(clientSaslMechanism, jaasContext);
170                   }
171                   channelBuilder = new SaslChannelBuilder(mode,
172                           jaasContexts,
173                           securityProtocol,
174                           listenerName,
175                           isInterBrokerListener,
176                           clientSaslMechanism,
177                           saslHandshakeRequestEnable,
178                           credentialCache,
179                           tokenCache,
180                           sslClientAuthOverride,
181                           time,
182                           logContext,
183                           apiVersionSupplier);
184                   break;
185               case PLAINTEXT:
186                   channelBuilder = new PlaintextChannelBuilder(listenerName);
187                   break;
188               default:
189                   throw new IllegalArgumentException("Unexpected securityProtocol " + securityProtocol);
190           }
191   
192           channelBuilder.configure(configs);
193           return channelBuilder;
194       }

196       /**
197        * @return a mutable RecordingMap. The elements got from RecordingMap are marked as "used".
198        */
199       @SuppressWarnings("unchecked")
200       static Map<String, Object> channelBuilderConfigs(final AbstractConfig config, final ListenerName listenerName) {
201           Map<String, Object> parsedConfigs;
202           if (listenerName == null)
203               parsedConfigs = (Map<String, Object>) config.values();
204           else
205               parsedConfigs = config.valuesWithPrefixOverride(listenerName.configPrefix());
206   
207           config.originals().entrySet().stream()
208               .filter(e -> !parsedConfigs.containsKey(e.getKey())) // exclude already parsed configs
209               // exclude already parsed listener prefix configs
210               .filter(e -> !(listenerName != null && e.getKey().startsWith(listenerName.configPrefix()) &&
211                   parsedConfigs.containsKey(e.getKey().substring(listenerName.configPrefix().length()))))
212               // exclude keys like `{mechanism}.some.prop` if "listener.name." prefix is present and key `some.prop` exists in parsed configs.
213               .filter(e -> !(listenerName != null && parsedConfigs.containsKey(e.getKey().substring(e.getKey().indexOf('.') + 1))))
214               .forEach(e -> parsedConfigs.put(e.getKey(), e.getValue()));
215           return parsedConfigs;
216       }

218       private static void requireNonNullMode(Mode mode, SecurityProtocol securityProtocol) {
219           if (mode == null)
220               throw new IllegalArgumentException("`mode` must be non-null if `securityProtocol` is `" + securityProtocol + "`");
221       }

223       public static KafkaPrincipalBuilder createPrincipalBuilder(Map<String, ?> configs,
224                                                                  KerberosShortNamer kerberosShortNamer,
225                                                                  SslPrincipalMapper sslPrincipalMapper) {
226           Class<?> principalBuilderClass = (Class<?>) configs.get(BrokerSecurityConfigs.PRINCIPAL_BUILDER_CLASS_CONFIG);
227           final KafkaPrincipalBuilder builder;
228   
229           if (principalBuilderClass == null || principalBuilderClass == DefaultKafkaPrincipalBuilder.class) {
230               builder = new DefaultKafkaPrincipalBuilder(kerberosShortNamer, sslPrincipalMapper);
231           } else if (KafkaPrincipalBuilder.class.isAssignableFrom(principalBuilderClass)) {
232               builder = (KafkaPrincipalBuilder) Utils.newInstance(principalBuilderClass);
233           } else {
234               throw new InvalidConfigurationException("Type " + principalBuilderClass.getName() + " is not " +
235                       "an instance of " + KafkaPrincipalBuilder.class.getName());
236           }
237   
238           if (builder instanceof Configurable)
239               ((Configurable) builder).configure(configs);
240   
241           return builder;
242       }
