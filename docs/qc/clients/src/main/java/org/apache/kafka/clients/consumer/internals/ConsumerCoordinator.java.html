268       private void maybeUpdateJoinedSubscription(Set<TopicPartition> assignedPartitions) {
269           if (subscriptions.hasPatternSubscription()) {
270               // Check if the assignment contains some topics that were not in the original
271               // subscription, if yes we will obey what leader has decided and add these topics
272               // into the subscriptions as long as they still match the subscribed pattern
273   
274               Set<String> addedTopics = new HashSet<>();
275               // this is a copy because its handed to listener below
276               for (TopicPartition tp : assignedPartitions) {
277                   if (!joinedSubscription.contains(tp.topic()))
278                       addedTopics.add(tp.topic());
279               }
280   
281               if (!addedTopics.isEmpty()) {
282                   Set<String> newSubscription = new HashSet<>(subscriptions.subscription());
283                   Set<String> newJoinedSubscription = new HashSet<>(joinedSubscription);
284                   newSubscription.addAll(addedTopics);
285                   newJoinedSubscription.addAll(addedTopics);
286   
287                   if (this.subscriptions.subscribeFromPattern(newSubscription))
288                       metadata.requestUpdateForNewTopics();
289                   this.joinedSubscription = newJoinedSubscription;
290               }
291           }

371       @Override
372       protected void onJoinComplete(int generation,
373                                     String memberId,
374                                     String assignmentStrategy,
375                                     ByteBuffer assignmentBuffer) {
376           log.debug("Executing onJoinComplete with generation {} and memberId {}", generation, memberId);
377   
378           // Only the leader is responsible for monitoring for metadata changes (i.e. partition changes)
379           if (!isLeader)
380               assignmentSnapshot = null;
381   
382           ConsumerPartitionAssignor assignor = lookupAssignor(assignmentStrategy);
383           if (assignor == null)
384               throw new IllegalStateException("Coordinator selected invalid assignment protocol: " + assignmentStrategy);
385   
386           // Give the assignor a chance to update internal state based on the received assignment
387           groupMetadata = new ConsumerGroupMetadata(rebalanceConfig.groupId, generation, memberId, rebalanceConfig.groupInstanceId);
388   
389           SortedSet<TopicPartition> ownedPartitions = new TreeSet<>(COMPARATOR);
390           ownedPartitions.addAll(subscriptions.assignedPartitions());
391   
392           // should at least encode the short version
393           if (assignmentBuffer.remaining() < 2)
394               throw new IllegalStateException("There are insufficient bytes available to read assignment from the sync-group response (" +
395                   "actual byte size " + assignmentBuffer.remaining() + ") , this is not expected; " +
396                   "it is possible that the leader's assign function is buggy and did not return any assignment for this member, " +
397                   "or because static member is configured and the protocol is buggy hence did not get the assignment for this member");
398   
399           Assignment assignment = ConsumerProtocol.deserializeAssignment(assignmentBuffer);
400   
401           SortedSet<TopicPartition> assignedPartitions = new TreeSet<>(COMPARATOR);
402           assignedPartitions.addAll(assignment.partitions());
403   
404           if (!subscriptions.checkAssignmentMatchedSubscription(assignedPartitions)) {
405               final String fullReason = String.format("received assignment %s does not match the current subscription %s; " +
406                       "it is likely that the subscription has changed since we joined the group, will re-join with current subscription",
407                       assignment.partitions(), subscriptions.prettyString());
408               requestRejoin("received assignment does not match the current subscription", fullReason);
409   
410               return;
411           }
412   
413           final AtomicReference<Exception> firstException = new AtomicReference<>(null);
414           SortedSet<TopicPartition> addedPartitions = new TreeSet<>(COMPARATOR);
415           addedPartitions.addAll(assignedPartitions);
416           addedPartitions.removeAll(ownedPartitions);
417   
418           if (protocol == RebalanceProtocol.COOPERATIVE) {
419               SortedSet<TopicPartition> revokedPartitions = new TreeSet<>(COMPARATOR);
420               revokedPartitions.addAll(ownedPartitions);
421               revokedPartitions.removeAll(assignedPartitions);
422   
423               log.info("Updating assignment with\n" +
424                       "\tAssigned partitions:                       {}\n" +
425                       "\tCurrent owned partitions:                  {}\n" +
426                       "\tAdded partitions (assigned - owned):       {}\n" +
427                       "\tRevoked partitions (owned - assigned):     {}\n",
428                   assignedPartitions,
429                   ownedPartitions,
430                   addedPartitions,
431                   revokedPartitions
432               );
433   
434               if (!revokedPartitions.isEmpty()) {
435                   // Revoke partitions that were previously owned but no longer assigned;
436                   // note that we should only change the assignment (or update the assignor's state)
437                   // AFTER we've triggered  the revoke callback
438                   firstException.compareAndSet(null, invokePartitionsRevoked(revokedPartitions));
439   
440                   // If revoked any partitions, need to re-join the group afterwards

490       /**
491        * Poll for coordinator events. This ensures that the coordinator is known and that the consumer
492        * has joined the group (if it is using group management). This also handles periodic offset commits
493        * if they are enabled.
494        * <p>
495        * Returns early if the timeout expires or if waiting on rejoin is not required
496        *
497        * @param timer Timer bounding how long this method can block
498        * @param waitForJoinGroup Boolean flag indicating if we should wait until re-join group completes
499        * @throws KafkaException if the rebalance callback throws an exception
500        * @return true iff the operation succeeded
501        */
502       public boolean poll(Timer timer, boolean waitForJoinGroup) {
503           maybeUpdateSubscriptionMetadata();
504   
505           invokeCompletedOffsetCommitCallbacks();
506   
507           if (subscriptions.hasAutoAssignedPartitions()) {
508               if (protocol == null) {
509                   throw new IllegalStateException("User configured " + ConsumerConfig.PARTITION_ASSIGNMENT_STRATEGY_CONFIG +
510                       " to empty while trying to subscribe for group protocol to auto assign partitions");
511               }
512               // Always update the heartbeat last poll time so that the heartbeat thread does not leave the
513               // group proactively due to application inactivity even if (say) the coordinator cannot be found.
514               pollHeartbeat(timer.currentTimeMs());
515               if (coordinatorUnknownAndUnready(timer)) {
516                   return false;
517               }
518   
519               if (rejoinNeededOrPending()) {
520                   // due to a race condition between the initial metadata fetch and the initial rebalance,
521                   // we need to ensure that the metadata is fresh before joining initially. This ensures
522                   // that we have matched the pattern against the cluster's topics at least once before joining.
523                   if (subscriptions.hasPatternSubscription()) {
524                       // For consumer group that uses pattern-based subscription, after a topic is created,
525                       // any consumer that discovers the topic after metadata refresh can trigger rebalance
526                       // across the entire consumer group. Multiple rebalances can be triggered after one topic
527                       // creation if consumers refresh metadata at vastly different times. We can significantly
528                       // reduce the number of rebalances caused by single topic creation by asking consumer to
529                       // refresh metadata before re-joining the group as long as the refresh backoff time has
530                       // passed.
531                       if (this.metadata.timeToAllowUpdate(timer.currentTimeMs()) == 0) {
532                           this.metadata.requestUpdate();
533                       }
534   
535                       if (!client.ensureFreshMetadata(timer)) {
536                           return false;
537                       }
538   
539                       maybeUpdateSubscriptionMetadata();
540                   }
541   
542                   // if not wait for join group, we would just use a timer of 0
543                   if (!ensureActiveGroup(waitForJoinGroup ? timer : time.timer(0L))) {
544                       // since we may use a different timer in the callee, we'd still need
545                       // to update the original timer's current time after the call
546                       timer.update(time.milliseconds());
547   
548                       return false;
549                   }
550               }
551           } else {
552               // For manually assigned partitions, we do not try to pro-actively lookup coordinator;
553               // instead we only try to refresh metadata when necessary.
554               // If connections to all nodes fail, wakeups triggered while attempting to send fetch
555               // requests result in polls returning immediately, causing a tight loop of polls. Without
556               // the wakeup, poll() with no channels would block for the timeout, delaying re-connection.
557               // awaitMetadataUpdate() in ensureCoordinatorReady initiates new connections with configured backoff and avoids the busy loop.
558               if (metadata.updateRequested() && !client.hasReadyNodes(timer.currentTimeMs())) {
559                   client.awaitMetadataUpdate(timer);
560               }
561   
562               // if there is pending coordinator requests, ensure they have a chance to be transmitted.
563               client.pollNoWakeup();
564           }
565   
566           maybeAutoCommitOffsetsAsync(timer.currentTimeMs());
567           return true;
568       }

572        * @param now current time in milliseconds
573        * @return the maximum time in milliseconds the caller should wait before the next invocation of poll()

707       /**
708        * Used by COOPERATIVE rebalance protocol only.
709        *
710        * Validate the assignments returned by the assignor such that no owned partitions are going to
711        * be reassigned to a different consumer directly: if the assignor wants to reassign an owned partition,
712        * it must first remove it from the new assignment of the current owner so that it is not assigned to any
713        * member, and then in the next rebalance it can finally reassign those partitions not owned by anyone to consumers.
714        */
715       private void validateCooperativeAssignment(final Map<String, List<TopicPartition>> ownedPartitions,
716                                                  final Map<String, Assignment> assignments) {
717           Set<TopicPartition> totalRevokedPartitions = new HashSet<>();
718           SortedSet<TopicPartition> totalAddedPartitions = new TreeSet<>(COMPARATOR);
719           for (final Map.Entry<String, Assignment> entry : assignments.entrySet()) {
720               final Assignment assignment = entry.getValue();
721               final Set<TopicPartition> addedPartitions = new HashSet<>(assignment.partitions());
722               addedPartitions.removeAll(ownedPartitions.get(entry.getKey()));
723               final Set<TopicPartition> revokedPartitions = new HashSet<>(ownedPartitions.get(entry.getKey()));
724               revokedPartitions.removeAll(assignment.partitions());
725   
726               totalAddedPartitions.addAll(addedPartitions);
727               totalRevokedPartitions.addAll(revokedPartitions);
728           }
729   
730           // if there are overlap between revoked partitions and added partitions, it means some partitions
731           // immediately gets re-assigned to another member while it is still claimed by some member
732           totalAddedPartitions.retainAll(totalRevokedPartitions);
733           if (!totalAddedPartitions.isEmpty()) {
734               log.error("With the COOPERATIVE protocol, owned partitions cannot be " +
735                   "reassigned to other members; however the assignor has reassigned partitions {} which are still owned " +
736                   "by some members", totalAddedPartitions);
737   
738               throw new IllegalStateException("Assignor supporting the COOPERATIVE protocol violates its requirements");
739           }

742       @Override
743       protected boolean onJoinPrepare(int generation, String memberId) {
744           log.debug("Executing onJoinPrepare with generation {} and memberId {}", generation, memberId);
745           boolean onJoinPrepareAsyncCommitCompleted = false;
746           // async commit offsets prior to rebalance if auto-commit enabled
747           RequestFuture<Void> future = maybeAutoCommitOffsetsAsync();
748           // return true when
749           // 1. future is null, which means no commit request sent, so it is still considered completed
750           // 2. offset commit completed
751           // 3. offset commit failed with non-retriable exception
752           if (future == null)
753               onJoinPrepareAsyncCommitCompleted = true;
754           else if (future.succeeded())
755               onJoinPrepareAsyncCommitCompleted = true;
756           else if (future.failed() && !future.isRetriable()) {
757               log.error("Asynchronous auto-commit of offsets failed: {}", future.exception().getMessage());
758               onJoinPrepareAsyncCommitCompleted = true;
759           }
760   
761   
762           // the generation / member-id can possibly be reset by the heartbeat thread
763           // upon getting errors or heartbeat timeouts; in this case whatever is previously
764           // owned partitions would be lost, we should trigger the callback and cleanup the assignment;
765           // otherwise we can proceed normally and revoke the partitions depending on the protocol,
766           // and in that case we should only change the assignment AFTER the revoke callback is triggered
767           // so that users can still access the previously owned partitions to commit offsets etc.
768           Exception exception = null;
769           final SortedSet<TopicPartition> revokedPartitions = new TreeSet<>(COMPARATOR);
770           if (generation == Generation.NO_GENERATION.generationId ||
771               memberId.equals(Generation.NO_GENERATION.memberId)) {
772               revokedPartitions.addAll(subscriptions.assignedPartitions());
773   
774               if (!revokedPartitions.isEmpty()) {
775                   log.info("Giving away all assigned partitions as lost since generation/memberID has been reset," +
776                       "indicating that consumer is in old state or no longer part of the group");
777                   exception = invokePartitionsLost(revokedPartitions);
778   
779                   subscriptions.assignFromSubscribed(Collections.emptySet());
780               }
781           } else {
782               switch (protocol) {
783                   case EAGER:
784                       // revoke all partitions
785                       revokedPartitions.addAll(subscriptions.assignedPartitions());
786                       exception = invokePartitionsRevoked(revokedPartitions);
787   
788                       subscriptions.assignFromSubscribed(Collections.emptySet());
789   
790                       break;
791   
792                   case COOPERATIVE:
793                       // only revoke those partitions that are not in the subscription any more.
794                       Set<TopicPartition> ownedPartitions = new HashSet<>(subscriptions.assignedPartitions());
795                       revokedPartitions.addAll(ownedPartitions.stream()
796                           .filter(tp -> !subscriptions.subscription().contains(tp.topic()))
797                           .collect(Collectors.toSet()));
798   
799                       if (!revokedPartitions.isEmpty()) {
800                           exception = invokePartitionsRevoked(revokedPartitions);
801   
802                           ownedPartitions.removeAll(revokedPartitions);
803                           subscriptions.assignFromSubscribed(ownedPartitions);
804                       }
805   
806                       break;
807               }
808           }
809   
810           isLeader = false;
811           subscriptions.resetGroupSubscription();
812   
813           if (exception != null) {
814               throw new KafkaException("User rebalance callback throws an error", exception);
815           }
816           return onJoinPrepareAsyncCommitCompleted;
817       }
