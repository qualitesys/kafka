17   package org.apache.kafka.common.security.oauthbearer.internals.expiring;
18   
19   import java.util.Map;
20   import java.util.Objects;
21   
22   import org.apache.kafka.common.config.SaslConfigs;
23   
24   /**
25    * Immutable refresh-related configuration for expiring credentials that can be
26    * parsed from a producer/consumer/broker config.
27    */
28   public class ExpiringCredentialRefreshConfig {
29       private final double loginRefreshWindowFactor;
30       private final double loginRefreshWindowJitter;
31       private final short loginRefreshMinPeriodSeconds;
32       private final short loginRefreshBufferSeconds;
33       private final boolean loginRefreshReloginAllowedBeforeLogout;
34   
35       /**
36        * Constructor based on producer/consumer/broker configs and the indicated value
37        * for whether or not client relogin is allowed before logout
38        * 
39        * @param configs
40        *            the mandatory (but possibly empty) producer/consumer/broker
41        *            configs upon which to build this instance
42        * @param clientReloginAllowedBeforeLogout
43        *            if the {@code LoginModule} and {@code SaslClient} implementations
44        *            support multiple simultaneous login contexts on a single
45        *            {@code Subject} at the same time. If true, then upon refresh,
46        *            logout will only be invoked on the original {@code LoginContext}
47        *            after a new one successfully logs in. This can be helpful if the
48        *            original credential still has some lifetime left when an attempt
49        *            to refresh the credential fails; the client will still be able to
50        *            create new connections as long as the original credential remains
51        *            valid. Otherwise, if logout is immediately invoked prior to
52        *            relogin, a relogin failure leaves the client without the ability
53        *            to connect until relogin does in fact succeed.
54        */
55       public ExpiringCredentialRefreshConfig(Map<String, ?> configs, boolean clientReloginAllowedBeforeLogout) {
56           Objects.requireNonNull(configs);
57           this.loginRefreshWindowFactor = (Double) configs.get(SaslConfigs.SASL_LOGIN_REFRESH_WINDOW_FACTOR);
58           this.loginRefreshWindowJitter = (Double) configs.get(SaslConfigs.SASL_LOGIN_REFRESH_WINDOW_JITTER);
59           this.loginRefreshMinPeriodSeconds = (Short) configs.get(SaslConfigs.SASL_LOGIN_REFRESH_MIN_PERIOD_SECONDS);
60           this.loginRefreshBufferSeconds = (Short) configs.get(SaslConfigs.SASL_LOGIN_REFRESH_BUFFER_SECONDS);
61           this.loginRefreshReloginAllowedBeforeLogout = clientReloginAllowedBeforeLogout;
62       }
63   
64       /**
65        * Background login refresh thread will sleep until the specified window factor
66        * relative to the credential's total lifetime has been reached, at which time
67        * it will try to refresh the credential.
68        * 
69        * @return the login refresh window factor
70        */
71       public double loginRefreshWindowFactor() {
72           return loginRefreshWindowFactor;
73       }
74   
75       /**
76        * Amount of random jitter added to the background login refresh thread's sleep
77        * time.
78        * 
79        * @return the login refresh window jitter
80        */
81       public double loginRefreshWindowJitter() {
82           return loginRefreshWindowJitter;
83       }
84   
85       /**
86        * The desired minimum time between checks by the background login refresh
87        * thread, in seconds
88        * 
89        * @return the desired minimum refresh period, in seconds
90        */
91       public short loginRefreshMinPeriodSeconds() {
92           return loginRefreshMinPeriodSeconds;
93       }
94   
95       /**
96        * The amount of buffer time before expiration to maintain when refreshing. If a
97        * refresh is scheduled to occur closer to expiration than the number of seconds
98        * defined here then the refresh will be moved up to maintain as much of the
99        * desired buffer as possible.
100        * 
101        * @return the refresh buffer, in seconds
102        */
103       public short loginRefreshBufferSeconds() {
104           return loginRefreshBufferSeconds;
105       }
106   
107       /**
108        * If the LoginModule and SaslClient implementations support multiple
109        * simultaneous login contexts on a single Subject at the same time. If true,
110        * then upon refresh, logout will only be invoked on the original LoginContext
111        * after a new one successfully logs in. This can be helpful if the original
112        * credential still has some lifetime left when an attempt to refresh the
113        * credential fails; the client will still be able to create new connections as
114        * long as the original credential remains valid. Otherwise, if logout is
115        * immediately invoked prior to relogin, a relogin failure leaves the client
116        * without the ability to connect until relogin does in fact succeed.
117        * 
118        * @return true if relogin is allowed prior to discarding an existing
119        *         (presumably unexpired) credential, otherwise false
120        */
121       public boolean loginRefreshReloginAllowedBeforeLogout() {
122           return loginRefreshReloginAllowedBeforeLogout;
123       }
124   }
