17   package org.apache.kafka.clients.producer;
18   
19   import org.apache.kafka.common.Configurable;
20   
21   /**
22    * A plugin interface that allows you to intercept (and possibly mutate) the records received by the producer before
23    * they are published to the Kafka cluster.
24    * <p>
25    * This class will get producer config properties via <code>configure()</code> method, including clientId assigned
26    * by KafkaProducer if not specified in the producer config. The interceptor implementation needs to be aware that it will be
27    * sharing producer config namespace with other interceptors and serializers, and ensure that there are no conflicts.
28    * <p>
29    * Exceptions thrown by ProducerInterceptor methods will be caught, logged, but not propagated further. As a result, if
30    * the user configures the interceptor with the wrong key and value type parameters, the producer will not throw an exception,
31    * just log the errors.
32    * <p>
33    * ProducerInterceptor callbacks may be called from multiple threads. Interceptor implementation must ensure thread-safety, if needed.
34    * <p>
35    * Implement {@link org.apache.kafka.common.ClusterResourceListener} to receive cluster metadata once it's available. Please see the class documentation for ClusterResourceListener for more information.
36    */
37   public interface ProducerInterceptor<K, V> extends Configurable, AutoCloseable {
38       /**
39        * This is called from {@link org.apache.kafka.clients.producer.KafkaProducer#send(ProducerRecord)} and
40        * {@link org.apache.kafka.clients.producer.KafkaProducer#send(ProducerRecord, Callback)} methods, before key and value
41        * get serialized and partition is assigned (if partition is not specified in ProducerRecord).
42        * <p>
43        * This method is allowed to modify the record, in which case, the new record will be returned. The implication of modifying
44        * key/value is that partition assignment (if not specified in ProducerRecord) will be done based on modified key/value,
45        * not key/value from the client. Consequently, key and value transformation done in onSend() needs to be consistent:
46        * same key and value should mutate to the same (modified) key and value. Otherwise, log compaction would not work
47        * as expected.
48        * <p>
49        * Similarly, it is up to interceptor implementation to ensure that correct topic/partition is returned in ProducerRecord.
50        * Most often, it should be the same topic/partition from 'record'.
51        * <p>
52        * Any exception thrown by this method will be caught by the caller and logged, but not propagated further.
53        * <p>
54        * Since the producer may run multiple interceptors, a particular interceptor's onSend() callback will be called in the order
55        * specified by {@link org.apache.kafka.clients.producer.ProducerConfig#INTERCEPTOR_CLASSES_CONFIG}. The first interceptor
56        * in the list gets the record passed from the client, the following interceptor will be passed the record returned by the
57        * previous interceptor, and so on. Since interceptors are allowed to modify records, interceptors may potentially get
58        * the record already modified by other interceptors. However, building a pipeline of mutable interceptors that depend on the output
59        * of the previous interceptor is discouraged, because of potential side-effects caused by interceptors potentially failing to
60        * modify the record and throwing an exception. If one of the interceptors in the list throws an exception from onSend(), the exception
61        * is caught, logged, and the next interceptor is called with the record returned by the last successful interceptor in the list,
62        * or otherwise the client.
63        *
64        * @param record the record from client or the record returned by the previous interceptor in the chain of interceptors.
65        * @return producer record to send to topic/partition
66        */
67       ProducerRecord<K, V> onSend(ProducerRecord<K, V> record);
68   
69       /**
70        * This method is called when the record sent to the server has been acknowledged, or when sending the record fails before
71        * it gets sent to the server.
72        * <p>
73        * This method is generally called just before the user callback is called, and in additional cases when <code>KafkaProducer.send()</code>
74        * throws an exception.
75        * <p>
76        * Any exception thrown by this method will be ignored by the caller.
77        * <p>
78        * This method will generally execute in the background I/O thread, so the implementation should be reasonably fast.
79        * Otherwise, sending of messages from other threads could be delayed.
80        *
81        * @param metadata The metadata for the record that was sent (i.e. the partition and offset).
82        *                 If an error occurred, metadata will contain only valid topic and maybe
83        *                 partition. If partition is not given in ProducerRecord and an error occurs
84        *                 before partition gets assigned, then partition will be set to RecordMetadata.NO_PARTITION.
85        *                 The metadata may be null if the client passed null record to
86        *                 {@link org.apache.kafka.clients.producer.KafkaProducer#send(ProducerRecord)}.
87        * @param exception The exception thrown during processing of this record. Null if no error occurred.
88        */
89       void onAcknowledgement(RecordMetadata metadata, Exception exception);
90   
91       /**
92        * This is called when interceptor is closed
93        */
94       void close();
95   }
