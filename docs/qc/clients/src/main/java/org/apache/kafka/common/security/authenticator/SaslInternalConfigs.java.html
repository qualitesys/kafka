21   public class SaslInternalConfigs {
22       /**
23        * The server (broker) specifies a positive session length in milliseconds to a
24        * SASL client when {@link BrokerSecurityConfigs#CONNECTIONS_MAX_REAUTH_MS} is
25        * positive as per <a href=
26        * "https://cwiki.apache.org/confluence/display/KAFKA/KIP-368%3A+Allow+SASL+Connections+to+Periodically+Re-Authenticate">KIP
27        * 368: Allow SASL Connections to Periodically Re-Authenticate</a>. The session
28        * length is the minimum of the configured value and any session length implied
29        * by the credential presented during authentication. The lifetime defined by
30        * the credential, in terms of milliseconds since the epoch, is available via a
31        * negotiated property on the SASL Server instance, and that value can be
32        * converted to a session length by subtracting the time at which authentication
33        * occurred. This variable defines the negotiated property key that is used to
34        * communicate the credential lifetime in milliseconds since the epoch.
35        */
36       public static final String CREDENTIAL_LIFETIME_MS_SASL_NEGOTIATED_PROPERTY_KEY = "CREDENTIAL.LIFETIME.MS";
37   
38       private SaslInternalConfigs() {
39           // empty
40       }
