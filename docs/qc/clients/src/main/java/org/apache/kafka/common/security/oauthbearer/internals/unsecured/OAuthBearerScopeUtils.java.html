17   package org.apache.kafka.common.security.oauthbearer.internals.unsecured;
18   
19   import java.util.ArrayList;
20   import java.util.Collections;
21   import java.util.List;
22   import java.util.Objects;
23   import java.util.regex.Pattern;
24   
25   /**
26    * Utility class for help dealing with
27    * <a href="https://tools.ietf.org/html/rfc6749#section-3.3">Access Token
28    * Scopes</a>
29    */
30   public class OAuthBearerScopeUtils {
31       private static final Pattern INDIVIDUAL_SCOPE_ITEM_PATTERN = Pattern.compile("[\\x23-\\x5B\\x5D-\\x7E\\x21]+");
32   
33       /**
34        * Return true if the given value meets the definition of a valid scope item as
35        * per <a href="https://tools.ietf.org/html/rfc6749#section-3.3">RFC 6749
36        * Section 3.3</a>, otherwise false
37        * 
38        * @param scopeItem
39        *            the mandatory scope item to check for validity
40        * @return true if the given value meets the definition of a valid scope item,
41        *         otherwise false
42        */
43       public static boolean isValidScopeItem(String scopeItem) {
44           return INDIVIDUAL_SCOPE_ITEM_PATTERN.matcher(Objects.requireNonNull(scopeItem)).matches();
45       }
46   
47       /**
48        * Convert a space-delimited list of scope values (for example,
49        * <code>"scope1 scope2"</code>) to a List containing the individual elements
50        * (<code>"scope1"</code> and <code>"scope2"</code>)
51        * 
52        * @param spaceDelimitedScope
53        *            the mandatory (but possibly empty) space-delimited scope values,
54        *            each of which must be valid according to
55        *            {@link #isValidScopeItem(String)}
56        * @return the list of the given (possibly empty) space-delimited values
57        * @throws OAuthBearerConfigException
58        *             if any of the individual scope values are malformed/illegal
59        */
60       public static List<String> parseScope(String spaceDelimitedScope) throws OAuthBearerConfigException {
61           List<String> retval = new ArrayList<>();
62           for (String individualScopeItem : Objects.requireNonNull(spaceDelimitedScope).split(" ")) {
63               if (!individualScopeItem.isEmpty()) {
64                   if (!isValidScopeItem(individualScopeItem))
65                       throw new OAuthBearerConfigException(String.format("Invalid scope value: %s", individualScopeItem));
66                   retval.add(individualScopeItem);
67               }
68           }
69           return Collections.unmodifiableList(retval);
70       }
71   
72       private OAuthBearerScopeUtils() {
73           // empty
74       }
75   }
