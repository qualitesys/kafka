43   /**
44    * This {@link RecordBatch} implementation is for magic versions 0 and 1. In addition to implementing
45    * {@link RecordBatch}, it also implements {@link Record}, which exposes the duality of the old message
46    * format in its handling of compressed messages. The wrapper record is considered the record batch in this
47    * interface, while the inner records are considered the log records (though they both share the same schema).

279       private static final class DataLogInputStream implements LogInputStream<AbstractLegacyRecordBatch> {
280           private final InputStream stream;
281           protected final int maxMessageSize;
282           private final ByteBuffer offsetAndSizeBuffer;
283   
284           DataLogInputStream(InputStream stream, int maxMessageSize) {
285               this.stream = stream;
286               this.maxMessageSize = maxMessageSize;
287               this.offsetAndSizeBuffer = ByteBuffer.allocate(Records.LOG_OVERHEAD);
288           }
289   
290           public AbstractLegacyRecordBatch nextBatch() throws IOException {
291               offsetAndSizeBuffer.clear();
292               Utils.readFully(stream, offsetAndSizeBuffer);
293               if (offsetAndSizeBuffer.hasRemaining())
294                   return null;
295   
296               long offset = offsetAndSizeBuffer.getLong(Records.OFFSET_OFFSET);
297               int size = offsetAndSizeBuffer.getInt(Records.SIZE_OFFSET);
298               if (size < LegacyRecord.RECORD_OVERHEAD_V0)
299                   throw new CorruptRecordException(String.format("Record size is less than the minimum record overhead (%d)", LegacyRecord.RECORD_OVERHEAD_V0));
300               if (size > maxMessageSize)
301                   throw new CorruptRecordException(String.format("Record size exceeds the largest allowable message size (%d).", maxMessageSize));
302   
303               ByteBuffer batchBuffer = ByteBuffer.allocate(size);
304               Utils.readFully(stream, batchBuffer);
305               if (batchBuffer.hasRemaining())
306                   return null;
307               batchBuffer.flip();
308   
309               return new BasicLegacyRecordBatch(offset, new LegacyRecord(batchBuffer));
310           }
311       }

313       private static class DeepRecordsIterator extends AbstractIterator<Record> implements CloseableIterator<Record> {
314           private final ArrayDeque<AbstractLegacyRecordBatch> innerEntries;
315           private final long absoluteBaseOffset;
316           private final byte wrapperMagic;
317   
318           private DeepRecordsIterator(AbstractLegacyRecordBatch wrapperEntry,
319                                       boolean ensureMatchingMagic,
320                                       int maxMessageSize,
321                                       BufferSupplier bufferSupplier) {
322               LegacyRecord wrapperRecord = wrapperEntry.outerRecord();
323               this.wrapperMagic = wrapperRecord.magic();
324               if (wrapperMagic != RecordBatch.MAGIC_VALUE_V0 && wrapperMagic != RecordBatch.MAGIC_VALUE_V1)
325                   throw new InvalidRecordException("Invalid wrapper magic found in legacy deep record iterator " + wrapperMagic);
326   
327               CompressionType compressionType = wrapperRecord.compressionType();
328               if (compressionType == CompressionType.ZSTD)
329                   throw new InvalidRecordException("Invalid wrapper compressionType found in legacy deep record iterator " + wrapperMagic);
330               ByteBuffer wrapperValue = wrapperRecord.value();
331               if (wrapperValue == null)
332                   throw new InvalidRecordException("Found invalid compressed record set with null value (magic = " +
333                           wrapperMagic + ")");
334   
335               InputStream stream = compressionType.wrapForInput(wrapperValue, wrapperRecord.magic(), bufferSupplier);
336               LogInputStream<AbstractLegacyRecordBatch> logStream = new DataLogInputStream(stream, maxMessageSize);
337   
338               long lastOffsetFromWrapper = wrapperEntry.lastOffset();
339               long timestampFromWrapper = wrapperRecord.timestamp();
340               this.innerEntries = new ArrayDeque<>();
341   
342               // If relative offset is used, we need to decompress the entire message first to compute
343               // the absolute offset. For simplicity and because it's a format that is on its way out, we
344               // do the same for message format version 0
345               try {
346                   while (true) {
347                       AbstractLegacyRecordBatch innerEntry = logStream.nextBatch();
348                       if (innerEntry == null)
349                           break;
350   
351                       LegacyRecord record = innerEntry.outerRecord();
352                       byte magic = record.magic();
353   
354                       if (ensureMatchingMagic && magic != wrapperMagic)
355                           throw new InvalidRecordException("Compressed message magic " + magic +
356                                   " does not match wrapper magic " + wrapperMagic);
357   
358                       if (magic == RecordBatch.MAGIC_VALUE_V1) {
359                           LegacyRecord recordWithTimestamp = new LegacyRecord(
360                                   record.buffer(),
361                                   timestampFromWrapper,
362                                   wrapperRecord.timestampType());
363                           innerEntry = new BasicLegacyRecordBatch(innerEntry.lastOffset(), recordWithTimestamp);
364                       }
365   
366                       innerEntries.addLast(innerEntry);
367                   }
368   
369                   if (innerEntries.isEmpty())
370                       throw new InvalidRecordException("Found invalid compressed record set with no inner records");
371   
372                   if (wrapperMagic == RecordBatch.MAGIC_VALUE_V1) {
373                       if (lastOffsetFromWrapper == 0) {
374                           // The outer offset may be 0 if this is produce data from certain versions of librdkafka.
375                           this.absoluteBaseOffset = 0;
376                       } else {
377                           long lastInnerOffset = innerEntries.getLast().offset();
378                           if (lastOffsetFromWrapper < lastInnerOffset)
379                               throw new InvalidRecordException("Found invalid wrapper offset in compressed v1 message set, " +
380                                       "wrapper offset '" + lastOffsetFromWrapper + "' is less than the last inner message " +
381                                       "offset '" + lastInnerOffset + "' and it is not zero.");
382                           this.absoluteBaseOffset = lastOffsetFromWrapper - lastInnerOffset;
383                       }
384                   } else {
385                       this.absoluteBaseOffset = -1;
386                   }
387               } catch (IOException e) {
388                   throw new KafkaException(e);
389               } finally {
390                   Utils.closeQuietly(stream, "records iterator stream");
391               }
392           }
393   
394           @Override
395           protected Record makeNext() {
396               if (innerEntries.isEmpty())
397                   return allDone();
398   
399               AbstractLegacyRecordBatch entry = innerEntries.remove();
400   
401               // Convert offset to absolute offset if needed.
402               if (wrapperMagic == RecordBatch.MAGIC_VALUE_V1) {
403                   long absoluteOffset = absoluteBaseOffset + entry.offset();
404                   entry = new BasicLegacyRecordBatch(absoluteOffset, entry.outerRecord());
405               }
406   
407               if (entry.isCompressed())
408                   throw new InvalidRecordException("Inner messages must not be compressed");
409   
410               return entry;
411           }
412   
413           @Override
414           public void close() {}
415       }
