285        * Begin connecting to the given node, return true if we are already connected and ready to send to that node.
286        *
287        * @param node The node to check

289        * @return True if we are ready to send to the given node

291       @Override
292       public boolean ready(Node node, long now) {
293           if (node.isEmpty())
294               throw new IllegalArgumentException("Cannot connect to empty node " + node);
295   
296           if (isReady(node, now))
297               return true;
298   
299           if (connectionStates.canConnect(node.idString(), now))
300               // if we are interested in sending to a node and we don't have a connection to it, initiate one
301               initiateConnect(node, now);
302   
303           return false;
304       }

311       /**
312        * Disconnects the connection to a particular node, if there is one.
313        * Any pending ClientRequests for this connection will receive disconnections.
314        *
315        * @param nodeId The id of the node
316        */
317       @Override
318       public void disconnect(String nodeId) {
319           if (connectionStates.isDisconnected(nodeId)) {
320               log.debug("Client requested disconnect from node {}, which is already disconnected", nodeId);
321               return;

325           selector.close(nodeId);
326           long now = time.milliseconds();

331       private void cancelInFlightRequests(String nodeId, long now, Collection<ClientResponse> responses) {
332           Iterable<InFlightRequest> inFlightRequests = this.inFlightRequests.clearAll(nodeId);
333           for (InFlightRequest request : inFlightRequests) {
334               if (log.isDebugEnabled()) {
335                   log.debug("Cancelled in-flight {} request with correlation id {} due to node {} being disconnected " +
336                           "(elapsed time since creation: {}ms, elapsed time since send: {}ms, request timeout: {}ms): {}",
337                       request.header.apiKey(), request.header.correlationId(), nodeId,
338                       request.timeElapsedSinceCreateMs(now), request.timeElapsedSinceSendMs(now),
339                       request.requestTimeoutMs, request.request);
340               } else {
341                   log.info("Cancelled in-flight {} request with correlation id {} due to node {} being disconnected " +
342                           "(elapsed time since creation: {}ms, elapsed time since send: {}ms, request timeout: {}ms)",
343                       request.header.apiKey(), request.header.correlationId(), nodeId,
344                       request.timeElapsedSinceCreateMs(now), request.timeElapsedSinceSendMs(now),
345                       request.requestTimeoutMs);
346               }
347   
348               if (!request.isInternalRequest) {
349                   if (responses != null)
350                       responses.add(request.disconnected(now, null));
351               } else if (request.header.apiKey() == ApiKeys.METADATA) {
352                   metadataUpdater.handleFailedRequest(now, Optional.empty());
353               }
354           }
355       }

469       private void doSend(ClientRequest clientRequest, boolean isInternalRequest, long now) {
470           ensureActive();
471           String nodeId = clientRequest.destination();
472           if (!isInternalRequest) {
473               // If this request came from outside the NetworkClient, validate
474               // that we can send data.  If the request is internal, we trust
475               // that internal code has done this validation.  Validation
476               // will be slightly different for some internal requests (for
477               // example, ApiVersionsRequests can be sent prior to being in
478               // READY state.)
479               if (!canSendRequest(nodeId, now))
480                   throw new IllegalStateException("Attempt to send a request to node " + nodeId + " which is not ready.");
481           }
482           AbstractRequest.Builder<?> builder = clientRequest.requestBuilder();
483           try {
484               NodeApiVersions versionInfo = apiVersions.get(nodeId);
485               short version;
486               // Note: if versionInfo is null, we have no server version information. This would be
487               // the case when sending the initial ApiVersionRequest which fetches the version
488               // information itself.  It is also the case when discoverBrokerVersions is set to false.
489               if (versionInfo == null) {
490                   version = builder.latestAllowedVersion();
491                   if (discoverBrokerVersions && log.isTraceEnabled())
492                       log.trace("No version information found when sending {} with correlation id {} to node {}. " +
493                               "Assuming version {}.", clientRequest.apiKey(), clientRequest.correlationId(), nodeId, version);
494               } else {
495                   version = versionInfo.latestUsableVersion(clientRequest.apiKey(), builder.oldestAllowedVersion(),
496                           builder.latestAllowedVersion());
497               }
498               // The call to build may also throw UnsupportedVersionException, if there are essential
499               // fields that cannot be represented in the chosen version.
500               doSend(clientRequest, isInternalRequest, now, builder.build(version));
501           } catch (UnsupportedVersionException unsupportedVersionException) {
502               // If the version is not supported, skip sending the request over the wire.
503               // Instead, simply add it to the local queue of aborted requests.
504               log.debug("Version mismatch when attempting to send {} with correlation id {} to {}", builder,
505                       clientRequest.correlationId(), clientRequest.destination(), unsupportedVersionException);
506               ClientResponse clientResponse = new ClientResponse(clientRequest.makeHeader(builder.latestAllowedVersion()),
507                       clientRequest.callback(), clientRequest.destination(), now, now,
508                       false, unsupportedVersionException, null, null);
509   
510               if (!isInternalRequest)
511                   abortedSends.add(clientResponse);
512               else if (clientRequest.apiKey() == ApiKeys.METADATA)
513                   metadataUpdater.handleFailedRequest(now, Optional.of(unsupportedVersionException));
514           }
515       }

517       private void doSend(ClientRequest clientRequest, boolean isInternalRequest, long now, AbstractRequest request) {
518           String destination = clientRequest.destination();
519           RequestHeader header = clientRequest.makeHeader(request.version());
520           if (log.isDebugEnabled()) {
521               log.debug("Sending {} request with header {} and timeout {} to node {}: {}",
522                   clientRequest.apiKey(), header, clientRequest.requestTimeoutMs(), destination, request);
523           }
524           Send send = request.toSend(header);
525           InFlightRequest inFlightRequest = new InFlightRequest(
526                   clientRequest,
527                   header,
528                   isInternalRequest,
529                   request,
530                   send,
531                   now);
532           this.inFlightRequests.add(inFlightRequest);
533           selector.send(new NetworkSend(clientRequest.destination(), send));
534       }

536       /**
537        * Do actual reads and writes to sockets.
538        *
539        * @param timeout The maximum amount of time to wait (in ms) for responses if there are none immediately,
540        *                must be non-negative. The actual timeout will be the minimum of timeout, request timeout and
541        *                metadata timeout
542        * @param now The current time in milliseconds
543        * @return The list of responses received
544        */
545       @Override
546       public List<ClientResponse> poll(long timeout, long now) {
547           ensureActive();
548   
549           if (!abortedSends.isEmpty()) {
550               // If there are aborted sends because of unsupported version exceptions or disconnects,
551               // handle them immediately without waiting for Selector#poll.
552               List<ClientResponse> responses = new ArrayList<>();
553               handleAbortedSends(responses);
554               completeResponses(responses);
555               return responses;
556           }
557   
558           long metadataTimeout = metadataUpdater.maybeUpdate(now);
559           try {
560               this.selector.poll(Utils.min(timeout, metadataTimeout, defaultRequestTimeoutMs));
561           } catch (IOException e) {
562               log.error("Unexpected error during I/O", e);
563           }
564   
565           // process completed actions
566           long updatedNow = this.time.milliseconds();
567           List<ClientResponse> responses = new ArrayList<>();
568           handleCompletedSends(responses, updatedNow);
569           handleCompletedReceives(responses, updatedNow);
570           handleDisconnections(responses, updatedNow);
571           handleConnections();
572           handleInitiateApiVersionRequests(updatedNow);
573           handleTimedOutConnections(responses, updatedNow);
574           handleTimedOutRequests(responses, updatedNow);
575           completeResponses(responses);
576   
577           return responses;
578       }

584               } catch (Exception e) {
585                   log.error("Uncaught error in request completion:", e);
586               }
587           }
588       }

660       /**
661        * Choose the node with the fewest outstanding requests which is at least eligible for connection. This method will
662        * prefer a node with an existing connection, but will potentially choose a node for which we don't yet have a
663        * connection if all existing connections are in use. If no connection exists, this method will prefer a node
664        * with least recent connection attempts. This method will never choose a node for which there is no
665        * existing connection and from which we have disconnected within the reconnect backoff period, or an active
666        * connection which is being throttled.
667        *
668        * @return The node with the fewest in-flight requests.
669        */
670       @Override
671       public Node leastLoadedNode(long now) {
672           List<Node> nodes = this.metadataUpdater.fetchNodes();
673           if (nodes.isEmpty())
674               throw new IllegalStateException("There are no nodes in the Kafka cluster");
675           int inflight = Integer.MAX_VALUE;
676   
677           Node foundConnecting = null;
678           Node foundCanConnect = null;
679           Node foundReady = null;
680   
681           int offset = this.randOffset.nextInt(nodes.size());
682           for (int i = 0; i < nodes.size(); i++) {
683               int idx = (offset + i) % nodes.size();
684               Node node = nodes.get(idx);
685               if (canSendRequest(node.idString(), now)) {
686                   int currInflight = this.inFlightRequests.count(node.idString());
687                   if (currInflight == 0) {
688                       // if we find an established connection with no in-flight requests we can stop right away
689                       log.trace("Found least loaded node {} connected with no in-flight requests", node);
690                       return node;
691                   } else if (currInflight < inflight) {
692                       // otherwise if this is the best we have found so far, record that
693                       inflight = currInflight;
694                       foundReady = node;
695                   }
696               } else if (connectionStates.isPreparingConnection(node.idString())) {
697                   foundConnecting = node;
698               } else if (canConnect(node, now)) {
699                   if (foundCanConnect == null ||
700                           this.connectionStates.lastConnectAttemptMs(foundCanConnect.idString()) >
701                                   this.connectionStates.lastConnectAttemptMs(node.idString())) {
702                       foundCanConnect = node;
703                   }
704               } else {
705                   log.trace("Removing node {} from least loaded node selection since it is neither ready " +
706                           "for sending or connecting", node);
707               }
708           }
709   
710           // We prefer established connections if possible. Otherwise, we will wait for connections
711           // which are being established before connecting to new nodes.
712           if (foundReady != null) {
713               log.trace("Found least loaded node {} with {} inflight requests", foundReady, inflight);
714               return foundReady;
715           } else if (foundConnecting != null) {
716               log.trace("Found least loaded connecting node {}", foundConnecting);
717               return foundConnecting;
718           } else if (foundCanConnect != null) {
719               log.trace("Found least loaded node {} with no active connection", foundCanConnect);
720               return foundCanConnect;
721           } else {
722               log.trace("Least loaded node selection failed to find an available node");
723               return null;
724           }
725       }

727       public static AbstractResponse parseResponse(ByteBuffer responseBuffer, RequestHeader requestHeader) {
728           try {
729               return AbstractResponse.parseResponse(responseBuffer, requestHeader);
730           } catch (BufferUnderflowException e) {
731               throw new SchemaException("Buffer underflow while parsing response for request with header " + requestHeader, e);
732           } catch (CorrelationIdMismatchException e) {
733               if (SaslClientAuthenticator.isReserved(requestHeader.correlationId())
734                   && !SaslClientAuthenticator.isReserved(e.responseCorrelationId()))
735                   throw new SchemaException("The response is unrelated to Sasl request since its correlation id is "
736                       + e.responseCorrelationId() + " and the reserved range for Sasl request is [ "
737                       + SaslClientAuthenticator.MIN_RESERVED_CORRELATION_ID + ","
738                       + SaslClientAuthenticator.MAX_RESERVED_CORRELATION_ID + "]");
739               else {
740                   throw e;
741               }
742           }

745       /**
746        * Post process disconnection of a node
747        *
748        * @param responses The list of responses to update
749        * @param nodeId Id of the node to be disconnected
750        * @param now The current time
751        * @param disconnectState The state of the disconnected channel
752        */
753       private void processDisconnection(List<ClientResponse> responses,
754                                         String nodeId,
755                                         long now,
756                                         ChannelState disconnectState) {
757           connectionStates.disconnected(nodeId, now);
758           apiVersions.remove(nodeId);
759           nodesNeedingApiVersionsFetch.remove(nodeId);
760           switch (disconnectState.state()) {
761               case AUTHENTICATION_FAILED:
762                   AuthenticationException exception = disconnectState.exception();
763                   connectionStates.authenticationFailed(nodeId, now, exception);
764                   log.error("Connection to node {} ({}) failed authentication due to: {}", nodeId,
765                       disconnectState.remoteAddress(), exception.getMessage());
766                   break;
767               case AUTHENTICATE:
768                   log.warn("Connection to node {} ({}) terminated during authentication. This may happen " +
769                       "due to any of the following reasons: (1) Authentication failed due to invalid " +
770                       "credentials with brokers older than 1.0.0, (2) Firewall blocking Kafka TLS " +
771                       "traffic (eg it may only allow HTTPS traffic), (3) Transient network issue.",
772                       nodeId, disconnectState.remoteAddress());
773                   break;
774               case NOT_CONNECTED:
775                   log.warn("Connection to node {} ({}) could not be established. Broker may not be available.", nodeId, disconnectState.remoteAddress());
776                   break;
777               default:
778                   break; // Disconnections in other states are logged at debug level in Selector
779           }
780   
781           cancelInFlightRequests(nodeId, now, responses);
782           metadataUpdater.handleServerDisconnect(now, nodeId, Optional.ofNullable(disconnectState.exception()));
783       }

786        * Iterate over all the inflight requests and expire any requests that have exceeded the configured requestTimeout.
787        * The connection to the node associated with the request will be terminated and will be treated as a disconnection.
788        *
789        * @param responses The list of responses to update

798               processDisconnection(responses, nodeId, now, ChannelState.LOCAL_CLOSE);
799           }

845       /**
846        * If a response from a node includes a non-zero throttle delay and client-side throttling has been enabled for
847        * the connection to the node, throttle the connection for the specified delay.
848        *
849        * @param response the response
850        * @param apiVersion the API version of the response
851        * @param nodeId the id of the node
852        * @param now The current time
853        */
854       private void maybeThrottle(AbstractResponse response, short apiVersion, String nodeId, long now) {
855           int throttleTimeMs = response.throttleTimeMs();
856           if (throttleTimeMs > 0 && response.shouldClientThrottle(apiVersion)) {
857               connectionStates.throttle(nodeId, now + throttleTimeMs);
858               log.trace("Connection to node {} is throttled for {} ms until timestamp {}", nodeId, throttleTimeMs,
859                         now + throttleTimeMs);
860           }
861       }

863       /**
864        * Handle any completed receives and update the response list with the responses received.
865        *
866        * @param responses The list of responses to update
867        * @param now The current time
868        */
869       private void handleCompletedReceives(List<ClientResponse> responses, long now) {
870           for (NetworkReceive receive : this.selector.completedReceives()) {
871               String source = receive.source();
872               InFlightRequest req = inFlightRequests.completeNext(source);
873   
874               AbstractResponse response = parseResponse(receive.payload(), req.header);
875               if (throttleTimeSensor != null)
876                   throttleTimeSensor.record(response.throttleTimeMs(), now);
877   
878               if (log.isDebugEnabled()) {
879                   log.debug("Received {} response from node {} for request with header {}: {}",
880                       req.header.apiKey(), req.destination, req.header, response);
881               }
882   
883               // If the received response includes a throttle delay, throttle the connection.
884               maybeThrottle(response, req.header.apiVersion(), req.destination, now);
885               if (req.isInternalRequest && response instanceof MetadataResponse)
886                   metadataUpdater.handleSuccessfulResponse(req.header, now, (MetadataResponse) response);
887               else if (req.isInternalRequest && response instanceof ApiVersionsResponse)
888                   handleApiVersionsResponse(responses, req, now, (ApiVersionsResponse) response);
889               else
890                   responses.add(req.completed(response, now));
891           }
892       }

894       private void handleApiVersionsResponse(List<ClientResponse> responses,
895                                              InFlightRequest req, long now, ApiVersionsResponse apiVersionsResponse) {
896           final String node = req.destination;
897           if (apiVersionsResponse.data().errorCode() != Errors.NONE.code()) {
898               if (req.request.version() == 0 || apiVersionsResponse.data().errorCode() != Errors.UNSUPPORTED_VERSION.code()) {
899                   log.warn("Received error {} from node {} when making an ApiVersionsRequest with correlation id {}. Disconnecting.",
900                           Errors.forCode(apiVersionsResponse.data().errorCode()), node, req.header.correlationId());
901                   this.selector.close(node);
902                   processDisconnection(responses, node, now, ChannelState.LOCAL_CLOSE);
903               } else {
904                   // Starting from Apache Kafka 2.4, ApiKeys field is populated with the supported versions of
905                   // the ApiVersionsRequest when an UNSUPPORTED_VERSION error is returned.
906                   // If not provided, the client falls back to version 0.
907                   short maxApiVersion = 0;
908                   if (apiVersionsResponse.data().apiKeys().size() > 0) {
909                       ApiVersion apiVersion = apiVersionsResponse.data().apiKeys().find(ApiKeys.API_VERSIONS.id);
910                       if (apiVersion != null) {
911                           maxApiVersion = apiVersion.maxVersion();
