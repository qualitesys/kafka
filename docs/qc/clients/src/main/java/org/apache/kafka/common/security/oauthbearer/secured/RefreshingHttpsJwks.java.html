41   /**
42    * Implementation of {@link HttpsJwks} that will periodically refresh the JWKS cache to reduce or
43    * even prevent HTTP/HTTPS traffic in the hot path of validation. It is assumed that it's
44    * possible to receive a JWT that contains a <code>kid</code> that points to yet-unknown JWK,
45    * thus requiring a connection to the OAuth/OIDC provider to be made. Hopefully, in practice,
46    * keys are made available for some amount of time before they're used within JWTs.
47    *
48    * This instance is created and provided to the
49    * {@link org.jose4j.keys.resolvers.HttpsJwksVerificationKeyResolver} that is used when using
50    * an HTTP-/HTTPS-based {@link org.jose4j.keys.resolvers.VerificationKeyResolver}, which is then
51    * provided to the {@link ValidatorAccessTokenValidator} to use in validating the signature of
52    * a JWT.
53    *
54    * @see org.jose4j.keys.resolvers.HttpsJwksVerificationKeyResolver
55    * @see org.jose4j.keys.resolvers.VerificationKeyResolver
56    * @see ValidatorAccessTokenValidator
57    */
58   
59   public final class RefreshingHttpsJwks implements Initable, Closeable {
60   
61       private static final Logger log = LoggerFactory.getLogger(RefreshingHttpsJwks.class);
62   
63       private static final int MISSING_KEY_ID_CACHE_MAX_ENTRIES = 16;
64   
65       static final long MISSING_KEY_ID_CACHE_IN_FLIGHT_MS = 60000;
66   
67       static final int MISSING_KEY_ID_MAX_KEY_LENGTH = 1000;
68   
69       private static final int SHUTDOWN_TIMEOUT = 10;
70   
71       private static final TimeUnit SHUTDOWN_TIME_UNIT = TimeUnit.SECONDS;
72   
73       /**
74        * {@link HttpsJwks} does the actual work of contacting the OAuth/OIDC endpoint to get the
75        * JWKS. In some cases, the call to {@link HttpsJwks#getJsonWebKeys()} will trigger a call
76        * to {@link HttpsJwks#refresh()} which will block the current thread in network I/O. We cache
77        * the JWKS ourselves (see {@link #jsonWebKeys}) to avoid the network I/O.
78        *
79        * We want to be very careful where we use the {@link HttpsJwks} instance so that we don't
80        * perform any operation (directly or indirectly) that could cause blocking. This is because
81        * the JWKS logic is part of the larger authentication logic which operates on Kafka's network
82        * thread. It's OK to execute {@link HttpsJwks#getJsonWebKeys()} (which calls
83        * {@link HttpsJwks#refresh()}) from within {@link #init()} as that method is called only at
84        * startup, and we can afford the blocking hit there.
85        */
86   
87       private final HttpsJwks httpsJwks;
88   
89       private final ScheduledExecutorService executorService;
90   
91       private final Time time;
92   
93       private final long refreshMs;
94   
95       private final long refreshRetryBackoffMs;
96   
97       private final long refreshRetryBackoffMaxMs;
98   
99       /**
100        * Protects {@link #missingKeyIds} and {@link #jsonWebKeys}.
101        */
102   
103       private final ReadWriteLock refreshLock = new ReentrantReadWriteLock();
104   
105       private final Map<String, Long> missingKeyIds;
106   
107       /**
108        * Flag to prevent concurrent refresh invocations.
109        */
110   
111       private final AtomicBoolean refreshInProgressFlag = new AtomicBoolean(false);
112   
113       /**
114        * As mentioned in the comments for {@link #httpsJwks}, we cache the JWKS ourselves so that
115        * we can return the list immediately without any network I/O. They are only cached within
116        * calls to {@link #refresh()}.
117        */
118   
119       private List<JsonWebKey> jsonWebKeys;
120   
121       private boolean isInitialized;
122   
123       /**
124        * Creates a <code>RefreshingHttpsJwks</code> that will be used by the
125        * {@link RefreshingHttpsJwksVerificationKeyResolver} to resolve new key IDs in JWTs.
126        *
127        * @param time                     {@link Time} instance
128        * @param httpsJwks                {@link HttpsJwks} instance from which to retrieve the JWKS
129        *                                 based on the OAuth/OIDC standard
130        * @param refreshMs                The number of milliseconds between refresh passes to connect
131        *                                 to the OAuth/OIDC JWKS endpoint to retrieve the latest set
132        * @param refreshRetryBackoffMs    Time for delay after initial failed attempt to retrieve JWKS
133        * @param refreshRetryBackoffMaxMs Maximum time to retrieve JWKS
134        */
135   
136       public RefreshingHttpsJwks(Time time,
137           HttpsJwks httpsJwks,
138           long refreshMs,
139           long refreshRetryBackoffMs,
140           long refreshRetryBackoffMaxMs) {
141           if (refreshMs <= 0)
142               throw new IllegalArgumentException("JWKS validation key refresh configuration value retryWaitMs value must be positive");
143   
144           this.httpsJwks = httpsJwks;
145           this.time = time;
146           this.refreshMs = refreshMs;
147           this.refreshRetryBackoffMs = refreshRetryBackoffMs;
148           this.refreshRetryBackoffMaxMs = refreshRetryBackoffMaxMs;
149           this.executorService = Executors.newSingleThreadScheduledExecutor();
150           this.missingKeyIds = new LinkedHashMap<String, Long>(MISSING_KEY_ID_CACHE_MAX_ENTRIES, .75f, true) {
151               @Override
152               protected boolean removeEldestEntry(Map.Entry<String, Long> eldest) {
153                   return this.size() > MISSING_KEY_ID_CACHE_MAX_ENTRIES;
154               }
155           };
156       }
157   
158       @Override
159       public void init() throws IOException {
160           try {
161               log.debug("init started");
162   
163               List<JsonWebKey> localJWKs;
164   
165               try {
166                   localJWKs = httpsJwks.getJsonWebKeys();
167               } catch (JoseException e) {
168                   throw new IOException("Could not refresh JWKS", e);
169               }
170   
171               try {
172                   refreshLock.writeLock().lock();
173                   jsonWebKeys = Collections.unmodifiableList(localJWKs);
174               } finally {
175                   refreshLock.writeLock().unlock();
176               }
177   
178               // Since we just grabbed the keys (which will have invoked a HttpsJwks.refresh()
179               // internally), we can delay our first invocation by refreshMs.
180               //
181               // Note: we refer to this as a _scheduled_ refresh.
182               executorService.scheduleAtFixedRate(this::refresh,
183                   refreshMs,
184                   refreshMs,
185                   TimeUnit.MILLISECONDS);
186   
187               log.info("JWKS validation key refresh thread started with a refresh interval of {} ms", refreshMs);
188           } finally {
189               isInitialized = true;
190   
191               log.debug("init completed");
192           }
193       }
194   
195       @Override
196       public void close() {
197           try {
198               log.debug("close started");
199   
200               try {
201                   log.debug("JWKS validation key refresh thread shutting down");
202                   executorService.shutdown();
203   
204                   if (!executorService.awaitTermination(SHUTDOWN_TIMEOUT, SHUTDOWN_TIME_UNIT)) {
205                       log.warn("JWKS validation key refresh thread termination did not end after {} {}",
206                           SHUTDOWN_TIMEOUT, SHUTDOWN_TIME_UNIT);
207                   }
208               } catch (InterruptedException e) {
209                   log.warn("JWKS validation key refresh thread error during close", e);
210               }
211           } finally {
212               log.debug("close completed");
213           }
214       }
215   
216       /**
217        * Our implementation avoids the blocking call within {@link HttpsJwks#refresh()} that is
218        * sometimes called internal to {@link HttpsJwks#getJsonWebKeys()}. We want to avoid any
219        * blocking I/O as this code is running in the authentication path on the Kafka network thread.
220        *
221        * The list may be stale up to {@link #refreshMs}.
222        *
223        * @return {@link List} of {@link JsonWebKey} instances
224        *
225        * @throws JoseException Thrown if a problem is encountered parsing the JSON content into JWKs
226        * @throws IOException Thrown f a problem is encountered making the HTTP request
227        */
228   
229       public List<JsonWebKey> getJsonWebKeys() throws JoseException, IOException {
230           if (!isInitialized)
231               throw new IllegalStateException("Please call init() first");
232   
233           try {
234               refreshLock.readLock().lock();
235               return jsonWebKeys;
236           } finally {
237               refreshLock.readLock().unlock();
238           }
239       }
240   
241       public String getLocation() {
242           return httpsJwks.getLocation();
243       }
244   
245       /**
246        * <p>
247        * <code>refresh</code> is an internal method that will refresh the JWKS cache and is
248        * invoked in one of two ways:
249        *
250        * <ol>
251        *     <li>Scheduled</li>
252        *     <li>Expedited</li>
253        * </ol>
254        * </p>
255        *
256        * <p>
257        * The <i>scheduled</i> refresh is scheduled in {@link #init()} and runs every
258        * {@link #refreshMs} milliseconds. An <i>expedited</i> refresh is performed when an
259        * incoming JWT refers to a key ID that isn't in our JWKS cache ({@link #jsonWebKeys})
260        * and we try to perform a refresh sooner than the next scheduled refresh.
261        * </p>
262        */
263   
264       private void refresh() {
265           if (!refreshInProgressFlag.compareAndSet(false, true)) {
266               log.debug("OAuth JWKS refresh is already in progress; ignoring concurrent refresh");
267               return;
268           }
269   
270           try {
271               log.info("OAuth JWKS refresh of {} starting", httpsJwks.getLocation());
272               Retry<List<JsonWebKey>> retry = new Retry<>(refreshRetryBackoffMs, refreshRetryBackoffMaxMs);
273               List<JsonWebKey> localJWKs = retry.execute(() -> {
274                   try {
275                       log.debug("JWKS validation key calling refresh of {} starting", httpsJwks.getLocation());
276                       // Call the *actual* refresh implementation that will more than likely issue
277                       // HTTP(S) calls over the network.
278                       httpsJwks.refresh();
279                       List<JsonWebKey> jwks = httpsJwks.getJsonWebKeys();
280                       log.debug("JWKS validation key refresh of {} complete", httpsJwks.getLocation());
281                       return jwks;
282                   } catch (Exception e) {
283                       throw new ExecutionException(e);
284                   }
285               });
286   
287               try {
288                   refreshLock.writeLock().lock();
289   
290                   for (JsonWebKey jwk : localJWKs)
291                       missingKeyIds.remove(jwk.getKeyId());
292   
293                   jsonWebKeys = Collections.unmodifiableList(localJWKs);
294               } finally {
295                   refreshLock.writeLock().unlock();
296               }
297   
298               log.info("OAuth JWKS refresh of {} complete", httpsJwks.getLocation());
299           } catch (ExecutionException e) {
300               log.warn("OAuth JWKS refresh of {} encountered an error; not updating local JWKS cache", httpsJwks.getLocation(), e);
301           } finally {
302               refreshInProgressFlag.set(false);
303           }
304       }
305   
306       /**
307        * <p>
308        * <code>maybeExpediteRefresh</code> is a public method that will trigger a refresh of
309        * the JWKS cache if all of the following conditions are met:
310        *
311        * <ul>
312        *     <li>The given <code>keyId</code> parameter is &lte; the
313        *     {@link #MISSING_KEY_ID_MAX_KEY_LENGTH}</li>
314        *     <li>The key isn't in the process of being expedited already</li>
315        * </ul>
316        *
317        * <p>
318        * This <i>expedited</i> refresh is scheduled immediately.
319        * </p>
320        *
321        * @param keyId JWT key ID
322        * @return <code>true</code> if an expedited refresh was scheduled, <code>false</code> otherwise
323        */
324   
325       public boolean maybeExpediteRefresh(String keyId) {
326           if (keyId.length() > MISSING_KEY_ID_MAX_KEY_LENGTH) {
327               // Although there's no limit on the length of the key ID, they're generally
328               // "reasonably" short. If we have a very long key ID length, we're going to assume
329               // the JWT is malformed, and we will not actually try to resolve the key.
330               //
331               // In this case, let's prevent blowing out our memory in two ways:
332               //
333               //     1. Don't try to resolve the key as the large ID will sit in our cache
334               //     2. Report the issue in the logs but include only the first N characters
335               int actualLength = keyId.length();
336               String s = keyId.substring(0, MISSING_KEY_ID_MAX_KEY_LENGTH);
337               String snippet = String.format("%s (trimmed to first %s characters out of %s total)", s, MISSING_KEY_ID_MAX_KEY_LENGTH, actualLength);
338               log.warn("Key ID {} was too long to cache", snippet);
339               return false;
340           } else {
341               try {
342                   refreshLock.writeLock().lock();
343   
344                   Long nextCheckTime = missingKeyIds.get(keyId);
345                   long currTime = time.milliseconds();
346                   log.debug("For key ID {}, nextCheckTime: {}, currTime: {}", keyId, nextCheckTime, currTime);
347   
348                   if (nextCheckTime == null || nextCheckTime <= currTime) {
349                       // If there's no entry in the missing key ID cache for the incoming key ID,
350                       // or it has expired, schedule a refresh ASAP.
351                       nextCheckTime = currTime + MISSING_KEY_ID_CACHE_IN_FLIGHT_MS;
352                       missingKeyIds.put(keyId, nextCheckTime);
353                       executorService.schedule(this::refresh, 0, TimeUnit.MILLISECONDS);
354                       return true;
355                   } else {
356                       return false;
357                   }
358               } finally {
359                   refreshLock.writeLock().unlock();
360               }
361           }
362       }
