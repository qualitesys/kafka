17   package org.apache.kafka.common;
18   
19   import java.nio.ByteBuffer;
20   import java.util.Base64;
21   
22   /**
23    * This class defines an immutable universally unique identifier (UUID). It represents a 128-bit value.
24    * More specifically, the random UUIDs generated by this class are variant 2 (Leach-Salz) version 4 UUIDs.
25    * This is the same type of UUID as the ones generated by java.util.UUID. The toString() method prints
26    * using the base64 string encoding. Likewise, the fromString method expects a base64 string encoding.
27    */
28   public class Uuid implements Comparable<Uuid> {
29   
30       /**
31        * A UUID for the metadata topic in KRaft mode. Will never be returned by the randomUuid method.
32        */
33       public static final Uuid METADATA_TOPIC_ID = new Uuid(0L, 1L);
34   
35       /**
36        * A UUID that represents a null or empty UUID. Will never be returned by the randomUuid method.
37        */
38       public static final Uuid ZERO_UUID = new Uuid(0L, 0L);
39   
40       private final long mostSignificantBits;
41       private final long leastSignificantBits;
42   
43       /**
44        * Constructs a 128-bit type 4 UUID where the first long represents the most significant 64 bits
45        * and the second long represents the least significant 64 bits.
46        */
47       public Uuid(long mostSigBits, long leastSigBits) {
48           this.mostSignificantBits = mostSigBits;
49           this.leastSignificantBits = leastSigBits;
50       }
51   
52       private static Uuid unsafeRandomUuid() {
53           java.util.UUID jUuid = java.util.UUID.randomUUID();
54           return new Uuid(jUuid.getMostSignificantBits(), jUuid.getLeastSignificantBits());
55       }
56   
57       /**
58        * Static factory to retrieve a type 4 (pseudo randomly generated) UUID.
59        *
60        * This will not generate a UUID equal to 0, 1, or one whose string representation starts with a dash ("-")
61        */
62       public static Uuid randomUuid() {
63           Uuid uuid = unsafeRandomUuid();
64           while (uuid.equals(METADATA_TOPIC_ID) || uuid.equals(ZERO_UUID) || uuid.toString().startsWith("-")) {
65               uuid = unsafeRandomUuid();
66           }
67           return uuid;
68       }
69   
70       /**
71        * Returns the most significant bits of the UUID's 128 value.
72        */
73       public long getMostSignificantBits() {
74           return this.mostSignificantBits;
75       }
76   
77       /**
78        * Returns the least significant bits of the UUID's 128 value.
79        */
80       public long getLeastSignificantBits() {
81           return this.leastSignificantBits;
82       }
83   
84       /**
85        * Returns true iff obj is another Uuid represented by the same two long values.
86        */
87       @Override
88       public boolean equals(Object obj) {
89           if ((null == obj) || (obj.getClass() != this.getClass()))
90               return false;
91           Uuid id = (Uuid) obj;
92           return this.mostSignificantBits == id.mostSignificantBits &&
93                   this.leastSignificantBits == id.leastSignificantBits;
94       }
95   
96       /**
97        * Returns a hash code for this UUID
98        */
99       @Override
100       public int hashCode() {
101           long xor = mostSignificantBits ^ leastSignificantBits;
102           return (int) (xor >> 32) ^ (int) xor;
103       }
104   
105       /**
106        * Returns a base64 string encoding of the UUID.
107        */
108       @Override
109       public String toString() {
110           return Base64.getUrlEncoder().withoutPadding().encodeToString(getBytesFromUuid());
111       }
112   
113       /**
114        * Creates a UUID based on a base64 string encoding used in the toString() method.
115        */
116       public static Uuid fromString(String str) {
117           if (str.length() > 24) {
118               throw new IllegalArgumentException("Input string with prefix `"
119                   + str.substring(0, 24) + "` is too long to be decoded as a base64 UUID");
120           }
121   
122           ByteBuffer uuidBytes = ByteBuffer.wrap(Base64.getUrlDecoder().decode(str));
123           if (uuidBytes.remaining() != 16) {
124               throw new IllegalArgumentException("Input string `" + str + "` decoded as "
125                   + uuidBytes.remaining() + " bytes, which is not equal to the expected 16 bytes "
126                   + "of a base64-encoded UUID");
127           }
128   
129           return new Uuid(uuidBytes.getLong(), uuidBytes.getLong());
130       }
131   
132       private byte[] getBytesFromUuid() {
133           // Extract bytes for uuid which is 128 bits (or 16 bytes) long.
134           ByteBuffer uuidBytes = ByteBuffer.wrap(new byte[16]);
135           uuidBytes.putLong(this.mostSignificantBits);
136           uuidBytes.putLong(this.leastSignificantBits);
137           return uuidBytes.array();
138       }
139   
140       @Override
141       public int compareTo(Uuid other) {
142           if (mostSignificantBits > other.mostSignificantBits) {
143               return 1;
144           } else if (mostSignificantBits < other.mostSignificantBits) {
145               return -1;
146           } else if (leastSignificantBits > other.leastSignificantBits) {
147               return 1;
148           } else if (leastSignificantBits < other.leastSignificantBits) {
149               return -1;
150           } else {
151               return 0;
152           }
153       }
154   }
