17   package org.apache.kafka.common.feature;
18   
19   import java.util.HashMap;
20   import java.util.Map;
21   import java.util.stream.Collectors;
22   import java.util.Objects;
23   
24   import static java.util.stream.Collectors.joining;
25   
26   /**
27    * Represents an immutable dictionary with key being feature name, and value being <VersionRangeType>.
28    * Also provides API to convert the features and their version ranges to/from a map.
29    *
30    * This class can be instantiated only using its factory functions, with the important ones being:
31    * Features.supportedFeatures(...) and Features.finalizedFeatures(...).
32    *
33    * @param <VersionRangeType> is the type of version range.
34    * @see SupportedVersionRange
35    */
36   public class Features<VersionRangeType extends BaseVersionRange> {
37       private final Map<String, VersionRangeType> features;
38   
39       /**
40        * Constructor is made private, as for readability it is preferred the caller uses one of the
41        * static factory functions for instantiation (see below).
42        *
43        * @param features   Map of feature name to a type of VersionRange.
44        */
45       private Features(Map<String, VersionRangeType> features) {
46           Objects.requireNonNull(features, "Provided features can not be null.");
47           this.features = features;
48       }
49   
50       /**
51        * @param features   Map of feature name to SupportedVersionRange.
52        *
53        * @return           Returns a new Features object representing supported features.
54        */
55       public static Features<SupportedVersionRange> supportedFeatures(Map<String, SupportedVersionRange> features) {
56           return new Features<>(features);
57       }
58   
59       public static Features<SupportedVersionRange> emptySupportedFeatures() {
60           return new Features<>(new HashMap<>());
61       }
62   
63       public Map<String, VersionRangeType> features() {
64           return features;
65       }
66   
67       public boolean empty() {
68           return features.isEmpty();
69       }
70   
71       /**
72        * @param  feature   name of the feature
73        *
74        * @return           the VersionRangeType corresponding to the feature name, or null if the
75        *                   feature is absent
76        */
77       public VersionRangeType get(String feature) {
78           return features.get(feature);
79       }
80   
81       public String toString() {
82           return String.format(
83               "Features{%s}",
84               features
85                   .entrySet()
86                   .stream()
87                   .map(entry -> String.format("(%s -> %s)", entry.getKey(), entry.getValue()))
88                   .collect(joining(", "))
89           );
90       }
91   
92       /**
93        * @return   A map representation of the underlying features. The returned value can be converted
94        *           back to Features using one of the from*FeaturesMap() APIs of this class.
95        */
96       public Map<String, Map<String, Short>> toMap() {
97           return features.entrySet().stream().collect(
98               Collectors.toMap(
99                   Map.Entry::getKey,
100                   entry -> entry.getValue().toMap()));
101       }
102   
103       /**
104        * An interface that defines behavior to convert from a Map to an object of type BaseVersionRange.
105        */
106       private interface MapToBaseVersionRangeConverter<V extends BaseVersionRange> {
107   
108           /**
109            * Convert the map representation of an object of type <V>, to an object of type <V>.
110            *
111            * @param  baseVersionRangeMap   the map representation of a BaseVersionRange object.
112            *
113            * @return                       the object of type <V>
114            */
115           V fromMap(Map<String, Short> baseVersionRangeMap);
116       }
117   
118       private static <V extends BaseVersionRange> Features<V> fromFeaturesMap(
119           Map<String, Map<String, Short>> featuresMap, MapToBaseVersionRangeConverter<V> converter) {
120           return new Features<>(featuresMap.entrySet().stream().collect(
121               Collectors.toMap(
122                   Map.Entry::getKey,
123                   entry -> converter.fromMap(entry.getValue()))));
124       }
125   
126       /**
127        * Converts from a map to Features<SupportedVersionRange>.
128        *
129        * @param featuresMap  the map representation of a Features<SupportedVersionRange> object,
130        *                     generated using the toMap() API.
131        *
132        * @return             the Features<SupportedVersionRange> object
133        */
134       public static Features<SupportedVersionRange> fromSupportedFeaturesMap(
135           Map<String, Map<String, Short>> featuresMap) {
136           return fromFeaturesMap(featuresMap, SupportedVersionRange::fromMap);
137       }
138   
139       @Override
140       public boolean equals(Object other) {
141           if (this == other) {
142               return true;
143           }
144           if (!(other instanceof Features)) {
145               return false;
146           }
147   
148           final Features that = (Features) other;
149           return Objects.equals(this.features, that.features);
150       }
151   
152       @Override
153       public int hashCode() {
154           return Objects.hash(features);
155       }
156   }
