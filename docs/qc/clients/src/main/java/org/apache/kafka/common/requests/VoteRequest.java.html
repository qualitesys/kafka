34           public Builder(VoteRequestData data) {

37           }

39           @Override
40           public VoteRequest build(short version) {
41               return new VoteRequest(data, version);
42           }
43   
44           @Override
45           public String toString() {

48       }
49   
50       private final VoteRequestData data;
51   
52       private VoteRequest(VoteRequestData data, short version) {
53           super(ApiKeys.VOTE, version);
54           this.data = data;
55       }
56   
57       @Override
58       public VoteRequestData data() {
59           return data;
60       }
61   
62       @Override
63       public AbstractResponse getErrorResponse(int throttleTimeMs, Throwable e) {
64           return new VoteResponse(new VoteResponseData()
65               .setErrorCode(Errors.forException(e).code()));
66       }
67   
68       public static VoteRequest parse(ByteBuffer buffer, short version) {
69           return new VoteRequest(new VoteRequestData(new ByteBufferAccessor(buffer), version), version);
70       }

85       public static VoteRequestData singletonRequest(TopicPartition topicPartition,
86                                                      String clusterId,
87                                                      int candidateEpoch,
88                                                      int candidateId,
89                                                      int lastEpoch,
90                                                      long lastEpochEndOffset) {
91           return new VoteRequestData()
92                      .setClusterId(clusterId)
93                      .setTopics(Collections.singletonList(
94                          new VoteRequestData.TopicData()
95                              .setTopicName(topicPartition.topic())
96                              .setPartitions(Collections.singletonList(
97                                  new VoteRequestData.PartitionData()
98                                      .setPartitionIndex(topicPartition.partition())
99                                      .setCandidateEpoch(candidateEpoch)
100                                      .setCandidateId(candidateId)
101                                      .setLastOffsetEpoch(lastEpoch)

103                              )));
104       }
