23   import java.util.Timer;

29   /**
30    * A base class for running a Unix command.
31    *
32    * <code>Shell</code> can be used to run unix commands like <code>du</code> or
33    * <code>df</code>.
34    */
35   abstract public class Shell {
36   
37       private static final Logger LOG = LoggerFactory.getLogger(Shell.class);
38   
39       /** Return an array containing the command name and its parameters */
40       protected abstract String[] execString();
41   
42       /** Parse the execution result */
43       protected abstract void parseExecResult(BufferedReader lines) throws IOException;
44   
45       private final long timeout;
46   
47       private int exitCode;
48       private Process process; // sub process used to execute the command
49   
50       /* If or not script finished executing */
51       private volatile AtomicBoolean completed;
52   
53       /**
54        * @param timeout Specifies the time in milliseconds, after which the command will be killed. -1 means no timeout.
55        */
56       public Shell(long timeout) {
57           this.timeout = timeout;
58       }
59   
60       /** get the exit code
61        * @return the exit code of the process
62        */
63       public int exitCode() {
64           return exitCode;
65       }
66   
67       /** get the current sub-process executing the given command
68        * @return process executing the command
69        */
70       public Process process() {
71           return process;
72       }
73   
74       protected void run() throws IOException {
75           exitCode = 0; // reset for next run
76           runCommand();
77       }
78   
79       /** Run a command */
80       private void runCommand() throws IOException {
81           ProcessBuilder builder = new ProcessBuilder(execString());
82           Timer timeoutTimer = null;
83           completed = new AtomicBoolean(false);
84   
85           process = builder.start();
86           if (timeout > -1) {
87               timeoutTimer = new Timer();
88               //One time scheduling.
89               timeoutTimer.schedule(new ShellTimeoutTimerTask(this), timeout);
90           }
91           final BufferedReader errReader = new BufferedReader(
92               new InputStreamReader(process.getErrorStream(), StandardCharsets.UTF_8));
93           BufferedReader inReader = new BufferedReader(
94               new InputStreamReader(process.getInputStream(), StandardCharsets.UTF_8));
95           final StringBuffer errMsg = new StringBuffer();
96   
97           // read error and input streams as this would free up the buffers
98           // free the error stream buffer
99           Thread errThread = KafkaThread.nonDaemon("kafka-shell-thread", new Runnable() {
100               @Override
101               public void run() {
102                   try {
103                       String line = errReader.readLine();
104                       while ((line != null) && !Thread.currentThread().isInterrupted()) {
105                           errMsg.append(line);
106                           errMsg.append(System.getProperty("line.separator"));
107                           line = errReader.readLine();
108                       }
109                   } catch (IOException ioe) {
110                       LOG.warn("Error reading the error stream", ioe);
111                   }
112               }
113           });
114           errThread.start();
115   
116           try {
117               parseExecResult(inReader); // parse the output
118               // wait for the process to finish and check the exit code
119               exitCode = process.waitFor();
120               try {
121                   // make sure that the error thread exits
122                   errThread.join();
123               } catch (InterruptedException ie) {
124                   LOG.warn("Interrupted while reading the error stream", ie);
125               }
126               completed.set(true);
127               //the timeout thread handling
128               //taken care in finally block
129               if (exitCode != 0) {
130                   throw new ExitCodeException(exitCode, errMsg.toString());
131               }
132           } catch (InterruptedException ie) {
133               throw new IOException(ie.toString());
134           } finally {
135               if (timeoutTimer != null)
136                   timeoutTimer.cancel();
137   
138               // close the input stream
139               try {
140                   inReader.close();
141               } catch (IOException ioe) {
142                   LOG.warn("Error while closing the input stream", ioe);
143               }
144               if (!completed.get())
145                   errThread.interrupt();
146   
147               try {
148                   errReader.close();
149               } catch (IOException ioe) {
150                   LOG.warn("Error while closing the error stream", ioe);
151               }
152   
153               process.destroy();
154           }
155       }
156   
157   
158       /**
159        * This is an IOException with exit code added.
160        */
161       @SuppressWarnings("serial")
162       public static class ExitCodeException extends IOException {
163           int exitCode;
164   
165           public ExitCodeException(int exitCode, String message) {
166               super(message);
167               this.exitCode = exitCode;
168           }
169   
170           public int getExitCode() {
171               return exitCode;
172           }
173       }
174   
175       /**
176        * A simple shell command executor.
177        *
178        * <code>ShellCommandExecutor</code>should be used in cases where the output
179        * of the command needs no explicit parsing and where the command, working
180        * directory and the environment remains unchanged. The output of the command
181        * is stored as-is and is expected to be small.
182        */
183       public static class ShellCommandExecutor extends Shell {
184   
185           private final String[] command;
186           private StringBuffer output;
187   
188           /**
189            * Create a new instance of the ShellCommandExecutor to execute a command.
190            *
191            * @param execString The command to execute with arguments
192            * @param timeout Specifies the time in milliseconds, after which the
193            *                command will be killed. -1 means no timeout.
194            */
195   
196           public ShellCommandExecutor(String[] execString, long timeout) {
197               super(timeout);
198               command = execString.clone();
199           }
200   
201   
202           /** Execute the shell command. */
203           public void execute() throws IOException {
204               this.run();
205           }
206   
207           protected String[] execString() {
208               return command;
209           }
210   
211           protected void parseExecResult(BufferedReader reader) throws IOException {
212               output = new StringBuffer();
213               char[] buf = new char[512];
214               int nRead;
215               while ((nRead = reader.read(buf, 0, buf.length)) > 0) {
216                   output.append(buf, 0, nRead);
217               }
218           }
219   
220           /** Get the output of the shell command.*/
221           public String output() {
222               return (output == null) ? "" : output.toString();
223           }
224   
225           /**
226            * Returns the commands of this instance.
227            * Arguments with spaces in are presented with quotes round; other
228            * arguments are presented raw
229            *
230            * @return a string representation of the object.
231            */
232           public String toString() {
233               StringBuilder builder = new StringBuilder();
234               String[] args = execString();
235               for (String s : args) {
236                   if (s.indexOf(' ') >= 0) {
237                       builder.append('"').append(s).append('"');
238                   } else {
239                       builder.append(s);
240                   }
241                   builder.append(' ');
242               }
243               return builder.toString();
244           }
245       }
246   
247       /**
248        * Static method to execute a shell command.
249        * Covers most of the simple cases without requiring the user to implement
250        * the <code>Shell</code> interface.
251        * @param cmd shell command to execute.
252        * @return the output of the executed command.
253        */
254       public static String execCommand(String... cmd) throws IOException {
255           return execCommand(cmd, -1);
256       }
257   
258       /**
259        * Static method to execute a shell command.
260        * Covers most of the simple cases without requiring the user to implement
261        * the <code>Shell</code> interface.
262        * @param cmd shell command to execute.
263        * @param timeout time in milliseconds after which script should be killed. -1 means no timeout.
264        * @return the output of the executed command.
265        */
266       public static String execCommand(String[] cmd, long timeout) throws IOException {
267           ShellCommandExecutor exec = new ShellCommandExecutor(cmd, timeout);
268           exec.execute();
269           return exec.output();
270       }
271   
272       /**
273        * Timer which is used to timeout scripts spawned off by shell.
274        */
275       private static class ShellTimeoutTimerTask extends TimerTask {
276   
277           private final Shell shell;
278   
279           public ShellTimeoutTimerTask(Shell shell) {
280               this.shell = shell;
281           }
282   
283           @Override
284           public void run() {
285               Process p = shell.process();
286               try {
287                   p.exitValue();
288               } catch (Exception e) {
289                   //Process has not terminated.
290                   //So check if it has completed
291                   //if not just destroy it.
292                   if (p != null && !shell.completed.get()) {
293                       p.destroy();
294                   }
295               }
296           }
297       }
