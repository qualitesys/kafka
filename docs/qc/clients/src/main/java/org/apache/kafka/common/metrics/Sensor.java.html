79       public enum RecordingLevel {
80           INFO(0, "INFO"), DEBUG(1, "DEBUG"), TRACE(2, "TRACE");
81   
82           private static final RecordingLevel[] ID_TO_TYPE;
83           private static final int MIN_RECORDING_LEVEL_KEY = 0;
84           public static final int MAX_RECORDING_LEVEL_KEY;
85   
86           static {
87               int maxRL = -1;
88               for (RecordingLevel level : RecordingLevel.values()) {
89                   maxRL = Math.max(maxRL, level.id);
90               }
91               RecordingLevel[] idToName = new RecordingLevel[maxRL + 1];
92               for (RecordingLevel level : RecordingLevel.values()) {
93                   idToName[level.id] = level;
94               }
95               ID_TO_TYPE = idToName;
96               MAX_RECORDING_LEVEL_KEY = maxRL;
97           }

103           public final short id;

110           public static RecordingLevel forId(int id) {
111               if (id < MIN_RECORDING_LEVEL_KEY || id > MAX_RECORDING_LEVEL_KEY)
112                   throw new IllegalArgumentException(String.format("Unexpected RecordLevel id `%d`, it should be between `%d` " +
113                       "and `%d` (inclusive)", id, MIN_RECORDING_LEVEL_KEY, MAX_RECORDING_LEVEL_KEY));
114               return ID_TO_TYPE[id];
115           }

117           /** Case insensitive lookup by protocol name */
118           public static RecordingLevel forName(String name) {
119               return RecordingLevel.valueOf(name.toUpperCase(Locale.ROOT));
120           }

122           public boolean shouldRecord(final int configId) {
123               if (configId == INFO.id) {
124                   return this.id == INFO.id;
125               } else if (configId == DEBUG.id) {
126                   return this.id == INFO.id || this.id == DEBUG.id;
127               } else if (configId == TRACE.id) {
128                   return true;
129               } else {
130                   throw new IllegalStateException("Did not recognize recording level " + configId);
131               }
132           }
133       }

142           this.parents = parents == null ? new Sensor[0] : parents;
143           this.metrics = new LinkedHashMap<>();
144           this.stats = new ArrayList<>();
145           this.config = config;
146           this.time = time;
147           this.inactiveSensorExpirationTimeMs = TimeUnit.MILLISECONDS.convert(inactiveSensorExpirationTimeSeconds, TimeUnit.SECONDS);
148           this.lastRecordTime = time.milliseconds();
149           this.recordingLevel = recordingLevel;
150           this.metricLock = new Object();
151           checkForest(new HashSet<>());
152       }

154       /* Validate that this sensor doesn't end up referencing itself */
155       private void checkForest(Set<Sensor> sensors) {
156           if (!sensors.add(this))
157               throw new IllegalArgumentException("Circular dependency in sensors: " + name() + " is its own parent.");
158           for (Sensor parent : parents)
159               parent.checkForest(sensors);
160       }

230       private void recordInternal(double value, long timeMs, boolean checkQuotas) {
231           this.lastRecordTime = timeMs;
232           synchronized (this) {
233               synchronized (metricLock()) {
234                   // increment all the stats
235                   for (StatAndConfig statAndConfig : this.stats) {
236                       statAndConfig.stat.record(statAndConfig.config(), value, timeMs);
237                   }
238               }
239               if (checkQuotas)
240                   checkQuotas(timeMs);
241           }
242           for (Sensor parent : parents)
243               parent.record(value, timeMs, checkQuotas);
244       }

247        * Check if we have violated our quota for any metric that has a configured quota
248        */
249       public void checkQuotas() {
250           checkQuotas(time.milliseconds());
251       }

253       public void checkQuotas(long timeMs) {
254           for (KafkaMetric metric : this.metrics.values()) {
255               MetricConfig config = metric.config();
256               if (config != null) {
257                   Quota quota = config.quota();
258                   if (quota != null) {
259                       double value = metric.measurableValue(timeMs);
260                       if (metric.measurable() instanceof TokenBucket) {
261                           if (value < 0) {
262                               throw new QuotaViolationException(metric, value, quota.bound());
263                           }
264                       } else {
265                           if (!quota.acceptable(value)) {
266                               throw new QuotaViolationException(metric, value, quota.bound());
267                           }
268                       }
269                   }
270               }
271           }
272       }

283       /**
284        * Register a compound statistic with this sensor which yields multiple measurable quantities (like a histogram)
285        * @param stat The stat to register
286        * @param config The configuration for this stat. If null then the stat will use the default configuration for this
287        *        sensor.
288        * @return true if stat is added to sensor, false if sensor is expired
289        */
290       public synchronized boolean add(CompoundStat stat, MetricConfig config) {
291           if (hasExpired())
292               return false;
293   
294           final MetricConfig statConfig = config == null ? this.config : config;
295           stats.add(new StatAndConfig(Objects.requireNonNull(stat), () -> statConfig));
296           Object lock = metricLock();
297           for (NamedMeasurable m : stat.stats()) {
298               final KafkaMetric metric = new KafkaMetric(lock, m.name(), m.stat(), statConfig, time);
299               if (!metrics.containsKey(metric.metricName())) {
300                   KafkaMetric existingMetric = registry.registerMetric(metric);
301                   if (existingMetric != null) {
302                       throw new IllegalArgumentException("A metric named '" + metric.metricName() + "' already exists, can't register another one.");
303                   }
304                   metrics.put(metric.metricName(), metric);
305               }
306           }
307           return true;
308       }

320       /**
321        * Register a metric with this sensor
322        *
323        * @param metricName The name of the metric
324        * @param stat       The statistic to keep
325        * @param config     A special configuration for this metric. If null use the sensor default configuration.
326        * @return true if metric is added to sensor, false if sensor is expired
327        */
328       public synchronized boolean add(final MetricName metricName, final MeasurableStat stat, final MetricConfig config) {
329           if (hasExpired()) {
330               return false;
331           } else if (metrics.containsKey(metricName)) {
332               return true;
333           } else {
334               final MetricConfig statConfig = config == null ? this.config : config;
335               final KafkaMetric metric = new KafkaMetric(
336                   metricLock(),
337                   Objects.requireNonNull(metricName),
338                   Objects.requireNonNull(stat),
339                   statConfig,
340                   time
341               );
342               KafkaMetric existingMetric = registry.registerMetric(metric);
343               if (existingMetric != null) {
344                   throw new IllegalArgumentException("A metric named '" + metricName + "' already exists, can't register another one.");
345               }
346               metrics.put(metric.metricName(), metric);
347               stats.add(new StatAndConfig(Objects.requireNonNull(stat), metric::config));
348               return true;
349           }
350       }

373       /**
374        * KafkaMetrics of sensors which use SampledStat should be synchronized on the same lock
375        * for sensor record and metric value read to allow concurrent reads and updates. For simplicity,
376        * all sensors are synchronized on this object.
377        * <p>
378        * Sensor object is not used as a lock for reading metric value since metrics reporter is
379        * invoked while holding Sensor and Metrics locks to report addition and removal of metrics
380        * and synchronized reporters may deadlock if Sensor lock is used for reading metrics values.
381        * Note that Sensor object itself is used as a lock to protect the access to stats and metrics
382        * while recording metric values, adding and deleting sensors.
383        * </p><p>
384        * Locking order (assume all MetricsReporter methods may be synchronized):
385        * <ul>
386        *   <li>Sensor#add: Sensor -> Metrics -> MetricsReporter</li>
387        *   <li>Metrics#removeSensor: Sensor -> Metrics -> MetricsReporter</li>
388        *   <li>KafkaMetric#metricValue: MetricsReporter -> Sensor#metricLock</li>
389        *   <li>Sensor#record: Sensor -> Sensor#metricLock</li>
390        * </ul>
391        * </p>
392        */
393       private Object metricLock() {
394           return metricLock;
395       }
