18   package org.apache.kafka.server.authorizer;
19   
20   import java.util.Objects;
21   import org.apache.kafka.common.acl.AclOperation;
22   import org.apache.kafka.common.annotation.InterfaceStability;
23   import org.apache.kafka.common.resource.ResourcePattern;
24   
25   @InterfaceStability.Evolving
26   public class Action {
27   
28       private final ResourcePattern resourcePattern;
29       private final AclOperation operation;
30       private final int resourceReferenceCount;
31       private final boolean logIfAllowed;
32       private final boolean logIfDenied;
33   
34       /**
35        * @param operation non-null operation being performed
36        * @param resourcePattern non-null resource pattern on which this action is being performed
37        */
38       public Action(AclOperation operation,
39                     ResourcePattern resourcePattern,
40                     int resourceReferenceCount,
41                     boolean logIfAllowed,
42                     boolean logIfDenied) {
43           this.operation = Objects.requireNonNull(operation, "operation can't be null");
44           this.resourcePattern = Objects.requireNonNull(resourcePattern, "resourcePattern can't be null");
45           this.logIfAllowed = logIfAllowed;
46           this.logIfDenied = logIfDenied;
47           this.resourceReferenceCount = resourceReferenceCount;
48       }
49   
50       /**
51        * @return a non-null resource pattern on which this action is being performed
52        */
53       public ResourcePattern resourcePattern() {
54           return resourcePattern;
55       }
56   
57       /**
58        *
59        * @return a non-null operation being performed
60        */
61       public AclOperation operation() {
62           return operation;
63       }
64   
65       /**
66        * Indicates if audit logs tracking ALLOWED access should include this action if result is
67        * ALLOWED. The flag is true if access to a resource is granted while processing the request as a
68        * result of this authorization. The flag is false only for requests used to describe access where
69        * no operation on the resource is actually performed based on the authorization result.
70        */
71       public boolean logIfAllowed() {
72           return logIfAllowed;
73       }
74   
75       /**
76        * Indicates if audit logs tracking DENIED access should include this action if result is
77        * DENIED. The flag is true if access to a resource was explicitly requested and request
78        * is denied as a result of this authorization request. The flag is false if request was
79        * filtering out authorized resources (e.g. to subscribe to regex pattern). The flag is also
80        * false if this is an optional authorization where an alternative resource authorization is
81        * applied if this fails (e.g. Cluster:Create which is subsequently overridden by Topic:Create).
82        */
83       public boolean logIfDenied() {
84           return logIfDenied;
85       }
86   
87       /**
88        * Number of times the resource being authorized is referenced within the request. For example, a single
89        * request may reference `n` topic partitions of the same topic. Brokers will authorize the topic once
90        * with `resourceReferenceCount=n`. Authorizers may include the count in audit logs.
91        */
92       public int resourceReferenceCount() {
93           return resourceReferenceCount;
94       }
95   
96       @Override
97       public boolean equals(Object o) {
98           if (this == o) {
99               return true;
100           }
101           if (!(o instanceof Action)) {
102               return false;
103           }
104   
105           Action that = (Action) o;
106           return Objects.equals(this.resourcePattern, that.resourcePattern) &&
107               Objects.equals(this.operation, that.operation) &&
108               this.resourceReferenceCount == that.resourceReferenceCount &&
109               this.logIfAllowed == that.logIfAllowed &&
110               this.logIfDenied == that.logIfDenied;
111   
112       }
113   
114       @Override
115       public int hashCode() {
116           return Objects.hash(resourcePattern, operation, resourceReferenceCount, logIfAllowed, logIfDenied);
117       }
118   
119       @Override
120       public String toString() {
121           return "Action(" +
122               "resourcePattern='" + resourcePattern + '\'' +
123               ", operation='" + operation + '\'' +
124               ", resourceReferenceCount='" + resourceReferenceCount + '\'' +
125               ", logIfAllowed='" + logIfAllowed + '\'' +
126               ", logIfDenied='" + logIfDenied + '\'' +
127               ')';
128       }
129   }
