17   package org.apache.kafka.clients.consumer.internals;
18   
19   import org.apache.kafka.common.errors.RetriableException;
20   import org.apache.kafka.common.protocol.Errors;
21   import org.apache.kafka.common.utils.Timer;
22   
23   import java.util.concurrent.CountDownLatch;
24   import java.util.concurrent.TimeUnit;
25   import java.util.concurrent.ConcurrentLinkedQueue;
26   import java.util.concurrent.atomic.AtomicReference;
27   
28   /**
29    * Result of an asynchronous request from {@link ConsumerNetworkClient}. Use {@link ConsumerNetworkClient#poll(Timer)}
30    * (and variants) to finish a request future. Use {@link #isDone()} to check if the future is complete, and
31    * {@link #succeeded()} to check if the request completed successfully. Typical usage might look like this:
32    *
33    * <pre>
34    *     RequestFuture<ClientResponse> future = client.send(api, request);
35    *     client.poll(future);
36    *
37    *     if (future.succeeded()) {
38    *         ClientResponse response = future.value();
39    *         // Handle response
40    *     } else {
41    *         throw future.exception();
42    *     }
43    * </pre>
44    *
45    * @param <T> Return type of the result (Can be Void if there is no response)
46    */
47   public class RequestFuture<T> implements ConsumerNetworkClient.PollCondition {
48   
49       private static final Object INCOMPLETE_SENTINEL = new Object();
50       private final AtomicReference<Object> result = new AtomicReference<>(INCOMPLETE_SENTINEL);
51       private final ConcurrentLinkedQueue<RequestFutureListener<T>> listeners = new ConcurrentLinkedQueue<>();
52       private final CountDownLatch completedLatch = new CountDownLatch(1);
53   
54       /**
55        * Check whether the response is ready to be handled
56        * @return true if the response is ready, false otherwise
57        */
58       public boolean isDone() {
59           return result.get() != INCOMPLETE_SENTINEL;
60       }
61   
62       public boolean awaitDone(long timeout, TimeUnit unit) throws InterruptedException {
63           return completedLatch.await(timeout, unit);
64       }
65   
66       /**
67        * Get the value corresponding to this request (only available if the request succeeded)
68        * @return the value set in {@link #complete(Object)}
69        * @throws IllegalStateException if the future is not complete or failed
70        */
71       @SuppressWarnings("unchecked")
72       public T value() {
73           if (!succeeded())
74               throw new IllegalStateException("Attempt to retrieve value from future which hasn't successfully completed");
75           return (T) result.get();
76       }
77   
78       /**
79        * Check if the request succeeded;
80        * @return true if the request completed and was successful
81        */
82       public boolean succeeded() {
83           return isDone() && !failed();
84       }
85   
86       /**
87        * Check if the request failed.
88        * @return true if the request completed with a failure
89        */
90       public boolean failed() {
91           return result.get() instanceof RuntimeException;
92       }
93   
94       /**
95        * Check if the request is retriable (convenience method for checking if
96        * the exception is an instance of {@link RetriableException}.
97        * @return true if it is retriable, false otherwise
98        * @throws IllegalStateException if the future is not complete or completed successfully
99        */
100       public boolean isRetriable() {
101           return exception() instanceof RetriableException;
102       }
103   
104       /**
105        * Get the exception from a failed result (only available if the request failed)
106        * @return the exception set in {@link #raise(RuntimeException)}
107        * @throws IllegalStateException if the future is not complete or completed successfully
108        */
109       public RuntimeException exception() {
110           if (!failed())
111               throw new IllegalStateException("Attempt to retrieve exception from future which hasn't failed");
112           return (RuntimeException) result.get();
113       }
114   
115       /**
116        * Complete the request successfully. After this call, {@link #succeeded()} will return true
117        * and the value can be obtained through {@link #value()}.
118        * @param value corresponding value (or null if there is none)
119        * @throws IllegalStateException if the future has already been completed
120        * @throws IllegalArgumentException if the argument is an instance of {@link RuntimeException}
121        */
122       public void complete(T value) {
123           try {
124               if (value instanceof RuntimeException)
125                   throw new IllegalArgumentException("The argument to complete can not be an instance of RuntimeException");
126   
127               if (!result.compareAndSet(INCOMPLETE_SENTINEL, value))
128                   throw new IllegalStateException("Invalid attempt to complete a request future which is already complete");
129               fireSuccess();
130           } finally {
131               completedLatch.countDown();
132           }
133       }
134   
135       /**
136        * Raise an exception. The request will be marked as failed, and the caller can either
137        * handle the exception or throw it.
138        * @param e corresponding exception to be passed to caller
139        * @throws IllegalStateException if the future has already been completed
140        */
141       public void raise(RuntimeException e) {
142           try {
143               if (e == null)
144                   throw new IllegalArgumentException("The exception passed to raise must not be null");
145   
146               if (!result.compareAndSet(INCOMPLETE_SENTINEL, e))
147                   throw new IllegalStateException("Invalid attempt to complete a request future which is already complete");
148   
149               fireFailure();
150           } finally {
151               completedLatch.countDown();
152           }
153       }
154   
155       /**
156        * Raise an error. The request will be marked as failed.
157        * @param error corresponding error to be passed to caller
158        */
159       public void raise(Errors error) {
160           raise(error.exception());
161       }
162   
163       private void fireSuccess() {
164           T value = value();
165           while (true) {
166               RequestFutureListener<T> listener = listeners.poll();
167               if (listener == null)
168                   break;
169               listener.onSuccess(value);
170           }
171       }
172   
173       private void fireFailure() {
174           RuntimeException exception = exception();
175           while (true) {
176               RequestFutureListener<T> listener = listeners.poll();
177               if (listener == null)
178                   break;
179               listener.onFailure(exception);
180           }
181       }
182   
183       /**
184        * Add a listener which will be notified when the future completes
185        * @param listener non-null listener to add
186        */
187       public void addListener(RequestFutureListener<T> listener) {
188           this.listeners.add(listener);
189           if (failed())
190               fireFailure();
191           else if (succeeded())
192               fireSuccess();
193       }
194   
195       /**
196        * Convert from a request future of one type to another type
197        * @param adapter The adapter which does the conversion
198        * @param <S> The type of the future adapted to
199        * @return The new future
200        */
201       public <S> RequestFuture<S> compose(final RequestFutureAdapter<T, S> adapter) {
202           final RequestFuture<S> adapted = new RequestFuture<>();
203           addListener(new RequestFutureListener<T>() {
204               @Override
205               public void onSuccess(T value) {
206                   adapter.onSuccess(value, adapted);
207               }
208   
209               @Override
210               public void onFailure(RuntimeException e) {
211                   adapter.onFailure(e, adapted);
212               }
213           });
214           return adapted;
215       }
216   
217       public void chain(final RequestFuture<T> future) {
218           addListener(new RequestFutureListener<T>() {
219               @Override
220               public void onSuccess(T value) {
221                   future.complete(value);
222               }
223   
224               @Override
225               public void onFailure(RuntimeException e) {
226                   future.raise(e);
227               }
228           });
229       }
230   
231       public static <T> RequestFuture<T> failure(RuntimeException e) {
232           RequestFuture<T> future = new RequestFuture<>();
233           future.raise(e);
234           return future;
235       }
236   
237       public static RequestFuture<Void> voidSuccess() {
238           RequestFuture<Void> future = new RequestFuture<>();
239           future.complete(null);
240           return future;
241       }
242   
243       public static <T> RequestFuture<T> coordinatorNotAvailable() {
244           return failure(Errors.COORDINATOR_NOT_AVAILABLE.exception());
245       }
246   
247       public static <T> RequestFuture<T> noBrokersAvailable() {
248           return failure(new NoAvailableBrokersException());
249       }
250   
251       @Override
252       public boolean shouldBlock() {
253           return !isDone();
254       }
255   }
