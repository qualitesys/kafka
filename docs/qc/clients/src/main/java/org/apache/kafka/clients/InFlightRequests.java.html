17   package org.apache.kafka.clients;
18   
19   import java.util.ArrayDeque;
20   import java.util.ArrayList;
21   import java.util.Collections;
22   import java.util.Deque;
23   import java.util.HashMap;
24   import java.util.List;
25   import java.util.Map;
26   import java.util.concurrent.atomic.AtomicInteger;
27   
28   /**
29    * The set of requests which have been sent or are being sent but haven't yet received a response
30    */
31   final class InFlightRequests {
32   
33       private final int maxInFlightRequestsPerConnection;
34       private final Map<String, Deque<NetworkClient.InFlightRequest>> requests = new HashMap<>();
35       /** Thread safe total number of in flight requests. */
36       private final AtomicInteger inFlightRequestCount = new AtomicInteger(0);
37   
38       public InFlightRequests(int maxInFlightRequestsPerConnection) {
39           this.maxInFlightRequestsPerConnection = maxInFlightRequestsPerConnection;
40       }
41   
42       /**
43        * Add the given request to the queue for the connection it was directed to
44        */
45       public void add(NetworkClient.InFlightRequest request) {
46           String destination = request.destination;
47           Deque<NetworkClient.InFlightRequest> reqs = this.requests.get(destination);
48           if (reqs == null) {
49               reqs = new ArrayDeque<>();
50               this.requests.put(destination, reqs);
51           }
52           reqs.addFirst(request);
53           inFlightRequestCount.incrementAndGet();
54       }
55   
56       /**
57        * Get the request queue for the given node
58        */
59       private Deque<NetworkClient.InFlightRequest> requestQueue(String node) {
60           Deque<NetworkClient.InFlightRequest> reqs = requests.get(node);
61           if (reqs == null || reqs.isEmpty())
62               throw new IllegalStateException("There are no in-flight requests for node " + node);
63           return reqs;
64       }
65   
66       /**
67        * Get the oldest request (the one that will be completed next) for the given node
68        */
69       public NetworkClient.InFlightRequest completeNext(String node) {
70           NetworkClient.InFlightRequest inFlightRequest = requestQueue(node).pollLast();
71           inFlightRequestCount.decrementAndGet();
72           return inFlightRequest;
73       }
74   
75       /**
76        * Get the last request we sent to the given node (but don't remove it from the queue)
77        * @param node The node id
78        */
79       public NetworkClient.InFlightRequest lastSent(String node) {
80           return requestQueue(node).peekFirst();
81       }
82   
83       /**
84        * Complete the last request that was sent to a particular node.
85        * @param node The node the request was sent to
86        * @return The request
87        */
88       public NetworkClient.InFlightRequest completeLastSent(String node) {
89           NetworkClient.InFlightRequest inFlightRequest = requestQueue(node).pollFirst();
90           inFlightRequestCount.decrementAndGet();
91           return inFlightRequest;
92       }
93   
94       /**
95        * Can we send more requests to this node?
96        *
97        * @param node Node in question
98        * @return true iff we have no requests still being sent to the given node
99        */
100       public boolean canSendMore(String node) {
101           Deque<NetworkClient.InFlightRequest> queue = requests.get(node);
102           return queue == null || queue.isEmpty() ||
103                  (queue.peekFirst().send.completed() && queue.size() < this.maxInFlightRequestsPerConnection);
104       }
105   
106       /**
107        * Return the number of in-flight requests directed at the given node
108        * @param node The node
109        * @return The request count.
110        */
111       public int count(String node) {
112           Deque<NetworkClient.InFlightRequest> queue = requests.get(node);
113           return queue == null ? 0 : queue.size();
114       }
115   
116       /**
117        * Return true if there is no in-flight request directed at the given node and false otherwise
118        */
119       public boolean isEmpty(String node) {
120           Deque<NetworkClient.InFlightRequest> queue = requests.get(node);
121           return queue == null || queue.isEmpty();
122       }
123   
124       /**
125        * Count all in-flight requests for all nodes. This method is thread safe, but may lag the actual count.
126        */
127       public int count() {
128           return inFlightRequestCount.get();
129       }
130   
131       /**
132        * Return true if there is no in-flight request and false otherwise
133        */
134       public boolean isEmpty() {
135           for (Deque<NetworkClient.InFlightRequest> deque : this.requests.values()) {
136               if (!deque.isEmpty())
137                   return false;
138           }
139           return true;
140       }
141   
142       /**
143        * Clear out all the in-flight requests for the given node and return them
144        *
145        * @param node The node
146        * @return All the in-flight requests for that node that have been removed
147        */
148       public Iterable<NetworkClient.InFlightRequest> clearAll(String node) {
149           Deque<NetworkClient.InFlightRequest> reqs = requests.get(node);
150           if (reqs == null) {
151               return Collections.emptyList();
152           } else {
153               final Deque<NetworkClient.InFlightRequest> clearedRequests = requests.remove(node);
154               inFlightRequestCount.getAndAdd(-clearedRequests.size());
155               return () -> clearedRequests.descendingIterator();
156           }
157       }
158   
159       private Boolean hasExpiredRequest(long now, Deque<NetworkClient.InFlightRequest> deque) {
160           for (NetworkClient.InFlightRequest request : deque) {
161               if (request.timeElapsedSinceSendMs(now) > request.requestTimeoutMs)
162                   return true;
163           }
164           return false;
165       }
166   
167       /**
168        * Returns a list of nodes with pending in-flight request, that need to be timed out
169        *
170        * @param now current time in milliseconds
171        * @return list of nodes
172        */
173       public List<String> nodesWithTimedOutRequests(long now) {
174           List<String> nodeIds = new ArrayList<>();
175           for (Map.Entry<String, Deque<NetworkClient.InFlightRequest>> requestEntry : requests.entrySet()) {
176               String nodeId = requestEntry.getKey();
177               Deque<NetworkClient.InFlightRequest> deque = requestEntry.getValue();
178               if (hasExpiredRequest(now, deque))
179                   nodeIds.add(nodeId);
180           }
181           return nodeIds;
182       }
