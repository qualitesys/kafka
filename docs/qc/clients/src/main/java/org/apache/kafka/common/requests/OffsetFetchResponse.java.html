44    * Possible error codes:
45    *
46    * - Partition errors:

137       /**
138        * Constructor with throttle time for version 0 to 7
139        * @param throttleTimeMs The time in milliseconds that this response was throttled
140        * @param error Potential coordinator or group level error code (for api version 2 and later)
141        * @param responseData Fetched offset information grouped by topic-partition
142        */
143       public OffsetFetchResponse(int throttleTimeMs, Errors error, Map<TopicPartition, PartitionData> responseData) {
144           super(ApiKeys.OFFSET_FETCH);
145           Map<String, OffsetFetchResponseTopic> offsetFetchResponseTopicMap = new HashMap<>();
146           for (Map.Entry<TopicPartition, PartitionData> entry : responseData.entrySet()) {
147               String topicName = entry.getKey().topic();
148               OffsetFetchResponseTopic topic = offsetFetchResponseTopicMap.getOrDefault(
149                   topicName, new OffsetFetchResponseTopic().setName(topicName));
150               PartitionData partitionData = entry.getValue();
151               topic.partitions().add(new OffsetFetchResponsePartition()
152                                          .setPartitionIndex(entry.getKey().partition())
153                                          .setErrorCode(partitionData.error.code())
154                                          .setCommittedOffset(partitionData.offset)
155                                          .setCommittedLeaderEpoch(
156                                              partitionData.leaderEpoch.orElse(NO_PARTITION_LEADER_EPOCH))
157                                          .setMetadata(partitionData.metadata)
158               );
159               offsetFetchResponseTopicMap.put(topicName, topic);
160           }
161   
162           this.data = new OffsetFetchResponseData()
163               .setTopics(new ArrayList<>(offsetFetchResponseTopicMap.values()))
164               .setErrorCode(error.code())
165               .setThrottleTimeMs(throttleTimeMs);
166           this.error = error;
167       }
168   
169       /**
170        * Constructor with throttle time for version 8 and above.
171        * @param throttleTimeMs The time in milliseconds that this response was throttled
172        * @param errors Potential coordinator or group level error code
173        * @param responseData Fetched offset information grouped by topic-partition and by group
174        */
175       public OffsetFetchResponse(int throttleTimeMs,
176                                  Map<String, Errors> errors, Map<String,
177                                  Map<TopicPartition, PartitionData>> responseData) {
178           super(ApiKeys.OFFSET_FETCH);
179           List<OffsetFetchResponseGroup> groupList = new ArrayList<>();
180           for (Entry<String, Map<TopicPartition, PartitionData>> entry : responseData.entrySet()) {
181               String groupName = entry.getKey();
182               Map<TopicPartition, PartitionData> partitionDataMap = entry.getValue();
183               Map<String, OffsetFetchResponseTopics> offsetFetchResponseTopicsMap = new HashMap<>();
184               for (Entry<TopicPartition, PartitionData> partitionEntry : partitionDataMap.entrySet()) {
185                   String topicName = partitionEntry.getKey().topic();
186                   OffsetFetchResponseTopics topic =
187                       offsetFetchResponseTopicsMap.getOrDefault(topicName,
188                           new OffsetFetchResponseTopics().setName(topicName));
189                   PartitionData partitionData = partitionEntry.getValue();
190                   topic.partitions().add(new OffsetFetchResponsePartitions()
191                       .setPartitionIndex(partitionEntry.getKey().partition())
192                       .setErrorCode(partitionData.error.code())
193                       .setCommittedOffset(partitionData.offset)
194                       .setCommittedLeaderEpoch(
195                           partitionData.leaderEpoch.orElse(NO_PARTITION_LEADER_EPOCH))
196                       .setMetadata(partitionData.metadata));
197                   offsetFetchResponseTopicsMap.put(topicName, topic);
198               }
199               groupList.add(new OffsetFetchResponseGroup()
200                   .setGroupId(groupName)
201                   .setTopics(new ArrayList<>(offsetFetchResponseTopicsMap.values()))
202                   .setErrorCode(errors.get(groupName).code()));
203               groupLevelErrors.put(groupName, errors.get(groupName));
204           }
205           this.data = new OffsetFetchResponseData()
206               .setGroups(groupList)
207               .setThrottleTimeMs(throttleTimeMs);
208           this.error = null;
209       }

211       public OffsetFetchResponse(OffsetFetchResponseData data, short version) {
212           super(ApiKeys.OFFSET_FETCH);
213           this.data = data;
214           // for version 2 and later use the top-level error code (in ERROR_CODE_KEY_NAME) from the response.
215           // for older versions there is no top-level error in the response and all errors are partition errors,
216           // so if there is a group or coordinator error at the partition level use that as the top-level error.
217           // this way clients can depend on the top-level error regardless of the offset fetch version.
218           // we return the error differently starting with version 8, so we will only populate the
219           // error field if we are between version 2 and 7. if we are in version 8 or greater, then
220           // we will populate the map of group id to error codes.
221           if (version < 8) {
222               this.error = version >= 2 ? Errors.forCode(data.errorCode()) : topLevelError(data);
223           } else {
224               for (OffsetFetchResponseGroup group : data.groups()) {
225                   this.groupLevelErrors.put(group.groupId(), Errors.forCode(group.errorCode()));
226               }
227               this.error = null;
228           }
229       }

231       private static Errors topLevelError(OffsetFetchResponseData data) {
232           for (OffsetFetchResponseTopic topic : data.topics()) {
233               for (OffsetFetchResponsePartition partition : topic.partitions()) {
234                   Errors partitionError = Errors.forCode(partition.errorCode());
235                   if (partitionError != Errors.NONE && !PARTITION_ERRORS.contains(partitionError)) {
236                       return partitionError;
237                   }
238               }
239           }
240           return Errors.NONE;
241       }

267       @Override
268       public Map<Errors, Integer> errorCounts() {
269           Map<Errors, Integer> counts = new HashMap<>();
270           if (!groupLevelErrors.isEmpty()) {
271               // built response with v8 or above
272               for (Map.Entry<String, Errors> entry : groupLevelErrors.entrySet()) {
273                   updateErrorCounts(counts, entry.getValue());
274               }
275               for (OffsetFetchResponseGroup group : data.groups()) {
276                   group.topics().forEach(topic ->
277                       topic.partitions().forEach(partition ->
278                           updateErrorCounts(counts, Errors.forCode(partition.errorCode()))));
279               }
280           } else {
281               // built response with v0-v7
282               updateErrorCounts(counts, error);
283               data.topics().forEach(topic ->
284                   topic.partitions().forEach(partition ->
285                       updateErrorCounts(counts, Errors.forCode(partition.errorCode()))));
286           }
287           return counts;
288       }

290       // package-private for testing purposes
291       Map<TopicPartition, PartitionData> responseDataV0ToV7() {
292           Map<TopicPartition, PartitionData> responseData = new HashMap<>();
293           for (OffsetFetchResponseTopic topic : data.topics()) {
294               for (OffsetFetchResponsePartition partition : topic.partitions()) {
295                   responseData.put(new TopicPartition(topic.name(), partition.partitionIndex()),
296                                    new PartitionData(partition.committedOffset(),
297                                                      RequestUtils.getLeaderEpoch(partition.committedLeaderEpoch()),
298                                                      partition.metadata(),
299                                                      Errors.forCode(partition.errorCode()))
300                   );
301               }
302           }
303           return responseData;
304       }

306       private Map<TopicPartition, PartitionData> buildResponseData(String groupId) {
307           Map<TopicPartition, PartitionData> responseData = new HashMap<>();
308           OffsetFetchResponseGroup group = data
309               .groups()
310               .stream()
311               .filter(g -> g.groupId().equals(groupId))
312               .collect(Collectors.toList())
313               .get(0);
314           for (OffsetFetchResponseTopics topic : group.topics()) {
315               for (OffsetFetchResponsePartitions partition : topic.partitions()) {
316                   responseData.put(new TopicPartition(topic.name(), partition.partitionIndex()),
317                       new PartitionData(partition.committedOffset(),
318                           RequestUtils.getLeaderEpoch(partition.committedLeaderEpoch()),
319                           partition.metadata(),
320                           Errors.forCode(partition.errorCode()))
321                   );
322               }
323           }
324           return responseData;
325       }
