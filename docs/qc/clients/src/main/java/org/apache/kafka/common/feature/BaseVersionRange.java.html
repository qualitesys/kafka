18   
19   import static java.util.stream.Collectors.joining;
20   
21   import java.util.Map;
22   import java.util.Objects;
23   
24   import org.apache.kafka.common.utils.Utils;
25   
26   /**
27    * Represents an immutable basic version range using 2 attributes: min and max, each of type short.
28    * The min and max attributes need to satisfy 2 rules:
29    *  - they are each expected to be >= 1, as we only consider positive version values to be valid.
30    *  - max should be >= min.
31    *
32    * The class also provides API to convert the version range to a map.
33    * The class allows for configurable labels for the min/max attributes, which can be specialized by
34    * sub-classes (if needed).
35    */
36   class BaseVersionRange {
37       // Non-empty label for the min version key, that's used only to convert to/from a map.
38       private final String minKeyLabel;
39   
40       // The value of the minimum version.
41       private final short minValue;
42   
43       // Non-empty label for the max version key, that's used only to convert to/from a map.
44       private final String maxKeyLabel;
45   
46       // The value of the maximum version.
47       private final short maxValue;
48   
49       /**
50        * Raises an exception unless the following condition is met:
51        * minValue >= 1 and maxValue >= 1 and maxValue >= minValue.
52        *
53        * @param minKeyLabel   Label for the min version key, that's used only to convert to/from a map.
54        * @param minValue      The minimum version value.
55        * @param maxKeyLabel   Label for the max version key, that's used only to convert to/from a map.
56        * @param maxValue      The maximum version value.
57        *
58        * @throws IllegalArgumentException   If any of the following conditions are true:
59        *                                     - (minValue < 1) OR (maxValue < 1) OR (maxValue < minValue).
60        *                                     - minKeyLabel is empty, OR, minKeyLabel is empty.
61        */
62       protected BaseVersionRange(String minKeyLabel, short minValue, String maxKeyLabel, short maxValue) {
63           if (minValue < 1 || maxValue < 1 || maxValue < minValue) {
64               throw new IllegalArgumentException(
65                   String.format(
66                       "Expected minValue >= 1, maxValue >= 1 and maxValue >= minValue, but received" +
67                       " minValue: %d, maxValue: %d", minValue, maxValue));
68           }
69           if (minKeyLabel.isEmpty()) {
70               throw new IllegalArgumentException("Expected minKeyLabel to be non-empty.");
71           }
72           if (maxKeyLabel.isEmpty()) {
73               throw new IllegalArgumentException("Expected maxKeyLabel to be non-empty.");
74           }
75           this.minKeyLabel = minKeyLabel;
76           this.minValue = minValue;
77           this.maxKeyLabel = maxKeyLabel;
78           this.maxValue = maxValue;
79       }
80   
81       public short min() {
82           return minValue;
83       }
84   
85       public short max() {
86           return maxValue;
87       }
88   
89       public String toString() {
90           return String.format(
91               "%s[%s]",
92               this.getClass().getSimpleName(),
93               mapToString(toMap()));
94       }
95   
96       public Map<String, Short> toMap() {
97           return Utils.mkMap(Utils.mkEntry(minKeyLabel, min()), Utils.mkEntry(maxKeyLabel, max()));
98       }
99   
100       private static String mapToString(final Map<String, Short> map) {
101           return map
102               .entrySet()
103               .stream()
104               .map(entry -> String.format("%s:%d", entry.getKey(), entry.getValue()))
105               .collect(joining(", "));
106       }
107   
108       @Override
109       public boolean equals(Object other) {
110           if (this == other) {
111               return true;
112           }
113   
114           if (other == null || getClass() != other.getClass()) {
115               return false;
116           }
117   
118           final BaseVersionRange that = (BaseVersionRange) other;
119           return Objects.equals(this.minKeyLabel, that.minKeyLabel) &&
120               this.minValue == that.minValue &&
121               Objects.equals(this.maxKeyLabel, that.maxKeyLabel) &&
122               this.maxValue == that.maxValue;
123       }
124   
125       @Override
126       public int hashCode() {
127           return Objects.hash(minKeyLabel, minValue, maxKeyLabel, maxValue);
128       }
129   
130       public static short valueOrThrow(String key, Map<String, Short> versionRangeMap) {
131           final Short value = versionRangeMap.get(key);
132           if (value == null) {
133               throw new IllegalArgumentException(String.format("%s absent in [%s]", key, mapToString(versionRangeMap)));
134           }
135           return value;
136       }
137   }
