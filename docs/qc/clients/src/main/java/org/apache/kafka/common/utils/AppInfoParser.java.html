34   public class AppInfoParser {
35       private static final Logger log = LoggerFactory.getLogger(AppInfoParser.class);
36       private static final String VERSION;
37       private static final String COMMIT_ID;
38   
39       protected static final String DEFAULT_VALUE = "unknown";
40   
41       static {
42           Properties props = new Properties();
43           try (InputStream resourceStream = AppInfoParser.class.getResourceAsStream("/kafka/kafka-version.properties")) {
44               props.load(resourceStream);
45           } catch (Exception e) {
46               log.warn("Error while loading kafka-version.properties: {}", e.getMessage());
47           }
48           VERSION = props.getProperty("version", DEFAULT_VALUE).trim();
49           COMMIT_ID = props.getProperty("commitId", DEFAULT_VALUE).trim();
50       }
51   
52       public static String getVersion() {
53           return VERSION;
54       }
55   
56       public static String getCommitId() {
57           return COMMIT_ID;
58       }
59   
60       public static synchronized void registerAppInfo(String prefix, String id, Metrics metrics, long nowMs) {
61           try {
62               ObjectName name = new ObjectName(prefix + ":type=app-info,id=" + Sanitizer.jmxSanitize(id));
63               AppInfo mBean = new AppInfo(nowMs);
64               ManagementFactory.getPlatformMBeanServer().registerMBean(mBean, name);
65   
66               registerMetrics(metrics, mBean); // prefix will be added later by JmxReporter
67           } catch (JMException e) {
68               log.warn("Error registering AppInfo mbean", e);
69           }
70       }
71   
72       public static synchronized void unregisterAppInfo(String prefix, String id, Metrics metrics) {
73           MBeanServer server = ManagementFactory.getPlatformMBeanServer();
74           try {
75               ObjectName name = new ObjectName(prefix + ":type=app-info,id=" + Sanitizer.jmxSanitize(id));
76               if (server.isRegistered(name))
77                   server.unregisterMBean(name);
78   
79               unregisterMetrics(metrics);
80           } catch (JMException e) {
81               log.warn("Error unregistering AppInfo mbean", e);
82           } finally {
83               log.info("App info {} for {} unregistered", prefix, id);
84           }
85       }
86   
87       private static MetricName metricName(Metrics metrics, String name) {
88           return metrics.metricName(name, "app-info", "Metric indicating " + name);
89       }
90   
91       private static void registerMetrics(Metrics metrics, AppInfo appInfo) {
92           if (metrics != null) {
93               metrics.addMetric(metricName(metrics, "version"), new ImmutableValue<>(appInfo.getVersion()));
94               metrics.addMetric(metricName(metrics, "commit-id"), new ImmutableValue<>(appInfo.getCommitId()));
95               metrics.addMetric(metricName(metrics, "start-time-ms"), new ImmutableValue<>(appInfo.getStartTimeMs()));
96           }
97       }
98   
99       private static void unregisterMetrics(Metrics metrics) {
100           if (metrics != null) {
101               metrics.removeMetric(metricName(metrics, "version"));
102               metrics.removeMetric(metricName(metrics, "commit-id"));
103               metrics.removeMetric(metricName(metrics, "start-time-ms"));
104           }
105       }
106   
107       public interface AppInfoMBean {
108           String getVersion();
109           String getCommitId();
110           Long getStartTimeMs();
111       }
112   
113       public static class AppInfo implements AppInfoMBean {
114   
115           private final Long startTimeMs;
116   
117           public AppInfo(long startTimeMs) {
118               this.startTimeMs = startTimeMs;
119               log.info("Kafka version: {}", AppInfoParser.getVersion());
120               log.info("Kafka commitId: {}", AppInfoParser.getCommitId());
121               log.info("Kafka startTimeMs: {}", startTimeMs);
122           }
123   
124           @Override
125           public String getVersion() {
126               return AppInfoParser.getVersion();
127           }
128   
129           @Override
130           public String getCommitId() {
131               return AppInfoParser.getCommitId();
132           }
133   
134           @Override
135           public Long getStartTimeMs() {
136               return startTimeMs;
137           }
138   
139       }

141       static class ImmutableValue<T> implements Gauge<T> {
142           private final T value;
143   
144           public ImmutableValue(T value) {
145               this.value = value;
146           }
147   
148           @Override
149           public T value(MetricConfig config, long now) {
150               return value;
151           }
152       }
