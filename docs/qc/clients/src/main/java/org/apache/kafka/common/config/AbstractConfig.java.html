65       /**
66        * Construct a configuration with a ConfigDef and the configuration properties, which can include properties
67        * for zero or more {@link ConfigProvider} that will be used to resolve variables in configuration property
68        * values.
69        *
70        * The originals is a name-value pair configuration properties and optional config provider configs. The
71        * value of the configuration can be a variable as defined below or the actual value. This constructor will
72        * first instantiate the ConfigProviders using the config provider configs, then it will find all the
73        * variables in the values of the originals configurations, attempt to resolve the variables using the named
74        * ConfigProviders, and then parse and validate the configurations.
75        *
76        * ConfigProvider configs can be passed either as configs in the originals map or in the separate
77        * configProviderProps map. If config providers properties are passed in the configProviderProps any config
78        * provider properties in originals map will be ignored. If ConfigProvider properties are not provided, the
79        * constructor will skip the variable substitution step and will simply validate and parse the supplied
80        * configuration.
81        *
82        * The "{@code config.providers}" configuration property and all configuration properties that begin with the
83        * "{@code config.providers.}" prefix are reserved. The "{@code config.providers}" configuration property
84        * specifies the names of the config providers, and properties that begin with the "{@code config.providers..}"
85        * prefix correspond to the properties for that named provider. For example, the "{@code config.providers..class}"
86        * property specifies the name of the {@link ConfigProvider} implementation class that should be used for
87        * the provider.
88        *
89        * The keys for ConfigProvider configs in both originals and configProviderProps will start with the above
90        * mentioned "{@code config.providers.}" prefix.
91        *
92        * Variables have the form "${providerName:[path:]key}", where "providerName" is the name of a ConfigProvider,
93        * "path" is an optional string, and "key" is a required string. This variable is resolved by passing the "key"
94        * and optional "path" to a ConfigProvider with the specified name, and the result from the ConfigProvider is
95        * then used in place of the variable. Variables that cannot be resolved by the AbstractConfig constructor will
96        * be left unchanged in the configuration.
97        *
98        *
99        * @param definition the definition of the configurations; may not be null
100        * @param originals the configuration properties plus any optional config provider properties;
101        * @param configProviderProps the map of properties of config providers which will be instantiated by
102        *        the constructor to resolve any variables in {@code originals}; may be null or empty
103        * @param doLog whether the configurations should be logged
104        */
105       @SuppressWarnings("unchecked")
106       public AbstractConfig(ConfigDef definition, Map<?, ?> originals,  Map<String, ?> configProviderProps, boolean doLog) {
107           /* check that all the keys are really strings */
108           for (Map.Entry<?, ?> entry : originals.entrySet())
109               if (!(entry.getKey() instanceof String))
110                   throw new ConfigException(entry.getKey().toString(), entry.getValue(), "Key must be a string.");
111   
112           this.originals = resolveConfigVariables(configProviderProps, (Map<String, Object>) originals);
113           this.values = definition.parse(this.originals);
114           Map<String, Object> configUpdates = postProcessParsedConfig(Collections.unmodifiableMap(this.values));
115           for (Map.Entry<String, Object> update : configUpdates.entrySet()) {
116               this.values.put(update.getKey(), update.getValue());
117           }
118           definition.parse(this.values);
119           this.definition = definition;
120           if (doLog)
121               logAll();
122       }

269        * Gets all original settings with the given prefix.
270        *
271        * @param prefix the prefix to use as a filter
272        * @param strip strip the prefix before adding to the output if set true
273        * @return a Map containing the settings with the prefix
274        */
275       public Map<String, Object> originalsWithPrefix(String prefix, boolean strip) {
276           Map<String, Object> result = new RecordingMap<>(prefix, false);
277           for (Map.Entry<String, ?> entry : originals.entrySet()) {
278               if (entry.getKey().startsWith(prefix) && entry.getKey().length() > prefix.length()) {
279                   if (strip)
280                       result.put(entry.getKey().substring(prefix.length()), entry.getValue());
281                   else
282                       result.put(entry.getKey(), entry.getValue());
283               }
284           }
285           return result;
286       }

288       /**
289        * Put all keys that do not start with {@code prefix} and their parsed values in the result map and then
290        * put all the remaining keys with the prefix stripped and their parsed values in the result map.
291        *
292        * This is useful if one wants to allow prefixed configs to override default ones.
293        * <p>
294        * Two forms of prefixes are supported:
295        * <ul>
296        *     <li>listener.name.{listenerName}.some.prop: If the provided prefix is `listener.name.{listenerName}.`,
297        *         the key `some.prop` with the value parsed using the definition of `some.prop` is returned.</li>
298        *     <li>listener.name.{listenerName}.{mechanism}.some.prop: If the provided prefix is `listener.name.{listenerName}.`,
299        *         the key `{mechanism}.some.prop` with the value parsed using the definition of `some.prop` is returned.
300        *          This is used to provide per-mechanism configs for a broker listener (e.g sasl.jaas.config)</li>
301        * </ul>
302        * </p>
303        */
304       public Map<String, Object> valuesWithPrefixOverride(String prefix) {
305           Map<String, Object> result = new RecordingMap<>(values(), prefix, true);
306           for (Map.Entry<String, ?> entry : originals.entrySet()) {
307               if (entry.getKey().startsWith(prefix) && entry.getKey().length() > prefix.length()) {
308                   String keyWithNoPrefix = entry.getKey().substring(prefix.length());
309                   ConfigDef.ConfigKey configKey = definition.configKeys().get(keyWithNoPrefix);
310                   if (configKey != null)
311                       result.put(keyWithNoPrefix, definition.parseValue(configKey, entry.getValue(), true));
312                   else {
313                       String keyWithNoSecondaryPrefix = keyWithNoPrefix.substring(keyWithNoPrefix.indexOf('.') + 1);
314                       configKey = definition.configKeys().get(keyWithNoSecondaryPrefix);
315                       if (configKey != null)
316                           result.put(keyWithNoPrefix, definition.parseValue(configKey, entry.getValue(), true));
317                   }
318               }
319           }
320           return result;
321       }

323       /**
324        * If at least one key with {@code prefix} exists, all prefixed values will be parsed and put into map.
325        * If no value with {@code prefix} exists all unprefixed values will be returned.
326        *
327        * This is useful if one wants to allow prefixed configs to override default ones, but wants to use either
328        * only prefixed configs or only regular configs, but not mix them.
329        */
330       public Map<String, Object> valuesWithPrefixAllOrNothing(String prefix) {
331           Map<String, Object> withPrefix = originalsWithPrefix(prefix, true);
332   
333           if (withPrefix.isEmpty()) {
334               return new RecordingMap<>(values(), "", true);
335           } else {
336               Map<String, Object> result = new RecordingMap<>(prefix, true);
337   
338               for (Map.Entry<String, ?> entry : withPrefix.entrySet()) {
339                   ConfigDef.ConfigKey configKey = definition.configKeys().get(entry.getKey());
340                   if (configKey != null)
341                       result.put(entry.getKey(), definition.parseValue(configKey, entry.getValue(), true));
342               }
343   
344               return result;
345           }
346       }

389       private <T> T getConfiguredInstance(Object klass, Class<T> t, Map<String, Object> configPairs) {
390           if (klass == null)
391               return null;
392   
393           Object o;
394           if (klass instanceof String) {
395               try {
396                   o = Utils.newInstance((String) klass, t);
397               } catch (ClassNotFoundException e) {
398                   throw new KafkaException("Class " + klass + " cannot be found", e);
399               }
400           } else if (klass instanceof Class<?>) {
401               o = Utils.newInstance((Class<?>) klass);
402           } else
403               throw new KafkaException("Unexpected element of type " + klass.getClass().getName() + ", expected String or Class");
404           if (!t.isInstance(o))
405               throw new KafkaException(klass + " is not an instance of " + t.getName());
406           if (o instanceof Configurable)
407               ((Configurable) o).configure(configPairs);
408   
409           return t.cast(o);
410       }

424       /**
425        * Get a configured instance of the give class specified by the given configuration key. If the object implements
426        * Configurable configure it using the configuration.
427        *

486       private Map<String, String> extractPotentialVariables(Map<?, ?>  configMap) {
487           // Variables are tuples of the form "${providerName:[path:]key}". From the configMap we extract the subset of configs with string
488           // values as potential variables.
489           Map<String, String> configMapAsString = new HashMap<>();
490           for (Map.Entry<?, ?> entry : configMap.entrySet()) {
491               if (entry.getValue() instanceof String)
492                   configMapAsString.put((String) entry.getKey(), (String) entry.getValue());
493           }
494   
495           return configMapAsString;
496       }

498       /**
499        * Instantiates given list of config providers and fetches the actual values of config variables from the config providers.
500        * returns a map of config key and resolved values.
501        * @param configProviderProps The map of config provider configs
502        * @param originals The map of raw configs.
503        * @return map of resolved config variable.
504        */
505       @SuppressWarnings("unchecked")
506       private  Map<String, ?> resolveConfigVariables(Map<String, ?> configProviderProps, Map<String, Object> originals) {
507           Map<String, String> providerConfigString;
508           Map<String, ?> configProperties;
509           Map<String, Object> resolvedOriginals = new HashMap<>();
510           // As variable configs are strings, parse the originals and obtain the potential variable configs.
511           Map<String, String> indirectVariables = extractPotentialVariables(originals);
512   
513           resolvedOriginals.putAll(originals);
514           if (configProviderProps == null || configProviderProps.isEmpty()) {
515               providerConfigString = indirectVariables;
516               configProperties = originals;
517           } else {
518               providerConfigString = extractPotentialVariables(configProviderProps);
519               configProperties = configProviderProps;
520           }
521           Map<String, ConfigProvider> providers = instantiateConfigProviders(providerConfigString, configProperties);
522   
523           if (!providers.isEmpty()) {
524               ConfigTransformer configTransformer = new ConfigTransformer(providers);
525               ConfigTransformerResult result = configTransformer.transform(indirectVariables);
526               if (!result.data().isEmpty()) {
527                   resolvedOriginals.putAll(result.data());
528               }
529           }
530           providers.values().forEach(x -> Utils.closeQuietly(x, "config provider"));
531   
532           return new ResolvingMap<>(resolvedOriginals, originals);
533       }

535       private Map<String, Object> configProviderProperties(String configProviderPrefix, Map<String, ?> providerConfigProperties) {
536           Map<String, Object> result = new HashMap<>();
537           for (Map.Entry<String, ?> entry : providerConfigProperties.entrySet()) {
538               String key = entry.getKey();
539               if (key.startsWith(configProviderPrefix) && key.length() > configProviderPrefix.length()) {
540                   result.put(key.substring(configProviderPrefix.length()), entry.getValue());
541               }
542           }
543           return result;
544       }

546       /**
547        * Instantiates and configures the ConfigProviders. The config providers configs are defined as follows:
548        * config.providers : A comma-separated list of names for providers.
549        * config.providers.{name}.class : The Java class name for a provider.
550        * config.providers.{name}.param.{param-name} : A parameter to be passed to the above Java class on initialization.
551        * returns a map of config provider name and its instance.
552        * @param indirectConfigs The map of potential variable configs
553        * @param providerConfigProperties The map of config provider configs
554        * @return map map of config provider name and its instance.
555        */
556       private Map<String, ConfigProvider> instantiateConfigProviders(Map<String, String> indirectConfigs, Map<String, ?> providerConfigProperties) {
557           final String configProviders = indirectConfigs.get(CONFIG_PROVIDERS_CONFIG);
558   
559           if (configProviders == null || configProviders.isEmpty()) {
560               return Collections.emptyMap();
561           }
562   
563           Map<String, String> providerMap = new HashMap<>();
564   
565           for (String provider: configProviders.split(",")) {
566               String providerClass = providerClassProperty(provider);
567               if (indirectConfigs.containsKey(providerClass))
568                   providerMap.put(provider, indirectConfigs.get(providerClass));
569   
570           }
571           // Instantiate Config Providers
572           Map<String, ConfigProvider> configProviderInstances = new HashMap<>();
573           for (Map.Entry<String, String> entry : providerMap.entrySet()) {
574               try {
575                   String prefix = CONFIG_PROVIDERS_CONFIG + "." + entry.getKey() + CONFIG_PROVIDERS_PARAM;
576                   Map<String, ?> configProperties = configProviderProperties(prefix, providerConfigProperties);
577                   ConfigProvider provider = Utils.newInstance(entry.getValue(), ConfigProvider.class);
578                   provider.configure(configProperties);
579                   configProviderInstances.put(entry.getKey(), provider);
580               } catch (ClassNotFoundException e) {
581                   log.error("Could not load config provider class " + entry.getValue(), e);
582                   throw new ConfigException(providerClassProperty(entry.getKey()), entry.getValue(), "Could not load config provider class or one of its dependencies");
583               }
584           }
585   
586           return configProviderInstances;
587       }

608       /**
609        * Marks keys retrieved via `get` as used. This is needed because `Configurable.configure` takes a `Map` instead
610        * of an `AbstractConfig` and we can't change that without breaking public API like `Partitioner`.
611        */
612       private class RecordingMap<V> extends HashMap<String, V> {
613   
614           private final String prefix;
615           private final boolean withIgnoreFallback;
616   
617           RecordingMap() {
618               this("", false);
619           }
620   
621           RecordingMap(String prefix, boolean withIgnoreFallback) {
622               this.prefix = prefix;
623               this.withIgnoreFallback = withIgnoreFallback;
624           }
625   
626           RecordingMap(Map<String, ? extends V> m) {
627               this(m, "", false);
628           }
629   
630           RecordingMap(Map<String, ? extends V> m, String prefix, boolean withIgnoreFallback) {
631               super(m);
632               this.prefix = prefix;
633               this.withIgnoreFallback = withIgnoreFallback;
634           }
635   
636           @Override
637           public V get(Object key) {
638               if (key instanceof String) {
639                   String stringKey = (String) key;
640                   String keyWithPrefix;
641                   if (prefix.isEmpty()) {
642                       keyWithPrefix = stringKey;
643                   } else {
644                       keyWithPrefix = prefix + stringKey;
645                   }
646                   ignore(keyWithPrefix);
647                   if (withIgnoreFallback)
648                       ignore(stringKey);
649               }
650               return super.get(key);
651           }
652       }

654       /**
655        * ResolvingMap keeps a track of the original map instance and the resolved configs.
656        * The originals are tracked in a separate nested map and may be a `RecordingMap`; thus
657        * any access to a value for a key needs to be recorded on the originals map.
658        * The resolved configs are kept in the inherited map and are therefore mutable, though any
659        * mutations are not applied to the originals.
660        */
661       private static class ResolvingMap<V> extends HashMap<String, V> {
662   
663           private final Map<String, ?> originals;
664   
665           ResolvingMap(Map<String, ? extends V> resolved, Map<String, ?> originals) {
666               super(resolved);
667               this.originals = Collections.unmodifiableMap(originals);
668           }
669   
670           @Override
671           public V get(Object key) {
672               if (key instanceof String && originals.containsKey(key)) {
673                   // Intentionally ignore the result; call just to mark the original entry as used
674                   originals.get(key);
675               }
676               // But always use the resolved entry
677               return super.get(key);
678           }
679       }
