593       private void maybeOverrideClientId(Map<String, Object> configs) {
594           final String clientId = this.getString(CLIENT_ID_CONFIG);
595           if (clientId == null || clientId.isEmpty()) {
596               final String groupId = this.getString(GROUP_ID_CONFIG);
597               String groupInstanceId = this.getString(GROUP_INSTANCE_ID_CONFIG);
598               if (groupInstanceId != null)
599                   JoinGroupRequest.validateGroupInstanceId(groupInstanceId);
600   
601               String groupInstanceIdPart = groupInstanceId != null ? groupInstanceId : CONSUMER_CLIENT_ID_SEQUENCE.getAndIncrement() + "";
602               String generatedClientId = String.format("consumer-%s-%s", groupId, groupInstanceIdPart);
603               configs.put(CLIENT_ID_CONFIG, generatedClientId);
604           }

610           // validate deserializer configuration, if the passed deserializer instance is null, the user must explicitly set a valid deserializer configuration value
611           Map<String, Object> newConfigs = new HashMap<>(configs);

614           else if (newConfigs.get(KEY_DESERIALIZER_CLASS_CONFIG) == null)
615               throw new ConfigException(KEY_DESERIALIZER_CLASS_CONFIG, null, "must be non-null.");
616           if (valueDeserializer != null)
617               newConfigs.put(VALUE_DESERIALIZER_CLASS_CONFIG, valueDeserializer.getClass());
618           else if (newConfigs.get(VALUE_DESERIALIZER_CLASS_CONFIG) == null)
619               throw new ConfigException(VALUE_DESERIALIZER_CLASS_CONFIG, null, "must be non-null.");

623       boolean maybeOverrideEnableAutoCommit() {
624           Optional<String> groupId = Optional.ofNullable(getString(CommonClientConfigs.GROUP_ID_CONFIG));
625           boolean enableAutoCommit = getBoolean(ConsumerConfig.ENABLE_AUTO_COMMIT_CONFIG);
626           if (!groupId.isPresent()) { // overwrite in case of default group id where the config is not explicitly provided
627               if (!originals().containsKey(ENABLE_AUTO_COMMIT_CONFIG)) {
628                   enableAutoCommit = false;
629               } else if (enableAutoCommit) {
630                   throw new InvalidConfigurationException(ConsumerConfig.ENABLE_AUTO_COMMIT_CONFIG + " cannot be set to true when default group id (null) is used.");
631               }
632           }
633           return enableAutoCommit;
634       }
