21   
22   import java.util.List;
23   import java.util.Map;

25   /**
26    * Describes new partitions for a particular topic in a call to {@link Admin#createPartitions(Map)}.
27    *
28    * The API of this class is evolving, see {@link Admin} for details.
29    */
30   @InterfaceStability.Evolving
31   public class NewPartitions {
32   
33       private int totalCount;
34   
35       private List<List<Integer>> newAssignments;
36   
37       private NewPartitions(int totalCount, List<List<Integer>> newAssignments) {
38           this.totalCount = totalCount;
39           this.newAssignments = newAssignments;
40       }
41   
42       /**
43        * Increase the partition count for a topic to the given {@code totalCount}.
44        * The assignment of new replicas to brokers will be decided by the broker.
45        *
46        * @param totalCount The total number of partitions after the operation succeeds.
47        */
48       public static NewPartitions increaseTo(int totalCount) {
49           return new NewPartitions(totalCount, null);
50       }
51   
52       /**
53        * <p>Increase the partition count for a topic to the given {@code totalCount}
54        * assigning the new partitions according to the given {@code newAssignments}.
55        * The length of the given {@code newAssignments} should equal {@code totalCount - oldCount}, since
56        * the assignment of existing partitions are not changed.
57        * Each inner list of {@code newAssignments} should have a length equal to
58        * the topic's replication factor.
59        * The first broker id in each inner list is the "preferred replica".</p>
60        *
61        * <p>For example, suppose a topic currently has a replication factor of 2, and
62        * has 3 partitions. The number of partitions can be increased to 6 using a
63        * {@code NewPartition} constructed like this:</p>
64        *
65        * <pre><code>
66        * NewPartitions.increaseTo(6, asList(asList(1, 2),
67        *                                    asList(2, 3),
68        *                                    asList(3, 1)))
69        * </code></pre>
70        * <p>In this example partition 3's preferred leader will be broker 1, partition 4's preferred leader will be
71        * broker 2 and partition 5's preferred leader will be broker 3.</p>
72        *
73        * @param totalCount The total number of partitions after the operation succeeds.
74        * @param newAssignments The replica assignments for the new partitions.
75        */
76       public static NewPartitions increaseTo(int totalCount, List<List<Integer>> newAssignments) {
77           return new NewPartitions(totalCount, newAssignments);
78       }
79   
80       /**
81        * The total number of partitions after the operation succeeds.
82        */
83       public int totalCount() {
84           return totalCount;
85       }
86   
87       /**
88        * The replica assignments for the new partitions, or null if the assignment will be done by the controller.
89        */
90       public List<List<Integer>> assignments() {
91           return newAssignments;
92       }
93   
94       @Override
95       public String toString() {
96           return "(totalCount=" + totalCount() + ", newAssignments=" + assignments() + ")";
97       }
98   
99   }
