47   /**
48    * <code>HttpAccessTokenRetriever</code> is an {@link AccessTokenRetriever} that will
49    * communicate with an OAuth/OIDC provider directly via HTTP to post client credentials
50    * ({@link OAuthBearerLoginCallbackHandler#CLIENT_ID_CONFIG}/{@link OAuthBearerLoginCallbackHandler#CLIENT_SECRET_CONFIG})
51    * to a publicized token endpoint URL
52    * ({@link SaslConfigs#SASL_OAUTHBEARER_TOKEN_ENDPOINT_URL}).
53    *
54    * @see AccessTokenRetriever
55    * @see OAuthBearerLoginCallbackHandler#CLIENT_ID_CONFIG
56    * @see OAuthBearerLoginCallbackHandler#CLIENT_SECRET_CONFIG
57    * @see OAuthBearerLoginCallbackHandler#SCOPE_CONFIG
58    * @see SaslConfigs#SASL_OAUTHBEARER_TOKEN_ENDPOINT_URL
59    */
60

65       private static final Set<Integer> UNRETRYABLE_HTTP_CODES;
66   
67       private static final int MAX_RESPONSE_BODY_LENGTH = 1000;
68   
69       public static final String AUTHORIZATION_HEADER = "Authorization";
70   
71       static {
72           // This does not have to be an exhaustive list. There are other HTTP codes that
73           // are defined in different RFCs (e.g. https://datatracker.ietf.org/doc/html/rfc6585)
74           // that we won't worry about yet. The worst case if a status code is missing from
75           // this set is that the request will be retried.
76           UNRETRYABLE_HTTP_CODES = new HashSet<>();
77           UNRETRYABLE_HTTP_CODES.add(HttpURLConnection.HTTP_BAD_REQUEST);
78           UNRETRYABLE_HTTP_CODES.add(HttpURLConnection.HTTP_UNAUTHORIZED);
79           UNRETRYABLE_HTTP_CODES.add(HttpURLConnection.HTTP_PAYMENT_REQUIRED);
80           UNRETRYABLE_HTTP_CODES.add(HttpURLConnection.HTTP_FORBIDDEN);
81           UNRETRYABLE_HTTP_CODES.add(HttpURLConnection.HTTP_NOT_FOUND);
82           UNRETRYABLE_HTTP_CODES.add(HttpURLConnection.HTTP_BAD_METHOD);
83           UNRETRYABLE_HTTP_CODES.add(HttpURLConnection.HTTP_NOT_ACCEPTABLE);
84           UNRETRYABLE_HTTP_CODES.add(HttpURLConnection.HTTP_PROXY_AUTH);
85           UNRETRYABLE_HTTP_CODES.add(HttpURLConnection.HTTP_CONFLICT);
86           UNRETRYABLE_HTTP_CODES.add(HttpURLConnection.HTTP_GONE);
87           UNRETRYABLE_HTTP_CODES.add(HttpURLConnection.HTTP_LENGTH_REQUIRED);
88           UNRETRYABLE_HTTP_CODES.add(HttpURLConnection.HTTP_PRECON_FAILED);
89           UNRETRYABLE_HTTP_CODES.add(HttpURLConnection.HTTP_ENTITY_TOO_LARGE);
90           UNRETRYABLE_HTTP_CODES.add(HttpURLConnection.HTTP_REQ_TOO_LONG);
91           UNRETRYABLE_HTTP_CODES.add(HttpURLConnection.HTTP_UNSUPPORTED_TYPE);
92           UNRETRYABLE_HTTP_CODES.add(HttpURLConnection.HTTP_NOT_IMPLEMENTED);
93           UNRETRYABLE_HTTP_CODES.add(HttpURLConnection.HTTP_VERSION);
94       }

134       /**
135        * Retrieves a JWT access token in its serialized three-part form. The implementation
136        * is free to determine how it should be retrieved but should not perform validation
137        * on the result.
138        *
139        * <b>Note</b>: This is a blocking function and callers should be aware that the
140        * implementation communicates over a network. The facility in the
141        * {@link javax.security.auth.spi.LoginModule} from which this is ultimately called
142        * does not provide an asynchronous approach.
143        *
144        * @return Non-<code>null</code> JWT access token string
145        *
146        * @throws IOException Thrown on errors related to IO during retrieval
147        */
148   
149       @Override
150       public String retrieve() throws IOException {
151           String authorizationHeader = formatAuthorizationHeader(clientId, clientSecret);
152           String requestBody = formatRequestBody(scope);
153           Retry<String> retry = new Retry<>(loginRetryBackoffMs, loginRetryBackoffMaxMs);
154           Map<String, String> headers = Collections.singletonMap(AUTHORIZATION_HEADER, authorizationHeader);
155   
156           String responseBody;
157   
158           try {
159               responseBody = retry.execute(() -> {
160                   HttpURLConnection con = null;
161   
162                   try {
163                       con = (HttpURLConnection) new URL(tokenEndpointUrl).openConnection();
164   
165                       if (sslSocketFactory != null && con instanceof HttpsURLConnection)
166                           ((HttpsURLConnection) con).setSSLSocketFactory(sslSocketFactory);
167   
168                       return post(con, headers, requestBody, loginConnectTimeoutMs, loginReadTimeoutMs);
169                   } catch (IOException e) {
170                       throw new ExecutionException(e);
171                   } finally {
172                       if (con != null)
173                           con.disconnect();
174                   }
175               });
176           } catch (ExecutionException e) {
177               if (e.getCause() instanceof IOException)
178                   throw (IOException) e.getCause();
179               else
180                   throw new KafkaException(e.getCause());
181           }
182   
183           return parseAccessToken(responseBody);
184       }

186       public static String post(HttpURLConnection con,
187           Map<String, String> headers,
188           String requestBody,
189           Integer connectTimeoutMs,
190           Integer readTimeoutMs)
191           throws IOException, UnretryableException {
192           handleInput(con, headers, requestBody, connectTimeoutMs, readTimeoutMs);
193           return handleOutput(con);
194       }
195   
196       private static void handleInput(HttpURLConnection con,
197           Map<String, String> headers,
198           String requestBody,
199           Integer connectTimeoutMs,
200           Integer readTimeoutMs)
201           throws IOException, UnretryableException {
202           log.debug("handleInput - starting post for {}", con.getURL());
203           con.setRequestMethod("POST");
204           con.setRequestProperty("Accept", "application/json");
205   
206           if (headers != null) {
207               for (Map.Entry<String, String> header : headers.entrySet())
208                   con.setRequestProperty(header.getKey(), header.getValue());
209           }
210   
211           con.setRequestProperty("Cache-Control", "no-cache");
212   
213           if (requestBody != null) {
214               con.setRequestProperty("Content-Length", String.valueOf(requestBody.length()));
215               con.setDoOutput(true);
216           }
217   
218           con.setUseCaches(false);
219   
220           if (connectTimeoutMs != null)
221               con.setConnectTimeout(connectTimeoutMs);
222   
223           if (readTimeoutMs != null)
224               con.setReadTimeout(readTimeoutMs);
225   
226           log.debug("handleInput - preparing to connect to {}", con.getURL());
227           con.connect();
228   
229           if (requestBody != null) {
230               try (OutputStream os = con.getOutputStream()) {
231                   ByteArrayInputStream is = new ByteArrayInputStream(requestBody.getBytes(StandardCharsets.UTF_8));
232                   log.debug("handleInput - preparing to write request body to {}", con.getURL());
233                   copy(is, os);
234               }
235           }
236       }
237   
238       static String handleOutput(final HttpURLConnection con) throws IOException {
239           int responseCode = con.getResponseCode();
240           log.debug("handleOutput - responseCode: {}", responseCode);
241   
242           String responseBody = null;
243   
244           try (InputStream is = con.getInputStream()) {
245               ByteArrayOutputStream os = new ByteArrayOutputStream();
246               log.debug("handleOutput - preparing to read response body from {}", con.getURL());
247               copy(is, os);
248               responseBody = os.toString(StandardCharsets.UTF_8.name());
249           } catch (Exception e) {
250               log.warn("handleOutput - error retrieving data", e);
251           }
252   
253           if (responseCode == HttpURLConnection.HTTP_OK || responseCode == HttpURLConnection.HTTP_CREATED) {
254               log.debug("handleOutput - responseCode: {}, response: {}", responseCode, responseBody);
255   
256               if (responseBody == null || responseBody.isEmpty())
257                   throw new IOException(String.format("The token endpoint response was unexpectedly empty despite response code %s from %s", responseCode, con.getURL()));
258   
259               return responseBody;
260           } else {
261               log.warn("handleOutput - error response code: {}, error response body: {}", responseCode, responseBody);
262   
263               if (UNRETRYABLE_HTTP_CODES.contains(responseCode)) {
264                   // We know that this is a non-transient error, so let's not keep retrying the
265                   // request unnecessarily.
266                   throw new UnretryableException(new IOException(String.format("The response code %s was encountered reading the token endpoint response; will not attempt further retries", responseCode)));
267               } else {
268                   // We don't know if this is a transient (retryable) error or not, so let's assume
269                   // it is.
270                   throw new IOException(String.format("The unexpected response code %s was encountered reading the token endpoint response", responseCode));
271               }
272           }
273       }

275       static void copy(InputStream is, OutputStream os) throws IOException {
276           byte[] buf = new byte[4096];
277           int b;
278   
279           while ((b = is.read(buf)) != -1)
280               os.write(buf, 0, b);
281       }

283       static String parseAccessToken(String responseBody) throws IOException {
284           log.debug("parseAccessToken - responseBody: {}", responseBody);
285           ObjectMapper mapper = new ObjectMapper();
286           JsonNode rootNode = mapper.readTree(responseBody);
287           JsonNode accessTokenNode = rootNode.at("/access_token");
288   
289           if (accessTokenNode == null) {
290               // Only grab the first N characters so that if the response body is huge, we don't
291               // blow up.
292               String snippet = responseBody;
293   
294               if (snippet.length() > MAX_RESPONSE_BODY_LENGTH) {
295                   int actualLength = responseBody.length();
296                   String s = responseBody.substring(0, MAX_RESPONSE_BODY_LENGTH);
297                   snippet = String.format("%s (trimmed to first %s characters out of %s total)", s, MAX_RESPONSE_BODY_LENGTH, actualLength);
298               }
299   
300               throw new IOException(String.format("The token endpoint response did not contain an access_token value. Response: (%s)", snippet));
301           }
302   
303           return sanitizeString("the token endpoint response's access_token JSON attribute", accessTokenNode.textValue());
304       }

306       static String formatAuthorizationHeader(String clientId, String clientSecret) {
307           clientId = sanitizeString("the token endpoint request client ID parameter", clientId);
308           clientSecret = sanitizeString("the token endpoint request client secret parameter", clientSecret);
309   
310           String s = String.format("%s:%s", clientId, clientSecret);
311           String encoded = Base64.getUrlEncoder().encodeToString(Utils.utf8(s));
312           return String.format("Basic %s", encoded);
313       }

315       static String formatRequestBody(String scope) throws IOException {
316           try {
317               StringBuilder requestParameters = new StringBuilder();
318               requestParameters.append("grant_type=client_credentials");
319   
320               if (scope != null && !scope.trim().isEmpty()) {
321                   scope = scope.trim();
322                   String encodedScope = URLEncoder.encode(scope, StandardCharsets.UTF_8.name());
323                   requestParameters.append("&scope=").append(encodedScope);
324               }
325   
326               return requestParameters.toString();
327           } catch (UnsupportedEncodingException e) {
328               // The world has gone crazy!
329               throw new IOException(String.format("Encoding %s not supported", StandardCharsets.UTF_8.name()));
330           }
331       }

333       private static String sanitizeString(String name, String value) {
334           if (value == null)
335               throw new IllegalArgumentException(String.format("The value for %s must be non-null", name));
336   
337           if (value.isEmpty())
338               throw new IllegalArgumentException(String.format("The value for %s must be non-empty", name));
339   
340           value = value.trim();
341   
342           if (value.isEmpty())
343               throw new IllegalArgumentException(String.format("The value for %s must not contain only whitespace", name));
344   
345           return value;
346       }
