17   package org.apache.kafka.common.protocol;
18   
19   import org.apache.kafka.common.network.ByteBufferSend;
20   import org.apache.kafka.common.network.Send;
21   import org.apache.kafka.common.record.BaseRecords;
22   import org.apache.kafka.common.record.MemoryRecords;
23   import org.apache.kafka.common.record.MultiRecordsSend;
24   import org.apache.kafka.common.record.UnalignedMemoryRecords;
25   import org.apache.kafka.common.requests.RequestHeader;
26   import org.apache.kafka.common.requests.ResponseHeader;
27   import org.apache.kafka.common.utils.ByteUtils;
28   
29   import java.nio.ByteBuffer;
30   import java.util.ArrayDeque;
31   import java.util.ArrayList;
32   import java.util.List;
33   import java.util.Queue;

35   /**
36    * This class provides a way to build {@link Send} objects for network transmission
37    * from generated {@link org.apache.kafka.common.protocol.ApiMessage} types without
38    * allocating new space for "zero-copy" fields (see {@link #writeByteBuffer(ByteBuffer)}
39    * and {@link #writeRecords(BaseRecords)}).
40    *
41    * See {@link org.apache.kafka.common.requests.EnvelopeRequest#toSend(RequestHeader)}
42    * for example usage.
43    */
44   public class SendBuilder implements Writable {
45       private final ByteBuffer buffer;
46   
47       private final Queue<Send> sends = new ArrayDeque<>(1);
48       private long sizeOfSends = 0;
49   
50       private final List<ByteBuffer> buffers = new ArrayList<>();
51       private long sizeOfBuffers = 0;
52   
53       SendBuilder(int size) {
54           this.buffer = ByteBuffer.allocate(size);
55           this.buffer.mark();
56       }
57   
58       @Override
59       public void writeByte(byte val) {
60           buffer.put(val);
61       }
62   
63       @Override
64       public void writeShort(short val) {
65           buffer.putShort(val);
66       }
67   
68       @Override
69       public void writeInt(int val) {
70           buffer.putInt(val);
71       }
72   
73       @Override
74       public void writeLong(long val) {
75           buffer.putLong(val);
76       }
77   
78       @Override
79       public void writeDouble(double val) {
80           buffer.putDouble(val);
81       }
82   
83       @Override
84       public void writeByteArray(byte[] arr) {
85           buffer.put(arr);
86       }
87   
88       @Override
89       public void writeUnsignedVarint(int i) {
90           ByteUtils.writeUnsignedVarint(i, buffer);
91       }
92   
93       /**
94        * Write a byte buffer. The reference to the underlying buffer will
95        * be retained in the result of {@link #build()}.
96        *
97        * @param buf the buffer to write
98        */
99       @Override
100       public void writeByteBuffer(ByteBuffer buf) {
101           flushPendingBuffer();
102           addBuffer(buf.duplicate());
103       }
104   
105       @Override
106       public void writeVarint(int i) {
107           ByteUtils.writeVarint(i, buffer);
108       }
109   
110       @Override
111       public void writeVarlong(long i) {
112           ByteUtils.writeVarlong(i, buffer);
113       }
114   
115       private void addBuffer(ByteBuffer buffer) {
116           buffers.add(buffer);
117           sizeOfBuffers += buffer.remaining();
118       }
119   
120       private void addSend(Send send) {
121           sends.add(send);
122           sizeOfSends += send.size();
123       }
124   
125       private void clearBuffers() {
126           buffers.clear();
127           sizeOfBuffers = 0;
128       }
129   
130       /**
131        * Write a record set. The underlying record data will be retained
132        * in the result of {@link #build()}. See {@link BaseRecords#toSend()}.
133        *
134        * @param records the records to write
135        */
136       @Override
137       public void writeRecords(BaseRecords records) {
138           if (records instanceof MemoryRecords) {
139               flushPendingBuffer();
140               addBuffer(((MemoryRecords) records).buffer());
141           } else if (records instanceof UnalignedMemoryRecords) {
142               flushPendingBuffer();
143               addBuffer(((UnalignedMemoryRecords) records).buffer());
144           } else {
145               flushPendingSend();
146               addSend(records.toSend());
147           }
148       }
149   
150       private void flushPendingSend() {
151           flushPendingBuffer();
152           if (!buffers.isEmpty()) {
153               ByteBuffer[] byteBufferArray = buffers.toArray(new ByteBuffer[0]);
154               addSend(new ByteBufferSend(byteBufferArray, sizeOfBuffers));
155               clearBuffers();
156           }
157       }
158   
159       private void flushPendingBuffer() {
160           int latestPosition = buffer.position();
161           buffer.reset();
162   
163           if (latestPosition > buffer.position()) {
164               buffer.limit(latestPosition);
165               addBuffer(buffer.slice());
166   
167               buffer.position(latestPosition);
168               buffer.limit(buffer.capacity());
169               buffer.mark();
170           }
171       }
172   
173       public Send build() {
174           flushPendingSend();
175   
176           if (sends.size() == 1) {
177               return sends.poll();
178           } else {
179               return new MultiRecordsSend(sends, sizeOfSends);
180           }
181       }
182   
183       public static Send buildRequestSend(
184           RequestHeader header,
185           Message apiRequest
186       ) {
187           return buildSend(
188               header.data(),
189               header.headerVersion(),
190               apiRequest,
191               header.apiVersion()
192           );
193       }
194   
195       public static Send buildResponseSend(
196           ResponseHeader header,
197           Message apiResponse,
198           short apiVersion
199       ) {
200           return buildSend(
201               header.data(),
202               header.headerVersion(),
203               apiResponse,
204               apiVersion
205           );
206       }
207   
208       private static Send buildSend(
209           Message header,
210           short headerVersion,
211           Message apiMessage,
212           short apiVersion
213       ) {
214           ObjectSerializationCache serializationCache = new ObjectSerializationCache();
215   
216           MessageSizeAccumulator messageSize = new MessageSizeAccumulator();
217           header.addSize(messageSize, serializationCache, headerVersion);
218           apiMessage.addSize(messageSize, serializationCache, apiVersion);
219   
220           SendBuilder builder = new SendBuilder(messageSize.sizeExcludingZeroCopy() + 4);
221           builder.writeInt(messageSize.totalSize());
222           header.write(builder, serializationCache, headerVersion);
223           apiMessage.write(builder, serializationCache, apiVersion);
224   
225           return builder.build();
226       }
