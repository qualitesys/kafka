35   /**
36    * A pool of ByteBuffers kept under a given memory limit. This class is fairly specific to the needs of the producer. In
37    * particular it has the following properties:
38    * <ol>
39    * <li>There is a special "poolable size" and buffers of this size are kept in a free list and recycled
40    * <li>It is fair. That is all memory is given to the longest waiting thread until it has sufficient memory. This
41    * prevents starvation or deadlock when a thread asks for a large chunk of memory and needs to block until multiple
42    * buffers are deallocated.
43    * </ol>
44    */
45   public class BufferPool {
46   
47       static final String WAIT_TIME_SENSOR_NAME = "bufferpool-wait-time";
48   
49       private final long totalMemory;
50       private final int poolableSize;
51       private final ReentrantLock lock;
52       private final Deque<ByteBuffer> free;
53       private final Deque<Condition> waiters;
54       /** Total available memory is the sum of nonPooledAvailableMemory and the number of byte buffers in free * poolableSize.  */
55       private long nonPooledAvailableMemory;
56       private final Metrics metrics;
57       private final Time time;
58       private final Sensor waitTime;
59       private boolean closed;
60   
61       /**
62        * Create a new buffer pool
63        *
64        * @param memory The maximum amount of memory that this buffer pool can allocate
65        * @param poolableSize The buffer size to cache in the free list rather than deallocating
66        * @param metrics instance of Metrics
67        * @param time time instance
68        * @param metricGrpName logical group name for metrics
69        */
70       public BufferPool(long memory, int poolableSize, Metrics metrics, Time time, String metricGrpName) {
71           this.poolableSize = poolableSize;
72           this.lock = new ReentrantLock();
73           this.free = new ArrayDeque<>();
74           this.waiters = new ArrayDeque<>();
75           this.totalMemory = memory;
76           this.nonPooledAvailableMemory = memory;
77           this.metrics = metrics;
78           this.time = time;
79           this.waitTime = this.metrics.sensor(WAIT_TIME_SENSOR_NAME);
80           MetricName rateMetricName = metrics.metricName("bufferpool-wait-ratio",
81                                                      metricGrpName,
82                                                      "The fraction of time an appender waits for space allocation.");
83           MetricName totalMetricName = metrics.metricName("bufferpool-wait-time-total",
84                                                      metricGrpName,
85                                                      "*Deprecated* The total time an appender waits for space allocation.");
86           MetricName totalNsMetricName = metrics.metricName("bufferpool-wait-time-ns-total",
87                                                       metricGrpName,
88                                                       "The total time in nanoseconds an appender waits for space allocation.");
89   
90           Sensor bufferExhaustedRecordSensor = metrics.sensor("buffer-exhausted-records");
91           MetricName bufferExhaustedRateMetricName = metrics.metricName("buffer-exhausted-rate", metricGrpName, "The average per-second number of record sends that are dropped due to buffer exhaustion");
92           MetricName bufferExhaustedTotalMetricName = metrics.metricName("buffer-exhausted-total", metricGrpName, "The total number of record sends that are dropped due to buffer exhaustion");
93           bufferExhaustedRecordSensor.add(new Meter(bufferExhaustedRateMetricName, bufferExhaustedTotalMetricName));
94   
95           this.waitTime.add(new Meter(TimeUnit.NANOSECONDS, rateMetricName, totalMetricName));
96           this.waitTime.add(new Meter(TimeUnit.NANOSECONDS, rateMetricName, totalNsMetricName));
97           this.closed = false;
98       }
99   
100       /**
101        * Allocate a buffer of the given size. This method blocks if there is not enough memory and the buffer pool
102        * is configured with blocking mode.
103        *
104        * @param size The buffer size to allocate in bytes
105        * @param maxTimeToBlockMs The maximum time in milliseconds to block for buffer memory to be available
106        * @return The buffer
107        * @throws InterruptedException If the thread is interrupted while blocked
108        * @throws IllegalArgumentException if size is larger than the total memory controlled by the pool (and hence we would block
109        *         forever)
110        */
111       public ByteBuffer allocate(int size, long maxTimeToBlockMs) throws InterruptedException {
112           if (size > this.totalMemory)
113               throw new IllegalArgumentException("Attempt to allocate " + size
114                                                  + " bytes, but there is a hard limit of "
115                                                  + this.totalMemory
116                                                  + " on memory allocations.");
117   
118           ByteBuffer buffer = null;
119           this.lock.lock();
120   
121           if (this.closed) {
122               this.lock.unlock();
123               throw new KafkaException("Producer closed while allocating memory");
124           }
125   
126           try {
127               // check if we have a free buffer of the right size pooled
128               if (size == poolableSize && !this.free.isEmpty())
129                   return this.free.pollFirst();
130   
131               // now check if the request is immediately satisfiable with the
132               // memory on hand or if we need to block
133               int freeListSize = freeSize() * this.poolableSize;
134               if (this.nonPooledAvailableMemory + freeListSize >= size) {
135                   // we have enough unallocated or pooled memory to immediately
136                   // satisfy the request, but need to allocate the buffer
137                   freeUp(size);
138                   this.nonPooledAvailableMemory -= size;
139               } else {
140                   // we are out of memory and will have to block
141                   int accumulated = 0;
142                   Condition moreMemory = this.lock.newCondition();
143                   try {
144                       long remainingTimeToBlockNs = TimeUnit.MILLISECONDS.toNanos(maxTimeToBlockMs);
145                       this.waiters.addLast(moreMemory);
146                       // loop over and over until we have a buffer or have reserved
147                       // enough memory to allocate one
148                       while (accumulated < size) {
149                           long startWaitNs = time.nanoseconds();
150                           long timeNs;
151                           boolean waitingTimeElapsed;
152                           try {
153                               waitingTimeElapsed = !moreMemory.await(remainingTimeToBlockNs, TimeUnit.NANOSECONDS);
154                           } finally {
155                               long endWaitNs = time.nanoseconds();
156                               timeNs = Math.max(0L, endWaitNs - startWaitNs);
157                               recordWaitTime(timeNs);
158                           }
159   
160                           if (this.closed)
161                               throw new KafkaException("Producer closed while allocating memory");
162   
163                           if (waitingTimeElapsed) {
164                               this.metrics.sensor("buffer-exhausted-records").record();
165                               throw new BufferExhaustedException("Failed to allocate " + size + " bytes within the configured max blocking time "
166                                   + maxTimeToBlockMs + " ms. Total memory: " + totalMemory() + " bytes. Available memory: " + availableMemory()
167                                   + " bytes. Poolable size: " + poolableSize() + " bytes");
168                           }
169   
170                           remainingTimeToBlockNs -= timeNs;
171   
172                           // check if we can satisfy this request from the free list,
173                           // otherwise allocate memory
174                           if (accumulated == 0 && size == this.poolableSize && !this.free.isEmpty()) {
175                               // just grab a buffer from the free list
176                               buffer = this.free.pollFirst();
177                               accumulated = size;
178                           } else {
179                               // we'll need to allocate memory, but we may only get
180                               // part of what we need on this iteration
181                               freeUp(size - accumulated);
182                               int got = (int) Math.min(size - accumulated, this.nonPooledAvailableMemory);
183                               this.nonPooledAvailableMemory -= got;
184                               accumulated += got;
185                           }
186                       }
187                       // Don't reclaim memory on throwable since nothing was thrown
188                       accumulated = 0;
189                   } finally {
190                       // When this loop was not able to successfully terminate don't loose available memory
191                       this.nonPooledAvailableMemory += accumulated;
192                       this.waiters.remove(moreMemory);
193                   }
194               }
195           } finally {
196               // signal any additional waiters if there is more memory left
197               // over for them
198               try {
199                   if (!(this.nonPooledAvailableMemory == 0 && this.free.isEmpty()) && !this.waiters.isEmpty())
200                       this.waiters.peekFirst().signal();
201               } finally {
202                   // Another finally... otherwise find bugs complains
203                   lock.unlock();
204               }
205           }
206   
207           if (buffer == null)
208               return safeAllocateByteBuffer(size);
209           else
210               return buffer;
211       }
212   
213       // Protected for testing
214       protected void recordWaitTime(long timeNs) {
215           this.waitTime.record(timeNs, time.milliseconds());
216       }
217   
218       /**
219        * Allocate a buffer.  If buffer allocation fails (e.g. because of OOM) then return the size count back to
220        * available memory and signal the next waiter if it exists.
221        */
222       private ByteBuffer safeAllocateByteBuffer(int size) {
223           boolean error = true;
224           try {
225               ByteBuffer buffer = allocateByteBuffer(size);
226               error = false;
227               return buffer;
228           } finally {
229               if (error) {
230                   this.lock.lock();
231                   try {
232                       this.nonPooledAvailableMemory += size;
233                       if (!this.waiters.isEmpty())
234                           this.waiters.peekFirst().signal();
235                   } finally {
236                       this.lock.unlock();
237                   }
238               }
239           }
240       }
241   
242       // Protected for testing.
243       protected ByteBuffer allocateByteBuffer(int size) {
244           return ByteBuffer.allocate(size);
245       }
246   
247       /**
248        * Attempt to ensure we have at least the requested number of bytes of memory for allocation by deallocating pooled
249        * buffers (if needed)
250        */
251       private void freeUp(int size) {
252           while (!this.free.isEmpty() && this.nonPooledAvailableMemory < size)
253               this.nonPooledAvailableMemory += this.free.pollLast().capacity();
254       }
255   
256       /**
257        * Return buffers to the pool. If they are of the poolable size add them to the free list, otherwise just mark the
258        * memory as free.
259        *
260        * @param buffer The buffer to return
261        * @param size The size of the buffer to mark as deallocated, note that this may be smaller than buffer.capacity
262        *             since the buffer may re-allocate itself during in-place compression
263        */
264       public void deallocate(ByteBuffer buffer, int size) {
265           lock.lock();
266           try {
267               if (size == this.poolableSize && size == buffer.capacity()) {
268                   buffer.clear();
269                   this.free.add(buffer);
270               } else {
271                   this.nonPooledAvailableMemory += size;
272               }
273               Condition moreMem = this.waiters.peekFirst();
274               if (moreMem != null)
275                   moreMem.signal();
276           } finally {
277               lock.unlock();
278           }
279       }
280   
281       public void deallocate(ByteBuffer buffer) {
282           if (buffer != null)
283               deallocate(buffer, buffer.capacity());
284       }
285   
286       /**
287        * the total free memory both unallocated and in the free list
288        */
289       public long availableMemory() {
290           lock.lock();
291           try {
292               return this.nonPooledAvailableMemory + freeSize() * (long) this.poolableSize;
293           } finally {
294               lock.unlock();
295           }
296       }
297   
298       // Protected for testing.
299       protected int freeSize() {
300           return this.free.size();
301       }
302   
303       /**
304        * Get the unallocated memory (not in the free list or in use)
305        */
306       public long unallocatedMemory() {
307           lock.lock();
308           try {
309               return this.nonPooledAvailableMemory;
310           } finally {
311               lock.unlock();
312           }
313       }
314   
315       /**
316        * The number of threads blocked waiting on memory
317        */
318       public int queued() {
319           lock.lock();
320           try {
321               return this.waiters.size();
322           } finally {
323               lock.unlock();
324           }
325       }
326   
327       /**
328        * The buffer size that will be retained in the free list after use
329        */
330       public int poolableSize() {
331           return this.poolableSize;
332       }
333   
334       /**
335        * The total memory managed by this pool
336        */
337       public long totalMemory() {
338           return this.totalMemory;
339       }
340   
341       // package-private method used only for testing
342       Deque<Condition> waiters() {
343           return this.waiters;
344       }
345   
346       /**
347        * Closes the buffer pool. Memory will be prevented from being allocated, but may be deallocated. All allocations
348        * awaiting available memory will be notified to abort.
349        */
350       public void close() {
351           this.lock.lock();
352           this.closed = true;
353           try {
354               for (Condition waiter : this.waiters)
355                   waiter.signal();
356           } finally {
357               this.lock.unlock();
358           }
359       }
