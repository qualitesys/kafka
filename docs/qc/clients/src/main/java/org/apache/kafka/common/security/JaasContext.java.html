38       private static final String GLOBAL_CONTEXT_NAME_SERVER = "KafkaServer";
39       private static final String GLOBAL_CONTEXT_NAME_CLIENT = "KafkaClient";

41       /**
42        * Returns an instance of this class.
43        *
44        * The context will contain the configuration specified by the JAAS configuration property
45        * {@link SaslConfigs#SASL_JAAS_CONFIG} with prefix `listener.name.{listenerName}.{mechanism}.`
46        * with listenerName and mechanism in lower case. The context `KafkaServer` will be returned
47        * with a single login context entry loaded from the property.
48        * <p>
49        * If the property is not defined, the context will contain the default Configuration and
50        * the context name will be one of:
51        * <ol>
52        *   <li>Lowercased listener name followed by a period and the string `KafkaServer`</li>
53        *   <li>The string `KafkaServer`</li>
54        *  </ol>
55        * If both are valid entries in the default JAAS configuration, the first option is chosen.
56        * </p>
57        *
58        * @throws IllegalArgumentException if listenerName or mechanism is not defined.
59        */
60       public static JaasContext loadServerContext(ListenerName listenerName, String mechanism, Map<String, ?> configs) {
61           if (listenerName == null)
62               throw new IllegalArgumentException("listenerName should not be null for SERVER");
63           if (mechanism == null)
64               throw new IllegalArgumentException("mechanism should not be null for SERVER");
65           String listenerContextName = listenerName.value().toLowerCase(Locale.ROOT) + "." + GLOBAL_CONTEXT_NAME_SERVER;
66           Password dynamicJaasConfig = (Password) configs.get(mechanism.toLowerCase(Locale.ROOT) + "." + SaslConfigs.SASL_JAAS_CONFIG);
67           if (dynamicJaasConfig == null && configs.get(SaslConfigs.SASL_JAAS_CONFIG) != null)
68               LOG.warn("Server config {} should be prefixed with SASL mechanism name, ignoring config", SaslConfigs.SASL_JAAS_CONFIG);
69           return load(Type.SERVER, listenerContextName, GLOBAL_CONTEXT_NAME_SERVER, dynamicJaasConfig);
70       }
71   
72       /**
73        * Returns an instance of this class.
74        *
75        * If JAAS configuration property @link SaslConfigs#SASL_JAAS_CONFIG} is specified,
76        * the configuration object is created by parsing the property value. Otherwise, the default Configuration
77        * is returned. The context name is always `KafkaClient`.
78        *
79        */
80       public static JaasContext loadClientContext(Map<String, ?> configs) {
81           Password dynamicJaasConfig = (Password) configs.get(SaslConfigs.SASL_JAAS_CONFIG);
82           return load(JaasContext.Type.CLIENT, null, GLOBAL_CONTEXT_NAME_CLIENT, dynamicJaasConfig);
83       }

85       static JaasContext load(JaasContext.Type contextType, String listenerContextName,
86                               String globalContextName, Password dynamicJaasConfig) {
87           if (dynamicJaasConfig != null) {
88               JaasConfig jaasConfig = new JaasConfig(globalContextName, dynamicJaasConfig.value());
89               AppConfigurationEntry[] contextModules = jaasConfig.getAppConfigurationEntry(globalContextName);
90               if (contextModules == null || contextModules.length == 0)
91                   throw new IllegalArgumentException("JAAS config property does not contain any login modules");
92               else if (contextModules.length != 1)
93                   throw new IllegalArgumentException("JAAS config property contains " + contextModules.length + " login modules, should be 1 module");
94               return new JaasContext(globalContextName, contextType, jaasConfig, dynamicJaasConfig);
95           } else
96               return defaultContext(contextType, listenerContextName, globalContextName);
97       }
98   
99       private static JaasContext defaultContext(JaasContext.Type contextType, String listenerContextName,
100                                                 String globalContextName) {
101           String jaasConfigFile = System.getProperty(JaasUtils.JAVA_LOGIN_CONFIG_PARAM);
102           if (jaasConfigFile == null) {
103               if (contextType == Type.CLIENT) {
104                   LOG.debug("System property '" + JaasUtils.JAVA_LOGIN_CONFIG_PARAM + "' and Kafka SASL property '" +
105                           SaslConfigs.SASL_JAAS_CONFIG + "' are not set, using default JAAS configuration.");
106               } else {
107                   LOG.debug("System property '" + JaasUtils.JAVA_LOGIN_CONFIG_PARAM + "' is not set, using default JAAS " +
108                           "configuration.");
109               }
110           }
111   
112           Configuration jaasConfig = Configuration.getConfiguration();
113   
114           AppConfigurationEntry[] configEntries = null;
115           String contextName = globalContextName;
116   
117           if (listenerContextName != null) {
118               configEntries = jaasConfig.getAppConfigurationEntry(listenerContextName);
119               if (configEntries != null)
120                   contextName = listenerContextName;
121           }
122   
123           if (configEntries == null)
124               configEntries = jaasConfig.getAppConfigurationEntry(globalContextName);
125   
126           if (configEntries == null) {
127               String listenerNameText = listenerContextName == null ? "" : " or '" + listenerContextName + "'";
128               String errorMessage = "Could not find a '" + globalContextName + "'" + listenerNameText + " entry in the JAAS " +
129                       "configuration. System property '" + JaasUtils.JAVA_LOGIN_CONFIG_PARAM + "' is " +
130                       (jaasConfigFile == null ? "not set" : jaasConfigFile);
131               throw new IllegalArgumentException(errorMessage);
132           }
133   
134           return new JaasContext(contextName, contextType, jaasConfig, null);
135       }

137       /**
138        * The type of the SASL login context, it should be SERVER for the broker and CLIENT for the clients (consumer, producer,
139        * etc.). This is used to validate behaviour (e.g. some functionality is only available in the broker or clients).
140        */
141       public enum Type { CLIENT, SERVER }
142

144       private final Type type;
145       private final Configuration configuration;
146       private final List<AppConfigurationEntry> configurationEntries;
147       private final Password dynamicJaasConfig;

149       public JaasContext(String name, Type type, Configuration configuration, Password dynamicJaasConfig) {
150           this.name = name;
151           this.type = type;
152           this.configuration = configuration;
153           AppConfigurationEntry[] entries = configuration.getAppConfigurationEntry(name);
154           if (entries == null)
155               throw new IllegalArgumentException("Could not find a '" + name + "' entry in this JAAS configuration.");
156           this.configurationEntries = Collections.unmodifiableList(new ArrayList<>(Arrays.asList(entries)));
157           this.dynamicJaasConfig = dynamicJaasConfig;
158       }

180       /**
181        * Returns the configuration option for <code>key</code> from this context.
182        * If login module name is specified, return option value only from that module.
183        */
184       public static String configEntryOption(List<AppConfigurationEntry> configurationEntries, String key, String loginModuleName) {
185           for (AppConfigurationEntry entry : configurationEntries) {
186               if (loginModuleName != null && !loginModuleName.equals(entry.getLoginModuleName()))
187                   continue;
188               Object val = entry.getOptions().get(key);
189               if (val != null)
190                   return (String) val;
191           }
192           return null;
193       }
