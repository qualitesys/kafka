18   
19   import org.apache.kafka.common.errors.GroupAuthorizationException;
20   import org.apache.kafka.common.errors.InvalidGroupIdException;
21   import org.apache.kafka.common.errors.TransactionalIdAuthorizationException;
22   import org.apache.kafka.common.message.FindCoordinatorRequestData;
23   import org.apache.kafka.common.message.FindCoordinatorResponseData.Coordinator;
24   import org.apache.kafka.common.protocol.Errors;
25   import org.apache.kafka.common.requests.AbstractResponse;
26   import org.apache.kafka.common.requests.FindCoordinatorRequest;
27   import org.apache.kafka.common.requests.FindCoordinatorRequest.CoordinatorType;
28   import org.apache.kafka.common.requests.FindCoordinatorResponse;
29   import org.apache.kafka.common.utils.LogContext;
30   import org.slf4j.Logger;
31   
32   import java.util.Collections;
33   import java.util.HashMap;
34   import java.util.Map;
35   import java.util.Objects;
36   import java.util.Set;
37   import java.util.stream.Collectors;
38   
39   public class CoordinatorStrategy implements AdminApiLookupStrategy<CoordinatorKey> {
40   
41       private static final ApiRequestScope BATCH_REQUEST_SCOPE = new ApiRequestScope() { };
42   
43       private final Logger log;
44       private final FindCoordinatorRequest.CoordinatorType type;
45       private Set<CoordinatorKey> unrepresentableKeys = Collections.emptySet();
46   
47       boolean batch = true;
48   
49       public CoordinatorStrategy(
50           FindCoordinatorRequest.CoordinatorType type,
51           LogContext logContext
52       ) {
53           this.type = type;
54           this.log = logContext.logger(CoordinatorStrategy.class);
55       }
56   
57       @Override
58       public ApiRequestScope lookupScope(CoordinatorKey key) {
59           if (batch) {
60               return BATCH_REQUEST_SCOPE;
61           } else {
62               // If the `FindCoordinator` API does not support batched lookups, we use a
63               // separate lookup context for each coordinator key we need to lookup
64               return new LookupRequestScope(key);
65           }
66       }
67   
68       @Override
69       public FindCoordinatorRequest.Builder buildRequest(Set<CoordinatorKey> keys) {
70           unrepresentableKeys = keys.stream().filter(k -> k == null || !isRepresentableKey(k.idValue)).collect(Collectors.toSet());
71           Set<CoordinatorKey> representableKeys = keys.stream().filter(k -> k != null && isRepresentableKey(k.idValue)).collect(Collectors.toSet());
72           if (batch) {
73               ensureSameType(representableKeys);
74               FindCoordinatorRequestData data = new FindCoordinatorRequestData()
75                       .setKeyType(type.id())
76                       .setCoordinatorKeys(representableKeys.stream().map(k -> k.idValue).collect(Collectors.toList()));
77               return new FindCoordinatorRequest.Builder(data);
78           } else {
79               CoordinatorKey key = requireSingletonAndType(representableKeys);
80               return new FindCoordinatorRequest.Builder(
81                   new FindCoordinatorRequestData()
82                       .setKey(key.idValue)
83                       .setKeyType(key.type.id())
84               );
85           }
86       }
87   
88       @Override
89       public LookupResult<CoordinatorKey> handleResponse(
90           Set<CoordinatorKey> keys,
91           AbstractResponse abstractResponse
92       ) {
93           Map<CoordinatorKey, Integer> mappedKeys = new HashMap<>();
94           Map<CoordinatorKey, Throwable> failedKeys = new HashMap<>();
95   
96           for (CoordinatorKey key : unrepresentableKeys) {
97               failedKeys.put(key, new InvalidGroupIdException("The given group id '" +
98                   key.idValue + "' cannot be represented in a request."));
99           }
100   
101           for (Coordinator coordinator : ((FindCoordinatorResponse) abstractResponse).coordinators()) {
102               CoordinatorKey key;
103               if (coordinator.key() == null) // old version without batching
104                   key = requireSingletonAndType(keys);
105               else {
106                   key = (type == CoordinatorType.GROUP)
107                           ? CoordinatorKey.byGroupId(coordinator.key())
108                           : CoordinatorKey.byTransactionalId(coordinator.key());
109               }
110               handleError(Errors.forCode(coordinator.errorCode()),
111                           key,
112                           coordinator.nodeId(),
113                           mappedKeys,
114                           failedKeys);
115           }
116           return new LookupResult<>(failedKeys, mappedKeys);
117       }
118   
119       public void disableBatch() {
120           batch = false;
121       }
122   
123       private CoordinatorKey requireSingletonAndType(Set<CoordinatorKey> keys) {
124           if (keys.size() != 1) {
125               throw new IllegalArgumentException("Unexpected size of key set: expected 1, but got " + keys.size());
126           }
127           CoordinatorKey key = keys.iterator().next();
128           if (key.type != type) {
129               throw new IllegalArgumentException("Unexpected key type: expected key to be of type " + type + ", but got " + key.type);
130           }
131           return key;
132       }
133   
134       private void ensureSameType(Set<CoordinatorKey> keys) {
135           if (keys.size() < 1) {
136               throw new IllegalArgumentException("Unexpected size of key set: expected >= 1, but got " + keys.size());
137           }
138           if (keys.stream().filter(k -> k.type == type).collect(Collectors.toSet()).size() != keys.size()) {
139               throw new IllegalArgumentException("Unexpected key set: expected all key to be of type " + type + ", but some key were not");
140           }
141       }
142   
143       private static boolean isRepresentableKey(String groupId) {
144           return groupId != null;
145       }
146   
147       private void handleError(Errors error, CoordinatorKey key, int nodeId, Map<CoordinatorKey, Integer> mappedKeys, Map<CoordinatorKey, Throwable> failedKeys) {
148           switch (error) {
149               case NONE:
150                   mappedKeys.put(key, nodeId);
151                   break;
152               case COORDINATOR_NOT_AVAILABLE:
153               case COORDINATOR_LOAD_IN_PROGRESS:
154                   log.debug("FindCoordinator request for key {} returned topic-level error {}. Will retry",
155                       key, error);
156                   break;
157               case GROUP_AUTHORIZATION_FAILED:
158                   failedKeys.put(key, new GroupAuthorizationException("FindCoordinator request for groupId " +
159                       "`" + key + "` failed due to authorization failure", key.idValue));
160                   break;
161               case TRANSACTIONAL_ID_AUTHORIZATION_FAILED:
162                   failedKeys.put(key, new TransactionalIdAuthorizationException("FindCoordinator request for " +
163                       "transactionalId `" + key + "` failed due to authorization failure"));
164                   break;
165               default:
166                   failedKeys.put(key, error.exception("FindCoordinator request for key " +
167                       "`" + key + "` failed due to an unexpected error"));
168           }
169       }
170   
171       private static class LookupRequestScope implements ApiRequestScope {
172           final CoordinatorKey key;
173   
174           private LookupRequestScope(CoordinatorKey key) {
175               this.key = key;
176           }
177   
178           @Override
179           public boolean equals(Object o) {
180               if (this == o) return true;
181               if (o == null || getClass() != o.getClass()) return false;
182               LookupRequestScope that = (LookupRequestScope) o;
183               return Objects.equals(key, that.key);
184           }
185   
186           @Override
187           public int hashCode() {
188               return Objects.hash(key);
189           }
190       }
191   }
