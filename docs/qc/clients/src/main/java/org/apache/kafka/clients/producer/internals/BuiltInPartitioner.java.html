32   /**
33    * Built-in default partitioner.  Note, that this is just a utility class that is used directly from
34    * RecordAccumulator, it does not implement the Partitioner interface.
35    *
36    * The class keeps track of various bookkeeping information required for adaptive sticky partitioning
37    * (described in detail in KIP-794).  There is one partitioner object per topic.
38    */
39   public class BuiltInPartitioner {
40       private final Logger log;
41       private final String topic;
42       private final int stickyBatchSize;
43   
44       private volatile PartitionLoadStats partitionLoadStats = null;
45       private final AtomicReference<StickyPartitionInfo> stickyPartitionInfo = new AtomicReference<>();
46   
47       // Visible and used for testing only.
48       static volatile public Supplier<Integer> mockRandom = null;
49   
50       /**
51        * BuiltInPartitioner constructor.
52        *
53        * @param topic The topic
54        * @param stickyBatchSize How much to produce to partition before switch
55        */
56       public BuiltInPartitioner(LogContext logContext, String topic, int stickyBatchSize) {
57           this.log = logContext.logger(BuiltInPartitioner.class);
58           this.topic = topic;
59           this.stickyBatchSize = stickyBatchSize;
60       }

62       /**
63        * Calculate the next partition for the topic based on the partition load stats.
64        */
65       private int nextPartition(Cluster cluster) {
66           int random = mockRandom != null ? mockRandom.get() : Utils.toPositive(ThreadLocalRandom.current().nextInt());
67   
68           // Cache volatile variable in local variable.
69           PartitionLoadStats partitionLoadStats = this.partitionLoadStats;
70           int partition;
71   
72           if (partitionLoadStats == null) {
73               // We don't have stats to do adaptive partitioning (or it's disabled), just switch to the next
74               // partition based on uniform distribution.
75               List<PartitionInfo> availablePartitions = cluster.availablePartitionsForTopic(topic);
76               if (availablePartitions.size() > 0) {
77                   partition = availablePartitions.get(random % availablePartitions.size()).partition();
78               } else {
79                   // We don't have available partitions, just pick one among all partitions.
80                   List<PartitionInfo> partitions = cluster.partitionsForTopic(topic);
81                   partition = random % partitions.size();
82               }
83           } else {
84               // Calculate next partition based on load distribution.
85               // Note that partitions without leader are excluded from the partitionLoadStats.
86               assert partitionLoadStats.length > 0;
87   
88               int[] cumulativeFrequencyTable = partitionLoadStats.cumulativeFrequencyTable;
89               int weightedRandom = random % cumulativeFrequencyTable[partitionLoadStats.length - 1];
90   
91               // By construction, the cumulative frequency table is sorted, so we can use binary
92               // search to find the desired index.
93               int searchResult = Arrays.binarySearch(cumulativeFrequencyTable, 0, partitionLoadStats.length, weightedRandom);
94   
95               // binarySearch results the index of the found element, or -(insertion_point) - 1
96               // (where insertion_point is the index of the first element greater than the key).
97               // We need to get the index of the first value that is strictly greater, which
98               // would be the insertion point, except if we found the element that's equal to
99               // the searched value (in this case we need to get next).  For example, if we have
100               //  4 5 8
101               // and we're looking for 3, then we'd get the insertion_point = 0, and the function
102               // would return -0 - 1 = -1, by adding 1 we'd get 0.  If we're looking for 4, we'd
103               // get 0, and we need the next one, so adding 1 works here as well.
104               int partitionIndex = Math.abs(searchResult + 1);
105               assert partitionIndex < partitionLoadStats.length;
106               partition = partitionLoadStats.partitionIds[partitionIndex];
107           }
108   
109           log.trace("Switching to partition {} in topic {}", partition, topic);
110           return partition;
111       }
112   
113       /**
114        * Test-only function.  When partition load stats are defined, return the end of range for the
115        * random number.
116        */
117       public int loadStatsRangeEnd() {
118           assert partitionLoadStats != null;
119           assert partitionLoadStats.length > 0;
120           return partitionLoadStats.cumulativeFrequencyTable[partitionLoadStats.length - 1];
121       }

123       /**
124        * Peek currently chosen sticky partition.  This method works in conjunction with {@link #isPartitionChanged}
125        * and {@link #updatePartitionInfo}.  The workflow is the following:
126        *
127        * 1. peekCurrentPartitionInfo is called to know which partition to lock.
128        * 2. Lock partition's batch queue.
129        * 3. isPartitionChanged under lock to make sure that nobody raced us.
130        * 4. Append data to buffer.
131        * 5. updatePartitionInfo to update produced bytes and maybe switch partition.
132        *
133        *  It's important that steps 3-5 are under partition's batch queue lock.
134        *
135        * @param cluster The cluster information (needed if there is no current partition)
136        * @return sticky partition info object
137        */
138       StickyPartitionInfo peekCurrentPartitionInfo(Cluster cluster) {
139           StickyPartitionInfo partitionInfo = stickyPartitionInfo.get();
140           if (partitionInfo != null)
141               return partitionInfo;
142   
143           // We're the first to create it.
144           partitionInfo = new StickyPartitionInfo(nextPartition(cluster));
145           if (stickyPartitionInfo.compareAndSet(null, partitionInfo))
146               return partitionInfo;
147   
148           // Someone has raced us.
149           return stickyPartitionInfo.get();
150       }
151   
152       /**
153        * Check if partition is changed by a concurrent thread.  NOTE this function needs to be called under
154        * the partition's batch queue lock.
155        *
156        * @param partitionInfo The sticky partition info object returned by peekCurrentPartitionInfo
157        * @return true if sticky partition object is changed (race condition)
158        */
159       boolean isPartitionChanged(StickyPartitionInfo partitionInfo) {
160           // partitionInfo may be null if the caller didn't use built-in partitioner.
161           return partitionInfo != null && stickyPartitionInfo.get() != partitionInfo;
162       }
163   
164       /**
165        * Update partition info with the number of bytes appended and maybe switch partition.
166        * NOTE this function needs to be called under the partition's batch queue lock.
167        *
168        * @param partitionInfo The sticky partition info object returned by peekCurrentPartitionInfo
169        * @param appendedBytes The number of bytes appended to this partition
170        * @param cluster The cluster information
171        */
172       void updatePartitionInfo(StickyPartitionInfo partitionInfo, int appendedBytes, Cluster cluster) {
173           // partitionInfo may be null if the caller didn't use built-in partitioner.
174           if (partitionInfo == null)
175               return;
176   
177           assert partitionInfo == stickyPartitionInfo.get();
178           int producedBytes = partitionInfo.producedBytes.addAndGet(appendedBytes);
179           if (producedBytes >= stickyBatchSize) {
180               // We've produced enough to this partition, switch to next.
181               StickyPartitionInfo newPartitionInfo = new StickyPartitionInfo(nextPartition(cluster));
182               stickyPartitionInfo.set(newPartitionInfo);
183           }
184       }

186       /**
187        * Update partition load stats from the queue sizes of each partition
188        * NOTE: queueSizes are modified in place to avoid allocations
189        *
190        * @param queueSizes The queue sizes, partitions without leaders are excluded
191        * @param partitionIds The partition ids for the queues, partitions without leaders are excluded
192        * @param length The logical length of the arrays (could be less): we may eliminate some partitions
193        *               based on latency, but to avoid reallocation of the arrays, we just decrement
194        *               logical length
195        * Visible for testing
196        */
197       public void updatePartitionLoadStats(int[] queueSizes, int[] partitionIds, int length) {
198           if (queueSizes == null) {
199               log.trace("No load stats for topic {}, not using adaptive", topic);
200               partitionLoadStats = null;
201               return;
202           }
203           assert queueSizes.length == partitionIds.length;
204           assert length <= queueSizes.length;
205   
206           // The queueSizes.length represents the number of all partitions in the topic and if we have
207           // less than 2 partitions, there is no need to do adaptive logic.
208           // If partitioner.availability.timeout.ms != 0, then partitions that experience high latencies
209           // (greater than partitioner.availability.timeout.ms) may be excluded, the length represents
210           // partitions that are not excluded.  If some partitions were excluded, we'd still want to
211           // go through adaptive logic, even if we have one partition.
212           // See also RecordAccumulator#partitionReady where the queueSizes are built.
213           if (length < 1 || queueSizes.length < 2) {
214               log.trace("The number of partitions is too small: available={}, all={}, not using adaptive for topic {}",
215                       length, queueSizes.length, topic);
216               partitionLoadStats = null;
217               return;
218           }
219   
220           // We build cumulative frequency table from the queue sizes in place.  At the beginning
221           // each entry contains queue size, then we invert it (so it represents the frequency)
222           // and convert to a running sum.  Then a uniformly distributed random variable
223           // in the range [0..last) would map to a partition with weighted probability.
224           // Example: suppose we have 3 partitions with the corresponding queue sizes:
225           //  0 3 1
226           // Then we can invert them by subtracting the queue size from the max queue size + 1 = 4:
227           //  4 1 3
228           // Then we can convert it into a running sum (next value adds previous value):
229           //  4 5 8
230           // Now if we get a random number in the range [0..8) and find the first value that
231           // is strictly greater than the number (e.g. for 4 it would be 5), then the index of
232           // the value is the index of the partition we're looking for.  In this example
233           // random numbers 0, 1, 2, 3 would map to partition[0], 4 would map to partition[1]
234           // and 5, 6, 7 would map to partition[2].
235   
236           // Calculate max queue size + 1 and check if all sizes are the same.
237           int maxSizePlus1 = queueSizes[0];
238           boolean allEqual = true;
239           for (int i = 1; i < length; i++) {
240               if (queueSizes[i] != maxSizePlus1)
241                   allEqual = false;
242               if (queueSizes[i] > maxSizePlus1)
243                   maxSizePlus1 = queueSizes[i];
244           }
245           ++maxSizePlus1;
246   
247           if (allEqual && length == queueSizes.length) {
248               // No need to have complex probability logic when all queue sizes are the same,
249               // and we didn't exclude partitions that experience high latencies (greater than
250               // partitioner.availability.timeout.ms).
251               log.trace("All queue lengths are the same, not using adaptive for topic {}", topic);
252               partitionLoadStats = null;
253               return;
254           }
255   
256           // Invert and fold the queue size, so that they become separator values in the CFT.
257           queueSizes[0] = maxSizePlus1 - queueSizes[0];
258           for (int i = 1; i < length; i++) {
259               queueSizes[i] = maxSizePlus1 - queueSizes[i] + queueSizes[i - 1];
260           }
261           log.trace("Partition load stats for topic {}: CFT={}, IDs={}, length={}",
262                   topic, queueSizes, partitionIds, length);
263           partitionLoadStats = new PartitionLoadStats(queueSizes, partitionIds, length);
264       }

267        * Info for the current sticky partition.
268        */
269       public static class StickyPartitionInfo {
270           private final int index;
271           private final AtomicInteger producedBytes = new AtomicInteger();

273           StickyPartitionInfo(int index) {
274               this.index = index;
275           }

279           }

282       /*
283        * Default hashing function to choose a partition from the serialized key bytes
284        */
285       public static int partitionForKey(final byte[] serializedKey, final int numPartitions) {
286           return Utils.toPositive(Utils.murmur2(serializedKey)) % numPartitions;
287       }

298               assert length <= cumulativeFrequencyTable.length;
299               this.cumulativeFrequencyTable = cumulativeFrequencyTable;
300               this.partitionIds = partitionIds;
