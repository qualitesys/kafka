17   package org.apache.kafka.common.record;
18   
19   import org.apache.kafka.common.utils.BufferSupplier;
20   import org.apache.kafka.common.utils.CloseableIterator;
21   
22   import java.nio.ByteBuffer;
23   import java.util.Iterator;
24   import java.util.OptionalLong;
25   
26   /**
27    * A record batch is a container for records. In old versions of the record format (versions 0 and 1),
28    * a batch consisted always of a single record if no compression was enabled, but could contain
29    * many records otherwise. Newer versions (magic versions 2 and above) will generally contain many records
30    * regardless of compression.
31    */
32   public interface RecordBatch extends Iterable<Record> {
33   
34       /**
35        * The "magic" values
36        */
37       byte MAGIC_VALUE_V0 = 0;
38       byte MAGIC_VALUE_V1 = 1;
39       byte MAGIC_VALUE_V2 = 2;
40   
41       /**
42        * The current "magic" value
43        */
44       byte CURRENT_MAGIC_VALUE = MAGIC_VALUE_V2;
45   
46       /**
47        * Timestamp value for records without a timestamp
48        */
49       long NO_TIMESTAMP = -1L;
50   
51       /**
52        * Values used in the v2 record format by non-idempotent/non-transactional producers or when
53        * up-converting from an older format.
54        */
55       long NO_PRODUCER_ID = -1L;
56       short NO_PRODUCER_EPOCH = -1;
57       int NO_SEQUENCE = -1;
58   
59       /**
60        * Used to indicate an unknown leader epoch, which will be the case when the record set is
61        * first created by the producer.
62        */
63       int NO_PARTITION_LEADER_EPOCH = -1;
64   
65       /**
66        * Check whether the checksum of this batch is correct.
67        *
68        * @return true If so, false otherwise
69        */
70       boolean isValid();
71   
72       /**
73        * Raise an exception if the checksum is not valid.
74        */
75       void ensureValid();
76   
77       /**
78        * Get the checksum of this record batch, which covers the batch header as well as all of the records.
79        *
80        * @return The 4-byte unsigned checksum represented as a long
81        */
82       long checksum();
83   
84       /**
85        * Get the max timestamp or log append time of this record batch.
86        *
87        * If the timestamp type is create time, this is the max timestamp among all records contained in this batch and
88        * the value is updated during compaction.
89        *
90        * @return The max timestamp
91        */
92       long maxTimestamp();
93   
94       /**
95        * Get the timestamp type of this record batch. This will be {@link TimestampType#NO_TIMESTAMP_TYPE}
96        * if the batch has magic 0.
97        *
98        * @return The timestamp type
99        */
100       TimestampType timestampType();
101   
102       /**
103        * Get the base offset contained in this record batch. For magic version prior to 2, the base offset will
104        * always be the offset of the first message in the batch. This generally requires deep iteration and will
105        * return the offset of the first record in the record batch. For magic version 2 and above, this will return
106        * the first offset of the original record batch (i.e. prior to compaction). For non-compacted topics, the
107        * behavior is equivalent.
108        *
109        * Because this requires deep iteration for older magic versions, this method should be used with
110        * caution. Generally {@link #lastOffset()} is safer since access is efficient for all magic versions.
111        *
112        * @return The base offset of this record batch (which may or may not be the offset of the first record
113        *         as described above).
114        */
115       long baseOffset();
116   
117       /**
118        * Get the last offset in this record batch (inclusive). Just like {@link #baseOffset()}, the last offset
119        * always reflects the offset of the last record in the original batch, even if it is removed during log
120        * compaction.
121        *
122        * @return The offset of the last record in this batch
123        */
124       long lastOffset();
125   
126       /**
127        * Get the offset following this record batch (i.e. the last offset contained in this batch plus one).
128        *
129        * @return the next consecutive offset following this batch
130        */
131       long nextOffset();
132   
133       /**
134        * Get the record format version of this record batch (i.e its magic value).
135        *
136        * @return the magic byte
137        */
138       byte magic();
139   
140       /**
141        * Get the producer id for this log record batch. For older magic versions, this will return -1.
142        *
143        * @return The producer id or -1 if there is none
144        */
145       long producerId();
146   
147       /**
148        * Get the producer epoch for this log record batch.
149        *
150        * @return The producer epoch, or -1 if there is none
151        */
152       short producerEpoch();
153   
154       /**
155        * Does the batch have a valid producer id set.
156        */
157       boolean hasProducerId();
158   
159       /**
160        * Get the base sequence number of this record batch. Like {@link #baseOffset()}, this value is not
161        * affected by compaction: it always retains the base sequence number from the original batch.
162        *
163        * @return The first sequence number or -1 if there is none
164        */
165       int baseSequence();
166   
167       /**
168        * Get the last sequence number of this record batch. Like {@link #lastOffset()}, the last sequence number
169        * always reflects the sequence number of the last record in the original batch, even if it is removed during log
170        * compaction.
171        *
172        * @return The last sequence number or -1 if there is none
173        */
174       int lastSequence();
175   
176       /**
177        * Get the compression type of this record batch.
178        *
179        * @return The compression type
180        */
181       CompressionType compressionType();
182   
183       /**
184        * Get the size in bytes of this batch, including the size of the record and the batch overhead.
185        * @return The size in bytes of this batch
186        */
187       int sizeInBytes();
188   
189       /**
190        * Get the count if it is efficiently supported by the record format (which is only the case
191        * for magic 2 and higher).
192        *
193        * @return The number of records in the batch or null for magic versions 0 and 1.
194        */
195       Integer countOrNull();
196   
197       /**
198        * Check whether this record batch is compressed.
199        * @return true if so, false otherwise
200        */
201       boolean isCompressed();
202   
203       /**
204        * Write this record batch into a buffer.
205        * @param buffer The buffer to write the batch to
206        */
207       void writeTo(ByteBuffer buffer);
208   
209       /**
210        * Whether or not this record batch is part of a transaction.
211        * @return true if it is, false otherwise
212        */
213       boolean isTransactional();
214   
215       /**
216        * Get the delete horizon, returns OptionalLong.EMPTY if the first timestamp is not the delete horizon
217        * @return timestamp of the delete horizon
218        */
219       OptionalLong deleteHorizonMs();
220   
221       /**
222        * Get the partition leader epoch of this record batch.
223        * @return The leader epoch or -1 if it is unknown
224        */
225       int partitionLeaderEpoch();
226   
227       /**
228        * Return a streaming iterator which basically delays decompression of the record stream until the records
229        * are actually asked for using {@link Iterator#next()}. If the message format does not support streaming
230        * iteration, then the normal iterator is returned. Either way, callers should ensure that the iterator is closed.
231        *
232        * @param decompressionBufferSupplier The supplier of ByteBuffer(s) used for decompression if supported.
233        *                                    For small record batches, allocating a potentially large buffer (64 KB for LZ4)
234        *                                    will dominate the cost of decompressing and iterating over the records in the
235        *                                    batch. As such, a supplier that reuses buffers will have a significant
236        *                                    performance impact.
237        * @return The closeable iterator
238        */
239       CloseableIterator<Record> streamingIterator(BufferSupplier decompressionBufferSupplier);
240   
241       /**
242        * Check whether this is a control batch (i.e. whether the control bit is set in the batch attributes).
243        * For magic versions prior to 2, this is always false.
244        *
245        * @return Whether this is a batch containing control records
246        */
247       boolean isControlBatch();
248   }
