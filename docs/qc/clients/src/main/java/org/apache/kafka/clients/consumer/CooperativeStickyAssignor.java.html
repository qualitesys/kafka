19   import java.nio.ByteBuffer;
20   import java.util.Arrays;
21   import java.util.HashMap;
22   import java.util.HashSet;
23   import java.util.List;
24   import java.util.Map;
25   import java.util.Optional;
26   import java.util.Set;
27   import org.apache.kafka.clients.consumer.internals.AbstractStickyAssignor;
28   import org.apache.kafka.common.TopicPartition;
29   import org.apache.kafka.common.protocol.types.Field;

31   import org.apache.kafka.common.protocol.types.Struct;
32   import org.apache.kafka.common.protocol.types.Type;

34   /**
35    * A cooperative version of the {@link AbstractStickyAssignor AbstractStickyAssignor}. This follows the same (sticky)
36    * assignment logic as {@link StickyAssignor StickyAssignor} but allows for cooperative rebalancing while the
37    * {@link StickyAssignor StickyAssignor} follows the eager rebalancing protocol. See
38    * {@link ConsumerPartitionAssignor.RebalanceProtocol} for an explanation of the rebalancing protocols.
39    * <p>
40    * Users should prefer this assignor for newer clusters.
41    * <p>
42    * To turn on cooperative rebalancing you must set all your consumers to use this {@code PartitionAssignor},
43    * or implement a custom one that returns {@code RebalanceProtocol.COOPERATIVE} in
44    * {@link CooperativeStickyAssignor#supportedProtocols supportedProtocols()}.
45    * <p>
46    * IMPORTANT: if upgrading from 2.3 or earlier, you must follow a specific upgrade path in order to safely turn on
47    * cooperative rebalancing. See the <a href="https://kafka.apache.org/documentation/#upgrade_240_notable">upgrade guide</a> for details.
48    */
49   public class CooperativeStickyAssignor extends AbstractStickyAssignor {
50       public static final String COOPERATIVE_STICKY_ASSIGNOR_NAME = "cooperative-sticky";
51   
52       // these schemas are used for preserving useful metadata for the assignment, such as the last stable generation
53       private static final String GENERATION_KEY_NAME = "generation";
54       private static final Schema COOPERATIVE_STICKY_ASSIGNOR_USER_DATA_V0 = new Schema(
55           new Field(GENERATION_KEY_NAME, Type.INT32));
56   
57       private int generation = DEFAULT_GENERATION; // consumer group generation
58   
59       @Override
60       public String name() {
61           return COOPERATIVE_STICKY_ASSIGNOR_NAME;
62       }
63   
64       @Override
65       public List<RebalanceProtocol> supportedProtocols() {
66           return Arrays.asList(RebalanceProtocol.COOPERATIVE, RebalanceProtocol.EAGER);
67       }
68   
69       @Override
70       public void onAssignment(Assignment assignment, ConsumerGroupMetadata metadata) {
71           this.generation = metadata.generationId();
72       }
73   
74       @Override
75       public ByteBuffer subscriptionUserData(Set<String> topics) {
76           Struct struct = new Struct(COOPERATIVE_STICKY_ASSIGNOR_USER_DATA_V0);
77   
78           struct.set(GENERATION_KEY_NAME, generation);
79           ByteBuffer buffer = ByteBuffer.allocate(COOPERATIVE_STICKY_ASSIGNOR_USER_DATA_V0.sizeOf(struct));
80           COOPERATIVE_STICKY_ASSIGNOR_USER_DATA_V0.write(buffer, struct);
81           buffer.flip();
82           return buffer;
83       }
84   
85       @Override
86       protected MemberData memberData(Subscription subscription) {
87           ByteBuffer buffer = subscription.userData();
88           Optional<Integer> encodedGeneration;
89           if (buffer == null) {
90               encodedGeneration = Optional.empty();
91           } else {
92               try {
93                   Struct struct = COOPERATIVE_STICKY_ASSIGNOR_USER_DATA_V0.read(buffer);
94                   encodedGeneration = Optional.of(struct.getInt(GENERATION_KEY_NAME));
95               } catch (Exception e) {
96                   encodedGeneration = Optional.of(DEFAULT_GENERATION);
97               }
98           }
99           return new MemberData(subscription.ownedPartitions(), encodedGeneration);
100       }
101   
102       @Override
103       public Map<String, List<TopicPartition>> assign(Map<String, Integer> partitionsPerTopic,
104                                                       Map<String, Subscription> subscriptions) {
105           Map<String, List<TopicPartition>> assignments = super.assign(partitionsPerTopic, subscriptions);
106   
107           Map<TopicPartition, String> partitionsTransferringOwnership = super.partitionsTransferringOwnership == null ?
108               computePartitionsTransferringOwnership(subscriptions, assignments) :
109               super.partitionsTransferringOwnership;
110   
111           adjustAssignment(assignments, partitionsTransferringOwnership);
112           return assignments;
113       }
114   
115       // Following the cooperative rebalancing protocol requires removing partitions that must first be revoked from the assignment
116       private void adjustAssignment(Map<String, List<TopicPartition>> assignments,
117                                     Map<TopicPartition, String> partitionsTransferringOwnership) {
118           for (Map.Entry<TopicPartition, String> partitionEntry : partitionsTransferringOwnership.entrySet()) {
119               assignments.get(partitionEntry.getValue()).remove(partitionEntry.getKey());
120           }
121       }
122   
123       private Map<TopicPartition, String> computePartitionsTransferringOwnership(Map<String, Subscription> subscriptions,
124                                                                                  Map<String, List<TopicPartition>> assignments) {
125           Map<TopicPartition, String> allAddedPartitions = new HashMap<>();
126           Set<TopicPartition> allRevokedPartitions = new HashSet<>();
127   
128           for (final Map.Entry<String, List<TopicPartition>> entry : assignments.entrySet()) {
129               String consumer = entry.getKey();
130   
131               List<TopicPartition> ownedPartitions = subscriptions.get(consumer).ownedPartitions();
132               List<TopicPartition> assignedPartitions = entry.getValue();
133   
134               Set<TopicPartition> ownedPartitionsSet = new HashSet<>(ownedPartitions);
135               for (TopicPartition tp : assignedPartitions) {
136                   if (!ownedPartitionsSet.contains(tp))
137                       allAddedPartitions.put(tp, consumer);
138               }
139   
140               Set<TopicPartition> assignedPartitionsSet = new HashSet<>(assignedPartitions);
141               for (TopicPartition tp : ownedPartitions) {
142                   if (!assignedPartitionsSet.contains(tp))
143                       allRevokedPartitions.add(tp);
144               }
145           }
146   
147           allAddedPartitions.keySet().retainAll(allRevokedPartitions);
148           return allAddedPartitions;
149       }
150   }
