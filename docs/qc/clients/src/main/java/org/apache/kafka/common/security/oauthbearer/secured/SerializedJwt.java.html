18   package org.apache.kafka.common.security.oauthbearer.secured;

20   /**
21    * SerializedJwt provides a modicum of structure and validation around a JWT's serialized form by
22    * splitting and making the three sections (header, payload, and signature) available to the user.
23    */
24   
25   public class SerializedJwt {
26   
27       private final String token;
28   
29       private final String header;
30   
31       private final String payload;
32   
33       private final String signature;
34   
35       public SerializedJwt(String token) {
36           if (token == null)
37               token = "";
38           else
39               token = token.trim();
40   
41           if (token.isEmpty())
42               throw new ValidateException("Empty JWT provided; expected three sections (header, payload, and signature)");
43   
44           String[] splits = token.split("\\.");
45   
46           if (splits.length != 3)
47               throw new ValidateException(String.format("Malformed JWT provided (%s); expected three sections (header, payload, and signature), but %s sections provided",
48                   token, splits.length));
49   
50           this.token = token.trim();
51           this.header = validateSection(splits[0], "header");
52           this.payload = validateSection(splits[1], "payload");
53           this.signature = validateSection(splits[2], "signature");
54       }
55   
56       /**
57        * Returns the entire base 64-encoded JWT.
58        *
59        * @return JWT
60        */
61   
62       public String getToken() {
63           return token;
64       }
65   
66       /**
67        * Returns the first section--the JWT header--in its base 64-encoded form.
68        *
69        * @return Header section of the JWT
70        */
71   
72       public String getHeader() {
73           return header;
74       }
75   
76       /**
77        * Returns the second section--the JWT payload--in its base 64-encoded form.
78        *
79        * @return Payload section of the JWT
80        */
81   
82       public String getPayload() {
83           return payload;
84       }
85   
86       /**
87        * Returns the third section--the JWT signature--in its base 64-encoded form.
88        *
89        * @return Signature section of the JWT
90        */
91   
92       public String getSignature() {
93           return signature;
94       }
95   
96       private String validateSection(String section, String sectionName) throws ValidateException {
97           section = section.trim();
98   
99           if (section.isEmpty())
100               throw new ValidateException(String.format(
101                   "Malformed JWT provided; expected at least three sections (header, payload, and signature), but %s section missing",
102                   sectionName));
103   
104           return section;
105       }
106   
107   }
