17   package org.apache.kafka.common.security;
18   
19   import java.io.IOException;
20   import java.io.StreamTokenizer;
21   import java.io.StringReader;
22   import java.util.ArrayList;
23   import java.util.HashMap;
24   import java.util.List;
25   import java.util.Locale;
26   import java.util.Map;
27   
28   import javax.security.auth.login.AppConfigurationEntry;
29   import javax.security.auth.login.Configuration;
30   import javax.security.auth.login.AppConfigurationEntry.LoginModuleControlFlag;
31   
32   import org.apache.kafka.common.KafkaException;
33   import org.apache.kafka.common.config.SaslConfigs;
34   
35   /**
36    * JAAS configuration parser that constructs a JAAS configuration object with a single
37    * login context from the Kafka configuration option {@link SaslConfigs#SASL_JAAS_CONFIG}.
38    * <p/>
39    * JAAS configuration file format is described <a href="http://docs.oracle.com/javase/8/docs/technotes/guides/security/jgss/tutorials/LoginConfigFile.html">here</a>.
40    * The format of the property value is:
41    * <pre>
42    * {@code
43    *   <loginModuleClass> <controlFlag> (<optionName>=<optionValue>)*;
44    * }
45    * </pre>
46    */
47   class JaasConfig extends Configuration {
48   
49       private final String loginContextName;
50       private final List<AppConfigurationEntry> configEntries;
51   
52       public JaasConfig(String loginContextName, String jaasConfigParams) {
53           StreamTokenizer tokenizer = new StreamTokenizer(new StringReader(jaasConfigParams));
54           tokenizer.slashSlashComments(true);
55           tokenizer.slashStarComments(true);
56           tokenizer.wordChars('-', '-');
57           tokenizer.wordChars('_', '_');
58           tokenizer.wordChars('$', '$');
59   
60           try {
61               configEntries = new ArrayList<>();
62               while (tokenizer.nextToken() != StreamTokenizer.TT_EOF) {
63                   configEntries.add(parseAppConfigurationEntry(tokenizer));
64               }
65               if (configEntries.isEmpty())
66                   throw new IllegalArgumentException("Login module not specified in JAAS config");
67   
68               this.loginContextName = loginContextName;
69   
70           } catch (IOException e) {
71               throw new KafkaException("Unexpected exception while parsing JAAS config");
72           }
73       }
74   
75       @Override
76       public AppConfigurationEntry[] getAppConfigurationEntry(String name) {
77           if (this.loginContextName.equals(name))
78               return configEntries.toArray(new AppConfigurationEntry[0]);
79           else
80               return  null;
81       }
82   
83       private LoginModuleControlFlag loginModuleControlFlag(String flag) {
84           if (flag == null)
85               throw new IllegalArgumentException("Login module control flag is not available in the JAAS config");
86   
87           LoginModuleControlFlag controlFlag;
88           switch (flag.toUpperCase(Locale.ROOT)) {
89               case "REQUIRED":
90                   controlFlag = LoginModuleControlFlag.REQUIRED;
91                   break;
92               case "REQUISITE":
93                   controlFlag = LoginModuleControlFlag.REQUISITE;
94                   break;
95               case "SUFFICIENT":
96                   controlFlag = LoginModuleControlFlag.SUFFICIENT;
97                   break;
98               case "OPTIONAL":
99                   controlFlag = LoginModuleControlFlag.OPTIONAL;
100                   break;
101               default:
102                   throw new IllegalArgumentException("Invalid login module control flag '" + flag + "' in JAAS config");
103           }
104           return controlFlag;
105       }
106   
107       private AppConfigurationEntry parseAppConfigurationEntry(StreamTokenizer tokenizer) throws IOException {
108           String loginModule = tokenizer.sval;
109           if (tokenizer.nextToken() == StreamTokenizer.TT_EOF)
110               throw new IllegalArgumentException("Login module control flag not specified in JAAS config");
111           LoginModuleControlFlag controlFlag = loginModuleControlFlag(tokenizer.sval);
112           Map<String, String> options = new HashMap<>();
113           while (tokenizer.nextToken() != StreamTokenizer.TT_EOF && tokenizer.ttype != ';') {
114               String key = tokenizer.sval;
115               if (tokenizer.nextToken() != '=' || tokenizer.nextToken() == StreamTokenizer.TT_EOF || tokenizer.sval == null)
116                   throw new IllegalArgumentException("Value not specified for key '" + key + "' in JAAS config");
117               String value = tokenizer.sval;
118               options.put(key, value);
119           }
120           if (tokenizer.ttype != ';')
121               throw new IllegalArgumentException("JAAS config entry not terminated by semi-colon");
122           return new AppConfigurationEntry(loginModule, controlFlag, options);
123       }
124   }
