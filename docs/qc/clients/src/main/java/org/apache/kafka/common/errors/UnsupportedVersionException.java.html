21   /**
22    * Indicates that a request API or version needed by the client is not supported by the broker. This is
23    * typically a fatal error as Kafka clients will downgrade request versions as needed except in cases where
24    * a needed feature is not available in old versions. Fatal errors can generally only be handled by closing
25    * the client instance, although in some cases it may be possible to continue without relying on the
26    * underlying feature. For example, when the producer is used with idempotence enabled, this error is fatal
27    * since the producer does not support reverting to weaker semantics. On the other hand, if this error
28    * is raised from {@link org.apache.kafka.clients.consumer.KafkaConsumer#offsetsForTimes(Map)}, it would
29    * be possible to revert to alternative logic to set the consumer's position.
30    */
31   public class UnsupportedVersionException extends ApiException {
32       private static final long serialVersionUID = 1L;
33   
34       public UnsupportedVersionException(String message, Throwable cause) {
35           super(message, cause);
36       }
37   
38       public UnsupportedVersionException(String message) {
39           super(message);
40       }
41   }
