17   package org.apache.kafka.common.requests;
18   
19   import java.util.regex.Pattern;
20   import org.apache.kafka.common.message.ApiVersionsRequestData;
21   import org.apache.kafka.common.message.ApiVersionsResponseData;
22   import org.apache.kafka.common.message.ApiVersionsResponseData.ApiVersionCollection;
23   import org.apache.kafka.common.protocol.ApiKeys;
24   import org.apache.kafka.common.protocol.ByteBufferAccessor;
25   import org.apache.kafka.common.protocol.Errors;
26   import org.apache.kafka.common.utils.AppInfoParser;
27   
28   import java.nio.ByteBuffer;
29   
30   public class ApiVersionsRequest extends AbstractRequest {
31   
32       public static class Builder extends AbstractRequest.Builder<ApiVersionsRequest> {
33           private static final String DEFAULT_CLIENT_SOFTWARE_NAME = "apache-kafka-java";
34   
35           private static final ApiVersionsRequestData DATA = new ApiVersionsRequestData()
36               .setClientSoftwareName(DEFAULT_CLIENT_SOFTWARE_NAME)
37               .setClientSoftwareVersion(AppInfoParser.getVersion());
38   
39           public Builder() {
40               super(ApiKeys.API_VERSIONS);
41           }
42   
43           public Builder(short version) {
44               super(ApiKeys.API_VERSIONS, version);
45           }
46   
47           @Override
48           public ApiVersionsRequest build(short version) {
49               return new ApiVersionsRequest(DATA, version);
50           }
51   
52           @Override
53           public String toString() {
54               return DATA.toString();
55           }
56       }
57   
58       private static final Pattern SOFTWARE_NAME_VERSION_PATTERN = Pattern.compile("[a-zA-Z0-9](?:[a-zA-Z0-9\\-.]*[a-zA-Z0-9])?");
59   
60       private final Short unsupportedRequestVersion;
61   
62       private final ApiVersionsRequestData data;
63   
64       public ApiVersionsRequest(ApiVersionsRequestData data, short version) {
65           this(data, version, null);
66       }
67   
68       public ApiVersionsRequest(ApiVersionsRequestData data, short version, Short unsupportedRequestVersion) {
69           super(ApiKeys.API_VERSIONS, version);
70           this.data = data;
71   
72           // Unlike other request types, the broker handles ApiVersion requests with higher versions than
73           // supported. It does so by treating the request as if it were v0 and returns a response using
74           // the v0 response schema. The reason for this is that the client does not yet know what versions
75           // a broker supports when this request is sent, so instead of assuming the lowest supported version,
76           // it can use the most recent version and only fallback to the old version when necessary.
77           this.unsupportedRequestVersion = unsupportedRequestVersion;
78       }
79   
80       public boolean hasUnsupportedRequestVersion() {
81           return unsupportedRequestVersion != null;
82       }
83   
84       public boolean isValid() {
85           if (version() >= 3) {
86               return SOFTWARE_NAME_VERSION_PATTERN.matcher(data.clientSoftwareName()).matches() &&
87                   SOFTWARE_NAME_VERSION_PATTERN.matcher(data.clientSoftwareVersion()).matches();
88           } else {
89               return true;
90           }
91       }
92   
93       @Override
94       public ApiVersionsRequestData data() {
95           return data;
96       }
97   
98       @Override
99       public ApiVersionsResponse getErrorResponse(int throttleTimeMs, Throwable e) {
100           ApiVersionsResponseData data = new ApiVersionsResponseData()
101               .setErrorCode(Errors.forException(e).code());
102   
103           if (version() >= 1) {
104               data.setThrottleTimeMs(throttleTimeMs);
105           }
106   
107           // Starting from Apache Kafka 2.4 (KIP-511), ApiKeys field is populated with the supported
108           // versions of the ApiVersionsRequest when an UNSUPPORTED_VERSION error is returned.
109           if (Errors.forException(e) == Errors.UNSUPPORTED_VERSION) {
110               ApiVersionCollection apiKeys = new ApiVersionCollection();
111               apiKeys.add(ApiVersionsResponse.toApiVersion(ApiKeys.API_VERSIONS));
112               data.setApiKeys(apiKeys);
113           }
114   
115           return new ApiVersionsResponse(data);
116       }
117   
118       public static ApiVersionsRequest parse(ByteBuffer buffer, short version) {
119           return new ApiVersionsRequest(new ApiVersionsRequestData(new ByteBufferAccessor(buffer), version), version);
120       }

122   }
