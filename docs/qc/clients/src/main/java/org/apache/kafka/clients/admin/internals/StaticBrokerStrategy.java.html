17   package org.apache.kafka.clients.admin.internals;
18   
19   import org.apache.kafka.common.requests.AbstractRequest;
20   import org.apache.kafka.common.requests.AbstractResponse;
21   
22   import java.util.OptionalInt;
23   import java.util.Set;
24   
25   /**
26    * This lookup strategy is used when we already know the destination broker ID
27    * and we have no need for an explicit lookup. By setting {@link ApiRequestScope#destinationBrokerId()}
28    * in the returned value for {@link #lookupScope(Object)}, the driver will
29    * skip the lookup.
30    */
31   public class StaticBrokerStrategy<K> implements AdminApiLookupStrategy<K> {
32       private final SingleBrokerScope scope;
33   
34       public StaticBrokerStrategy(int brokerId) {
35           this.scope = new SingleBrokerScope(brokerId);
36       }
37   
38       @Override
39       public ApiRequestScope lookupScope(K key) {
40           return scope;
41       }
42   
43       @Override
44       public AbstractRequest.Builder<?> buildRequest(Set<K> keys) {
45           throw new UnsupportedOperationException();
46       }
47   
48       @Override
49       public LookupResult<K> handleResponse(Set<K> keys, AbstractResponse response) {
50           throw new UnsupportedOperationException();
51       }
52   
53       private static class SingleBrokerScope implements ApiRequestScope {
54           private final int brokerId;
55   
56           private SingleBrokerScope(int brokerId) {
57               this.brokerId = brokerId;
58           }
59   
60           @Override
61           public OptionalInt destinationBrokerId() {
62               return OptionalInt.of(brokerId);
63           }
64       }
65   }
