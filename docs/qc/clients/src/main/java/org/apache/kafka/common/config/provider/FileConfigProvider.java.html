17   package org.apache.kafka.common.config.provider;
18   
19   import org.apache.kafka.common.config.ConfigData;
20   import org.apache.kafka.common.config.ConfigException;
21   import org.slf4j.Logger;
22   import org.slf4j.LoggerFactory;
23   
24   import java.io.IOException;
25   import java.io.Reader;
26   import java.nio.file.Files;
27   import java.nio.file.Paths;
28   import java.util.Enumeration;
29   import java.util.HashMap;
30   import java.util.Map;
31   import java.util.Properties;
32   import java.util.Set;
33   
34   /**
35    * An implementation of {@link ConfigProvider} that represents a Properties file.
36    * All property keys and values are stored as cleartext.
37    */
38   public class FileConfigProvider implements ConfigProvider {
39   
40       private static final Logger log = LoggerFactory.getLogger(FileConfigProvider.class);
41   
42       public void configure(Map<String, ?> configs) {
43       }
44   
45       /**
46        * Retrieves the data at the given Properties file.
47        *
48        * @param path the file where the data resides
49        * @return the configuration data
50        */
51       public ConfigData get(String path) {
52           Map<String, String> data = new HashMap<>();
53           if (path == null || path.isEmpty()) {
54               return new ConfigData(data);
55           }
56           try (Reader reader = reader(path)) {
57               Properties properties = new Properties();
58               properties.load(reader);
59               Enumeration<Object> keys = properties.keys();
60               while (keys.hasMoreElements()) {
61                   String key = keys.nextElement().toString();
62                   String value = properties.getProperty(key);
63                   if (value != null) {
64                       data.put(key, value);
65                   }
66               }
67               return new ConfigData(data);
68           } catch (IOException e) {
69               log.error("Could not read properties from file {}", path, e);
70               throw new ConfigException("Could not read properties from file " + path);
71           }
72       }
73   
74       /**
75        * Retrieves the data with the given keys at the given Properties file.
76        *
77        * @param path the file where the data resides
78        * @param keys the keys whose values will be retrieved
79        * @return the configuration data
80        */
81       public ConfigData get(String path, Set<String> keys) {
82           Map<String, String> data = new HashMap<>();
83           if (path == null || path.isEmpty()) {
84               return new ConfigData(data);
85           }
86           try (Reader reader = reader(path)) {
87               Properties properties = new Properties();
88               properties.load(reader);
89               for (String key : keys) {
90                   String value = properties.getProperty(key);
91                   if (value != null) {
92                       data.put(key, value);
93                   }
94               }
95               return new ConfigData(data);
96           } catch (IOException e) {
97               log.error("Could not read properties from file {}", path, e);
98               throw new ConfigException("Could not read properties from file " + path);
99           }
100       }
101   
102       // visible for testing
103       protected Reader reader(String path) throws IOException {
104           return Files.newBufferedReader(Paths.get(path));
105       }
106   
107       public void close() {
108       }
109   }
