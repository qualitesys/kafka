46   public class DescribeTransactionsHandler extends AdminApiHandler.Batched<CoordinatorKey, TransactionDescription> {
47       private final Logger log;
48       private final AdminApiLookupStrategy<CoordinatorKey> lookupStrategy;
49   
50       public DescribeTransactionsHandler(
51           LogContext logContext
52       ) {
53           this.log = logContext.logger(DescribeTransactionsHandler.class);
54           this.lookupStrategy = new CoordinatorStrategy(CoordinatorType.TRANSACTION, logContext);
55       }
56   
57       public static AdminApiFuture.SimpleAdminApiFuture<CoordinatorKey, TransactionDescription> newFuture(
58           Collection<String> transactionalIds
59       ) {
60           return AdminApiFuture.forKeys(buildKeySet(transactionalIds));
61       }
62   
63       private static Set<CoordinatorKey> buildKeySet(Collection<String> transactionalIds) {
64           return transactionalIds.stream()
65               .map(CoordinatorKey::byTransactionalId)
66               .collect(Collectors.toSet());
67       }
68   
69       @Override
70       public String apiName() {
71           return "describeTransactions";
72       }
73   
74       @Override
75       public AdminApiLookupStrategy<CoordinatorKey> lookupStrategy() {
76           return lookupStrategy;
77       }
78   
79       @Override
80       public DescribeTransactionsRequest.Builder buildBatchedRequest(
81           int brokerId,
82           Set<CoordinatorKey> keys
83       ) {
84           DescribeTransactionsRequestData request = new DescribeTransactionsRequestData();
85           List<String> transactionalIds = keys.stream().map(key -> {
86               if (key.type != FindCoordinatorRequest.CoordinatorType.TRANSACTION) {
87                   throw new IllegalArgumentException("Invalid group coordinator key " + key +
88                       " when building `DescribeTransaction` request");
89               }
90               return key.idValue;
91           }).collect(Collectors.toList());
92           request.setTransactionalIds(transactionalIds);
93           return new DescribeTransactionsRequest.Builder(request);
94       }
95   
96       @Override
97       public ApiResult<CoordinatorKey, TransactionDescription> handleResponse(
98           Node broker,
99           Set<CoordinatorKey> keys,
100           AbstractResponse abstractResponse
101       ) {
102           DescribeTransactionsResponse response = (DescribeTransactionsResponse) abstractResponse;
103           Map<CoordinatorKey, TransactionDescription> completed = new HashMap<>();
104           Map<CoordinatorKey, Throwable> failed = new HashMap<>();
105           List<CoordinatorKey> unmapped = new ArrayList<>();
106   
107           for (DescribeTransactionsResponseData.TransactionState transactionState : response.data().transactionStates()) {
108               CoordinatorKey transactionalIdKey = CoordinatorKey.byTransactionalId(
109                   transactionState.transactionalId());
110               if (!keys.contains(transactionalIdKey)) {
111                   log.warn("Response included transactionalId `{}`, which was not requested",
112                       transactionState.transactionalId());
113                   continue;
114               }
115   
116               Errors error = Errors.forCode(transactionState.errorCode());
117               if (error != Errors.NONE) {
118                   handleError(transactionalIdKey, error, failed, unmapped);
119                   continue;
120               }
121   
122               OptionalLong transactionStartTimeMs = transactionState.transactionStartTimeMs() < 0 ?
123                   OptionalLong.empty() :
124                   OptionalLong.of(transactionState.transactionStartTimeMs());
125   
126               completed.put(transactionalIdKey, new TransactionDescription(
127                   broker.id(),
128                   TransactionState.parse(transactionState.transactionState()),
129                   transactionState.producerId(),
130                   transactionState.producerEpoch(),
131                   transactionState.transactionTimeoutMs(),
132                   transactionStartTimeMs,
133                   collectTopicPartitions(transactionState)
134               ));
135           }
136   
137           return new ApiResult<>(completed, failed, unmapped);
138       }
139   
140       private Set<TopicPartition> collectTopicPartitions(
141           DescribeTransactionsResponseData.TransactionState transactionState
142       ) {
143           Set<TopicPartition> res = new HashSet<>();
144           for (DescribeTransactionsResponseData.TopicData topicData : transactionState.topics()) {
145               String topic = topicData.topic();
146               for (Integer partitionId : topicData.partitions()) {
147                   res.add(new TopicPartition(topic, partitionId));
148               }
149           }
150           return res;
151       }
152   
153       private void handleError(
154           CoordinatorKey transactionalIdKey,
155           Errors error,
156           Map<CoordinatorKey, Throwable> failed,
157           List<CoordinatorKey> unmapped
158       ) {
159           switch (error) {
160               case TRANSACTIONAL_ID_AUTHORIZATION_FAILED:
161                   failed.put(transactionalIdKey, new TransactionalIdAuthorizationException(
162                       "DescribeTransactions request for transactionalId `" + transactionalIdKey.idValue + "` " +
163                           "failed due to authorization failure"));
164                   break;
165   
166               case TRANSACTIONAL_ID_NOT_FOUND:
167                   failed.put(transactionalIdKey, new TransactionalIdNotFoundException(
168                       "DescribeTransactions request for transactionalId `" + transactionalIdKey.idValue + "` " +
169                           "failed because the ID could not be found"));
170                   break;
171   
172               case COORDINATOR_LOAD_IN_PROGRESS:
173                   // If the coordinator is in the middle of loading, then we just need to retry
174                   log.debug("DescribeTransactions request for transactionalId `{}` failed because the " +
175                           "coordinator is still in the process of loading state. Will retry",
176                       transactionalIdKey.idValue);
177                   break;
178   
179               case NOT_COORDINATOR:
180               case COORDINATOR_NOT_AVAILABLE:
181                   // If the coordinator is unavailable or there was a coordinator change, then we unmap
182                   // the key so that we retry the `FindCoordinator` request
183                   unmapped.add(transactionalIdKey);
184                   log.debug("DescribeTransactions request for transactionalId `{}` returned error {}. Will attempt " +
185                           "to find the coordinator again and retry", transactionalIdKey.idValue, error);
186                   break;
187   
188               default:
189                   failed.put(transactionalIdKey, error.exception("DescribeTransactions request for " +
190                       "transactionalId `" + transactionalIdKey.idValue + "` failed due to unexpected error"));
191           }
192       }
