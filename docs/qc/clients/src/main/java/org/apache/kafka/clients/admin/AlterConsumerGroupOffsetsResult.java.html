17   package org.apache.kafka.clients.admin;
18   
19   import java.util.List;
20   import java.util.Map;
21   import java.util.stream.Collectors;
22   
23   import org.apache.kafka.common.KafkaFuture;
24   import org.apache.kafka.common.TopicPartition;
25   import org.apache.kafka.common.annotation.InterfaceStability;
26   import org.apache.kafka.common.internals.KafkaFutureImpl;
27   import org.apache.kafka.common.protocol.Errors;
28   
29   /**
30    * The result of the {@link AdminClient#alterConsumerGroupOffsets(String, Map)} call.
31    *
32    * The API of this class is evolving, see {@link AdminClient} for details.
33    */
34   @InterfaceStability.Evolving
35   public class AlterConsumerGroupOffsetsResult {
36   
37       private final KafkaFuture<Map<TopicPartition, Errors>> future;
38   
39       AlterConsumerGroupOffsetsResult(KafkaFuture<Map<TopicPartition, Errors>> future) {
40           this.future = future;
41       }
42   
43       /**
44        * Return a future which can be used to check the result for a given partition.
45        */
46       public KafkaFuture<Void> partitionResult(final TopicPartition partition) {
47           final KafkaFutureImpl<Void> result = new KafkaFutureImpl<>();
48   
49           this.future.whenComplete((topicPartitions, throwable) -> {
50               if (throwable != null) {
51                   result.completeExceptionally(throwable);
52               } else if (!topicPartitions.containsKey(partition)) {
53                   result.completeExceptionally(new IllegalArgumentException(
54                       "Alter offset for partition \"" + partition + "\" was not attempted"));
55               } else {
56                   final Errors error = topicPartitions.get(partition);
57                   if (error == Errors.NONE) {
58                       result.complete(null);
59                   } else {
60                       result.completeExceptionally(error.exception());
61                   }
62               }
63           });
64   
65           return result;
66       }
67   
68       /**
69        * Return a future which succeeds if all the alter offsets succeed.
70        */
71       public KafkaFuture<Void> all() {
72           return this.future.thenApply(topicPartitionErrorsMap ->  {
73               List<TopicPartition> partitionsFailed = topicPartitionErrorsMap.entrySet()
74                   .stream()
75                   .filter(e -> e.getValue() != Errors.NONE)
76                   .map(Map.Entry::getKey)
77                   .collect(Collectors.toList());
78               for (Errors error : topicPartitionErrorsMap.values()) {
79                   if (error != Errors.NONE) {
80                       throw error.exception(
81                           "Failed altering consumer group offsets for the following partitions: " + partitionsFailed);
82                   }
83               }
84               return null;
85           });
86       }
87   }
