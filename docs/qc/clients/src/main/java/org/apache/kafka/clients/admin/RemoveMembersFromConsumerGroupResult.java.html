19   import org.apache.kafka.common.KafkaException;
20   import org.apache.kafka.common.KafkaFuture;
21   import org.apache.kafka.common.internals.KafkaFutureImpl;
22   import org.apache.kafka.common.message.LeaveGroupRequestData.MemberIdentity;
23   import org.apache.kafka.common.protocol.Errors;
24   
25   import java.util.Map;
26   import java.util.Set;
27   
28   /**
29    * The result of the {@link Admin#removeMembersFromConsumerGroup(String, RemoveMembersFromConsumerGroupOptions)} call.
30    *
31    * The API of this class is evolving, see {@link Admin} for details.
32    */
33   public class RemoveMembersFromConsumerGroupResult {
34   
35       private final KafkaFuture<Map<MemberIdentity, Errors>> future;
36       private final Set<MemberToRemove> memberInfos;
37   
38       RemoveMembersFromConsumerGroupResult(KafkaFuture<Map<MemberIdentity, Errors>> future,
39                                            Set<MemberToRemove> memberInfos) {
40           this.future = future;
41           this.memberInfos = memberInfos;
42       }
43   
44       /**
45        * Returns a future which indicates whether the request was 100% success, i.e. no
46        * either top level or member level error.
47        * If not, the first member error shall be returned.
48        */
49       public KafkaFuture<Void> all() {
50           final KafkaFutureImpl<Void> result = new KafkaFutureImpl<>();
51           this.future.whenComplete((memberErrors, throwable) -> {
52               if (throwable != null) {
53                   result.completeExceptionally(throwable);
54               } else {
55                   if (removeAll()) {
56                       for (Map.Entry<MemberIdentity, Errors> entry: memberErrors.entrySet()) {
57                           Exception exception = entry.getValue().exception();
58                           if (exception != null) {
59                               Throwable ex = new KafkaException("Encounter exception when trying to remove: "
60                                       + entry.getKey(), exception);
61                               result.completeExceptionally(ex);
62                               return;
63                           }
64                       }
65                   } else {
66                       for (MemberToRemove memberToRemove : memberInfos) {
67                           if (maybeCompleteExceptionally(memberErrors, memberToRemove.toMemberIdentity(), result)) {
68                               return;
69                           }
70                       }
71                   }
72                   result.complete(null);
73               }
74           });
75           return result;
76       }
77   
78       /**
79        * Returns the selected member future.
80        */
81       public KafkaFuture<Void> memberResult(MemberToRemove member) {
82           if (removeAll()) {
83               throw new IllegalArgumentException("The method: memberResult is not applicable in 'removeAll' mode");
84           }
85           if (!memberInfos.contains(member)) {
86               throw new IllegalArgumentException("Member " + member + " was not included in the original request");
87           }
88   
89           final KafkaFutureImpl<Void> result = new KafkaFutureImpl<>();
90           this.future.whenComplete((memberErrors, throwable) -> {
91               if (throwable != null) {
92                   result.completeExceptionally(throwable);
93               } else if (!maybeCompleteExceptionally(memberErrors, member.toMemberIdentity(), result)) {
94                   result.complete(null);
95               }
96           });
97           return result;
98       }
99   
100       private boolean maybeCompleteExceptionally(Map<MemberIdentity, Errors> memberErrors,
101                                                  MemberIdentity member,
102                                                  KafkaFutureImpl<Void> result) {
103           Throwable exception = KafkaAdminClient.getSubLevelError(memberErrors, member,
104               "Member \"" + member + "\" was not included in the removal response");
105           if (exception != null) {
106               result.completeExceptionally(exception);
107               return true;
108           } else {
109               return false;
110           }
111       }
112   
113       private boolean removeAll() {
114           return memberInfos.isEmpty();
115       }
116   }
