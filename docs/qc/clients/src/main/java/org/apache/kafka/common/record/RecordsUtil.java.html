17   package org.apache.kafka.common.record;
18   
19   import org.apache.kafka.common.errors.UnsupportedCompressionTypeException;
20   import org.apache.kafka.common.utils.Time;
21   import org.apache.kafka.common.utils.Utils;
22   
23   import java.nio.ByteBuffer;
24   import java.util.ArrayList;
25   import java.util.List;
26   
27   public class RecordsUtil {
28       /**
29        * Down convert batches to the provided message format version. The first offset parameter is only relevant in the
30        * conversion from uncompressed v2 or higher to v1 or lower. The reason is that uncompressed records in v0 and v1
31        * are not batched (put another way, each batch always has 1 record).
32        *
33        * If a client requests records in v1 format starting from the middle of an uncompressed batch in v2 format, we
34        * need to drop records from the batch during the conversion. Some versions of librdkafka rely on this for
35        * correctness.
36        *
37        * The temporaryMemoryBytes computation assumes that the batches are not loaded into the heap
38        * (via classes like FileChannelRecordBatch) before this method is called. This is the case in the broker (we
39        * only load records into the heap when down converting), but it's not for the producer. However, down converting
40        * in the producer is very uncommon and the extra complexity to handle that case is not worth it.
41        */
42       protected static ConvertedRecords<MemoryRecords> downConvert(Iterable<? extends RecordBatch> batches, byte toMagic,
43                                                                    long firstOffset, Time time) {
44           // maintain the batch along with the decompressed records to avoid the need to decompress again
45           List<RecordBatchAndRecords> recordBatchAndRecordsList = new ArrayList<>();
46           int totalSizeEstimate = 0;
47           long startNanos = time.nanoseconds();
48   
49           for (RecordBatch batch : batches) {
50               if (toMagic < RecordBatch.MAGIC_VALUE_V2) {
51                   if (batch.isControlBatch())
52                       continue;
53   
54                   if (batch.compressionType() == CompressionType.ZSTD)
55                       throw new UnsupportedCompressionTypeException("Down-conversion of zstandard-compressed batches " +
56                           "is not supported");
57               }
58   
59               if (batch.magic() <= toMagic) {
60                   totalSizeEstimate += batch.sizeInBytes();
61                   recordBatchAndRecordsList.add(new RecordBatchAndRecords(batch, null, null));
62               } else {
63                   List<Record> records = new ArrayList<>();
64                   for (Record record : batch) {
65                       // See the method javadoc for an explanation
66                       if (toMagic > RecordBatch.MAGIC_VALUE_V1 || batch.isCompressed() || record.offset() >= firstOffset)
67                           records.add(record);
68                   }
69                   if (records.isEmpty())
70                       continue;
71                   final long baseOffset;
72                   if (batch.magic() >= RecordBatch.MAGIC_VALUE_V2 && toMagic >= RecordBatch.MAGIC_VALUE_V2)
73                       baseOffset = batch.baseOffset();
74                   else
75                       baseOffset = records.get(0).offset();
76                   totalSizeEstimate += AbstractRecords.estimateSizeInBytes(toMagic, baseOffset, batch.compressionType(), records);
77                   recordBatchAndRecordsList.add(new RecordBatchAndRecords(batch, records, baseOffset));
78               }
79           }
80   
81           ByteBuffer buffer = ByteBuffer.allocate(totalSizeEstimate);
82           long temporaryMemoryBytes = 0;
83           int numRecordsConverted = 0;
84   
85           for (RecordBatchAndRecords recordBatchAndRecords : recordBatchAndRecordsList) {
86               temporaryMemoryBytes += recordBatchAndRecords.batch.sizeInBytes();
87               if (recordBatchAndRecords.batch.magic() <= toMagic) {
88                   buffer = Utils.ensureCapacity(buffer, buffer.position() + recordBatchAndRecords.batch.sizeInBytes());
89                   recordBatchAndRecords.batch.writeTo(buffer);
90               } else {
91                   MemoryRecordsBuilder builder = convertRecordBatch(toMagic, buffer, recordBatchAndRecords);
92                   buffer = builder.buffer();
93                   temporaryMemoryBytes += builder.uncompressedBytesWritten();
94                   numRecordsConverted += builder.numRecords();
95               }
96           }
97   
98           buffer.flip();
99           RecordConversionStats stats = new RecordConversionStats(temporaryMemoryBytes, numRecordsConverted,
100                   time.nanoseconds() - startNanos);
101           return new ConvertedRecords<>(MemoryRecords.readableRecords(buffer), stats);
102       }
103   
104       /**
105        * Return a buffer containing the converted record batches. The returned buffer may not be the same as the received
106        * one (e.g. it may require expansion).
107        */
108       private static MemoryRecordsBuilder convertRecordBatch(byte magic, ByteBuffer buffer, RecordBatchAndRecords recordBatchAndRecords) {
109           RecordBatch batch = recordBatchAndRecords.batch;
110           final TimestampType timestampType = batch.timestampType();
111           long logAppendTime = timestampType == TimestampType.LOG_APPEND_TIME ? batch.maxTimestamp() : RecordBatch.NO_TIMESTAMP;
112   
113           MemoryRecordsBuilder builder = MemoryRecords.builder(buffer, magic, batch.compressionType(),
114                   timestampType, recordBatchAndRecords.baseOffset, logAppendTime);
115           for (Record record : recordBatchAndRecords.records) {
116               // Down-convert this record. Ignore headers when down-converting to V0 and V1 since they are not supported
117               if (magic > RecordBatch.MAGIC_VALUE_V1)
118                   builder.append(record);
119               else
120                   builder.appendWithOffset(record.offset(), record.timestamp(), record.key(), record.value());
121           }
122   
123           builder.close();
124           return builder;
125       }
126   
127   
128       private static class RecordBatchAndRecords {
129           private final RecordBatch batch;
130           private final List<Record> records;
131           private final Long baseOffset;
132   
133           private RecordBatchAndRecords(RecordBatch batch, List<Record> records, Long baseOffset) {
134               this.batch = batch;
135               this.records = records;
136               this.baseOffset = baseOffset;
137           }
138       }
