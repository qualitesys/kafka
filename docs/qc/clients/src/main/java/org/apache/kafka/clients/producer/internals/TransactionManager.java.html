155           private boolean isTransitionValid(State source, State target) {
156               switch (target) {
157                   case UNINITIALIZED:
158                       return source == READY;
159                   case INITIALIZING:
160                       return source == UNINITIALIZED || source == ABORTING_TRANSACTION;
161                   case READY:
162                       return source == INITIALIZING || source == COMMITTING_TRANSACTION || source == ABORTING_TRANSACTION;
163                   case IN_TRANSACTION:
164                       return source == READY;
165                   case COMMITTING_TRANSACTION:
166                       return source == IN_TRANSACTION;
167                   case ABORTING_TRANSACTION:
168                       return source == IN_TRANSACTION || source == ABORTABLE_ERROR;
169                   case ABORTABLE_ERROR:
170                       return source == IN_TRANSACTION || source == COMMITTING_TRANSACTION || source == ABORTABLE_ERROR;
171                   case FATAL_ERROR:
172                   default:
173                       // We can transition to FATAL_ERROR unconditionally.
174                       // FATAL_ERROR is never a valid starting state for any transition. So the only option is to close the
175                       // producer or do purely non transactional requests.
176                       return true;
177               }
178           }

277       private TransactionalRequestResult beginCompletingTransaction(TransactionResult transactionResult) {
278           if (!newPartitionsInTransaction.isEmpty())
279               enqueueRequest(addPartitionsToTransactionHandler());
280   
281           // If the error is an INVALID_PRODUCER_ID_MAPPING error, the server will not accept an EndTxnRequest, so skip
282           // directly to InitProducerId. Otherwise, we must first abort the transaction, because the producer will be
283           // fenced if we directly call InitProducerId.
284           if (!(lastError instanceof InvalidPidMappingException)) {
285               EndTxnRequest.Builder builder = new EndTxnRequest.Builder(
286                       new EndTxnRequestData()
287                               .setTransactionalId(transactionalId)
288                               .setProducerId(producerIdAndEpoch.producerId)
289                               .setProducerEpoch(producerIdAndEpoch.epoch)
290                               .setCommitted(transactionResult.id));
291   
292               EndTxnHandler handler = new EndTxnHandler(builder);
293               enqueueRequest(handler);
294               if (!epochBumpRequired) {
295                   return handler.result;
296               }
297           }

328           maybeFailWithError();

624       synchronized void handleFailedBatch(ProducerBatch batch, RuntimeException exception, boolean adjustSequenceNumbers) {
625           maybeTransitionToErrorState(exception);
626           removeInFlightBatch(batch);
627   
628           if (hasFatalError()) {
629               log.debug("Ignoring batch {} with producer id {}, epoch {}, and sequence number {} " +
630                               "since the producer is already in fatal error state", batch, batch.producerId(),
631                       batch.producerEpoch(), batch.baseSequence(), exception);
632               return;
633           }
634   
635           if (exception instanceof OutOfOrderSequenceException && !isTransactional()) {
636               log.error("The broker returned {} for topic-partition {} with producerId {}, epoch {}, and sequence number {}",
637                       exception, batch.topicPartition, batch.producerId(), batch.producerEpoch(), batch.baseSequence());
638   
639               // If we fail with an OutOfOrderSequenceException, we have a gap in the log. Bump the epoch for this
640               // partition, which will reset the sequence number to 0 and allow us to continue
641               requestEpochBumpForPartition(batch.topicPartition);
642           } else if (exception instanceof UnknownProducerIdException) {
643               // If we get an UnknownProducerId for a partition, then the broker has no state for that producer. It will
644               // therefore accept a write with sequence number 0. We reset the sequence number for the partition here so
645               // that the producer can continue after aborting the transaction. All inflight-requests to this partition
646               // will also fail with an UnknownProducerId error, so the sequence will remain at 0. Note that if the
647               // broker supports bumping the epoch, we will later reset all sequence numbers after calling InitProducerId
648               resetSequenceForPartition(batch.topicPartition);
649           } else {
650               if (adjustSequenceNumbers) {
651                   if (!isTransactional()) {
652                       requestEpochBumpForPartition(batch.topicPartition);
653                   } else {
654                       adjustSequencesDueToFailedBatch(batch);
655                   }
656               }
657           }
658       }
659   
660       // If a batch is failed fatally, the sequence numbers for future batches bound for the partition must be adjusted
661       // so that they don't fail with the OutOfOrderSequenceException.
662       //
663       // This method must only be called when we know that the batch is question has been unequivocally failed by the broker,
664       // ie. it has received a confirmed fatal status code like 'Message Too Large' or something similar.
665       private void adjustSequencesDueToFailedBatch(ProducerBatch batch) {
666           if (!txnPartitionMap.contains(batch.topicPartition))
667               // Sequence numbers are not being tracked for this partition. This could happen if the producer id was just
668               // reset due to a previous OutOfOrderSequenceException.
669               return;
670           log.debug("producerId: {}, send to partition {} failed fatally. Reducing future sequence numbers by {}",
671                   batch.producerId(), batch.topicPartition, batch.recordCount);
672           int currentSequence = sequenceNumber(batch.topicPartition);
673           currentSequence -= batch.recordCount;
674           if (currentSequence < 0)
675               throw new IllegalStateException("Sequence number for partition " + batch.topicPartition + " is going to become negative: " + currentSequence);
676   
677           setNextSequence(batch.topicPartition, currentSequence);
678   
679           txnPartitionMap.get(batch.topicPartition).resetSequenceNumbers(inFlightBatch -> {
680               if (inFlightBatch.baseSequence() < batch.baseSequence())
681                   return;
682   
683               int newSequence = inFlightBatch.baseSequence() - batch.recordCount;
684               if (newSequence < 0)
685                   throw new IllegalStateException("Sequence number for batch with sequence " + inFlightBatch.baseSequence()
686                           + " for partition " + batch.topicPartition + " is going to become negative: " + newSequence);
687   
688               inFlightBatch.resetProducerState(new ProducerIdAndEpoch(inFlightBatch.producerId(), inFlightBatch.producerEpoch()), newSequence, inFlightBatch.isTransactional());
689           });
690       }

717       // unresolved, either bump the epoch if possible, or transition to a fatal error
718       synchronized void maybeResolveSequences() {
719           for (Iterator<TopicPartition> iter = partitionsWithUnresolvedSequences.keySet().iterator(); iter.hasNext(); ) {

727                       iter.remove();
728                   } else {
729                       // We would enter this branch if all in flight batches were ultimately expired in the producer.
730                       if (isTransactional()) {
731                           // For the transactional producer, we bump the epoch if possible, otherwise we transition to a fatal error

733                                   "sent messages and can no longer retry them. ";

737                                       "the transaction and continue.");

739                           } else {
740                               KafkaException exception = new KafkaException(unackedMessagesErr + "It isn't safe to continue.");
741                               transitionToFatalError(exception);
742                           }
743                       } else {

745                           log.info("No inflight batches remaining for {}, last ack'd sequence for partition is {}, next sequence is {}. " +

748                           requestEpochBumpForPartition(topicPartition);

754           }

882           Errors error = response.error;
883   
884           // An UNKNOWN_PRODUCER_ID means that we have lost the producer state on the broker. Depending on the log start
885           // offset, we may want to retry these, as described for each case below. If none of those apply, then for the
886           // idempotent producer, we will locally bump the epoch and reset the sequence numbers of in-flight batches from
887           // sequence 0, then retry the failed batch, which should now succeed. For the transactional producer, allow the
888           // batch to fail. When processing the failed batch, we will transition to an abortable error and set a flag
889           // indicating that we need to bump the epoch (if supported by the broker).
890           if (error == Errors.UNKNOWN_PRODUCER_ID) {
891               if (response.logStartOffset == -1) {
892                   // We don't know the log start offset with this response. We should just retry the request until we get it.
893                   // The UNKNOWN_PRODUCER_ID error code was added along with the new ProduceResponse which includes the
894                   // logStartOffset. So the '-1' sentinel is not for backward compatibility. Instead, it is possible for
895                   // a broker to not know the logStartOffset at when it is returning the response because the partition
896                   // may have moved away from the broker from the time the error was initially raised to the time the
897                   // response was being constructed. In these cases, we should just retry the request: we are guaranteed
898                   // to eventually get a logStartOffset once things settle down.
899                   return true;
900               }
901   
902               if (batch.sequenceHasBeenReset()) {
903                   // When the first inflight batch fails due to the truncation case, then the sequences of all the other
904                   // in flight batches would have been restarted from the beginning. However, when those responses
905                   // come back from the broker, they would also come with an UNKNOWN_PRODUCER_ID error. In this case, we should not
906                   // reset the sequence numbers to the beginning.
907                   return true;
908               } else if (lastAckedOffset(batch.topicPartition).orElse(NO_LAST_ACKED_SEQUENCE_NUMBER) < response.logStartOffset) {
909                   // The head of the log has been removed, probably due to the retention time elapsing. In this case,
910                   // we expect to lose the producer state. For the transactional producer, reset the sequences of all
911                   // inflight batches to be from the beginning and retry them, so that the transaction does not need to
912                   // be aborted. For the idempotent producer, bump the epoch to avoid reusing (sequence, epoch) pairs
913                   if (isTransactional()) {
914                       txnPartitionMap.startSequencesAtBeginning(batch.topicPartition, this.producerIdAndEpoch);
915                   } else {
916                       requestEpochBumpForPartition(batch.topicPartition);
917                   }
918                   return true;
919               }
920   
921               if (!isTransactional()) {
922                   // For the idempotent producer, always retry UNKNOWN_PRODUCER_ID errors. If the batch has the current
923                   // producer ID and epoch, request a bump of the epoch. Otherwise just retry the produce.
924                   requestEpochBumpForPartition(batch.topicPartition);
925                   return true;
926               }
927           } else if (error == Errors.OUT_OF_ORDER_SEQUENCE_NUMBER) {
928               if (!hasUnresolvedSequence(batch.topicPartition) &&
929                       (batch.sequenceHasBeenReset() || !isNextSequence(batch.topicPartition, batch.baseSequence()))) {
930                   // We should retry the OutOfOrderSequenceException if the batch is _not_ the next batch, ie. its base

932                   return true;
933               } else if (!isTransactional()) {
934                   // For the idempotent producer, retry all OUT_OF_ORDER_SEQUENCE_NUMBER errors. If there are no
935                   // unresolved sequences, or this batch is the one immediately following an unresolved sequence, we know
936                   // there is actually a gap in the sequences, and we bump the epoch. Otherwise, retry without bumping
937                   // and wait to see if the sequence resolves
938                   if (!hasUnresolvedSequence(batch.topicPartition) ||
939                           isNextSequenceForUnresolvedPartition(batch.topicPartition, batch.baseSequence())) {

944           }
