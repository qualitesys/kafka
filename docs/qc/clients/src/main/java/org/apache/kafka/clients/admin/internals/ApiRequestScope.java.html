19   import java.util.OptionalInt;
20   
21   /**
22    * This interface is used by {@link AdminApiDriver} to bridge the gap
23    * to the internal `NodeProvider` defined in
24    * {@link org.apache.kafka.clients.admin.KafkaAdminClient}. However, a
25    * request scope is more than just a target broker specification. It also
26    * provides a way to group key lookups according to different batching
27    * mechanics. See {@link AdminApiLookupStrategy#lookupScope(Object)} for
28    * more detail.
29    */
30   public interface ApiRequestScope {
31   
32       /**
33        * Get the target broker ID that a request is intended for or
34        * empty if the request can be sent to any broker.
35        *
36        * Note that if the destination broker ID is present in the
37        * {@link ApiRequestScope} returned by {@link AdminApiLookupStrategy#lookupScope(Object)},
38        * then no lookup will be attempted.
39        *
40        * @return optional broker ID
41        */
42       default OptionalInt destinationBrokerId() {
43           return OptionalInt.empty();
44       }
45   
46   }
