17   package org.apache.kafka.common.record;
18   
19   import org.apache.kafka.common.network.Send;
20   import org.apache.kafka.common.network.TransferableChannel;
21   
22   import java.io.EOFException;
23   import java.io.IOException;
24   import java.nio.ByteBuffer;
25   
26   public abstract class RecordsSend<T extends BaseRecords> implements Send {
27       private static final ByteBuffer EMPTY_BYTE_BUFFER = ByteBuffer.allocate(0);
28   
29       private final T records;
30       private final int maxBytesToWrite;
31       private int remaining;
32       private boolean pending = false;
33   
34       protected RecordsSend(T records, int maxBytesToWrite) {
35           this.records = records;
36           this.maxBytesToWrite = maxBytesToWrite;
37           this.remaining = maxBytesToWrite;
38       }
39   
40       @Override
41       public boolean completed() {
42           return remaining <= 0 && !pending;
43       }
44   
45       @Override
46       public final long writeTo(TransferableChannel channel) throws IOException {
47           long written = 0;
48   
49           if (remaining > 0) {
50               written = writeTo(channel, size() - remaining, remaining);
51               if (written < 0)
52                   throw new EOFException("Wrote negative bytes to channel. This shouldn't happen.");
53               remaining -= written;
54           }
55   
56           pending = channel.hasPendingWrites();
57           if (remaining <= 0 && pending)
58               channel.write(EMPTY_BYTE_BUFFER);
59   
60           return written;
61       }
62   
63       @Override
64       public long size() {
65           return maxBytesToWrite;
66       }
67   
68       protected T records() {
69           return records;
70       }
71   
72       /**
73        * Write records up to `remaining` bytes to `channel`. The implementation is allowed to be stateful. The contract
74        * from the caller is that the first invocation will be with `previouslyWritten` equal to 0, and `remaining` equal to
75        * the to maximum bytes we want to write the to `channel`. `previouslyWritten` and `remaining` will be adjusted
76        * appropriately for every subsequent invocation. See {@link #writeTo} for example expected usage.
77        * @param channel The channel to write to
78        * @param previouslyWritten Bytes written in previous calls to {@link #writeTo(TransferableChannel, long, int)}; 0 if being called for the first time
79        * @param remaining Number of bytes remaining to be written
80        * @return The number of bytes actually written
81        * @throws IOException For any IO errors
82        */
83       protected abstract long writeTo(TransferableChannel channel, long previouslyWritten, int remaining) throws IOException;
84   }
