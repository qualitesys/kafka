20   import org.apache.kafka.common.KafkaFuture;
21   import org.apache.kafka.common.annotation.InterfaceStability;
22   import java.util.HashMap;
23   import java.util.Collection;
24   import java.util.Map;
25   import java.util.concurrent.ExecutionException;
26   import java.util.stream.Collectors;
27   import java.util.stream.Stream;
28   
29   import org.apache.kafka.common.protocol.Errors;
30   import org.apache.kafka.common.requests.DescribeLogDirsResponse;
31   
32   
33   /**
34    * The result of the {@link Admin#describeLogDirs(Collection)} call.
35    *
36    * The API of this class is evolving, see {@link Admin} for details.
37    */
38   @InterfaceStability.Evolving
39   public class DescribeLogDirsResult {
40       private final Map<Integer, KafkaFuture<Map<String, LogDirDescription>>> futures;
41   
42       DescribeLogDirsResult(Map<Integer, KafkaFuture<Map<String, LogDirDescription>>> futures) {
43           this.futures = futures;
44       }
45   
46       /**
47        * Return a map from brokerId to future which can be used to check the information of partitions on each individual broker.
48        * @deprecated Deprecated Since Kafka 2.7. Use {@link #descriptions()}.
49        */
50       @Deprecated
51       @SuppressWarnings("deprecation")
52       public Map<Integer, KafkaFuture<Map<String, DescribeLogDirsResponse.LogDirInfo>>> values() {
53           return descriptions().entrySet().stream()
54               .collect(Collectors.toMap(
55                   Map.Entry::getKey,
56                   entry -> entry.getValue().thenApply(map -> convertMapValues(map))));
57       }
58   
59       @SuppressWarnings("deprecation")
60       private Map<String, DescribeLogDirsResponse.LogDirInfo> convertMapValues(Map<String, LogDirDescription> map) {
61           Stream<Map.Entry<String, LogDirDescription>> stream = map.entrySet().stream();
62           return stream.collect(Collectors.toMap(
63               Map.Entry::getKey,
64               infoEntry -> {
65                   LogDirDescription logDir = infoEntry.getValue();
66                   return new DescribeLogDirsResponse.LogDirInfo(logDir.error() == null ? Errors.NONE : Errors.forException(logDir.error()),
67                       logDir.replicaInfos().entrySet().stream().collect(Collectors.toMap(
68                           Map.Entry::getKey,
69                           replicaEntry -> new DescribeLogDirsResponse.ReplicaInfo(
70                               replicaEntry.getValue().size(),
71                               replicaEntry.getValue().offsetLag(),
72                               replicaEntry.getValue().isFuture())
73                   )));
74               }));
75       }
76   
77       /**
78        * Return a map from brokerId to future which can be used to check the information of partitions on each individual broker.
79        * The result of the future is a map from broker log directory path to a description of that log directory.
80        */
81       public Map<Integer, KafkaFuture<Map<String, LogDirDescription>>> descriptions() {
82           return futures;
83       }
84   
85       /**
86        * Return a future which succeeds only if all the brokers have responded without error
87        * @deprecated Deprecated Since Kafka 2.7. Use {@link #allDescriptions()}.
88        */
89       @Deprecated
90       @SuppressWarnings("deprecation")
91       public KafkaFuture<Map<Integer, Map<String, DescribeLogDirsResponse.LogDirInfo>>> all() {
92           return allDescriptions().thenApply(map -> map.entrySet().stream().collect(Collectors.toMap(
93               entry -> entry.getKey(),
94               entry -> convertMapValues(entry.getValue())
95           )));
96       }
97   
98       /**
99        * Return a future which succeeds only if all the brokers have responded without error.
100        * The result of the future is a map from brokerId to a map from broker log directory path
101        * to a description of that log directory.
102        */
103       public KafkaFuture<Map<Integer, Map<String, LogDirDescription>>> allDescriptions() {
104           return KafkaFuture.allOf(futures.values().toArray(new KafkaFuture[0])).
105               thenApply(v -> {
106                   Map<Integer, Map<String, LogDirDescription>> descriptions = new HashMap<>(futures.size());
107                   for (Map.Entry<Integer, KafkaFuture<Map<String, LogDirDescription>>> entry : futures.entrySet()) {
108                       try {
109                           descriptions.put(entry.getKey(), entry.getValue().get());
110                       } catch (InterruptedException | ExecutionException e) {
111                           // This should be unreachable, because allOf ensured that all the futures completed successfully.
112                           throw new RuntimeException(e);
113                       }
114                   }
115                   return descriptions;
116               });
117       }
118   }
