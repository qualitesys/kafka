35   /**
36    * <p>The sticky assignor serves two purposes. First, it guarantees an assignment that is as balanced as possible, meaning either:
37    * <ul>
38    * <li>the numbers of topic partitions assigned to consumers differ by at most one; or</li>
39    * <li>each consumer that has 2+ fewer topic partitions than some other consumer cannot get any of those topic partitions transferred to it.</li>
40    * </ul>
41    * Second, it preserved as many existing assignment as possible when a reassignment occurs. This helps in saving some of the
42    * overhead processing when topic partitions move from one consumer to another.</p>
43    *
44    * <p>Starting fresh it would work by distributing the partitions over consumers as evenly as possible. Even though this may sound similar to
45    * how round robin assignor works, the second example below shows that it is not.
46    * During a reassignment it would perform the reassignment in such a way that in the new assignment
47    * <ol>
48    * <li>topic partitions are still distributed as evenly as possible, and</li>
49    * <li>topic partitions stay with their previously assigned consumers as much as possible.</li>
50    * </ol>
51    * Of course, the first goal above takes precedence over the second one.</p>
52    *
53    * <p><b>Example 1.</b> Suppose there are three consumers <code>C0</code>, <code>C1</code>, <code>C2</code>,
54    * four topics <code>t0,</code> <code>t1</code>, <code>t2</code>, <code>t3</code>, and each topic has 2 partitions,
55    * resulting in partitions <code>t0p0</code>, <code>t0p1</code>, <code>t1p0</code>, <code>t1p1</code>, <code>t2p0</code>,
56    * <code>t2p1</code>, <code>t3p0</code>, <code>t3p1</code>. Each consumer is subscribed to all three topics.
57    *
58    * The assignment with both sticky and round robin assignors will be:
59    * <ul>
60    * <li><code>C0: [t0p0, t1p1, t3p0]</code></li>
61    * <li><code>C1: [t0p1, t2p0, t3p1]</code></li>
62    * <li><code>C2: [t1p0, t2p1]</code></li>
63    * </ul>
64    *
65    * Now, let's assume <code>C1</code> is removed and a reassignment is about to happen. The round robin assignor would produce:
66    * <ul>
67    * <li><code>C0: [t0p0, t1p0, t2p0, t3p0]</code></li>
68    * <li><code>C2: [t0p1, t1p1, t2p1, t3p1]</code></li>
69    * </ul>
70    *
71    * while the sticky assignor would result in:
72    * <ul>
73    * <li><code>C0 [t0p0, t1p1, t3p0, t2p0]</code></li>
74    * <li><code>C2 [t1p0, t2p1, t0p1, t3p1]</code></li>
75    * </ul>
76    * preserving all the previous assignments (unlike the round robin assignor).
77    *</p>
78    * <p><b>Example 2.</b> There are three consumers <code>C0</code>, <code>C1</code>, <code>C2</code>,
79    * and three topics <code>t0</code>, <code>t1</code>, <code>t2</code>, with 1, 2, and 3 partitions respectively.
80    * Therefore, the partitions are <code>t0p0</code>, <code>t1p0</code>, <code>t1p1</code>, <code>t2p0</code>,
81    * <code>t2p1</code>, <code>t2p2</code>. <code>C0</code> is subscribed to <code>t0</code>; <code>C1</code> is subscribed to
82    * <code>t0</code>, <code>t1</code>; and <code>C2</code> is subscribed to <code>t0</code>, <code>t1</code>, <code>t2</code>.
83    *
84    * The round robin assignor would come up with the following assignment:
85    * <ul>
86    * <li><code>C0 [t0p0]</code></li>
87    * <li><code>C1 [t1p0]</code></li>
88    * <li><code>C2 [t1p1, t2p0, t2p1, t2p2]</code></li>
89    * </ul>
90    *
91    * which is not as balanced as the assignment suggested by sticky assignor:
92    * <ul>
93    * <li><code>C0 [t0p0]</code></li>
94    * <li><code>C1 [t1p0, t1p1]</code></li>
95    * <li><code>C2 [t2p0, t2p1, t2p2]</code></li>
96    * </ul>
97    *
98    * Now, if consumer <code>C0</code> is removed, these two assignors would produce the following assignments.
99    * Round Robin (preserves 3 partition assignments):
100    * <ul>
101    * <li><code>C1 [t0p0, t1p1]</code></li>
102    * <li><code>C2 [t1p0, t2p0, t2p1, t2p2]</code></li>
103    * </ul>
104    *
105    * Sticky (preserves 5 partition assignments):
106    * <ul>
107    * <li><code>C1 [t1p0, t1p1, t0p0]</code></li>
108    * <li><code>C2 [t2p0, t2p1, t2p2]</code></li>
109    * </ul>
110    *</p>
111    * <h3>Impact on <code>ConsumerRebalanceListener</code></h3>
112    * The sticky assignment strategy can provide some optimization to those consumers that have some partition cleanup code
113    * in their <code>onPartitionsRevoked()</code> callback listeners. The cleanup code is placed in that callback listener
114    * because the consumer has no assumption or hope of preserving any of its assigned partitions after a rebalance when it
115    * is using range or round robin assignor. The listener code would look like this:
116    * <pre>
117    * {@code
118    * class TheOldRebalanceListener implements ConsumerRebalanceListener {
119    *
120    *   void onPartitionsRevoked(Collection<TopicPartition> partitions) {
121    *     for (TopicPartition partition: partitions) {
122    *       commitOffsets(partition);
123    *       cleanupState(partition);
124    *     }
125    *   }
126    *
127    *   void onPartitionsAssigned(Collection<TopicPartition> partitions) {
128    *     for (TopicPartition partition: partitions) {
129    *       initializeState(partition);
130    *       initializeOffset(partition);
131    *     }
132    *   }
133    * }
134    * }
135    * </pre>
136    *
137    * As mentioned above, one advantage of the sticky assignor is that, in general, it reduces the number of partitions that
138    * actually move from one consumer to another during a reassignment. Therefore, it allows consumers to do their cleanup
139    * more efficiently. Of course, they still can perform the partition cleanup in the <code>onPartitionsRevoked()</code>
140    * listener, but they can be more efficient and make a note of their partitions before and after the rebalance, and do the
141    * cleanup after the rebalance only on the partitions they have lost (which is normally not a lot). The code snippet below
142    * clarifies this point:
143    * <pre>
144    * {@code
145    * class TheNewRebalanceListener implements ConsumerRebalanceListener {
146    *   Collection<TopicPartition> lastAssignment = Collections.emptyList();
147    *
148    *   void onPartitionsRevoked(Collection<TopicPartition> partitions) {
149    *     for (TopicPartition partition: partitions)
150    *       commitOffsets(partition);
151    *   }
152    *
153    *   void onPartitionsAssigned(Collection<TopicPartition> assignment) {
154    *     for (TopicPartition partition: difference(lastAssignment, assignment))
155    *       cleanupState(partition);
156    *
157    *     for (TopicPartition partition: difference(assignment, lastAssignment))
158    *       initializeState(partition);
159    *
160    *     for (TopicPartition partition: assignment)
161    *       initializeOffset(partition);
162    *
163    *     this.lastAssignment = assignment;
164    *   }
165    * }
166    * }
167    * </pre>
168    *
169    * Any consumer that uses sticky assignment can leverage this listener like this:
170    * <code>consumer.subscribe(topics, new TheNewRebalanceListener());</code>
171    *
172    * Note that you can leverage the {@link CooperativeStickyAssignor} so that only partitions which are being
173    * reassigned to another consumer will be revoked. That is the preferred assignor for newer cluster. See
174    * {@link ConsumerPartitionAssignor.RebalanceProtocol} for a detailed explanation of cooperative rebalancing.
175    */
176   public class StickyAssignor extends AbstractStickyAssignor {
177       public static final String STICKY_ASSIGNOR_NAME = "sticky";
178   
179       // these schemas are used for preserving consumer's previously assigned partitions
180       // list and sending it as user data to the leader during a rebalance
181       static final String TOPIC_PARTITIONS_KEY_NAME = "previous_assignment";
182       static final String TOPIC_KEY_NAME = "topic";
183       static final String PARTITIONS_KEY_NAME = "partitions";
184       private static final String GENERATION_KEY_NAME = "generation";
185   
186       static final Schema TOPIC_ASSIGNMENT = new Schema(
187           new Field(TOPIC_KEY_NAME, Type.STRING),
188           new Field(PARTITIONS_KEY_NAME, new ArrayOf(Type.INT32)));
189       static final Schema STICKY_ASSIGNOR_USER_DATA_V0 = new Schema(
190           new Field(TOPIC_PARTITIONS_KEY_NAME, new ArrayOf(TOPIC_ASSIGNMENT)));
191       private static final Schema STICKY_ASSIGNOR_USER_DATA_V1 = new Schema(
192           new Field(TOPIC_PARTITIONS_KEY_NAME, new ArrayOf(TOPIC_ASSIGNMENT)),
193           new Field(GENERATION_KEY_NAME, Type.INT32));
194   
195       private List<TopicPartition> memberAssignment = null;
196       private int generation = DEFAULT_GENERATION; // consumer group generation
197   
198       @Override
199       public String name() {
200           return STICKY_ASSIGNOR_NAME;
201       }
202   
203       @Override
204       public void onAssignment(Assignment assignment, ConsumerGroupMetadata metadata) {
205           memberAssignment = assignment.partitions();
206           this.generation = metadata.generationId();
207       }
208   
209       @Override
210       public ByteBuffer subscriptionUserData(Set<String> topics) {
211           if (memberAssignment == null)
212               return null;
213   
214           return serializeTopicPartitionAssignment(new MemberData(memberAssignment, Optional.of(generation)));
215       }
216   
217       @Override
218       protected MemberData memberData(Subscription subscription) {
219           ByteBuffer userData = subscription.userData();
220           if (userData == null || !userData.hasRemaining()) {
221               return new MemberData(Collections.emptyList(), Optional.empty());
222           }
223           return deserializeTopicPartitionAssignment(userData);
224       }
225   
226       // visible for testing
227       static ByteBuffer serializeTopicPartitionAssignment(MemberData memberData) {
228           Struct struct = new Struct(STICKY_ASSIGNOR_USER_DATA_V1);
229           List<Struct> topicAssignments = new ArrayList<>();
230           for (Map.Entry<String, List<Integer>> topicEntry : CollectionUtils.groupPartitionsByTopic(memberData.partitions).entrySet()) {
231               Struct topicAssignment = new Struct(TOPIC_ASSIGNMENT);
232               topicAssignment.set(TOPIC_KEY_NAME, topicEntry.getKey());
233               topicAssignment.set(PARTITIONS_KEY_NAME, topicEntry.getValue().toArray());
234               topicAssignments.add(topicAssignment);
235           }
236           struct.set(TOPIC_PARTITIONS_KEY_NAME, topicAssignments.toArray());
237           if (memberData.generation.isPresent())
238               struct.set(GENERATION_KEY_NAME, memberData.generation.get());
239           ByteBuffer buffer = ByteBuffer.allocate(STICKY_ASSIGNOR_USER_DATA_V1.sizeOf(struct));
240           STICKY_ASSIGNOR_USER_DATA_V1.write(buffer, struct);
241           buffer.flip();
242           return buffer;
243       }
244   
245       private static MemberData deserializeTopicPartitionAssignment(ByteBuffer buffer) {
246           Struct struct;
247           ByteBuffer copy = buffer.duplicate();
248           try {
249               struct = STICKY_ASSIGNOR_USER_DATA_V1.read(buffer);
250           } catch (Exception e1) {
251               try {
252                   // fall back to older schema
253                   struct = STICKY_ASSIGNOR_USER_DATA_V0.read(copy);
254               } catch (Exception e2) {
255                   // ignore the consumer's previous assignment if it cannot be parsed
256                   return new MemberData(Collections.emptyList(), Optional.of(DEFAULT_GENERATION));
257               }
258           }
259   
260           List<TopicPartition> partitions = new ArrayList<>();
261           for (Object structObj : struct.getArray(TOPIC_PARTITIONS_KEY_NAME)) {
262               Struct assignment = (Struct) structObj;
263               String topic = assignment.getString(TOPIC_KEY_NAME);
264               for (Object partitionObj : assignment.getArray(PARTITIONS_KEY_NAME)) {
265                   Integer partition = (Integer) partitionObj;
266                   partitions.add(new TopicPartition(topic, partition));
267               }
268           }
269           // make sure this is backward compatible
270           Optional<Integer> generation = struct.hasField(GENERATION_KEY_NAME) ? Optional.of(struct.getInt(GENERATION_KEY_NAME)) : Optional.empty();
271           return new MemberData(partitions, generation);
272       }
273   }
