17   package org.apache.kafka.common.metrics;
18   
19   import java.util.Collections;
20   import java.util.List;
21   import java.util.Map;
22   import java.util.Set;
23   
24   import org.apache.kafka.common.Reconfigurable;
25   import org.apache.kafka.common.annotation.InterfaceStability;
26   import org.apache.kafka.common.config.ConfigException;
27   
28   /**
29    * A plugin interface to allow things to listen as new metrics are created so they can be reported.
30    * <p>
31    * Implement {@link org.apache.kafka.common.ClusterResourceListener} to receive cluster metadata once it's available. Please see the class documentation for ClusterResourceListener for more information.
32    */
33   public interface MetricsReporter extends Reconfigurable, AutoCloseable {
34   
35       /**
36        * This is called when the reporter is first registered to initially register all existing metrics
37        * @param metrics All currently existing metrics
38        */
39       void init(List<KafkaMetric> metrics);
40   
41       /**
42        * This is called whenever a metric is updated or added
43        * @param metric
44        */
45       void metricChange(KafkaMetric metric);
46   
47       /**
48        * This is called whenever a metric is removed
49        * @param metric
50        */
51       void metricRemoval(KafkaMetric metric);
52   
53       /**
54        * Called when the metrics repository is closed.
55        */
56       void close();
57   
58       // default methods for backwards compatibility with reporters that only implement Configurable
59       default Set<String> reconfigurableConfigs() {
60           return Collections.emptySet();
61       }
62   
63       default void validateReconfiguration(Map<String, ?> configs) throws ConfigException {
64       }
65   
66       default void reconfigure(Map<String, ?> configs) {
67       }
68   
69       /**
70        * Sets the context labels for the service or library exposing metrics. This will be called before {@link #init(List)} and may be called anytime after that.
71        *
72        * @param metricsContext the metric context
73        */
74       @InterfaceStability.Evolving
75       default void contextChange(MetricsContext metricsContext) {
76       }
77   }
