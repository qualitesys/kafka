18   package org.apache.kafka.common.utils;
19   
20   /**
21    * This is a helper class which makes blocking methods with a timeout easier to implement.
22    * In particular it enables use cases where a high-level blocking call with a timeout is
23    * composed of several lower level calls, each of which has their own respective timeouts. The idea
24    * is to create a single timer object for the high level timeout and carry it along to
25    * all of the lower level methods. This class also handles common problems such as integer overflow.
26    * This class also ensures monotonic updates to the timer even if the underlying clock is subject
27    * to non-monotonic behavior. For example, the remaining time returned by {@link #remainingMs()} is
28    * guaranteed to decrease monotonically until it hits zero.
29    *
30    * Note that it is up to the caller to ensure progress of the timer using one of the
31    * {@link #update()} methods or {@link #sleep(long)}. The timer will cache the current time and
32    * return it indefinitely until the timer has been updated. This allows the caller to limit
33    * unnecessary system calls and update the timer only when needed. For example, a timer which is
34    * waiting a request sent through the {@link org.apache.kafka.clients.NetworkClient} should call
35    * {@link #update()} following each blocking call to
36    * {@link org.apache.kafka.clients.NetworkClient#poll(long, long)}.
37    *
38    * A typical usage might look something like this:
39    *
40    * <pre>
41    *     Time time = Time.SYSTEM;
42    *     Timer timer = time.timer(500);
43    *
44    *     while (!conditionSatisfied() && timer.notExpired) {
45    *         client.poll(timer.remainingMs(), timer.currentTimeMs());
46    *         timer.update();
47    *     }
48    * </pre>
49    */
50   public class Timer {
51       private final Time time;
52       private long startMs;
53       private long currentTimeMs;
54       private long deadlineMs;
55       private long timeoutMs;
56   
57       Timer(Time time, long timeoutMs) {
58           this.time = time;
59           update();
60           reset(timeoutMs);
61       }
62   
63       /**
64        * Check timer expiration. Like {@link #remainingMs()}, this depends on the current cached
65        * time in milliseconds, which is only updated through one of the {@link #update()} methods
66        * or with {@link #sleep(long)};
67        *
68        * @return true if the timer has expired, false otherwise
69        */
70       public boolean isExpired() {
71           return currentTimeMs >= deadlineMs;
72       }
73   
74       /**
75        * Check whether the timer has not yet expired.
76        * @return true if there is still time remaining before expiration
77        */
78       public boolean notExpired() {
79           return !isExpired();
80       }
81   
82       /**
83        * Reset the timer to the specific timeout. This will use the underlying {@link #Timer(Time, long)}
84        * implementation to update the current cached time in milliseconds and it will set a new timer
85        * deadline.
86        *
87        * @param timeoutMs The new timeout in milliseconds
88        */
89       public void updateAndReset(long timeoutMs) {
90           update();
91           reset(timeoutMs);
92       }
93   
94       /**
95        * Reset the timer using a new timeout. Note that this does not update the cached current time
96        * in milliseconds, so it typically must be accompanied with a separate call to {@link #update()}.
97        * Typically, you can just use {@link #updateAndReset(long)}.
98        *
99        * @param timeoutMs The new timeout in milliseconds
100        */
101       public void reset(long timeoutMs) {
102           if (timeoutMs < 0)
103               throw new IllegalArgumentException("Invalid negative timeout " + timeoutMs);
104   
105           this.timeoutMs = timeoutMs;
106           this.startMs = this.currentTimeMs;
107   
108           if (currentTimeMs > Long.MAX_VALUE - timeoutMs)
109               this.deadlineMs = Long.MAX_VALUE;
110           else
111               this.deadlineMs = currentTimeMs + timeoutMs;
112       }
113   
114       /**
115        * Reset the timer's deadline directly.
116        *
117        * @param deadlineMs The new deadline in milliseconds
118        */
119       public void resetDeadline(long deadlineMs) {
120           if (deadlineMs < 0)
121               throw new IllegalArgumentException("Invalid negative deadline " + deadlineMs);
122   
123           this.timeoutMs = Math.max(0, deadlineMs - this.currentTimeMs);
124           this.startMs = this.currentTimeMs;
125           this.deadlineMs = deadlineMs;
126       }
127   
128       /**
129        * Use the underlying {@link Time} implementation to update the current cached time. If
130        * the underlying time returns a value which is smaller than the current cached time,
131        * the update will be ignored.
132        */
133       public void update() {
134           update(time.milliseconds());
135       }
136   
137       /**
138        * Update the cached current time to a specific value. In some contexts, the caller may already
139        * have an accurate time, so this avoids unnecessary calls to system time.
140        *
141        * Note that if the updated current time is smaller than the cached time, then the update
142        * is ignored.
143        *
144        * @param currentTimeMs The current time in milliseconds to cache
145        */
146       public void update(long currentTimeMs) {
147           this.currentTimeMs = Math.max(currentTimeMs, this.currentTimeMs);
148       }
149   
150       /**
151        * Get the remaining time in milliseconds until the timer expires. Like {@link #currentTimeMs},
152        * this depends on the cached current time, so the returned value will not change until the timer
153        * has been updated using one of the {@link #update()} methods or {@link #sleep(long)}.
154        *
155        * @return The cached remaining time in milliseconds until timer expiration
156        */
157       public long remainingMs() {
158           return Math.max(0, deadlineMs - currentTimeMs);
159       }
160   
161       /**
162        * Get the current time in milliseconds. This will return the same cached value until the timer
163        * has been updated using one of the {@link #update()} methods or {@link #sleep(long)} is used.
164        *
165        * Note that the value returned is guaranteed to increase monotonically even if the underlying
166        * {@link Time} implementation goes backwards. Effectively, the timer will just wait for the
167        * time to catch up.
168        *
169        * @return The current cached time in milliseconds
170        */
171       public long currentTimeMs() {
172           return currentTimeMs;
173       }
174   
175       /**
176        * Get the amount of time that has elapsed since the timer began. If the timer was reset, this
177        * will be the amount of time since the last reset.
178        *
179        * @return The elapsed time since construction or the last reset
180        */
181       public long elapsedMs() {
182           return currentTimeMs - startMs;
183       }
184   
185       /**
186        * Get the current timeout value specified through {@link #reset(long)} or {@link #resetDeadline(long)}.
187        * This value is constant until altered by one of these API calls.
188        *
189        * @return The timeout in milliseconds
190        */
191       public long timeoutMs() {
192           return timeoutMs;
193       }
194   
195       /**
196        * Sleep for the requested duration and update the timer. Return when either the duration has
197        * elapsed or the timer has expired.
198        *
199        * @param durationMs The duration in milliseconds to sleep
200        */
201       public void sleep(long durationMs) {
202           long sleepDurationMs = Math.min(durationMs, remainingMs());
203           time.sleep(sleepDurationMs);
204           update();
205       }
206   }
