112   /**
113    * This class manages the fetching process with the brokers.
114    * <p>
115    * Thread-safety:
116    * Requests and responses of Fetcher may be processed by different threads since heartbeat
117    * thread may process responses. Other operations are single-threaded and invoked only from
118    * the thread polling the consumer.
119    * <ul>
120    *     <li>If a response handler accesses any shared state of the Fetcher (e.g. FetchSessionHandler),
121    *     all access to that state must be synchronized on the Fetcher instance.</li>
122    *     <li>If a response handler accesses any shared state of the coordinator (e.g. SubscriptionState),
123    *     it is assumed that all access to that state is synchronized on the coordinator instance by
124    *     the caller.</li>
125    *     <li>Responses that collate partial responses from multiple brokers (e.g. to list offsets) are
126    *     synchronized on the response future.</li>
127    *     <li>At most one request is pending for each node at any time. Nodes with pending requests are
128    *     tracked and updated after processing the response. This ensures that any state (e.g. epoch)
129    *     updated while processing responses on one thread are visible while creating the subsequent request
130    *     on a different thread.</li>
131    * </ul>
132    */

243       /**
244        * Set-up a fetch request for any node that we have assigned partitions for which doesn't already have
245        * an in-flight fetch or pending fetch data.
246        * @return number of fetches sent
247        */
248       public synchronized int sendFetches() {
249           // Update metrics in case there was an assignment change
250           sensors.maybeUpdateAssignment(subscriptions);
251   
252           Map<Node, FetchSessionHandler.FetchRequestData> fetchRequestMap = prepareFetchRequests();
253           for (Map.Entry<Node, FetchSessionHandler.FetchRequestData> entry : fetchRequestMap.entrySet()) {
254               final Node fetchTarget = entry.getKey();
255               final FetchSessionHandler.FetchRequestData data = entry.getValue();
256               final short maxVersion;
257               if (!data.canUseTopicIds()) {
258                   maxVersion = (short) 12;
259               } else {
260                   maxVersion = ApiKeys.FETCH.latestVersion();
261               }
262               final FetchRequest.Builder request = FetchRequest.Builder
263                       .forConsumer(maxVersion, this.maxWaitMs, this.minBytes, data.toSend())
264                       .isolationLevel(isolationLevel)
265                       .setMaxBytes(this.maxBytes)
266                       .metadata(data.metadata())
267                       .removed(data.toForget())
268                       .replaced(data.toReplace())
269                       .rackId(clientRackId);
270   
271               if (log.isDebugEnabled()) {
272                   log.debug("Sending {} {} to broker {}", isolationLevel, data.toString(), fetchTarget);
273               }
274               RequestFuture<ClientResponse> future = client.send(fetchTarget, request);
275               // We add the node to the set of nodes with pending fetch requests before adding the
276               // listener because the future may have been fulfilled on another thread (e.g. during a
277               // disconnection being handled by the heartbeat thread) which will mean the listener
278               // will be invoked synchronously.
279               this.nodesWithPendingFetchRequests.add(entry.getKey().id());
280               future.addListener(new RequestFutureListener<ClientResponse>() {
281                   @Override
282                   public void onSuccess(ClientResponse resp) {
283                       synchronized (Fetcher.this) {
284                           try {
285                               FetchResponse response = (FetchResponse) resp.responseBody();
286                               FetchSessionHandler handler = sessionHandler(fetchTarget.id());
287                               if (handler == null) {
288                                   log.error("Unable to find FetchSessionHandler for node {}. Ignoring fetch response.",
289                                           fetchTarget.id());
290                                   return;
291                               }
292                               if (!handler.handleResponse(response, resp.requestHeader().apiVersion())) {
293                                   if (response.error() == Errors.FETCH_SESSION_TOPIC_ID_ERROR) {
294                                       metadata.requestUpdate();
295                                   }
296                                   return;
297                               }
298   
299                               Map<TopicPartition, FetchResponseData.PartitionData> responseData = response.responseData(handler.sessionTopicNames(), resp.requestHeader().apiVersion());
300                               Set<TopicPartition> partitions = new HashSet<>(responseData.keySet());
301                               FetchResponseMetricAggregator metricAggregator = new FetchResponseMetricAggregator(sensors, partitions);
302   
303                               for (Map.Entry<TopicPartition, FetchResponseData.PartitionData> entry : responseData.entrySet()) {
304                                   TopicPartition partition = entry.getKey();
305                                   FetchRequest.PartitionData requestData = data.sessionPartitions().get(partition);
306                                   if (requestData == null) {
307                                       String message;
308                                       if (data.metadata().isFull()) {
309                                           message = MessageFormatter.arrayFormat(
310                                                   "Response for missing full request partition: partition={}; metadata={}",
311                                                   new Object[]{partition, data.metadata()}).getMessage();
312                                       } else {
313                                           message = MessageFormatter.arrayFormat(
314                                                   "Response for missing session request partition: partition={}; metadata={}; toSend={}; toForget={}; toReplace={}",
315                                                   new Object[]{partition, data.metadata(), data.toSend(), data.toForget(), data.toReplace()}).getMessage();
316                                       }
317   
318                                       // Received fetch response for missing session partition
319                                       throw new IllegalStateException(message);
320                                   } else {
321                                       long fetchOffset = requestData.fetchOffset;
322                                       FetchResponseData.PartitionData partitionData = entry.getValue();
323   
324                                       log.debug("Fetch {} at offset {} for partition {} returned fetch data {}",
325                                               isolationLevel, fetchOffset, partition, partitionData);
326   
327                                       Iterator<? extends RecordBatch> batches = FetchResponse.recordsOrFail(partitionData).batches().iterator();
328                                       short responseVersion = resp.requestHeader().apiVersion();
329   
330                                       completedFetches.add(new CompletedFetch(partition, partitionData,
331                                               metricAggregator, batches, fetchOffset, responseVersion));
332                                   }
333                               }
334   
335                               sensors.fetchLatency.record(resp.requestLatencyMs());
336                           } finally {
337                               nodesWithPendingFetchRequests.remove(fetchTarget.id());
338                           }
339                       }
340                   }

351                               nodesWithPendingFetchRequests.remove(fetchTarget.id());
352                           }
353                       }
354                   }
355               });
356   
357           }
358           return fetchRequestMap.size();
359       }

370       /**
371        * Get metadata for all topics present in Kafka cluster
372        *
373        * @param request The MetadataRequest to send
374        * @param timer Timer bounding how long this method can block
375        * @return The map of topics with their partition information
376        */
377       public Map<String, List<PartitionInfo>> getTopicMetadata(MetadataRequest.Builder request, Timer timer) {
378           // Save the round trip if no topics are requested.
379           if (!request.isAllTopics() && request.emptyTopicList())
380               return Collections.emptyMap();
381   
382           do {
383               RequestFuture<ClientResponse> future = sendMetadataRequest(request);
384               client.poll(future, timer);
385   
386               if (future.failed() && !future.isRetriable())
387                   throw future.exception();
388   
389               if (future.succeeded()) {
390                   MetadataResponse response = (MetadataResponse) future.value().responseBody();
391                   Cluster cluster = response.buildCluster();
392   
393                   Set<String> unauthorizedTopics = cluster.unauthorizedTopics();
394                   if (!unauthorizedTopics.isEmpty())
395                       throw new TopicAuthorizationException(unauthorizedTopics);
396   
397                   boolean shouldRetry = false;
398                   Map<String, Errors> errors = response.errors();
399                   if (!errors.isEmpty()) {
400                       // if there were errors, we need to check whether they were fatal or whether
401                       // we should just retry
402   
403                       log.debug("Topic metadata fetch included errors: {}", errors);
404   
405                       for (Map.Entry<String, Errors> errorEntry : errors.entrySet()) {
406                           String topic = errorEntry.getKey();
407                           Errors error = errorEntry.getValue();
408   
409                           if (error == Errors.INVALID_TOPIC_EXCEPTION)
410                               throw new InvalidTopicException("Topic '" + topic + "' is invalid");
411                           else if (error == Errors.UNKNOWN_TOPIC_OR_PARTITION)
412                               // if a requested topic is unknown, we just continue and let it be absent
413                               // in the returned map
414                               continue;
415                           else if (error.exception() instanceof RetriableException)
416                               shouldRetry = true;
417                           else
418                               throw new KafkaException("Unexpected error fetching metadata for topic " + topic,
419                                       error.exception());
420                       }
421                   }
422   
423                   if (!shouldRetry) {
424                       HashMap<String, List<PartitionInfo>> topicsPartitionInfos = new HashMap<>();
425                       for (String topic : cluster.topics())
426                           topicsPartitionInfos.put(topic, cluster.partitionsForTopic(topic));
427                       return topicsPartitionInfos;
428                   }
429               }
430   
431               timer.sleep(retryBackoffMs);
432           } while (timer.notExpired());

635       /**
636        * Return the fetched records, empty the record buffer and update the consumed position.
637        *
638        * NOTE: returning an {@link Fetch#isEmpty empty} fetch guarantees the consumed position is not updated.
639        *
640        * @return A {@link Fetch} for the requested partitions
641        * @throws OffsetOutOfRangeException If there is OffsetOutOfRange error in fetchResponse and
642        *         the defaultResetPolicy is NONE
643        * @throws TopicAuthorizationException If there is TopicAuthorization error in fetchResponse.
644        */
645       public Fetch<K, V> collectFetch() {
646           Fetch<K, V> fetch = Fetch.empty();
647           Queue<CompletedFetch> pausedCompletedFetches = new ArrayDeque<>();
648           int recordsRemaining = maxPollRecords;
649   
650           try {
651               while (recordsRemaining > 0) {
652                   if (nextInLineFetch == null || nextInLineFetch.isConsumed) {
653                       CompletedFetch records = completedFetches.peek();
654                       if (records == null) break;
655   
656                       if (records.notInitialized()) {
657                           try {
658                               nextInLineFetch = initializeCompletedFetch(records);
659                           } catch (Exception e) {
660                               // Remove a completedFetch upon a parse with exception if (1) it contains no records, and
661                               // (2) there are no fetched records with actual content preceding this exception.
662                               // The first condition ensures that the completedFetches is not stuck with the same completedFetch
663                               // in cases such as the TopicAuthorizationException, and the second condition ensures that no
664                               // potential data loss due to an exception in a following record.
665                               FetchResponseData.PartitionData partition = records.partitionData;
666                               if (fetch.isEmpty() && FetchResponse.recordsOrFail(partition).sizeInBytes() == 0) {
667                                   completedFetches.poll();
668                               }
669                               throw e;
670                           }
671                       } else {
672                           nextInLineFetch = records;
673                       }
674                       completedFetches.poll();
675                   } else if (subscriptions.isPaused(nextInLineFetch.partition)) {
676                       // when the partition is paused we add the records back to the completedFetches queue instead of draining
677                       // them so that they can be returned on a subsequent poll if the partition is resumed at that time
678                       log.debug("Skipping fetching records for assigned partition {} because it is paused", nextInLineFetch.partition);
679                       pausedCompletedFetches.add(nextInLineFetch);
680                       nextInLineFetch = null;
681                   } else {
682                       Fetch<K, V> nextFetch = fetchRecords(nextInLineFetch, recordsRemaining);
683                       recordsRemaining -= nextFetch.numRecords();
684                       fetch.add(nextFetch);
685                   }
686               }
687           } catch (KafkaException e) {
688               if (fetch.isEmpty())
689                   throw e;
690           } finally {
691               // add any polled completed fetches for paused partitions back to the completed fetches queue to be
692               // re-evaluated in the next poll
693               completedFetches.addAll(pausedCompletedFetches);
694           }
695   
696           return fetch;
697       }

699       private Fetch<K, V> fetchRecords(CompletedFetch completedFetch, int maxRecords) {
700           if (!subscriptions.isAssigned(completedFetch.partition)) {
701               // this can happen when a rebalance happened before fetched records are returned to the consumer's poll call
702               log.debug("Not returning fetched records for partition {} since it is no longer assigned",
703                       completedFetch.partition);
704           } else if (!subscriptions.isFetchable(completedFetch.partition)) {
705               // this can happen when a partition is paused before fetched records are returned to the consumer's
706               // poll call or if the offset is being reset
707               log.debug("Not returning fetched records for assigned partition {} since it is no longer fetchable",
708                       completedFetch.partition);
709           } else {
710               FetchPosition position = subscriptions.position(completedFetch.partition);
711               if (position == null) {
712                   throw new IllegalStateException("Missing position for fetchable partition " + completedFetch.partition);
713               }
714   
715               if (completedFetch.nextFetchOffset == position.offset) {
716                   List<ConsumerRecord<K, V>> partRecords = completedFetch.fetchRecords(maxRecords);
717   
718                   log.trace("Returning {} fetched records at offset {} for assigned partition {}",
719                           partRecords.size(), position, completedFetch.partition);
720   
721                   boolean positionAdvanced = false;
722   
723                   if (completedFetch.nextFetchOffset > position.offset) {
724                       FetchPosition nextPosition = new FetchPosition(
725                               completedFetch.nextFetchOffset,
726                               completedFetch.lastEpoch,
727                               position.currentLeader);
728                       log.trace("Updating fetch position from {} to {} for partition {} and returning {} records from `poll()`",
729                               position, nextPosition, completedFetch.partition, partRecords.size());
730                       subscriptions.position(completedFetch.partition, nextPosition);
731                       positionAdvanced = true;
732                   }
733   
734                   Long partitionLag = subscriptions.partitionLag(completedFetch.partition, isolationLevel);
735                   if (partitionLag != null)
736                       this.sensors.recordPartitionLag(completedFetch.partition, partitionLag);
737   
738                   Long lead = subscriptions.partitionLead(completedFetch.partition);
739                   if (lead != null) {
740                       this.sensors.recordPartitionLead(completedFetch.partition, lead);
741                   }
742   
743                   return Fetch.forPartition(completedFetch.partition, partRecords, positionAdvanced);
744               } else {
745                   // these records aren't next in line based on the last consumed position, ignore them
746                   // they must be from an obsolete request
747                   log.debug("Ignoring fetched records for {} at offset {} since the current position is {}",
748                           completedFetch.partition, completedFetch.nextFetchOffset, position);
749               }
750           }
751   
752           log.trace("Draining fetched records for partition {}", completedFetch.partition);
753           completedFetch.drain();
754   
755           return Fetch.empty();
756       }

813       /**

815        * with the epoch less than or equal to the epoch the partition last saw.
816        *
817        * Requests are grouped by Node for efficiency.
818        */
819       private void validateOffsetsAsync(Map<TopicPartition, FetchPosition> partitionsToValidate) {
820           final Map<Node, Map<TopicPartition, FetchPosition>> regrouped =
821               regroupFetchPositionsByLeader(partitionsToValidate);
822   
823           long nextResetTimeMs = time.milliseconds() + requestTimeoutMs;
824           regrouped.forEach((node, fetchPositions) -> {
825               if (node.isEmpty()) {
826                   metadata.requestUpdate();
827                   return;
828               }
829   
830               NodeApiVersions nodeApiVersions = apiVersions.get(node.idString());
831               if (nodeApiVersions == null) {
832                   client.tryConnect(node);
833                   return;
834               }
835   
836               if (!hasUsableOffsetForLeaderEpochVersion(nodeApiVersions)) {
837                   log.debug("Skipping validation of fetch offsets for partitions {} since the broker does not " +
838                                 "support the required protocol version (introduced in Kafka 2.3)",
839                       fetchPositions.keySet());
840                   for (TopicPartition partition : fetchPositions.keySet()) {
841                       subscriptions.completeValidation(partition);
842                   }
843                   return;
844               }
845   
846               subscriptions.setNextAllowedRetry(fetchPositions.keySet(), nextResetTimeMs);
847   
848               RequestFuture<OffsetForEpochResult> future =
849                   offsetsForLeaderEpochClient.sendAsyncRequest(node, fetchPositions);
850   
851               future.addListener(new RequestFutureListener<OffsetForEpochResult>() {
852                   @Override
853                   public void onSuccess(OffsetForEpochResult offsetsResult) {
854                       List<SubscriptionState.LogTruncation> truncations = new ArrayList<>();
855                       if (!offsetsResult.partitionsToRetry().isEmpty()) {
856                           subscriptions.setNextAllowedRetry(offsetsResult.partitionsToRetry(), time.milliseconds() + retryBackoffMs);
857                           metadata.requestUpdate();
858                       }

860                       // For each OffsetsForLeader response, check if the end-offset is lower than our current offset
861                       // for the partition. If so, it means we have experienced log truncation and need to reposition
862                       // that partition's offset.
863                       //
864                       // In addition, check whether the returned offset and epoch are valid. If not, then we should reset
