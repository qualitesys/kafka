58   /**
59    * A nioSelector interface for doing non-blocking multi-connection network I/O.
60    * <p>
61    * This class works with {@link NetworkSend} and {@link NetworkReceive} to transmit size-delimited network requests and
62    * responses.
63    * <p>
64    * A connection can be added to the nioSelector associated with an integer id by doing
65    *
66    * <pre>
67    * nioSelector.connect(&quot;42&quot;, new InetSocketAddress(&quot;google.com&quot;, server.port), 64000, 64000);
68    * </pre>
69    *
70    * The connect call does not block on the creation of the TCP connection, so the connect method only begins initiating
71    * the connection. The successful invocation of this method does not mean a valid connection has been established.
72    *
73    * Sending requests, receiving responses, processing connection completions, and disconnections on the existing
74    * connections are all done using the <code>poll()</code> call.
75    *
76    * <pre>
77    * nioSelector.send(new NetworkSend(myDestination, myBytes));
78    * nioSelector.send(new NetworkSend(myOtherDestination, myOtherBytes));
79    * nioSelector.poll(TIMEOUT_MS);
80    * </pre>
81    *
82    * The nioSelector maintains several lists that are reset by each call to <code>poll()</code> which are available via
83    * various getters. These are reset by each call to <code>poll()</code>.
84    *
85    * This class is not thread safe!
86    */
87   public class Selector implements Selectable, AutoCloseable {

235       /**
236        * Begin connecting to the given address and add the connection to this nioSelector associated with the given id
237        * number.
238        * <p>
239        * Note that this call only initiates the connection, which will be completed on a future {@link #poll(long)}
240        * call. Check {@link #connected()} to see which (if any) connections have completed after a given poll call.
241        * @param id The id for the new connection
242        * @param address The address to connect to
243        * @param sendBufferSize The send buffer for the new connection
244        * @param receiveBufferSize The receive buffer for the new connection
245        * @throws IllegalStateException if there is already a connection for that id
246        * @throws IOException if DNS resolution fails on the hostname or if the broker is down
247        */
248       @Override
249       public void connect(String id, InetSocketAddress address, int sendBufferSize, int receiveBufferSize) throws IOException {
250           ensureNotRegistered(id);
251           SocketChannel socketChannel = SocketChannel.open();
252           SelectionKey key = null;
253           try {
254               configureSocketChannel(socketChannel, sendBufferSize, receiveBufferSize);
255               boolean connected = doConnect(socketChannel, address);
256               key = registerChannel(id, socketChannel, SelectionKey.OP_CONNECT);
257   
258               if (connected) {
259                   // OP_CONNECT won't trigger for immediately connected channels
260                   log.debug("Immediately connected to node {}", id);
261                   immediatelyConnectedKeys.add(key);
262                   key.interestOps(0);
263               }
264           } catch (IOException | RuntimeException e) {
265               if (key != null)
266                   immediatelyConnectedKeys.remove(key);
267               channels.remove(id);
268               socketChannel.close();
269               throw e;
270           }
271       }

297        * Use this on server-side, when a connection is accepted by a different thread but processed by the Selector
298        * <p>
299        * If a connection already exists with the same connection id in `channels` or `closingChannels`,

301        * Kafka brokers add an incrementing index to the connection id to avoid reuse in the timing window
302        * where an existing connection may not yet have been closed by the broker when a new connection with
303        * the same remote host:port is processed.
304        * </p><p>
305        * If a `KafkaChannel` cannot be created for this connection, the `socketChannel` is closed
306        * and its selection key cancelled.
307        * </p>

360       /**
361        * Close this selector and all associated connections
362        */
363       @Override
364       public void close() {
365           List<String> connections = new ArrayList<>(channels.keySet());
366           AtomicReference<Throwable> firstException = new AtomicReference<>();
367           Utils.closeAllQuietly(firstException, "release connections",
368                   connections.stream().map(id -> (AutoCloseable) () -> close(id)).toArray(AutoCloseable[]::new));
369           // If there is any exception thrown in close(id), we should still be able
370           // to close the remaining objects, especially the sensors because keeping
371           // the sensors may lead to failure to start up the ReplicaFetcherThread if
372           // the old sensors with the same names has not yet been cleaned up.
373           Utils.closeQuietly(nioSelector, "nioSelector", firstException);
374           Utils.closeQuietly(sensors, "sensors", firstException);
375           Utils.closeQuietly(channelBuilder, "channelBuilder", firstException);
376           Throwable exception = firstException.get();
377           if (exception instanceof RuntimeException && !(exception instanceof SecurityException)) {
378               throw (RuntimeException) exception;
379           }
380       }

382       /**
383        * Queue the given request for sending in the subsequent {@link #poll(long)} calls
384        * @param send The request to send
385        */
386       public void send(NetworkSend send) {
387           String connectionId = send.destinationId();
388           KafkaChannel channel = openOrClosingChannelOrFail(connectionId);
389           if (closingChannels.containsKey(connectionId)) {
390               // ensure notification via `disconnected`, leave channel in the state in which closing was triggered
391               this.failedSends.add(connectionId);
392           } else {
393               try {
394                   channel.setSend(send);
395               } catch (Exception e) {
396                   // update the state for consistency, the channel will be discarded after `close`
397                   channel.state(ChannelState.FAILED_SEND);
398                   // ensure notification via `disconnected` when `failedSends` are processed in the next poll
399                   this.failedSends.add(connectionId);
400                   close(channel, CloseMode.DISCARD_NO_NOTIFY);
401                   if (!(e instanceof CancelledKeyException)) {
402                       log.error("Unexpected exception during send, closing connection {} and rethrowing exception {}",
403                               connectionId, e);
404                       throw e;
405                   }
406               }
407           }

413        *
414        * When this call is completed the user can check for completed sends, receives, connections or disconnects using
415        * {@link #completedSends()}, {@link #completedReceives()}, {@link #connected()}, {@link #disconnected()}. These
416        * lists will be cleared at the beginning of each `poll` call and repopulated by the call if there is
417        * any completed I/O.
418        *
419        * In the "Plaintext" setting, we are using socketChannel to read & write to the network. But for the "SSL" setting,
420        * we encrypt the data before we use socketChannel to write data to the network, and decrypt before we return the responses.
421        * This requires additional buffers to be maintained as we are reading from network, since the data on the wire is encrypted
422        * we won't be able to read exact no.of bytes as kafka protocol requires. We read as many bytes as we can, up to SSLEngine's
423        * application buffer size. This means we might be reading additional bytes than the requested size.
424        * If there is no further data to read from socketChannel selector won't invoke that channel and we have additional bytes
425        * in the buffer. To overcome this issue we added "keysWithBufferedRead" map which tracks channels which have data in the SSL
426        * buffers. If there are channels with buffered data that can by processed, we set "timeout" to 0 and process the data even
427        * if there is no more data to read from the socket.
428        *
429        * Atmost one entry is added to "completedReceives" for a channel in each poll. This is necessary to guarantee that
430        * requests from a channel are processed on the broker in the order they are sent. Since outstanding requests added
431        * by SocketServer to the request queue may be processed by different request handler threads, requests on each
432        * channel must be processed one-at-a-time to guarantee ordering.
433        *
434        * @param timeout The amount of time to wait, in milliseconds, which must be non-negative
435        * @throws IllegalArgumentException If `timeout` is negative
436        * @throws IllegalStateException If a send is given for which we have no existing connection or for which there is
437        *         already an in-progress send
438        */
439       @Override
440       public void poll(long timeout) throws IOException {
441           if (timeout < 0)
442               throw new IllegalArgumentException("timeout should be >= 0");
443   
444           boolean madeReadProgressLastCall = madeReadProgressLastPoll;
445           clear();
446   
447           boolean dataInBuffers = !keysWithBufferedRead.isEmpty();
448   
449           if (!immediatelyConnectedKeys.isEmpty() || (madeReadProgressLastCall && dataInBuffers))
450               timeout = 0;
451   
452           if (!memoryPool.isOutOfMemory() && outOfMemory) {
453               //we have recovered from memory pressure. unmute any channel not explicitly muted for other reasons
454               log.trace("Broker no longer low on memory - unmuting incoming sockets");
455               for (KafkaChannel channel : channels.values()) {
456                   if (channel.isInMutableState() && !explicitlyMutedChannels.contains(channel)) {
457                       channel.maybeUnmute();
458                   }
459               }
460               outOfMemory = false;
461           }
462   
463           /* check ready keys */
464           long startSelect = time.nanoseconds();
465           int numReadyKeys = select(timeout);
466           long endSelect = time.nanoseconds();
467           this.sensors.selectTime.record(endSelect - startSelect, time.milliseconds());
468   
469           if (numReadyKeys > 0 || !immediatelyConnectedKeys.isEmpty() || dataInBuffers) {
470               Set<SelectionKey> readyKeys = this.nioSelector.selectedKeys();
471   
472               // Poll from channels that have buffered data (but nothing more from the underlying socket)
473               if (dataInBuffers) {
474                   keysWithBufferedRead.removeAll(readyKeys); //so no channel gets polled twice
475                   Set<SelectionKey> toPoll = keysWithBufferedRead;
476                   keysWithBufferedRead = new HashSet<>(); //poll() calls will repopulate if needed
477                   pollSelectionKeys(toPoll, false, endSelect);
478               }
479   
480               // Poll from channels where the underlying socket has more data
481               pollSelectionKeys(readyKeys, false, endSelect);
482               // Clear all selected keys so that they are included in the ready count for the next select
483               readyKeys.clear();
484   
485               pollSelectionKeys(immediatelyConnectedKeys, true, endSelect);
486               immediatelyConnectedKeys.clear();
487           } else {
488               madeReadProgressLastPoll = true; //no work is also "progress"
489           }
490   
491           long endIo = time.nanoseconds();
492           this.sensors.ioTime.record(endIo - endSelect, time.milliseconds());
493   
494           // Close channels that were delayed and are now ready to be closed
495           completeDelayedChannelClose(endIo);
496   
497           // we use the time at the end of select to ensure that we don't close any connections that
498           // have just been processed in pollSelectionKeys
499           maybeCloseOldestConnection(endSelect);
500       }
501   
502       /**
503        * handle any ready I/O on a set of selection keys
504        * @param selectionKeys set of keys to handle
505        * @param isImmediatelyConnected true if running over a set of keys for just-connected sockets
506        * @param currentTimeNanos time at which set of keys was determined
507        */
508       // package-private for testing
509       void pollSelectionKeys(Set<SelectionKey> selectionKeys,
510                              boolean isImmediatelyConnected,
511                              long currentTimeNanos) {
512           for (SelectionKey key : determineHandlingOrder(selectionKeys)) {
513               KafkaChannel channel = channel(key);
514               long channelStartTimeNanos = recordTimePerConnection ? time.nanoseconds() : 0;
515               boolean sendFailed = false;
516               String nodeId = channel.id();
517   
518               // register all per-connection metrics at once
519               sensors.maybeRegisterConnectionMetrics(nodeId);
520               if (idleExpiryManager != null)
521                   idleExpiryManager.update(nodeId, currentTimeNanos);
522   
523               try {
524                   /* complete any connections that have finished their handshake (either normally or immediately) */
525                   if (isImmediatelyConnected || key.isConnectable()) {
526                       if (channel.finishConnect()) {
527                           this.connected.add(nodeId);
528                           this.sensors.connectionCreated.record();
529   
530                           SocketChannel socketChannel = (SocketChannel) key.channel();
531                           log.debug("Created socket with SO_RCVBUF = {}, SO_SNDBUF = {}, SO_TIMEOUT = {} to node {}",
532                                   socketChannel.socket().getReceiveBufferSize(),
533                                   socketChannel.socket().getSendBufferSize(),
534                                   socketChannel.socket().getSoTimeout(),
535                                   nodeId);
536                       } else {
537                           continue;
538                       }
539                   }
540   
541                   /* if channel is not ready finish prepare */
542                   if (channel.isConnected() && !channel.ready()) {
543                       channel.prepare();
544                       if (channel.ready()) {
545                           long readyTimeMs = time.milliseconds();
546                           boolean isReauthentication = channel.successfulAuthentications() > 1;
547                           if (isReauthentication) {
548                               sensors.successfulReauthentication.record(1.0, readyTimeMs);
549                               if (channel.reauthenticationLatencyMs() == null)
550                                   log.warn(
551                                       "Should never happen: re-authentication latency for a re-authenticated channel was null; continuing...");
552                               else
553                                   sensors.reauthenticationLatency
554                                       .record(channel.reauthenticationLatencyMs().doubleValue(), readyTimeMs);
555                           } else {
556                               sensors.successfulAuthentication.record(1.0, readyTimeMs);
557                               if (!channel.connectedClientSupportsReauthentication())
558                                   sensors.successfulAuthenticationNoReauth.record(1.0, readyTimeMs);
559                           }
560                           log.debug("Successfully {}authenticated with {}", isReauthentication ?
561                               "re-" : "", channel.socketDescription());
562                       }
563                   }
564                   if (channel.ready() && channel.state() == ChannelState.NOT_CONNECTED)
565                       channel.state(ChannelState.READY);
566                   Optional<NetworkReceive> responseReceivedDuringReauthentication = channel.pollResponseReceivedDuringReauthentication();
567                   responseReceivedDuringReauthentication.ifPresent(receive -> {
568                       long currentTimeMs = time.milliseconds();
569                       addToCompletedReceives(channel, receive, currentTimeMs);
570                   });
571   
572                   //if channel is ready and has bytes to read from socket or buffer, and has no
573                   //previous completed receive then read from it
574                   if (channel.ready() && (key.isReadable() || channel.hasBytesBuffered()) && !hasCompletedReceive(channel)
575                           && !explicitlyMutedChannels.contains(channel)) {
576                       attemptRead(channel);
577                   }
578   
579                   if (channel.hasBytesBuffered() && !explicitlyMutedChannels.contains(channel)) {
580                       //this channel has bytes enqueued in intermediary buffers that we could not read
581                       //(possibly because no memory). it may be the case that the underlying socket will
582                       //not come up in the next poll() and so we need to remember this channel for the
583                       //next poll call otherwise data may be stuck in said buffers forever. If we attempt
584                       //to process buffered data and no progress is made, the channel buffered status is
585                       //cleared to avoid the overhead of checking every time.
586                       keysWithBufferedRead.add(key);
587                   }
588   
589                   /* if channel is ready write to any sockets that have space in their buffer and for which we have data */
590   
591                   long nowNanos = channelStartTimeNanos != 0 ? channelStartTimeNanos : currentTimeNanos;
592                   try {
593                       attemptWrite(key, channel, nowNanos);
594                   } catch (Exception e) {
595                       sendFailed = true;
596                       throw e;
597                   }
598   
599                   /* cancel any defunct sockets */
600                   if (!key.isValid())
601                       close(channel, CloseMode.GRACEFUL);
602   
603               } catch (Exception e) {
604                   String desc = String.format("%s (channelId=%s)", channel.socketDescription(), channel.id());
605                   if (e instanceof IOException) {
606                       log.debug("Connection with {} disconnected", desc, e);
607                   } else if (e instanceof AuthenticationException) {
608                       boolean isReauthentication = channel.successfulAuthentications() > 0;
609                       if (isReauthentication)
610                           sensors.failedReauthentication.record();
611                       else
612                           sensors.failedAuthentication.record();
613                       String exceptionMessage = e.getMessage();
614                       if (e instanceof DelayedResponseAuthenticationException)
615                           exceptionMessage = e.getCause().getMessage();
616                       log.info("Failed {}authentication with {} ({})", isReauthentication ? "re-" : "",
617                           desc, exceptionMessage);
618                   } else {
619                       log.warn("Unexpected error from {}; closing connection", desc, e);
620                   }
621   
622                   if (e instanceof DelayedResponseAuthenticationException)
623                       maybeDelayCloseOnAuthenticationFailure(channel);
624                   else
625                       close(channel, sendFailed ? CloseMode.NOTIFY_ONLY : CloseMode.GRACEFUL);
626               } finally {
627                   maybeRecordTimePerConnection(channel, channelStartTimeNanos);
628               }
629           }
630       }

826        * a poll() when all the results from the previous poll are expected to have been handled.
827        * <p>
828        * SocketServer uses {@link #clearCompletedSends()} and {@link #clearCompletedReceives()} to
829        * clear `completedSends` and `completedReceives` as soon as they are processed to avoid

835       private void clear() {
836           this.completedSends.clear();
837           this.completedReceives.clear();
838           this.connected.clear();
839           this.disconnected.clear();
840   
841           // Remove closed channels after all their buffered receives have been processed or if a send was requested
