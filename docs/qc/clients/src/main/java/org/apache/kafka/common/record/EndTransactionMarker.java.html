29   /**
30    * This class represents the control record which is written to the log to indicate the completion
31    * of a transaction. The record key specifies the {@link ControlRecordType control type} and the
32    * value embeds information useful for write validation (for now, just the coordinator epoch).
33    */
34   public class EndTransactionMarker {
35       private static final Logger log = LoggerFactory.getLogger(EndTransactionMarker.class);
36   
37       private static final short CURRENT_END_TXN_MARKER_VERSION = 0;
38       private static final Schema END_TXN_MARKER_SCHEMA_VERSION_V0 = new Schema(
39               new Field("version", Type.INT16),
40               new Field("coordinator_epoch", Type.INT32));
41       static final int CURRENT_END_TXN_MARKER_VALUE_SIZE = 6;
42       static final int CURRENT_END_TXN_SCHEMA_RECORD_SIZE = DefaultRecord.sizeInBytes(0, 0L,
43               ControlRecordType.CURRENT_CONTROL_RECORD_KEY_SIZE,
44               EndTransactionMarker.CURRENT_END_TXN_MARKER_VALUE_SIZE,
45               Record.EMPTY_HEADERS);
46   
47       private final ControlRecordType type;
48       private final int coordinatorEpoch;
49   
50       public EndTransactionMarker(ControlRecordType type, int coordinatorEpoch) {
51           ensureTransactionMarkerControlType(type);
52           this.type = type;
53           this.coordinatorEpoch = coordinatorEpoch;
54       }
55   
56       public int coordinatorEpoch() {
57           return coordinatorEpoch;
58       }
59   
60       public ControlRecordType controlType() {
61           return type;
62       }
63   
64       private Struct buildRecordValue() {
65           Struct struct = new Struct(END_TXN_MARKER_SCHEMA_VERSION_V0);
66           struct.set("version", CURRENT_END_TXN_MARKER_VERSION);
67           struct.set("coordinator_epoch", coordinatorEpoch);
68           return struct;
69       }
70   
71       public ByteBuffer serializeValue() {
72           Struct valueStruct = buildRecordValue();
73           ByteBuffer value = ByteBuffer.allocate(valueStruct.sizeOf());
74           valueStruct.writeTo(value);
75           value.flip();
76           return value;
77       }
78   
79       @Override
80       public boolean equals(Object o) {
81           if (this == o) return true;
82           if (o == null || getClass() != o.getClass()) return false;
83   
84           EndTransactionMarker that = (EndTransactionMarker) o;
85           return coordinatorEpoch == that.coordinatorEpoch && type == that.type;
86       }
87   
88       @Override
89       public int hashCode() {
90           int result = type != null ? type.hashCode() : 0;
91           result = 31 * result + coordinatorEpoch;
92           return result;
93       }
94   
95       private static void ensureTransactionMarkerControlType(ControlRecordType type) {
96           if (type != ControlRecordType.COMMIT && type != ControlRecordType.ABORT)
97               throw new IllegalArgumentException("Invalid control record type for end transaction marker" + type);
98       }
99   
100       public static EndTransactionMarker deserialize(Record record) {
101           ControlRecordType type = ControlRecordType.parse(record.key());
102           return deserializeValue(type, record.value());
103       }
104   
105       static EndTransactionMarker deserializeValue(ControlRecordType type, ByteBuffer value) {
106           ensureTransactionMarkerControlType(type);
107   
108           if (value.remaining() < CURRENT_END_TXN_MARKER_VALUE_SIZE)
109               throw new InvalidRecordException("Invalid value size found for end transaction marker. Must have " +
110                       "at least " + CURRENT_END_TXN_MARKER_VALUE_SIZE + " bytes, but found only " + value.remaining());
111   
112           short version = value.getShort(0);
113           if (version < 0)
114               throw new InvalidRecordException("Invalid version found for end transaction marker: " + version +
115                       ". May indicate data corruption");
116   
117           if (version > CURRENT_END_TXN_MARKER_VERSION)
118               log.debug("Received end transaction marker value version {}. Parsing as version {}", version,
119                       CURRENT_END_TXN_MARKER_VERSION);
120   
121           int coordinatorEpoch = value.getInt(2);
122           return new EndTransactionMarker(type, coordinatorEpoch);
123       }
