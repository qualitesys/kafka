42   /**
43    * {@code SaslClient} implementation for SASL/OAUTHBEARER in Kafka. This
44    * implementation requires an instance of {@code AuthenticateCallbackHandler}
45    * that can handle an instance of {@link OAuthBearerTokenCallback} and return
46    * the {@link OAuthBearerToken} generated by the {@code login()} event on the
47    * {@code LoginContext}. Said handler can also optionally handle an instance of {@link SaslExtensionsCallback}
48    * to return any extensions generated by the {@code login()} event on the {@code LoginContext}.
49    *
50    * @see <a href="https://tools.ietf.org/html/rfc6750#section-2.1">RFC 6750,
51    *      Section 2.1</a>
52    *
53    */
54   public class OAuthBearerSaslClient implements SaslClient {
55       static final byte BYTE_CONTROL_A = (byte) 0x01;
56       private static final Logger log = LoggerFactory.getLogger(OAuthBearerSaslClient.class);
57       private final CallbackHandler callbackHandler;
58   
59       enum State {
60           SEND_CLIENT_FIRST_MESSAGE, RECEIVE_SERVER_FIRST_MESSAGE, RECEIVE_SERVER_MESSAGE_AFTER_FAILURE, COMPLETE, FAILED
61       }
62   
63       private State state;
64   
65       public OAuthBearerSaslClient(AuthenticateCallbackHandler callbackHandler) {
66           this.callbackHandler = Objects.requireNonNull(callbackHandler);
67           setState(State.SEND_CLIENT_FIRST_MESSAGE);
68       }
69   
70       public CallbackHandler callbackHandler() {
71           return callbackHandler;
72       }
73   
74       @Override
75       public String getMechanismName() {
76           return OAuthBearerLoginModule.OAUTHBEARER_MECHANISM;
77       }
78   
79       @Override
80       public boolean hasInitialResponse() {
81           return true;
82       }
83   
84       @Override
85       public byte[] evaluateChallenge(byte[] challenge) throws SaslException {
86           try {
87               OAuthBearerTokenCallback callback = new OAuthBearerTokenCallback();
88               switch (state) {
89                   case SEND_CLIENT_FIRST_MESSAGE:
90                       if (challenge != null && challenge.length != 0)
91                           throw new SaslException("Expected empty challenge");
92                       callbackHandler().handle(new Callback[] {callback});
93                       SaslExtensions extensions = retrieveCustomExtensions();
94   
95                       setState(State.RECEIVE_SERVER_FIRST_MESSAGE);
96   
97                       return new OAuthBearerClientInitialResponse(callback.token().value(), extensions).toBytes();
98                   case RECEIVE_SERVER_FIRST_MESSAGE:
99                       if (challenge != null && challenge.length != 0) {
100                           String jsonErrorResponse = new String(challenge, StandardCharsets.UTF_8);
101                           if (log.isDebugEnabled())
102                               log.debug("Sending %%x01 response to server after receiving an error: {}",
103                                       jsonErrorResponse);
104                           setState(State.RECEIVE_SERVER_MESSAGE_AFTER_FAILURE);
105                           return new byte[] {BYTE_CONTROL_A};
106                       }
107                       callbackHandler().handle(new Callback[] {callback});
108                       if (log.isDebugEnabled())
109                           log.debug("Successfully authenticated as {}", callback.token().principalName());
110                       setState(State.COMPLETE);
111                       return null;
112                   default:
113                       throw new IllegalSaslStateException("Unexpected challenge in Sasl client state " + state);
114               }
115           } catch (SaslException e) {
116               setState(State.FAILED);
117               throw e;
118           } catch (IOException | UnsupportedCallbackException e) {
119               setState(State.FAILED);
120               throw new SaslException(e.getMessage(), e);
121           }
122       }
123   
124       @Override
125       public boolean isComplete() {
126           return state == State.COMPLETE;
127       }
128   
129       @Override
130       public byte[] unwrap(byte[] incoming, int offset, int len) {
131           if (!isComplete())
132               throw new IllegalStateException("Authentication exchange has not completed");
133           return Arrays.copyOfRange(incoming, offset, offset + len);
134       }
135   
136       @Override
137       public byte[] wrap(byte[] outgoing, int offset, int len) {
138           if (!isComplete())
139               throw new IllegalStateException("Authentication exchange has not completed");
140           return Arrays.copyOfRange(outgoing, offset, offset + len);
141       }
142   
143       @Override
144       public Object getNegotiatedProperty(String propName) {
145           if (!isComplete())
146               throw new IllegalStateException("Authentication exchange has not completed");
147           return null;
148       }
149   
150       @Override
151       public void dispose() {
152       }
153   
154       private void setState(State state) {
155           log.debug("Setting SASL/{} client state to {}", OAuthBearerLoginModule.OAUTHBEARER_MECHANISM, state);
156           this.state = state;
157       }
158   
159       private SaslExtensions retrieveCustomExtensions() throws SaslException {
160           SaslExtensionsCallback extensionsCallback = new SaslExtensionsCallback();
161           try {
162               callbackHandler().handle(new Callback[] {extensionsCallback});
163           } catch (UnsupportedCallbackException e) {
164               log.debug("Extensions callback is not supported by client callback handler {}, no extensions will be added",
165                       callbackHandler());
166           } catch (Exception e) {
167               throw new SaslException("SASL extensions could not be obtained", e);
168           }
169   
170           return extensionsCallback.extensions();
171       }
172   
173       public static class OAuthBearerSaslClientFactory implements SaslClientFactory {
174           @Override
175           public SaslClient createSaslClient(String[] mechanisms, String authorizationId, String protocol,
176                   String serverName, Map<String, ?> props, CallbackHandler callbackHandler) {
177               String[] mechanismNamesCompatibleWithPolicy = getMechanismNames(props);
178               for (String mechanism : mechanisms) {
179                   for (int i = 0; i < mechanismNamesCompatibleWithPolicy.length; i++) {
180                       if (mechanismNamesCompatibleWithPolicy[i].equals(mechanism)) {
181                           if (!(Objects.requireNonNull(callbackHandler) instanceof AuthenticateCallbackHandler))
182                               throw new IllegalArgumentException(String.format(
183                                       "Callback handler must be castable to %s: %s",
184                                       AuthenticateCallbackHandler.class.getName(), callbackHandler.getClass().getName()));
185                           return new OAuthBearerSaslClient((AuthenticateCallbackHandler) callbackHandler);
186                       }
187                   }
188               }
189               return null;
190           }
191   
192           @Override
193           public String[] getMechanismNames(Map<String, ?> props) {
194               return OAuthBearerSaslServer.mechanismNamesCompatibleWithPolicy(props);
195           }
196       }
