17   package org.apache.kafka.common.config.provider;
18   
19   import org.apache.kafka.common.Configurable;
20   import org.apache.kafka.common.config.ConfigChangeCallback;
21   import org.apache.kafka.common.config.ConfigData;
22   
23   import java.io.Closeable;
24   import java.util.Set;
25   
26   /**
27    * A provider of configuration data, which may optionally support subscriptions to configuration changes.
28    * Implementations are required to safely support concurrent calls to any of the methods in this interface.
29    * Kafka Connect discovers configuration providers using Java's Service Provider mechanism (see {@code java.util.ServiceLoader}).
30    * To support this, implementations of this interface should also contain a service provider configuration file in {@code  META-INF/service/org.apache.kafka.common.config.provider.ConfigProvider}.
31    */
32   public interface ConfigProvider extends Configurable, Closeable {
33   
34       /**
35        * Retrieves the data at the given path.
36        *
37        * @param path the path where the data resides
38        * @return the configuration data
39        */
40       ConfigData get(String path);
41   
42       /**
43        * Retrieves the data with the given keys at the given path.
44        *
45        * @param path the path where the data resides
46        * @param keys the keys whose values will be retrieved
47        * @return the configuration data
48        */
49       ConfigData get(String path, Set<String> keys);
50   
51       /**
52        * Subscribes to changes for the given keys at the given path (optional operation).
53        *
54        * @param path the path where the data resides
55        * @param keys the keys whose values will be retrieved
56        * @param callback the callback to invoke upon change
57        * @throws {@link UnsupportedOperationException} if the subscribe operation is not supported
58        */
59       default void subscribe(String path, Set<String> keys, ConfigChangeCallback callback) {
60           throw new UnsupportedOperationException();
61       }
62   
63       /**
64        * Unsubscribes to changes for the given keys at the given path (optional operation).
65        *
66        * @param path the path where the data resides
67        * @param keys the keys whose values will be retrieved
68        * @param callback the callback to be unsubscribed from changes
69        * @throws {@link UnsupportedOperationException} if the unsubscribe operation is not supported
70        */
71       default void unsubscribe(String path, Set<String> keys, ConfigChangeCallback callback) {
72           throw new UnsupportedOperationException();
73       }
74   
75       /**
76        * Clears all subscribers (optional operation).
77        *
78        * @throws {@link UnsupportedOperationException} if the unsubscribeAll operation is not supported
79        */
80       default void unsubscribeAll() {
81           throw new UnsupportedOperationException();
82       }
83   }
