18   package org.apache.kafka.common.resource;
19   
20   import org.apache.kafka.common.annotation.InterfaceStability;
21   
22   import java.util.HashMap;
23   import java.util.Locale;
24   
25   /**
26    * Represents a type of resource which an ACL can be applied to.
27    *
28    * The API for this class is still evolving and we may break compatibility in minor releases, if necessary.
29    */
30   @InterfaceStability.Evolving
31   public enum ResourceType {
32       /**
33        * Represents any ResourceType which this client cannot understand,
34        * perhaps because this client is too old.
35        */
36       UNKNOWN((byte) 0),
37   
38       /**
39        * In a filter, matches any ResourceType.
40        */
41       ANY((byte) 1),
42   
43       /**
44        * A Kafka topic.
45        */
46       TOPIC((byte) 2),
47   
48       /**
49        * A consumer group.
50        */
51       GROUP((byte) 3),
52   
53       /**
54        * The cluster as a whole.
55        */
56       CLUSTER((byte) 4),
57   
58       /**
59        * A transactional ID.
60        */
61       TRANSACTIONAL_ID((byte) 5),
62   
63       /**
64        * A token ID.
65        */
66       DELEGATION_TOKEN((byte) 6),
67   
68       /**
69        * A user principal
70        */
71       USER((byte) 7);
72   
73       private final static HashMap<Byte, ResourceType> CODE_TO_VALUE = new HashMap<>();
74   
75       static {
76           for (ResourceType resourceType : ResourceType.values()) {
77               CODE_TO_VALUE.put(resourceType.code, resourceType);
78           }
79       }
80   
81       /**
82        * Parse the given string as an ACL resource type.
83        *
84        * @param str    The string to parse.
85        *
86        * @return       The ResourceType, or UNKNOWN if the string could not be matched.
87        */
88       public static ResourceType fromString(String str) throws IllegalArgumentException {
89           try {
90               return ResourceType.valueOf(str.toUpperCase(Locale.ROOT));
91           } catch (IllegalArgumentException e) {
92               return UNKNOWN;
93           }
94       }
95   
96       /**
97        * Return the ResourceType with the provided code or `ResourceType.UNKNOWN` if one cannot be found.
98        */
99       public static ResourceType fromCode(byte code) {
100           ResourceType resourceType = CODE_TO_VALUE.get(code);
101           if (resourceType == null) {
102               return UNKNOWN;
103           }
104           return resourceType;
105       }
106   
107       private final byte code;
108   
109       ResourceType(byte code) {
110           this.code = code;
111       }
112   
113       /**
114        * Return the code of this resource.
115        */
116       public byte code() {
117           return code;
118       }
119   
120       /**
121        * Return whether this resource type is UNKNOWN.
122        */
123       public boolean isUnknown() {
124           return this == UNKNOWN;
125       }
126   }
