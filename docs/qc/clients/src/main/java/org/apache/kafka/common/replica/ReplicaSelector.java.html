27   /**
28    * Plug-able interface for selecting a preferred read replica given the current set of replicas for a partition
29    * and metadata from the client.

32   
33       /**
34        * Select the preferred replica a client should use for fetching. If no replica is available, this will return an
35        * empty optional.
36        */
37       Optional<ReplicaView> select(TopicPartition topicPartition,
38                                    ClientMetadata clientMetadata,
39                                    PartitionView partitionView);

41       default void close() throws IOException {
42           // No-op by default
43       }
44

47           // No-op by default
48       }
49   }
