187       /**
188        * Postprocess the configuration so that exponential backoff is disabled when reconnect backoff
189        * is explicitly configured but the maximum reconnect backoff is not explicitly configured.
190        *
191        * @param config                    The config object.
192        * @param parsedValues              The parsedValues as provided to postProcessParsedConfig.
193        *
194        * @return                          The new values which have been set as described in postProcessParsedConfig.
195        */
196       public static Map<String, Object> postProcessReconnectBackoffConfigs(AbstractConfig config,
197                                                       Map<String, Object> parsedValues) {
198           HashMap<String, Object> rval = new HashMap<>();
199           Map<String, Object> originalConfig = config.originals();
200           if ((!originalConfig.containsKey(RECONNECT_BACKOFF_MAX_MS_CONFIG)) &&
201               originalConfig.containsKey(RECONNECT_BACKOFF_MS_CONFIG)) {
202               log.debug("Disabling exponential reconnect backoff because {} is set, but {} is not.",
203                       RECONNECT_BACKOFF_MS_CONFIG, RECONNECT_BACKOFF_MAX_MS_CONFIG);
204               rval.put(RECONNECT_BACKOFF_MAX_MS_CONFIG, parsedValues.get(RECONNECT_BACKOFF_MS_CONFIG));
205           }
206           return rval;
207       }

209       public static void postValidateSaslMechanismConfig(AbstractConfig config) {
210           SecurityProtocol securityProtocol = SecurityProtocol.forName(config.getString(CommonClientConfigs.SECURITY_PROTOCOL_CONFIG));
211           String clientSaslMechanism = config.getString(SaslConfigs.SASL_MECHANISM);
212           if (securityProtocol == SecurityProtocol.SASL_PLAINTEXT || securityProtocol == SecurityProtocol.SASL_SSL) {
213               if (clientSaslMechanism == null || clientSaslMechanism.isEmpty()) {
214                   throw new ConfigException(SaslConfigs.SASL_MECHANISM, null, "When the " + CommonClientConfigs.SECURITY_PROTOCOL_CONFIG +
215                           " configuration enables SASL, mechanism must be non-null and non-empty string.");
216               }
217           }
