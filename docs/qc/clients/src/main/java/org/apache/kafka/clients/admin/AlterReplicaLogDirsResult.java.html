25   import org.apache.kafka.common.errors.ClusterAuthorizationException;

32   /**
33    * The result of {@link Admin#alterReplicaLogDirs(Map, AlterReplicaLogDirsOptions)}.
34    *
35    * To retrieve the detailed result per specified {@link TopicPartitionReplica}, use {@link #values()}. To retrieve the
36    * overall result only, use {@link #all()}.
37    */
38   @InterfaceStability.Evolving
39   public class AlterReplicaLogDirsResult {
40       private final Map<TopicPartitionReplica, KafkaFuture<Void>> futures;
41   
42       AlterReplicaLogDirsResult(Map<TopicPartitionReplica, KafkaFuture<Void>> futures) {
43           this.futures = futures;
44       }
45   
46       /**
47        * Return a map from {@link TopicPartitionReplica} to {@link KafkaFuture} which holds the status of individual
48        * replica movement.
49        *
50        * To check the result of individual replica movement, call {@link KafkaFuture#get()} from the value contained
51        * in the returned map. If there is no error, it will return silently; if not, an {@link Exception} will be thrown
52        * like the following:
53        *
54        * <ul>
55        *   <li>{@link CancellationException}: The task was canceled.</li>
56        *   <li>{@link InterruptedException}: Interrupted while joining I/O thread.</li>
57        *   <li>{@link ExecutionException}: Execution failed with the following causes:</li>
58        *   <ul>
59        *     <li>{@link ClusterAuthorizationException}: Authorization failed. (CLUSTER_AUTHORIZATION_FAILED, 31)</li>
60        *     <li>{@link InvalidTopicException}: The specified topic name is too long. (INVALID_TOPIC_EXCEPTION, 17)</li>
61        *     <li>{@link LogDirNotFoundException}: The specified log directory is not found in the broker. (LOG_DIR_NOT_FOUND, 57)</li>
62        *     <li>{@link ReplicaNotAvailableException}: The replica does not exist on the broker. (REPLICA_NOT_AVAILABLE, 9)</li>
63        *     <li>{@link KafkaStorageException}: Disk error occurred. (KAFKA_STORAGE_ERROR, 56)</li>
64        *     <li>{@link UnknownServerException}: Unknown. (UNKNOWN_SERVER_ERROR, -1)</li>
65        *   </ul>
66        * </ul>
67        */
68       public Map<TopicPartitionReplica, KafkaFuture<Void>> values() {
69           return futures;
70       }
71   
72       /**
73        * Return a {@link KafkaFuture} which succeeds on {@link KafkaFuture#get()} if all the replica movement have succeeded.
74        * if not, it throws an {@link Exception} described in {@link #values()} method.
75        */
76       public KafkaFuture<Void> all() {
77           return KafkaFuture.allOf(futures.values().toArray(new KafkaFuture[0]));
78       }
79   }
