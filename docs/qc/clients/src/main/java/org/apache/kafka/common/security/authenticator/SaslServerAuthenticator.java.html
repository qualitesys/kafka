90   public class SaslServerAuthenticator implements Authenticator {
91       // GSSAPI limits requests to 64K, but we allow a bit extra for custom SASL mechanisms
92       static final int MAX_RECEIVE_SIZE = 524288;
93       private static final Logger LOG = LoggerFactory.getLogger(SaslServerAuthenticator.class);

95       /**
96        * The internal state transitions for initial authentication of a channel on the
97        * server side are declared in order, starting with {@link #INITIAL_REQUEST} and
98        * ending in either {@link #COMPLETE} or {@link #FAILED}.
99        * <p>
100        * Re-authentication of a channel on the server side starts with the state
101        * {@link #REAUTH_PROCESS_HANDSHAKE}. It may then flow to
102        * {@link #REAUTH_BAD_MECHANISM} before a transition to {@link #FAILED} if
103        * re-authentication is attempted with a mechanism different than the original
104        * one; otherwise it joins the authentication flow at the {@link #AUTHENTICATE}
105        * state and likewise ends at either {@link #COMPLETE} or {@link #FAILED}.
106        */
107       private enum SaslState {
108           INITIAL_REQUEST,               // May be GSSAPI token, SaslHandshake or ApiVersions for authentication
109           HANDSHAKE_OR_VERSIONS_REQUEST, // May be SaslHandshake or ApiVersions
110           HANDSHAKE_REQUEST,             // After an ApiVersions request, next request must be SaslHandshake
111           AUTHENTICATE,                  // Authentication tokens (SaslHandshake v1 and above indicate SaslAuthenticate headers)
112           COMPLETE,                      // Authentication completed successfully
113           FAILED,                        // Authentication failed
114           REAUTH_PROCESS_HANDSHAKE,      // Initial state for re-authentication, processes SASL handshake request
115           REAUTH_BAD_MECHANISM,          // When re-authentication requested with wrong mechanism, generate exception
116       }

194       private void createSaslServer(String mechanism) throws IOException {
195           this.saslMechanism = mechanism;

199               saslServer = createSaslKerberosServer(callbackHandler, configs, subject);

203                       Sasl.createSaslServer(saslMechanism, "kafka", serverAddress().getHostName(), configs, callbackHandler));

206                   }
207               } catch (PrivilegedActionException e) {
208                   throw new SaslException("Kafka Server failed to create a SaslServer to interact with a client during session authentication with server mechanism " + saslMechanism, e.getCause());
209               }

213       private SaslServer createSaslKerberosServer(final AuthenticateCallbackHandler saslServerCallbackHandler, final Map<String, ?> configs, Subject subject) throws IOException {
214           // server is using a JAAS-authenticated subject: determine service principal name and hostname from kafka server's subject.
215           final String servicePrincipal = SaslClientAuthenticator.firstPrincipal(subject);
216           KerberosName kerberosName;
217           try {
218               kerberosName = KerberosName.parse(servicePrincipal);
219           } catch (IllegalArgumentException e) {
220               throw new KafkaException("Principal has name with unexpected format " + servicePrincipal);
221           }
222           final String servicePrincipalName = kerberosName.serviceName();
223           final String serviceHostname = kerberosName.hostName();
224   
225           LOG.debug("Creating SaslServer for {} with mechanism {}", kerberosName, saslMechanism);
226   
227           try {
228               return Subject.doAs(subject, (PrivilegedExceptionAction<SaslServer>) () ->
229                       Sasl.createSaslServer(saslMechanism, servicePrincipalName, serviceHostname, configs, saslServerCallbackHandler));
230           } catch (PrivilegedActionException e) {
231               throw new SaslException("Kafka Server failed to create a SaslServer to interact with a client during session authentication", e.getCause());
232           }
233       }

235       /**
236        * Evaluates client responses via `SaslServer.evaluateResponse` and returns the issued challenge to the client until
237        * authentication succeeds or fails.
238        *
239        * The messages are sent and received as size delimited bytes that consists of a 4 byte network-ordered size N
240        * followed by N bytes representing the opaque payload.
241        */
242       @SuppressWarnings("fallthrough")
243       @Override
244       public void authenticate() throws IOException {
245           if (saslState != SaslState.REAUTH_PROCESS_HANDSHAKE) {
246               if (netOutBuffer != null && !flushNetOutBufferAndUpdateInterestOps())
247                   return;
248   
249               if (saslServer != null && saslServer.isComplete()) {
250                   setSaslState(SaslState.COMPLETE);
251                   return;
252               }
253   
254               // allocate on heap (as opposed to any socket server memory pool)
255               if (netInBuffer == null) netInBuffer = new NetworkReceive(MAX_RECEIVE_SIZE, connectionId);
256   
257               netInBuffer.readFrom(transportLayer);
258               if (!netInBuffer.complete())
259                   return;
260               netInBuffer.payload().rewind();
261           }
262           byte[] clientToken = new byte[netInBuffer.payload().remaining()];
263           netInBuffer.payload().get(clientToken, 0, clientToken.length);
264           netInBuffer = null; // reset the networkReceive as we read all the data.
265           try {
266               switch (saslState) {
267                   case REAUTH_PROCESS_HANDSHAKE:
268                   case HANDSHAKE_OR_VERSIONS_REQUEST:
269                   case HANDSHAKE_REQUEST:
270                       handleKafkaRequest(clientToken);
271                       break;
272                   case REAUTH_BAD_MECHANISM:
273                       throw new SaslAuthenticationException(reauthInfo.badMechanismErrorMessage);
274                   case INITIAL_REQUEST:
275                       if (handleKafkaRequest(clientToken))
276                           break;
277                       // For default GSSAPI, fall through to authenticate using the client token as the first GSSAPI packet.
278                       // This is required for interoperability with 0.9.0.x clients which do not send handshake request
279                   case AUTHENTICATE:
280                       handleSaslToken(clientToken);
281                       // When the authentication exchange is complete and no more tokens are expected from the client,
282                       // update SASL state. Current SASL state will be updated when outgoing writes to the client complete.
283                       if (saslServer.isComplete())
284                           setSaslState(SaslState.COMPLETE);
285                       break;
286                   default:
287                       break;
288               }
289           } catch (AuthenticationException e) {
290               // Exception will be propagated after response is sent to client
291               setSaslState(SaslState.FAILED, e);
292           } catch (Exception e) {
293               // In the case of IOExceptions and other unexpected exceptions, fail immediately
294               saslState = SaslState.FAILED;
295               LOG.debug("Failed during {}: {}", reauthInfo.authenticationOrReauthenticationText(), e.getMessage());
296               throw e;
297           }
298       }

336       @Override
337       public void reauthenticate(ReauthenticationContext reauthenticationContext) throws IOException {
338           NetworkReceive saslHandshakeReceive = reauthenticationContext.networkReceive();
339           if (saslHandshakeReceive == null)
340               throw new IllegalArgumentException(
341                       "Invalid saslHandshakeReceive in server-side re-authentication context: null");
342           SaslServerAuthenticator previousSaslServerAuthenticator = (SaslServerAuthenticator) reauthenticationContext.previousAuthenticator();
343           reauthInfo.reauthenticating(previousSaslServerAuthenticator.saslMechanism,
344                   previousSaslServerAuthenticator.principal(), reauthenticationContext.reauthenticationBeginNanos());

347           LOG.debug("Beginning re-authentication: {}", this);
348           netInBuffer.payload().rewind();
349           setSaslState(SaslState.REAUTH_PROCESS_HANDSHAKE);
350           authenticate();
351       }

372       private void setSaslState(SaslState saslState, AuthenticationException exception) {
373           if (netOutBuffer != null && !netOutBuffer.completed()) {
374               pendingSaslState = saslState;
375               pendingException = exception;
376           } else {
377               this.saslState = saslState;
378               LOG.debug("Set SASL server state to {} during {}", saslState, reauthInfo.authenticationOrReauthenticationText());
379               this.pendingSaslState = null;
380               this.pendingException = null;
381               if (exception != null)
382                   throw exception;
383           }
384       }

386       private boolean flushNetOutBufferAndUpdateInterestOps() throws IOException {
387           boolean flushedCompletely = flushNetOutBuffer();
388           if (flushedCompletely) {
389               transportLayer.removeInterestOps(SelectionKey.OP_WRITE);
390               if (pendingSaslState != null)
391                   setSaslState(pendingSaslState, pendingException);
392           } else
393               transportLayer.addInterestOps(SelectionKey.OP_WRITE);
394           return flushedCompletely;
395       }

411       private void handleSaslToken(byte[] clientToken) throws IOException {
412           if (!enableKafkaSaslAuthenticateHeaders) {
413               byte[] response = saslServer.evaluateResponse(clientToken);
414               if (saslServer.isComplete()) {
415                   reauthInfo.calcCompletionTimesAndReturnSessionLifetimeMs();
416                   if (reauthInfo.reauthenticating())
417                       reauthInfo.ensurePrincipalUnchanged(principal());
418               }
419               if (response != null) {
420                   netOutBuffer = ByteBufferSend.sizePrefixed(ByteBuffer.wrap(response));
421                   flushNetOutBufferAndUpdateInterestOps();
422               }
423           } else {
424               ByteBuffer requestBuffer = ByteBuffer.wrap(clientToken);
425               RequestHeader header = RequestHeader.parse(requestBuffer);
426               ApiKeys apiKey = header.apiKey();
427               short version = header.apiVersion();
428               RequestContext requestContext = new RequestContext(header, connectionId, clientAddress(),
429                       KafkaPrincipal.ANONYMOUS, listenerName, securityProtocol, ClientInformation.EMPTY, false);
430               RequestAndSize requestAndSize = requestContext.parseRequest(requestBuffer);
431               if (apiKey != ApiKeys.SASL_AUTHENTICATE) {
432                   IllegalSaslStateException e = new IllegalSaslStateException("Unexpected Kafka request of type " + apiKey + " during SASL authentication.");
433                   buildResponseOnAuthenticateFailure(requestContext, requestAndSize.request.getErrorResponse(e));
434                   throw e;
435               }
436               if (!apiKey.isVersionSupported(version)) {
437                   // We cannot create an error response if the request version of SaslAuthenticate is not supported
438                   // This should not normally occur since clients typically check supported versions using ApiVersionsRequest
439                   throw new UnsupportedVersionException("Version " + version + " is not supported for apiKey " + apiKey);
440               }
441               /*
442                * The client sends multiple SASL_AUTHENTICATE requests, and the client is known
443                * to support the required version if any one of them indicates it supports that
444                * version.
445                */
446               if (!reauthInfo.connectedClientSupportsReauthentication)
447                   reauthInfo.connectedClientSupportsReauthentication = version > 0;
448               SaslAuthenticateRequest saslAuthenticateRequest = (SaslAuthenticateRequest) requestAndSize.request;
449   
450               try {
451                   byte[] responseToken = saslServer.evaluateResponse(
452                           Utils.copyArray(saslAuthenticateRequest.data().authBytes()));
453                   if (reauthInfo.reauthenticating() && saslServer.isComplete())
454                       reauthInfo.ensurePrincipalUnchanged(principal());
455                   // For versions with SASL_AUTHENTICATE header, send a response to SASL_AUTHENTICATE request even if token is empty.
456                   byte[] responseBytes = responseToken == null ? new byte[0] : responseToken;
457                   long sessionLifetimeMs = !saslServer.isComplete() ? 0L
458                           : reauthInfo.calcCompletionTimesAndReturnSessionLifetimeMs();
459                   sendKafkaResponse(requestContext, new SaslAuthenticateResponse(
460                           new SaslAuthenticateResponseData()
461                           .setErrorCode(Errors.NONE.code())
462                           .setAuthBytes(responseBytes)
463                           .setSessionLifetimeMs(sessionLifetimeMs)));
464               } catch (SaslAuthenticationException e) {
465                   buildResponseOnAuthenticateFailure(requestContext,
466                           new SaslAuthenticateResponse(
467                                   new SaslAuthenticateResponseData()
468                                   .setErrorCode(Errors.SASL_AUTHENTICATION_FAILED.code())
469                                   .setErrorMessage(e.getMessage())));
470                   throw e;
471               } catch (SaslException e) {
472                   KerberosError kerberosError = KerberosError.fromException(e);
473                   if (kerberosError != null && kerberosError.retriable()) {
474                       // Handle retriable Kerberos exceptions as I/O exceptions rather than authentication exceptions
475                       throw e;
476                   } else {
477                       // DO NOT include error message from the `SaslException` in the client response since it may
478                       // contain sensitive data like the existence of the user.
479                       String errorMessage = "Authentication failed during "
480                               + reauthInfo.authenticationOrReauthenticationText()
481                               + " due to invalid credentials with SASL mechanism " + saslMechanism;
482                       buildResponseOnAuthenticateFailure(requestContext, new SaslAuthenticateResponse(
483                               new SaslAuthenticateResponseData()
484                               .setErrorCode(Errors.SASL_AUTHENTICATION_FAILED.code())
485                               .setErrorMessage(errorMessage)));
486                       throw new SaslAuthenticationException(errorMessage, e);
487                   }
488               }
489           }
490       }

492       private boolean handleKafkaRequest(byte[] requestBytes) throws IOException, AuthenticationException {
493           boolean isKafkaRequest = false;
494           String clientMechanism = null;
495           try {
496               ByteBuffer requestBuffer = ByteBuffer.wrap(requestBytes);
497               RequestHeader header = RequestHeader.parse(requestBuffer);
498               ApiKeys apiKey = header.apiKey();
499   
500               // A valid Kafka request header was received. SASL authentication tokens are now expected only
501               // following a SaslHandshakeRequest since this is not a GSSAPI client token from a Kafka 0.9.0.x client.
502               if (saslState == SaslState.INITIAL_REQUEST)
503                   setSaslState(SaslState.HANDSHAKE_OR_VERSIONS_REQUEST);
504               isKafkaRequest = true;
505   
506               // Raise an error prior to parsing if the api cannot be handled at this layer. This avoids
507               // unnecessary exposure to some of the more complex schema types.
508               if (apiKey != ApiKeys.API_VERSIONS && apiKey != ApiKeys.SASL_HANDSHAKE)
509                   throw new IllegalSaslStateException("Unexpected Kafka request of type " + apiKey + " during SASL handshake.");
510   
511               LOG.debug("Handling Kafka request {} during {}", apiKey, reauthInfo.authenticationOrReauthenticationText());
512   
513   
514               RequestContext requestContext = new RequestContext(header, connectionId, clientAddress(),
515                       KafkaPrincipal.ANONYMOUS, listenerName, securityProtocol, ClientInformation.EMPTY, false);
516               RequestAndSize requestAndSize = requestContext.parseRequest(requestBuffer);
517               if (apiKey == ApiKeys.API_VERSIONS)
518                   handleApiVersionsRequest(requestContext, (ApiVersionsRequest) requestAndSize.request);
519               else
520                   clientMechanism = handleHandshakeRequest(requestContext, (SaslHandshakeRequest) requestAndSize.request);
521           } catch (InvalidRequestException e) {
522               if (saslState == SaslState.INITIAL_REQUEST) {
523                   // InvalidRequestException is thrown if the request is not in Kafka format or if the API key
524                   // is invalid. For compatibility with 0.9.0.x where the first packet is a GSSAPI token
525                   // starting with 0x60, revert to GSSAPI for both these exceptions.
526                   if (LOG.isDebugEnabled()) {
527                       StringBuilder tokenBuilder = new StringBuilder();
528                       for (byte b : requestBytes) {
529                           tokenBuilder.append(String.format("%02x", b));
530                           if (tokenBuilder.length() >= 20)
531                                break;
532                       }
533                       LOG.debug("Received client packet of length {} starting with bytes 0x{}, process as GSSAPI packet", requestBytes.length, tokenBuilder);
534                   }
535                   if (enabledMechanisms.contains(SaslConfigs.GSSAPI_MECHANISM)) {
536                       LOG.debug("First client packet is not a SASL mechanism request, using default mechanism GSSAPI");
537                       clientMechanism = SaslConfigs.GSSAPI_MECHANISM;
538                   } else
539                       throw new UnsupportedSaslMechanismException("Exception handling first SASL packet from client, GSSAPI is not supported by server", e);
540               } else
541                   throw e;
542           }
543           if (clientMechanism != null && (!reauthInfo.reauthenticating()
544                   || reauthInfo.saslMechanismUnchanged(clientMechanism))) {
545               createSaslServer(clientMechanism);
546               setSaslState(SaslState.AUTHENTICATE);
547           }
548           return isKafkaRequest;
549       }

551       private String handleHandshakeRequest(RequestContext context, SaslHandshakeRequest handshakeRequest) throws IOException, UnsupportedSaslMechanismException {
552           String clientMechanism = handshakeRequest.data().mechanism();
553           short version = context.header.apiVersion();
554           if (version >= 1)
555               this.enableKafkaSaslAuthenticateHeaders(true);
556           if (enabledMechanisms.contains(clientMechanism)) {
557               LOG.debug("Using SASL mechanism '{}' provided by client", clientMechanism);
558               sendKafkaResponse(context, new SaslHandshakeResponse(
559                       new SaslHandshakeResponseData().setErrorCode(Errors.NONE.code()).setMechanisms(enabledMechanisms)));
560               return clientMechanism;
561           } else {
562               LOG.debug("SASL mechanism '{}' requested by client is not supported", clientMechanism);
563               buildResponseOnAuthenticateFailure(context, new SaslHandshakeResponse(
564                       new SaslHandshakeResponseData().setErrorCode(Errors.UNSUPPORTED_SASL_MECHANISM.code()).setMechanisms(enabledMechanisms)));
565               throw new UnsupportedSaslMechanismException("Unsupported SASL mechanism " + clientMechanism);
566           }
567       }

669           private long calcCompletionTimesAndReturnSessionLifetimeMs() {
670               long retvalSessionLifetimeMs = 0L;
671               long authenticationEndMs = time.milliseconds();
672               authenticationEndNanos = time.nanoseconds();
673               Long credentialExpirationMs = (Long) saslServer
674                       .getNegotiatedProperty(SaslInternalConfigs.CREDENTIAL_LIFETIME_MS_SASL_NEGOTIATED_PROPERTY_KEY);
675               Long connectionsMaxReauthMs = connectionsMaxReauthMsByMechanism.get(saslMechanism);
676               boolean maxReauthSet = connectionsMaxReauthMs != null && connectionsMaxReauthMs > 0;
677   
678               if (credentialExpirationMs != null || maxReauthSet) {
679                   if (credentialExpirationMs == null)
680                       retvalSessionLifetimeMs = zeroIfNegative(connectionsMaxReauthMs);
681                   else if (!maxReauthSet)
682                       retvalSessionLifetimeMs = zeroIfNegative(credentialExpirationMs - authenticationEndMs);
683                   else
684                       retvalSessionLifetimeMs = zeroIfNegative(Math.min(credentialExpirationMs - authenticationEndMs, connectionsMaxReauthMs));
685   
686                   sessionExpirationTimeNanos = authenticationEndNanos + 1000 * 1000 * retvalSessionLifetimeMs;
687               }
688   
689               if (credentialExpirationMs != null) {
690                   LOG.debug(
691                           "Authentication complete; session max lifetime from broker config={} ms, credential expiration={} ({} ms); session expiration = {} ({} ms), sending {} ms to client",
692                           connectionsMaxReauthMs, new Date(credentialExpirationMs),
693                           credentialExpirationMs - authenticationEndMs,
694                           new Date(authenticationEndMs + retvalSessionLifetimeMs), retvalSessionLifetimeMs,
695                           retvalSessionLifetimeMs);
696               } else {
697                   if (sessionExpirationTimeNanos != null)
698                       LOG.debug(
699                               "Authentication complete; session max lifetime from broker config={} ms, no credential expiration; session expiration = {} ({} ms), sending {} ms to client",
700                               connectionsMaxReauthMs, new Date(authenticationEndMs + retvalSessionLifetimeMs),
701                               retvalSessionLifetimeMs, retvalSessionLifetimeMs);
702                   else
703                       LOG.debug(
704                               "Authentication complete; session max lifetime from broker config={} ms, no credential expiration; no session expiration, sending 0 ms to client",
705                               connectionsMaxReauthMs);
706               }
707               return retvalSessionLifetimeMs;
708           }

710           public Long reauthenticationLatencyMs() {
711               if (!reauthenticating())
712                   return null;
713               // record at least 1 ms if there is some latency
714               long latencyNanos = authenticationEndNanos - reauthenticationBeginNanos;
715               return latencyNanos == 0L ? 0L : Math.max(1L, Math.round(latencyNanos / 1000.0 / 1000.0));
716           }
