35   /**
36    * ConsumerProtocol contains the schemas for consumer subscriptions and assignments for use with
37    * Kafka's generalized group management protocol.
38    *
39    * The current implementation assumes that future versions will not break compatibility. When
40    * it encounters a newer version, it parses it using the current format. This basically means
41    * that new versions cannot remove or reorder any of the existing fields.
42    */
43   public class ConsumerProtocol {
44       public static final String PROTOCOL_TYPE = "consumer";
45   
46       static {
47           // Safety check to ensure that both parts of the consumer protocol remain in sync.
48           if (ConsumerProtocolSubscription.LOWEST_SUPPORTED_VERSION
49                   != ConsumerProtocolAssignment.LOWEST_SUPPORTED_VERSION)
50               throw new IllegalStateException("Subscription and Assignment schemas must have the " +
51                   "same lowest version");
52   
53           if (ConsumerProtocolSubscription.HIGHEST_SUPPORTED_VERSION
54                   != ConsumerProtocolAssignment.HIGHEST_SUPPORTED_VERSION)
55               throw new IllegalStateException("Subscription and Assignment schemas must have the " +
56                   "same highest version");
57       }
58   
59       public static short deserializeVersion(final ByteBuffer buffer) {
60           try {
61               return buffer.getShort();
62           } catch (BufferUnderflowException e) {
63               throw new SchemaException("Buffer underflow while parsing consumer protocol's header", e);
64           }
65       }
66   
67       public static ByteBuffer serializeSubscription(final Subscription subscription) {
68           return serializeSubscription(subscription, ConsumerProtocolSubscription.HIGHEST_SUPPORTED_VERSION);
69       }
70   
71       public static ByteBuffer serializeSubscription(final Subscription subscription, short version) {
72           version = checkSubscriptionVersion(version);
73   
74           ConsumerProtocolSubscription data = new ConsumerProtocolSubscription();
75   
76           List<String> topics = new ArrayList<>(subscription.topics());
77           Collections.sort(topics);
78           data.setTopics(topics);
79   
80           data.setUserData(subscription.userData() != null ? subscription.userData().duplicate() : null);
81   
82           List<TopicPartition> ownedPartitions = new ArrayList<>(subscription.ownedPartitions());
83           ownedPartitions.sort(Comparator.comparing(TopicPartition::topic).thenComparing(TopicPartition::partition));
84           ConsumerProtocolSubscription.TopicPartition partition = null;
85           for (TopicPartition tp : ownedPartitions) {
86               if (partition == null || !partition.topic().equals(tp.topic())) {
87                   partition = new ConsumerProtocolSubscription.TopicPartition().setTopic(tp.topic());
88                   data.ownedPartitions().add(partition);
89               }
90               partition.partitions().add(tp.partition());
91           }
92   
93           return MessageUtil.toVersionPrefixedByteBuffer(version, data);
94       }
95   
96       public static Subscription deserializeSubscription(final ByteBuffer buffer, short version) {
97           version = checkSubscriptionVersion(version);
98   
99           try {
100               ConsumerProtocolSubscription data =
101                   new ConsumerProtocolSubscription(new ByteBufferAccessor(buffer), version);
102   
103               List<TopicPartition> ownedPartitions = new ArrayList<>();
104               for (ConsumerProtocolSubscription.TopicPartition tp : data.ownedPartitions()) {
105                   for (Integer partition : tp.partitions()) {
106                       ownedPartitions.add(new TopicPartition(tp.topic(), partition));
107                   }
108               }
109   
110               return new Subscription(
111                   data.topics(),
112                   data.userData() != null ? data.userData().duplicate() : null,
113                   ownedPartitions);
114           } catch (BufferUnderflowException e) {
115               throw new SchemaException("Buffer underflow while parsing consumer protocol's subscription", e);
116           }
117       }
118   
119       public static Subscription deserializeSubscription(final ByteBuffer buffer) {
120           return deserializeSubscription(buffer, deserializeVersion(buffer));
121       }
122   
123       public static ByteBuffer serializeAssignment(final Assignment assignment) {
124           return serializeAssignment(assignment, ConsumerProtocolAssignment.HIGHEST_SUPPORTED_VERSION);
125       }
126   
127       public static ByteBuffer serializeAssignment(final Assignment assignment, short version) {
128           version = checkAssignmentVersion(version);
129   
130           ConsumerProtocolAssignment data = new ConsumerProtocolAssignment();
131           data.setUserData(assignment.userData() != null ? assignment.userData().duplicate() : null);
132           assignment.partitions().forEach(tp -> {
133               ConsumerProtocolAssignment.TopicPartition partition = data.assignedPartitions().find(tp.topic());
134               if (partition == null) {
135                   partition = new ConsumerProtocolAssignment.TopicPartition().setTopic(tp.topic());
136                   data.assignedPartitions().add(partition);
137               }
138               partition.partitions().add(tp.partition());
139           });
140           return MessageUtil.toVersionPrefixedByteBuffer(version, data);
141       }
142   
143       public static Assignment deserializeAssignment(final ByteBuffer buffer, short version) {
144           version = checkAssignmentVersion(version);
145   
146           try {
147               ConsumerProtocolAssignment data =
148                   new ConsumerProtocolAssignment(new ByteBufferAccessor(buffer), version);
149   
150               List<TopicPartition> assignedPartitions = new ArrayList<>();
151               for (ConsumerProtocolAssignment.TopicPartition tp : data.assignedPartitions()) {
152                   for (Integer partition : tp.partitions()) {
153                       assignedPartitions.add(new TopicPartition(tp.topic(), partition));
154                   }
155               }
156   
157               return new Assignment(
158                   assignedPartitions,
159                   data.userData() != null ? data.userData().duplicate() : null);
160           } catch (BufferUnderflowException e) {
161               throw new SchemaException("Buffer underflow while parsing consumer protocol's assignment", e);
162           }
163       }
164   
165       public static Assignment deserializeAssignment(final ByteBuffer buffer) {
166           return deserializeAssignment(buffer, deserializeVersion(buffer));
167       }
168   
169       private static short checkSubscriptionVersion(final short version) {
170           if (version < ConsumerProtocolSubscription.LOWEST_SUPPORTED_VERSION)
171               throw new SchemaException("Unsupported subscription version: " + version);
172           else if (version > ConsumerProtocolSubscription.HIGHEST_SUPPORTED_VERSION)
173               return ConsumerProtocolSubscription.HIGHEST_SUPPORTED_VERSION;
174           else
175               return version;
176       }
177   
178       private static short checkAssignmentVersion(final short version) {
179           if (version < ConsumerProtocolAssignment.LOWEST_SUPPORTED_VERSION)
180               throw new SchemaException("Unsupported assignment version: " + version);
181           else if (version > ConsumerProtocolAssignment.HIGHEST_SUPPORTED_VERSION)
182               return ConsumerProtocolAssignment.HIGHEST_SUPPORTED_VERSION;
183           else
184               return version;
185       }
