20   import static org.apache.kafka.common.config.SaslConfigs.SASL_LOGIN_CONNECT_TIMEOUT_MS;
21   import static org.apache.kafka.common.config.SaslConfigs.SASL_LOGIN_READ_TIMEOUT_MS;
22   import static org.apache.kafka.common.config.SaslConfigs.SASL_LOGIN_RETRY_BACKOFF_MAX_MS;
23   import static org.apache.kafka.common.config.SaslConfigs.SASL_LOGIN_RETRY_BACKOFF_MS;
24   import static org.apache.kafka.common.config.SaslConfigs.SASL_OAUTHBEARER_TOKEN_ENDPOINT_URL;
25   import static org.apache.kafka.common.security.oauthbearer.secured.OAuthBearerLoginCallbackHandler.CLIENT_ID_CONFIG;
26   import static org.apache.kafka.common.security.oauthbearer.secured.OAuthBearerLoginCallbackHandler.CLIENT_SECRET_CONFIG;
27   import static org.apache.kafka.common.security.oauthbearer.secured.OAuthBearerLoginCallbackHandler.SCOPE_CONFIG;
28   
29   import java.net.URL;
30   import java.util.Locale;
31   import java.util.Map;
32   import javax.net.ssl.SSLSocketFactory;
33   
34   public class AccessTokenRetrieverFactory  {
35   
36       /**
37        * Create an {@link AccessTokenRetriever} from the given SASL and JAAS configuration.
38        *
39        * <b>Note</b>: the returned <code>AccessTokenRetriever</code> is <em>not</em> initialized
40        * here and must be done by the caller prior to use.
41        *
42        * @param configs    SASL configuration
43        * @param jaasConfig JAAS configuration
44        *
45        * @return Non-<code>null</code> {@link AccessTokenRetriever}
46        */
47   
48       public static AccessTokenRetriever create(Map<String, ?> configs, Map<String, Object> jaasConfig) {
49           return create(configs, null, jaasConfig);
50       }
51   
52       public static AccessTokenRetriever create(Map<String, ?> configs,
53           String saslMechanism,
54           Map<String, Object> jaasConfig) {
55           ConfigurationUtils cu = new ConfigurationUtils(configs, saslMechanism);
56           URL tokenEndpointUrl = cu.validateUrl(SASL_OAUTHBEARER_TOKEN_ENDPOINT_URL);
57   
58           if (tokenEndpointUrl.getProtocol().toLowerCase(Locale.ROOT).equals("file")) {
59               return new FileTokenRetriever(cu.validateFile(SASL_OAUTHBEARER_TOKEN_ENDPOINT_URL));
60           } else {
61               JaasOptionsUtils jou = new JaasOptionsUtils(jaasConfig);
62               String clientId = jou.validateString(CLIENT_ID_CONFIG);
63               String clientSecret = jou.validateString(CLIENT_SECRET_CONFIG);
64               String scope = jou.validateString(SCOPE_CONFIG, false);
65   
66               SSLSocketFactory sslSocketFactory = null;
67   
68               if (jou.shouldCreateSSLSocketFactory(tokenEndpointUrl))
69                   sslSocketFactory = jou.createSSLSocketFactory();
70   
71               return new HttpAccessTokenRetriever(clientId,
72                   clientSecret,
73                   scope,
74                   sslSocketFactory,
75                   tokenEndpointUrl.toString(),
76                   cu.validateLong(SASL_LOGIN_RETRY_BACKOFF_MS),
77                   cu.validateLong(SASL_LOGIN_RETRY_BACKOFF_MAX_MS),
78                   cu.validateInteger(SASL_LOGIN_CONNECT_TIMEOUT_MS, false),
79                   cu.validateInteger(SASL_LOGIN_READ_TIMEOUT_MS, false));
80           }
81       }
