17   package org.apache.kafka.common.utils;
18   
19   import org.slf4j.Logger;
20   import org.slf4j.LoggerFactory;
21   
22   import java.lang.reflect.Constructor;
23   import java.lang.reflect.InvocationHandler;
24   import java.lang.reflect.Method;
25   import java.lang.reflect.Proxy;
26   import java.util.Arrays;
27   import java.util.List;
28   import java.util.Map;
29   import java.util.concurrent.ConcurrentHashMap;
30   
31   public class LoggingSignalHandler {
32   
33       private static final Logger log = LoggerFactory.getLogger(LoggingSignalHandler.class);
34   
35       private static final List<String> SIGNALS = Arrays.asList("TERM", "INT", "HUP");
36   
37       private final Constructor<?> signalConstructor;
38       private final Class<?> signalHandlerClass;
39       private final Class<?> signalClass;
40       private final Method signalHandleMethod;
41       private final Method signalGetNameMethod;
42       private final Method signalHandlerHandleMethod;
43   
44       /**
45        * Create an instance of this class.
46        *
47        * @throws ReflectiveOperationException if the underlying API has changed in an incompatible manner.
48        */
49       public LoggingSignalHandler() throws ReflectiveOperationException {
50           signalClass = Class.forName("sun.misc.Signal");
51           signalConstructor = signalClass.getConstructor(String.class);
52           signalHandlerClass = Class.forName("sun.misc.SignalHandler");
53           signalHandlerHandleMethod = signalHandlerClass.getMethod("handle", signalClass);
54           signalHandleMethod = signalClass.getMethod("handle", signalClass, signalHandlerClass);
55           signalGetNameMethod = signalClass.getMethod("getName");
56       }
57   
58       /**
59        * Register signal handler to log termination due to SIGTERM, SIGHUP and SIGINT (control-c). This method
60        * does not currently work on Windows.
61        *
62        * @implNote sun.misc.Signal and sun.misc.SignalHandler are described as "not encapsulated" in
63        * http://openjdk.java.net/jeps/260. However, they are not available in the compile classpath if the `--release`
64        * flag is used. As a workaround, we rely on reflection.
65        */
66       public void register() throws ReflectiveOperationException {
67           Map<String, Object> jvmSignalHandlers = new ConcurrentHashMap<>();
68   
69           for (String signal : SIGNALS) {
70               register(signal, jvmSignalHandlers);
71           }
72           log.info("Registered signal handlers for " + String.join(", ", SIGNALS));
73       }
74   
75       private Object createSignalHandler(final Map<String, Object> jvmSignalHandlers) {
76           InvocationHandler invocationHandler = new InvocationHandler() {
77   
78               private String getName(Object signal) throws ReflectiveOperationException {
79                   return (String) signalGetNameMethod.invoke(signal);
80               }
81   
82               private void handle(Object signalHandler, Object signal) throws ReflectiveOperationException {
83                   signalHandlerHandleMethod.invoke(signalHandler, signal);
84               }
85   
86               @Override
87               public Object invoke(Object proxy, Method method, Object[] args) throws Throwable {
88                   Object signal = args[0];
89                   log.info("Terminating process due to signal {}", signal);
90                   Object handler = jvmSignalHandlers.get(getName(signal));
91                   if (handler != null)
92                       handle(handler, signal);
93                   return null;
94               }
95           };
96           return Proxy.newProxyInstance(Utils.getContextOrKafkaClassLoader(), new Class[] {signalHandlerClass},
97                   invocationHandler);
98       }
99   
100       private void register(String signalName, final Map<String, Object> jvmSignalHandlers) throws ReflectiveOperationException {
101           Object signal = signalConstructor.newInstance(signalName);
102           Object signalHandler = createSignalHandler(jvmSignalHandlers);
103           Object oldHandler = signalHandleMethod.invoke(null, signal, signalHandler);
104           if (oldHandler != null)
105               jvmSignalHandlers.put(signalName, oldHandler);
106       }
107   }
