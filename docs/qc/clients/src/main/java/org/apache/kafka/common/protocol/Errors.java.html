469        * If there are multiple matches in the class hierarchy, the first match starting from the bottom is used.
470        */
471       public static Errors forException(Throwable t) {
472           Class<?> clazz = t.getClass();
473           while (clazz != null) {
474               Errors error = classToError.get(clazz);
475               if (error != null)
476                   return error;
477               clazz = clazz.getSuperclass();
478           }
479           return UNKNOWN_SERVER_ERROR;
480       }
