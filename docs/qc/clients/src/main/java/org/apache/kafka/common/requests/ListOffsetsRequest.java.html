98       private ListOffsetsRequest(ListOffsetsRequestData data, short version) {
99           super(ApiKeys.LIST_OFFSETS, version);
100           this.data = data;
101           duplicatePartitions = new HashSet<>();
102           Set<TopicPartition> partitions = new HashSet<>();
103           for (ListOffsetsTopic topic : data.topics()) {
104               for (ListOffsetsPartition partition : topic.partitions()) {
105                   TopicPartition tp = new TopicPartition(topic.name(), partition.partitionIndex());
106                   if (!partitions.add(tp)) {
107                       duplicatePartitions.add(tp);
108                   }
109               }
110           }
111       }

113       @Override
114       public AbstractResponse getErrorResponse(int throttleTimeMs, Throwable e) {
115           short versionId = version();
116           short errorCode = Errors.forException(e).code();
117   
118           List<ListOffsetsTopicResponse> responses = new ArrayList<>();
119           for (ListOffsetsTopic topic : data.topics()) {
120               ListOffsetsTopicResponse topicResponse = new ListOffsetsTopicResponse().setName(topic.name());
121               List<ListOffsetsPartitionResponse> partitions = new ArrayList<>();
122               for (ListOffsetsPartition partition : topic.partitions()) {
123                   ListOffsetsPartitionResponse partitionResponse = new ListOffsetsPartitionResponse()
124                           .setErrorCode(errorCode)
125                           .setPartitionIndex(partition.partitionIndex());
126                   if (versionId == 0) {
127                       partitionResponse.setOldStyleOffsets(Collections.emptyList());
128                   } else {
129                       partitionResponse.setOffset(ListOffsetsResponse.UNKNOWN_OFFSET)
130                                        .setTimestamp(ListOffsetsResponse.UNKNOWN_TIMESTAMP);
131                   }
132                   partitions.add(partitionResponse);
133               }
134               topicResponse.setPartitions(partitions);
135               responses.add(topicResponse);
136           }
137           ListOffsetsResponseData responseData = new ListOffsetsResponseData()
138                   .setThrottleTimeMs(throttleTimeMs)
139                   .setTopics(responses);
140           return new ListOffsetsResponse(responseData);
141       }

168       public static List<ListOffsetsTopic> toListOffsetsTopics(Map<TopicPartition, ListOffsetsPartition> timestampsToSearch) {
169           Map<String, ListOffsetsTopic> topics = new HashMap<>();
170           for (Map.Entry<TopicPartition, ListOffsetsPartition> entry : timestampsToSearch.entrySet()) {
171               TopicPartition tp = entry.getKey();
172               ListOffsetsTopic topic = topics.computeIfAbsent(tp.topic(), k -> new ListOffsetsTopic().setName(tp.topic()));
173               topic.partitions().add(entry.getValue());
174           }
175           return new ArrayList<>(topics.values());
176       }
