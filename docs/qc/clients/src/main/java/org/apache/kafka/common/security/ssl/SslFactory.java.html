145       private SslEngineFactory createNewSslEngineFactory(Map<String, ?> newConfigs) {
146           if (sslEngineFactory == null) {
147               throw new IllegalStateException("SslFactory has not been configured.");
148           }
149           Map<String, Object> nextConfigs = new HashMap<>(sslEngineFactoryConfig);
150           copyMapEntries(nextConfigs, newConfigs, reconfigurableConfigs());
151           if (clientAuthConfigOverride != null) {
152               nextConfigs.put(BrokerSecurityConfigs.SSL_CLIENT_AUTH_CONFIG, clientAuthConfigOverride);
153           }
154           if (!sslEngineFactory.shouldBeRebuilt(nextConfigs)) {
155               return sslEngineFactory;
156           }
157           try {
158               SslEngineFactory newSslEngineFactory = instantiateSslEngineFactory(nextConfigs);
159               if (sslEngineFactory.keystore() == null) {
160                   if (newSslEngineFactory.keystore() != null) {
161                       throw new ConfigException("Cannot add SSL keystore to an existing listener for " +
162                               "which no keystore was configured.");
163                   }
164               } else {
165                   if (newSslEngineFactory.keystore() == null) {
166                       throw new ConfigException("Cannot remove the SSL keystore from an existing listener for " +
167                               "which a keystore was configured.");
168                   }
169   
170                   CertificateEntries.ensureCompatible(newSslEngineFactory.keystore(), sslEngineFactory.keystore());
171               }
172               if (sslEngineFactory.truststore() == null && newSslEngineFactory.truststore() != null) {
173                   throw new ConfigException("Cannot add SSL truststore to an existing listener for which no " +
174                           "truststore was configured.");
175               }
176               if (keystoreVerifiableUsingTruststore) {
177                   if (sslEngineFactory.truststore() != null || sslEngineFactory.keystore() != null) {
178                       SslEngineValidator.validate(sslEngineFactory, newSslEngineFactory);
179                   }
180               }
181               return newSslEngineFactory;
182           } catch (Exception e) {
183               log.debug("Validation of dynamic config update of SSLFactory failed.", e);
184               throw new ConfigException("Validation of dynamic config update of SSLFactory failed: " + e);
185           }
186       }

207       /**
208        * Returns host/IP address of remote host without reverse DNS lookup to be used as the host
209        * for creating SSL engine. This is used as a hint for session reuse strategy and also for
210        * hostname verification of server hostnames.
211        * <p>
212        * Scenarios:
213        * <ul>
214        *   <li>Server-side
215        *   <ul>
216        *     <li>Server accepts connection from a client. Server knows only client IP
217        *     address. We want to avoid reverse DNS lookup of the client IP address since the server
218        *     does not verify or use client hostname. The IP address can be used directly.</li>
219        *   </ul>
220        *   </li>
221        *   <li>Client-side
222        *   <ul>
223        *     <li>Client connects to server using hostname. No lookup is necessary
224        *     and the hostname should be used to create the SSL engine. This hostname is validated
225        *     against the hostname in SubjectAltName (dns) or CommonName in the certificate if
226        *     hostname verification is enabled. Authentication fails if hostname does not match.</li>
227        *     <li>Client connects to server using IP address, but certificate contains only
228        *     SubjectAltName (dns). Use of reverse DNS lookup to determine hostname introduces
229        *     a security vulnerability since authentication would be reliant on a secure DNS.
230        *     Hence hostname verification should fail in this case.</li>
231        *     <li>Client connects to server using IP address and certificate contains
232        *     SubjectAltName (ipaddress). This could be used when Kafka is on a private network.
233        *     If reverse DNS lookup is used, authentication would succeed using IP address if lookup
234        *     fails and IP address is used, but authentication would fail if lookup succeeds and
235        *     dns name is used. For consistency and to avoid dependency on a potentially insecure
236        *     DNS, reverse DNS lookup should be avoided and the IP address specified by the client for
237        *     connection should be used to create the SSL engine.</li>
238        *   </ul></li>
239        * </ul>
240        */
241       private String peerHost(Socket socket) {
242           return new InetSocketAddress(socket.getInetAddress(), 0).getHostString();
243       }

288       static class CertificateEntries {
289           private final String alias;
290           private final Principal subjectPrincipal;
291           private final Set<List<?>> subjectAltNames;
292   
293           static List<CertificateEntries> create(KeyStore keystore) throws GeneralSecurityException {
294               Enumeration<String> aliases = keystore.aliases();
295               List<CertificateEntries> entries = new ArrayList<>();
296               while (aliases.hasMoreElements()) {
297                   String alias = aliases.nextElement();
298                   Certificate cert  = keystore.getCertificate(alias);
299                   if (cert instanceof X509Certificate)
300                       entries.add(new CertificateEntries(alias, (X509Certificate) cert));
301               }
302               return entries;
303           }
304   
305           static void ensureCompatible(KeyStore newKeystore, KeyStore oldKeystore) throws GeneralSecurityException {
306               List<CertificateEntries> newEntries = CertificateEntries.create(newKeystore);
307               List<CertificateEntries> oldEntries = CertificateEntries.create(oldKeystore);
308               if (newEntries.size() != oldEntries.size()) {
309                   throw new ConfigException(String.format("Keystore entries do not match, existing store contains %d entries, new store contains %d entries",
310                       oldEntries.size(), newEntries.size()));
311               }
312               for (int i = 0; i < newEntries.size(); i++) {
313                   CertificateEntries newEntry = newEntries.get(i);
314                   CertificateEntries oldEntry = oldEntries.get(i);
315                   if (!Objects.equals(newEntry.subjectPrincipal, oldEntry.subjectPrincipal)) {
316                       throw new ConfigException(String.format("Keystore DistinguishedName does not match: " +
317                           " existing={alias=%s, DN=%s}, new={alias=%s, DN=%s}",
318                           oldEntry.alias, oldEntry.subjectPrincipal, newEntry.alias, newEntry.subjectPrincipal));
319                   }
320                   if (!newEntry.subjectAltNames.containsAll(oldEntry.subjectAltNames)) {
321                       throw new ConfigException(String.format("Keystore SubjectAltNames do not match: " +
322                               " existing={alias=%s, SAN=%s}, new={alias=%s, SAN=%s}",
323                           oldEntry.alias, oldEntry.subjectAltNames, newEntry.alias, newEntry.subjectAltNames));
324                   }
325               }
326           }
327   
328           CertificateEntries(String alias, X509Certificate cert) throws GeneralSecurityException {
329               this.alias = alias;
330               this.subjectPrincipal = cert.getSubjectX500Principal();
331               Collection<List<?>> altNames = cert.getSubjectAlternativeNames();
332               // use a set for comparison
333               this.subjectAltNames = altNames != null ? new HashSet<>(altNames) : Collections.emptySet();
334           }

336           @Override
337           public int hashCode() {

339           }

357       /**
358        * Validator used to verify dynamic update of keystore used in inter-broker communication.
359        * The validator checks that a successful handshake can be performed using the keystore and
360        * truststore configured on this SslFactory.
361        */
362       private static class SslEngineValidator {
363           private static final ByteBuffer EMPTY_BUF = ByteBuffer.allocate(0);
364           private final SSLEngine sslEngine;
365           private SSLEngineResult handshakeResult;
366           private ByteBuffer appBuffer;
367           private ByteBuffer netBuffer;
368   
369           static void validate(SslEngineFactory oldEngineBuilder,
370                                SslEngineFactory newEngineBuilder) throws SSLException {
371               validate(createSslEngineForValidation(oldEngineBuilder, Mode.SERVER),
372                       createSslEngineForValidation(newEngineBuilder, Mode.CLIENT));
373               validate(createSslEngineForValidation(newEngineBuilder, Mode.SERVER),
374                       createSslEngineForValidation(oldEngineBuilder, Mode.CLIENT));
375           }
376   
377           private static SSLEngine createSslEngineForValidation(SslEngineFactory sslEngineFactory, Mode mode) {
378               // Use empty hostname, disable hostname verification
379               if (mode == Mode.SERVER) {
380                   return sslEngineFactory.createServerSslEngine("", 0);
381               } else {
382                   return sslEngineFactory.createClientSslEngine("", 0, "");
383               }
384           }
385   
386           static void validate(SSLEngine clientEngine, SSLEngine serverEngine) throws SSLException {
387               SslEngineValidator clientValidator = new SslEngineValidator(clientEngine);
388               SslEngineValidator serverValidator = new SslEngineValidator(serverEngine);
389               try {
390                   clientValidator.beginHandshake();
391                   serverValidator.beginHandshake();
392                   while (!serverValidator.complete() || !clientValidator.complete()) {
393                       clientValidator.handshake(serverValidator);
394                       serverValidator.handshake(clientValidator);
395                   }
396               } finally {
397                   clientValidator.close();
398                   serverValidator.close();
399               }
400           }
401   
402           private SslEngineValidator(SSLEngine engine) {
403               this.sslEngine = engine;
404               appBuffer = ByteBuffer.allocate(sslEngine.getSession().getApplicationBufferSize());
405               netBuffer = ByteBuffer.allocate(sslEngine.getSession().getPacketBufferSize());
406           }
407   
408           void beginHandshake() throws SSLException {
409               sslEngine.beginHandshake();
410           }
411           void handshake(SslEngineValidator peerValidator) throws SSLException {
412               SSLEngineResult.HandshakeStatus handshakeStatus = sslEngine.getHandshakeStatus();
413               while (true) {
414                   switch (handshakeStatus) {
415                       case NEED_WRAP:
416                           if (netBuffer.position() != 0) // Wait for peer to consume previously wrapped data
417                               return;
418                           handshakeResult = sslEngine.wrap(EMPTY_BUF, netBuffer);
419                           switch (handshakeResult.getStatus()) {
420                               case OK: break;
421                               case BUFFER_OVERFLOW:
422                                   netBuffer.compact();
423                                   netBuffer = Utils.ensureCapacity(netBuffer, sslEngine.getSession().getPacketBufferSize());
424                                   netBuffer.flip();
425                                   break;
426                               case BUFFER_UNDERFLOW:
427                               case CLOSED:
428                               default:
429                                   throw new SSLException("Unexpected handshake status: " + handshakeResult.getStatus());
430                           }
431                           return;
432                       case NEED_UNWRAP:
433                           if (peerValidator.netBuffer.position() == 0) // no data to unwrap, return to process peer
434                               return;
435                           peerValidator.netBuffer.flip(); // unwrap the data from peer
436                           handshakeResult = sslEngine.unwrap(peerValidator.netBuffer, appBuffer);
437                           peerValidator.netBuffer.compact();
438                           handshakeStatus = handshakeResult.getHandshakeStatus();
439                           switch (handshakeResult.getStatus()) {
440                               case OK: break;
441                               case BUFFER_OVERFLOW:
442                                   appBuffer = Utils.ensureCapacity(appBuffer, sslEngine.getSession().getApplicationBufferSize());
443                                   break;
444                               case BUFFER_UNDERFLOW:
445                                   netBuffer = Utils.ensureCapacity(netBuffer, sslEngine.getSession().getPacketBufferSize());
446                                   break;
447                               case CLOSED:
448                               default:
449                                   throw new SSLException("Unexpected handshake status: " + handshakeResult.getStatus());
450                           }
451                           break;
452                       case NEED_TASK:
453                           sslEngine.getDelegatedTask().run();
454                           handshakeStatus = sslEngine.getHandshakeStatus();
455                           break;
456                       case FINISHED:
457                           return;
458                       case NOT_HANDSHAKING:
459                           if (handshakeResult.getHandshakeStatus() != SSLEngineResult.HandshakeStatus.FINISHED)
460                               throw new SSLException("Did not finish handshake");
461                           return;
462                       default:
463                           throw new IllegalStateException("Unexpected handshake status " + handshakeStatus);
464                   }
465               }
466           }
467   
468           boolean complete() {
469               return sslEngine.getHandshakeStatus() == SSLEngineResult.HandshakeStatus.FINISHED ||
470                       sslEngine.getHandshakeStatus() == SSLEngineResult.HandshakeStatus.NOT_HANDSHAKING;
471           }
472   
473           void close() {
474               sslEngine.closeOutbound();
475               try {
476                   sslEngine.closeInbound();
477               } catch (Exception e) {
478                   // ignore
479               }
480           }
481       }
482   }
