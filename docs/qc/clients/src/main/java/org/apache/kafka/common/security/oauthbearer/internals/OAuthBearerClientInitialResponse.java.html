17   package org.apache.kafka.common.security.oauthbearer.internals;
18   
19   import org.apache.kafka.common.security.auth.SaslExtensions;
20   import org.apache.kafka.common.utils.Utils;
21   
22   import javax.security.sasl.SaslException;
23   import java.nio.charset.StandardCharsets;
24   import java.util.Map;
25   import java.util.Objects;
26   import java.util.regex.Matcher;
27   import java.util.regex.Pattern;
28   
29   public class OAuthBearerClientInitialResponse {
30       static final String SEPARATOR = "\u0001";
31   
32       private static final String SASLNAME = "(?:[\\x01-\\x7F&&[^=,]]|=2C|=3D)+";
33       private static final String KEY = "[A-Za-z]+";
34       private static final String VALUE = "[\\x21-\\x7E \t\r\n]+";
35   
36       private static final String KVPAIRS = String.format("(%s=%s%s)*", KEY, VALUE, SEPARATOR);
37       private static final Pattern AUTH_PATTERN = Pattern.compile("(?<scheme>[\\w]+)[ ]+(?<token>[-_\\.a-zA-Z0-9]+)");
38       private static final Pattern CLIENT_INITIAL_RESPONSE_PATTERN = Pattern.compile(
39               String.format("n,(a=(?<authzid>%s))?,%s(?<kvpairs>%s)%s", SASLNAME, SEPARATOR, KVPAIRS, SEPARATOR));
40       public static final String AUTH_KEY = "auth";
41   
42       private final String tokenValue;
43       private final String authorizationId;
44       private SaslExtensions saslExtensions;
45   
46       public static final Pattern EXTENSION_KEY_PATTERN = Pattern.compile(KEY);
47       public static final Pattern EXTENSION_VALUE_PATTERN = Pattern.compile(VALUE);
48   
49       public OAuthBearerClientInitialResponse(byte[] response) throws SaslException {
50           String responseMsg = new String(response, StandardCharsets.UTF_8);
51           Matcher matcher = CLIENT_INITIAL_RESPONSE_PATTERN.matcher(responseMsg);
52           if (!matcher.matches())
53               throw new SaslException("Invalid OAUTHBEARER client first message");
54           String authzid = matcher.group("authzid");
55           this.authorizationId = authzid == null ? "" : authzid;
56           String kvPairs = matcher.group("kvpairs");
57           Map<String, String> properties = Utils.parseMap(kvPairs, "=", SEPARATOR);
58           String auth = properties.get(AUTH_KEY);
59           if (auth == null)
60               throw new SaslException("Invalid OAUTHBEARER client first message: 'auth' not specified");
61           properties.remove(AUTH_KEY);
62           SaslExtensions extensions = new SaslExtensions(properties);
63           validateExtensions(extensions);
64           this.saslExtensions = extensions;
65   
66           Matcher authMatcher = AUTH_PATTERN.matcher(auth);
67           if (!authMatcher.matches())
68               throw new SaslException("Invalid OAUTHBEARER client first message: invalid 'auth' format");
69           if (!"bearer".equalsIgnoreCase(authMatcher.group("scheme"))) {
70               String msg = String.format("Invalid scheme in OAUTHBEARER client first message: %s",
71                       matcher.group("scheme"));
72               throw new SaslException(msg);
73           }
74           this.tokenValue = authMatcher.group("token");
75       }
76   
77       /**
78        * Constructor
79        * 
80        * @param tokenValue
81        *            the mandatory token value
82        * @param extensions
83        *            the optional extensions
84        * @throws SaslException
85        *             if any extension name or value fails to conform to the required
86        *             regular expression as defined by the specification, or if the
87        *             reserved {@code auth} appears as a key
88        */
89       public OAuthBearerClientInitialResponse(String tokenValue, SaslExtensions extensions) throws SaslException {
90           this(tokenValue, "", extensions);
91       }
92   
93       /**
94        * Constructor
95        * 
96        * @param tokenValue
97        *            the mandatory token value
98        * @param authorizationId
99        *            the optional authorization ID
100        * @param extensions
101        *            the optional extensions
102        * @throws SaslException
103        *             if any extension name or value fails to conform to the required
104        *             regular expression as defined by the specification, or if the
105        *             reserved {@code auth} appears as a key
106        */
107       public OAuthBearerClientInitialResponse(String tokenValue, String authorizationId, SaslExtensions extensions) throws SaslException {
108           this.tokenValue = Objects.requireNonNull(tokenValue, "token value must not be null");
109           this.authorizationId = authorizationId == null ? "" : authorizationId;
110           validateExtensions(extensions);
111           this.saslExtensions = extensions != null ? extensions : SaslExtensions.NO_SASL_EXTENSIONS;
112       }
113   
114       /**
115        * Return the always non-null extensions
116        * 
117        * @return the always non-null extensions
118        */
119       public SaslExtensions extensions() {
120           return saslExtensions;
121       }
122   
123       public byte[] toBytes() {
124           String authzid = authorizationId.isEmpty() ? "" : "a=" + authorizationId;
125           String extensions = extensionsMessage();
126           if (extensions.length() > 0)
127               extensions = SEPARATOR + extensions;
128   
129           String message = String.format("n,%s,%sauth=Bearer %s%s%s%s", authzid,
130                   SEPARATOR, tokenValue, extensions, SEPARATOR, SEPARATOR);
131   
132           return message.getBytes(StandardCharsets.UTF_8);
133       }
134   
135       /**
136        * Return the always non-null token value
137        * 
138        * @return the always non-null toklen value
139        */
140       public String tokenValue() {
141           return tokenValue;
142       }
143   
144       /**
145        * Return the always non-null authorization ID
146        * 
147        * @return the always non-null authorization ID
148        */
149       public String authorizationId() {
150           return authorizationId;
151       }
152   
153       /**
154        * Validates that the given extensions conform to the standard. They should also not contain the reserve key name {@link OAuthBearerClientInitialResponse#AUTH_KEY}
155        *
156        * @param extensions
157        *            optional extensions to validate
158        * @throws SaslException
159        *             if any extension name or value fails to conform to the required
160        *             regular expression as defined by the specification, or if the
161        *             reserved {@code auth} appears as a key
162        *
163        * @see <a href="https://tools.ietf.org/html/rfc7628#section-3.1">RFC 7628,
164        *  Section 3.1</a>
165        */
166       public static void validateExtensions(SaslExtensions extensions) throws SaslException {
167           if (extensions == null)
168               return;
169           if (extensions.map().containsKey(OAuthBearerClientInitialResponse.AUTH_KEY))
170               throw new SaslException("Extension name " + OAuthBearerClientInitialResponse.AUTH_KEY + " is invalid");
171   
172           for (Map.Entry<String, String> entry : extensions.map().entrySet()) {
173               String extensionName = entry.getKey();
174               String extensionValue = entry.getValue();
175   
176               if (!EXTENSION_KEY_PATTERN.matcher(extensionName).matches())
177                   throw new SaslException("Extension name " + extensionName + " is invalid");
178               if (!EXTENSION_VALUE_PATTERN.matcher(extensionValue).matches())
179                   throw new SaslException("Extension value (" + extensionValue + ") for extension " + extensionName + " is invalid");
180           }
181       }
182   
183       /**
184        * Converts the SASLExtensions to an OAuth protocol-friendly string
185        */
186       private String extensionsMessage() {
187           return Utils.mkString(saslExtensions.map(), "", "", "=", SEPARATOR);
188       }
189   }
