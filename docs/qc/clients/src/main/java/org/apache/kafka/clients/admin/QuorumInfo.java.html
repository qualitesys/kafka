23   /**
24    * This class is used to describe the state of the quorum received in DescribeQuorumResponse.
25    */
26   public class QuorumInfo {
27       private final Integer leaderId;
28       private final List<ReplicaState> voters;
29       private final List<ReplicaState> observers;
30   
31       QuorumInfo(Integer leaderId, List<ReplicaState> voters, List<ReplicaState> observers) {
32           this.leaderId = leaderId;
33           this.voters = voters;
34           this.observers = observers;
35       }
36   
37       public Integer leaderId() {
38           return leaderId;
39       }
40   
41       public List<ReplicaState> voters() {
42           return voters;
43       }
44   
45       public List<ReplicaState> observers() {
46           return observers;
47       }
48   
49       @Override
50       public boolean equals(Object o) {
51           if (this == o) return true;
52           if (o == null || getClass() != o.getClass()) return false;
53           QuorumInfo that = (QuorumInfo) o;
54           return leaderId.equals(that.leaderId)
55               && voters.equals(that.voters)
56               && observers.equals(that.observers);
57       }
58   
59       @Override
60       public int hashCode() {
61           return Objects.hash(leaderId, voters, observers);
62       }
63   
64       @Override
65       public String toString() {
66           return "QuorumInfo(" +
67               "leaderId=" + leaderId +
68               ", voters=" + voters +
69               ", observers=" + observers +
70               ')';
71       }

73       public static class ReplicaState {
74           private final int replicaId;
75           private final long logEndOffset;
76           private final OptionalLong lastFetchTimeMs;
77           private final OptionalLong lastCaughtUpTimeMs;
78   
79           ReplicaState() {
80               this(0, 0, OptionalLong.empty(), OptionalLong.empty());
81           }
82   
83           ReplicaState(
84               int replicaId,
85               long logEndOffset,
86               OptionalLong lastFetchTimeMs,
87               OptionalLong lastCaughtUpTimeMs
88           ) {
89               this.replicaId = replicaId;
90               this.logEndOffset = logEndOffset;
91               this.lastFetchTimeMs = lastFetchTimeMs;
92               this.lastCaughtUpTimeMs = lastCaughtUpTimeMs;
93           }
94   
95           /**
96            * Return the ID for this replica.
97            * @return The ID for this replica
98            */
99           public int replicaId() {
100               return replicaId;
101           }
102   
103           /**
104            * Return the logEndOffset known by the leader for this replica.
105            * @return The logEndOffset for this replica
106            */
107           public long logEndOffset() {
108               return logEndOffset;
109           }
110   
111           /**
112            * Return the lastFetchTime in milliseconds for this replica.
113            * @return The value of the lastFetchTime if known, empty otherwise
114            */
115           public OptionalLong lastFetchTimeMs() {
116               return lastFetchTimeMs;
117           }
118   
119           /**
120            * Return the lastCaughtUpTime in milliseconds for this replica.
121            * @return The value of the lastCaughtUpTime if known, empty otherwise
122            */
123           public OptionalLong lastCaughtUpTimeMs() {
124               return lastCaughtUpTimeMs;
125           }
126   
127           @Override
128           public boolean equals(Object o) {
129               if (this == o) return true;
130               if (o == null || getClass() != o.getClass()) return false;
131               ReplicaState that = (ReplicaState) o;
132               return replicaId == that.replicaId

134                   && lastFetchTimeMs.equals(that.lastFetchTimeMs)
135                   && lastCaughtUpTimeMs.equals(that.lastCaughtUpTimeMs);
136           }
137   
138           @Override
139           public int hashCode() {
140               return Objects.hash(replicaId, logEndOffset, lastFetchTimeMs, lastCaughtUpTimeMs);
141           }
142   
143           @Override

145               return "ReplicaState(" +

148                   ", lastFetchTimeMs=" + lastFetchTimeMs +
149                   ", lastCaughtUpTimeMs=" + lastCaughtUpTimeMs +
150                   ')';
151           }
