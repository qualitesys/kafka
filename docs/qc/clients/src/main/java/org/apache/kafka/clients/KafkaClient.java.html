17   package org.apache.kafka.clients;
18   
19   import org.apache.kafka.common.Node;
20   import org.apache.kafka.common.errors.AuthenticationException;
21   import org.apache.kafka.common.requests.AbstractRequest;
22   
23   import java.io.Closeable;
24   import java.util.List;
25   
26   /**
27    * The interface for {@link NetworkClient}
28    */
29   public interface KafkaClient extends Closeable {
30   
31       /**
32        * Check if we are currently ready to send another request to the given node but don't attempt to connect if we
33        * aren't.
34        *
35        * @param node The node to check
36        * @param now The current timestamp
37        */
38       boolean isReady(Node node, long now);
39   
40       /**
41        * Initiate a connection to the given node (if necessary), and return true if already connected. The readiness of a
42        * node will change only when poll is invoked.
43        *
44        * @param node The node to connect to.
45        * @param now The current time
46        * @return true iff we are ready to immediately initiate the sending of another request to the given node.
47        */
48       boolean ready(Node node, long now);
49   
50       /**
51        * Return the number of milliseconds to wait, based on the connection state, before attempting to send data. When
52        * disconnected, this respects the reconnect backoff time. When connecting or connected, this handles slow/stalled
53        * connections.
54        *
55        * @param node The node to check
56        * @param now The current timestamp
57        * @return The number of milliseconds to wait.
58        */
59       long connectionDelay(Node node, long now);
60   
61       /**
62        * Return the number of milliseconds to wait, based on the connection state and the throttle time, before
63        * attempting to send data. If the connection has been established but being throttled, return throttle delay.
64        * Otherwise, return connection delay.
65        *
66        * @param node the connection to check
67        * @param now the current time in ms
68        */
69       long pollDelayMs(Node node, long now);
70   
71       /**
72        * Check if the connection of the node has failed, based on the connection state. Such connection failure are
73        * usually transient and can be resumed in the next {@link #ready(org.apache.kafka.common.Node, long)} }
74        * call, but there are cases where transient failures needs to be caught and re-acted upon.
75        *
76        * @param node the node to check
77        * @return true iff the connection has failed and the node is disconnected
78        */
79       boolean connectionFailed(Node node);
80   
81       /**
82        * Check if authentication to this node has failed, based on the connection state. Authentication failures are
83        * propagated without any retries.
84        *
85        * @param node the node to check
86        * @return an AuthenticationException iff authentication has failed, null otherwise
87        */
88       AuthenticationException authenticationException(Node node);
89   
90       /**
91        * Queue up the given request for sending. Requests can only be sent on ready connections.
92        * @param request The request
93        * @param now The current timestamp
94        */
95       void send(ClientRequest request, long now);
96   
97       /**
98        * Do actual reads and writes from sockets.
99        *
100        * @param timeout The maximum amount of time to wait for responses in ms, must be non-negative. The implementation
101        *                is free to use a lower value if appropriate (common reasons for this are a lower request or
102        *                metadata update timeout)
103        * @param now The current time in ms
104        * @throws IllegalStateException If a request is sent to an unready node
105        */
106       List<ClientResponse> poll(long timeout, long now);
107   
108       /**
109        * Disconnects the connection to a particular node, if there is one.
110        * Any pending ClientRequests for this connection will receive disconnections.
111        *
112        * @param nodeId The id of the node
113        */
114       void disconnect(String nodeId);
115   
116       /**
117        * Closes the connection to a particular node (if there is one).
118        * All requests on the connection will be cleared.  ClientRequest callbacks will not be invoked
119        * for the cleared requests, nor will they be returned from poll().
120        *
121        * @param nodeId The id of the node
122        */
123       void close(String nodeId);
124   
125       /**
126        * Choose the node with the fewest outstanding requests. This method will prefer a node with an existing connection,
127        * but will potentially choose a node for which we don't yet have a connection if all existing connections are in
128        * use.
129        *
130        * @param now The current time in ms
131        * @return The node with the fewest in-flight requests.
132        */
133       Node leastLoadedNode(long now);
134   
135       /**
136        * The number of currently in-flight requests for which we have not yet returned a response
137        */
138       int inFlightRequestCount();
139   
140       /**
141        * Return true if there is at least one in-flight request and false otherwise.
142        */
143       boolean hasInFlightRequests();
144   
145       /**
146        * Get the total in-flight requests for a particular node
147        *
148        * @param nodeId The id of the node
149        */
150       int inFlightRequestCount(String nodeId);
151   
152       /**
153        * Return true if there is at least one in-flight request for a particular node and false otherwise.
154        */
155       boolean hasInFlightRequests(String nodeId);
156   
157       /**
158        * Return true if there is at least one node with connection in the READY state and not throttled. Returns false
159        * otherwise.
160        *
161        * @param now the current time
162        */
163       boolean hasReadyNodes(long now);
164   
165       /**
166        * Wake up the client if it is currently blocked waiting for I/O
167        */
168       void wakeup();
169   
170       /**
171        * Create a new ClientRequest.
172        *
173        * @param nodeId the node to send to
174        * @param requestBuilder the request builder to use
175        * @param createdTimeMs the time in milliseconds to use as the creation time of the request
176        * @param expectResponse true iff we expect a response
177        */
178       ClientRequest newClientRequest(String nodeId, AbstractRequest.Builder<?> requestBuilder,
179                                      long createdTimeMs, boolean expectResponse);
180   
181       /**
182        * Create a new ClientRequest.
183        *
184        * @param nodeId the node to send to
185        * @param requestBuilder the request builder to use
186        * @param createdTimeMs the time in milliseconds to use as the creation time of the request
187        * @param expectResponse true iff we expect a response
188        * @param requestTimeoutMs Upper bound time in milliseconds to await a response before disconnecting the socket and
189        *                         cancelling the request. The request may get cancelled sooner if the socket disconnects
190        *                         for any reason including if another pending request to the same node timed out first.
191        * @param callback the callback to invoke when we get a response
192        */
193       ClientRequest newClientRequest(String nodeId,
194                                      AbstractRequest.Builder<?> requestBuilder,
195                                      long createdTimeMs,
196                                      boolean expectResponse,
197                                      int requestTimeoutMs,
198                                      RequestCompletionHandler callback);
199   
200   
201   
202       /**
203        * Initiates shutdown of this client. This method may be invoked from another thread while this
204        * client is being polled. No further requests may be sent using the client. The current poll()
205        * will be terminated using wakeup(). The client should be explicitly shutdown using {@link #close()}
206        * after poll returns. Note that {@link #close()} should not be invoked concurrently while polling.
207        */
208       void initiateClose();
209   
210       /**
211        * Returns true if the client is still active. Returns false if {@link #initiateClose()} or {@link #close()}
212        * was invoked for this client.
213        */
214       boolean active();
215   
216   }
