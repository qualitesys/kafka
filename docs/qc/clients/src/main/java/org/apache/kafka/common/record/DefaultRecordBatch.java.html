44   /**
45    * RecordBatch implementation for magic 2 and above. The schema is given below:
46    *
47    * RecordBatch =>
48    *  BaseOffset => Int64
49    *  Length => Int32
50    *  PartitionLeaderEpoch => Int32
51    *  Magic => Int8
52    *  CRC => Uint32
53    *  Attributes => Int16
54    *  LastOffsetDelta => Int32 // also serves as LastSequenceDelta
55    *  BaseTimestamp => Int64
56    *  MaxTimestamp => Int64
57    *  ProducerId => Int64
58    *  ProducerEpoch => Int16
59    *  BaseSequence => Int32
60    *  Records => [Record]
61    *
62    * Note that when compression is enabled (see attributes below), the compressed record data is serialized
63    * directly following the count of the number of records.
64    *
65    * The CRC covers the data from the attributes to the end of the batch (i.e. all the bytes that follow the CRC). It is
66    * located after the magic byte, which means that clients must parse the magic byte before deciding how to interpret
67    * the bytes between the batch length and the magic byte. The partition leader epoch field is not included in the CRC
68    * computation to avoid the need to recompute the CRC when this field is assigned for every batch that is received by
69    * the broker. The CRC-32C (Castagnoli) polynomial is used for the computation.
70    *
71    * On Compaction: Unlike the older message formats, magic v2 and above preserves the first and last offset/sequence
72    * numbers from the original batch when the log is cleaned. This is required in order to be able to restore the
73    * producer's state when the log is reloaded. If we did not retain the last sequence number, then following
74    * a partition leader failure, once the new leader has rebuilt the producer state from the log, the next sequence
75    * expected number would no longer be in sync with what was written by the client. This would cause an
76    * unexpected OutOfOrderSequence error, which is typically fatal. The base sequence number must be preserved for
77    * duplicate checking: the broker checks incoming Produce requests for duplicates by verifying that the first and
78    * last sequence numbers of the incoming batch match the last from that producer.
79    *
80    * Note that if all of the records in a batch are removed during compaction, the broker may still retain an empty
81    * batch header in order to preserve the producer sequence information as described above. These empty batches
82    * are retained only until either a new sequence number is written by the corresponding producer or the producerId
83    * is expired from lack of activity.
84    *
85    * There is no similar need to preserve the timestamp from the original batch after compaction. The BaseTimestamp
86    * field therefore reflects the timestamp of the first record in the batch in most cases. If the batch is empty, the
87    * BaseTimestamp will be set to -1 (NO_TIMESTAMP). If the delete horizon flag is set to 1, the BaseTimestamp
88    * will be set to the time at which tombstone records and aborted transaction markers in the batch should be removed.
89    *
90    * Similarly, the MaxTimestamp field reflects the maximum timestamp of the current records if the timestamp type
91    * is CREATE_TIME. For LOG_APPEND_TIME, on the other hand, the MaxTimestamp field reflects the timestamp set
92    * by the broker and is preserved after compaction. Additionally, the MaxTimestamp of an empty batch always retains
93    * the previous value prior to becoming empty.
94    *
95    * The current attributes are given below:
96    *
97    *  ---------------------------------------------------------------------------------------------------------------------------
98    *  | Unused (7-15) | Delete Horizon Flag (6) | Control (5) | Transactional (4) | Timestamp Type (3) | Compression Type (0-2) |
99    *  ---------------------------------------------------------------------------------------------------------------------------
100    */

279       private CloseableIterator<Record> compressedIterator(BufferSupplier bufferSupplier, boolean skipKeyValue) {
280           final DataInputStream inputStream = recordInputStream(bufferSupplier);
281   
282           if (skipKeyValue) {
283               // this buffer is used to skip length delimited fields like key, value, headers
284               byte[] skipArray = new byte[MAX_SKIP_BUFFER_SIZE];
285   
286               return new StreamRecordIterator(inputStream) {
287                   @Override
288                   protected Record doReadRecord(long baseOffset, long baseTimestamp, int baseSequence, Long logAppendTime) throws IOException {
289                       return DefaultRecord.readPartiallyFrom(inputStream, skipArray, baseOffset, baseTimestamp, baseSequence, logAppendTime);
290                   }
291               };
292           } else {
293               return new StreamRecordIterator(inputStream) {
294                   @Override
295                   protected Record doReadRecord(long baseOffset, long baseTimestamp, int baseSequence, Long logAppendTime) throws IOException {
296                       return DefaultRecord.readFrom(inputStream, baseOffset, baseTimestamp, baseSequence, logAppendTime);
297                   }
298               };
299           }
300       }

302       private CloseableIterator<Record> uncompressedIterator() {
303           final ByteBuffer buffer = this.buffer.duplicate();
304           buffer.position(RECORDS_OFFSET);
305           return new RecordIterator() {
306               @Override
307               protected Record readNext(long baseOffset, long baseTimestamp, int baseSequence, Long logAppendTime) {
308                   try {
309                       return DefaultRecord.readFrom(buffer, baseOffset, baseTimestamp, baseSequence, logAppendTime);
310                   } catch (BufferUnderflowException e) {
311                       throw new InvalidRecordException("Incorrect declared batch size, premature EOF reached");
312                   }
313               }
314               @Override
315               protected boolean ensureNoneRemaining() {
316                   return !buffer.hasRemaining();
317               }
318               @Override
319               public void close() {}
320           };
321       }

323       @Override
324       public Iterator<Record> iterator() {
325           if (count() == 0)
326               return Collections.emptyIterator();
327   
328           if (!isCompressed())
329               return uncompressedIterator();
330   
331           // for a normal iterator, we cannot ensure that the underlying compression stream is closed,
332           // so we decompress the full record set here. Use cases which call for a lower memory footprint
333           // can use `streamingIterator` at the cost of additional complexity
334           try (CloseableIterator<Record> iterator = compressedIterator(BufferSupplier.NO_CACHING, false)) {
335               List<Record> records = new ArrayList<>(count());
336               while (iterator.hasNext())
337                   records.add(iterator.next());
338               return records.iterator();
339           }
340       }

342       @Override
343       public CloseableIterator<Record> skipKeyValueIterator(BufferSupplier bufferSupplier) {
344           if (count() == 0) {
345               return CloseableIterator.wrap(Collections.emptyIterator());
346           }
347   
348           /*
349            * For uncompressed iterator, it is actually not worth skipping key / value / headers at all since
350            * its ByteBufferInputStream's skip() function is less efficient compared with just reading it actually
351            * as it will allocate new byte array.
352            */
353           if (!isCompressed())
354               return uncompressedIterator();
355   
356           // we define this to be a closable iterator so that caller (i.e. the log validator) needs to close it
357           // while we can save memory footprint of not decompressing the full record set ahead of time
358           return compressedIterator(bufferSupplier, true);
359       }

374       @Override

376           long currentMaxTimestamp = maxTimestamp();
377           // We don't need to recompute crc if the timestamp is not updated.
378           if (timestampType() == timestampType && currentMaxTimestamp == maxTimestamp)
379               return;

383           buffer.putLong(MAX_TIMESTAMP_OFFSET, maxTimestamp);
384           long crc = computeChecksum();
385           ByteUtils.writeUnsignedInt(buffer, CRC_OFFSET, crc);
386       }

427       private static byte computeAttributes(CompressionType type, TimestampType timestampType,
428                                             boolean isTransactional, boolean isControl, boolean isDeleteHorizonSet) {
429           if (timestampType == TimestampType.NO_TIMESTAMP_TYPE)
430               throw new IllegalArgumentException("Timestamp type must be provided to compute attributes for message " +
431                       "format v2 and above");
432   
433           byte attributes = isTransactional ? TRANSACTIONAL_FLAG_MASK : 0;
434           if (isControl)
435               attributes |= CONTROL_FLAG_MASK;
436           if (type.id > 0)
437               attributes |= COMPRESSION_CODEC_MASK & type.id;
438           if (timestampType == TimestampType.LOG_APPEND_TIME)
439               attributes |= TIMESTAMP_TYPE_MASK;
440           if (isDeleteHorizonSet)
441               attributes |= DELETE_HORIZON_FLAG_MASK;
442           return attributes;
443       }

463       public static void writeHeader(ByteBuffer buffer,
464                                      long baseOffset,
465                                      int lastOffsetDelta,
466                                      int sizeInBytes,
467                                      byte magic,
468                                      CompressionType compressionType,
469                                      TimestampType timestampType,
470                                      long baseTimestamp,
471                                      long maxTimestamp,
472                                      long producerId,
473                                      short epoch,
474                                      int sequence,
475                                      boolean isTransactional,
476                                      boolean isControlBatch,
477                                      boolean isDeleteHorizonSet,
478                                      int partitionLeaderEpoch,
479                                      int numRecords) {
480           if (magic < RecordBatch.CURRENT_MAGIC_VALUE)
481               throw new IllegalArgumentException("Invalid magic value " + magic);
482           if (baseTimestamp < 0 && baseTimestamp != NO_TIMESTAMP)
483               throw new IllegalArgumentException("Invalid message timestamp " + baseTimestamp);
484   
485           short attributes = computeAttributes(compressionType, timestampType, isTransactional, isControlBatch, isDeleteHorizonSet);
486   
487           int position = buffer.position();
488           buffer.putLong(position + BASE_OFFSET_OFFSET, baseOffset);
489           buffer.putInt(position + LENGTH_OFFSET, sizeInBytes - LOG_OVERHEAD);
490           buffer.putInt(position + PARTITION_LEADER_EPOCH_OFFSET, partitionLeaderEpoch);
491           buffer.put(position + MAGIC_OFFSET, magic);
492           buffer.putShort(position + ATTRIBUTES_OFFSET, attributes);
493           buffer.putLong(position + BASE_TIMESTAMP_OFFSET, baseTimestamp);
494           buffer.putLong(position + MAX_TIMESTAMP_OFFSET, maxTimestamp);
495           buffer.putInt(position + LAST_OFFSET_DELTA_OFFSET, lastOffsetDelta);
496           buffer.putLong(position + PRODUCER_ID_OFFSET, producerId);
497           buffer.putShort(position + PRODUCER_EPOCH_OFFSET, epoch);
498           buffer.putInt(position + BASE_SEQUENCE_OFFSET, sequence);
499           buffer.putInt(position + RECORDS_COUNT_OFFSET, numRecords);
500           long crc = Crc32C.compute(buffer, ATTRIBUTES_OFFSET, sizeInBytes - ATTRIBUTES_OFFSET);
501           buffer.putInt(position + CRC_OFFSET, (int) crc);
502           buffer.position(position + RECORD_BATCH_OVERHEAD);
503       }

513       public static int sizeInBytes(long baseOffset, Iterable<Record> records) {
514           Iterator<Record> iterator = records.iterator();
515           if (!iterator.hasNext())
516               return 0;
517   
518           int size = RECORD_BATCH_OVERHEAD;
519           Long baseTimestamp = null;
520           while (iterator.hasNext()) {
521               Record record = iterator.next();
522               int offsetDelta = (int) (record.offset() - baseOffset);
523               if (baseTimestamp == null)
524                   baseTimestamp = record.timestamp();
525               long timestampDelta = record.timestamp() - baseTimestamp;
526               size += DefaultRecord.sizeInBytes(offsetDelta, timestampDelta, record.key(), record.value(),
527                       record.headers());
528           }
529           return size;
530       }

572       private abstract class RecordIterator implements CloseableIterator<Record> {
573           private final Long logAppendTime;
574           private final long baseOffset;
575           private final long baseTimestamp;
576           private final int baseSequence;
577           private final int numRecords;
578           private int readRecords = 0;
579   
580           RecordIterator() {
581               this.logAppendTime = timestampType() == TimestampType.LOG_APPEND_TIME ? maxTimestamp() : null;
582               this.baseOffset = baseOffset();
583               this.baseTimestamp = baseTimestamp();
584               this.baseSequence = baseSequence();
585               int numRecords = count();
586               if (numRecords < 0)
587                   throw new InvalidRecordException("Found invalid record count " + numRecords + " in magic v" +
588                           magic() + " batch");
589               this.numRecords = numRecords;
590           }
591   
592           @Override
593           public boolean hasNext() {
594               return readRecords < numRecords;
595           }
596   
597           @Override
598           public Record next() {
599               if (readRecords >= numRecords)
600                   throw new NoSuchElementException();
601   
602               readRecords++;
603               Record rec = readNext(baseOffset, baseTimestamp, baseSequence, logAppendTime);
604               if (readRecords == numRecords) {
605                   // Validate that the actual size of the batch is equal to declared size
606                   // by checking that after reading declared number of items, there no items left
607                   // (overflow case, i.e. reading past buffer end is checked elsewhere).
608                   if (!ensureNoneRemaining())
609                       throw new InvalidRecordException("Incorrect declared batch size, records still remaining in file");
610               }
611               return rec;
612           }
