42   /**
43    * The `KafkaAdminClient`'s internal `Call` primitive is not a good fit for multi-stage
44    * request workflows such as we see with the group coordinator APIs or any request which
45    * needs to be sent to a partition leader. Typically these APIs have two concrete stages:
46    *
47    * 1. Lookup: Find the broker that can fulfill the request (e.g. partition leader or group
48    *            coordinator)
49    * 2. Fulfillment: Send the request to the broker found in the first step
50    *
51    * This is complicated by the fact that `Admin` APIs are typically batched, which
52    * means the Lookup stage may result in a set of brokers. For example, take a `ListOffsets`
53    * request for a set of topic partitions. In the Lookup stage, we will find the partition
54    * leaders for this set of partitions; in the Fulfillment stage, we will group together
55    * partition according to the IDs of the discovered leaders.
56    *
57    * Additionally, the flow between these two stages is bi-directional. We may find after
58    * sending a `ListOffsets` request to an expected leader that there was a leader change.
59    * This would result in a topic partition being sent back to the Lookup stage.
60    *
61    * Managing this complexity by chaining together `Call` implementations is challenging
62    * and messy, so instead we use this class to do the bookkeeping. It handles both the
63    * batching aspect as well as the transitions between the Lookup and Fulfillment stages.
64    *
65    * Note that the interpretation of the `retries` configuration becomes ambiguous
66    * for this kind of pipeline. We could treat it as an overall limit on the number
67    * of requests that can be sent, but that is not very useful because each pipeline
68    * has a minimum number of requests that need to be sent in order to satisfy the request.
69    * Instead, we treat this number of retries independently at each stage so that each
70    * stage has at least one opportunity to complete. So if a user sets `retries=1`, then
71    * the full pipeline can still complete as long as there are no request failures.
72    *
73    * @param <K> The key type, which is also the granularity of the request routing (e.g.
74    *            this could be `TopicPartition` in the case of requests intended for a partition
75    *            leader or the `GroupId` in the case of consumer group requests intended for
76    *            the group coordinator)
77    * @param <V> The fulfillment type for each key (e.g. this could be consumer group state
78    *            when the key type is a consumer `GroupId`)
79    */
80   public class AdminApiDriver<K, V> {
81       private final Logger log;
82       private final long retryBackoffMs;
83       private final long deadlineMs;
84       private final AdminApiHandler<K, V> handler;
85       private final AdminApiFuture<K, V> future;
86   
87       private final BiMultimap<ApiRequestScope, K> lookupMap = new BiMultimap<>();
88       private final BiMultimap<FulfillmentScope, K> fulfillmentMap = new BiMultimap<>();
89       private final Map<ApiRequestScope, RequestState> requestStates = new HashMap<>();
90   
91       public AdminApiDriver(
92           AdminApiHandler<K, V> handler,
93           AdminApiFuture<K, V> future,
94           long deadlineMs,
95           long retryBackoffMs,
96           LogContext logContext
97       ) {
98           this.handler = handler;
99           this.future = future;
100           this.deadlineMs = deadlineMs;
101           this.retryBackoffMs = retryBackoffMs;
102           this.log = logContext.logger(AdminApiDriver.class);
103           retryLookup(future.lookupKeys());
104       }
105   
106       /**
107        * Associate a key with a brokerId. This is called after a response in the Lookup
108        * stage reveals the mapping (e.g. when the `FindCoordinator` tells us the group
109        * coordinator for a specific consumer group).
110        */
111       private void map(K key, Integer brokerId) {
112           lookupMap.remove(key);
113           fulfillmentMap.put(new FulfillmentScope(brokerId), key);
114       }
115   
116       /**
117        * Disassociate a key from the currently mapped brokerId. This will send the key
118        * back to the Lookup stage, which will allow us to attempt lookup again.
119        */
120       private void unmap(K key) {
121           fulfillmentMap.remove(key);
122   
123           ApiRequestScope lookupScope = handler.lookupStrategy().lookupScope(key);
124           OptionalInt destinationBrokerId = lookupScope.destinationBrokerId();
125   
126           if (destinationBrokerId.isPresent()) {
127               fulfillmentMap.put(new FulfillmentScope(destinationBrokerId.getAsInt()), key);
128           } else {
129               lookupMap.put(handler.lookupStrategy().lookupScope(key), key);
130           }
131       }
132   
133       private void clear(Collection<K> keys) {
134           keys.forEach(key -> {
135               lookupMap.remove(key);
136               fulfillmentMap.remove(key);
137           });
138       }
139   
140       OptionalInt keyToBrokerId(K key) {
141           Optional<FulfillmentScope> scope = fulfillmentMap.getKey(key);
142           return scope
143               .map(fulfillmentScope -> OptionalInt.of(fulfillmentScope.destinationBrokerId))
144               .orElseGet(OptionalInt::empty);
145       }
146   
147       /**
148        * Complete the future associated with the given key exceptionally. After is called,
149        * the key will be taken out of both the Lookup and Fulfillment stages so that request
150        * are not retried.
151        */
152       private void completeExceptionally(Map<K, Throwable> errors) {
153           if (!errors.isEmpty()) {
154               future.completeExceptionally(errors);
155               clear(errors.keySet());
156           }
157       }
158   
159       private void completeLookupExceptionally(Map<K, Throwable> errors) {
160           if (!errors.isEmpty()) {
161               future.completeLookupExceptionally(errors);
162               clear(errors.keySet());
163           }
164       }
165   
166       private void retryLookup(Collection<K> keys) {
167           keys.forEach(this::unmap);
168       }
169   
170       /**
171        * Complete the future associated with the given key. After this is called, all keys will
172        * be taken out of both the Lookup and Fulfillment stages so that request are not retried.
173        */
174       private void complete(Map<K, V> values) {
175           if (!values.isEmpty()) {
176               future.complete(values);
177               clear(values.keySet());
178           }
179       }
180   
181       private void completeLookup(Map<K, Integer> brokerIdMapping) {
182           if (!brokerIdMapping.isEmpty()) {
183               future.completeLookup(brokerIdMapping);
184               brokerIdMapping.forEach(this::map);
185           }
186       }
187   
188       /**
189        * Check whether any requests need to be sent. This should be called immediately
190        * after the driver is constructed and then again after each request returns
191        * (i.e. after {@link #onFailure(long, RequestSpec, Throwable)} or
192        * {@link #onResponse(long, RequestSpec, AbstractResponse, Node)}).
193        *
194        * @return A list of requests that need to be sent
195        */
196       public List<RequestSpec<K>> poll() {
197           List<RequestSpec<K>> requests = new ArrayList<>();
198           collectLookupRequests(requests);
199           collectFulfillmentRequests(requests);
200           return requests;
201       }

203       /**
204        * Callback that is invoked when a `Call` returns a response successfully.
205        */
206       public void onResponse(
207           long currentTimeMs,
208           RequestSpec<K> spec,
209           AbstractResponse response,
210           Node node
211       ) {
212           clearInflightRequest(currentTimeMs, spec);
213   
214           if (spec.scope instanceof FulfillmentScope) {
215               AdminApiHandler.ApiResult<K, V> result = handler.handleResponse(
216                   node,
217                   spec.keys,
218                   response
219               );
220               complete(result.completedKeys);
221               completeExceptionally(result.failedKeys);
222               retryLookup(result.unmappedKeys);
223           } else {
224               AdminApiLookupStrategy.LookupResult<K> result = handler.lookupStrategy().handleResponse(
225                   spec.keys,
226                   response
227               );
228   
229               result.completedKeys.forEach(lookupMap::remove);
230               completeLookup(result.mappedKeys);
231               completeLookupExceptionally(result.failedKeys);
232           }
233       }
234   
235       /**
236        * Callback that is invoked when a `Call` is failed.
237        */
238       public void onFailure(
239           long currentTimeMs,
240           RequestSpec<K> spec,
241           Throwable t
242       ) {
243           clearInflightRequest(currentTimeMs, spec);
244           if (t instanceof DisconnectException) {
245               log.debug("Node disconnected before response could be received for request {}. " +
246                   "Will attempt retry", spec.request);
247   
248               // After a disconnect, we want the driver to attempt to lookup the key
249               // again. This gives us a chance to find a new coordinator or partition
250               // leader for example.
251               Set<K> keysToUnmap = spec.keys.stream()
252                   .filter(future.lookupKeys()::contains)
253                   .collect(Collectors.toSet());
254               retryLookup(keysToUnmap);
255   
256           } else if (t instanceof NoBatchedFindCoordinatorsException) {
257               ((CoordinatorStrategy) handler.lookupStrategy()).disableBatch();
258               Set<K> keysToUnmap = spec.keys.stream()
259                   .filter(future.lookupKeys()::contains)
260                   .collect(Collectors.toSet());
261               retryLookup(keysToUnmap);
262           } else {
263               Map<K, Throwable> errors = spec.keys.stream().collect(Collectors.toMap(
264                   Function.identity(),
265                   key -> t
266               ));
267   
268               if (spec.scope instanceof FulfillmentScope) {
269                   completeExceptionally(errors);
270               } else {
271                   completeLookupExceptionally(errors);
272               }
273           }
274       }

276       private void clearInflightRequest(long currentTimeMs, RequestSpec<K> spec) {
277           RequestState requestState = requestStates.get(spec.scope);
278           if (requestState != null) {
279               // Only apply backoff if it's not a retry of a lookup request
280               if (spec.scope instanceof FulfillmentScope) {
281                   requestState.clearInflight(currentTimeMs + retryBackoffMs);
282               } else {
283                   requestState.clearInflight(currentTimeMs);
284               }
285           }
286       }

288       private <T extends ApiRequestScope> void collectRequests(
289           List<RequestSpec<K>> requests,
290           BiMultimap<T, K> multimap,
291           BiFunction<Set<K>, T, Collection<AdminApiHandler.RequestAndKeys<K>>> buildRequest
292       ) {
293           for (Map.Entry<T, Set<K>> entry : multimap.entrySet()) {
294               T scope = entry.getKey();
295   
296               Set<K> keys = entry.getValue();
297               if (keys.isEmpty()) {
298                   continue;
299               }
300   
301               RequestState requestState = requestStates.computeIfAbsent(scope, c -> new RequestState());
302               if (requestState.hasInflight()) {
303                   continue;
304               }
305   
306               // Copy the keys to avoid exposing the underlying mutable set
307               Set<K> copyKeys = Collections.unmodifiableSet(new HashSet<>(keys));
308   
309               Collection<AdminApiHandler.RequestAndKeys<K>> newRequests = buildRequest.apply(copyKeys, scope);
310               if (newRequests.isEmpty()) {
311                   return;
312               }
313   
314               // Only process the first request; all the remaining requests will be targeted at the same broker
315               // and we don't want to issue more than one fulfillment request per broker at a time
316               AdminApiHandler.RequestAndKeys<K> newRequest = newRequests.iterator().next();
317               RequestSpec<K> spec = new RequestSpec<>(
318                   handler.apiName() + "(api=" + newRequest.request.apiKey() + ")",
319                   scope,
320                   newRequest.keys,
321                   newRequest.request,
322                   requestState.nextAllowedRetryMs,
323                   deadlineMs,
324                   requestState.tries
325               );
326   
327               requestState.setInflight(spec);
328               requests.add(spec);
329           }
330       }
331   
332       private void collectLookupRequests(List<RequestSpec<K>> requests) {
333           collectRequests(
334               requests,
335               lookupMap,
336               (keys, scope) -> Collections.singletonList(new AdminApiHandler.RequestAndKeys<>(handler.lookupStrategy().buildRequest(keys), keys))
337           );
338       }
339   
340       private void collectFulfillmentRequests(List<RequestSpec<K>> requests) {
341           collectRequests(
342               requests,
343               fulfillmentMap,
344               (keys, scope) -> handler.buildRequest(scope.destinationBrokerId, keys)
345           );
346       }

394       /**
395        * Helper class used to track the request state within each request scope.
396        * This class enforces a maximum number of inflight request and keeps track
397        * of backoff/retry state.
398        */
399       private class RequestState {
400           private Optional<RequestSpec<K>> inflightRequest = Optional.empty();
401           private int tries = 0;
402           private long nextAllowedRetryMs = 0;
403   
404           boolean hasInflight() {
405               return inflightRequest.isPresent();
406           }
407   
408           public void clearInflight(long nextAllowedRetryMs) {
409               this.inflightRequest = Optional.empty();
410               this.nextAllowedRetryMs = nextAllowedRetryMs;
411           }
412   
413           public void setInflight(RequestSpec<K> spec) {
414               this.inflightRequest = Optional.of(spec);
415               this.tries++;
416           }
417       }

419       /**
420        * Completion of the Lookup stage results in a destination broker to send the
421        * fulfillment request to. Each destination broker in the Fulfillment stage
422        * gets its own request scope.
423        */
424       private static class FulfillmentScope implements ApiRequestScope {
425           public final int destinationBrokerId;
426   
427           private FulfillmentScope(int destinationBrokerId) {
428               this.destinationBrokerId = destinationBrokerId;
429           }
430   
431           @Override
432           public OptionalInt destinationBrokerId() {
433               return OptionalInt.of(destinationBrokerId);
434           }
435   
436           @Override
437           public boolean equals(Object o) {
438               if (this == o) return true;
439               if (o == null || getClass() != o.getClass()) return false;
440               FulfillmentScope that = (FulfillmentScope) o;
441               return destinationBrokerId == that.destinationBrokerId;
442           }
443   
444           @Override
445           public int hashCode() {
446               return Objects.hash(destinationBrokerId);
447           }
448       }

450       /**
451        * Helper class which maintains a bi-directional mapping from a key to a set of values.
452        * Each value can map to one and only one key, but many values can be associated with
453        * a single key.
454        *
455        * @param <K> The key type
456        * @param <V> The value type
457        */
458       private static class BiMultimap<K, V> {
459           private final Map<V, K> reverseMap = new HashMap<>();
460           private final Map<K, Set<V>> map = new HashMap<>();
461   
462           void put(K key, V value) {
463               remove(value);
464               reverseMap.put(value, key);
465               map.computeIfAbsent(key, k -> new HashSet<>()).add(value);
466           }
467   
468           void remove(V value) {
469               K key = reverseMap.remove(value);
470               if (key != null) {
471                   Set<V> set = map.get(key);
472                   if (set != null) {
473                       set.remove(value);
474                       if (set.isEmpty()) {
475                           map.remove(key);
476                       }
