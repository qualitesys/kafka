49    * Register metrics in JMX as dynamic mbeans based on the metric names

54

59       public static final String INCLUDE_CONFIG_ALIAS = METRICS_CONFIG_PREFIX + "whitelist";

106       @Override
107       public void reconfigure(Map<String, ?> configs) {
108           synchronized (LOCK) {
109               this.mbeanPredicate = JmxReporter.compilePredicate(configs);
110   
111               mbeans.forEach((name, mbean) -> {
112                   if (mbeanPredicate.test(name)) {
113                       reregister(mbean);
114                   } else {
115                       unregister(mbean);
116                   }
117               });
118           }
119       }

125                   addAttribute(metric);
126   
127               mbeans.forEach((name, mbean) -> {
128                   if (mbeanPredicate.test(name)) {
129                       reregister(mbean);
130                   }
131               });
132           }
133       }

139       @Override
140       public void metricChange(KafkaMetric metric) {
141           synchronized (LOCK) {
142               String mbeanName = addAttribute(metric);
143               if (mbeanName != null && mbeanPredicate.test(mbeanName)) {
144                   reregister(mbeans.get(mbeanName));
145               }
146           }
147       }

149       @Override
150       public void metricRemoval(KafkaMetric metric) {
151           synchronized (LOCK) {
152               MetricName metricName = metric.metricName();
153               String mBeanName = getMBeanName(prefix, metricName);
154               KafkaMbean mbean = removeAttribute(metric, mBeanName);
155               if (mbean != null) {
156                   if (mbean.metrics.isEmpty()) {
157                       unregister(mbean);
158                       mbeans.remove(mBeanName);
159                   } else if (mbeanPredicate.test(mBeanName))
160                       reregister(mbean);
161               }
162           }
163       }

165       private KafkaMbean removeAttribute(KafkaMetric metric, String mBeanName) {
166           MetricName metricName = metric.metricName();
167           KafkaMbean mbean = this.mbeans.get(mBeanName);
168           if (mbean != null)
169               mbean.removeAttribute(metricName.name());
170           return mbean;
171       }
172   
173       private String addAttribute(KafkaMetric metric) {
174           try {
175               MetricName metricName = metric.metricName();
176               String mBeanName = getMBeanName(prefix, metricName);
177               if (!this.mbeans.containsKey(mBeanName))
178                   mbeans.put(mBeanName, new KafkaMbean(mBeanName));
179               KafkaMbean mbean = this.mbeans.get(mBeanName);
180               mbean.setAttribute(metricName.name(), metric);
181               return mBeanName;
182           } catch (JMException e) {
183               throw new KafkaException("Error creating mbean attribute for metricName :" + metric.metricName(), e);
184           }
185       }
186   
187       /**
188        * @param metricName
189        * @return standard JMX MBean name in the following format domainName:type=metricType,key1=val1,key2=val2
190        */
191       static String getMBeanName(String prefix, MetricName metricName) {
192           StringBuilder mBeanName = new StringBuilder();
193           mBeanName.append(prefix);
194           mBeanName.append(":type=");
195           mBeanName.append(metricName.group());
196           for (Map.Entry<String, String> entry : metricName.tags().entrySet()) {
197               if (entry.getKey().length() <= 0 || entry.getValue().length() <= 0)
198                   continue;
199               mBeanName.append(",");
200               mBeanName.append(entry.getKey());
201               mBeanName.append("=");
202               mBeanName.append(Sanitizer.jmxSanitize(entry.getValue()));
203           }
204           return mBeanName.toString();
205       }

207       public void close() {
208           synchronized (LOCK) {
209               for (KafkaMbean mbean : this.mbeans.values())
210                   unregister(mbean);
211           }
212       }

214       private void unregister(KafkaMbean mbean) {
215           MBeanServer server = ManagementFactory.getPlatformMBeanServer();
216           try {
217               if (server.isRegistered(mbean.name()))
218                   server.unregisterMBean(mbean.name());
219           } catch (JMException e) {
220               throw new KafkaException("Error unregistering mbean", e);
221           }
222       }
223   
224       private void reregister(KafkaMbean mbean) {
225           unregister(mbean);
226           try {
227               ManagementFactory.getPlatformMBeanServer().registerMBean(mbean, mbean.name());
228           } catch (JMException e) {
229               throw new KafkaException("Error registering mbean " + mbean.name(), e);
230           }

275           @Override
276           public MBeanInfo getMBeanInfo() {
277               MBeanAttributeInfo[] attrs = new MBeanAttributeInfo[metrics.size()];
278               int i = 0;
279               for (Map.Entry<String, KafkaMetric> entry : this.metrics.entrySet()) {
280                   String attribute = entry.getKey();
281                   KafkaMetric metric = entry.getValue();
282                   attrs[i] = new MBeanAttributeInfo(attribute,
283                                                     double.class.getName(),
284                                                     metric.metricName().description(),
285                                                     true,
286                                                     false,
287                                                     false);
288                   i += 1;
289               }
290               return new MBeanInfo(this.getClass().getName(), "", attrs, null, null, null);
291           }

310       public static Predicate<String> compilePredicate(Map<String, ?> originalConfig) {
311           Map<String, ?> configs = ConfigUtils.translateDeprecatedConfigs(
312               originalConfig, new String[][]{{INCLUDE_CONFIG, INCLUDE_CONFIG_ALIAS},
313                                              {EXCLUDE_CONFIG, EXCLUDE_CONFIG_ALIAS}});
314           String include = (String) configs.get(INCLUDE_CONFIG);
315           String exclude = (String) configs.get(EXCLUDE_CONFIG);
316   
317           if (include == null) {
318               include = DEFAULT_INCLUDE;
319           }
320   
321           if (exclude == null) {
322               exclude = DEFAULT_EXCLUDE;
323           }
324   
325           try {
326               Pattern includePattern = Pattern.compile(include);
327               Pattern excludePattern = Pattern.compile(exclude);
328   
329               return s -> includePattern.matcher(s).matches()
330                           && !excludePattern.matcher(s).matches();
331           } catch (PatternSyntaxException e) {
332               throw new ConfigException("JMX filter for configuration" + METRICS_CONFIG_PREFIX
333                                         + ".(include/exclude) is not a valid regular expression");
334           }
335       }

337       @Override
338       public void contextChange(MetricsContext metricsContext) {
339           String namespace = metricsContext.contextLabels().get(MetricsContext.NAMESPACE);
340           Objects.requireNonNull(namespace);
341           synchronized (LOCK) {
342               if (!mbeans.isEmpty()) {
343                   throw new IllegalStateException("JMX MetricsContext can only be updated before JMX metrics are created");
344               }
345   
346               // prevent prefix from getting reset back to empty for backwards compatibility
347               // with the deprecated JmxReporter(String prefix) constructor, in case contextChange gets called
348               // via one of the Metrics() constructor with a default empty MetricsContext()
349               if (namespace.isEmpty()) {
350                   return;
351               }
352   
353               prefix = namespace;
354           }
355       }
