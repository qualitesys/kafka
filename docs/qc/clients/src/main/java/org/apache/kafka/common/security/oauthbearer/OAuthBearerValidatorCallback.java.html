17   package org.apache.kafka.common.security.oauthbearer;
18   
19   import java.util.Objects;
20   
21   import javax.security.auth.callback.Callback;
22   
23   import org.apache.kafka.common.annotation.InterfaceStability;
24   
25   /**
26    * A {@code Callback} for use by the {@code SaslServer} implementation when it
27    * needs to provide an OAuth 2 bearer token compact serialization for
28    * validation. Callback handlers should use the
29    * {@link #error(String, String, String)} method to communicate errors back to
30    * the SASL Client as per
31    * <a href="https://tools.ietf.org/html/rfc6749#section-5.2">RFC 6749: The OAuth
32    * 2.0 Authorization Framework</a> and the <a href=
33    * "https://www.iana.org/assignments/oauth-parameters/oauth-parameters.xhtml#extensions-error">IANA
34    * OAuth Extensions Error Registry</a>. Callback handlers should communicate
35    * other problems by raising an {@code IOException}.
36    * <p>
37    * This class was introduced in 2.0.0 and, while it feels stable, it could
38    * evolve. We will try to evolve the API in a compatible manner, but we reserve
39    * the right to make breaking changes in minor releases, if necessary. We will
40    * update the {@code InterfaceStability} annotation and this notice once the API
41    * is considered stable.
42    */
43   @InterfaceStability.Evolving
44   public class OAuthBearerValidatorCallback implements Callback {
45       private final String tokenValue;
46       private OAuthBearerToken token = null;
47       private String errorStatus = null;
48       private String errorScope = null;
49       private String errorOpenIDConfiguration = null;
50   
51       /**
52        * Constructor
53        * 
54        * @param tokenValue
55        *            the mandatory/non-blank token value
56        */
57       public OAuthBearerValidatorCallback(String tokenValue) {
58           if (Objects.requireNonNull(tokenValue).isEmpty())
59               throw new IllegalArgumentException("token value must not be empty");
60           this.tokenValue = tokenValue;
61       }
62   
63       /**
64        * Return the (always non-null) token value
65        * 
66        * @return the (always non-null) token value
67        */
68       public String tokenValue() {
69           return tokenValue;
70       }
71   
72       /**
73        * Return the (potentially null) token
74        * 
75        * @return the (potentially null) token
76        */
77       public OAuthBearerToken token() {
78           return token;
79       }
80   
81       /**
82        * Return the (potentially null) error status value as per
83        * <a href="https://tools.ietf.org/html/rfc7628#section-3.2.2">RFC 7628: A Set
84        * of Simple Authentication and Security Layer (SASL) Mechanisms for OAuth</a>
85        * and the <a href=
86        * "https://www.iana.org/assignments/oauth-parameters/oauth-parameters.xhtml#extensions-error">IANA
87        * OAuth Extensions Error Registry</a>.
88        * 
89        * @return the (potentially null) error status value
90        */
91       public String errorStatus() {
92           return errorStatus;
93       }
94   
95       /**
96        * Return the (potentially null) error scope value as per
97        * <a href="https://tools.ietf.org/html/rfc7628#section-3.2.2">RFC 7628: A Set
98        * of Simple Authentication and Security Layer (SASL) Mechanisms for OAuth</a>.
99        * 
100        * @return the (potentially null) error scope value
101        */
102       public String errorScope() {
103           return errorScope;
104       }
105   
106       /**
107        * Return the (potentially null) error openid-configuration value as per
108        * <a href="https://tools.ietf.org/html/rfc7628#section-3.2.2">RFC 7628: A Set
109        * of Simple Authentication and Security Layer (SASL) Mechanisms for OAuth</a>.
110        * 
111        * @return the (potentially null) error openid-configuration value
112        */
113       public String errorOpenIDConfiguration() {
114           return errorOpenIDConfiguration;
115       }
116   
117       /**
118        * Set the token. The token value is unchanged and is expected to match the
119        * provided token's value. All error values are cleared.
120        * 
121        * @param token
122        *            the mandatory token to set
123        */
124       public void token(OAuthBearerToken token) {
125           this.token = Objects.requireNonNull(token);
126           this.errorStatus = null;
127           this.errorScope = null;
128           this.errorOpenIDConfiguration = null;
129       }
130   
131       /**
132        * Set the error values as per
133        * <a href="https://tools.ietf.org/html/rfc7628#section-3.2.2">RFC 7628: A Set
134        * of Simple Authentication and Security Layer (SASL) Mechanisms for OAuth</a>.
135        * Any token is cleared.
136        * 
137        * @param errorStatus
138        *            the mandatory error status value from the <a href=
139        *            "https://www.iana.org/assignments/oauth-parameters/oauth-parameters.xhtml#extensions-error">IANA
140        *            OAuth Extensions Error Registry</a> to set
141        * @param errorScope
142        *            the optional error scope value to set
143        * @param errorOpenIDConfiguration
144        *            the optional error openid-configuration value to set
145        */
146       public void error(String errorStatus, String errorScope, String errorOpenIDConfiguration) {
147           if (Objects.requireNonNull(errorStatus).isEmpty())
148               throw new IllegalArgumentException("error status must not be empty");
149           this.errorStatus = errorStatus;
150           this.errorScope = errorScope;
151           this.errorOpenIDConfiguration = errorOpenIDConfiguration;
152           this.token = null;
153       }
154   }
