46   /**
47    * FetchSessionHandler maintains the fetch session state for connecting to a broker.
48    *
49    * Using the protocol outlined by KIP-227, clients can create incremental fetch sessions.
50    * These sessions allow the client to fetch information about a set of partition over
51    * and over, without explicitly enumerating all the partitions in the request and the
52    * response.
53    *
54    * FetchSessionHandler tracks the partitions which are in the session.  It also
55    * determines which partitions need to be included in each fetch request, and what
56    * the attached fetch session metadata should be for each request.  The corresponding
57    * class on the receiving broker side is FetchManager.
58    */

225       public class Builder {
226           /**
227            * The next partitions which we want to fetch.
228            *
229            * It is important to maintain the insertion order of this list by using a LinkedHashMap rather
230            * than a regular Map.
231            *
232            * One reason is that when dealing with FULL fetch requests, if there is not enough response
233            * space to return data from all partitions, the server will only return data from partitions
234            * early in this list.
235            *
236            * Another reason is because we make use of the list ordering to optimize the preparation of
237            * incremental fetch requests (see below).
238            */
239           private LinkedHashMap<TopicPartition, PartitionData> next;
240           private Map<Uuid, String> topicNames;
241           private final boolean copySessionPartitions;
242           private int partitionsWithoutTopicIds = 0;
243   
244           Builder() {
245               this.next = new LinkedHashMap<>();
246               this.topicNames = new HashMap<>();
247               this.copySessionPartitions = true;
248           }

256           /**
257            * Mark that we want data from this partition in the upcoming fetch.
258            */
259           public void add(TopicPartition topicPartition, PartitionData data) {
260               next.put(topicPartition, data);
261               // topicIds should not change between adding partitions and building, so we can use putIfAbsent
262               if (data.topicId.equals(Uuid.ZERO_UUID)) {
263                   partitionsWithoutTopicIds++;
264               } else {
265                   topicNames.putIfAbsent(data.topicId, topicPartition.topic());
266               }
267           }

269           public FetchRequestData build() {
270               boolean canUseTopicIds = partitionsWithoutTopicIds == 0;
271   
272               if (nextMetadata.isFull()) {
273                   if (log.isDebugEnabled()) {
274                       log.debug("Built full fetch {} for node {} with {}.",
275                               nextMetadata, node, topicPartitionsToLogString(next.keySet()));
276                   }
277                   sessionPartitions = next;
278                   next = null;
279                   // Only add topic IDs to the session if we are using topic IDs.
280                   if (canUseTopicIds) {
281                       sessionTopicNames = topicNames;
282                   } else {
283                       sessionTopicNames = Collections.emptyMap();
284                   }
285                   Map<TopicPartition, PartitionData> toSend =
286                           Collections.unmodifiableMap(new LinkedHashMap<>(sessionPartitions));
287                   return new FetchRequestData(toSend, Collections.emptyList(), Collections.emptyList(), toSend, nextMetadata, canUseTopicIds);
288               }
289   
290               List<TopicIdPartition> added = new ArrayList<>();
291               List<TopicIdPartition> removed = new ArrayList<>();
292               List<TopicIdPartition> altered = new ArrayList<>();
293               List<TopicIdPartition> replaced = new ArrayList<>();
294               for (Iterator<Entry<TopicPartition, PartitionData>> iter =
295                    sessionPartitions.entrySet().iterator(); iter.hasNext(); ) {
296                   Entry<TopicPartition, PartitionData> entry = iter.next();
297                   TopicPartition topicPartition = entry.getKey();
298                   PartitionData prevData = entry.getValue();
299                   PartitionData nextData = next.remove(topicPartition);
300                   if (nextData != null) {
301                       // We basically check if the new partition had the same topic ID. If not,
302                       // we add it to the "replaced" set. If the request is version 13 or higher, the replaced
303                       // partition will be forgotten. In any case, we will send the new partition in the request.
304                       if (!prevData.topicId.equals(nextData.topicId)
305                               && !prevData.topicId.equals(Uuid.ZERO_UUID)
306                               && !nextData.topicId.equals(Uuid.ZERO_UUID)) {
307                           // Re-add the replaced partition to the end of 'next'
308                           next.put(topicPartition, nextData);
309                           entry.setValue(nextData);
310                           replaced.add(new TopicIdPartition(prevData.topicId, topicPartition));
311                       } else if (!prevData.equals(nextData)) {
312                           // Re-add the altered partition to the end of 'next'
313                           next.put(topicPartition, nextData);
314                           entry.setValue(nextData);
315                           altered.add(new TopicIdPartition(nextData.topicId, topicPartition));
316                       }
317                   } else {
318                       // Remove this partition from the session.
319                       iter.remove();
320                       // Indicate that we no longer want to listen to this partition.
321                       removed.add(new TopicIdPartition(prevData.topicId, topicPartition));
322                       // If we do not have this topic ID in the builder or the session, we can not use topic IDs.
323                       if (canUseTopicIds && prevData.topicId.equals(Uuid.ZERO_UUID))
324                           canUseTopicIds = false;
325                   }
326               }
327               // Add any new partitions to the session.
328               for (Entry<TopicPartition, PartitionData> entry : next.entrySet()) {
329                   TopicPartition topicPartition = entry.getKey();
330                   PartitionData nextData = entry.getValue();
331                   if (sessionPartitions.containsKey(topicPartition)) {
332                       // In the previous loop, all the partitions which existed in both sessionPartitions
333                       // and next were moved to the end of next, or removed from next.  Therefore,
334                       // once we hit one of them, we know there are no more unseen entries to look
335                       // at in next.
336                       break;
337                   }
338                   sessionPartitions.put(topicPartition, nextData);
339                   added.add(new TopicIdPartition(nextData.topicId, topicPartition));
340               }
341   
342               // Add topic IDs to session if we can use them. If an ID is inconsistent, we will handle in the receiving broker.
343               // If we switched from using topic IDs to not using them (or vice versa), that error will also be handled in the receiving broker.
344               if (canUseTopicIds) {
345                   sessionTopicNames = topicNames;
346               } else {
347                   sessionTopicNames = Collections.emptyMap();
348               }
349   
350               if (log.isDebugEnabled()) {
351                   log.debug("Built incremental fetch {} for node {}. Added {}, altered {}, removed {}, " +
352                             "replaced {} out of {}", nextMetadata, node, topicIdPartitionsToLogString(added),
353                             topicIdPartitionsToLogString(altered), topicIdPartitionsToLogString(removed),
354                             topicIdPartitionsToLogString(replaced), topicPartitionsToLogString(sessionPartitions.keySet()));
355               }
356               Map<TopicPartition, PartitionData> toSend = Collections.unmodifiableMap(next);
357               Map<TopicPartition, PartitionData> curSessionPartitions = copySessionPartitions
358                       ? Collections.unmodifiableMap(new LinkedHashMap<>(sessionPartitions))
359                       : Collections.unmodifiableMap(sessionPartitions);
360               next = null;
361               return new FetchRequestData(toSend,
362                       Collections.unmodifiableList(removed),
363                       Collections.unmodifiableList(replaced),
364                       curSessionPartitions,
365                       nextMetadata,
366                       canUseTopicIds);
367           }
368       }

400       /**
401        * Return missing items which are expected to be in a particular set, but which are not.
402        *
403        * @param toFind    The items to look for.
404        * @param toSearch  The set of items to search.
405        * @return          Empty set if all items were found; some of the missing ones in a set, if not.
406        */
407       static <T> Set<T> findMissing(Set<T> toFind, Set<T> toSearch) {
408           Set<T> ret = new LinkedHashSet<>();
409           for (T toFindItem: toFind) {
410               if (!toSearch.contains(toFindItem)) {
411                   ret.add(toFindItem);
412               }
413           }
414           return ret;
415       }

417       /**
418        * Verify that a full fetch response contains all the partitions in the fetch session.
419        *
420        * @param topicPartitions  The topicPartitions from the FetchResponse.
421        * @param ids              The topic IDs from the FetchResponse.
422        * @param version          The version of the FetchResponse.
423        * @return                 null if the full fetch response partitions are valid; human-readable problem description otherwise.
424        */
425       String verifyFullFetchResponsePartitions(Set<TopicPartition> topicPartitions, Set<Uuid> ids, short version) {
426           StringBuilder bld = new StringBuilder();
427           Set<TopicPartition> extra =
428               findMissing(topicPartitions, sessionPartitions.keySet());
429           Set<TopicPartition> omitted =
430               findMissing(sessionPartitions.keySet(), topicPartitions);
431           Set<Uuid> extraIds = new HashSet<>();
432           if (version >= 13) {
433               extraIds = findMissing(ids, sessionTopicNames.keySet());
434           }
435           if (!omitted.isEmpty()) {
436               bld.append("omittedPartitions=(").append(Utils.join(omitted, ", ")).append(", ");
437           }
438           if (!extra.isEmpty()) {
439               bld.append("extraPartitions=(").append(Utils.join(extra, ", ")).append(", ");
440           }
441           if (!extraIds.isEmpty()) {
442               bld.append("extraIds=(").append(Utils.join(extraIds, ", ")).append(", ");
443           }
444           if ((!omitted.isEmpty()) || (!extra.isEmpty()) || (!extraIds.isEmpty())) {
445               bld.append("response=(").append(Utils.join(topicPartitions, ", ")).append(")");
446               return bld.toString();
447           }
448           return null;
449       }

451       /**
452        * Verify that the partitions in an incremental fetch response are contained in the session.
453        *
454        * @param topicPartitions  The topicPartitions from the FetchResponse.
455        * @param ids              The topic IDs from the FetchResponse.
456        * @param version          The version of the FetchResponse.
457        * @return                 null if the incremental fetch response partitions are valid; human-readable problem description otherwise.
458        */
459       String verifyIncrementalFetchResponsePartitions(Set<TopicPartition> topicPartitions, Set<Uuid> ids, short version) {
460           Set<Uuid> extraIds = new HashSet<>();
461           if (version >= 13) {
462               extraIds = findMissing(ids, sessionTopicNames.keySet());
463           }
464           Set<TopicPartition> extra =
465               findMissing(topicPartitions, sessionPartitions.keySet());
466           StringBuilder bld = new StringBuilder();
467           if (!extra.isEmpty())
468               bld.append("extraPartitions=(").append(Utils.join(extra, ", ")).append("), ");
469           if (!extraIds.isEmpty())
470               bld.append("extraIds=(").append(Utils.join(extraIds, ", ")).append("), ");
471           if ((!extra.isEmpty()) || (!extraIds.isEmpty())) {
472               bld.append("response=(").append(Utils.join(topicPartitions, ", ")).append(")");
473               return bld.toString();
474           }
475           return null;
476       }
477   
478       /**
479        * Create a string describing the partitions in a FetchResponse.
480        *
481        * @param topicPartitions  The topicPartitions from the FetchResponse.
482        * @return                 The string to log.
483        */
484       private String responseDataToLogString(Set<TopicPartition> topicPartitions) {
485           if (!log.isTraceEnabled()) {
486               int implied = sessionPartitions.size() - topicPartitions.size();
487               if (implied > 0) {
488                   return String.format(" with %d response partition(s), %d implied partition(s)",
489                       topicPartitions.size(), implied);
490               } else {
491                   return String.format(" with %d response partition(s)",
492                       topicPartitions.size());
493               }
494           }
495           StringBuilder bld = new StringBuilder();
496           bld.append(" with response=(").
497               append(Utils.join(topicPartitions, ", ")).
498               append(")");
499           String prefix = ", implied=(";
500           String suffix = "";
501           for (TopicPartition partition : sessionPartitions.keySet()) {
502               if (!topicPartitions.contains(partition)) {
503                   bld.append(prefix);
504                   bld.append(partition);
505                   prefix = ", ";
506                   suffix = ")";
507               }
508           }
509           bld.append(suffix);
510           return bld.toString();
511       }

513       /**
514        * Handle the fetch response.
515        *
516        * @param response  The response.
517        * @param version   The version of the request.
518        * @return          True if the response is well-formed; false if it can't be processed
519        *                  because of missing or unexpected partitions.
520        */
521       public boolean handleResponse(FetchResponse response, short version) {
522           if (response.error() != Errors.NONE) {
523               log.info("Node {} was unable to process the fetch request with {}: {}.",
524                   node, nextMetadata, response.error());
525               if (response.error() == Errors.FETCH_SESSION_ID_NOT_FOUND) {
526                   nextMetadata = FetchMetadata.INITIAL;
527               } else {
528                   nextMetadata = nextMetadata.nextCloseExisting();
529               }
530               return false;
531           }
532           Set<TopicPartition> topicPartitions = response.responseData(sessionTopicNames, version).keySet();
533           if (nextMetadata.isFull()) {
534               if (topicPartitions.isEmpty() && response.throttleTimeMs() > 0) {
535                   // Normally, an empty full fetch response would be invalid.  However, KIP-219
536                   // specifies that if the broker wants to throttle the client, it will respond
537                   // to a full fetch request with an empty response and a throttleTimeMs
538                   // value set.  We don't want to log this with a warning, since it's not an error.
539                   // However, the empty full fetch response can't be processed, so it's still appropriate
540                   // to return false here.
541                   if (log.isDebugEnabled()) {
542                       log.debug("Node {} sent a empty full fetch response to indicate that this " +
543                           "client should be throttled for {} ms.", node, response.throttleTimeMs());
544                   }
545                   nextMetadata = FetchMetadata.INITIAL;
546                   return false;
547               }
548               String problem = verifyFullFetchResponsePartitions(topicPartitions, response.topicIds(), version);
549               if (problem != null) {
550                   log.info("Node {} sent an invalid full fetch response with {}", node, problem);
551                   nextMetadata = FetchMetadata.INITIAL;
552                   return false;
553               } else if (response.sessionId() == INVALID_SESSION_ID) {
554                   if (log.isDebugEnabled())
555                       log.debug("Node {} sent a full fetch response{}", node, responseDataToLogString(topicPartitions));
556                   nextMetadata = FetchMetadata.INITIAL;
557                   return true;
558               } else {
559                   // The server created a new incremental fetch session.
560                   if (log.isDebugEnabled())
561                       log.debug("Node {} sent a full fetch response that created a new incremental " +
562                               "fetch session {}{}", node, response.sessionId(), responseDataToLogString(topicPartitions));
563                   nextMetadata = FetchMetadata.newIncremental(response.sessionId());
564                   return true;
565               }
566           } else {
567               String problem = verifyIncrementalFetchResponsePartitions(topicPartitions, response.topicIds(), version);
568               if (problem != null) {
569                   log.info("Node {} sent an invalid incremental fetch response with {}", node, problem);
570                   nextMetadata = nextMetadata.nextCloseExisting();
571                   return false;
572               } else if (response.sessionId() == INVALID_SESSION_ID) {
573                   // The incremental fetch session was closed by the server.
574                   if (log.isDebugEnabled())
575                       log.debug("Node {} sent an incremental fetch response closing session {}{}",
576                               node, nextMetadata.sessionId(), responseDataToLogString(topicPartitions));
577                   nextMetadata = FetchMetadata.INITIAL;
578                   return true;
579               } else {
580                   // The incremental fetch session was continued by the server.
581                   // We don't have to do anything special here to support KIP-219, since an empty incremental
582                   // fetch request is perfectly valid.
583                   if (log.isDebugEnabled())
584                       log.debug("Node {} sent an incremental fetch response with throttleTimeMs = {} " +
585                           "for session {}{}", node, response.throttleTimeMs(), response.sessionId(),
586                           responseDataToLogString(topicPartitions));
587                   nextMetadata = nextMetadata.nextIncremental();
588                   return true;
589               }
590           }
591       }
592   
593       /**
594        * Handle an error sending the prepared request.
595        *
596        * When a network error occurs, we close any existing fetch session on our next request,
597        * and try to create a new session.
598        *
599        * @param t     The exception.
600        */
601       public void handleError(Throwable t) {
602           log.info("Error sending fetch request {} to node {}:", nextMetadata, node, t);
603           nextMetadata = nextMetadata.nextCloseExisting();
604       }
