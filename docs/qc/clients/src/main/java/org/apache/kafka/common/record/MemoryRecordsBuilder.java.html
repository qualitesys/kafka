36   /**
37    * This class is used to write new log data in memory, i.e. this is the write path for {@link MemoryRecords}.
38    * It transparently handles compression and exposes methods for appending new records, possibly with message
39    * format conversion.
40    *
41    * In cases where keeping memory retention low is important and there's a gap between the time that record appends stop
42    * and the builder is closed (e.g. the Producer), it's important to call `closeForRecordAppends` when the former happens.
43    * This will release resources like compression buffers that can be relatively large (64 KB for LZ4).
44    */

243       /**

245        *
246        * If the log append time is used, the offset will be the last offset unless no compression is used and
247        * the message format version is 0 or 1, in which case, it will be the first offset.

258               if (compressionType != CompressionType.NONE || magic >= RecordBatch.MAGIC_VALUE_V2)
259                   shallowOffsetOfMaxTimestamp = lastOffset;
260               else
261                   shallowOffsetOfMaxTimestamp = baseOffset;
262               return new RecordsInfo(logAppendTime, shallowOffsetOfMaxTimestamp);
263           } else if (maxTimestamp == RecordBatch.NO_TIMESTAMP) {
264               return new RecordsInfo(RecordBatch.NO_TIMESTAMP, lastOffset);
265           } else {
266               long shallowOffsetOfMaxTimestamp;
267               // Use the last offset when dealing with record batches
268               if (compressionType != CompressionType.NONE || magic >= RecordBatch.MAGIC_VALUE_V2)
269                   shallowOffsetOfMaxTimestamp = lastOffset;
270               else
271                   shallowOffsetOfMaxTimestamp = offsetOfMaxTimestamp;
272               return new RecordsInfo(maxTimestamp, shallowOffsetOfMaxTimestamp);

287       public void setProducerState(long producerId, short producerEpoch, int baseSequence, boolean isTransactional) {
288           if (isClosed()) {

294           }
295           this.producerId = producerId;
296           this.producerEpoch = producerEpoch;
297           this.baseSequence = baseSequence;
298           this.isTransactional = isTransactional;
299       }

340       public void close() {
341           if (aborted)
342               throw new IllegalStateException("Cannot close MemoryRecordsBuilder as it has already been aborted");
343   
344           if (builtRecords != null)
345               return;
346   
347           validateProducerState();
348   
349           closeForRecordAppends();
350   
351           if (numRecords == 0L) {
352               buffer().position(initialPosition);
353               builtRecords = MemoryRecords.EMPTY;
354           } else {
355               if (magic > RecordBatch.MAGIC_VALUE_V1)
356                   this.actualCompressionRatio = (float) writeDefaultBatchHeader() / this.uncompressedRecordsSizeInBytes;
357               else if (compressionType != CompressionType.NONE)
358                   this.actualCompressionRatio = (float) writeLegacyCompressedWrapperHeader() / this.uncompressedRecordsSizeInBytes;
359   
360               ByteBuffer buffer = buffer().duplicate();
361               buffer.flip();
362               buffer.position(initialPosition);
363               builtRecords = MemoryRecords.readableRecords(buffer.slice());
364           }
365       }

367       private void validateProducerState() {
368           if (isTransactional && producerId == RecordBatch.NO_PRODUCER_ID)
369               throw new IllegalArgumentException("Cannot write transactional messages without a valid producer ID");
370   
371           if (producerId != RecordBatch.NO_PRODUCER_ID) {
372               if (producerEpoch == RecordBatch.NO_PRODUCER_EPOCH)
373                   throw new IllegalArgumentException("Invalid negative producer epoch");
374   
375               if (baseSequence < 0 && !isControlBatch)
376                   throw new IllegalArgumentException("Invalid negative sequence number used");
377   
378               if (magic < RecordBatch.MAGIC_VALUE_V2)
379                   throw new IllegalArgumentException("Idempotent messages are not supported for magic " + magic);
380           }
381       }

383       /**
384        * Write the header to the default batch.
385        * @return the written compressed bytes.
386        */
387       private int writeDefaultBatchHeader() {
388           ensureOpenForRecordBatchWrite();
389           ByteBuffer buffer = bufferStream.buffer();
390           int pos = buffer.position();
391           buffer.position(initialPosition);
392           int size = pos - initialPosition;
393           int writtenCompressed = size - DefaultRecordBatch.RECORD_BATCH_OVERHEAD;
394           int offsetDelta = (int) (lastOffset - baseOffset);
395   
396           final long maxTimestamp;
397           if (timestampType == TimestampType.LOG_APPEND_TIME)
398               maxTimestamp = logAppendTime;
399           else
400               maxTimestamp = this.maxTimestamp;
401   
402           DefaultRecordBatch.writeHeader(buffer, baseOffset, offsetDelta, size, magic, compressionType, timestampType,
403                   baseTimestamp, maxTimestamp, producerId, producerEpoch, baseSequence, isTransactional, isControlBatch,
404                   hasDeleteHorizonMs(), partitionLeaderEpoch, numRecords);
405   
406           buffer.position(pos);
407           return writtenCompressed;
408       }

410       /**
411        * Write the header to the legacy batch.
412        * @return the written compressed bytes.
413        */
414       private int writeLegacyCompressedWrapperHeader() {
415           ensureOpenForRecordBatchWrite();
416           ByteBuffer buffer = bufferStream.buffer();
417           int pos = buffer.position();
418           buffer.position(initialPosition);
419   
420           int wrapperSize = pos - initialPosition - Records.LOG_OVERHEAD;
421           int writtenCompressed = wrapperSize - LegacyRecord.recordOverhead(magic);
422           AbstractLegacyRecordBatch.writeHeader(buffer, lastOffset, wrapperSize);
423   
424           long timestamp = timestampType == TimestampType.LOG_APPEND_TIME ? logAppendTime : maxTimestamp;
425           LegacyRecord.writeCompressedRecordHeader(buffer, magic, wrapperSize, timestamp, compressionType, timestampType);
426   
427           buffer.position(pos);
428           return writtenCompressed;
429       }

431       /**
432        * Append a new record at the given offset.
433        */
434       private void appendWithOffset(long offset, boolean isControlRecord, long timestamp, ByteBuffer key,
435                                     ByteBuffer value, Header[] headers) {
436           try {
437               if (isControlRecord != isControlBatch)
438                   throw new IllegalArgumentException("Control records can only be appended to control batches");
439   
440               if (lastOffset != null && offset <= lastOffset)
441                   throw new IllegalArgumentException(String.format("Illegal offset %s following previous offset %s " +
442                           "(Offsets must increase monotonically).", offset, lastOffset));
443   
444               if (timestamp < 0 && timestamp != RecordBatch.NO_TIMESTAMP)
445                   throw new IllegalArgumentException("Invalid negative timestamp " + timestamp);
446   
447               if (magic < RecordBatch.MAGIC_VALUE_V2 && headers != null && headers.length > 0)
448                   throw new IllegalArgumentException("Magic v" + magic + " does not support record headers");

720       private long appendLegacyRecord(long offset, long timestamp, ByteBuffer key, ByteBuffer value, byte magic) throws IOException {
721           ensureOpenForRecordAppend();
722           if (compressionType == CompressionType.NONE && timestampType == TimestampType.LOG_APPEND_TIME)
723               timestamp = logAppendTime;
724   
725           int size = LegacyRecord.recordSize(magic, key, value);
726           AbstractLegacyRecordBatch.writeHeader(appendStream, toInnerOffset(offset), size);
727   
728           if (timestampType == TimestampType.LOG_APPEND_TIME)
729               timestamp = logAppendTime;
730           long crc = LegacyRecord.write(appendStream, magic, timestamp, key, value, CompressionType.NONE, timestampType);
731           recordWritten(offset, timestamp, size + Records.LOG_OVERHEAD);
732           return crc;
733       }

742       private void recordWritten(long offset, long timestamp, int size) {
743           if (numRecords == Integer.MAX_VALUE)
744               throw new IllegalArgumentException("Maximum number of records per batch exceeded, max records: " + Integer.MAX_VALUE);
745           if (offset - baseOffset > Integer.MAX_VALUE)
746               throw new IllegalArgumentException("Maximum offset delta exceeded, base offset: " + baseOffset +
747                       ", last offset: " + offset);
748   
749           numRecords += 1;
750           uncompressedRecordsSizeInBytes += size;
751           lastOffset = offset;
752   
753           if (magic > RecordBatch.MAGIC_VALUE_V0 && timestamp > maxTimestamp) {
754               maxTimestamp = timestamp;
755               offsetOfMaxTimestamp = offset;
756           }

794        */

799       /**
800        * Check if we have room for a new record containing the given key/value pair. If no records have been
801        * appended, then this returns true.
802        *
803        * Note that the return value is based on the estimate of the bytes written to the compressor, which may not be
804        * accurate if compression is used. When this happens, the following append may cause dynamic buffer
805        * re-allocation in the underlying byte buffer stream.
806        */
807       public boolean hasRoomFor(long timestamp, ByteBuffer key, ByteBuffer value, Header[] headers) {
808           if (isFull())
809               return false;
810   
811           // We always allow at least one record to be appended (the ByteBufferOutputStream will grow as needed)
812           if (numRecords == 0)
813               return true;
814   
815           final int recordSize;
816           if (magic < RecordBatch.MAGIC_VALUE_V2) {
817               recordSize = Records.LOG_OVERHEAD + LegacyRecord.recordSize(magic, key, value);
818           } else {
819               int nextOffsetDelta = lastOffset == null ? 0 : (int) (lastOffset - baseOffset + 1);
820               long timestampDelta = baseTimestamp == null ? 0 : timestamp - baseTimestamp;
821               recordSize = DefaultRecord.sizeInBytes(nextOffsetDelta, timestampDelta, key, value, headers);
822           }
823   
824           // Be conservative and not take compression of the new record into consideration.
825           return this.writeLimit >= estimatedBytesWritten() + recordSize;
826       }
