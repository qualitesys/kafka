40       /**
41        * Translates deprecated configurations into their non-deprecated equivalents
42        *
43        * This is a convenience method for {@link ConfigUtils#translateDeprecatedConfigs(Map, Map)}
44        * until we can use Java 9+ {@code Map.of(..)} and {@code Set.of(...)}
45        *
46        * @param configs the input configuration
47        * @param aliasGroups An array of arrays of synonyms.  Each synonym array begins with the non-deprecated synonym
48        *                    For example, new String[][] { { a, b }, { c, d, e} }
49        *                    would declare b as a deprecated synonym for a,
50        *                    and d and e as deprecated synonyms for c.
51        *                    The ordering of synonyms determines the order of precedence
52        *                    (e.g. the first synonym takes precedence over the second one)
53        * @return a new configuration map with deprecated  keys translated to their non-deprecated equivalents
54        */
55       public static <T> Map<String, T> translateDeprecatedConfigs(Map<String, T> configs, String[][] aliasGroups) {
56           return translateDeprecatedConfigs(configs, Stream.of(aliasGroups)
57               .collect(Collectors.toMap(x -> x[0], x -> Stream.of(x).skip(1).collect(Collectors.toList()))));
58       }

60       /**
61        * Translates deprecated configurations into their non-deprecated equivalents
62        *
63        * @param configs the input configuration
64        * @param aliasGroups A map of config to synonyms.  Each key is the non-deprecated synonym
65        *                    For example, Map.of(a , Set.of(b), c, Set.of(d, e))
66        *                    would declare b as a deprecated synonym for a,
67        *                    and d and e as deprecated synonyms for c.
68        *                    The ordering of synonyms determines the order of precedence
69        *                    (e.g. the first synonym takes precedence over the second one)
70        * @return a new configuration map with deprecated  keys translated to their non-deprecated equivalents
71        */
72       public static <T> Map<String, T> translateDeprecatedConfigs(Map<String, T> configs,
73                                                                   Map<String, List<String>> aliasGroups) {
74           Set<String> aliasSet = Stream.concat(
75               aliasGroups.keySet().stream(),
76               aliasGroups.values().stream().flatMap(Collection::stream))
77               .collect(Collectors.toSet());
78   
79           // pass through all configurations without aliases
80           Map<String, T> newConfigs = configs.entrySet().stream()
81               .filter(e -> !aliasSet.contains(e.getKey()))
82               // filter out null values
83               .filter(e -> Objects.nonNull(e.getValue()))
84               .collect(Collectors.toMap(Map.Entry::getKey, Map.Entry::getValue));
85   
86           aliasGroups.forEach((target, aliases) -> {
87               List<String> deprecated = aliases.stream()
88                   .filter(configs::containsKey)
89                   .collect(Collectors.toList());
90   
91               if (deprecated.isEmpty()) {
92                   // No deprecated key(s) found.
93                   if (configs.containsKey(target)) {
94                       newConfigs.put(target, configs.get(target));
95                   }
96                   return;
97               }
98   
99               String aliasString = String.join(", ", deprecated);
100   
101               if (configs.containsKey(target)) {
102                   // Ignore the deprecated key(s) because the actual key was set.
103                   log.error(target + " was configured, as well as the deprecated alias(es) " +
104                             aliasString + ".  Using the value of " + target);
105                   newConfigs.put(target, configs.get(target));
106               } else if (deprecated.size() > 1) {
107                   log.error("The configuration keys " + aliasString + " are deprecated and may be " +
108                             "removed in the future.  Additionally, this configuration is ambigous because " +
109                             "these configuration keys are all aliases for " + target + ".  Please update " +
110                             "your configuration to have only " + target + " set.");
111                   newConfigs.put(target, configs.get(deprecated.get(0)));
112               } else {
113                   log.warn("Configuration key " + deprecated.get(0) + " is deprecated and may be removed " +
114                            "in the future.  Please update your configuration to use " + target + " instead.");
115                   newConfigs.put(target, configs.get(deprecated.get(0)));
116               }
117           });
118   
119           return newConfigs;
120       }

122       public static String configMapToRedactedString(Map<String, Object> map, ConfigDef configDef) {
123           StringBuilder bld = new StringBuilder("{");
124           List<String> keys = new ArrayList<>(map.keySet());
125           Collections.sort(keys);
126           String prefix = "";
127           for (String key : keys) {
128               bld.append(prefix).append(key).append("=");
129               ConfigKey configKey = configDef.configKeys().get(key);
130               if (configKey == null || configKey.type().isSensitive()) {
131                   bld.append("(redacted)");
132               } else {
133                   Object value = map.get(key);
134                   if (value == null) {
135                       bld.append("null");
136                   } else if (configKey.type() == Type.STRING) {
137                       bld.append("\"").append(value).append("\"");
138                   } else {
139                       bld.append(value);
140                   }
141               }
142               prefix = ", ";
143           }
144           bld.append("}");
145           return bld.toString();
146       }
