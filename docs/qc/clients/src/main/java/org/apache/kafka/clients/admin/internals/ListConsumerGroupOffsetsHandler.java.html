39   public class ListConsumerGroupOffsetsHandler extends AdminApiHandler.Batched<CoordinatorKey, Map<TopicPartition, OffsetAndMetadata>> {

42       private final List<TopicPartition> partitions;

65           this.lookupStrategy = new CoordinatorStrategy(CoordinatorType.GROUP, logContext);

84       private void validateKeys(Set<CoordinatorKey> groupIds) {
85           if (!groupIds.equals(Collections.singleton(groupId))) {
86               throw new IllegalArgumentException("Received unexpected group ids " + groupIds +
87                   " (expected only " + Collections.singleton(groupId) + ")");
88           }
89       }
90   
91       @Override
92       public OffsetFetchRequest.Builder buildBatchedRequest(int coordinatorId, Set<CoordinatorKey> groupIds) {
93           validateKeys(groupIds);
94           return new OffsetFetchRequest.Builder(groupId.idValue, requireStable, partitions, false);
95       }
96   
97       @Override
98       public ApiResult<CoordinatorKey, Map<TopicPartition, OffsetAndMetadata>> handleResponse(
99           Node coordinator,
100           Set<CoordinatorKey> groupIds,
101           AbstractResponse abstractResponse
102       ) {
103           validateKeys(groupIds);
104   
105           final OffsetFetchResponse response = (OffsetFetchResponse) abstractResponse;
106   
107           // the groupError will contain the group level error for v0-v8 OffsetFetchResponse
108           Errors groupError = response.groupLevelError(groupId.idValue);
109           if (groupError != Errors.NONE) {
110               final Map<CoordinatorKey, Throwable> failed = new HashMap<>();
111               final Set<CoordinatorKey> groupsToUnmap = new HashSet<>();
112   
113               handleGroupError(groupId, groupError, failed, groupsToUnmap);
114   
115               return new ApiResult<>(Collections.emptyMap(), failed, new ArrayList<>(groupsToUnmap));
116           } else {
117               final Map<TopicPartition, OffsetAndMetadata> groupOffsetsListing = new HashMap<>();
118   
119               response.partitionDataMap(groupId.idValue).forEach((topicPartition, partitionData) -> {
120                   final Errors error = partitionData.error;
121                   if (error == Errors.NONE) {
122                       final long offset = partitionData.offset;
123                       final String metadata = partitionData.metadata;
124                       final Optional<Integer> leaderEpoch = partitionData.leaderEpoch;
125                       // Negative offset indicates that the group has no committed offset for this partition
126                       if (offset < 0) {
127                           groupOffsetsListing.put(topicPartition, null);
128                       } else {
129                           groupOffsetsListing.put(topicPartition, new OffsetAndMetadata(offset, leaderEpoch, metadata));
130                       }
131                   } else {
132                       log.warn("Skipping return offset for {} due to error {}.", topicPartition, error);
133                   }
134               });
135   
136               return ApiResult.completed(groupId, groupOffsetsListing);
137           }
138       }
139   
140       private void handleGroupError(
141           CoordinatorKey groupId,
142           Errors error,
143           Map<CoordinatorKey, Throwable> failed,
144           Set<CoordinatorKey> groupsToUnmap
145       ) {
146           switch (error) {
147               case GROUP_AUTHORIZATION_FAILED:
148                   log.debug("`OffsetFetch` request for group id {} failed due to error {}", groupId.idValue, error);
149                   failed.put(groupId, error.exception());
150                   break;
151               case COORDINATOR_LOAD_IN_PROGRESS:
152                   // If the coordinator is in the middle of loading, then we just need to retry
153                   log.debug("`OffsetFetch` request for group id {} failed because the coordinator " +
154                       "is still in the process of loading state. Will retry", groupId.idValue);
155                   break;
156   
157               case COORDINATOR_NOT_AVAILABLE:
158               case NOT_COORDINATOR:
159                   // If the coordinator is unavailable or there was a coordinator change, then we unmap
160                   // the key so that we retry the `FindCoordinator` request
161                   log.debug("`OffsetFetch` request for group id {} returned error {}. " +
162                       "Will attempt to find the coordinator again and retry", groupId.idValue, error);
163                   groupsToUnmap.add(groupId);
164                   break;
165   
166               default:
167                   log.error("`OffsetFetch` request for group id {} failed due to unexpected error {}", groupId.idValue, error);
168                   failed.put(groupId, error.exception());
169           }
170       }
