17   package org.apache.kafka.clients.consumer.internals;
18   
19   import org.apache.kafka.clients.consumer.ConsumerRecord;
20   import org.apache.kafka.common.TopicPartition;
21   
22   import java.util.ArrayList;
23   import java.util.Collections;
24   import java.util.HashMap;
25   import java.util.List;
26   import java.util.Map;
27   import java.util.Objects;
28   
29   import static org.apache.kafka.common.utils.Utils.mkEntry;
30   import static org.apache.kafka.common.utils.Utils.mkMap;
31   
32   public class Fetch<K, V> {
33       private final Map<TopicPartition, List<ConsumerRecord<K, V>>> records;
34       private boolean positionAdvanced;
35       private int numRecords;
36   
37       public static <K, V> Fetch<K, V> empty() {
38           return new Fetch<>(new HashMap<>(), false, 0);
39       }
40   
41       public static <K, V> Fetch<K, V> forPartition(
42               TopicPartition partition,
43               List<ConsumerRecord<K, V>> records,
44               boolean positionAdvanced
45       ) {
46           Map<TopicPartition, List<ConsumerRecord<K, V>>> recordsMap = records.isEmpty()
47                   ? new HashMap<>()
48                   : mkMap(mkEntry(partition, records));
49           return new Fetch<>(recordsMap, positionAdvanced, records.size());
50       }
51   
52       private Fetch(
53               Map<TopicPartition, List<ConsumerRecord<K, V>>> records,
54               boolean positionAdvanced,
55               int numRecords
56       ) {
57           this.records = records;
58           this.positionAdvanced = positionAdvanced;
59           this.numRecords = numRecords;
60       }
61   
62       /**
63        * Add another {@link Fetch} to this one; all of its records will be added to this fetch's
64        * {@link #records()} records}, and if the other fetch
65        * {@link #positionAdvanced() advanced the consume position for any topic partition},
66        * this fetch will be marked as having advanced the consume position as well.
67        * @param fetch the other fetch to add; may not be null
68        */
69       public void add(Fetch<K, V> fetch) {
70           Objects.requireNonNull(fetch);
71           addRecords(fetch.records);
72           this.positionAdvanced |= fetch.positionAdvanced;
73       }
74   
75       /**
76        * @return all of the non-control messages for this fetch, grouped by partition
77        */
78       public Map<TopicPartition, List<ConsumerRecord<K, V>>> records() {
79           return Collections.unmodifiableMap(records);
80       }
81   
82       /**
83        * @return whether the fetch caused the consumer's
84        * {@link org.apache.kafka.clients.consumer.KafkaConsumer#position(TopicPartition) position} to advance for at
85        * least one of the topic partitions in this fetch
86        */
87       public boolean positionAdvanced() {
88           return positionAdvanced;
89       }
90   
91       /**
92        * @return the total number of non-control messages for this fetch, across all partitions
93        */
94       public int numRecords() {
95           return numRecords;
96       }
97   
98       /**
99        * @return {@code true} if and only if this fetch did not return any user-visible (i.e., non-control) records, and
100        * did not cause the consumer position to advance for any topic partitions
101        */
102       public boolean isEmpty() {
103           return numRecords == 0 && !positionAdvanced;
104       }
105   
106       private void addRecords(Map<TopicPartition, List<ConsumerRecord<K, V>>> records) {
107           records.forEach((partition, partRecords) -> {
108               this.numRecords += partRecords.size();
109               List<ConsumerRecord<K, V>> currentRecords = this.records.get(partition);
110               if (currentRecords == null) {
111                   this.records.put(partition, partRecords);
112               } else {
113                   // this case shouldn't usually happen because we only send one fetch at a time per partition,
114                   // but it might conceivably happen in some rare cases (such as partition leader changes).
115                   // we have to copy to a new list because the old one may be immutable
116                   List<ConsumerRecord<K, V>> newRecords = new ArrayList<>(currentRecords.size() + partRecords.size());
117                   newRecords.addAll(currentRecords);
118                   newRecords.addAll(partRecords);
119                   this.records.put(partition, newRecords);
120               }
121           });
122       }
123   }
