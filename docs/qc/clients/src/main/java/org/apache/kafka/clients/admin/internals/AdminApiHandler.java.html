17   package org.apache.kafka.clients.admin.internals;
18   
19   import org.apache.kafka.common.Node;
20   import org.apache.kafka.common.requests.AbstractRequest;
21   import org.apache.kafka.common.requests.AbstractResponse;
22   
23   import java.util.Collection;
24   import java.util.Collections;
25   import java.util.List;
26   import java.util.Map;
27   import java.util.Set;
28   import java.util.stream.Collectors;
29   
30   public interface AdminApiHandler<K, V> {
31   
32       /**
33        * Get a user-friendly name for the API this handler is implementing.
34        */
35       String apiName();
36   
37       /**
38        * Build the requests necessary for the given keys. The set of keys is derived by
39        * {@link AdminApiDriver} during the lookup stage as the set of keys which all map
40        * to the same destination broker. Handlers can choose to issue a single request for
41        * all of the provided keys (see {@link Batched}, issue one request per key (see
42        * {@link Unbatched}, or implement their own custom grouping logic if necessary.
43        *
44        * @param brokerId the target brokerId for the request
45        * @param keys the set of keys that should be handled by this request
46        *
47        * @return a collection of {@link RequestAndKeys} for the requests containing the given keys
48        */
49       Collection<RequestAndKeys<K>> buildRequest(int brokerId, Set<K> keys);
50   
51       /**
52        * Callback that is invoked when a request returns successfully.
53        * The handler should parse the response, check for errors, and return a
54        * result which indicates which keys (if any) have either been completed or
55        * failed with an unrecoverable error.
56        *
57        * It is also possible that the response indicates an incorrect target brokerId
58        * (e.g. in the case of a NotLeader error when the request is bound for a partition
59        * leader). In this case the key will be "unmapped" from the target brokerId
60        * and lookup will be retried.
61        *
62        * Note that keys which received a retriable error should be left out of the
63        * result. They will be retried automatically.
64        *
65        * @param broker the broker that the associated request was sent to
66        * @param keys the set of keys from the associated request
67        * @param response the response received from the broker
68        *
69        * @return result indicating key completion, failure, and unmapping
70        */
71       ApiResult<K, V> handleResponse(Node broker, Set<K> keys, AbstractResponse response);
72   
73       /**
74        * Get the lookup strategy that is responsible for finding the brokerId
75        * which will handle each respective key.
76        *
77        * @return non-null lookup strategy
78        */
79       AdminApiLookupStrategy<K> lookupStrategy();
80   
81       class ApiResult<K, V> {
82           public final Map<K, V> completedKeys;
83           public final Map<K, Throwable> failedKeys;
84           public final List<K> unmappedKeys;
85   
86           public ApiResult(
87               Map<K, V> completedKeys,
88               Map<K, Throwable> failedKeys,
89               List<K> unmappedKeys
90           ) {
91               this.completedKeys = Collections.unmodifiableMap(completedKeys);
92               this.failedKeys = Collections.unmodifiableMap(failedKeys);
93               this.unmappedKeys = Collections.unmodifiableList(unmappedKeys);
94           }
95   
96           public static <K, V> ApiResult<K, V> completed(K key, V value) {
97               return new ApiResult<>(
98                   Collections.singletonMap(key, value),
99                   Collections.emptyMap(),
100                   Collections.emptyList()
101               );
102           }
103   
104           public static <K, V> ApiResult<K, V> failed(K key, Throwable t) {
105               return new ApiResult<>(
106                   Collections.emptyMap(),
107                   Collections.singletonMap(key, t),
108                   Collections.emptyList()
109               );
110           }
111   
112           public static <K, V> ApiResult<K, V> unmapped(List<K> keys) {
113               return new ApiResult<>(
114                   Collections.emptyMap(),
115                   Collections.emptyMap(),
116                   keys
117               );
118           }
119   
120           public static <K, V> ApiResult<K, V> empty() {
121               return new ApiResult<>(
122                   Collections.emptyMap(),
123                   Collections.emptyMap(),
124                   Collections.emptyList()
125               );
126           }
127       }
128   
129       class RequestAndKeys<K> {
130           public final AbstractRequest.Builder<?> request;
131           public final Set<K> keys;
132   
133           public RequestAndKeys(AbstractRequest.Builder<?> request, Set<K> keys) {
134               this.request = request;
135               this.keys = keys;
136           }
137       }
138   
139       /**
140        * An {@link AdminApiHandler} that will group multiple keys into a single request when possible.
141        * Keys will be grouped together whenever they target the same broker. This type of handler
142        * should be used when interacting with broker APIs that can act on multiple keys at once, such
143        * as describing or listing transactions.
144        */
145       abstract class Batched<K, V> implements AdminApiHandler<K, V> {
146           abstract AbstractRequest.Builder<?> buildBatchedRequest(int brokerId, Set<K> keys);
147   
148           @Override
149           public final Collection<RequestAndKeys<K>> buildRequest(int brokerId, Set<K> keys) {
150               return Collections.singleton(new RequestAndKeys<>(buildBatchedRequest(brokerId, keys), keys));
151           }
152       }
153   
154       /**
155        * An {@link AdminApiHandler} that will create one request per key, not performing any grouping based
156        * on the targeted broker. This type of handler should only be used for broker APIs that do not accept
157        * multiple keys at once, such as initializing a transactional producer.
158        */
159       abstract class Unbatched<K, V> implements AdminApiHandler<K, V> {
160           abstract AbstractRequest.Builder<?> buildSingleRequest(int brokerId, K key);
161           abstract ApiResult<K, V> handleSingleResponse(Node broker, K key, AbstractResponse response);
162   
163           @Override
164           public final Collection<RequestAndKeys<K>> buildRequest(int brokerId, Set<K> keys) {
165               return keys.stream()
166                   .map(key -> new RequestAndKeys<>(buildSingleRequest(brokerId, key), Collections.singleton(key)))
167                   .collect(Collectors.toSet());
168           }
169   
170           @Override
171           public final ApiResult<K, V> handleResponse(Node broker, Set<K> keys, AbstractResponse response) {
172               if (keys.size() != 1) {
173                   throw new IllegalArgumentException("Unbatched admin handler should only be required to handle responses for a single key at a time");
174               }
175               K key = keys.iterator().next();
176               return handleSingleResponse(broker, key, response);
177           }
178       }
179   }
