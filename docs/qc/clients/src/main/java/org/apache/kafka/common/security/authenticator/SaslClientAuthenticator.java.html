77   public class SaslClientAuthenticator implements Authenticator {
78       /**
79        * The internal state transitions for initial authentication of a channel are
80        * declared in order, starting with {@link #SEND_APIVERSIONS_REQUEST} and ending
81        * in either {@link #COMPLETE} or {@link #FAILED}.
82        * <p>
83        * Re-authentication of a channel starts with the state
84        * {@link #REAUTH_PROCESS_ORIG_APIVERSIONS_RESPONSE} and then flows to
85        * {@link #REAUTH_SEND_HANDSHAKE_REQUEST} followed by
86        * {@link #REAUTH_RECEIVE_HANDSHAKE_OR_OTHER_RESPONSE} and then
87        * {@link #REAUTH_INITIAL}; after that the flow joins the authentication flow
88        * at the {@link #INTERMEDIATE} state and ends at either {@link #COMPLETE} or
89        * {@link #FAILED}.
90        */
91       public enum SaslState {
92           SEND_APIVERSIONS_REQUEST,                   // Initial state for authentication: client sends ApiVersionsRequest in this state when authenticating
93           RECEIVE_APIVERSIONS_RESPONSE,               // Awaiting ApiVersionsResponse from server
94           SEND_HANDSHAKE_REQUEST,                     // Received ApiVersionsResponse, send SaslHandshake request
95           RECEIVE_HANDSHAKE_RESPONSE,                 // Awaiting SaslHandshake response from server when authenticating
96           INITIAL,                                    // Initial authentication state starting SASL token exchange for configured mechanism, send first token
97           INTERMEDIATE,                               // Intermediate state during SASL token exchange, process challenges and send responses
98           CLIENT_COMPLETE,                            // Sent response to last challenge. If using SaslAuthenticate, wait for authentication status from server, else COMPLETE
99           COMPLETE,                                   // Authentication sequence complete. If using SaslAuthenticate, this state implies successful authentication.
100           FAILED,                                     // Failed authentication due to an error at some stage
101           REAUTH_PROCESS_ORIG_APIVERSIONS_RESPONSE,   // Initial state for re-authentication: process ApiVersionsResponse from original authentication
102           REAUTH_SEND_HANDSHAKE_REQUEST,              // Processed original ApiVersionsResponse, send SaslHandshake request as part of re-authentication
103           REAUTH_RECEIVE_HANDSHAKE_OR_OTHER_RESPONSE, // Awaiting SaslHandshake response from server when re-authenticating, and may receive other, in-flight responses sent prior to start of re-authentication as well
104           REAUTH_INITIAL,                             // Initial re-authentication state starting SASL token exchange for configured mechanism, send first token
105       }

107       private static final short DISABLE_KAFKA_SASL_AUTHENTICATE_HEADER = -1;
108       private static final Random RNG = new Random();

110       /**
111        * the reserved range of correlation id for Sasl requests.
112        *
113        * Noted: there is a story about reserved range. The response of LIST_OFFSET is compatible to response of SASL_HANDSHAKE.
114        * Hence, we could miss the schema error when using schema of SASL_HANDSHAKE to parse response of LIST_OFFSET.
115        * For example: the IllegalStateException caused by mismatched correlation id is thrown if following steps happens.
116        * 1) sent LIST_OFFSET
117        * 2) sent SASL_HANDSHAKE
118        * 3) receive response of LIST_OFFSET
119        * 4) succeed to use schema of SASL_HANDSHAKE to parse response of LIST_OFFSET
120        * 5) throw IllegalStateException due to mismatched correlation id
121        * As a simple approach, we force Sasl requests to use a reserved correlation id which is separated from those
122        * used in NetworkClient for Kafka requests. Hence, we can guarantee that every SASL request will throw
123        * SchemaException due to correlation id mismatch during reauthentication
124        */
125       public static final int MAX_RESERVED_CORRELATION_ID = Integer.MAX_VALUE;
126   
127       /**
128        * We only expect one request in-flight a time during authentication so the small range is fine.
129        */
130       public static final int MIN_RESERVED_CORRELATION_ID = MAX_RESERVED_CORRELATION_ID - 7;
131   
132       /**
133        * @return true if the correlation id is reserved for SASL request. otherwise, false
134        */
135       public static boolean isReserved(int correlationId) {
136           return correlationId >= MIN_RESERVED_CORRELATION_ID;
137       }

230       /**
231        * Sends an empty message to the server to initiate the authentication process. It then evaluates server challenges
232        * via `SaslClient.evaluateChallenge` and returns client responses until authentication succeeds or fails.
233        *
234        * The messages are sent and received as size delimited bytes that consists of a 4 byte network-ordered size N
235        * followed by N bytes representing the opaque payload.
236        */
237       @SuppressWarnings("fallthrough")
238       public void authenticate() throws IOException {
239           if (netOutBuffer != null && !flushNetOutBufferAndUpdateInterestOps())
240               return;
241   
242           switch (saslState) {
243               case SEND_APIVERSIONS_REQUEST:
244                   // Always use version 0 request since brokers treat requests with schema exceptions as GSSAPI tokens
245                   ApiVersionsRequest apiVersionsRequest = new ApiVersionsRequest.Builder().build((short) 0);
246                   send(apiVersionsRequest.toSend(nextRequestHeader(ApiKeys.API_VERSIONS, apiVersionsRequest.version())));
247                   setSaslState(SaslState.RECEIVE_APIVERSIONS_RESPONSE);
248                   break;
249               case RECEIVE_APIVERSIONS_RESPONSE:
250                   ApiVersionsResponse apiVersionsResponse = (ApiVersionsResponse) receiveKafkaResponse();
251                   if (apiVersionsResponse == null)
252                       break;
253                   else {
254                       setSaslAuthenticateAndHandshakeVersions(apiVersionsResponse);
255                       reauthInfo.apiVersionsResponseReceivedFromBroker = apiVersionsResponse;
256                       setSaslState(SaslState.SEND_HANDSHAKE_REQUEST);
257                       // Fall through to send handshake request with the latest supported version
258                   }
259               case SEND_HANDSHAKE_REQUEST:
260                   sendHandshakeRequest(saslHandshakeVersion);
261                   setSaslState(SaslState.RECEIVE_HANDSHAKE_RESPONSE);
262                   break;
263               case RECEIVE_HANDSHAKE_RESPONSE:
264                   SaslHandshakeResponse handshakeResponse = (SaslHandshakeResponse) receiveKafkaResponse();
265                   if (handshakeResponse == null)
266                       break;
267                   else {
268                       handleSaslHandshakeResponse(handshakeResponse);
269                       setSaslState(SaslState.INITIAL);
270                       // Fall through and start SASL authentication using the configured client mechanism
271                   }
272               case INITIAL:
273                   sendInitialToken();
274                   setSaslState(SaslState.INTERMEDIATE);
275                   break;
276               case REAUTH_PROCESS_ORIG_APIVERSIONS_RESPONSE:
277                   setSaslAuthenticateAndHandshakeVersions(reauthInfo.apiVersionsResponseFromOriginalAuthentication);
278                   setSaslState(SaslState.REAUTH_SEND_HANDSHAKE_REQUEST); // Will set immediately
279                   // Fall through to send handshake request with the latest supported version
280               case REAUTH_SEND_HANDSHAKE_REQUEST:
281                   sendHandshakeRequest(saslHandshakeVersion);
282                   setSaslState(SaslState.REAUTH_RECEIVE_HANDSHAKE_OR_OTHER_RESPONSE);
283                   break;
284               case REAUTH_RECEIVE_HANDSHAKE_OR_OTHER_RESPONSE:
285                   handshakeResponse = (SaslHandshakeResponse) receiveKafkaResponse();
286                   if (handshakeResponse == null)
287                       break;
288                   handleSaslHandshakeResponse(handshakeResponse);
289                   setSaslState(SaslState.REAUTH_INITIAL); // Will set immediately
290                   /*
291                    * Fall through and start SASL authentication using the configured client
292                    * mechanism. Note that we have to either fall through or add a loop to enter
293                    * the switch statement again. We will fall through to avoid adding the loop and
294                    * therefore minimize the changes to authentication-related code due to the
295                    * changes related to re-authentication.
296                    */
297               case REAUTH_INITIAL:
298                   sendInitialToken();
299                   setSaslState(SaslState.INTERMEDIATE);
300                   break;
301               case INTERMEDIATE:
302                   byte[] serverToken = receiveToken();
303                   boolean noResponsesPending = serverToken != null && !sendSaslClientToken(serverToken, false);
304                   // For versions without SASL_AUTHENTICATE header, SASL exchange may be complete after a token is sent to server.
305                   // For versions with SASL_AUTHENTICATE header, server always sends a response to each SASL_AUTHENTICATE request.
306                   if (saslClient.isComplete()) {
307                       if (saslAuthenticateVersion == DISABLE_KAFKA_SASL_AUTHENTICATE_HEADER || noResponsesPending)
308                           setSaslState(SaslState.COMPLETE);
309                       else
310                           setSaslState(SaslState.CLIENT_COMPLETE);
311                   }
312                   break;
313               case CLIENT_COMPLETE:
314                   byte[] serverResponse = receiveToken();
315                   if (serverResponse != null)
316                       setSaslState(SaslState.COMPLETE);
317                   break;
318               case COMPLETE:
319                   break;
320               case FAILED:
321                   // Should never get here since exception would have been propagated earlier
322                   throw new IllegalStateException("SASL handshake has already failed");
323           }
324       }

372       private RequestHeader nextRequestHeader(ApiKeys apiKey, short version) {
373           String clientId = (String) configs.get(CommonClientConfigs.CLIENT_ID_CONFIG);
374           short requestApiKey = apiKey.id;
375           currentRequestHeader = new RequestHeader(
376               new RequestHeaderData().
377                   setRequestApiKey(requestApiKey).
378                   setRequestApiVersion(version).
379                   setClientId(clientId).
380                   setCorrelationId(nextCorrelationId()),
381               apiKey.requestHeaderVersion(version));
382           return currentRequestHeader;
383       }

391       // Visible to override for testing

445                   send(send);

448           }

474           if (netInBuffer == null) netInBuffer = new NetworkReceive(node);
475           netInBuffer.readFrom(transportLayer);

482           }
483           return serverPacket;

526       private byte[] createSaslToken(final byte[] saslToken, boolean isInitial) throws SaslException {
527           if (saslToken == null)
528               throw new IllegalSaslStateException("Error authenticating with the Kafka Broker: received a `null` saslToken.");
529   
530           try {
531               if (isInitial && !saslClient.hasInitialResponse())
532                   return saslToken;
533               else
534                   return Subject.doAs(subject, (PrivilegedExceptionAction<byte[]>) () -> saslClient.evaluateChallenge(saslToken));
535           } catch (PrivilegedActionException e) {
536               String error = "An error: (" + e + ") occurred when evaluating SASL token received from the Kafka Broker.";
537               KerberosError kerberosError = KerberosError.fromException(e);
538               // Try to provide hints to use about what went wrong so they can fix their configuration.
539               if (kerberosError == KerberosError.SERVER_NOT_FOUND) {
540                   error += " This may be caused by Java's being unable to resolve the Kafka Broker's" +
541                       " hostname correctly. You may want to try to adding" +
542                       " '-Dsun.net.spi.nameservice.provider.1=dns,sun' to your client's JVMFLAGS environment." +
543                       " Users must configure FQDN of kafka brokers when authenticating using SASL and" +
544                       " `socketChannel.socket().getInetAddress().getHostName()` must match the hostname in `principal/hostname@realm`";
545               }
546               //Unwrap the SaslException inside `PrivilegedActionException`
547               Throwable cause = e.getCause();
548               // Treat transient Kerberos errors as non-fatal SaslExceptions that are processed as I/O exceptions
549               // and all other failures as fatal SaslAuthenticationException.
550               if ((kerberosError != null && kerberosError.retriable()) || (kerberosError == null && KerberosError.isRetriableClientGssException(e))) {
551                   error += " Kafka Client will retry.";
552                   throw new SaslException(error, cause);
553               } else {
554                   error += " Kafka Client will go to AUTHENTICATION_FAILED state.";
555                   throw new SaslAuthenticationException(error, cause);
556               }
557           }
558       }

567       private AbstractResponse receiveKafkaResponse() throws IOException {
568           if (netInBuffer == null)
569               netInBuffer = new NetworkReceive(node);
570           NetworkReceive receive = netInBuffer;
571           try {
572               byte[] responseBytes = receiveResponseOrToken();
573               if (responseBytes == null)
574                   return null;
575               else {
576                   AbstractResponse response = NetworkClient.parseResponse(ByteBuffer.wrap(responseBytes), currentRequestHeader);
577                   currentRequestHeader = null;
578                   return response;
579               }
580           } catch (BufferUnderflowException | SchemaException | IllegalArgumentException e) {
581               /*
582                * Account for the fact that during re-authentication there may be responses
583                * arriving for requests that were sent in the past.
584                */
585               if (reauthInfo.reauthenticating()) {
586                   /*
587                    * It didn't match the current request header, so it must be unrelated to
588                    * re-authentication. Save it so it can be processed later.
589                    */
590                   receive.payload().rewind();
591                   reauthInfo.pendingAuthenticatedReceives.add(receive);
592                   return null;
593               }
594               log.debug("Invalid SASL mechanism response, server may be expecting only GSSAPI tokens");
595               setSaslState(SaslState.FAILED);
596               throw new IllegalSaslStateException("Invalid SASL mechanism response, server may be expecting a different protocol", e);
597           }
598       }

600       private void handleSaslHandshakeResponse(SaslHandshakeResponse response) {
601           Errors error = response.error();
602           if (error != Errors.NONE)
603               setSaslState(SaslState.FAILED);
604           switch (error) {
605               case NONE:
606                   break;
607               case UNSUPPORTED_SASL_MECHANISM:
608                   throw new UnsupportedSaslMechanismException(String.format("Client SASL mechanism '%s' not enabled in the server, enabled mechanisms are %s",
609                       mechanism, response.enabledMechanisms()));
610               case ILLEGAL_SASL_STATE:
611                   throw new IllegalSaslStateException(String.format("Unexpected handshake request with client mechanism %s, enabled mechanisms are %s",
612                       mechanism, response.enabledMechanisms()));
613               default:
614                   throw new IllegalSaslStateException(String.format("Unknown error code %s, client mechanism is %s, enabled mechanisms are %s",
615                       response.error(), mechanism, response.enabledMechanisms()));
616           }
617       }

633           }

682           public void setAuthenticationEndAndSessionReauthenticationTimes(long nowNanos) {
683               authenticationEndNanos = nowNanos;
684               long sessionLifetimeMsToUse = 0;
685               if (positiveSessionLifetimeMs != null) {
686                   // pick a random percentage between 85% and 95% for session re-authentication
687                   double pctWindowFactorToTakeNetworkLatencyAndClockDriftIntoAccount = 0.85;
688                   double pctWindowJitterToAvoidReauthenticationStormAcrossManyChannelsSimultaneously = 0.10;
689                   double pctToUse = pctWindowFactorToTakeNetworkLatencyAndClockDriftIntoAccount + RNG.nextDouble()
690                           * pctWindowJitterToAvoidReauthenticationStormAcrossManyChannelsSimultaneously;
691                   sessionLifetimeMsToUse = (long) (positiveSessionLifetimeMs * pctToUse);
692                   clientSessionReauthenticationTimeNanos = authenticationEndNanos + 1000 * 1000 * sessionLifetimeMsToUse;
693                   log.debug(
694                           "Finished {} with session expiration in {} ms and session re-authentication on or after {} ms",
695                           authenticationOrReauthenticationText(), positiveSessionLifetimeMs, sessionLifetimeMsToUse);
696               } else
697                   log.debug("Finished {} with no session expiration and no session re-authentication",
698                           authenticationOrReauthenticationText());
699           }
