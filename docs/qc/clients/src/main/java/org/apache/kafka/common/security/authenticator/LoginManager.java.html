55       private LoginManager(JaasContext jaasContext, String saslMechanism, Map<String, ?> configs,
56                            LoginMetadata<?> loginMetadata) throws LoginException {
57           this.loginMetadata = loginMetadata;
58           this.login = Utils.newInstance(loginMetadata.loginClass);
59           loginCallbackHandler = Utils.newInstance(loginMetadata.loginCallbackClass);
60           loginCallbackHandler.configure(configs, saslMechanism, jaasContext.configurationEntries());
61           login.configure(configs, jaasContext.name(), jaasContext.configuration(), loginCallbackHandler);
62           login.login();
63       }
64   
65       /**
66        * Returns an instance of `LoginManager` and increases its reference count.
67        *
68        * `release()` should be invoked when the `LoginManager` is no longer needed. This method will try to reuse an
69        * existing `LoginManager` for the provided context type. If `jaasContext` was loaded from a dynamic config,
70        * login managers are reused for the same dynamic config value. For `jaasContext` loaded from static JAAS
71        * configuration, login managers are reused for static contexts with the same login context name.
72        *
73        * This is a bit ugly and it would be nicer if we could pass the `LoginManager` to `ChannelBuilders.create` and
74        * shut it down when the broker or clients are closed. It's straightforward to do the former, but it's more
75        * complicated to do the latter without making the consumer API more complex.
76        *
77        * @param jaasContext Static or dynamic JAAS context. `jaasContext.dynamicJaasConfig()` is non-null for dynamic context.
78        *                    For static contexts, this may contain multiple login modules if the context type is SERVER.
79        *                    For CLIENT static contexts and dynamic contexts of CLIENT and SERVER, 'jaasContext` contains
80        *                    only one login module.
81        * @param saslMechanism SASL mechanism for which login manager is being acquired. For dynamic contexts, the single
82        *                      login module in `jaasContext` corresponds to this SASL mechanism. Hence `Login` class is
83        *                      chosen based on this mechanism.
84        * @param defaultLoginClass Default login class to use if an override is not specified in `configs`
85        * @param configs Config options used to configure `Login` if a new login manager is created.
86        *
87        */
88       public static LoginManager acquireLoginManager(JaasContext jaasContext, String saslMechanism,
89                                                      Class<? extends Login> defaultLoginClass,
90                                                      Map<String, ?> configs) throws LoginException {
91           Class<? extends Login> loginClass = configuredClassOrDefault(configs, jaasContext,
92                   saslMechanism, SaslConfigs.SASL_LOGIN_CLASS, defaultLoginClass);
93           Class<? extends AuthenticateCallbackHandler> defaultLoginCallbackHandlerClass = OAuthBearerLoginModule.OAUTHBEARER_MECHANISM
94                   .equals(saslMechanism) ? OAuthBearerUnsecuredLoginCallbackHandler.class
95                           : AbstractLogin.DefaultLoginCallbackHandler.class;
96           Class<? extends AuthenticateCallbackHandler> loginCallbackClass = configuredClassOrDefault(configs, jaasContext,
97                   saslMechanism, SaslConfigs.SASL_LOGIN_CALLBACK_HANDLER_CLASS, defaultLoginCallbackHandlerClass);
98           synchronized (LoginManager.class) {
99               LoginManager loginManager;
100               Password jaasConfigValue = jaasContext.dynamicJaasConfig();
101               if (jaasConfigValue != null) {
102                   LoginMetadata<Password> loginMetadata = new LoginMetadata<>(jaasConfigValue, loginClass, loginCallbackClass);
103                   loginManager = DYNAMIC_INSTANCES.get(loginMetadata);
104                   if (loginManager == null) {
105                       loginManager = new LoginManager(jaasContext, saslMechanism, configs, loginMetadata);
106                       DYNAMIC_INSTANCES.put(loginMetadata, loginManager);
107                   }
108               } else {
109                   LoginMetadata<String> loginMetadata = new LoginMetadata<>(jaasContext.name(), loginClass, loginCallbackClass);
110                   loginManager = STATIC_INSTANCES.get(loginMetadata);
111                   if (loginManager == null) {
112                       loginManager = new LoginManager(jaasContext, saslMechanism, configs, loginMetadata);
113                       STATIC_INSTANCES.put(loginMetadata, loginManager);
114                   }
115               }
116               SecurityUtils.addConfiguredSecurityProviders(configs);
117               return loginManager.acquire();
118           }
119       }

134       private LoginManager acquire() {
135           ++refCount;
136           LOGGER.trace("{} acquired", this);
137           return this;
138       }
139   
140       /**
141        * Decrease the reference count for this instance and release resources if it reaches 0.
142        */
143       public void release() {
144           synchronized (LoginManager.class) {
145               if (refCount == 0)
146                   throw new IllegalStateException("release() called on disposed " + this);
147               else if (refCount == 1) {
148                   if (loginMetadata.configInfo instanceof Password) {
149                       DYNAMIC_INSTANCES.remove(loginMetadata);
150                   } else {
151                       STATIC_INSTANCES.remove(loginMetadata);
152                   }
153                   login.close();
154                   loginCallbackHandler.close();
155               }
156               --refCount;
157               LOGGER.trace("{} released", this);
158           }
159       }

161       @Override
162       public String toString() {
163           return "LoginManager(serviceName=" + serviceName() +
164                   // subject.toString() exposes private credentials, so we can't use it
165                   ", publicCredentials=" + subject().getPublicCredentials() +
166                   ", refCount=" + refCount + ')';
167       }

169       /* Should only be used in tests. */
170       public static void closeAll() {
171           synchronized (LoginManager.class) {
172               for (LoginMetadata<String> key : new ArrayList<>(STATIC_INSTANCES.keySet()))
173                   STATIC_INSTANCES.remove(key).login.close();
174               for (LoginMetadata<Password> key : new ArrayList<>(DYNAMIC_INSTANCES.keySet()))
175                   DYNAMIC_INSTANCES.remove(key).login.close();
176           }
177       }

179       private static <T> Class<? extends T> configuredClassOrDefault(Map<String, ?> configs,
180                                                        JaasContext jaasContext,
181                                                        String saslMechanism,
182                                                        String configName,
183                                                        Class<? extends T> defaultClass) {
184           String prefix  = jaasContext.type() == JaasContext.Type.SERVER ? ListenerName.saslMechanismPrefix(saslMechanism) : "";
185           @SuppressWarnings("unchecked")
186           Class<? extends T> clazz = (Class<? extends T>) configs.get(prefix + configName);
187           if (clazz != null && jaasContext.configurationEntries().size() != 1) {
188               String errorMessage = configName + " cannot be specified with multiple login modules in the JAAS context. " +
189                       SaslConfigs.SASL_JAAS_CONFIG + " must be configured to override mechanism-specific configs.";
190               throw new ConfigException(errorMessage);
191           }
192           if (clazz == null)
193               clazz = defaultClass;
194           return clazz;
195       }

197       private static class LoginMetadata<T> {
198           final T configInfo;
199           final Class<? extends Login> loginClass;
200           final Class<? extends AuthenticateCallbackHandler> loginCallbackClass;
201   
202           LoginMetadata(T configInfo, Class<? extends Login> loginClass,
203                         Class<? extends AuthenticateCallbackHandler> loginCallbackClass) {
204               this.configInfo = configInfo;
205               this.loginClass = loginClass;
206               this.loginCallbackClass = loginCallbackClass;
207           }
208   
209           @Override
210           public int hashCode() {
211               return Objects.hash(configInfo, loginClass, loginCallbackClass);
212           }
213   
214           @Override
215           public boolean equals(Object o) {
216               if (this == o) return true;
217               if (o == null || getClass() != o.getClass()) return false;
218   
219               LoginMetadata<?> loginMetadata = (LoginMetadata<?>) o;
220               return Objects.equals(configInfo, loginMetadata.configInfo) &&
221                      Objects.equals(loginClass, loginMetadata.loginClass) &&
222                      Objects.equals(loginCallbackClass, loginMetadata.loginCallbackClass);
223           }
224       }
