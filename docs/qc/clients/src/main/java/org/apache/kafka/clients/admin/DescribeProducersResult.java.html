19   import org.apache.kafka.common.KafkaException;
20   import org.apache.kafka.common.KafkaFuture;
21   import org.apache.kafka.common.TopicPartition;
22   import org.apache.kafka.common.annotation.InterfaceStability;
23   
24   import java.util.HashMap;
25   import java.util.List;
26   import java.util.Map;
27   import java.util.concurrent.ExecutionException;
28   
29   @InterfaceStability.Evolving
30   public class DescribeProducersResult {
31   
32       private final Map<TopicPartition, KafkaFuture<PartitionProducerState>> futures;
33   
34       DescribeProducersResult(Map<TopicPartition, KafkaFuture<PartitionProducerState>> futures) {
35           this.futures = futures;
36       }
37   
38       public KafkaFuture<PartitionProducerState> partitionResult(final TopicPartition partition) {
39           KafkaFuture<PartitionProducerState> future = futures.get(partition);
40           if (future == null) {
41               throw new IllegalArgumentException("Topic partition " + partition +
42                   " was not included in the request");
43           }
44           return future;
45       }
46   
47       public KafkaFuture<Map<TopicPartition, PartitionProducerState>> all() {
48           return KafkaFuture.allOf(futures.values().toArray(new KafkaFuture[0]))
49               .thenApply(nil -> {
50                   Map<TopicPartition, PartitionProducerState> results = new HashMap<>(futures.size());
51                   for (Map.Entry<TopicPartition, KafkaFuture<PartitionProducerState>> entry : futures.entrySet()) {
52                       try {
53                           results.put(entry.getKey(), entry.getValue().get());
54                       } catch (InterruptedException | ExecutionException e) {
55                           // This should be unreachable, because allOf ensured that all the futures completed successfully.
56                           throw new KafkaException(e);
57                       }
58                   }
59                   return results;
60               });
61       }
62   
63       public static class PartitionProducerState {
64           private final List<ProducerState> activeProducers;
65   
66           public PartitionProducerState(List<ProducerState> activeProducers) {
67               this.activeProducers = activeProducers;
68           }
69   
70           public List<ProducerState> activeProducers() {
71               return activeProducers;
72           }
73   
74           @Override
75           public String toString() {
76               return "PartitionProducerState(" +
77                   "activeProducers=" + activeProducers +
78                   ')';
79           }
80       }
