51       public static List<InetSocketAddress> parseAndValidateAddresses(List<String> urls, ClientDnsLookup clientDnsLookup) {
52           List<InetSocketAddress> addresses = new ArrayList<>();
53           for (String url : urls) {
54               if (url != null && !url.isEmpty()) {
55                   try {
56                       String host = getHost(url);
57                       Integer port = getPort(url);
58                       if (host == null || port == null)
59                           throw new ConfigException("Invalid url in " + CommonClientConfigs.BOOTSTRAP_SERVERS_CONFIG + ": " + url);
60   
61                       if (clientDnsLookup == ClientDnsLookup.RESOLVE_CANONICAL_BOOTSTRAP_SERVERS_ONLY) {
62                           InetAddress[] inetAddresses = InetAddress.getAllByName(host);
63                           for (InetAddress inetAddress : inetAddresses) {
64                               String resolvedCanonicalName = inetAddress.getCanonicalHostName();
65                               InetSocketAddress address = new InetSocketAddress(resolvedCanonicalName, port);
66                               if (address.isUnresolved()) {
67                                   log.warn("Couldn't resolve server {} from {} as DNS resolution of the canonical hostname {} failed for {}", url, CommonClientConfigs.BOOTSTRAP_SERVERS_CONFIG, resolvedCanonicalName, host);
68                               } else {
69                                   addresses.add(address);
70                               }
71                           }
72                       } else {
73                           InetSocketAddress address = new InetSocketAddress(host, port);
74                           if (address.isUnresolved()) {
75                               log.warn("Couldn't resolve server {} from {} as DNS resolution failed for {}", url, CommonClientConfigs.BOOTSTRAP_SERVERS_CONFIG, host);
76                           } else {
77                               addresses.add(address);
78                           }
79                       }
80   
81                   } catch (IllegalArgumentException e) {
82                       throw new ConfigException("Invalid port in " + CommonClientConfigs.BOOTSTRAP_SERVERS_CONFIG + ": " + url);
83                   } catch (UnknownHostException e) {
84                       throw new ConfigException("Unknown host in " + CommonClientConfigs.BOOTSTRAP_SERVERS_CONFIG + ": " + url);
85                   }
86               }
87           }
88           if (addresses.isEmpty())
89               throw new ConfigException("No resolvable bootstrap urls given in " + CommonClientConfigs.BOOTSTRAP_SERVERS_CONFIG);
90           return addresses;
91       }

93       /**
94        * Create a new channel builder from the provided configuration.
95        *
96        * @param config client configs
97        * @param time the time implementation
98        * @param logContext the logging context
99        *
100        * @return configured ChannelBuilder based on the configs.
101        */
102       public static ChannelBuilder createChannelBuilder(AbstractConfig config, Time time, LogContext logContext) {
103           SecurityProtocol securityProtocol = SecurityProtocol.forName(config.getString(CommonClientConfigs.SECURITY_PROTOCOL_CONFIG));
104           String clientSaslMechanism = config.getString(SaslConfigs.SASL_MECHANISM);
105           return ChannelBuilders.clientChannelBuilder(securityProtocol, JaasContext.Type.CLIENT, config, null,
106                   clientSaslMechanism, time, true, logContext);
107       }

109       static List<InetAddress> resolve(String host, HostResolver hostResolver) throws UnknownHostException {
110           InetAddress[] addresses = hostResolver.resolve(host);
111           List<InetAddress> result = filterPreferredAddresses(addresses);
112           if (log.isDebugEnabled())
113               log.debug("Resolved host {} as {}", host, result.stream().map(i -> i.getHostAddress()).collect(Collectors.joining(",")));
114           return result;
115       }

117       /**
118        * Return a list containing the first address in `allAddresses` and subsequent addresses
119        * that are a subtype of the first address.
120        *
121        * The outcome is that all returned addresses are either IPv4 or IPv6 (InetAddress has two
122        * subclasses: Inet4Address and Inet6Address).
123        */
124       static List<InetAddress> filterPreferredAddresses(InetAddress[] allAddresses) {
125           List<InetAddress> preferredAddresses = new ArrayList<>();
126           Class<? extends InetAddress> clazz = null;
127           for (InetAddress address : allAddresses) {
128               if (clazz == null) {
129                   clazz = address.getClass();
130               }
131               if (clazz.isInstance(address)) {
132                   preferredAddresses.add(address);
133               }
134           }
135           return preferredAddresses;
136       }
