18   package org.apache.kafka.common.acl;
19   
20   import org.apache.kafka.common.annotation.InterfaceStability;
21   
22   import java.util.Objects;
23   
24   /**
25    * Represents a filter which matches access control entries.
26    *
27    * The API for this class is still evolving and we may break compatibility in minor releases, if necessary.
28    */
29   @InterfaceStability.Evolving
30   public class AccessControlEntryFilter {
31       private final AccessControlEntryData data;
32   
33       /**
34        * Matches any access control entry.
35        */
36       public static final AccessControlEntryFilter ANY =
37           new AccessControlEntryFilter(null, null, AclOperation.ANY, AclPermissionType.ANY);
38   
39       /**
40        * Create an instance of an access control entry filter with the provided parameters.
41        *
42        * @param principal the principal or null
43        * @param host the host or null
44        * @param operation non-null operation
45        * @param permissionType non-null permission type
46        */
47       public AccessControlEntryFilter(String principal, String host, AclOperation operation, AclPermissionType permissionType) {
48           Objects.requireNonNull(operation);
49           Objects.requireNonNull(permissionType);
50           this.data = new AccessControlEntryData(principal, host, operation, permissionType);
51       }
52   
53       /**
54        * This is a non-public constructor used in AccessControlEntry#toFilter
55        *
56        * @param data     The access control data.
57        */
58       AccessControlEntryFilter(AccessControlEntryData data) {
59           this.data = data;
60       }
61   
62       /**
63        * Return the principal or null.
64        */
65       public String principal() {
66           return data.principal();
67       }
68   
69       /**
70        * Return the host or null. The value `*` means any host.
71        */
72       public String host() {
73           return data.host();
74       }
75   
76       /**
77        * Return the AclOperation.
78        */
79       public AclOperation operation() {
80           return data.operation();
81       }
82   
83       /**
84        * Return the AclPermissionType.
85        */
86       public AclPermissionType permissionType() {
87           return data.permissionType();
88       }
89   
90       @Override
91       public String toString() {
92           return data.toString();
93       }
94   
95       /**
96        * Return true if there are any UNKNOWN components.
97        */
98       public boolean isUnknown() {
99           return data.isUnknown();
100       }
101   
102       /**
103        * Returns true if this filter matches the given AccessControlEntry.
104        */
105       public boolean matches(AccessControlEntry other) {
106           if ((principal() != null) && (!principal().equals(other.principal())))
107               return false;
108           if ((host() != null) && (!host().equals(other.host())))
109               return false;
110           if ((operation() != AclOperation.ANY) && (!operation().equals(other.operation())))
111               return false;
112           return (permissionType() == AclPermissionType.ANY) || (permissionType().equals(other.permissionType()));
113       }
114   
115       /**
116        * Returns true if this filter could only match one ACE -- in other words, if
117        * there are no ANY or UNKNOWN fields.
118        */
119       public boolean matchesAtMostOne() {
120           return findIndefiniteField() == null;
121       }
122   
123       /**
124        * Returns a string describing an ANY or UNKNOWN field, or null if there is
125        * no such field.
126        */
127       public String findIndefiniteField() {
128           return data.findIndefiniteField();
129       }
130   
131       @Override
132       public boolean equals(Object o) {
133           if (!(o instanceof AccessControlEntryFilter))
134               return false;
135           AccessControlEntryFilter other = (AccessControlEntryFilter) o;
136           return data.equals(other.data);
137       }
138   
139       @Override
140       public int hashCode() {
141           return data.hashCode();
142       }
143   }
