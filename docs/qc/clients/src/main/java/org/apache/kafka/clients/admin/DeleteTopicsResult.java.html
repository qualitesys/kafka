18   package org.apache.kafka.clients.admin;
19   
20   import org.apache.kafka.common.KafkaFuture;
21   import org.apache.kafka.common.TopicCollection;
22   import org.apache.kafka.common.Uuid;
23   import org.apache.kafka.common.annotation.InterfaceStability;
24   
25   import java.util.Collection;
26   import java.util.Map;
27   
28   /**
29    * The result of the {@link Admin#deleteTopics(Collection)} call.
30    *
31    * The API of this class is evolving, see {@link Admin} for details.
32    */
33   @InterfaceStability.Evolving
34   public class DeleteTopicsResult {
35       private final Map<Uuid, KafkaFuture<Void>> topicIdFutures;
36       private final Map<String, KafkaFuture<Void>> nameFutures;
37   
38       protected DeleteTopicsResult(Map<Uuid, KafkaFuture<Void>> topicIdFutures, Map<String, KafkaFuture<Void>> nameFutures) {
39           if (topicIdFutures != null && nameFutures != null)
40               throw new IllegalArgumentException("topicIdFutures and nameFutures cannot both be specified.");
41           if (topicIdFutures == null && nameFutures == null)
42               throw new IllegalArgumentException("topicIdFutures and nameFutures cannot both be null.");
43           this.topicIdFutures = topicIdFutures;
44           this.nameFutures = nameFutures;
45       }
46   
47       static DeleteTopicsResult ofTopicIds(Map<Uuid, KafkaFuture<Void>> topicIdFutures) {
48           return new DeleteTopicsResult(topicIdFutures, null);
49       }
50   
51       static DeleteTopicsResult ofTopicNames(Map<String, KafkaFuture<Void>> nameFutures) {
52           return new DeleteTopicsResult(null, nameFutures);
53       }
54   
55       /**
56        * Use when {@link Admin#deleteTopics(TopicCollection, DeleteTopicsOptions)} used a TopicIdCollection
57        * @return a map from topic IDs to futures which can be used to check the status of
58        * individual deletions if the deleteTopics request used topic IDs. Otherwise return null.
59        */
60       public Map<Uuid, KafkaFuture<Void>> topicIdValues() {
61           return topicIdFutures;
62       }
63   
64       /**
65        * Use when {@link Admin#deleteTopics(TopicCollection, DeleteTopicsOptions)} used a TopicNameCollection
66        * @return a map from topic names to futures which can be used to check the status of
67        * individual deletions if the deleteTopics request used topic names. Otherwise return null.
68        */
69       public Map<String, KafkaFuture<Void>> topicNameValues() {
70           return nameFutures;
71       }
72   
73       /**
74        * @return a map from topic names to futures which can be used to check the status of
75        * individual deletions if the deleteTopics request used topic names. Otherwise return null.
76        * @deprecated Since 3.0 use {@link #topicNameValues} instead
77        */
78       @Deprecated
79       public Map<String, KafkaFuture<Void>> values() {
80           return nameFutures;
81       }
82   
83       /**
84        * @return a future which succeeds only if all the topic deletions succeed.
85        */
86       public KafkaFuture<Void> all() {
87           return (topicIdFutures == null) ? KafkaFuture.allOf(nameFutures.values().toArray(new KafkaFuture[0])) :
88               KafkaFuture.allOf(topicIdFutures.values().toArray(new KafkaFuture[0]));
89       }
90   }
