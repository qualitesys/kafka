18   package org.apache.kafka.common.security.token.delegation.internals;
19   
20   import org.apache.kafka.common.security.authenticator.CredentialCache;
21   import org.apache.kafka.common.security.scram.ScramCredential;
22   import org.apache.kafka.common.security.scram.internals.ScramCredentialUtils;
23   import org.apache.kafka.common.security.scram.internals.ScramMechanism;
24   import org.apache.kafka.common.security.token.delegation.DelegationToken;
25   import org.apache.kafka.common.security.token.delegation.TokenInformation;
26   
27   import java.util.Collection;
28   import java.util.HashMap;
29   import java.util.Map;
30   import java.util.concurrent.ConcurrentHashMap;
31   
32   public class DelegationTokenCache {
33   
34       private CredentialCache credentialCache = new CredentialCache();
35   
36       //Cache to hold all the tokens
37       private Map<String, TokenInformation> tokenCache = new ConcurrentHashMap<>();
38   
39       //Cache to hold hmac->tokenId mapping. This is required for renew, expire requests
40       private Map<String, String> hmacTokenIdCache = new ConcurrentHashMap<>();
41   
42       //Cache to hold tokenId->hmac mapping. This is required for removing entry from hmacTokenIdCache using tokenId.
43       private Map<String, String> tokenIdHmacCache = new ConcurrentHashMap<>();
44   
45       public DelegationTokenCache(Collection<String> scramMechanisms) {
46           //Create caches for scramMechanisms
47           ScramCredentialUtils.createCache(credentialCache, scramMechanisms);
48       }
49   
50       public ScramCredential credential(String mechanism, String tokenId) {
51           CredentialCache.Cache<ScramCredential> cache = credentialCache.cache(mechanism, ScramCredential.class);
52           return cache == null ? null : cache.get(tokenId);
53       }
54   
55       public String owner(String tokenId) {
56           TokenInformation tokenInfo = tokenCache.get(tokenId);
57           return tokenInfo == null ? null : tokenInfo.owner().getName();
58       }
59   
60       public void updateCache(DelegationToken token, Map<String, ScramCredential> scramCredentialMap) {
61           //Update TokenCache
62           String tokenId =  token.tokenInfo().tokenId();
63           addToken(tokenId, token.tokenInfo());
64           String hmac = token.hmacAsBase64String();
65           //Update Scram Credentials
66           updateCredentials(tokenId, scramCredentialMap);
67           //Update hmac-id cache
68           hmacTokenIdCache.put(hmac, tokenId);
69           tokenIdHmacCache.put(tokenId, hmac);
70       }
71   
72       public void removeCache(String tokenId) {
73           removeToken(tokenId);
74           updateCredentials(tokenId, new HashMap<>());
75       }
76   
77       public String tokenIdForHmac(String base64hmac) {
78           return hmacTokenIdCache.get(base64hmac);
79       }
80   
81       public TokenInformation tokenForHmac(String base64hmac) {
82           String tokenId = hmacTokenIdCache.get(base64hmac);
83           return tokenId == null ? null : tokenCache.get(tokenId);
84       }
85   
86       public TokenInformation addToken(String tokenId, TokenInformation tokenInfo) {
87           return tokenCache.put(tokenId, tokenInfo);
88       }
89   
90       public void removeToken(String tokenId) {
91           TokenInformation tokenInfo = tokenCache.remove(tokenId);
92           if (tokenInfo != null) {
93               String hmac = tokenIdHmacCache.remove(tokenInfo.tokenId());
94               if (hmac != null) {
95                   hmacTokenIdCache.remove(hmac);
96               }
97           }
98       }
99   
100       public Collection<TokenInformation> tokens() {
101           return tokenCache.values();
102       }
103   
104       public TokenInformation token(String tokenId) {
105           return tokenCache.get(tokenId);
106       }
107   
108       public CredentialCache.Cache<ScramCredential> credentialCache(String mechanism) {
109           return credentialCache.cache(mechanism, ScramCredential.class);
110       }
111   
112       private void updateCredentials(String tokenId, Map<String, ScramCredential> scramCredentialMap) {
113           for (String mechanism : ScramMechanism.mechanismNames()) {
114               CredentialCache.Cache<ScramCredential> cache = credentialCache.cache(mechanism, ScramCredential.class);
115               if (cache != null) {
116                   ScramCredential credential = scramCredentialMap.get(mechanism);
117                   if (credential == null) {
118                       cache.remove(tokenId);
119                   } else {
120                       cache.put(tokenId, credential);
121                   }
122               }
123           }
124       }
125   }
