41   /**
42    * <p>
43    * <code>OAuthBearerValidatorCallbackHandler</code> is an {@link AuthenticateCallbackHandler} that
44    * accepts {@link OAuthBearerValidatorCallback} and {@link OAuthBearerExtensionsValidatorCallback}
45    * callbacks to implement OAuth/OIDC validation. This callback handler is intended only to be used
46    * on the Kafka broker side as it will receive a {@link OAuthBearerValidatorCallback} that includes
47    * the JWT provided by the Kafka client. That JWT is validated in terms of format, expiration,
48    * signature, and audience and issuer (if desired). This callback handler is the broker side of the
49    * OAuth functionality, whereas {@link OAuthBearerLoginCallbackHandler} is used by clients.
50    * </p>
51    *
52    * <p>
53    * This {@link AuthenticateCallbackHandler} is enabled in the broker configuration by setting the
54    * {@link org.apache.kafka.common.config.internals.BrokerSecurityConfigs#SASL_SERVER_CALLBACK_HANDLER_CLASS}
55    * like so:
56    *
57    * <code>
58    * listener.name.<listener name>.oauthbearer.sasl.server.callback.handler.class=org.apache.kafka.common.security.oauthbearer.secured.OAuthBearerValidatorCallbackHandler
59    * </code>
60    * </p>
61    *
62    * <p>
63    * The JAAS configuration for OAuth is also needed. If using OAuth for inter-broker communication,
64    * the options are those specified in {@link OAuthBearerLoginCallbackHandler}.
65    * </p>
66    *
67    * <p>
68    * The configuration option
69    * {@link org.apache.kafka.common.config.SaslConfigs#SASL_OAUTHBEARER_JWKS_ENDPOINT_URL}
70    * is also required in order to contact the OAuth/OIDC provider to retrieve the JWKS for use in
71    * JWT signature validation. For example:
72    *
73    * <code>
74    * listener.name.<listener name>.oauthbearer.sasl.oauthbearer.jwks.endpoint.url=https://example.com/oauth2/v1/keys
75    * </code>
76    *
77    * Please see the OAuth/OIDC providers documentation for the JWKS endpoint URL.
78    * </p>
79    *
80    * <p>
81    * The following is a list of all the configuration options that are available for the broker
82    * validation callback handler:
83    *
84    * <ul>
85    *   <li>{@link org.apache.kafka.common.config.internals.BrokerSecurityConfigs#SASL_SERVER_CALLBACK_HANDLER_CLASS}</li>
86    *   <li>{@link org.apache.kafka.common.config.SaslConfigs#SASL_JAAS_CONFIG}</li>
87    *   <li>{@link org.apache.kafka.common.config.SaslConfigs#SASL_OAUTHBEARER_CLOCK_SKEW_SECONDS}</li>
88    *   <li>{@link org.apache.kafka.common.config.SaslConfigs#SASL_OAUTHBEARER_EXPECTED_AUDIENCE}</li>
89    *   <li>{@link org.apache.kafka.common.config.SaslConfigs#SASL_OAUTHBEARER_EXPECTED_ISSUER}</li>
90    *   <li>{@link org.apache.kafka.common.config.SaslConfigs#SASL_OAUTHBEARER_JWKS_ENDPOINT_REFRESH_MS}</li>
91    *   <li>{@link org.apache.kafka.common.config.SaslConfigs#SASL_OAUTHBEARER_JWKS_ENDPOINT_RETRY_BACKOFF_MAX_MS}</li>
92    *   <li>{@link org.apache.kafka.common.config.SaslConfigs#SASL_OAUTHBEARER_JWKS_ENDPOINT_RETRY_BACKOFF_MS}</li>
93    *   <li>{@link org.apache.kafka.common.config.SaslConfigs#SASL_OAUTHBEARER_JWKS_ENDPOINT_URL}</li>
94    *   <li>{@link org.apache.kafka.common.config.SaslConfigs#SASL_OAUTHBEARER_SCOPE_CLAIM_NAME}</li>
95    *   <li>{@link org.apache.kafka.common.config.SaslConfigs#SASL_OAUTHBEARER_SUB_CLAIM_NAME}</li>
96    * </ul>
97    * </p>
98    */
99   
100   public class OAuthBearerValidatorCallbackHandler implements AuthenticateCallbackHandler {
101   
102       private static final Logger log = LoggerFactory.getLogger(OAuthBearerValidatorCallbackHandler.class);
103   
104       /**
105        * Because a {@link CloseableVerificationKeyResolver} instance can spawn threads and issue
106        * HTTP(S) calls ({@link RefreshingHttpsJwksVerificationKeyResolver}), we only want to create
107        * a new instance for each particular set of configuration. Because each set of configuration
108        * may have multiple instances, we want to reuse the single instance.
109        */
110   
111       private static final Map<VerificationKeyResolverKey, CloseableVerificationKeyResolver> VERIFICATION_KEY_RESOLVER_CACHE = new HashMap<>();
112   
113       private CloseableVerificationKeyResolver verificationKeyResolver;
114   
115       private AccessTokenValidator accessTokenValidator;
116   
117       private boolean isInitialized = false;
118   
119       @Override
120       public void configure(Map<String, ?> configs, String saslMechanism, List<AppConfigurationEntry> jaasConfigEntries) {
121           Map<String, Object> moduleOptions = JaasOptionsUtils.getOptions(saslMechanism, jaasConfigEntries);
122           CloseableVerificationKeyResolver verificationKeyResolver;
123   
124           // Here's the logic which keeps our VerificationKeyResolvers down to a single instance.
125           synchronized (VERIFICATION_KEY_RESOLVER_CACHE) {
126               VerificationKeyResolverKey key = new VerificationKeyResolverKey(configs, moduleOptions);
127               verificationKeyResolver = VERIFICATION_KEY_RESOLVER_CACHE.computeIfAbsent(key, k ->
128                   new RefCountingVerificationKeyResolver(VerificationKeyResolverFactory.create(configs, saslMechanism, moduleOptions)));
129           }
130   
131           AccessTokenValidator accessTokenValidator = AccessTokenValidatorFactory.create(configs, saslMechanism, verificationKeyResolver);
132           init(verificationKeyResolver, accessTokenValidator);
133       }
134   
135       /*
136        * Package-visible for testing.
137        */
138   
139       void init(CloseableVerificationKeyResolver verificationKeyResolver, AccessTokenValidator accessTokenValidator) {
140           this.verificationKeyResolver = verificationKeyResolver;
141           this.accessTokenValidator = accessTokenValidator;
142   
143           try {
144               verificationKeyResolver.init();
145           } catch (Exception e) {
146               throw new KafkaException("The OAuth validator configuration encountered an error when initializing the VerificationKeyResolver", e);
147           }
148   
149           isInitialized = true;
150       }
151   
152       @Override
153       public void close() {
154           if (verificationKeyResolver != null) {
155               try {
156                   verificationKeyResolver.close();
157               } catch (Exception e) {
158                   log.error(e.getMessage(), e);
159               }
160           }
161       }
162   
163       @Override
164       public void handle(Callback[] callbacks) throws IOException, UnsupportedCallbackException {
165           checkInitialized();
166   
167           for (Callback callback : callbacks) {
168               if (callback instanceof OAuthBearerValidatorCallback) {
169                   handleValidatorCallback((OAuthBearerValidatorCallback) callback);
170               } else if (callback instanceof OAuthBearerExtensionsValidatorCallback) {
171                   handleExtensionsValidatorCallback((OAuthBearerExtensionsValidatorCallback) callback);
172               } else {
173                   throw new UnsupportedCallbackException(callback);
174               }
175           }
176       }
177   
178       private void handleValidatorCallback(OAuthBearerValidatorCallback callback) {
179           checkInitialized();
180   
181           OAuthBearerToken token;
182   
183           try {
184               token = accessTokenValidator.validate(callback.tokenValue());
185               callback.token(token);
186           } catch (ValidateException e) {
187               log.warn(e.getMessage(), e);
188               callback.error("invalid_token", null, null);
189           }
190       }
191   
192       private void handleExtensionsValidatorCallback(OAuthBearerExtensionsValidatorCallback extensionsValidatorCallback) {
193           checkInitialized();
194   
195           extensionsValidatorCallback.inputExtensions().map().forEach((extensionName, v) -> extensionsValidatorCallback.valid(extensionName));
196       }
197   
198       private void checkInitialized() {
199           if (!isInitialized)
200               throw new IllegalStateException(String.format("To use %s, first call the configure or init method", getClass().getSimpleName()));
201       }
202   
203       /**
204        * <code>VkrKey</code> is a simple structure which encapsulates the criteria for different
205        * sets of configuration. This will allow us to use this object as a key in a {@link Map}
206        * to keep a single instance per key.
207        */
208   
209       private static class VerificationKeyResolverKey {
210   
211           private final Map<String, ?> configs;
212   
213           private final Map<String, Object> moduleOptions;
214   
215           public VerificationKeyResolverKey(Map<String, ?> configs, Map<String, Object> moduleOptions) {
216               this.configs = configs;
217               this.moduleOptions = moduleOptions;
218           }
219   
220           @Override
221           public boolean equals(Object o) {
222               if (this == o) {
223                   return true;
224               }
225   
226               if (o == null || getClass() != o.getClass()) {
227                   return false;
228               }
229   
230               VerificationKeyResolverKey that = (VerificationKeyResolverKey) o;
231               return configs.equals(that.configs) && moduleOptions.equals(that.moduleOptions);
232           }
233   
234           @Override
235           public int hashCode() {
236               return Objects.hash(configs, moduleOptions);
237           }
238   
239       }
240   
241       /**
242        * <code>RefCountingVerificationKeyResolver</code> allows us to share a single
243        * {@link CloseableVerificationKeyResolver} instance between multiple
244        * {@link AuthenticateCallbackHandler} instances and perform the lifecycle methods the
245        * appropriate number of times.
246        */
247   
248       private static class RefCountingVerificationKeyResolver implements CloseableVerificationKeyResolver {
249   
250           private final CloseableVerificationKeyResolver delegate;
251   
252           private final AtomicInteger count = new AtomicInteger(0);
253   
254           public RefCountingVerificationKeyResolver(CloseableVerificationKeyResolver delegate) {
255               this.delegate = delegate;
256           }
257   
258           @Override
259           public Key resolveKey(JsonWebSignature jws, List<JsonWebStructure> nestingContext) throws UnresolvableKeyException {
260               return delegate.resolveKey(jws, nestingContext);
261           }
262   
263           @Override
264           public void init() throws IOException {
265               if (count.incrementAndGet() == 1)
266                   delegate.init();
267           }
268   
269           @Override
270           public void close() throws IOException {
271               if (count.decrementAndGet() == 0)
272                   delegate.close();
273           }
274   
275       }
276   
277   }
