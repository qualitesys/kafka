17   package org.apache.kafka.common.network;
18   
19   import java.io.IOException;
20   import java.nio.channels.FileChannel;
21   import java.nio.channels.GatheringByteChannel;
22   
23   /**
24    * Extends GatheringByteChannel with the minimal set of methods required by the Send interface. Supporting TLS and
25    * efficient zero copy transfers are the main reasons for the additional methods.
26    * 
27    * @see SslTransportLayer
28    */
29   public interface TransferableChannel extends GatheringByteChannel {
30   
31       /**
32        * @return true if there are any pending writes. false if the implementation directly write all data to output.
33        */
34       boolean hasPendingWrites();
35   
36       /**
37        * Transfers bytes from `fileChannel` to this `TransferableChannel`.
38        *
39        * This method will delegate to {@link FileChannel#transferTo(long, long, java.nio.channels.WritableByteChannel)},
40        * but it will unwrap the destination channel, if possible, in order to benefit from zero copy. This is required
41        * because the fast path of `transferTo` is only executed if the destination buffer inherits from an internal JDK
42        * class.
43        *
44        * @param fileChannel The source channel
45        * @param position The position within the file at which the transfer is to begin; must be non-negative
46        * @param count The maximum number of bytes to be transferred; must be non-negative
47        * @return The number of bytes, possibly zero, that were actually transferred
48        * @see FileChannel#transferTo(long, long, java.nio.channels.WritableByteChannel)
49        */
50       long transferFrom(FileChannel fileChannel, long position, long count) throws IOException;
51   }
