155       private MetricName createMetricName(String name, String description) {
156           return this.metrics.metricInstance(createTemplate(name, KafkaProducerMetrics.GROUP, description, this.tags));
157       }
158   
159       private MetricNameTemplate createTopicTemplate(String name, String description) {
160           return createTemplate(name, TOPIC_METRIC_GROUP_NAME, description, this.topicTags);
161       }

216       private MetricNameTemplate createTemplate(String name, String group, String description, Set<String> tags) {
217           MetricNameTemplate template = new MetricNameTemplate(name, group, description, tags);
218           this.allTemplates.add(template);
