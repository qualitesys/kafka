17   package org.apache.kafka.common.record;
18   
19   import org.apache.kafka.common.errors.CorruptRecordException;
20   
21   import java.nio.ByteBuffer;
22   
23   import static org.apache.kafka.common.record.Records.HEADER_SIZE_UP_TO_MAGIC;
24   import static org.apache.kafka.common.record.Records.LOG_OVERHEAD;
25   import static org.apache.kafka.common.record.Records.MAGIC_OFFSET;
26   import static org.apache.kafka.common.record.Records.SIZE_OFFSET;
27   
28   /**
29    * A byte buffer backed log input stream. This class avoids the need to copy records by returning
30    * slices from the underlying byte buffer.
31    */
32   class ByteBufferLogInputStream implements LogInputStream<MutableRecordBatch> {
33       private final ByteBuffer buffer;
34       private final int maxMessageSize;
35   
36       ByteBufferLogInputStream(ByteBuffer buffer, int maxMessageSize) {
37           this.buffer = buffer;
38           this.maxMessageSize = maxMessageSize;
39       }
40   
41       public MutableRecordBatch nextBatch() {
42           int remaining = buffer.remaining();
43   
44           Integer batchSize = nextBatchSize();
45           if (batchSize == null || remaining < batchSize)
46               return null;
47   
48           byte magic = buffer.get(buffer.position() + MAGIC_OFFSET);
49   
50           ByteBuffer batchSlice = buffer.slice();
51           batchSlice.limit(batchSize);
52           buffer.position(buffer.position() + batchSize);
53   
54           if (magic > RecordBatch.MAGIC_VALUE_V1)
55               return new DefaultRecordBatch(batchSlice);
56           else
57               return new AbstractLegacyRecordBatch.ByteBufferLegacyRecordBatch(batchSlice);
58       }
59   
60       /**
61        * Validates the header of the next batch and returns batch size.
62        * @return next batch size including LOG_OVERHEAD if buffer contains header up to
63        *         magic byte, null otherwise
64        * @throws CorruptRecordException if record size or magic is invalid
65        */
66       Integer nextBatchSize() throws CorruptRecordException {
67           int remaining = buffer.remaining();
68           if (remaining < LOG_OVERHEAD)
69               return null;
70           int recordSize = buffer.getInt(buffer.position() + SIZE_OFFSET);
71           // V0 has the smallest overhead, stricter checking is done later
72           if (recordSize < LegacyRecord.RECORD_OVERHEAD_V0)
73               throw new CorruptRecordException(String.format("Record size %d is less than the minimum record overhead (%d)",
74                       recordSize, LegacyRecord.RECORD_OVERHEAD_V0));
75           if (recordSize > maxMessageSize)
76               throw new CorruptRecordException(String.format("Record size %d exceeds the largest allowable message size (%d).",
77                       recordSize, maxMessageSize));
78   
79           if (remaining < HEADER_SIZE_UP_TO_MAGIC)
80               return null;
81   
82           byte magic = buffer.get(buffer.position() + MAGIC_OFFSET);
83           if (magic < 0 || magic > RecordBatch.CURRENT_MAGIC_VALUE)
84               throw new CorruptRecordException("Invalid magic found in record: " + magic);
85   
86           return recordSize + LOG_OVERHEAD;
87       }
88   }
