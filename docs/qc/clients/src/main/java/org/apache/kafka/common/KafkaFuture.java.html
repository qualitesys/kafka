17   package org.apache.kafka.common;
18   
19   import org.apache.kafka.common.internals.KafkaFutureImpl;
20   
21   import java.util.Arrays;
22   import java.util.concurrent.CompletableFuture;
23   import java.util.concurrent.CompletionStage;
24   import java.util.concurrent.ExecutionException;
25   import java.util.concurrent.Future;
26   import java.util.concurrent.TimeUnit;
27   import java.util.concurrent.TimeoutException;
28   
29   /**
30    * A flexible future which supports call chaining and other asynchronous programming patterns.
31    *
32    * <h3>Relation to {@code CompletionStage}</h3>
33    * <p>It is possible to obtain a {@code CompletionStage} from a
34    * {@code KafkaFuture} instance by calling {@link #toCompletionStage()}.
35    * If converting {@link KafkaFuture#whenComplete(BiConsumer)} or {@link KafkaFuture#thenApply(BaseFunction)} to
36    * {@link CompletableFuture#whenComplete(java.util.function.BiConsumer)} or
37    * {@link CompletableFuture#thenApply(java.util.function.Function)} be aware that the returned
38    * {@code KafkaFuture} will fail with an {@code ExecutionException}, whereas a {@code CompletionStage} fails
39    * with a {@code CompletionException}.
40    */
41   public abstract class KafkaFuture<T> implements Future<T> {
42       /**
43        * A function which takes objects of type A and returns objects of type B.
44        */
45       @FunctionalInterface
46       public interface BaseFunction<A, B> {
47           B apply(A a);
48       }
49   
50       /**
51        * A function which takes objects of type A and returns objects of type B.
52        *
53        * @deprecated Since Kafka 3.0. Use the {@link BaseFunction} functional interface.
54        */
55       @Deprecated
56       public static abstract class Function<A, B> implements BaseFunction<A, B> { }
57   
58       /**
59        * A consumer of two different types of object.
60        */
61       @FunctionalInterface
62       public interface BiConsumer<A, B> {
63           void accept(A a, B b);
64       }
65   
66       /** 
67        * Returns a new KafkaFuture that is already completed with the given value.
68        */
69       public static <U> KafkaFuture<U> completedFuture(U value) {
70           KafkaFuture<U> future = new KafkaFutureImpl<>();
71           future.complete(value);
72           return future;
73       }
74   
75       /** 
76        * Returns a new KafkaFuture that is completed when all the given futures have completed.  If
77        * any future throws an exception, the returned future returns it.  If multiple futures throw
78        * an exception, which one gets returned is arbitrarily chosen.
79        */
80       public static KafkaFuture<Void> allOf(KafkaFuture<?>... futures) {
81           KafkaFutureImpl<Void> result = new KafkaFutureImpl<>();
82           CompletableFuture.allOf(Arrays.stream(futures)
83                   .map(kafkaFuture -> {
84                       // Safe since KafkaFuture's only subclass is KafkaFuture for which toCompletionStage()
85                       // always return a CF.
86                       return (CompletableFuture<?>) kafkaFuture.toCompletionStage();
87                   })
88                   .toArray(CompletableFuture[]::new)).whenComplete((value, ex) -> {
89                       if (ex == null) {
90                           result.complete(value);
91                       } else {
92                           // Have to unwrap the CompletionException which allOf() introduced
93                           result.completeExceptionally(ex.getCause());
94                       }
95                   });
96   
97           return result;
98       }
99   
100       /**
101        * Gets a {@code CompletionStage} with the same completion properties as this {@code KafkaFuture}.
102        * The returned instance will complete when this future completes and in the same way
103        * (with the same result or exception).
104        *
105        * <p>Calling {@code toCompletableFuture()} on the returned instance will yield a {@code CompletableFuture},
106        * but invocation of the completion methods ({@code complete()} and other methods in the {@code complete*()}
107        * and {@code obtrude*()} families) on that {@code CompletableFuture} instance will result in
108        * {@code UnsupportedOperationException} being thrown. Unlike a "minimal" {@code CompletableFuture},
109        * the {@code get*()} and other methods of {@code CompletableFuture} that are not inherited from
110        * {@code CompletionStage} will work normally.
111        *
112        * <p>If you want to block on the completion of a KafkaFuture you should use
113        * {@link #get()}, {@link #get(long, TimeUnit)} or {@link #getNow(Object)}, rather then calling
114        * {@code .toCompletionStage().toCompletableFuture().get()} etc.
115        *
116        * @since Kafka 3.0
117        */
118       public abstract CompletionStage<T> toCompletionStage();
119   
120       /**
121        * Returns a new KafkaFuture that, when this future completes normally, is executed with this
122        * futures's result as the argument to the supplied function.
123        *
124        * The function may be invoked by the thread that calls {@code thenApply} or it may be invoked by the thread that
125        * completes the future.
126        */
127       public abstract <R> KafkaFuture<R> thenApply(BaseFunction<T, R> function);
128   
129       /**
130        * @see KafkaFuture#thenApply(BaseFunction)
131        *
132        * Prefer {@link KafkaFuture#thenApply(BaseFunction)} as this function is here for backwards compatibility reasons
133        * and might be deprecated/removed in a future release.
134        */
135       public abstract <R> KafkaFuture<R> thenApply(Function<T, R> function);
136   
137       /**
138        * Returns a new KafkaFuture with the same result or exception as this future, that executes the given action
139        * when this future completes.
140        *
141        * When this future is done, the given action is invoked with the result (or null if none) and the exception
142        * (or null if none) of this future as arguments.
143        *
144        * The returned future is completed when the action returns.
145        * The supplied action should not throw an exception. However, if it does, the following rules apply:
146        * if this future completed normally but the supplied action throws an exception, then the returned future completes
147        * exceptionally with the supplied action's exception.
148        * Or, if this future completed exceptionally and the supplied action throws an exception, then the returned future
149        * completes exceptionally with this future's exception.
150        *
151        * The action may be invoked by the thread that calls {@code whenComplete} or it may be invoked by the thread that
152        * completes the future.
153        *
154        * @param action the action to preform
155        * @return the new future
156        */
157       public abstract KafkaFuture<T> whenComplete(BiConsumer<? super T, ? super Throwable> action);
158   
159       /**
160        * If not already completed, sets the value returned by get() and related methods to the given
161        * value.
162        */
163       protected abstract boolean complete(T newValue);
164   
165       /**
166        * If not already completed, causes invocations of get() and related methods to throw the given
167        * exception.
168        */
169       protected abstract boolean completeExceptionally(Throwable newException);
170   
171       /**
172        * If not already completed, completes this future with a CancellationException.  Dependent
173        * futures that have not already completed will also complete exceptionally, with a
174        * CompletionException caused by this CancellationException.
175        */
176       @Override
177       public abstract boolean cancel(boolean mayInterruptIfRunning);
178   
179       /**
180        * Waits if necessary for this future to complete, and then returns its result.
181        */
182       @Override
183       public abstract T get() throws InterruptedException, ExecutionException;
184   
185       /**
186        * Waits if necessary for at most the given time for this future to complete, and then returns
187        * its result, if available.
188        */
189       @Override
190       public abstract T get(long timeout, TimeUnit unit) throws InterruptedException, ExecutionException,
191           TimeoutException;
192   
193       /**
194        * Returns the result value (or throws any encountered exception) if completed, else returns
195        * the given valueIfAbsent.
196        */
197       public abstract T getNow(T valueIfAbsent) throws InterruptedException, ExecutionException;
198   
199       /**
200        * Returns true if this CompletableFuture was cancelled before it completed normally.
201        */
202       @Override
203       public abstract boolean isCancelled();
204   
205       /**
206        * Returns true if this CompletableFuture completed exceptionally, in any way.
207        */
208       public abstract boolean isCompletedExceptionally();
209   
210       /**
211        * Returns true if completed in any fashion: normally, exceptionally, or via cancellation.
212        */
213       @Override
214       public abstract boolean isDone();
215   }
