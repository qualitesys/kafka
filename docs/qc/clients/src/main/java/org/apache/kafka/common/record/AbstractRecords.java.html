30       @Override
31       public boolean hasMatchingMagic(byte magic) {
32           for (RecordBatch batch : batches())
33               if (batch.magic() != magic)
34                   return false;
35           return true;
36       }

38       public RecordBatch firstBatch() {
39           Iterator<? extends RecordBatch> iterator = batches().iterator();
40   
41           if (!iterator.hasNext())
42               return null;
43   
44           return iterator.next();
45       }

61       private Iterator<Record> recordsIterator() {
62           return new AbstractIterator<Record>() {
63               private final Iterator<? extends RecordBatch> batches = batches().iterator();
64               private Iterator<Record> records;
65   
66               @Override
67               protected Record makeNext() {
68                   if (records != null && records.hasNext())
69                       return records.next();
70   
71                   if (batches.hasNext()) {
72                       records = batches.next().iterator();
73                       return makeNext();
74                   }
75   
76                   return allDone();
77               }
78           };
79       }

81       public static int estimateSizeInBytes(byte magic,
82                                             long baseOffset,
83                                             CompressionType compressionType,
84                                             Iterable<Record> records) {
85           int size = 0;
86           if (magic <= RecordBatch.MAGIC_VALUE_V1) {
87               for (Record record : records)
88                   size += Records.LOG_OVERHEAD + LegacyRecord.recordSize(magic, record.key(), record.value());
89           } else {
90               size = DefaultRecordBatch.sizeInBytes(baseOffset, records);
91           }
92           return estimateCompressedSizeInBytes(size, compressionType);
93       }
94   
95       public static int estimateSizeInBytes(byte magic,
96                                             CompressionType compressionType,
97                                             Iterable<SimpleRecord> records) {
98           int size = 0;
99           if (magic <= RecordBatch.MAGIC_VALUE_V1) {
100               for (SimpleRecord record : records)
101                   size += Records.LOG_OVERHEAD + LegacyRecord.recordSize(magic, record.key(), record.value());
102           } else {
103               size = DefaultRecordBatch.sizeInBytes(records);
104           }
105           return estimateCompressedSizeInBytes(size, compressionType);
106       }
107   
108       private static int estimateCompressedSizeInBytes(int size, CompressionType compressionType) {
109           return compressionType == CompressionType.NONE ? size : Math.min(Math.max(size / 2, 1024), 1 << 16);
110       }

112       /**
113        * Get an upper bound estimate on the batch size needed to hold a record with the given fields. This is only
114        * an estimate because it does not take into account overhead from the compression algorithm.
115        */
116       public static int estimateSizeInBytesUpperBound(byte magic, CompressionType compressionType, byte[] key, byte[] value, Header[] headers) {
117           return estimateSizeInBytesUpperBound(magic, compressionType, Utils.wrapNullable(key), Utils.wrapNullable(value), headers);
118       }
119   
120       /**
121        * Get an upper bound estimate on the batch size needed to hold a record with the given fields. This is only
122        * an estimate because it does not take into account overhead from the compression algorithm.
123        */
124       public static int estimateSizeInBytesUpperBound(byte magic, CompressionType compressionType, ByteBuffer key,
125                                                       ByteBuffer value, Header[] headers) {
126           if (magic >= RecordBatch.MAGIC_VALUE_V2)
127               return DefaultRecordBatch.estimateBatchSizeUpperBound(key, value, headers);
128           else if (compressionType != CompressionType.NONE)
129               return Records.LOG_OVERHEAD + LegacyRecord.recordOverhead(magic) + LegacyRecord.recordSize(magic, key, value);
130           else
131               return Records.LOG_OVERHEAD + LegacyRecord.recordSize(magic, key, value);
132       }
133   
134       /**
135        * Return the size of the record batch header.
136        *
137        * For V0 and V1 with no compression, it's unclear if Records.LOG_OVERHEAD or 0 should be chosen. There is no header
138        * per batch, but a sequence of batches is preceded by the offset and size. This method returns `0` as it's what
139        * `MemoryRecordsBuilder` requires.
140        */
141       public static int recordBatchHeaderSizeInBytes(byte magic, CompressionType compressionType) {
142           if (magic > RecordBatch.MAGIC_VALUE_V1) {
143               return DefaultRecordBatch.RECORD_BATCH_OVERHEAD;
144           } else if (compressionType != CompressionType.NONE) {
145               return Records.LOG_OVERHEAD + LegacyRecord.recordOverhead(magic);
146           } else {
147               return 0;
148           }
149       }
