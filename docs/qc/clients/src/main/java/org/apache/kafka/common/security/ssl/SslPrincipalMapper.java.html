19   import java.io.IOException;
20   import java.util.List;
21   import java.util.ArrayList;
22   import java.util.Locale;
23   import java.util.regex.Matcher;
24   import java.util.regex.Pattern;
25   
26   import static org.apache.kafka.common.config.internals.BrokerSecurityConfigs.DEFAULT_SSL_PRINCIPAL_MAPPING_RULES;
27   
28   public class SslPrincipalMapper {
29   
30       private static final String RULE_PATTERN = "(DEFAULT)|RULE:((\\\\.|[^\\\\/])*)/((\\\\.|[^\\\\/])*)/([LU]?).*?|(.*?)";
31       private static final Pattern RULE_SPLITTER = Pattern.compile("\\s*(" + RULE_PATTERN + ")\\s*(,\\s*|$)");
32       private static final Pattern RULE_PARSER = Pattern.compile(RULE_PATTERN);
33   
34       private final List<Rule> rules;
35   
36       public SslPrincipalMapper(String sslPrincipalMappingRules) {
37           this.rules = parseRules(splitRules(sslPrincipalMappingRules));
38       }
39   
40       public static SslPrincipalMapper fromRules(String sslPrincipalMappingRules) {
41           return new SslPrincipalMapper(sslPrincipalMappingRules);
42       }
43   
44       private static List<String> splitRules(String sslPrincipalMappingRules) {
45           if (sslPrincipalMappingRules == null) {
46               sslPrincipalMappingRules = DEFAULT_SSL_PRINCIPAL_MAPPING_RULES;
47           }
48   
49           List<String> result = new ArrayList<>();
50           Matcher matcher = RULE_SPLITTER.matcher(sslPrincipalMappingRules.trim());
51           while (matcher.find()) {
52               result.add(matcher.group(1));
53           }
54   
55           return result;
56       }
57   
58       private static List<Rule> parseRules(List<String> rules) {
59           List<Rule> result = new ArrayList<>();
60           for (String rule : rules) {
61               Matcher matcher = RULE_PARSER.matcher(rule);
62               if (!matcher.lookingAt()) {
63                   throw new IllegalArgumentException("Invalid rule: " + rule);
64               }
65               if (rule.length() != matcher.end()) {
66                   throw new IllegalArgumentException("Invalid rule: `" + rule + "`, unmatched substring: `" + rule.substring(matcher.end()) + "`");
67               }
68   
69               // empty rules are ignored
70               if (matcher.group(1) != null) {
71                   result.add(new Rule());
72               } else if (matcher.group(2) != null) {
73                   result.add(new Rule(matcher.group(2),
74                                       matcher.group(4),
75                                       "L".equals(matcher.group(6)),
76                                       "U".equals(matcher.group(6))));
77               }
78           }
79   
80           return result;
81       }
82   
83       public String getName(String distinguishedName) throws IOException {
84           for (Rule r : rules) {
85               String principalName = r.apply(distinguishedName);
86               if (principalName != null) {
87                   return principalName;
88               }
89           }
90           throw new NoMatchingRule("No rules apply to " + distinguishedName + ", rules " + rules);
91       }
92   
93       @Override
94       public String toString() {
95           return "SslPrincipalMapper(rules = " + rules + ")";
96       }
97   
98       public static class NoMatchingRule extends IOException {
99           NoMatchingRule(String msg) {
100               super(msg);
101           }
102       }
103   
104       private static class Rule {
105           private static final Pattern BACK_REFERENCE_PATTERN = Pattern.compile("\\$(\\d+)");
106   
107           private final boolean isDefault;
108           private final Pattern pattern;
109           private final String replacement;
110           private final boolean toLowerCase;
111           private final boolean toUpperCase;
112   
113           Rule() {
114               isDefault = true;
115               pattern = null;
116               replacement = null;
117               toLowerCase = false;
118               toUpperCase = false;
119           }
120   
121           Rule(String pattern, String replacement, boolean toLowerCase, boolean toUpperCase) {
122               isDefault = false;
123               this.pattern = pattern == null ? null : Pattern.compile(pattern);
124               this.replacement = replacement;
125               this.toLowerCase = toLowerCase;
126               this.toUpperCase = toUpperCase;
127           }
128   
129           String apply(String distinguishedName) {
130               if (isDefault) {
131                   return distinguishedName;
132               }
133   
134               String result = null;
135               final Matcher m = pattern.matcher(distinguishedName);
136   
137               if (m.matches()) {
138                   result = distinguishedName.replaceAll(pattern.pattern(), escapeLiteralBackReferences(replacement, m.groupCount()));
139               }
140   
141               if (toLowerCase && result != null) {
142                   result = result.toLowerCase(Locale.ENGLISH);
143               } else if (toUpperCase & result != null) {
144                   result = result.toUpperCase(Locale.ENGLISH);
145               }
146   
147               return result;
148           }
149   
150           //If we find a back reference that is not valid, then we will treat it as a literal string. For example, if we have 3 capturing
151           //groups and the Replacement Value has the value is "$1@$4", then we want to treat the $4 as a literal "$4", rather
152           //than attempting to use it as a back reference.
153           //This method was taken from Apache Nifi project : org.apache.nifi.authorization.util.IdentityMappingUtil
154           private String escapeLiteralBackReferences(final String unescaped, final int numCapturingGroups) {
155               if (numCapturingGroups == 0) {
156                   return unescaped;
157               }
158   
159               String value = unescaped;
160               final Matcher backRefMatcher = BACK_REFERENCE_PATTERN.matcher(value);
161               while (backRefMatcher.find()) {
162                   final String backRefNum = backRefMatcher.group(1);
163                   if (backRefNum.startsWith("0")) {
164                       continue;
165                   }
166                   int backRefIndex = Integer.parseInt(backRefNum);
167   
168   
169                   // if we have a replacement value like $123, and we have less than 123 capturing groups, then
170                   // we want to truncate the 3 and use capturing group 12; if we have less than 12 capturing groups,
171                   // then we want to truncate the 2 and use capturing group 1; if we don't have a capturing group then
172                   // we want to truncate the 1 and get 0.
173                   while (backRefIndex > numCapturingGroups && backRefIndex >= 10) {
174                       backRefIndex /= 10;
175                   }
176   
177                   if (backRefIndex > numCapturingGroups) {
178                       final StringBuilder sb = new StringBuilder(value.length() + 1);
179                       final int groupStart = backRefMatcher.start(1);
180   
181                       sb.append(value.substring(0, groupStart - 1));
182                       sb.append("\\");
183                       sb.append(value.substring(groupStart - 1));
184                       value = sb.toString();
185                   }
186               }
187   
188               return value;
189           }
190   
191           @Override
192           public String toString() {
193               StringBuilder buf = new StringBuilder();
194               if (isDefault) {
195                   buf.append("DEFAULT");
196               } else {
197                   buf.append("RULE:");
198                   if (pattern != null) {
199                       buf.append(pattern);
200                   }
201                   if (replacement != null) {
202                       buf.append("/");
203                       buf.append(replacement);
204                   }
205                   if (toLowerCase) {
206                       buf.append("/L");
207                   } else if (toUpperCase) {
208                       buf.append("/U");
209                   }
210               }
211               return buf.toString();
212           }

214       }
