175       /**
176        *  Get the in-flight batches that has reached delivery timeout.
177        */
178       private List<ProducerBatch> getExpiredInflightBatches(long now) {
179           List<ProducerBatch> expiredBatches = new ArrayList<>();
180   
181           for (Iterator<Map.Entry<TopicPartition, List<ProducerBatch>>> batchIt = inFlightBatches.entrySet().iterator(); batchIt.hasNext();) {
182               Map.Entry<TopicPartition, List<ProducerBatch>> entry = batchIt.next();
183               List<ProducerBatch> partitionInFlightBatches = entry.getValue();
184               if (partitionInFlightBatches != null) {
185                   Iterator<ProducerBatch> iter = partitionInFlightBatches.iterator();
186                   while (iter.hasNext()) {
187                       ProducerBatch batch = iter.next();
188                       if (batch.hasReachedDeliveryTimeout(accumulator.getDeliveryTimeoutMs(), now)) {
189                           iter.remove();
190                           // expireBatches is called in Sender.sendProducerData, before client.poll.
191                           // The !batch.isDone() invariant should always hold. An IllegalStateException
192                           // exception will be thrown if the invariant is violated.
193                           if (!batch.isDone()) {
194                               expiredBatches.add(batch);
195                           } else {
196                               throw new IllegalStateException(batch.topicPartition + " batch created at " +
197                                   batch.createdMs + " gets unexpected final state " + batch.finalState());
198                           }
199                       } else {
200                           accumulator.maybeUpdateNextBatchExpiryTime(batch);
201                           break;
202                       }
203                   }
204                   if (partitionInFlightBatches.isEmpty()) {
205                       batchIt.remove();
206                   }
207               }
208           }
209           return expiredBatches;
210       }

233       /**
234        * The main run loop for the sender thread
235        */
236       @Override
237       public void run() {
238           log.debug("Starting Kafka producer I/O thread.");
239   
240           // main loop, runs until close is called
241           while (running) {
242               try {
243                   runOnce();
244               } catch (Exception e) {
245                   log.error("Uncaught error in kafka producer I/O thread: ", e);
246               }
247           }
248   
249           log.debug("Beginning shutdown of Kafka producer I/O thread, sending remaining records.");
250   
251           // okay we stopped accepting requests but there may still be
252           // requests in the transaction manager, accumulator or waiting for acknowledgment,
253           // wait until these are completed.
254           while (!forceClose && ((this.accumulator.hasUndrained() || this.client.inFlightRequestCount() > 0) || hasPendingTransactionalRequests())) {
255               try {
256                   runOnce();
257               } catch (Exception e) {
258                   log.error("Uncaught error in kafka producer I/O thread: ", e);
259               }
260           }
261   
262           // Abort the transaction if any commit or abort didn't go through the transaction manager's queue
263           while (!forceClose && transactionManager != null && transactionManager.hasOngoingTransaction()) {
264               if (!transactionManager.isCompleting()) {
265                   log.info("Aborting incomplete transaction due to shutdown");
266                   transactionManager.beginAbort();
267               }
268               try {
269                   runOnce();
270               } catch (Exception e) {
271                   log.error("Uncaught error in kafka producer I/O thread: ", e);
272               }
273           }
274   
275           if (forceClose) {
276               // We need to fail all the incomplete transactional requests and batches and wake up the threads waiting on
277               // the futures.
278               if (transactionManager != null) {
279                   log.debug("Aborting incomplete transactional requests due to forced shutdown");
280                   transactionManager.close();
281               }
282               log.debug("Aborting incomplete batches due to forced shutdown");
283               this.accumulator.abortIncompleteBatches();
284           }
285           try {
286               this.client.close();
287           } catch (Exception e) {
288               log.error("Failed to close network client", e);
289           }
290   
291           log.debug("Shutdown of Kafka producer I/O thread has completed.");
292       }
293   
294       /**
295        * Run a single iteration of sending
296        *
297        */
298       void runOnce() {
299           if (transactionManager != null) {
300               try {
301                   transactionManager.maybeResolveSequences();
302   
303                   // do not continue sending if the transaction manager is in a failed state
304                   if (transactionManager.hasFatalError()) {
305                       RuntimeException lastError = transactionManager.lastError();
306                       if (lastError != null)
307                           maybeAbortBatches(lastError);
308                       client.poll(retryBackoffMs, time.milliseconds());
309                       return;
310                   }
311   
312                   // Check whether we need a new producerId. If so, we will enqueue an InitProducerId
313                   // request which will be sent below
314                   transactionManager.bumpIdempotentEpochAndResetIdIfNeeded();
315   
316                   if (maybeSendAndPollTransactionalRequest()) {
317                       return;
318                   }
319               } catch (AuthenticationException e) {
320                   // This is already logged as error, but propagated here to perform any clean ups.
321                   log.trace("Authentication exception while processing transactional request", e);
322                   transactionManager.authenticationFailed(e);
323               }
324           }
325   
326           long currentTimeMs = time.milliseconds();
327           long pollTimeout = sendProducerData(currentTimeMs);
328           client.poll(pollTimeout, currentTimeMs);
329       }

331       private long sendProducerData(long now) {
332           Cluster cluster = metadata.fetch();
333           // get the list of partitions with data ready to send
334           RecordAccumulator.ReadyCheckResult result = this.accumulator.ready(cluster, now);
335   
336           // if there are any partitions whose leaders are not known yet, force metadata update
337           if (!result.unknownLeaderTopics.isEmpty()) {
338               // The set of topics with unknown leader contains topics with leader election pending as well as
339               // topics which may have expired. Add the topic again to metadata to ensure it is included
340               // and request metadata update, since there are messages to send to the topic.
341               for (String topic : result.unknownLeaderTopics)
342                   this.metadata.add(topic, now);
343   
344               log.debug("Requesting metadata update due to unknown leader topics from the batched records: {}",
345                   result.unknownLeaderTopics);
346               this.metadata.requestUpdate();
347           }
348   
349           // remove any nodes we aren't ready to send to
350           Iterator<Node> iter = result.readyNodes.iterator();
351           long notReadyTimeout = Long.MAX_VALUE;
352           while (iter.hasNext()) {
353               Node node = iter.next();
354               if (!this.client.ready(node, now)) {
355                   // Update just the readyTimeMs of the latency stats, so that it moves forward
356                   // every time the batch is ready (then the difference between readyTimeMs and
357                   // drainTimeMs would represent how long data is waiting for the node).
358                   this.accumulator.updateNodeLatencyStats(node.id(), now, false);
359                   iter.remove();
360                   notReadyTimeout = Math.min(notReadyTimeout, this.client.pollDelayMs(node, now));
361               } else {
362                   // Update both readyTimeMs and drainTimeMs, this would "reset" the node
363                   // latency.
364                   this.accumulator.updateNodeLatencyStats(node.id(), now, true);
365               }
366           }
367   
368           // create produce requests
369           Map<Integer, List<ProducerBatch>> batches = this.accumulator.drain(cluster, result.readyNodes, this.maxRequestSize, now);
370           addToInflightBatches(batches);
371           if (guaranteeMessageOrder) {
372               // Mute all the partitions drained
373               for (List<ProducerBatch> batchList : batches.values()) {
374                   for (ProducerBatch batch : batchList)
375                       this.accumulator.mutePartition(batch.topicPartition);
376               }
377           }
378   
379           accumulator.resetNextBatchExpiryTime();
380           List<ProducerBatch> expiredInflightBatches = getExpiredInflightBatches(now);
381           List<ProducerBatch> expiredBatches = this.accumulator.expiredBatches(now);
382           expiredBatches.addAll(expiredInflightBatches);
383   
384           // Reset the producer id if an expired batch has previously been sent to the broker. Also update the metrics
385           // for expired batches. see the documentation of @TransactionState.resetIdempotentProducerId to understand why
386           // we need to reset the producer id here.
387           if (!expiredBatches.isEmpty())
388               log.trace("Expired {} batches in accumulator", expiredBatches.size());
389           for (ProducerBatch expiredBatch : expiredBatches) {
390               String errorMessage = "Expiring " + expiredBatch.recordCount + " record(s) for " + expiredBatch.topicPartition
391                   + ":" + (now - expiredBatch.createdMs) + " ms has passed since batch creation";
392               failBatch(expiredBatch, new TimeoutException(errorMessage), false);
393               if (transactionManager != null && expiredBatch.inRetry()) {
394                   // This ensures that no new batches are drained until the current in flight batches are fully resolved.
395                   transactionManager.markSequenceUnresolved(expiredBatch);
396               }
397           }
398           sensors.updateProduceRequestMetrics(batches);
399   
400           // If we have any nodes that are ready to send + have sendable data, poll with 0 timeout so this can immediately
401           // loop and try sending more data. Otherwise, the timeout will be the smaller value between next batch expiry
402           // time, and the delay time for checking data availability. Note that the nodes may have data that isn't yet
403           // sendable due to lingering, backing off, etc. This specifically does not include nodes with sendable data
404           // that aren't ready to send since they would cause busy looping.
405           long pollTimeout = Math.min(result.nextReadyCheckDelayMs, notReadyTimeout);
406           pollTimeout = Math.min(pollTimeout, this.accumulator.nextExpiryTimeMs() - now);
407           pollTimeout = Math.max(pollTimeout, 0);
408           if (!result.readyNodes.isEmpty()) {
409               log.trace("Nodes with data ready to send: {}", result.readyNodes);
410               // if some partitions are already ready to be sent, the select time would be 0;
411               // otherwise if some partition already has some data accumulated but not ready yet,
412               // the select time will be the time difference between now and its linger expiry time;
413               // otherwise the select time will be the time difference between now and the metadata expiry time;
414               pollTimeout = 0;
415           }
416           sendProduceRequests(batches, now);
417           return pollTimeout;
418       }

420       /**
421        * Returns true if a transactional request is sent or polled, or if a FindCoordinator request is enqueued
422        */
423       private boolean maybeSendAndPollTransactionalRequest() {
424           if (transactionManager.hasInFlightRequest()) {
425               // as long as there are outstanding transactional requests, we simply wait for them to return
426               client.poll(retryBackoffMs, time.milliseconds());
427               return true;
428           }
429   
430           if (transactionManager.hasAbortableError() || transactionManager.isAborting()) {
431               if (accumulator.hasIncomplete()) {
432                   // Attempt to get the last error that caused this abort.
433                   RuntimeException exception = transactionManager.lastError();
434                   // If there was no error, but we are still aborting,
435                   // then this is most likely a case where there was no fatal error.
436                   if (exception == null) {
437                       exception = new TransactionAbortedException();
438                   }
439                   accumulator.abortUndrainedBatches(exception);
440               }
441           }
442   
443           TransactionManager.TxnRequestHandler nextRequestHandler = transactionManager.nextRequest(accumulator.hasIncomplete());
444           if (nextRequestHandler == null)
445               return false;
446   
447           AbstractRequest.Builder<?> requestBuilder = nextRequestHandler.requestBuilder();
448           Node targetNode = null;
449           try {
450               FindCoordinatorRequest.CoordinatorType coordinatorType = nextRequestHandler.coordinatorType();
451               targetNode = coordinatorType != null ?
452                       transactionManager.coordinator(coordinatorType) :
453                       client.leastLoadedNode(time.milliseconds());
454               if (targetNode != null) {
455                   if (!awaitNodeReady(targetNode, coordinatorType)) {
456                       log.trace("Target node {} not ready within request timeout, will retry when node is ready.", targetNode);
457                       maybeFindCoordinatorAndRetry(nextRequestHandler);
458                       return true;
459                   }
460               } else if (coordinatorType != null) {
461                   log.trace("Coordinator not known for {}, will retry {} after finding coordinator.", coordinatorType, requestBuilder.apiKey());
462                   maybeFindCoordinatorAndRetry(nextRequestHandler);
463                   return true;
464               } else {
465                   log.trace("No nodes available to send requests, will poll and retry when until a node is ready.");
466                   transactionManager.retry(nextRequestHandler);
467                   client.poll(retryBackoffMs, time.milliseconds());
468                   return true;
469               }
470   
471               if (nextRequestHandler.isRetry())
472                   time.sleep(nextRequestHandler.retryBackoffMs());
473   
474               long currentTimeMs = time.milliseconds();
475               ClientRequest clientRequest = client.newClientRequest(targetNode.idString(), requestBuilder, currentTimeMs,
476                   true, requestTimeoutMs, nextRequestHandler);
477               log.debug("Sending transactional request {} to node {} with correlation ID {}", requestBuilder, targetNode, clientRequest.correlationId());
478               client.send(clientRequest, currentTimeMs);
479               transactionManager.setInFlightCorrelationId(clientRequest.correlationId());
480               client.poll(retryBackoffMs, time.milliseconds());
481               return true;
482           } catch (IOException e) {
483               log.debug("Disconnect from {} while trying to send request {}. Going " +
484                       "to back off and retry.", targetNode, requestBuilder, e);
485               // We break here so that we pick up the FindCoordinator request immediately.
486               maybeFindCoordinatorAndRetry(nextRequestHandler);
487               return true;
488           }
489       }

501       }

546       /**
547        * Handle a produce response
548        */
549       private void handleProduceResponse(ClientResponse response, Map<TopicPartition, ProducerBatch> batches, long now) {
550           RequestHeader requestHeader = response.requestHeader();
551           int correlationId = requestHeader.correlationId();
552           if (response.wasDisconnected()) {
553               log.trace("Cancelled request with header {} due to node {} being disconnected",
554                   requestHeader, response.destination());
555               for (ProducerBatch batch : batches.values())
556                   completeBatch(batch, new ProduceResponse.PartitionResponse(Errors.NETWORK_EXCEPTION, String.format("Disconnected from node %s", response.destination())),
557                           correlationId, now);
558           } else if (response.versionMismatch() != null) {
559               log.warn("Cancelled request {} due to a version mismatch with node {}",
560                       response, response.destination(), response.versionMismatch());
561               for (ProducerBatch batch : batches.values())
562                   completeBatch(batch, new ProduceResponse.PartitionResponse(Errors.UNSUPPORTED_VERSION), correlationId, now);
563           } else {
564               log.trace("Received produce response from node {} with correlation id {}", response.destination(), correlationId);
565               // if we have a response, parse it
566               if (response.hasResponse()) {
567                   // Sender should exercise PartitionProduceResponse rather than ProduceResponse.PartitionResponse
568                   // https://issues.apache.org/jira/browse/KAFKA-10696
569                   ProduceResponse produceResponse = (ProduceResponse) response.responseBody();
570                   produceResponse.data().responses().forEach(r -> r.partitionResponses().forEach(p -> {
571                       TopicPartition tp = new TopicPartition(r.name(), p.index());
572                       ProduceResponse.PartitionResponse partResp = new ProduceResponse.PartitionResponse(
573                               Errors.forCode(p.errorCode()),
574                               p.baseOffset(),
575                               p.logAppendTimeMs(),
576                               p.logStartOffset(),
577                               p.recordErrors()
578                                   .stream()
579                                   .map(e -> new ProduceResponse.RecordError(e.batchIndex(), e.batchIndexErrorMessage()))
580                                   .collect(Collectors.toList()),
581                               p.errorMessage());
582                       ProducerBatch batch = batches.get(tp);
583                       completeBatch(batch, partResp, correlationId, now);
584                   }));
585                   this.sensors.recordLatency(response.destination(), response.requestLatencyMs());
586               } else {
587                   // this is the acks = 0 case, just complete all requests
588                   for (ProducerBatch batch : batches.values()) {
589                       completeBatch(batch, new ProduceResponse.PartitionResponse(Errors.NONE), correlationId, now);
590                   }
591               }
592           }
593       }
594   
595       /**
596        * Complete or retry the given batch of records.
597        *
598        * @param batch The record batch
599        * @param response The produce response
600        * @param correlationId The correlation id for the request
601        * @param now The current POSIX timestamp in milliseconds
602        */
603       private void completeBatch(ProducerBatch batch, ProduceResponse.PartitionResponse response, long correlationId,
604                                  long now) {
605           Errors error = response.error;
606   
607           if (error == Errors.MESSAGE_TOO_LARGE && batch.recordCount > 1 && !batch.isDone() &&
608                   (batch.magic() >= RecordBatch.MAGIC_VALUE_V2 || batch.isCompressed())) {
609               // If the batch is too large, we split the batch and send the split batches again. We do not decrement
610               // the retry attempts in this case.
611               log.warn(
612                   "Got error produce response in correlation id {} on topic-partition {}, splitting and retrying ({} attempts left). Error: {}",
613                   correlationId,
614                   batch.topicPartition,
615                   this.retries - batch.attempts(),
616                   formatErrMsg(response));
617               if (transactionManager != null)
618                   transactionManager.removeInFlightBatch(batch);
619               this.accumulator.splitAndReenqueue(batch);
620               maybeRemoveAndDeallocateBatch(batch);
621               this.sensors.recordBatchSplit();
622           } else if (error != Errors.NONE) {
623               if (canRetry(batch, response, now)) {
624                   log.warn(
625                       "Got error produce response with correlation id {} on topic-partition {}, retrying ({} attempts left). Error: {}",
626                       correlationId,
627                       batch.topicPartition,
628                       this.retries - batch.attempts() - 1,
629                       formatErrMsg(response));
630                   reenqueueBatch(batch, now);
631               } else if (error == Errors.DUPLICATE_SEQUENCE_NUMBER) {
632                   // If we have received a duplicate sequence error, it means that the sequence number has advanced beyond
633                   // the sequence of the current batch, and we haven't retained batch metadata on the broker to return
634                   // the correct offset and timestamp.
635                   //
636                   // The only thing we can do is to return success to the user and not return a valid offset and timestamp.
637                   completeBatch(batch, response);
638               } else {
639                   // tell the user the result of their request. We only adjust sequence numbers if the batch didn't exhaust
640                   // its retries -- if it did, we don't know whether the sequence number was accepted or not, and
641                   // thus it is not safe to reassign the sequence.
642                   failBatch(batch, response, batch.attempts() < this.retries);
643               }
644               if (error.exception() instanceof InvalidMetadataException) {
645                   if (error.exception() instanceof UnknownTopicOrPartitionException) {
646                       log.warn("Received unknown topic or partition error in produce request on partition {}. The " +
647                               "topic-partition may not exist or the user may not have Describe access to it",
648                           batch.topicPartition);
649                   } else {
650                       log.warn("Received invalid metadata error in produce request on partition {} due to {}. Going " +
651                               "to request metadata update now", batch.topicPartition,
652                               error.exception(response.errorMessage).toString());
653                   }
654                   metadata.requestUpdate();
655               }
656           } else {
657               completeBatch(batch, response);
658           }
659   
660           // Unmute the completed partition.
661           if (guaranteeMessageOrder)
662               this.accumulator.unmutePartition(batch.topicPartition);
663       }
664   
665       /**
666        * Format the error from a {@link ProduceResponse.PartitionResponse} in a user-friendly string
667        * e.g "NETWORK_EXCEPTION. Error Message: Disconnected from node 0"
668        */
669       private String formatErrMsg(ProduceResponse.PartitionResponse response) {
670           String errorMessageSuffix = (response.errorMessage == null || response.errorMessage.isEmpty()) ?
671                   "" : String.format(". Error Message: %s", response.errorMessage);
672           return String.format("%s%s", response.error, errorMessageSuffix);
673       }

691       private void failBatch(ProducerBatch batch,
692                              ProduceResponse.PartitionResponse response,
693                              boolean adjustSequenceNumbers) {
694           final RuntimeException topLevelException;
695           if (response.error == Errors.TOPIC_AUTHORIZATION_FAILED)
696               topLevelException = new TopicAuthorizationException(Collections.singleton(batch.topicPartition.topic()));
697           else if (response.error == Errors.CLUSTER_AUTHORIZATION_FAILED)
698               topLevelException = new ClusterAuthorizationException("The producer is not authorized to do idempotent sends");
699           else
700               topLevelException = response.error.exception(response.errorMessage);
701   
702           if (response.recordErrors == null || response.recordErrors.isEmpty()) {
703               failBatch(batch, topLevelException, adjustSequenceNumbers);
704           } else {
705               Map<Integer, RuntimeException> recordErrorMap = new HashMap<>(response.recordErrors.size());
706               for (ProduceResponse.RecordError recordError : response.recordErrors) {
707                   // The API leaves us with some awkwardness interpreting the errors in the response.
708                   // We cannot differentiate between different error cases (such as INVALID_TIMESTAMP)
709                   // from the single error code at the partition level, so instead we use INVALID_RECORD
710                   // for all failed records and rely on the message to distinguish the cases.
711                   final String errorMessage;
712                   if (recordError.message != null) {
713                       errorMessage = recordError.message;
714                   } else if (response.errorMessage != null) {
715                       errorMessage = response.errorMessage;
716                   } else {
717                       errorMessage = response.error.message();
718                   }
719   
720                   // If the batch contained only a single record error, then we can unambiguously
721                   // use the exception type corresponding to the partition-level error code.
722                   if (response.recordErrors.size() == 1) {
723                       recordErrorMap.put(recordError.batchIndex, response.error.exception(errorMessage));
724                   } else {
725                       recordErrorMap.put(recordError.batchIndex, new InvalidRecordException(errorMessage));
726                   }
727               }
728   
729               Function<Integer, RuntimeException> recordExceptions = batchIndex -> {
730                   RuntimeException exception = recordErrorMap.get(batchIndex);
731                   if (exception != null) {
732                       return exception;
733                   } else {
734                       // If the response contains record errors, then the records which failed validation
735                       // will be present in the response. To avoid confusion for the remaining records, we
736                       // return a generic exception.
737                       return new KafkaException("Failed to append record because it was part of a batch " +
738                           "which had one more more invalid records");
739                   }
740               };
741   
742               failBatch(batch, topLevelException, recordExceptions, adjustSequenceNumbers);
743           }
744       }

746       private void failBatch(
747           ProducerBatch batch,
748           RuntimeException topLevelException,
749           boolean adjustSequenceNumbers
750       ) {
751           failBatch(batch, topLevelException, batchIndex -> topLevelException, adjustSequenceNumbers);
752       }

754       private void failBatch(
755           ProducerBatch batch,
756           RuntimeException topLevelException,
757           Function<Integer, RuntimeException> recordExceptions,
758           boolean adjustSequenceNumbers
759       ) {
760           if (transactionManager != null) {
761               transactionManager.handleFailedBatch(batch, topLevelException, adjustSequenceNumbers);
762           }
763   
764           this.sensors.recordErrors(batch.topicPartition.topic(), batch.recordCount);
765   
766           if (batch.completeExceptionally(topLevelException, recordExceptions)) {
767               maybeRemoveAndDeallocateBatch(batch);
768           }
769       }
770   
771       /**
772        * We can retry a send if the error is transient and the number of attempts taken is fewer than the maximum allowed.
773        * We can also retry OutOfOrderSequence exceptions for future batches, since if the first batch has failed, the
774        * future batches are certain to fail with an OutOfOrderSequence exception.
775        */
776       private boolean canRetry(ProducerBatch batch, ProduceResponse.PartitionResponse response, long now) {
777           return !batch.hasReachedDeliveryTimeout(accumulator.getDeliveryTimeoutMs(), now) &&
778               batch.attempts() < this.retries &&
779               !batch.isDone() &&
780               (transactionManager == null ?
781                       response.error.exception() instanceof RetriableException :
782                       transactionManager.canRetry(response, batch));
783       }

786        * Transfer the record batches into a list of produce requests on a per-node basis
787        */
788       private void sendProduceRequests(Map<Integer, List<ProducerBatch>> collated, long now) {
789           for (Map.Entry<Integer, List<ProducerBatch>> entry : collated.entrySet())
790               sendProduceRequest(now, entry.getKey(), acks, requestTimeoutMs, entry.getValue());
791       }

793       /**
794        * Create a produce request from the given record batches
795        */
796       private void sendProduceRequest(long now, int destination, short acks, int timeout, List<ProducerBatch> batches) {
797           if (batches.isEmpty())
798               return;
799   
800           final Map<TopicPartition, ProducerBatch> recordsByPartition = new HashMap<>(batches.size());
801   
802           // find the minimum magic version used when creating the record sets
803           byte minUsedMagic = apiVersions.maxUsableProduceMagic();
804           for (ProducerBatch batch : batches) {
805               if (batch.magic() < minUsedMagic)
806                   minUsedMagic = batch.magic();
807           }
808           ProduceRequestData.TopicProduceDataCollection tpd = new ProduceRequestData.TopicProduceDataCollection();
809           for (ProducerBatch batch : batches) {
810               TopicPartition tp = batch.topicPartition;
811               MemoryRecords records = batch.records();
812   
813               // down convert if necessary to the minimum magic used. In general, there can be a delay between the time
814               // that the producer starts building the batch and the time that we send the request, and we may have
815               // chosen the message format based on out-dated metadata. In the worst case, we optimistically chose to use
816               // the new message format, but found that the broker didn't support it, so we need to down-convert on the
817               // client before sending. This is intended to handle edge cases around cluster upgrades where brokers may
818               // not all support the same message format version. For example, if a partition migrates from a broker
819               // which is supporting the new magic version to one which doesn't, then we will need to convert.
820               if (!records.hasMatchingMagic(minUsedMagic))
821                   records = batch.records().downConvert(minUsedMagic, 0, time).records();
822               ProduceRequestData.TopicProduceData tpData = tpd.find(tp.topic());
823               if (tpData == null) {
824                   tpData = new ProduceRequestData.TopicProduceData().setName(tp.topic());
825                   tpd.add(tpData);
826               }
827               tpData.partitionData().add(new ProduceRequestData.PartitionProduceData()
828                       .setIndex(tp.partition())
829                       .setRecords(records));
830               recordsByPartition.put(tp, batch);
831           }
832   
833           String transactionalId = null;
834           if (transactionManager != null && transactionManager.isTransactional()) {
835               transactionalId = transactionManager.transactionalId();
836           }
837   
838           ProduceRequest.Builder requestBuilder = ProduceRequest.forMagic(minUsedMagic,
839                   new ProduceRequestData()
840                           .setAcks(acks)
841                           .setTimeoutMs(timeout)
842                           .setTransactionalId(transactionalId)
843                           .setTopicData(tpd));
844           RequestCompletionHandler callback = response -> handleProduceResponse(response, recordsByPartition, time.milliseconds());
845   
846           String nodeId = Integer.toString(destination);

848                   requestTimeoutMs, callback);
849           client.send(clientRequest, now);
850           log.trace("Sent produce request to {}: {}", nodeId, requestBuilder);
851       }

962           public void updateProduceRequestMetrics(Map<Integer, List<ProducerBatch>> batches) {
963               long now = time.milliseconds();
964               for (List<ProducerBatch> nodeBatch : batches.values()) {
965                   int records = 0;
966                   for (ProducerBatch batch : nodeBatch) {
967                       // register all per-topic metrics at once
968                       String topic = batch.topicPartition.topic();
969                       maybeRegisterTopicMetrics(topic);
970   
971                       // per-topic record send rate
972                       String topicRecordsCountName = "topic." + topic + ".records-per-batch";
973                       Sensor topicRecordCount = Objects.requireNonNull(this.metrics.getSensor(topicRecordsCountName));
974                       topicRecordCount.record(batch.recordCount);
975   
976                       // per-topic bytes send rate
977                       String topicByteRateName = "topic." + topic + ".bytes";
978                       Sensor topicByteRate = Objects.requireNonNull(this.metrics.getSensor(topicByteRateName));
979                       topicByteRate.record(batch.estimatedSizeInBytes());
980   
981                       // per-topic compression rate
982                       String topicCompressionRateName = "topic." + topic + ".compression-rate";
983                       Sensor topicCompressionRate = Objects.requireNonNull(this.metrics.getSensor(topicCompressionRateName));
984                       topicCompressionRate.record(batch.compressionRatio());
985   
986                       // global metrics
987                       this.batchSizeSensor.record(batch.estimatedSizeInBytes(), now);
988                       this.queueTimeSensor.record(batch.queueTimeMs(), now);
989                       this.compressionRateSensor.record(batch.compressionRatio());
990                       this.maxRecordSizeSensor.record(batch.maxRecordSize, now);
991                       records += batch.recordCount;
992                   }
993                   this.recordsPerRequestSensor.record(records, now);
994               }
995           }
