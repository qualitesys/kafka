39   /**
40    * This class is used for specifying the set of expected configurations. For each configuration, you can specify
41    * the name, the type, the default value, the documentation, the group information, the order in the group,
42    * the width of the configuration value and the name suitable for display in the UI.
43    *
44    * You can provide special validation logic used for single configuration validation by overriding {@link Validator}.
45    *
46    * Moreover, you can specify the dependents of a configuration. The valid values and visibility of a configuration
47    * may change according to the values of other configurations. You can override {@link Recommender} to get valid
48    * values and set visibility of a configuration given the current configuration values.
49    *
50    * <p/>
51    * To use the class:
52    * <p/>
53    * <pre>
54    * ConfigDef defs = new ConfigDef();
55    *
56    * defs.define(&quot;config_with_default&quot;, Type.STRING, &quot;default string value&quot;, &quot;Configuration with default value.&quot;);
57    * defs.define(&quot;config_with_validator&quot;, Type.INT, 42, Range.atLeast(0), &quot;Configuration with user provided validator.&quot;);
58    * defs.define(&quot;config_with_dependents&quot;, Type.INT, &quot;Configuration with dependents.&quot;, &quot;group&quot;, 1, &quot;Config With Dependents&quot;, Arrays.asList(&quot;config_with_default&quot;,&quot;config_with_validator&quot;));
59    *
60    * Map&lt;String, String&gt; props = new HashMap&lt;&gt;();
61    * props.put(&quot;config_with_default&quot;, &quot;some value&quot;);
62    * props.put(&quot;config_with_dependents&quot;, &quot;some other value&quot;);
63    *
64    * Map&lt;String, Object&gt; configs = defs.parse(props);
65    * // will return &quot;some value&quot;
66    * String someConfig = (String) configs.get(&quot;config_with_default&quot;);
67    * // will return default value of 42
68    * int anotherConfig = (Integer) configs.get(&quot;config_with_validator&quot;);
69    *
70    * To validate the full configuration, use:
71    * List&lt;Config&gt; configs = defs.validate(props);
72    * The {@link Config} contains updated configuration information given the current configuration values.
73    * </pre>
74    * <p/>
75    * This class can be used standalone or in combination with {@link AbstractConfig} which provides some additional
76    * functionality for accessing configs.
77    */

114       public Map<String, Object> defaultValues() {
115           Map<String, Object> defaultValues = new HashMap<>();
116           for (ConfigKey key : configKeys.values()) {
117               if (key.defaultValue != NO_DEFAULT_VALUE)
118                   defaultValues.put(key.name, key.defaultValue);
119           }
120           return defaultValues;
121       }

123       public ConfigDef define(ConfigKey key) {
124           if (configKeys.containsKey(key.name)) {
125               throw new ConfigException("Configuration " + key.name + " is defined twice.");
126           }
127           if (key.group != null && !groups.contains(key.group)) {
128               groups.add(key.group);
129           }
130           configKeys.put(key.name, key);
131           return this;
132       }

463       /**
464        * Parse and validate configs against this configuration definition. The input is a map of configs. It is expected
465        * that the keys of the map are strings, but the values can either be strings or they may already be of the
466        * appropriate type (int, string, etc). This will work equally well with either java.util.Properties instances or a
467        * programmatically constructed map.
468        *
469        * @param props The configs to parse and validate.
470        * @return Parsed and validated configs. The key will be the config name and the value will be the value parsed into
471        * the appropriate type (int, string, etc).
472        */
473       public Map<String, Object> parse(Map<?, ?> props) {
474           // Check all configurations are defined
475           List<String> undefinedConfigKeys = undefinedDependentConfigs();
476           if (!undefinedConfigKeys.isEmpty()) {
477               String joined = Utils.join(undefinedConfigKeys, ",");
478               throw new ConfigException("Some configurations in are referred in the dependents, but not defined: " + joined);
479           }
480           // parse all known keys
481           Map<String, Object> values = new HashMap<>();
482           for (ConfigKey key : configKeys.values())
483               values.put(key.name, parseValue(key, props.get(key.name), props.containsKey(key.name)));
484           return values;
485       }
486   
487       Object parseValue(ConfigKey key, Object value, boolean isSet) {
488           Object parsedValue;
489           if (isSet) {
490               parsedValue = parseType(key.name, value, key.type);
491           // props map doesn't contain setting, the key is required because no default value specified - its an error
492           } else if (NO_DEFAULT_VALUE.equals(key.defaultValue)) {
493               throw new ConfigException("Missing required configuration \"" + key.name + "\" which has no default value.");
494           } else {
495               // otherwise assign setting its default value
496               parsedValue = key.defaultValue;
497           }
498           if (key.validator != null) {
499               key.validator.ensureValid(key.name, parsedValue);
500           }
501           return parsedValue;
502       }

581       private void parseForValidate(String name, Map<String, String> props, Map<String, Object> parsed, Map<String, ConfigValue> configs) {
582           if (!configKeys.containsKey(name)) {
583               return;
584           }
585           ConfigKey key = configKeys.get(name);
586           ConfigValue config = configs.get(name);
587   
588           Object value = null;
589           if (props.containsKey(key.name)) {
590               try {
591                   value = parseType(key.name, props.get(key.name), key.type);
592               } catch (ConfigException e) {
593                   config.addErrorMessage(e.getMessage());
594               }
595           } else if (NO_DEFAULT_VALUE.equals(key.defaultValue)) {
596               config.addErrorMessage("Missing required configuration \"" + key.name + "\" which has no default value.");
597           } else {
598               value = key.defaultValue;
599           }
600   
601           if (key.validator != null) {
602               try {
603                   key.validator.ensureValid(key.name, value);
604               } catch (ConfigException e) {
605                   config.addErrorMessage(e.getMessage());
606               }
607           }
608           config.value(value);
609           parsed.put(name, value);
610           for (String dependent: key.dependents) {
611               parseForValidate(dependent, props, parsed, configs);
612           }
613       }

615       private void validate(String name, Map<String, Object> parsed, Map<String, ConfigValue> configs) {
616           if (!configKeys.containsKey(name)) {
617               return;
618           }
619           ConfigKey key = configKeys.get(name);
620           ConfigValue value = configs.get(name);
621           if (key.recommender != null) {
622               try {
623                   List<Object> recommendedValues = key.recommender.validValues(name, parsed);
624                   List<Object> originalRecommendedValues = value.recommendedValues();
625                   if (!originalRecommendedValues.isEmpty()) {
626                       Set<Object> originalRecommendedValueSet = new HashSet<>(originalRecommendedValues);
627                       recommendedValues.removeIf(o -> !originalRecommendedValueSet.contains(o));
628                   }
629                   value.recommendedValues(recommendedValues);
630                   value.visible(key.recommender.visible(name, parsed));
631               } catch (ConfigException e) {
632                   value.addErrorMessage(e.getMessage());
633               }
634           }
635   
636           configs.put(name, value);
637           for (String dependent: key.dependents) {
638               validate(dependent, parsed, configs);
639           }
640       }

642       /**
643        * Parse a value according to its expected type.
644        * @param name  The config name
645        * @param value The config value
646        * @param type  The expected type
647        * @return The parsed object
648        */
649       public static Object parseType(String name, Object value, Type type) {
650           try {
651               if (value == null) return null;
652   
653               String trimmed = null;
654               if (value instanceof String)
655                   trimmed = ((String) value).trim();
656   
657               switch (type) {
658                   case BOOLEAN:
659                       if (value instanceof String) {
660                           if (trimmed.equalsIgnoreCase("true"))
661                               return true;
662                           else if (trimmed.equalsIgnoreCase("false"))
663                               return false;
664                           else
665                               throw new ConfigException(name, value, "Expected value to be either true or false");
666                       } else if (value instanceof Boolean)
667                           return value;
668                       else
669                           throw new ConfigException(name, value, "Expected value to be either true or false");
670                   case PASSWORD:
671                       if (value instanceof Password)
672                           return value;
673                       else if (value instanceof String)
674                           return new Password(trimmed);
675                       else
676                           throw new ConfigException(name, value, "Expected value to be a string, but it was a " + value.getClass().getName());
677                   case STRING:
678                       if (value instanceof String)
679                           return trimmed;
680                       else
681                           throw new ConfigException(name, value, "Expected value to be a string, but it was a " + value.getClass().getName());
682                   case INT:
683                       if (value instanceof Integer) {
684                           return value;
685                       } else if (value instanceof String) {
686                           return Integer.parseInt(trimmed);
687                       } else {
688                           throw new ConfigException(name, value, "Expected value to be a 32-bit integer, but it was a " + value.getClass().getName());
689                       }
690                   case SHORT:
691                       if (value instanceof Short) {
692                           return value;
693                       } else if (value instanceof String) {
694                           return Short.parseShort(trimmed);
695                       } else {
696                           throw new ConfigException(name, value, "Expected value to be a 16-bit integer (short), but it was a " + value.getClass().getName());
697                       }
698                   case LONG:
699                       if (value instanceof Integer)
700                           return ((Integer) value).longValue();
701                       if (value instanceof Long)
702                           return value;
703                       else if (value instanceof String)
704                           return Long.parseLong(trimmed);
705                       else
706                           throw new ConfigException(name, value, "Expected value to be a 64-bit integer (long), but it was a " + value.getClass().getName());
707                   case DOUBLE:
708                       if (value instanceof Number)
709                           return ((Number) value).doubleValue();
710                       else if (value instanceof String)
711                           return Double.parseDouble(trimmed);
712                       else
713                           throw new ConfigException(name, value, "Expected value to be a double, but it was a " + value.getClass().getName());
714                   case LIST:
715                       if (value instanceof List)
716                           return value;
717                       else if (value instanceof String)
718                           if (trimmed.isEmpty())
719                               return Collections.emptyList();
720                           else
721                               return Arrays.asList(COMMA_WITH_WHITESPACE.split(trimmed, -1));
722                       else
723                           throw new ConfigException(name, value, "Expected a comma separated list.");
724                   case CLASS:
725                       if (value instanceof Class)
726                           return value;
727                       else if (value instanceof String) {
728                           ClassLoader contextOrKafkaClassLoader = Utils.getContextOrKafkaClassLoader();
729                           // Use loadClass here instead of Class.forName because the name we use here may be an alias
730                           // and not match the name of the class that gets loaded. If that happens, Class.forName can
731                           // throw an exception.
732                           Class<?> klass = contextOrKafkaClassLoader.loadClass(trimmed);
733                           // Invoke forName here with the true name of the requested class to cause class
734                           // initialization to take place.
735                           return Class.forName(klass.getName(), true, contextOrKafkaClassLoader);
736                       } else
737                           throw new ConfigException(name, value, "Expected a Class instance or class name.");
738                   default:
739                       throw new IllegalStateException("Unknown type.");
740               }
741           } catch (NumberFormatException e) {
742               throw new ConfigException(name, value, "Not a number of type " + type);
743           } catch (ClassNotFoundException e) {
744               throw new ConfigException(name, value, "Class " + value + " could not be found.");
745           }
746       }

748       public static String convertToString(Object parsedValue, Type type) {
749           if (parsedValue == null) {
750               return null;
751           }
752   
753           if (type == null) {
754               return parsedValue.toString();
755           }
756   
757           switch (type) {
758               case BOOLEAN:
759               case SHORT:
760               case INT:
761               case LONG:
762               case DOUBLE:
763               case STRING:
764               case PASSWORD:
765                   return parsedValue.toString();
766               case LIST:
767                   List<?> valueList = (List<?>) parsedValue;
768                   return Utils.join(valueList, ",");
769               case CLASS:
770                   Class<?> clazz = (Class<?>) parsedValue;
771                   return clazz.getName();
772               default:
773                   throw new IllegalStateException("Unknown type.");
774           }
