501       private void postProcessAndValidateIdempotenceConfigs(final Map<String, Object> configs) {
502           final Map<String, Object> originalConfigs = this.originals();
503           final String acksStr = parseAcks(this.getString(ACKS_CONFIG));
504           configs.put(ACKS_CONFIG, acksStr);
505           final boolean userConfiguredIdempotence = this.originals().containsKey(ENABLE_IDEMPOTENCE_CONFIG);
506           boolean idempotenceEnabled = this.getBoolean(ENABLE_IDEMPOTENCE_CONFIG);
507           boolean shouldDisableIdempotence = false;
508   
509           // For idempotence producers, values for `retries` and `acks` and `max.in.flight.requests.per.connection` need validation
510           if (idempotenceEnabled) {
511               final int retries = this.getInt(RETRIES_CONFIG);
512               if (retries == 0) {
513                   if (userConfiguredIdempotence) {
514                       throw new ConfigException("Must set " + RETRIES_CONFIG + " to non-zero when using the idempotent producer.");
515                   }
516                   log.info("Idempotence will be disabled because {} is set to 0.", RETRIES_CONFIG);
517                   shouldDisableIdempotence = true;
518               }
519   
520               final short acks = Short.parseShort(acksStr);
521               if (acks != (short) -1) {
522                   if (userConfiguredIdempotence) {
523                       throw new ConfigException("Must set " + ACKS_CONFIG + " to all in order to use the idempotent " +
524                           "producer. Otherwise we cannot guarantee idempotence.");
525                   }
526                   log.info("Idempotence will be disabled because {} is set to {}, not set to 'all'.", ACKS_CONFIG, acks);
527                   shouldDisableIdempotence = true;
528               }
529   
530               final int inFlightConnection = this.getInt(MAX_IN_FLIGHT_REQUESTS_PER_CONNECTION);
531               if (MAX_IN_FLIGHT_REQUESTS_PER_CONNECTION_FOR_IDEMPOTENCE < inFlightConnection) {
532                   if (userConfiguredIdempotence) {
533                       throw new ConfigException("Must set " + MAX_IN_FLIGHT_REQUESTS_PER_CONNECTION + " to at most 5" +
534                           " to use the idempotent producer.");
535                   }
536                   log.warn("Idempotence will be disabled because {} is set to {}, which is greater than 5. " +
537                       "Please note that in v4.0.0 and onward, this will become an error.", MAX_IN_FLIGHT_REQUESTS_PER_CONNECTION, inFlightConnection);
538                   shouldDisableIdempotence = true;
539               }
540           }
541   
542           if (shouldDisableIdempotence) {
543               configs.put(ENABLE_IDEMPOTENCE_CONFIG, false);
544               idempotenceEnabled = false;
545           }
546   
547           // validate `transaction.id` after validating idempotence dependant configs because `enable.idempotence` config might be overridden
548           boolean userConfiguredTransactions = originalConfigs.containsKey(TRANSACTIONAL_ID_CONFIG);
549           if (!idempotenceEnabled && userConfiguredTransactions) {
550               throw new ConfigException("Cannot set a " + ProducerConfig.TRANSACTIONAL_ID_CONFIG + " without also enabling idempotence.");
551           }
552       }
