49   public class DescribeProducersHandler extends AdminApiHandler.Batched<TopicPartition, PartitionProducerState> {
50       private final Logger log;
51       private final DescribeProducersOptions options;
52       private final AdminApiLookupStrategy<TopicPartition> lookupStrategy;
53   
54       public DescribeProducersHandler(
55           DescribeProducersOptions options,
56           LogContext logContext
57       ) {
58           this.options = options;
59           this.log = logContext.logger(DescribeProducersHandler.class);
60   
61           if (options.brokerId().isPresent()) {
62               this.lookupStrategy = new StaticBrokerStrategy<>(options.brokerId().getAsInt());
63           } else {
64               this.lookupStrategy = new PartitionLeaderStrategy(logContext);
65           }
66       }

68       public static AdminApiFuture.SimpleAdminApiFuture<TopicPartition, PartitionProducerState> newFuture(
69           Collection<TopicPartition> topicPartitions
70       ) {
71           return AdminApiFuture.forKeys(new HashSet<>(topicPartitions));
72       }

76           return "describeProducers";
77       }

84       @Override
85       public DescribeProducersRequest.Builder buildBatchedRequest(
86           int brokerId,
87           Set<TopicPartition> topicPartitions
88       ) {
89           DescribeProducersRequestData request = new DescribeProducersRequestData();
90           DescribeProducersRequest.Builder builder = new DescribeProducersRequest.Builder(request);
91   
92           CollectionUtils.groupPartitionsByTopic(
93               topicPartitions,
94               builder::addTopic,
95               (topicRequest, partitionId) -> topicRequest.partitionIndexes().add(partitionId)

99       }

101       private void handlePartitionError(
102           TopicPartition topicPartition,
103           ApiError apiError,
104           Map<TopicPartition, Throwable> failed,
105           List<TopicPartition> unmapped
106       ) {
107           switch (apiError.error()) {
108               case NOT_LEADER_OR_FOLLOWER:
109                   if (options.brokerId().isPresent()) {
110                       // Typically these errors are retriable, but if the user specified the brokerId
111                       // explicitly, then they are fatal.
112                       int brokerId = options.brokerId().getAsInt();
113                       log.error("Not leader error in `DescribeProducers` response for partition {} " +
114                           "for brokerId {} set in options", topicPartition, brokerId, apiError.exception());
115                       failed.put(topicPartition, apiError.error().exception("Failed to describe active producers " +
116                           "for partition " + topicPartition + " on brokerId " + brokerId));
117                   } else {
118                       // Otherwise, we unmap the partition so that we can find the new leader
119                       log.debug("Not leader error in `DescribeProducers` response for partition {}. " +
120                           "Will retry later.", topicPartition);
121                       unmapped.add(topicPartition);
122                   }
123                   break;
124   
125               case UNKNOWN_TOPIC_OR_PARTITION:
126                   log.debug("Unknown topic/partition error in `DescribeProducers` response for partition {}. " +
127                       "Will retry later.", topicPartition);
128                   break;
129   
130               case INVALID_TOPIC_EXCEPTION:
131                   log.error("Invalid topic in `DescribeProducers` response for partition {}",
132                       topicPartition, apiError.exception());
133                   failed.put(topicPartition, new InvalidTopicException(
134                       "Failed to fetch metadata for partition " + topicPartition
135                           + " due to invalid topic error: " + apiError.messageWithFallback(),
136                       Collections.singleton(topicPartition.topic())));
137                   break;
138   
139               case TOPIC_AUTHORIZATION_FAILED:
140                   log.error("Authorization failed in `DescribeProducers` response for partition {}",
141                       topicPartition, apiError.exception());
142                   failed.put(topicPartition, new TopicAuthorizationException("Failed to describe " +
143                       "active producers for partition " + topicPartition + " due to authorization failure on topic" +
144                       " `" + topicPartition.topic() + "`", Collections.singleton(topicPartition.topic())));
145                   break;
146   
147               default:
148                   log.error("Unexpected error in `DescribeProducers` response for partition {}",
149                       topicPartition, apiError.exception());
150                   failed.put(topicPartition, apiError.error().exception("Failed to describe active " +
151                       "producers for partition " + topicPartition + " due to unexpected error"));
152                   break;
153           }
154       }

156       @Override
157       public ApiResult<TopicPartition, PartitionProducerState> handleResponse(
158           Node broker,
159           Set<TopicPartition> keys,
160           AbstractResponse abstractResponse
161       ) {
162           DescribeProducersResponse response = (DescribeProducersResponse) abstractResponse;
163           Map<TopicPartition, PartitionProducerState> completed = new HashMap<>();
164           Map<TopicPartition, Throwable> failed = new HashMap<>();
165           List<TopicPartition> unmapped = new ArrayList<>();
166   
167           for (DescribeProducersResponseData.TopicResponse topicResponse : response.data().topics()) {
168               for (DescribeProducersResponseData.PartitionResponse partitionResponse : topicResponse.partitions()) {
169                   TopicPartition topicPartition = new TopicPartition(
170                       topicResponse.name(), partitionResponse.partitionIndex());
171   
172                   Errors error = Errors.forCode(partitionResponse.errorCode());
173                   if (error != Errors.NONE) {
174                       ApiError apiError = new ApiError(error, partitionResponse.errorMessage());
175                       handlePartitionError(topicPartition, apiError, failed, unmapped);
176                       continue;
177                   }
178   
179                   List<ProducerState> activeProducers = partitionResponse.activeProducers().stream()
180                       .map(activeProducer -> {
181                           OptionalLong currentTransactionFirstOffset =
182                               activeProducer.currentTxnStartOffset() < 0 ?
183                                   OptionalLong.empty() :
184                                   OptionalLong.of(activeProducer.currentTxnStartOffset());
185                           OptionalInt coordinatorEpoch =
186                               activeProducer.coordinatorEpoch() < 0 ?
187                                   OptionalInt.empty() :
188                                   OptionalInt.of(activeProducer.coordinatorEpoch());
189   
190                           return new ProducerState(
191                               activeProducer.producerId(),
192                               activeProducer.producerEpoch(),
193                               activeProducer.lastSequence(),
194                               activeProducer.lastTimestamp(),
195                               coordinatorEpoch,
196                               currentTransactionFirstOffset
197                           );
198                       }).collect(Collectors.toList());
199   
200                   completed.put(topicPartition, new PartitionProducerState(activeProducers));
201               }
202           }
203           return new ApiResult<>(completed, failed, unmapped);
204       }
