216       /**
217        * Add a record to the accumulator, return the append result
218        * <p>
219        * The append result will contain the future metadata, and flag for whether the appended batch is full or a new batch is created
220        * <p>
221        *
222        * @param topic The topic to which this record is being sent
223        * @param partition The partition to which this record is being sent or RecordMetadata.UNKNOWN_PARTITION
224        *                  if any partition could be used
225        * @param timestamp The timestamp of the record
226        * @param key The key for the record
227        * @param value The value for the record
228        * @param headers the Headers for the record
229        * @param callbacks The callbacks to execute
230        * @param maxTimeToBlock The maximum time in milliseconds to block for buffer memory to be available
231        * @param abortOnNewBatch A boolean that indicates returning before a new batch is created and
232        *                        running the partitioner's onNewBatch method before trying to append again
233        * @param nowMs The current time, in milliseconds
234        * @param cluster The cluster metadata
235        */
236       public RecordAppendResult append(String topic,
237                                        int partition,
238                                        long timestamp,
239                                        byte[] key,
240                                        byte[] value,
241                                        Header[] headers,
242                                        AppendCallbacks callbacks,
243                                        long maxTimeToBlock,
244                                        boolean abortOnNewBatch,
245                                        long nowMs,
246                                        Cluster cluster) throws InterruptedException {
247           TopicInfo topicInfo = topicInfoMap.computeIfAbsent(topic, k -> new TopicInfo(logContext, k, batchSize));
248   
249           // We keep track of the number of appending thread to make sure we do not miss batches in
250           // abortIncompleteBatches().
251           appendsInProgress.incrementAndGet();
252           ByteBuffer buffer = null;
253           if (headers == null) headers = Record.EMPTY_HEADERS;
254           try {
255               // Loop to retry in case we encounter partitioner's race conditions.
256               while (true) {
257                   // If the message doesn't have any partition affinity, so we pick a partition based on the broker
258                   // availability and performance.  Note, that here we peek current partition before we hold the
259                   // deque lock, so we'll need to make sure that it's not changed while we were waiting for the
260                   // deque lock.
261                   final BuiltInPartitioner.StickyPartitionInfo partitionInfo;
262                   final int effectivePartition;
263                   if (partition == RecordMetadata.UNKNOWN_PARTITION) {
264                       partitionInfo = topicInfo.builtInPartitioner.peekCurrentPartitionInfo(cluster);
265                       effectivePartition = partitionInfo.partition();
266                   } else {
267                       partitionInfo = null;
268                       effectivePartition = partition;
269                   }
270   
271                   // Now that we know the effective partition, let the caller know.
272                   setPartition(callbacks, effectivePartition);
273   
274                   // check if we have an in-progress batch
275                   Deque<ProducerBatch> dq = topicInfo.batches.computeIfAbsent(effectivePartition, k -> new ArrayDeque<>());
276                   synchronized (dq) {
277                       // After taking the lock, validate that the partition hasn't changed and retry.
278                       if (topicInfo.builtInPartitioner.isPartitionChanged(partitionInfo)) {
279                           log.trace("Partition {} for topic {} switched by a concurrent append, retrying",
280                                   partitionInfo.partition(), topic);
281                           continue;
282                       }
283                       RecordAppendResult appendResult = tryAppend(timestamp, key, value, headers, callbacks, dq, nowMs);
284                       if (appendResult != null) {
285                           topicInfo.builtInPartitioner.updatePartitionInfo(partitionInfo, appendResult.appendedBytes, cluster);
286                           return appendResult;
287                       }
288                   }
289   
290                   // we don't have an in-progress record batch try to allocate a new batch
291                   if (abortOnNewBatch) {
292                       // Return a result that will cause another call to append.
293                       return new RecordAppendResult(null, false, false, true, 0);
294                   }
295   
296                   if (buffer == null) {
297                       byte maxUsableMagic = apiVersions.maxUsableProduceMagic();
298                       int size = Math.max(this.batchSize, AbstractRecords.estimateSizeInBytesUpperBound(maxUsableMagic, compression, key, value, headers));
299                       log.trace("Allocating a new {} byte message buffer for topic {} partition {} with remaining timeout {}ms", size, topic, partition, maxTimeToBlock);
300                       buffer = free.allocate(size, maxTimeToBlock);
301                   }
302   
303                   synchronized (dq) {
304                       // After taking the lock, validate that the partition hasn't changed and retry.
305                       if (topicInfo.builtInPartitioner.isPartitionChanged(partitionInfo)) {
306                           log.trace("Partition {} for topic {} switched by a concurrent append, retrying",
307                                   partitionInfo.partition(), topic);
308                           continue;
309                       }
310                       RecordAppendResult appendResult = appendNewBatch(topic, effectivePartition, dq, timestamp, key, value, headers, callbacks, buffer);
311                       // Set buffer to null, so that deallocate doesn't return it back to free pool, since it's used in the batch.
312                       if (appendResult.newBatchCreated)
313                           buffer = null;
314                       topicInfo.builtInPartitioner.updatePartitionInfo(partitionInfo, appendResult.appendedBytes, cluster);
315                       return appendResult;
316                   }
317               }
318           } finally {
319               free.deallocate(buffer);
320               appendsInProgress.decrementAndGet();
321           }
322       }

324       /**
325        * Append a new batch to the queue
326        *
327        * @param topic The topic
328        * @param partition The partition (cannot be RecordMetadata.UNKNOWN_PARTITION)
329        * @param dq The queue
330        * @param timestamp The timestamp of the record
331        * @param key The key for the record
332        * @param value The value for the record
333        * @param headers the Headers for the record
334        * @param callbacks The callbacks to execute
335        * @param buffer The buffer for the new batch
336        */
337       private RecordAppendResult appendNewBatch(String topic,
338                                                 int partition,
339                                                 Deque<ProducerBatch> dq,
340                                                 long timestamp,
341                                                 byte[] key,
342                                                 byte[] value,
343                                                 Header[] headers,
344                                                 AppendCallbacks callbacks,
345                                                 ByteBuffer buffer) {
346           assert partition != RecordMetadata.UNKNOWN_PARTITION;
347   
348           // Update the current time in case the buffer allocation blocked above.
349           long nowMs = time.milliseconds();
350           RecordAppendResult appendResult = tryAppend(timestamp, key, value, headers, callbacks, dq, nowMs);
351           if (appendResult != null) {
352               // Somebody else found us a batch, return the one we waited for! Hopefully this doesn't happen often...
353               return appendResult;
354           }
355   
356           MemoryRecordsBuilder recordsBuilder = recordsBuilder(buffer, apiVersions.maxUsableProduceMagic());
357           ProducerBatch batch = new ProducerBatch(new TopicPartition(topic, partition), recordsBuilder, nowMs);
358           FutureRecordMetadata future = Objects.requireNonNull(batch.tryAppend(timestamp, key, value, headers,
359                   callbacks, nowMs));
360   
361           dq.addLast(batch);
362           incomplete.add(batch);
363   
364           return new RecordAppendResult(future, dq.size() > 1 || batch.isFull(), true, false, batch.estimatedSizeInBytes());
365       }

367       private MemoryRecordsBuilder recordsBuilder(ByteBuffer buffer, byte maxUsableMagic) {
368           if (transactionManager != null && maxUsableMagic < RecordBatch.MAGIC_VALUE_V2) {
369               throw new UnsupportedVersionException("Attempting to use idempotence with a broker which does not " +
370                   "support the required message format (v2). The broker must be version 0.11 or later.");
371           }
372           return MemoryRecords.builder(buffer, maxUsableMagic, compression, TimestampType.CREATE_TIME, 0L);
373       }

375       /**
376        *  Try to append to a ProducerBatch.
377        *
378        *  If it is full, we return null and a new batch is created. We also close the batch for record appends to free up
379        *  resources like compression buffers. The batch will be fully closed (ie. the record batch headers will be written
380        *  and memory records built) in one of the following cases (whichever comes first): right before send,
381        *  if it is expired, or when the producer is closed.
382        */
383       private RecordAppendResult tryAppend(long timestamp, byte[] key, byte[] value, Header[] headers,
384                                            Callback callback, Deque<ProducerBatch> deque, long nowMs) {
385           if (closed)
386               throw new KafkaException("Producer closed while send in progress");
387           ProducerBatch last = deque.peekLast();
388           if (last != null) {
389               int initialBytes = last.estimatedSizeInBytes();
390               FutureRecordMetadata future = last.tryAppend(timestamp, key, value, headers, callback, nowMs);
391               if (future == null) {
392                   last.closeForRecordAppends();
393               } else {
394                   int appendedBytes = last.estimatedSizeInBytes() - initialBytes;
395                   return new RecordAppendResult(future, deque.size() > 1 || last.isFull(), false, false, appendedBytes);
396               }
397           }
398           return null;
399       }

414           } else {
415               log.warn("Skipping next batch expiry time update due to addition overflow: "
416                   + "batch.createMs={}, deliveryTimeoutMs={}", batch.createdMs, deliveryTimeoutMs);
417           }

438                           }
439                       }
440                   }
441               }

450       /**
451        * Re-enqueue the given record batch in the accumulator. In Sender.completeBatch method, we check
452        * whether the batch has reached deliveryTimeoutMs or not. Hence we do not do the delivery timeout check here.
453        */
454       public void reenqueue(ProducerBatch batch, long now) {
455           batch.reenqueued(now);
456           Deque<ProducerBatch> deque = getOrCreateDeque(batch.topicPartition);
457           synchronized (deque) {
458               if (transactionManager != null)
459                   insertInSequenceOrder(deque, batch);
460               else
461                   deque.addFirst(batch);
462           }

465       /**
466        * Split the big batch that has been rejected and reenqueue the split batches in to the accumulator.
467        * @return the number of split batches.
468        */
469       public int splitAndReenqueue(ProducerBatch bigBatch) {
470           // Reset the estimated compression ratio to the initial value or the big batch compression ratio, whichever
471           // is bigger. There are several different ways to do the reset. We chose the most conservative one to ensure
472           // the split doesn't happen too often.
473           CompressionRatioEstimator.setEstimation(bigBatch.topicPartition.topic(), compression,
474                                                   Math.max(1.0f, (float) bigBatch.compressionRatio()));
475           Deque<ProducerBatch> dq = bigBatch.split(this.batchSize);
476           int numSplitBatches = dq.size();
477           Deque<ProducerBatch> partitionDequeue = getOrCreateDeque(bigBatch.topicPartition);
478           while (!dq.isEmpty()) {
479               ProducerBatch batch = dq.pollLast();
480               incomplete.add(batch);
481               // We treat the newly split batches as if they are not even tried.
482               synchronized (partitionDequeue) {
483                   if (transactionManager != null) {
484                       // We should track the newly created batches since they already have assigned sequences.
485                       transactionManager.addInFlightBatch(batch);
486                       insertInSequenceOrder(partitionDequeue, batch);
487                   } else {
488                       partitionDequeue.addFirst(batch);
489                   }
490               }
491           }
492           return numSplitBatches;
493       }
494   
495       // We will have to do extra work to ensure the queue is in order when requests are being retried and there are
496       // multiple requests in flight to that partition. If the first in flight request fails to append, then all the
497       // subsequent in flight requests will also fail because the sequence numbers will not be accepted.
498       //
499       // Further, once batches are being retried, we are reduced to a single in flight request for that partition. So when
500       // the subsequent batches come back in sequence order, they will have to be placed further back in the queue.
501       //
502       // Note that this assumes that all the batches in the queue which have an assigned sequence also have the current
503       // producer id. We will not attempt to reorder messages if the producer id has changed, we will throw an
504       // IllegalStateException instead.
505       private void insertInSequenceOrder(Deque<ProducerBatch> deque, ProducerBatch batch) {
506           // When we are re-enqueueing and have enabled idempotence, the re-enqueued batch must always have a sequence.
507           if (batch.baseSequence() == RecordBatch.NO_SEQUENCE)
508               throw new IllegalStateException("Trying to re-enqueue a batch which doesn't have a sequence even " +
509                   "though idempotency is enabled.");
510   
511           if (!transactionManager.hasInflightBatches(batch.topicPartition))
512               throw new IllegalStateException("We are re-enqueueing a batch which is not tracked as part of the in flight " +
513                   "requests. batch.topicPartition: " + batch.topicPartition + "; batch.baseSequence: " + batch.baseSequence());
514   
515           ProducerBatch firstBatchInQueue = deque.peekFirst();
516           if (firstBatchInQueue != null && firstBatchInQueue.hasSequence() && firstBatchInQueue.baseSequence() < batch.baseSequence()) {
517               // The incoming batch can't be inserted at the front of the queue without violating the sequence ordering.
518               // This means that the incoming batch should be placed somewhere further back.
519               // We need to find the right place for the incoming batch and insert it there.
520               // We will only enter this branch if we have multiple inflights sent to different brokers and we need to retry
521               // the inflight batches.
522               //
523               // Since we reenqueue exactly one batch a time and ensure that the queue is ordered by sequence always, it
524               // is a simple linear scan of a subset of the in flight batches to find the right place in the queue each time.
525               List<ProducerBatch> orderedBatches = new ArrayList<>();
526               while (deque.peekFirst() != null && deque.peekFirst().hasSequence() && deque.peekFirst().baseSequence() < batch.baseSequence())
527                   orderedBatches.add(deque.pollFirst());
528   
529               log.debug("Reordered incoming batch with sequence {} for partition {}. It was placed in the queue at " +
530                   "position {}", batch.baseSequence(), batch.topicPartition, orderedBatches.size());
531               // Either we have reached a point where there are batches without a sequence (ie. never been drained
532               // and are hence in order by default), or the batch at the front of the queue has a sequence greater
533               // than the incoming batch. This is the right place to add the incoming batch.
534               deque.addFirst(batch);
535   
536               // Now we have to re insert the previously queued batches in the right order.
537               for (int i = orderedBatches.size() - 1; i >= 0; --i) {
538                   deque.addFirst(orderedBatches.get(i));
539               }
540   
541               // At this point, the incoming batch has been queued in the correct place according to its sequence.
542           } else {
543               deque.addFirst(batch);
544           }
545       }

547       /**
548        * Add the leader to the ready nodes if the batch is ready
549        *
550        * @param nowMs The current time
551        * @param exhausted 'true' is the buffer pool is exhausted
552        * @param part The partition
553        * @param leader The leader for the partition
554        * @param waitedTimeMs How long batch waited
555        * @param backingOff Is backing off
556        * @param full Is batch full
557        * @param nextReadyCheckDelayMs The delay for next check
558        * @param readyNodes The set of ready nodes (to be filled in)
559        * @return The delay for next check
560        */
561       private long batchReady(long nowMs, boolean exhausted, TopicPartition part, Node leader,
562                               long waitedTimeMs, boolean backingOff, boolean full,
563                               long nextReadyCheckDelayMs, Set<Node> readyNodes) {
564           if (!readyNodes.contains(leader) && !isMuted(part)) {
565               long timeToWaitMs = backingOff ? retryBackoffMs : lingerMs;
566               boolean expired = waitedTimeMs >= timeToWaitMs;
567               boolean transactionCompleting = transactionManager != null && transactionManager.isCompleting();
568               boolean sendable = full
569                       || expired
570                       || exhausted
571                       || closed
572                       || flushInProgress()
573                       || transactionCompleting;
574               if (sendable && !backingOff) {
575                   readyNodes.add(leader);
576               } else {
577                   long timeLeftMs = Math.max(timeToWaitMs - waitedTimeMs, 0);
578                   // Note that this results in a conservative estimate since an un-sendable partition may have
579                   // a leader that will later be found to have sendable data. However, this is good enough
580                   // since we'll just wake up and then sleep again for the remaining time.
581                   nextReadyCheckDelayMs = Math.min(timeLeftMs, nextReadyCheckDelayMs);
582               }
583           }
584           return nextReadyCheckDelayMs;
585       }

587       /**
588        * Iterate over partitions to see which one have batches ready and collect leaders of those partitions
589        * into the set of ready nodes.  If partition has no leader, add the topic to the set of topics with
590        * no leader.  This function also calculates stats for adaptive partitioning.
591        *
592        * @param cluster The cluster metadata
593        * @param nowMs The current time
594        * @param topic The topic
595        * @param topicInfo The topic info
596        * @param nextReadyCheckDelayMs The delay for next check
597        * @param readyNodes The set of ready nodes (to be filled in)
598        * @param unknownLeaderTopics The set of topics with no leader (to be filled in)
599        * @return The delay for next check
600        */
601       private long partitionReady(Cluster cluster, long nowMs, String topic,
602                                   TopicInfo topicInfo,
603                                   long nextReadyCheckDelayMs, Set<Node> readyNodes, Set<String> unknownLeaderTopics) {
604           ConcurrentMap<Integer, Deque<ProducerBatch>> batches = topicInfo.batches;
605           // Collect the queue sizes for available partitions to be used in adaptive partitioning.
606           int[] queueSizes = null;
607           int[] partitionIds = null;
608           if (enableAdaptivePartitioning && batches.size() >= cluster.partitionsForTopic(topic).size()) {
609               // We don't do adaptive partitioning until we scheduled at least a batch for all
610               // partitions (i.e. we have the corresponding entries in the batches map), we just
611               // do uniform.  The reason is that we build queue sizes from the batches map,
612               // and if an entry is missing in the batches map, then adaptive partitioning logic
613               // won't know about it and won't switch to it.
614               queueSizes = new int[batches.size()];
615               partitionIds = new int[queueSizes.length];
616           }
617   
618           int queueSizesIndex = -1;
619           boolean exhausted = this.free.queued() > 0;
620           for (Map.Entry<Integer, Deque<ProducerBatch>> entry : batches.entrySet()) {
621               TopicPartition part = new TopicPartition(topic, entry.getKey());
622               // Advance queueSizesIndex so that we properly index available
623               // partitions.  Do it here so that it's done for all code paths.
624               Node leader = cluster.leaderFor(part);
625               if (leader != null && queueSizes != null) {
626                   ++queueSizesIndex;
627                   assert queueSizesIndex < queueSizes.length;
628                   partitionIds[queueSizesIndex] = part.partition();
629               }
630   
631               Deque<ProducerBatch> deque = entry.getValue();
632   
633               final long waitedTimeMs;
634               final boolean backingOff;
635               final int dequeSize;
636               final boolean full;
637   
638               // This loop is especially hot with large partition counts.
639   
640               // We are careful to only perform the minimum required inside the
641               // synchronized block, as this lock is also used to synchronize producer threads
642               // attempting to append() to a partition/batch.
643   
644               synchronized (deque) {
645                   // Deques are often empty in this path, esp with large partition counts,
646                   // so we exit early if we can.
647                   ProducerBatch batch = deque.peekFirst();
648                   if (batch == null) {
649                       continue;
650                   }
651   
652                   waitedTimeMs = batch.waitedTimeMs(nowMs);
653                   backingOff = batch.attempts() > 0 && waitedTimeMs < retryBackoffMs;
654                   dequeSize = deque.size();
655                   full = dequeSize > 1 || batch.isFull();
656               }
657   
658               if (leader == null) {
659                   // This is a partition for which leader is not known, but messages are available to send.
660                   // Note that entries are currently not removed from batches when deque is empty.
661                   unknownLeaderTopics.add(part.topic());
662               } else {
663                   if (queueSizes != null)
664                       queueSizes[queueSizesIndex] = dequeSize;
665                   if (partitionAvailabilityTimeoutMs > 0) {
666                       // Check if we want to exclude the partition from the list of available partitions
667                       // if the broker hasn't responded for some time.
668                       NodeLatencyStats nodeLatencyStats = nodeStats.get(leader.id());
669                       if (nodeLatencyStats != null) {
670                           // NOTE: there is no synchronization between reading metrics,
671                           // so we read ready time first to avoid accidentally marking partition
672                           // unavailable if we read while the metrics are being updated.
673                           long readyTimeMs = nodeLatencyStats.readyTimeMs;
674                           if (readyTimeMs - nodeLatencyStats.drainTimeMs > partitionAvailabilityTimeoutMs)
675                               --queueSizesIndex;
676                       }
677                   }
678   
679                   nextReadyCheckDelayMs = batchReady(nowMs, exhausted, part, leader, waitedTimeMs, backingOff,
680                       full, nextReadyCheckDelayMs, readyNodes);
681               }
682           }
683   
684           // We've collected the queue sizes for partitions of this topic, now we can calculate
685           // load stats.  NOTE: the stats are calculated in place, modifying the
686           // queueSizes array.
687           topicInfo.builtInPartitioner.updatePartitionLoadStats(queueSizes, partitionIds, queueSizesIndex + 1);
688           return nextReadyCheckDelayMs;
689       }
690   
691       /**
692        * Get a list of nodes whose partitions are ready to be sent, and the earliest time at which any non-sendable
693        * partition will be ready; Also return the flag for whether there are any unknown leaders for the accumulated
694        * partition batches.
695        * <p>
696        * A destination node is ready to send data if:
697        * <ol>
698        * <li>There is at least one partition that is not backing off its send
699        * <li><b>and</b> those partitions are not muted (to prevent reordering if
700        *   {@value org.apache.kafka.clients.producer.ProducerConfig#MAX_IN_FLIGHT_REQUESTS_PER_CONNECTION}
701        *   is set to one)</li>
702        * <li><b>and <i>any</i></b> of the following are true</li>
703        * <ul>
704        *     <li>The record set is full</li>
705        *     <li>The record set has sat in the accumulator for at least lingerMs milliseconds</li>
706        *     <li>The accumulator is out of memory and threads are blocking waiting for data (in this case all partitions
707        *     are immediately considered ready).</li>
708        *     <li>The accumulator has been closed</li>
709        * </ul>
710        * </ol>
711        */
712       public ReadyCheckResult ready(Cluster cluster, long nowMs) {
713           Set<Node> readyNodes = new HashSet<>();
714           long nextReadyCheckDelayMs = Long.MAX_VALUE;
715           Set<String> unknownLeaderTopics = new HashSet<>();
716           // Go topic by topic so that we can get queue sizes for partitions in a topic and calculate
717           // cumulative frequency table (used in partitioner).
718           for (Map.Entry<String, TopicInfo> topicInfoEntry : this.topicInfoMap.entrySet()) {
719               final String topic = topicInfoEntry.getKey();
720               nextReadyCheckDelayMs = partitionReady(cluster, nowMs, topic, topicInfoEntry.getValue(), nextReadyCheckDelayMs, readyNodes, unknownLeaderTopics);
721           }
722           return new ReadyCheckResult(readyNodes, nextReadyCheckDelayMs, unknownLeaderTopics);
723       }

731                   synchronized (deque) {
732                       if (!deque.isEmpty())
733                           return true;
734                   }
735               }
736           }

740       private boolean shouldStopDrainBatchesForPartition(ProducerBatch first, TopicPartition tp) {
741           ProducerIdAndEpoch producerIdAndEpoch = null;
742           if (transactionManager != null) {
743               if (!transactionManager.isSendToPartitionAllowed(tp))
744                   return true;
745   
746               producerIdAndEpoch = transactionManager.producerIdAndEpoch();
747               if (!producerIdAndEpoch.isValid())
748                   // we cannot send the batch until we have refreshed the producer id
749                   return true;
750   
751               if (!first.hasSequence()) {
752                   if (transactionManager.hasInflightBatches(tp) && transactionManager.hasStaleProducerIdAndEpoch(tp)) {
753                       // Don't drain any new batches while the partition has in-flight batches with a different epoch
754                       // and/or producer ID. Otherwise, a batch with a new epoch and sequence number
755                       // 0 could be written before earlier batches complete, which would cause out of sequence errors
756                       return true;
757                   }
758   
759                   if (transactionManager.hasUnresolvedSequence(first.topicPartition))
760                       // Don't drain any new batches while the state of previous sequence numbers
761                       // is unknown. The previous batches would be unknown if they were aborted
762                       // on the client after being sent to the broker at least once.
763                       return true;
764               }
765   
766               int firstInFlightSequence = transactionManager.firstInFlightSequence(first.topicPartition);
767               if (firstInFlightSequence != RecordBatch.NO_SEQUENCE && first.hasSequence()
768                   && first.baseSequence() != firstInFlightSequence)
769                   // If the queued batch already has an assigned sequence, then it is being retried.
770                   // In this case, we wait until the next immediate batch is ready and drain that.
771                   // We only move on when the next in line batch is complete (either successfully or due to
772                   // a fatal broker error). This effectively reduces our in flight request count to 1.
773                   return true;
774           }
775           return false;
776       }

778       private List<ProducerBatch> drainBatchesForOneNode(Cluster cluster, Node node, int maxSize, long now) {
779           int size = 0;
780           List<PartitionInfo> parts = cluster.partitionsForNode(node.id());
781           List<ProducerBatch> ready = new ArrayList<>();
782           /* to make starvation less likely each node has it's own drainIndex */
783           int drainIndex = getDrainIndex(node.idString());
784           int start = drainIndex = drainIndex % parts.size();
785           do {
786               PartitionInfo part = parts.get(drainIndex);
787               TopicPartition tp = new TopicPartition(part.topic(), part.partition());
788               updateDrainIndex(node.idString(), drainIndex);
789               drainIndex = (drainIndex + 1) % parts.size();
790               // Only proceed if the partition has no in-flight batches.
791               if (isMuted(tp))
792                   continue;
793   
794               Deque<ProducerBatch> deque = getDeque(tp);
795               if (deque == null)
796                   continue;
797   
798               final ProducerBatch batch;
799               synchronized (deque) {
800                   // invariant: !isMuted(tp,now) && deque != null
801                   ProducerBatch first = deque.peekFirst();
802                   if (first == null)
803                       continue;
804   
805                   // first != null
806                   boolean backoff = first.attempts() > 0 && first.waitedTimeMs(now) < retryBackoffMs;
807                   // Only drain the batch if it is not during backoff period.
808                   if (backoff)
809                       continue;
810   
811                   if (size + first.estimatedSizeInBytes() > maxSize && !ready.isEmpty()) {
812                       // there is a rare case that a single batch size is larger than the request size due to
813                       // compression; in this case we will still eventually send this batch in a single request
814                       break;
815                   } else {
816                       if (shouldStopDrainBatchesForPartition(first, tp))
817                           break;
818                   }
819   
820                   batch = deque.pollFirst();
821   
822                   boolean isTransactional = transactionManager != null && transactionManager.isTransactional();
823                   ProducerIdAndEpoch producerIdAndEpoch =
824                       transactionManager != null ? transactionManager.producerIdAndEpoch() : null;
825                   if (producerIdAndEpoch != null && !batch.hasSequence()) {
826                       // If the producer id/epoch of the partition do not match the latest one
827                       // of the producer, we update it and reset the sequence. This should be
828                       // only done when all its in-flight batches have completed. This is guarantee
829                       // in `shouldStopDrainBatchesForPartition`.
830                       transactionManager.maybeUpdateProducerIdAndEpoch(batch.topicPartition);
831   
832                       // If the batch already has an assigned sequence, then we should not change the producer id and
833                       // sequence number, since this may introduce duplicates. In particular, the previous attempt
834                       // may actually have been accepted, and if we change the producer id and sequence here, this
835                       // attempt will also be accepted, causing a duplicate.
836                       //
837                       // Additionally, we update the next sequence number bound for the partition, and also have
838                       // the transaction manager track the batch so as to ensure that sequence ordering is maintained
839                       // even if we receive out of order responses.
840                       batch.setProducerState(producerIdAndEpoch, transactionManager.sequenceNumber(batch.topicPartition), isTransactional);
841                       transactionManager.incrementSequenceNumber(batch.topicPartition, batch.recordCount);
842                       log.debug("Assigned producerId {} and producerEpoch {} to batch with base sequence " +
843                               "{} being sent to partition {}", producerIdAndEpoch.producerId,
844                           producerIdAndEpoch.epoch, batch.baseSequence(), tp);
845   
846                       transactionManager.addInFlightBatch(batch);
847                   }
848               }
849   
850               // the rest of the work by processing outside the lock
851               // close() is particularly expensive
852   
853               batch.close();
854               size += batch.records().sizeInBytes();
855               ready.add(batch);
856   
857               batch.drained(now);
858           } while (start != drainIndex);
859           return ready;
860       }
