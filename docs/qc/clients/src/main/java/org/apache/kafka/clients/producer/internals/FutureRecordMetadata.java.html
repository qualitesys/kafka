17   package org.apache.kafka.clients.producer.internals;
18   
19   import org.apache.kafka.clients.producer.RecordMetadata;
20   import org.apache.kafka.common.utils.Time;
21   
22   import java.util.concurrent.ExecutionException;
23   import java.util.concurrent.Future;
24   import java.util.concurrent.TimeUnit;
25   import java.util.concurrent.TimeoutException;
26   
27   /**
28    * The future result of a record send
29    */
30   public final class FutureRecordMetadata implements Future<RecordMetadata> {
31   
32       private final ProduceRequestResult result;
33       private final int batchIndex;
34       private final long createTimestamp;
35       private final int serializedKeySize;
36       private final int serializedValueSize;
37       private final Time time;
38       private volatile FutureRecordMetadata nextRecordMetadata = null;
39   
40       public FutureRecordMetadata(ProduceRequestResult result, int batchIndex, long createTimestamp, int serializedKeySize,
41                                   int serializedValueSize, Time time) {
42           this.result = result;
43           this.batchIndex = batchIndex;
44           this.createTimestamp = createTimestamp;
45           this.serializedKeySize = serializedKeySize;
46           this.serializedValueSize = serializedValueSize;
47           this.time = time;
48       }
49   
50       @Override
51       public boolean cancel(boolean interrupt) {
52           return false;
53       }
54   
55       @Override
56       public boolean isCancelled() {
57           return false;
58       }
59   
60       @Override
61       public RecordMetadata get() throws InterruptedException, ExecutionException {
62           this.result.await();
63           if (nextRecordMetadata != null)
64               return nextRecordMetadata.get();
65           return valueOrError();
66       }
67   
68       @Override
69       public RecordMetadata get(long timeout, TimeUnit unit) throws InterruptedException, ExecutionException, TimeoutException {
70           // Handle overflow.
71           long now = time.milliseconds();
72           long timeoutMillis = unit.toMillis(timeout);
73           long deadline = Long.MAX_VALUE - timeoutMillis < now ? Long.MAX_VALUE : now + timeoutMillis;
74           boolean occurred = this.result.await(timeout, unit);
75           if (!occurred)
76               throw new TimeoutException("Timeout after waiting for " + timeoutMillis + " ms.");
77           if (nextRecordMetadata != null)
78               return nextRecordMetadata.get(deadline - time.milliseconds(), TimeUnit.MILLISECONDS);
79           return valueOrError();
80       }
81   
82       /**
83        * This method is used when we have to split a large batch in smaller ones. A chained metadata will allow the
84        * future that has already returned to the users to wait on the newly created split batches even after the
85        * old big batch has been deemed as done.
86        */
87       void chain(FutureRecordMetadata futureRecordMetadata) {
88           if (nextRecordMetadata == null)
89               nextRecordMetadata = futureRecordMetadata;
90           else
91               nextRecordMetadata.chain(futureRecordMetadata);
92       }
93   
94       RecordMetadata valueOrError() throws ExecutionException {
95           RuntimeException exception = this.result.error(batchIndex);
96           if (exception != null)
97               throw new ExecutionException(exception);
98           else
99               return value();
100       }
101   
102       RecordMetadata value() {
103           if (nextRecordMetadata != null)
104               return nextRecordMetadata.value();
105           return new RecordMetadata(result.topicPartition(), this.result.baseOffset(), this.batchIndex,
106                                     timestamp(), this.serializedKeySize, this.serializedValueSize);
107       }
108   
109       private long timestamp() {
110           return result.hasLogAppendTime() ? result.logAppendTime() : createTimestamp;
111       }
112   
113       @Override
114       public boolean isDone() {
115           if (nextRecordMetadata != null)
116               return nextRecordMetadata.isDone();
117           return this.result.completed();
118       }
119   
120   }
