30   /**
31    * The result of {@link Admin#describeReplicaLogDirs(Collection)}.
32    *
33    * The API of this class is evolving, see {@link Admin} for details.
34    */
35   @InterfaceStability.Evolving
36   public class DescribeReplicaLogDirsResult {
37       private final Map<TopicPartitionReplica, KafkaFuture<ReplicaLogDirInfo>> futures;
38   
39       DescribeReplicaLogDirsResult(Map<TopicPartitionReplica, KafkaFuture<ReplicaLogDirInfo>> futures) {
40           this.futures = futures;
41       }
42   
43       /**
44        * Return a map from replica to future which can be used to check the log directory information of individual replicas
45        */
46       public Map<TopicPartitionReplica, KafkaFuture<ReplicaLogDirInfo>> values() {
47           return futures;
48       }
49   
50       /**
51        * Return a future which succeeds if log directory information of all replicas are available
52        */
53       public KafkaFuture<Map<TopicPartitionReplica, ReplicaLogDirInfo>> all() {
54           return KafkaFuture.allOf(futures.values().toArray(new KafkaFuture[0])).
55               thenApply(new KafkaFuture.BaseFunction<Void, Map<TopicPartitionReplica, ReplicaLogDirInfo>>() {
56                   @Override
57                   public Map<TopicPartitionReplica, ReplicaLogDirInfo> apply(Void v) {
58                       Map<TopicPartitionReplica, ReplicaLogDirInfo> replicaLogDirInfos = new HashMap<>();
59                       for (Map.Entry<TopicPartitionReplica, KafkaFuture<ReplicaLogDirInfo>> entry : futures.entrySet()) {
60                           try {
61                               replicaLogDirInfos.put(entry.getKey(), entry.getValue().get());
62                           } catch (InterruptedException | ExecutionException e) {
63                               // This should be unreachable, because allOf ensured that all the futures completed successfully.
64                               throw new RuntimeException(e);
65                           }
66                       }
67                       return replicaLogDirInfos;
68                   }
69               });
70       }
71   
72       static public class ReplicaLogDirInfo {
73           // The current log directory of the replica of this partition on the given broker.
74           // Null if no replica is not found for this partition on the given broker.
75           private final String currentReplicaLogDir;
76           // Defined as max(HW of partition - LEO of the replica, 0).
77           private final long currentReplicaOffsetLag;
78           // The future log directory of the replica of this partition on the given broker.
79           // Null if the replica of this partition is not being moved to another log directory on the given broker.
80           private final String futureReplicaLogDir;
81           // The LEO of the replica - LEO of the future log of this replica in the destination log directory.
82           // -1 if either there is not replica for this partition or the replica of this partition is not being moved to another log directory on the given broker.
83           private final long futureReplicaOffsetLag;
84   
85           ReplicaLogDirInfo() {
86               this(null, DescribeLogDirsResponse.INVALID_OFFSET_LAG, null, DescribeLogDirsResponse.INVALID_OFFSET_LAG);
87           }
88   
89           ReplicaLogDirInfo(String currentReplicaLogDir,
90                             long currentReplicaOffsetLag,
91                             String futureReplicaLogDir,
92                             long futureReplicaOffsetLag) {
93               this.currentReplicaLogDir = currentReplicaLogDir;
94               this.currentReplicaOffsetLag = currentReplicaOffsetLag;
95               this.futureReplicaLogDir = futureReplicaLogDir;
96               this.futureReplicaOffsetLag = futureReplicaOffsetLag;
97           }
98   
99           public String getCurrentReplicaLogDir() {
100               return currentReplicaLogDir;
101           }
102   
103           public long getCurrentReplicaOffsetLag() {
104               return currentReplicaOffsetLag;
105           }
106   
107           public String getFutureReplicaLogDir() {
108               return futureReplicaLogDir;
109           }
110   
111           public long getFutureReplicaOffsetLag() {
112               return futureReplicaOffsetLag;
113           }
114   
115           @Override
116           public String toString() {
117               StringBuilder builder = new StringBuilder();
118               if (futureReplicaLogDir != null) {
119                   builder.append("(currentReplicaLogDir=")
120                       .append(currentReplicaLogDir)
121                       .append(", futureReplicaLogDir=")
122                       .append(futureReplicaLogDir)
123                       .append(", futureReplicaOffsetLag=")
124                       .append(futureReplicaOffsetLag)
125                       .append(")");
126               } else {
127                   builder.append("ReplicaLogDirInfo(currentReplicaLogDir=").append(currentReplicaLogDir).append(")");
128               }
129               return builder.toString();
130           }
131       }
