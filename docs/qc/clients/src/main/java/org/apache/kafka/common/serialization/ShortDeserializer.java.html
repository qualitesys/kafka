19   import org.apache.kafka.common.errors.SerializationException;
20   
21   public class ShortDeserializer implements Deserializer<Short> {
22   
23       public Short deserialize(String topic, byte[] data) {
24           if (data == null)
25               return null;
26           if (data.length != 2) {
27               throw new SerializationException("Size of data received by ShortDeserializer is not 2");
28           }
29   
30           short value = 0;
31           for (byte b : data) {
32               value <<= 8;
33               value |= b & 0xFF;
34           }
35           return value;
36       }
37   }
