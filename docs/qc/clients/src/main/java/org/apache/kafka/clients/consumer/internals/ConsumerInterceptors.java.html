17   package org.apache.kafka.clients.consumer.internals;
18   
19   
20   import org.apache.kafka.clients.consumer.ConsumerInterceptor;
21   import org.apache.kafka.clients.consumer.ConsumerRecords;
22   import org.apache.kafka.clients.consumer.OffsetAndMetadata;
23   import org.apache.kafka.common.TopicPartition;
24   import org.slf4j.Logger;
25   import org.slf4j.LoggerFactory;
26   
27   import java.io.Closeable;
28   import java.util.List;
29   import java.util.Map;
30   
31   /**
32    * A container that holds the list {@link org.apache.kafka.clients.consumer.ConsumerInterceptor}
33    * and wraps calls to the chain of custom interceptors.
34    */
35   public class ConsumerInterceptors<K, V> implements Closeable {
36       private static final Logger log = LoggerFactory.getLogger(ConsumerInterceptors.class);
37       private final List<ConsumerInterceptor<K, V>> interceptors;
38   
39       public ConsumerInterceptors(List<ConsumerInterceptor<K, V>> interceptors) {
40           this.interceptors = interceptors;
41       }
42   
43       /**
44        * This is called when the records are about to be returned to the user.
45        * <p>
46        * This method calls {@link ConsumerInterceptor#onConsume(ConsumerRecords)} for each
47        * interceptor. Records returned from each interceptor get passed to onConsume() of the next interceptor
48        * in the chain of interceptors.
49        * <p>
50        * This method does not throw exceptions. If any of the interceptors in the chain throws an exception,
51        * it gets caught and logged, and next interceptor in the chain is called with 'records' returned by the
52        * previous successful interceptor onConsume call.
53        *
54        * @param records records to be consumed by the client.
55        * @return records that are either modified by interceptors or same as records passed to this method.
56        */
57       public ConsumerRecords<K, V> onConsume(ConsumerRecords<K, V> records) {
58           ConsumerRecords<K, V> interceptRecords = records;
59           for (ConsumerInterceptor<K, V> interceptor : this.interceptors) {
60               try {
61                   interceptRecords = interceptor.onConsume(interceptRecords);
62               } catch (Exception e) {
63                   // do not propagate interceptor exception, log and continue calling other interceptors
64                   log.warn("Error executing interceptor onConsume callback", e);
65               }
66           }
67           return interceptRecords;
68       }
69   
70       /**
71        * This is called when commit request returns successfully from the broker.
72        * <p>
73        * This method calls {@link ConsumerInterceptor#onCommit(Map)} method for each interceptor.
74        * <p>
75        * This method does not throw exceptions. Exceptions thrown by any of the interceptors in the chain are logged, but not propagated.
76        *
77        * @param offsets A map of offsets by partition with associated metadata
78        */
79       public void onCommit(Map<TopicPartition, OffsetAndMetadata> offsets) {
80           for (ConsumerInterceptor<K, V> interceptor : this.interceptors) {
81               try {
82                   interceptor.onCommit(offsets);
83               } catch (Exception e) {
84                   // do not propagate interceptor exception, just log
85                   log.warn("Error executing interceptor onCommit callback", e);
86               }
87           }
88       }
89   
90       /**
91        * Closes every interceptor in a container.
92        */
93       @Override
94       public void close() {
95           for (ConsumerInterceptor<K, V> interceptor : this.interceptors) {
96               try {
97                   interceptor.close();
98               } catch (Exception e) {
99                   log.error("Failed to close consumer interceptor ", e);
100               }
101           }
102       }
103   }
