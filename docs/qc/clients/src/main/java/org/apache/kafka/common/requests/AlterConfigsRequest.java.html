20   import org.apache.kafka.common.config.ConfigResource;

23   import org.apache.kafka.common.protocol.ApiKeys;
24   import org.apache.kafka.common.protocol.ByteBufferAccessor;
25   
26   import java.nio.ByteBuffer;
27   import java.util.Collection;
28   import java.util.Map;
29   import java.util.Objects;
30   import java.util.stream.Collectors;
31   
32   public class AlterConfigsRequest extends AbstractRequest {
33   
34       public static class Config {
35           private final Collection<ConfigEntry> entries;
36   
37           public Config(Collection<ConfigEntry> entries) {
38               this.entries = Objects.requireNonNull(entries, "entries");
39           }
40   
41           public Collection<ConfigEntry> entries() {
42               return entries;
43           }
44       }
45   
46       public static class ConfigEntry {
47           private final String name;
48           private final String value;
49   
50           public ConfigEntry(String name, String value) {
51               this.name = Objects.requireNonNull(name, "name");
52               this.value = Objects.requireNonNull(value, "value");
53           }
54   
55           public String name() {
56               return name;
57           }
58   
59           public String value() {
60               return value;
61           }
62   
63       }
64   
65       public static class Builder extends AbstractRequest.Builder<AlterConfigsRequest> {
66   
67           private final AlterConfigsRequestData data = new AlterConfigsRequestData();
68   
69           public Builder(Map<ConfigResource, Config> configs, boolean validateOnly) {
70               super(ApiKeys.ALTER_CONFIGS);
71               Objects.requireNonNull(configs, "configs");
72               for (Map.Entry<ConfigResource, Config> entry : configs.entrySet()) {
73                   AlterConfigsRequestData.AlterConfigsResource resource =
74                       new AlterConfigsRequestData.AlterConfigsResource()
75                           .setResourceName(entry.getKey().name())
76                           .setResourceType(entry.getKey().type().id());
77                   for (ConfigEntry x : entry.getValue().entries) {
78                       resource.configs().add(new AlterConfigsRequestData.AlterableConfig()
79                                                  .setName(x.name())
80                                                  .setValue(x.value()));
81                   }
82                   this.data.resources().add(resource);
83               }
84               this.data.setValidateOnly(validateOnly);
85           }
86   
87           @Override
88           public AlterConfigsRequest build(short version) {
89               return new AlterConfigsRequest(data, version);
90           }
91       }
92   
93       private final AlterConfigsRequestData data;
94   
95       public AlterConfigsRequest(AlterConfigsRequestData data, short version) {
96           super(ApiKeys.ALTER_CONFIGS, version);
97           this.data = data;
98       }
99   
100       public Map<ConfigResource, Config> configs() {
101           return data.resources().stream().collect(Collectors.toMap(
102               resource -> new ConfigResource(
103                       ConfigResource.Type.forId(resource.resourceType()),
104                       resource.resourceName()),
105               resource -> new Config(resource.configs().stream()
106                       .map(entry -> new ConfigEntry(entry.name(), entry.value()))
107                       .collect(Collectors.toList()))));
108       }
109   
110       public boolean validateOnly() {
111           return data.validateOnly();
112       }
113   
114       @Override
115       public AlterConfigsRequestData data() {
116           return data;
117       }
118   
119       @Override
120       public AbstractResponse getErrorResponse(int throttleTimeMs, Throwable e) {
121           ApiError error = ApiError.fromThrowable(e);
122           AlterConfigsResponseData data = new AlterConfigsResponseData()
123                   .setThrottleTimeMs(throttleTimeMs);
124           for (AlterConfigsRequestData.AlterConfigsResource resource : this.data.resources()) {
125               data.responses().add(new AlterConfigsResponseData.AlterConfigsResourceResponse()
126                       .setResourceType(resource.resourceType())
127                       .setResourceName(resource.resourceName())
128                       .setErrorMessage(error.message())
129                       .setErrorCode(error.error().code()));
130           }
131           return new AlterConfigsResponse(data);
132   
133       }
134   
135       public static AlterConfigsRequest parse(ByteBuffer buffer, short version) {
136           return new AlterConfigsRequest(new AlterConfigsRequestData(new ByteBufferAccessor(buffer), version), version);
137       }
