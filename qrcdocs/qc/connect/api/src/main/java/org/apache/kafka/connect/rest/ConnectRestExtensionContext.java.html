18   package org.apache.kafka.connect.rest;
19   
20   import org.apache.kafka.connect.health.ConnectClusterState;
21   
22   import javax.ws.rs.core.Configurable;
23   
24   /**
25    * The interface provides the ability for {@link ConnectRestExtension} implementations to access the JAX-RS
26    * {@link javax.ws.rs.core.Configurable} and cluster state {@link ConnectClusterState}. The implementation for the interface is provided
27    * by the Connect framework.
28    */
29   public interface ConnectRestExtensionContext {
30   
31       /**
32        * Provides an implementation of {@link javax.ws.rs.core.Configurable} that be used to register JAX-RS resources.
33        *
34        * @return @return the JAX-RS {@link javax.ws.rs.core.Configurable}; never {@code null}
35        */
36       Configurable<? extends Configurable<?>> configurable();
37   
38       /**
39        * Provides the cluster state and health information about the connectors and tasks.
40        *
41        * @return the cluster state information; never {@code null}
42        */
43       ConnectClusterState clusterState();
44   }
