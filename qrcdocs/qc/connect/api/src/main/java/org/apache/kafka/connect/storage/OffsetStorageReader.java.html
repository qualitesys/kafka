17   package org.apache.kafka.connect.storage;
18   
19   import java.util.Collection;
20   import java.util.Map;
21   
22   /**
23    * <p>
24    * OffsetStorageReader provides access to the offset storage used by sources. This can be used by
25    * connectors to determine offsets to start consuming data from. This is most commonly used during
26    * initialization of a task, but can also be used during runtime, e.g. when reconfiguring a task.
27    * </p>
28    * <p>
29    * Offsets are always defined as Maps of Strings to primitive types, i.e. all types supported by
30    * {@link org.apache.kafka.connect.data.Schema} other than Array, Map, and Struct.
31    * </p>
32    */
33   public interface OffsetStorageReader {
34       /**
35        * Get the offset for the specified partition. If the data isn't already available locally, this
36        * gets it from the backing store, which may require some network round trips.
37        *
38        * @param partition object uniquely identifying the partition of data
39        * @return object uniquely identifying the offset in the partition of data
40        */
41       <T> Map<String, Object> offset(Map<String, T> partition);
42   
43       /**
44        * <p>
45        * Get a set of offsets for the specified partition identifiers. This may be more efficient
46        * than calling {@link #offset(Map)} repeatedly.
47        * </p>
48        * <p>
49        * Note that when errors occur, this method omits the associated data and tries to return as
50        * many of the requested values as possible. This allows a task that's managing many partitions to
51        * still proceed with any available data. Therefore, implementations should take care to check
52        * that the data is actually available in the returned response. The only case when an
53        * exception will be thrown is if the entire request failed, e.g. because the underlying
54        * storage was unavailable.
55        * </p>
56        *
57        * @param partitions set of identifiers for partitions of data
58        * @return a map of partition identifiers to decoded offsets
59        */
60       <T> Map<Map<String, T>, Map<String, Object>> offsets(Collection<Map<String, T>> partitions);
61   }
