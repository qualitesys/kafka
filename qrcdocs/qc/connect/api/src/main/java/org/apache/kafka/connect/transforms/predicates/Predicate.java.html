17   package org.apache.kafka.connect.transforms.predicates;
18   
19   import org.apache.kafka.common.Configurable;
20   import org.apache.kafka.common.config.ConfigDef;
21   import org.apache.kafka.connect.connector.ConnectRecord;
22   
23   /**
24    * <p>A predicate on records.
25    * Predicates can be used to conditionally apply a {@link org.apache.kafka.connect.transforms.Transformation}
26    * by configuring the transformation's {@code predicate} (and {@code negate}) configuration parameters.
27    * In particular, the {@code Filter} transformation can be conditionally applied in order to filter
28    * certain records from further processing.
29    *
30    * <p>Implementations of this interface must be public and have a public constructor with no parameters.
31    *
32    * @param <R> The type of record.
33    */
34   public interface Predicate<R extends ConnectRecord<R>> extends Configurable, AutoCloseable {
35   
36       /**
37        * Configuration specification for this predicate.
38        *
39        * @return the configuration definition for this predicate; never null
40        */
41       ConfigDef config();
42   
43       /**
44        * Returns whether the given record satisfies this predicate.
45        *
46        * @param record the record to evaluate; may not be null
47        * @return true if the predicate matches, or false otherwise
48        */
49       boolean test(R record);
50   
51       @Override
52       void close();
53   }
