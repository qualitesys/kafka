17   package org.apache.kafka.connect.storage;
18   
19   import org.apache.kafka.common.Configurable;
20   import org.apache.kafka.common.config.ConfigDef;
21   import org.apache.kafka.connect.data.Schema;
22   import org.apache.kafka.connect.data.SchemaAndValue;
23   import org.apache.kafka.connect.header.Header;
24   
25   import java.io.Closeable;
26   
27   public interface HeaderConverter extends Configurable, Closeable {
28   
29       /**
30        * Convert the header name and byte array value into a {@link Header} object.
31        * @param topic the name of the topic for the record containing the header
32        * @param headerKey the header's key; may not be null
33        * @param value the header's raw value; may be null
34        * @return the {@link SchemaAndValue}; may not be null
35        */
36       SchemaAndValue toConnectHeader(String topic, String headerKey, byte[] value);
37   
38       /**
39        * Convert the {@link Header}'s {@link Header#value() value} into its byte array representation.
40        * @param topic the name of the topic for the record containing the header
41        * @param headerKey the header's key; may not be null
42        * @param schema the schema for the header's value; may be null
43        * @param value the header's value to convert; may be null
44        * @return the byte array form of the Header's value; may be null if the value is null
45        */
46       byte[] fromConnectHeader(String topic, String headerKey, Schema schema, Object value);
47   
48       /**
49        * Configuration specification for this set of header converters.
50        * @return the configuration specification; may not be null
51        */
52       ConfigDef config();
53   }
