26   /**
27    * <p>
28    * SourceRecords are generated by SourceTasks and passed to Kafka Connect for storage in
29    * Kafka. In addition to the standard fields in {@link ConnectRecord} which specify where data is stored
30    * in Kafka, they also include a sourcePartition and sourceOffset.
31    * </p>
32    * <p>
33    * The sourcePartition represents a single input sourcePartition that the record came from (e.g. a filename, table
34    * name, or topic-partition). The sourceOffset represents a position in that sourcePartition which can be used
35    * to resume consumption of data.
36    * </p>
37    * <p>
38    * These values can have arbitrary structure and should be represented using
39    * org.apache.kafka.connect.data objects (or primitive values). For example, a database connector
40    * might specify the sourcePartition as a record containing { "db": "database_name", "table":
41    * "table_name"} and the sourceOffset as a Long containing the timestamp of the row.
42    * </p>
43    */
44   public class SourceRecord extends ConnectRecord<SourceRecord> {
45       private final Map<String, ?> sourcePartition;
46       private final Map<String, ?> sourceOffset;
47   
48       public SourceRecord(Map<String, ?> sourcePartition, Map<String, ?> sourceOffset,
49                           String topic, Integer partition, Schema valueSchema, Object value) {
50           this(sourcePartition, sourceOffset, topic, partition, null, null, valueSchema, value);
51       }
52   
53       public SourceRecord(Map<String, ?> sourcePartition, Map<String, ?> sourceOffset,
54                           String topic, Schema valueSchema, Object value) {
55           this(sourcePartition, sourceOffset, topic, null, null, null, valueSchema, value);
56       }
57   
58       public SourceRecord(Map<String, ?> sourcePartition, Map<String, ?> sourceOffset,
59                           String topic, Schema keySchema, Object key, Schema valueSchema, Object value) {
60           this(sourcePartition, sourceOffset, topic, null, keySchema, key, valueSchema, value);
61       }
62   
63       public SourceRecord(Map<String, ?> sourcePartition, Map<String, ?> sourceOffset,
64                           String topic, Integer partition,
65                           Schema keySchema, Object key, Schema valueSchema, Object value) {
66           this(sourcePartition, sourceOffset, topic, partition, keySchema, key, valueSchema, value, null);
67       }
68   
69       public SourceRecord(Map<String, ?> sourcePartition, Map<String, ?> sourceOffset,
70                           String topic, Integer partition,
71                           Schema keySchema, Object key,
72                           Schema valueSchema, Object value,
73                           Long timestamp) {
74           this(sourcePartition, sourceOffset, topic, partition, keySchema, key, valueSchema, value, timestamp, null);
75       }
76   
77       public SourceRecord(Map<String, ?> sourcePartition, Map<String, ?> sourceOffset,
78                           String topic, Integer partition,
79                           Schema keySchema, Object key,
80                           Schema valueSchema, Object value,
81                           Long timestamp, Iterable<Header> headers) {
82           super(topic, partition, keySchema, key, valueSchema, value, timestamp, headers);
83           this.sourcePartition = sourcePartition;
84           this.sourceOffset = sourceOffset;
85       }
86   
87       public Map<String, ?> sourcePartition() {
88           return sourcePartition;
89       }
90   
91       public Map<String, ?> sourceOffset() {
92           return sourceOffset;
93       }
94   
95       @Override
96       public SourceRecord newRecord(String topic, Integer kafkaPartition, Schema keySchema, Object key, Schema valueSchema, Object value, Long timestamp) {
97           return newRecord(topic, kafkaPartition, keySchema, key, valueSchema, value, timestamp, headers().duplicate());
98       }
99   
100       @Override
101       public SourceRecord newRecord(String topic, Integer kafkaPartition, Schema keySchema, Object key, Schema valueSchema, Object value,
102                                     Long timestamp, Iterable<Header> headers) {
103           return new SourceRecord(sourcePartition, sourceOffset, topic, kafkaPartition, keySchema, key, valueSchema, value, timestamp, headers);
104       }
105   
106       @Override
107       public boolean equals(Object o) {
108           if (this == o)
109               return true;
110           if (o == null || getClass() != o.getClass())
111               return false;
112           if (!super.equals(o))
113               return false;
114   
115           SourceRecord that = (SourceRecord) o;
116   
117           return Objects.equals(sourcePartition, that.sourcePartition) &&
118                   Objects.equals(sourceOffset, that.sourceOffset);
119       }
120   
121       @Override
122       public int hashCode() {
123           int result = super.hashCode();
124           result = 31 * result + (sourcePartition != null ? sourcePartition.hashCode() : 0);
125           result = 31 * result + (sourceOffset != null ? sourceOffset.hashCode() : 0);
126           return result;
127       }
128   
129       @Override
130       public String toString() {
131           return "SourceRecord{" +
132                   "sourcePartition=" + sourcePartition +
133                   ", sourceOffset=" + sourceOffset +
134                   "} " + super.toString();
135       }
136   }
