17   package org.apache.kafka.connect.data;
18   
19   import org.apache.kafka.connect.errors.DataException;
20   
21   import java.math.BigDecimal;
22   import java.math.BigInteger;
23   
24   /**
25    * <p>
26    *     An arbitrary-precision signed decimal number. The value is unscaled * 10 ^ -scale where:
27    *     <ul>
28    *         <li>unscaled is an integer </li>
29    *         <li>scale is an integer representing how many digits the decimal point should be shifted on the unscaled value</li>
30    *     </ul>
31    * </p>
32    * <p>
33    *     Decimal does not provide a fixed schema because it is parameterized by the scale, which is fixed on the schema
34    *     rather than being part of the value.
35    * </p>
36    * <p>
37    *     The underlying representation of this type is bytes containing a two's complement integer
38    * </p>
39    */
40   public class Decimal {
41       public static final String LOGICAL_NAME = "org.apache.kafka.connect.data.Decimal";
42       public static final String SCALE_FIELD = "scale";
43   
44       /**
45        * Returns a SchemaBuilder for a Decimal with the given scale factor. By returning a SchemaBuilder you can override
46        * additional schema settings such as required/optional, default value, and documentation.
47        * @param scale the scale factor to apply to unscaled values
48        * @return a SchemaBuilder
49        */
50       public static SchemaBuilder builder(int scale) {
51           return SchemaBuilder.bytes()
52                   .name(LOGICAL_NAME)
53                   .parameter(SCALE_FIELD, Integer.toString(scale))
54                   .version(1);
55       }
56   
57       public static Schema schema(int scale) {
58           return builder(scale).build();
59       }
60   
61       /**
62        * Convert a value from its logical format (BigDecimal) to it's encoded format.
63        * @param value the logical value
64        * @return the encoded value
65        */
66       public static byte[] fromLogical(Schema schema, BigDecimal value) {
67           int schemaScale = scale(schema);
68           if (value.scale() != schemaScale)
69               throw new DataException(String.format(
70                   "Decimal value has mismatching scale for given Decimal schema. "
71                       + "Schema has scale %d, value has scale %d.",
72                   schemaScale,
73                   value.scale()
74               ));
75           return value.unscaledValue().toByteArray();
76       }
77   
78       public static BigDecimal toLogical(Schema schema, byte[] value) {
79           return new BigDecimal(new BigInteger(value), scale(schema));
80       }
81   
82       private static int scale(Schema schema) {
83           String scaleString = schema.parameters().get(SCALE_FIELD);
84           if (scaleString == null)
85               throw new DataException("Invalid Decimal schema: scale parameter not found.");
86           try {
87               return Integer.parseInt(scaleString);
88           } catch (NumberFormatException e) {
89               throw new DataException("Invalid scale parameter found in Decimal schema: ", e);
90           }
91       }
92   }
