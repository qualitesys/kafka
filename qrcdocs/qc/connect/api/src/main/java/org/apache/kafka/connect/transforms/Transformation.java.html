17   package org.apache.kafka.connect.transforms;
18   
19   import org.apache.kafka.common.Configurable;
20   import org.apache.kafka.common.config.ConfigDef;
21   import org.apache.kafka.connect.connector.ConnectRecord;
22   
23   import java.io.Closeable;
24   
25   /**
26    * Single message transformation for Kafka Connect record types.
27    *
28    * Connectors can be configured with transformations to make lightweight message-at-a-time modifications.
29    */
30   public interface Transformation<R extends ConnectRecord<R>> extends Configurable, Closeable {
31   
32       /**
33        * Apply transformation to the {@code record} and return another record object (which may be {@code record} itself) or {@code null},
34        * corresponding to a map or filter operation respectively.
35        *
36        * A transformation must not mutate objects reachable from the given {@code record}
37        * (including, but not limited to, {@link org.apache.kafka.connect.header.Headers Headers},
38        * {@link org.apache.kafka.connect.data.Struct Structs}, {@code Lists}, and {@code Maps}).
39        * If such objects need to be changed, a new ConnectRecord should be created and returned.
40        *
41        * The implementation must be thread-safe.
42        */
43       R apply(R record);
44   
45       /** Configuration specification for this transformation. **/
46       ConfigDef config();
47   
48       /** Signal that this transformation instance will no longer will be used. **/
49       @Override
50       void close();
51   
52   }
