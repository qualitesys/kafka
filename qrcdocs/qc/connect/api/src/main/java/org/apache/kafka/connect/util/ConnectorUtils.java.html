17   package org.apache.kafka.connect.util;
18   
19   import java.util.ArrayList;
20   import java.util.List;
21   
22   /**
23    * Utilities that connector implementations might find useful. Contains common building blocks
24    * for writing connectors.
25    */
26   public class ConnectorUtils {
27       /**
28        * Given a list of elements and a target number of groups, generates list of groups of
29        * elements to match the target number of groups, spreading them evenly among the groups.
30        * This generates groups with contiguous elements, which results in intuitive ordering if
31        * your elements are also ordered (e.g. alphabetical lists of table names if you sort
32        * table names alphabetically to generate the raw partitions) or can result in efficient
33        * partitioning if elements are sorted according to some criteria that affects performance
34        * (e.g. topic partitions with the same leader).
35        *
36        * @param elements list of elements to partition
37        * @param numGroups the number of output groups to generate.
38        */
39       public static <T> List<List<T>> groupPartitions(List<T> elements, int numGroups) {
40           if (numGroups <= 0)
41               throw new IllegalArgumentException("Number of groups must be positive.");
42   
43           List<List<T>> result = new ArrayList<>(numGroups);
44   
45           // Each group has either n+1 or n raw partitions
46           int perGroup = elements.size() / numGroups;
47           int leftover = elements.size() - (numGroups * perGroup);
48   
49           int assigned = 0;
50           for (int group = 0; group < numGroups; group++) {
51               int numThisGroup = group < leftover ? perGroup + 1 : perGroup;
52               List<T> groupList = new ArrayList<>(numThisGroup);
53               for (int i = 0; i < numThisGroup; i++) {
54                   groupList.add(elements.get(assigned));
55                   assigned++;
56               }
57               result.add(groupList);
58           }
59   
60           return result;
61       }
62   }
