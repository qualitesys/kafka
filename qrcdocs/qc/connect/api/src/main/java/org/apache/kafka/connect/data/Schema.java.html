23   /**
24    * <p>
25    *     Definition of an abstract data type. Data types can be primitive types (integer types, floating point types,
26    *     boolean, strings, and bytes) or complex types (typed arrays, maps with one key schema and value schema,
27    *     and structs that have a fixed set of field names each with an associated value schema). Any type can be specified
28    *     as optional, allowing it to be omitted (resulting in null values when it is missing) and can specify a default
29    *     value.
30    * </p>
31    * <p>
32    *     All schemas may have some associated metadata: a name, version, and documentation. These are all considered part
33    *     of the schema itself and included when comparing schemas. Besides adding important metadata, these fields enable
34    *     the specification of logical types that specify additional constraints and semantics (e.g. UNIX timestamps are
35    *     just an int64, but the user needs the know about the additional semantics to interpret it properly).
36    * </p>
37    * <p>
38    *     Schemas can be created directly, but in most cases using {@link SchemaBuilder} will be simpler.
39    * </p>
40    */
41   public interface Schema {
42       /**
43        * The type of a schema. These only include the core types; logical types must be determined by checking the schema name.
44        */
45       enum Type {
46           /**
47            *  8-bit signed integer
48            *
49            *  Note that if you have an unsigned 8-bit data source, {@link Type#INT16} will be required to safely capture all valid values
50            */
51           INT8,
52           /**
53            *  16-bit signed integer
54            *
55            *  Note that if you have an unsigned 16-bit data source, {@link Type#INT32} will be required to safely capture all valid values
56            */
57           INT16,
58           /**
59            *  32-bit signed integer
60            *
61            *  Note that if you have an unsigned 32-bit data source, {@link Type#INT64} will be required to safely capture all valid values
62            */
63           INT32,
64           /**
65            *  64-bit signed integer
66            *
67            *  Note that if you have an unsigned 64-bit data source, the {@link Decimal} logical type (encoded as {@link Type#BYTES})
68            *  will be required to safely capture all valid values
69            */
70           INT64,
71           /**
72            *  32-bit IEEE 754 floating point number
73            */
74           FLOAT32,
75           /**
76            *  64-bit IEEE 754 floating point number
77            */
78           FLOAT64,
79           /**
80            * Boolean value (true or false)
81            */
82           BOOLEAN,
83           /**
84            * Character string that supports all Unicode characters.
85            *
86            * Note that this does not imply any specific encoding (e.g. UTF-8) as this is an in-memory representation.
87            */
88           STRING,
89           /**
90            * Sequence of unsigned 8-bit bytes
91            */
92           BYTES,
93           /**
94            * An ordered sequence of elements, each of which shares the same type.
95            */
96           ARRAY,
97           /**
98            * A mapping from keys to values. Both keys and values can be arbitrarily complex types, including complex types
99            * such as {@link Struct}.
100            */
101           MAP,
102           /**
103            * A structured record containing a set of named fields, each field using a fixed, independent {@link Schema}.
104            */
105           STRUCT;
106   
107           private String name;
108   
109           Type() {
110               this.name = this.name().toLowerCase(Locale.ROOT);
111           }
112   
113           public String getName() {
114               return name;
115           }
116   
117           public boolean isPrimitive() {
118               switch (this) {
119                   case INT8:
120                   case INT16:
121                   case INT32:
122                   case INT64:
123                   case FLOAT32:
124                   case FLOAT64:
125                   case BOOLEAN:
126                   case STRING:
127                   case BYTES:
128                       return true;
129               }
130               return false;
131           }
132       }

156        * @return the type of this schema
157        */
158       Type type();
159   
160       /**
161        * @return true if this field is optional, false otherwise
162        */
163       boolean isOptional();
164   
165       /**
166        * @return the default value for this schema
167        */
168       Object defaultValue();

171        * @return the name of this schema
172        */
173       String name();
174   
175       /**
176        * Get the optional version of the schema. If a version is included, newer versions *must* be larger than older ones.
177        * @return the version of this schema
178        */
179       Integer version();
180   
181       /**
182        * @return the documentation for this schema
183        */
184       String doc();
185   
186       /**
187        * Get a map of schema parameters.
188        * @return Map containing parameters for this schema, or null if there are no parameters
189        */
190       Map<String, String> parameters();
191   
192       /**
193        * Get the key schema for this map schema. Throws a DataException if this schema is not a map.
194        * @return the key schema
195        */
196       Schema keySchema();
197   
198       /**
199        * Get the value schema for this map or array schema. Throws a DataException if this schema is not a map or array.
200        * @return the value schema
201        */
202       Schema valueSchema();
203   
204       /**
205        * Get the list of fields for this Schema. Throws a DataException if this schema is not a struct.
206        * @return the list of fields for this Schema
207        */
208       List<Field> fields();
209   
210       /**
211        * Get a field for this Schema by name. Throws a DataException if this schema is not a struct.
212        * @param fieldName the name of the field to look up
213        * @return the Field object for the specified field, or null if there is no field with the given name
214        */
215       Field field(String fieldName);
216   
217       /**
218        * Return a concrete instance of the {@link Schema}
219        * @return the {@link Schema}
220        */
221       Schema schema();
222   }
