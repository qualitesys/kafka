19   import org.apache.kafka.common.config.ConfigDef;
20   import org.apache.kafka.common.config.ConfigException;
21   import org.apache.kafka.connect.connector.ConnectRecord;
22   import org.apache.kafka.connect.data.ConnectSchema;
23   import org.apache.kafka.connect.data.Field;
24   import org.apache.kafka.connect.data.Schema;
25   import org.apache.kafka.connect.data.Struct;
26   import org.apache.kafka.connect.transforms.util.SimpleConfig;
27   import org.slf4j.Logger;
28   import org.slf4j.LoggerFactory;
29   
30   import java.util.Map;
31   
32   import static org.apache.kafka.connect.transforms.util.Requirements.requireSchema;
33   
34   public abstract class SetSchemaMetadata<R extends ConnectRecord<R>> implements Transformation<R> {
35       private static final Logger log = LoggerFactory.getLogger(SetSchemaMetadata.class);
36   
37       public static final String OVERVIEW_DOC =
38               "Set the schema name, version or both on the record's key (<code>" + Key.class.getName() + "</code>)"
39                       + " or value (<code>" + Value.class.getName() + "</code>) schema.";
40   
41       private interface ConfigName {
42           String SCHEMA_NAME = "schema.name";
43           String SCHEMA_VERSION = "schema.version";
44       }
45   
46       public static final ConfigDef CONFIG_DEF = new ConfigDef()
47               .define(ConfigName.SCHEMA_NAME, ConfigDef.Type.STRING, null, ConfigDef.Importance.HIGH, "Schema name to set.")
48               .define(ConfigName.SCHEMA_VERSION, ConfigDef.Type.INT, null, ConfigDef.Importance.HIGH, "Schema version to set.");
49   
50       private String schemaName;
51       private Integer schemaVersion;
52   
53       @Override
54       public void configure(Map<String, ?> configs) {
55           final SimpleConfig config = new SimpleConfig(CONFIG_DEF, configs);
56           schemaName = config.getString(ConfigName.SCHEMA_NAME);
57           schemaVersion = config.getInt(ConfigName.SCHEMA_VERSION);
58   
59           if (schemaName == null && schemaVersion == null) {
60               throw new ConfigException("Neither schema name nor version configured");
61           }
62       }
63   
64       @Override
65       public R apply(R record) {
66           final Object value = operatingValue(record);
67           final Schema schema = operatingSchema(record);
68           if (value == null && schema == null) {
69               return record;
70           }
71           requireSchema(schema, "updating schema metadata");
72           final boolean isArray = schema.type() == Schema.Type.ARRAY;
73           final boolean isMap = schema.type() == Schema.Type.MAP;
74           final Schema updatedSchema = new ConnectSchema(
75                   schema.type(),
76                   schema.isOptional(),
77                   schema.defaultValue(),
78                   schemaName != null ? schemaName : schema.name(),
79                   schemaVersion != null ? schemaVersion : schema.version(),
80                   schema.doc(),
81                   schema.parameters(),
82                   schema.fields(),
83                   isMap ? schema.keySchema() : null,
84                   isMap || isArray ? schema.valueSchema() : null
85           );
86           log.trace("Applying SetSchemaMetadata SMT. Original schema: {}, updated schema: {}",
87               schema, updatedSchema);
88           return newRecord(record, updatedSchema);
89       }
90   
91       @Override
92       public ConfigDef config() {
93           return CONFIG_DEF;
94       }
95   
96       @Override
97       public void close() {
98       }
99   
100       protected abstract Schema operatingSchema(R record);
101   
102       protected abstract Object operatingValue(R record);
103   
104       protected abstract R newRecord(R record, Schema updatedSchema);
105   
106       /**
107        * Set the schema name, version or both on the record's key schema.
108        */
109       public static class Key<R extends ConnectRecord<R>> extends SetSchemaMetadata<R> {
110           @Override
111           protected Schema operatingSchema(R record) {
112               return record.keySchema();
113           }
114   
115           @Override
116           protected Object operatingValue(R record) {
117               return record.key();
118           }
119   
120           @Override
121           protected R newRecord(R record, Schema updatedSchema) {
122               Object updatedKey = updateSchemaIn(record.key(), updatedSchema);
123               return record.newRecord(record.topic(), record.kafkaPartition(), updatedSchema, updatedKey, record.valueSchema(), record.value(), record.timestamp());
124           }
125       }
126   
127       /**
128        * Set the schema name, version or both on the record's value schema.
129        */
130       public static class Value<R extends ConnectRecord<R>> extends SetSchemaMetadata<R> {
131           @Override
132           protected Schema operatingSchema(R record) {
133               return record.valueSchema();
134           }
135   
136           @Override
137           protected Object operatingValue(R record) {
138               return record.value();
139           }
140   
141           @Override
142           protected R newRecord(R record, Schema updatedSchema) {
143               Object updatedValue = updateSchemaIn(record.value(), updatedSchema);
144               return record.newRecord(record.topic(), record.kafkaPartition(), record.keySchema(), record.key(), updatedSchema, updatedValue, record.timestamp());
145           }
146       }
147   
148       /**
149        * Utility to check the supplied key or value for references to the old Schema,
150        * and if so to return an updated key or value object that references the new Schema.
151        * Note that this method assumes that the new Schema may have a different name and/or version,
152        * but has fields that exactly match those of the old Schema.
153        * <p>
154        * Currently only {@link Struct} objects have references to the {@link Schema}.
155        *
156        * @param keyOrValue    the key or value object; may be null
157        * @param updatedSchema the updated schema that has been potentially renamed
158        * @return the original key or value object if it does not reference the old schema, or
159        * a copy of the key or value object with updated references to the new schema.
160        */
161       protected static Object updateSchemaIn(Object keyOrValue, Schema updatedSchema) {
162           if (keyOrValue instanceof Struct) {
163               Struct origStruct = (Struct) keyOrValue;
164               Struct newStruct = new Struct(updatedSchema);
165               for (Field field : updatedSchema.fields()) {
166                   // assume both schemas have exact same fields with same names and schemas ...
167                   newStruct.put(field, origStruct.get(field));
168               }
169               return newStruct;
170           }
171           return keyOrValue;
172       }
173   }
