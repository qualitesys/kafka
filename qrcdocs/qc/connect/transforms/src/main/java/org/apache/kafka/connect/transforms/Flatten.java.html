43               "Flatten a nested data structure, generating names for each field by concatenating the field names at each "
44                       + "level with a configurable delimiter character. Applies to Struct when schema present, or a Map "

96       private R applySchemaless(R record) {
97           final Map<String, Object> value = requireMap(operatingValue(record), PURPOSE);
98           final Map<String, Object> newValue = new LinkedHashMap<>();
99           applySchemaless(value, "", newValue);
100           return newRecord(record, null, newValue);
101       }
102   
103       private void applySchemaless(Map<String, Object> originalRecord, String fieldNamePrefix, Map<String, Object> newRecord) {
104           for (Map.Entry<String, Object> entry : originalRecord.entrySet()) {
105               final String fieldName = fieldName(fieldNamePrefix, entry.getKey());
106               Object value = entry.getValue();
107               if (value == null) {
108                   newRecord.put(fieldName(fieldNamePrefix, entry.getKey()), null);
109                   continue;
110               }
111   
112               Schema.Type inferredType = ConnectSchema.schemaType(value.getClass());
113               if (inferredType == null) {
114                   throw new DataException("Flatten transformation was passed a value of type " + value.getClass()
115                           + " which is not supported by Connect's data API");
116               }
117               switch (inferredType) {
118                   case INT8:
119                   case INT16:
120                   case INT32:
121                   case INT64:
122                   case FLOAT32:
123                   case FLOAT64:
124                   case BOOLEAN:
125                   case STRING:
126                   case BYTES:
127                   case ARRAY:
128                       newRecord.put(fieldName(fieldNamePrefix, entry.getKey()), entry.getValue());
129                       break;
130                   case MAP:
131                       final Map<String, Object> fieldValue = requireMap(entry.getValue(), PURPOSE);
132                       applySchemaless(fieldValue, fieldName, newRecord);
133                       break;
134                   default:
135                       throw new DataException("Flatten transformation does not support " + entry.getValue().getClass()
136                               + " for record without schemas (for field " + fieldName + ").");
137               }
138           }
139       }
140   
141       private R applyWithSchema(R record) {
142           final Struct value = requireStructOrNull(operatingValue(record), PURPOSE);
143   
144           Schema schema = operatingSchema(record);
145           Schema updatedSchema = schemaUpdateCache.get(schema);
146           if (updatedSchema == null) {
147               final SchemaBuilder builder = SchemaUtil.copySchemaBasics(schema, SchemaBuilder.struct());
148               Struct defaultValue = (Struct) schema.defaultValue();
149               buildUpdatedSchema(schema, "", builder, schema.isOptional(), defaultValue);
150               updatedSchema = builder.build();
151               schemaUpdateCache.put(schema, updatedSchema);
152           }
153           if (value == null) {
154               return newRecord(record, updatedSchema, null);
155           } else {
156               final Struct updatedValue = new Struct(updatedSchema);
157               buildWithSchema(value, "", updatedValue);
158               return newRecord(record, updatedSchema, updatedValue);
159           }
160       }

162       /**
163        * Build an updated Struct Schema which flattens all nested fields into a single struct, handling cases where
164        * optionality and default values of the flattened fields are affected by the optionality and default values of
165        * parent/ancestor schemas (e.g. flattened field is optional because the parent schema was optional, even if the
166        * schema itself is marked as required).
167        * @param schema the schema to translate
168        * @param fieldNamePrefix the prefix to use on field names, i.e. the delimiter-joined set of ancestor field names
169        * @param newSchema the flattened schema being built
170        * @param optional true if any ancestor schema is optional
171        * @param defaultFromParent the default value, if any, included via the parent/ancestor schemas
172        */
173       private void buildUpdatedSchema(Schema schema, String fieldNamePrefix, SchemaBuilder newSchema, boolean optional, Struct defaultFromParent) {
174           for (Field field : schema.fields()) {
175               final String fieldName = fieldName(fieldNamePrefix, field.name());
176               final boolean fieldIsOptional = optional || field.schema().isOptional();
177               Object fieldDefaultValue = null;
178               if (field.schema().defaultValue() != null) {
179                   fieldDefaultValue = field.schema().defaultValue();
180               } else if (defaultFromParent != null) {
181                   fieldDefaultValue = defaultFromParent.get(field);
182               }
183               switch (field.schema().type()) {
184                   case INT8:
185                   case INT16:
186                   case INT32:
187                   case INT64:
188                   case FLOAT32:
189                   case FLOAT64:
190                   case BOOLEAN:
191                   case STRING:
192                   case BYTES:
193                   case ARRAY:
194                       newSchema.field(fieldName, convertFieldSchema(field.schema(), fieldIsOptional, fieldDefaultValue));
195                       break;
196                   case STRUCT:
197                       buildUpdatedSchema(field.schema(), fieldName, newSchema, fieldIsOptional, (Struct) fieldDefaultValue);
198                       break;
199                   default:
200                       throw new DataException("Flatten transformation does not support " + field.schema().type()
201                               + " for record with schemas (for field " + fieldName + ").");
202               }
203           }
204       }

206       /**
207        * Convert the schema for a field of a Struct with a primitive schema to the schema to be used for the flattened
208        * version, taking into account that we may need to override optionality and default values in the flattened version
209        * to take into account the optionality and default values of parent/ancestor schemas
210        * @param orig the original schema for the field
211        * @param optional whether the new flattened field should be optional
212        * @param defaultFromParent the default value either taken from the existing field or provided by the parent
213        */
214       private Schema convertFieldSchema(Schema orig, boolean optional, Object defaultFromParent) {
215           // Note that we don't use the schema translation cache here. It might save us a bit of effort, but we really
216           // only care about caching top-level schema translations.
217   
218           final SchemaBuilder builder = SchemaUtil.copySchemaBasics(orig);
219           if (optional)
220               builder.optional();
221           if (defaultFromParent != null)
222               builder.defaultValue(defaultFromParent);
223           return builder.build();
224       }
225   
226       private void buildWithSchema(Struct record, String fieldNamePrefix, Struct newRecord) {
227           if (record == null) {
228               return;
229           }
230           for (Field field : record.schema().fields()) {
231               final String fieldName = fieldName(fieldNamePrefix, field.name());
232               switch (field.schema().type()) {
233                   case INT8:
234                   case INT16:
235                   case INT32:
236                   case INT64:
237                   case FLOAT32:
238                   case FLOAT64:
239                   case BOOLEAN:
240                   case STRING:
241                   case BYTES:
242                   case ARRAY:
243                       newRecord.put(fieldName, record.get(field));
244                       break;
245                   case STRUCT:
246                       buildWithSchema(record.getStruct(field.name()), fieldName, newRecord);
247                       break;
248                   default:
249                       throw new DataException("Flatten transformation does not support " + field.schema().type()
250                               + " for record with schemas (for field " + fieldName + ").");
251               }
252           }
253       }
254   
255       private String fieldName(String prefix, String fieldName) {
256           return prefix.isEmpty() ? fieldName : (prefix + delimiter + fieldName);
257       }
