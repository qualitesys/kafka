18   package org.apache.kafka.connect.rest.basic.auth.extension;
19   
20   import org.apache.kafka.common.utils.AppInfoParser;
21   import org.apache.kafka.connect.errors.ConnectException;
22   import org.apache.kafka.connect.rest.ConnectRestExtension;
23   import org.apache.kafka.connect.rest.ConnectRestExtensionContext;
24   import org.slf4j.Logger;
25   import org.slf4j.LoggerFactory;
26   
27   import javax.security.auth.login.Configuration;
28   import java.io.IOException;
29   import java.util.Map;
30   import java.util.function.Supplier;
31   
32   /**
33    * Provides the ability to authenticate incoming BasicAuth credentials using the configured JAAS {@link
34    * javax.security.auth.spi.LoginModule}. An entry with the name {@code KafkaConnect} is expected in the JAAS config file configured in the
35    * JVM. An implementation of {@link javax.security.auth.spi.LoginModule} needs to be provided in the JAAS config file. The {@code
36    * LoginModule} implementation should configure the {@link javax.security.auth.callback.CallbackHandler} with only {@link
37    * javax.security.auth.callback.NameCallback} and {@link javax.security.auth.callback.PasswordCallback}.
38    *
39    * <p>To use this extension, one needs to add the following config in the {@code worker.properties}
40    * <pre>
41    *     rest.extension.classes = org.apache.kafka.connect.rest.basic.auth.extension.BasicAuthSecurityRestExtension
42    * </pre>
43    *
44    * <p> An example JAAS config would look as below
45    * <Pre>
46    *         KafkaConnect {
47    *              org.apache.kafka.connect.rest.basic.auth.extension.PropertyFileLoginModule required
48    *              file="/mnt/secret/credentials.properties";
49    *         };
50    *</Pre>
51    *
52    * <p>This is a reference implementation of the {@link ConnectRestExtension} interface. It registers an implementation of {@link
53    * javax.ws.rs.container.ContainerRequestFilter} that does JAAS based authentication of incoming Basic Auth credentials. {@link
54    * ConnectRestExtension} implementations are loaded via the plugin class loader using {@link java.util.ServiceLoader} mechanism and hence
55    * the packaged jar includes {@code META-INF/services/org.apache.kafka.connect.rest.extension.ConnectRestExtension} with the entry
56    * {@code org.apache.kafka.connect.extension.auth.jaas.BasicAuthSecurityRestExtension}
57    *
58    * <p><b>NOTE: The implementation ships with a default {@link PropertyFileLoginModule} that helps authenticate the request against a
59    * property file. {@link PropertyFileLoginModule} is NOT intended to be used in production since the credentials are stored in PLAINTEXT. One can use
60    * this extension in production by using their own implementation of {@link javax.security.auth.spi.LoginModule} that authenticates against
61    * stores like LDAP, DB, etc.</b>
62    */
63   public class BasicAuthSecurityRestExtension implements ConnectRestExtension {
64   
65       private static final Logger log = LoggerFactory.getLogger(BasicAuthSecurityRestExtension.class);
66   
67       private static final Supplier<Configuration> CONFIGURATION = initializeConfiguration(Configuration::getConfiguration);
68   
69       // Capture the JVM's global JAAS configuration as soon as possible, as it may be altered later
70       // by connectors, converters, other REST extensions, etc.
71       static Supplier<Configuration> initializeConfiguration(Supplier<Configuration> configurationSupplier) {
72           try {
73               Configuration configuration = configurationSupplier.get();
74               return () -> configuration;
75           } catch (Exception e) {
76               // We have to be careful not to throw anything here as this static block gets executed during plugin scanning and any exceptions will
77               // cause the worker to fail during startup, even if it's not configured to use the basic auth extension.
78               return () -> {
79                   throw new ConnectException("Failed to retrieve JAAS configuration", e);
80               };
81           }
82       }
83   
84       private final Supplier<Configuration> configuration;
85   
86       public BasicAuthSecurityRestExtension() {
87           this(CONFIGURATION);
88       }
89   
90       // For testing
91       BasicAuthSecurityRestExtension(Supplier<Configuration> configuration) {
92           this.configuration = configuration;
93       }
94   
95       @Override
96       public void register(ConnectRestExtensionContext restPluginContext) {
97           log.trace("Registering JAAS basic auth filter");
98           restPluginContext.configurable().register(new JaasBasicAuthFilter(configuration.get()));
99           log.trace("Finished registering JAAS basic auth filter");
100       }
101   
102       @Override
103       public void close() throws IOException {
104   
105       }
106   
107       @Override
108       public void configure(Map<String, ?> configs) {
109           // If we failed to retrieve a JAAS configuration during startup, throw that exception now
110           configuration.get();
111       }
112   
113       @Override
114       public String version() {
115           return AppInfoParser.getVersion();
116       }
117   }
