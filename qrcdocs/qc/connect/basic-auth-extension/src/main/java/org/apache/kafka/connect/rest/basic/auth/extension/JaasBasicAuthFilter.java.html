49   public class JaasBasicAuthFilter implements ContainerRequestFilter {
50   
51       private static final Logger log = LoggerFactory.getLogger(JaasBasicAuthFilter.class);
52       private static final Set<RequestMatcher> INTERNAL_REQUEST_MATCHERS = new HashSet<>(Arrays.asList(
53               new RequestMatcher(HttpMethod.POST, "/?connectors/([^/]+)/tasks/?"),
54               new RequestMatcher(HttpMethod.PUT, "/?connectors/[^/]+/fence/?")
55       ));
56       private static final String CONNECT_LOGIN_MODULE = "KafkaConnect";
57   
58       static final String AUTHORIZATION = "Authorization";
59   
60       // Package-private for testing
61       final Configuration configuration;
62   
63       private static class RequestMatcher implements Predicate<ContainerRequestContext> {
64           private final String method;
65           private final Pattern path;
66   
67           public RequestMatcher(String method, String path) {
68               this.method = method;
69               this.path = Pattern.compile(path);
70           }
71   
72           @Override
73           public boolean test(ContainerRequestContext requestContext) {
74               return requestContext.getMethod().equals(method)
75                       && path.matcher(requestContext.getUriInfo().getPath()).matches();
76           }
77       }
78   
79       public JaasBasicAuthFilter(Configuration configuration) {
80           this.configuration = configuration;
81       }
82   
83       @Override
84       public void filter(ContainerRequestContext requestContext) throws IOException {
85           if (isInternalRequest(requestContext)) {
86               log.trace("Skipping authentication for internal request");
87               return;
88           }
89   
90           try {
91               log.debug("Authenticating request");
92               LoginContext loginContext = new LoginContext(
93                   CONNECT_LOGIN_MODULE,
94                   null,
95                   new BasicAuthCallBackHandler(requestContext.getHeaderString(AUTHORIZATION)),
96                   configuration);
97               loginContext.login();
98           } catch (LoginException | ConfigException e) {
99               // Log at debug here in order to avoid polluting log files whenever someone mistypes their credentials
100               log.debug("Request failed authentication", e);
101               requestContext.abortWith(
102                   Response.status(Response.Status.UNAUTHORIZED)
103                       .entity("User cannot access the resource.")
104                       .build());
105           }
106       }
107   
108       private boolean isInternalRequest(ContainerRequestContext requestContext) {
109           return INTERNAL_REQUEST_MATCHERS.stream().anyMatch(m -> m.test(requestContext));
110       }
111   
112       public static class BasicAuthCallBackHandler implements CallbackHandler {
113   
114           private static final String BASIC = "basic";
115           private static final char COLON = ':';
116           private static final char SPACE = ' ';
117           private String username;
118           private String password;
119   
120           public BasicAuthCallBackHandler(String credentials) {
121               if (credentials == null) {
122                   log.trace("No credentials were provided with the request");
123                   return;
124               }
125   
126               int space = credentials.indexOf(SPACE);
127               if (space <= 0) {
128                   log.trace("Request credentials were malformed; no space present in value for authorization header");
129                   return;
130               }
131   
132               String method = credentials.substring(0, space);
133               if (!BASIC.equalsIgnoreCase(method)) {
134                   log.trace("Request credentials used {} authentication, but only {} supported; ignoring", method, BASIC);
135                   return;
136               }
137   
138               credentials = credentials.substring(space + 1);
139               credentials = new String(Base64.getDecoder().decode(credentials),
140                                        StandardCharsets.UTF_8);
141               int i = credentials.indexOf(COLON);
142               if (i <= 0) {
143                   log.trace("Request credentials were malformed; no colon present between username and password");
144                   return;
145               }
146   
147               username = credentials.substring(0, i);
148               password = credentials.substring(i + 1);
149           }
150   
151           @Override
152           public void handle(Callback[] callbacks) throws UnsupportedCallbackException {
153               List<Callback> unsupportedCallbacks = new ArrayList<>();
154               for (Callback callback : callbacks) {
155                   if (callback instanceof NameCallback) {
156                       ((NameCallback) callback).setName(username);
157                   } else if (callback instanceof PasswordCallback) {
158                       ((PasswordCallback) callback).setPassword(password != null
159                           ? password.toCharArray()
160                           : null
161                       );
162                   } else {
163                       unsupportedCallbacks.add(callback);
164                   }
165               }
166               if (!unsupportedCallbacks.isEmpty())
167                   throw new ConnectException(String.format(
168                       "Unsupported callbacks %s; request authentication will fail. "
169                           + "This indicates the Connect worker was configured with a JAAS "
170                           + "LoginModule that is incompatible with the %s, and will need to be "
171                           + "corrected and restarted.",
172                       unsupportedCallbacks,
173                       BasicAuthSecurityRestExtension.class.getSimpleName()
174                   ));
175           }
176       }
177   }
