45   /** Top-level config describing replication flows between multiple Kafka clusters.
46    *
47    *  Supports cluster-level properties of the form cluster.x.y.z, and replication-level
48    *  properties of the form source->target.x.y.z.
49    *  e.g.
50    *
51    *      clusters = A, B, C
52    *      A.bootstrap.servers = aaa:9092
53    *      A.security.protocol = SSL
54    *      --->%---
55    *      A->B.enabled = true
56    *      A->B.producer.client.id = "A-B-producer"
57    *      --->%---
58    *
59    */
60   public class MirrorMakerConfig extends AbstractConfig {

94       public List<SourceAndTarget> clusterPairs() {
95           List<SourceAndTarget> pairs = new ArrayList<>();
96           Set<String> clusters = clusters();
97           Map<String, String> originalStrings = originalsStrings();
98           boolean globalHeartbeatsEnabled = MirrorConnectorConfig.EMIT_HEARTBEATS_ENABLED_DEFAULT;
99           if (originalStrings.containsKey(MirrorConnectorConfig.EMIT_HEARTBEATS_ENABLED)) {
100               globalHeartbeatsEnabled = Boolean.parseBoolean(originalStrings.get(MirrorConnectorConfig.EMIT_HEARTBEATS_ENABLED));
101           }
102   
103           for (String source : clusters) {
104               for (String target : clusters) {
105                   if (!source.equals(target)) {
106                       String clusterPairConfigPrefix = source + "->" + target + ".";
107                       boolean clusterPairEnabled = Boolean.parseBoolean(originalStrings.get(clusterPairConfigPrefix + "enabled"));
108                       boolean clusterPairHeartbeatsEnabled = globalHeartbeatsEnabled;
109                       if (originalStrings.containsKey(clusterPairConfigPrefix + MirrorConnectorConfig.EMIT_HEARTBEATS_ENABLED)) {
110                           clusterPairHeartbeatsEnabled = Boolean.parseBoolean(originalStrings.get(clusterPairConfigPrefix + MirrorConnectorConfig.EMIT_HEARTBEATS_ENABLED));
111                       }
112   
113                       // By default, all source->target Herder combinations are created even if `x->y.enabled=false`
114                       // Unless `emit.heartbeats.enabled=false` or `x->y.emit.heartbeats.enabled=false`
115                       // Reason for this behavior: for a given replication flow A->B with heartbeats, 2 herders are required :
116                       // B->A for the MirrorHeartbeatConnector (emits heartbeats into A for monitoring replication health)
117                       // A->B for the MirrorSourceConnector (actual replication flow)
118                       if (clusterPairEnabled || clusterPairHeartbeatsEnabled) {
119                           pairs.add(new SourceAndTarget(source, target));
120                       }
121                   }
122               }
123           }
124           return pairs;
125       }

127       /** Construct a MirrorClientConfig from properties of the form cluster.x.y.z.
128         * Use to connect to a cluster based on the MirrorMaker top-level config file.

133           props.putAll(clusterProps(cluster));
134           return new MirrorClientConfig(transform(props));
135       }

137       // loads properties of the form cluster.x.y.z
138       Map<String, String> clusterProps(String cluster) {
139           Map<String, String> props = new HashMap<>();
140           Map<String, String> strings = originalsStrings();
141   
142           props.putAll(stringsWithPrefixStripped(cluster + "."));
143   
144           for (String k : MirrorClientConfig.CLIENT_CONFIG_DEF.names()) {
145               String v = props.get(k);
146               if (v != null) {
147                   props.putIfAbsent("producer." + k, v);
148                   props.putIfAbsent("consumer." + k, v);
149                   props.putIfAbsent("admin." + k, v);
150               }
151           }
152   
153           for (String k : MirrorClientConfig.CLIENT_CONFIG_DEF.names()) {
154               String v = strings.get(k);
155               if (v != null) {
156                   props.putIfAbsent("producer." + k, v);
157                   props.putIfAbsent("consumer." + k, v);
158                   props.putIfAbsent("admin." + k, v);
159                   props.putIfAbsent(k, v);
160               }
161           }
162    
163           return props;
164       }

166       // loads worker configs based on properties of the form x.y.z and cluster.x.y.z 
167       public Map<String, String> workerConfig(SourceAndTarget sourceAndTarget) {
168           Map<String, String> props = new HashMap<>();
169           props.putAll(clusterProps(sourceAndTarget.target()));
170         
171           // Accept common top-level configs that are otherwise ignored by MM2.
172           // N.B. all other worker properties should be configured for specific herders,
173           // e.g. primary->backup.client.id
174           props.putAll(stringsWithPrefix("offset.storage"));
175           props.putAll(stringsWithPrefix("config.storage"));
176           props.putAll(stringsWithPrefix("status.storage"));
177           props.putAll(stringsWithPrefix("key.converter")); 
178           props.putAll(stringsWithPrefix("value.converter")); 
179           props.putAll(stringsWithPrefix("header.converter"));
180           props.putAll(stringsWithPrefix("task"));
181           props.putAll(stringsWithPrefix("worker"));
182           props.putAll(stringsWithPrefix("replication.policy"));
183    
184           // transform any expression like ${provider:path:key}, since the worker doesn't do so
185           props = transform(props);
186           props.putAll(stringsWithPrefix(CONFIG_PROVIDERS_CONFIG));
187   
188           // fill in reasonable defaults
189           props.putIfAbsent(GROUP_ID_CONFIG, sourceAndTarget.source() + "-mm2");
190           props.putIfAbsent(DistributedConfig.OFFSET_STORAGE_TOPIC_CONFIG, "mm2-offsets."
191                   + sourceAndTarget.source() + ".internal");
192           props.putIfAbsent(DistributedConfig.STATUS_STORAGE_TOPIC_CONFIG, "mm2-status."
193                   + sourceAndTarget.source() + ".internal");
194           props.putIfAbsent(DistributedConfig.CONFIG_TOPIC_CONFIG, "mm2-configs."
195                   + sourceAndTarget.source() + ".internal");
196           props.putIfAbsent(KEY_CONVERTER_CLASS_CONFIG, BYTE_ARRAY_CONVERTER_CLASS); 
197           props.putIfAbsent(VALUE_CONVERTER_CLASS_CONFIG, BYTE_ARRAY_CONVERTER_CLASS); 
198           props.putIfAbsent(HEADER_CONVERTER_CLASS_CONFIG, BYTE_ARRAY_CONVERTER_CLASS);
199   
200           return props;
201       }

203       // loads properties of the form cluster.x.y.z and source->target.x.y.z
204       public Map<String, String> connectorBaseConfig(SourceAndTarget sourceAndTarget, Class<?> connectorClass) {
205           Map<String, String> props = new HashMap<>();
206   
207           props.putAll(originalsStrings());
208           props.keySet().retainAll(MirrorConnectorConfig.CONNECTOR_CONFIG_DEF.names());
209           
210           props.putAll(stringsWithPrefix(CONFIG_PROVIDERS_CONFIG));
211           props.putAll(stringsWithPrefix("replication.policy"));
212   
213           Map<String, String> sourceClusterProps = clusterProps(sourceAndTarget.source());
214           // attrs non prefixed with producer|consumer|admin
215           props.putAll(clusterConfigsWithPrefix(SOURCE_CLUSTER_PREFIX, sourceClusterProps));
216           // attrs prefixed with producer|consumer|admin
217           props.putAll(clientConfigsWithPrefix(SOURCE_PREFIX, sourceClusterProps));
218   
219           Map<String, String> targetClusterProps = clusterProps(sourceAndTarget.target());
220           props.putAll(clusterConfigsWithPrefix(TARGET_CLUSTER_PREFIX, targetClusterProps));
221           props.putAll(clientConfigsWithPrefix(TARGET_PREFIX, targetClusterProps));
222   
223           props.putIfAbsent(NAME, connectorClass.getSimpleName());
224           props.putIfAbsent(CONNECTOR_CLASS, connectorClass.getName());
225           props.putIfAbsent(SOURCE_CLUSTER_ALIAS, sourceAndTarget.source());
226           props.putIfAbsent(TARGET_CLUSTER_ALIAS, sourceAndTarget.target());
227   
228           // override with connector-level properties
229           props.putAll(stringsWithPrefixStripped(sourceAndTarget.source() + "->"
230               + sourceAndTarget.target() + "."));
231   
232           // disabled by default
233           props.putIfAbsent(MirrorConnectorConfig.ENABLED, "false");
234   
235           // don't transform -- the worker will handle transformation of Connector and Task configs
236           return props;
237       }

239       List<String> configProviders() {
240           return getList(CONFIG_PROVIDERS_CONFIG);
241       } 
242   
243       Map<String, String> transform(Map<String, String> props) {
244           // transform worker config according to config.providers
245           List<String> providerNames = configProviders();
246           Map<String, ConfigProvider> providers = new HashMap<>();
247           for (String name : providerNames) {
248               ConfigProvider configProvider = plugins.newConfigProvider(
249                       this,
250                       CONFIG_PROVIDERS_CONFIG + "." + name,
251                       Plugins.ClassLoaderUsage.PLUGINS
252               );
253               providers.put(name, configProvider);
254           }
255           ConfigTransformer transformer = new ConfigTransformer(providers);
256           Map<String, String> transformed = transformer.transform(props).data();
257           providers.values().forEach(x -> Utils.closeQuietly(x, "config provider"));
258           return transformed;
259       }

268                   in(Utils.enumOptions(SecurityProtocol.class)),

271               .withClientSslSupport()
272               .withClientSaslSupport();

274       private Map<String, String> stringsWithPrefixStripped(String prefix) {
275           return originalsStrings().entrySet().stream()
276               .filter(x -> x.getKey().startsWith(prefix))
277               .collect(Collectors.toMap(x -> x.getKey().substring(prefix.length()), Entry::getValue));
278       }
279   
280       private Map<String, String> stringsWithPrefix(String prefix) {
281           Map<String, String> strings = originalsStrings();
282           strings.keySet().removeIf(x -> !x.startsWith(prefix));
283           return strings;
284       }

286       static Map<String, String> clusterConfigsWithPrefix(String prefix, Map<String, String> props) {
287           return props.entrySet().stream()
288                   .filter(x -> !x.getKey().matches("(^consumer.*|^producer.*|^admin.*)"))
289                   .collect(Collectors.toMap(x -> prefix + x.getKey(), Entry::getValue));
290       }
291   
292       static Map<String, String> clientConfigsWithPrefix(String prefix, Map<String, String> props) {
293           return props.entrySet().stream()
294                   .filter(x -> x.getKey().matches("(^consumer.*|^producer.*|^admin.*)"))
295                   .collect(Collectors.toMap(x -> prefix + x.getKey(), Entry::getValue));
296       }
