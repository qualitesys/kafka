17   package org.apache.kafka.connect.json;
18   
19   import java.util.Locale;
20   import org.apache.kafka.common.config.ConfigDef;
21   import org.apache.kafka.common.config.ConfigDef.Importance;
22   import org.apache.kafka.common.config.ConfigDef.Type;
23   import org.apache.kafka.common.config.ConfigDef.Width;
24   import org.apache.kafka.connect.storage.ConverterConfig;
25   
26   import java.util.Map;
27   
28   /**
29    * Configuration options for {@link JsonConverter} instances.
30    */
31   public class JsonConverterConfig extends ConverterConfig {
32   
33       public static final String SCHEMAS_ENABLE_CONFIG = "schemas.enable";
34       public static final boolean SCHEMAS_ENABLE_DEFAULT = true;
35       private static final String SCHEMAS_ENABLE_DOC = "Include schemas within each of the serialized values and keys.";
36       private static final String SCHEMAS_ENABLE_DISPLAY = "Enable Schemas";
37   
38       public static final String SCHEMAS_CACHE_SIZE_CONFIG = "schemas.cache.size";
39       public static final int SCHEMAS_CACHE_SIZE_DEFAULT = 1000;
40       private static final String SCHEMAS_CACHE_SIZE_DOC = "The maximum number of schemas that can be cached in this converter instance.";
41       private static final String SCHEMAS_CACHE_SIZE_DISPLAY = "Schema Cache Size";
42   
43       public static final String DECIMAL_FORMAT_CONFIG = "decimal.format";
44       public static final String DECIMAL_FORMAT_DEFAULT = DecimalFormat.BASE64.name();
45       private static final String DECIMAL_FORMAT_DOC = "Controls which format this converter will serialize decimals in."
46           + " This value is case insensitive and can be either 'BASE64' (default) or 'NUMERIC'";
47       private static final String DECIMAL_FORMAT_DISPLAY = "Decimal Format";
48   
49       private final static ConfigDef CONFIG;
50   
51       static {
52           String group = "Schemas";
53           int orderInGroup = 0;
54           CONFIG = ConverterConfig.newConfigDef();
55           CONFIG.define(SCHEMAS_ENABLE_CONFIG, Type.BOOLEAN, SCHEMAS_ENABLE_DEFAULT, Importance.HIGH, SCHEMAS_ENABLE_DOC, group,
56                         orderInGroup++, Width.MEDIUM, SCHEMAS_ENABLE_DISPLAY);
57           CONFIG.define(SCHEMAS_CACHE_SIZE_CONFIG, Type.INT, SCHEMAS_CACHE_SIZE_DEFAULT, Importance.HIGH, SCHEMAS_CACHE_SIZE_DOC, group,
58                         orderInGroup++, Width.MEDIUM, SCHEMAS_CACHE_SIZE_DISPLAY);
59   
60           group = "Serialization";
61           orderInGroup = 0;
62           CONFIG.define(
63               DECIMAL_FORMAT_CONFIG, Type.STRING, DECIMAL_FORMAT_DEFAULT,
64               ConfigDef.CaseInsensitiveValidString.in(
65                   DecimalFormat.BASE64.name(),
66                   DecimalFormat.NUMERIC.name()),
67               Importance.LOW, DECIMAL_FORMAT_DOC, group, orderInGroup++,
68               Width.MEDIUM, DECIMAL_FORMAT_DISPLAY);
69       }
70   
71       public static ConfigDef configDef() {
72           return CONFIG;
73       }
74   
75       // cached config values
76       private final boolean schemasEnabled;
77       private final int schemaCacheSize;
78       private final DecimalFormat decimalFormat;
79   
80       public JsonConverterConfig(Map<String, ?> props) {
81           super(CONFIG, props);
82           this.schemasEnabled = getBoolean(SCHEMAS_ENABLE_CONFIG);
83           this.schemaCacheSize = getInt(SCHEMAS_CACHE_SIZE_CONFIG);
84           this.decimalFormat = DecimalFormat.valueOf(getString(DECIMAL_FORMAT_CONFIG).toUpperCase(Locale.ROOT));
85       }
86   
87       /**
88        * Return whether schemas are enabled.
89        *
90        * @return true if enabled, or false otherwise
91        */
92       public boolean schemasEnabled() {
93           return schemasEnabled;
94       }
95   
96       /**
97        * Get the cache size.
98        *
99        * @return the cache size
100        */
101       public int schemaCacheSize() {
102           return schemaCacheSize;
103       }
104   
105       /**
106        * Get the serialization format for decimal types.
107        *
108        * @return the decimal serialization format
109        */
110       public DecimalFormat decimalFormat() {
111           return decimalFormat;
112       }
113   
114   }
