61       public static ObjectNode envelope(JsonNode schema, JsonNode payload) {
62           ObjectNode result = JsonNodeFactory.instance.objectNode();
63           result.set(ENVELOPE_SCHEMA_FIELD_NAME, schema);
64           result.set(ENVELOPE_PAYLOAD_FIELD_NAME, payload);
65           return result;
66       }

72           public Envelope(JsonNode schema, JsonNode payload) {
73               this.schema = schema;

75           }

77           public ObjectNode toJsonNode() {
78               return envelope(schema, payload);
