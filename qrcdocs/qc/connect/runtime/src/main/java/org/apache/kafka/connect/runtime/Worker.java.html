210       /**
211        * Stop worker.
212        */
213       public void stop() {
214           log.info("Worker stopping");
215   
216           long started = time.milliseconds();
217           long limit = started + config.getLong(WorkerConfig.TASK_SHUTDOWN_GRACEFUL_TIMEOUT_MS_CONFIG);
218   
219           if (!connectors.isEmpty()) {
220               log.warn("Shutting down connectors {} uncleanly; herder should have shut down connectors before the Worker is stopped", connectors.keySet());
221               stopAndAwaitConnectors();
222           }
223   
224           if (!tasks.isEmpty()) {
225               log.warn("Shutting down tasks {} uncleanly; herder should have shut down tasks before the Worker is stopped", tasks.keySet());
226               stopAndAwaitTasks();
227           }
228   
229           long timeoutMs = limit - time.milliseconds();
230           sourceTaskOffsetCommitter.ifPresent(committer -> committer.close(timeoutMs));
231   
232           globalOffsetBackingStore.stop();
233           metrics.stop();
234   
235           log.info("Worker stopped");
236   
237           workerMetricsGroup.close();
238           connectorStatusMetricsGroup.close();
239   
240           workerConfigTransformer.close();
241           executor.shutdown();
242           try {
243               // Wait a while for existing tasks to terminate
244               if (!executor.awaitTermination(EXECUTOR_SHUTDOWN_TERMINATION_TIMEOUT_MS, TimeUnit.MILLISECONDS)) {
245                   executor.shutdownNow(); //cancel current executing threads
246                   // Wait a while for tasks to respond to being cancelled
247                   if (!executor.awaitTermination(EXECUTOR_SHUTDOWN_TERMINATION_TIMEOUT_MS, TimeUnit.MILLISECONDS))
248                       log.error("Executor did not terminate in time");
249               }
250           } catch (InterruptedException e) {
251               executor.shutdownNow(); // (Re-)Cancel if current thread also interrupted
252               // Preserve interrupt status
253               Thread.currentThread().interrupt();
254           }
255       }

257       /**
258        * Start a connector managed by this worker.
259        *
260        * @param connName the connector name.
261        * @param connProps the properties of the connector.
262        * @param ctx the connector runtime context.
263        * @param statusListener a listener for the runtime status transitions of the connector.
264        * @param initialState the initial state of the connector.
265        * @param onConnectorStateChange invoked when the initial state change of the connector is completed
266        */
267       public void startConnector(
268               String connName,
269               Map<String, String> connProps,
270               CloseableConnectorContext ctx,
271               ConnectorStatus.Listener statusListener,
272               TargetState initialState,
273               Callback<TargetState> onConnectorStateChange
274       ) {
275           final ConnectorStatus.Listener connectorStatusListener = workerMetricsGroup.wrapStatusListener(statusListener);
276           try (LoggingContext loggingContext = LoggingContext.forConnector(connName)) {
277               if (connectors.containsKey(connName)) {
278                   onConnectorStateChange.onCompletion(
279                           new ConnectException("Connector with name " + connName + " already exists"),
280                           null);
281                   return;
282               }
283   
284               final WorkerConnector workerConnector;
285               ClassLoader savedLoader = plugins.currentThreadLoader();
286               try {
287                   // By the time we arrive here, CONNECTOR_CLASS_CONFIG has been validated already
288                   // Getting this value from the unparsed map will allow us to instantiate the
289                   // right config (source or sink)
290                   final String connClass = connProps.get(ConnectorConfig.CONNECTOR_CLASS_CONFIG);
291                   ClassLoader connectorLoader = plugins.delegatingLoader().connectorLoader(connClass);
292                   savedLoader = Plugins.compareAndSwapLoaders(connectorLoader);
293   
294                   log.info("Creating connector {} of type {}", connName, connClass);
295                   final Connector connector = plugins.newConnector(connClass);
296                   final ConnectorConfig connConfig;
297                   final CloseableOffsetStorageReader offsetReader;
298                   final ConnectorOffsetBackingStore offsetStore;
299                   if (ConnectUtils.isSinkConnector(connector)) {
300                       connConfig = new SinkConnectorConfig(plugins, connProps);
301                       offsetReader = null;
302                       offsetStore = null;
303                   } else {
304                       SourceConnectorConfig sourceConfig = new SourceConnectorConfig(plugins, connProps, config.topicCreationEnable());
305                       connConfig = sourceConfig;
306   
307                       // Set up the offset backing store for this connector instance
308                       // (This logic is implemented in a follow-up pull request: https://github.com/apache/kafka/pull/11781)
309                       offsetStore = new ConnectorOffsetBackingStore(globalOffsetBackingStore, "TODO");
310                       offsetStore.configure(config);
311                       offsetReader = new OffsetStorageReaderImpl(offsetStore, connName, internalKeyConverter, internalValueConverter);
312                   }
313                   workerConnector = new WorkerConnector(
314                           connName, connector, connConfig, ctx, metrics, connectorStatusListener, offsetReader, offsetStore, connectorLoader);
315                   log.info("Instantiated connector {} with version {} of type {}", connName, connector.version(), connector.getClass());
316                   workerConnector.transitionTo(initialState, onConnectorStateChange);
317                   Plugins.compareAndSwapLoaders(savedLoader);
318               } catch (Throwable t) {
319                   log.error("Failed to start connector {}", connName, t);
320                   // Can't be put in a finally block because it needs to be swapped before the call on
321                   // statusListener
322                   Plugins.compareAndSwapLoaders(savedLoader);
323                   connectorStatusListener.onFailure(connName, t);
324                   onConnectorStateChange.onCompletion(t, null);
325                   return;
326               }
327   
328               WorkerConnector existing = connectors.putIfAbsent(connName, workerConnector);
329               if (existing != null) {
330                   onConnectorStateChange.onCompletion(
331                           new ConnectException("Connector with name " + connName + " already exists"),
332                           null);
333                   // Don't need to do any cleanup of the WorkerConnector instance (such as calling
334                   // shutdown() on it) here because it hasn't actually started running yet
335                   return;
336               }
337   
338               executor.submit(workerConnector);
339   
340               log.info("Finished creating connector {}", connName);
341           }
342       }

371       public List<Map<String, String>> connectorTaskConfigs(String connName, ConnectorConfig connConfig) {
372           List<Map<String, String>> result = new ArrayList<>();
373           try (LoggingContext loggingContext = LoggingContext.forConnector(connName)) {
374               log.trace("Reconfiguring connector tasks for {}", connName);
375   
376               WorkerConnector workerConnector = connectors.get(connName);
377               if (workerConnector == null)
378                   throw new ConnectException("Connector " + connName + " not found in this worker.");
379   
380               int maxTasks = connConfig.getInt(ConnectorConfig.TASKS_MAX_CONFIG);
381               Map<String, String> connOriginals = connConfig.originalsStrings();
382   
383               Connector connector = workerConnector.connector();
384               ClassLoader savedLoader = plugins.currentThreadLoader();
385               try {
386                   savedLoader = Plugins.compareAndSwapLoaders(workerConnector.loader());
387                   String taskClassName = connector.taskClass().getName();
388                   for (Map<String, String> taskProps : connector.taskConfigs(maxTasks)) {
389                       // Ensure we don't modify the connector's copy of the config
390                       Map<String, String> taskConfig = new HashMap<>(taskProps);
391                       taskConfig.put(TaskConfig.TASK_CLASS_CONFIG, taskClassName);
392                       if (connOriginals.containsKey(SinkTask.TOPICS_CONFIG)) {
393                           taskConfig.put(SinkTask.TOPICS_CONFIG, connOriginals.get(SinkTask.TOPICS_CONFIG));
394                       }
395                       if (connOriginals.containsKey(SinkTask.TOPICS_REGEX_CONFIG)) {
396                           taskConfig.put(SinkTask.TOPICS_REGEX_CONFIG, connOriginals.get(SinkTask.TOPICS_REGEX_CONFIG));
397                       }
398                       result.add(taskConfig);
399                   }
400               } finally {
401                   Plugins.compareAndSwapLoaders(savedLoader);
402               }
403           }
404   
405           return result;
406       }

440       private void awaitStopConnector(String connName, long timeout) {
441           try (LoggingContext loggingContext = LoggingContext.forConnector(connName)) {
442               WorkerConnector connector = connectors.remove(connName);
443               if (connector == null) {
444                   log.warn("Ignoring await stop request for non-present connector {}", connName);
445                   return;
446               }
447

597        * @param taskProps the tasks properties.

601        */
602       private boolean startTask(
603               ConnectorTaskId id,
604               Map<String, String> connProps,
605               Map<String, String> taskProps,
606               TaskStatus.Listener statusListener,
607               TaskBuilder taskBuilder
608       ) {
609           final WorkerTask workerTask;
610           final TaskStatus.Listener taskStatusListener = workerMetricsGroup.wrapStatusListener(statusListener);
611           try (LoggingContext loggingContext = LoggingContext.forTask(id)) {
612               log.info("Creating task {}", id);
613   
614               if (tasks.containsKey(id))
615                   throw new ConnectException("Task already exists in this worker: " + id);
616   
617               connectorStatusMetricsGroup.recordTaskAdded(id);
618               ClassLoader savedLoader = plugins.currentThreadLoader();
619               try {
620                   String connType = connProps.get(ConnectorConfig.CONNECTOR_CLASS_CONFIG);
621                   ClassLoader connectorLoader = plugins.delegatingLoader().connectorLoader(connType);
622                   savedLoader = Plugins.compareAndSwapLoaders(connectorLoader);
623                   final ConnectorConfig connConfig = new ConnectorConfig(plugins, connProps);
624                   final TaskConfig taskConfig = new TaskConfig(taskProps);
625                   final Class<? extends Task> taskClass = taskConfig.getClass(TaskConfig.TASK_CLASS_CONFIG).asSubclass(Task.class);
626                   final Task task = plugins.newTask(taskClass);
627                   log.info("Instantiated task {} with version {} of type {}", id, task.version(), taskClass.getName());
628   
629                   // By maintaining connector's specific class loader for this thread here, we first
630                   // search for converters within the connector dependencies.
631                   // If any of these aren't found, that means the connector didn't configure specific converters,
632                   // so we should instantiate based upon the worker configuration
633                   Converter keyConverter = plugins.newConverter(connConfig, WorkerConfig.KEY_CONVERTER_CLASS_CONFIG, ClassLoaderUsage
634                                                                                                                              .CURRENT_CLASSLOADER);
635                   Converter valueConverter = plugins.newConverter(connConfig, WorkerConfig.VALUE_CONVERTER_CLASS_CONFIG, ClassLoaderUsage.CURRENT_CLASSLOADER);
636                   HeaderConverter headerConverter = plugins.newHeaderConverter(connConfig, WorkerConfig.HEADER_CONVERTER_CLASS_CONFIG,
637                                                                                ClassLoaderUsage.CURRENT_CLASSLOADER);
638                   if (keyConverter == null) {
639                       keyConverter = plugins.newConverter(config, WorkerConfig.KEY_CONVERTER_CLASS_CONFIG, ClassLoaderUsage.PLUGINS);
640                       log.info("Set up the key converter {} for task {} using the worker config", keyConverter.getClass(), id);
641                   } else {
642                       log.info("Set up the key converter {} for task {} using the connector config", keyConverter.getClass(), id);
643                   }
644                   if (valueConverter == null) {
645                       valueConverter = plugins.newConverter(config, WorkerConfig.VALUE_CONVERTER_CLASS_CONFIG, ClassLoaderUsage.PLUGINS);
646                       log.info("Set up the value converter {} for task {} using the worker config", valueConverter.getClass(), id);
647                   } else {
648                       log.info("Set up the value converter {} for task {} using the connector config", valueConverter.getClass(), id);
649                   }
650                   if (headerConverter == null) {
651                       headerConverter = plugins.newHeaderConverter(config, WorkerConfig.HEADER_CONVERTER_CLASS_CONFIG, ClassLoaderUsage
652                                                                                                                                .PLUGINS);
653                       log.info("Set up the header converter {} for task {} using the worker config", headerConverter.getClass(), id);
654                   } else {
655                       log.info("Set up the header converter {} for task {} using the connector config", headerConverter.getClass(), id);
656                   }
657   
658                   workerTask = taskBuilder
659                           .withTask(task)
660                           .withConnectorConfig(connConfig)
661                           .withKeyConverter(keyConverter)
662                           .withValueConverter(valueConverter)
663                           .withHeaderConverter(headerConverter)
664                           .withClassloader(connectorLoader)
665                           .build();
666   
667                   workerTask.initialize(taskConfig);
668                   Plugins.compareAndSwapLoaders(savedLoader);
669               } catch (Throwable t) {
670                   log.error("Failed to start task {}", id, t);
671                   // Can't be put in a finally block because it needs to be swapped before the call on
672                   // statusListener
673                   Plugins.compareAndSwapLoaders(savedLoader);
674                   connectorStatusMetricsGroup.recordTaskRemoved(id);
675                   taskStatusListener.onFailure(id, t);
676                   return false;
677               }
678   
679               WorkerTask existing = tasks.putIfAbsent(id, workerTask);
680               if (existing != null)
681                   throw new ConnectException("Task already exists in this worker: " + id);
682   
683               executor.submit(workerTask);
684               if (workerTask instanceof WorkerSourceTask) {
685                   sourceTaskOffsetCommitter.ifPresent(committer -> committer.schedule(id, (WorkerSourceTask) workerTask));
686               }
687               return true;
688           }
