17   package org.apache.kafka.connect.storage;
18   
19   import org.apache.kafka.connect.data.SchemaAndValue;
20   import org.apache.kafka.connect.errors.ConnectException;
21   import org.slf4j.Logger;
22   import org.slf4j.LoggerFactory;
23   
24   import java.nio.ByteBuffer;
25   import java.util.Arrays;
26   import java.util.Collection;
27   import java.util.Collections;
28   import java.util.HashMap;
29   import java.util.HashSet;
30   import java.util.Map;
31   import java.util.Set;
32   import java.util.concurrent.CancellationException;
33   import java.util.concurrent.Future;
34   import java.util.concurrent.atomic.AtomicBoolean;
35   
36   /**
37    * Implementation of OffsetStorageReader. Unlike OffsetStorageWriter which is implemented
38    * directly, the interface is only separate from this implementation because it needs to be
39    * included in the public API package.
40    */
41   public class OffsetStorageReaderImpl implements CloseableOffsetStorageReader {
42       private static final Logger log = LoggerFactory.getLogger(OffsetStorageReaderImpl.class);
43   
44       private final OffsetBackingStore backingStore;
45       private final String namespace;
46       private final Converter keyConverter;
47       private final Converter valueConverter;
48       private final AtomicBoolean closed;
49       private final Set<Future<Map<ByteBuffer, ByteBuffer>>> offsetReadFutures;
50   
51       public OffsetStorageReaderImpl(OffsetBackingStore backingStore, String namespace,
52                                      Converter keyConverter, Converter valueConverter) {
53           this.backingStore = backingStore;
54           this.namespace = namespace;
55           this.keyConverter = keyConverter;
56           this.valueConverter = valueConverter;
57           this.closed = new AtomicBoolean(false);
58           this.offsetReadFutures = new HashSet<>();
59       }
60   
61       @Override
62       public <T> Map<String, Object> offset(Map<String, T> partition) {
63           return offsets(Collections.singletonList(partition)).get(partition);
64       }
65   
66       @Override
67       @SuppressWarnings("unchecked")
68       public <T> Map<Map<String, T>, Map<String, Object>> offsets(Collection<Map<String, T>> partitions) {
69           // Serialize keys so backing store can work with them
70           Map<ByteBuffer, Map<String, T>> serializedToOriginal = new HashMap<>(partitions.size());
71           for (Map<String, T> key : partitions) {
72               try {
73                   // Offsets are treated as schemaless, their format is only validated here (and the returned value below)
74                   OffsetUtils.validateFormat(key);
75                   byte[] keySerialized = keyConverter.fromConnectData(namespace, null, Arrays.asList(namespace, key));
76                   ByteBuffer keyBuffer = (keySerialized != null) ? ByteBuffer.wrap(keySerialized) : null;
77                   serializedToOriginal.put(keyBuffer, key);
78               } catch (Throwable t) {
79                   log.error("CRITICAL: Failed to serialize partition key when getting offsets for task with "
80                           + "namespace {}. No value for this data will be returned, which may break the "
81                           + "task or cause it to skip some data.", namespace, t);
82               }
83           }
84   
85           // Get serialized key -> serialized value from backing store
86           Map<ByteBuffer, ByteBuffer> raw;
87           try {
88               Future<Map<ByteBuffer, ByteBuffer>> offsetReadFuture;
89               synchronized (offsetReadFutures) {
90                   if (closed.get()) {
91                       throw new ConnectException(
92                           "Offset reader is closed. This is likely because the task has already been "
93                               + "scheduled to stop but has taken longer than the graceful shutdown "
94                               + "period to do so.");
95                   }
96                   offsetReadFuture = backingStore.get(serializedToOriginal.keySet());
97                   offsetReadFutures.add(offsetReadFuture);
98               }
99   
100               try {
101                   raw = offsetReadFuture.get();
102               } catch (CancellationException e) {
103                   throw new ConnectException(
104                       "Offset reader closed while attempting to read offsets. This is likely because "
105                           + "the task was been scheduled to stop but has taken longer than the "
106                           + "graceful shutdown period to do so.");
107               } finally {
108                   synchronized (offsetReadFutures) {
109                       offsetReadFutures.remove(offsetReadFuture);
110                   }
111               }
112           } catch (Exception e) {
113               log.error("Failed to fetch offsets from namespace {}: ", namespace, e);
114               throw new ConnectException("Failed to fetch offsets.", e);
115           }
116   
117           // Deserialize all the values and map back to the original keys
118           Map<Map<String, T>, Map<String, Object>> result = new HashMap<>(partitions.size());
119           for (Map.Entry<ByteBuffer, ByteBuffer> rawEntry : raw.entrySet()) {
120               try {
121                   // Since null could be a valid key, explicitly check whether map contains the key
122                   if (!serializedToOriginal.containsKey(rawEntry.getKey())) {
123                       log.error("Should be able to map {} back to a requested partition-offset key, backing "
124                               + "store may have returned invalid data", rawEntry.getKey());
125                       continue;
126                   }
127                   Map<String, T> origKey = serializedToOriginal.get(rawEntry.getKey());
128                   SchemaAndValue deserializedSchemaAndValue = valueConverter.toConnectData(namespace, rawEntry.getValue() != null ? rawEntry.getValue().array() : null);
129                   Object deserializedValue = deserializedSchemaAndValue.value();
130                   OffsetUtils.validateFormat(deserializedValue);
131   
132                   result.put(origKey, (Map<String, Object>) deserializedValue);
133               } catch (Throwable t) {
134                   log.error("CRITICAL: Failed to deserialize offset data when getting offsets for task with"
135                           + " namespace {}. No value for this data will be returned, which may break the "
136                           + "task or cause it to skip some data. This could either be due to an error in "
137                           + "the connector implementation or incompatible schema.", namespace, t);
138               }
139           }
140   
141           return result;
142       }
143   
144       @Override
145       public void close() {
146           if (!closed.getAndSet(true)) {
147               synchronized (offsetReadFutures) {
148                   for (Future<Map<ByteBuffer, ByteBuffer>> offsetReadFuture : offsetReadFutures) {
149                       try {
150                           offsetReadFuture.cancel(true);
151                       } catch (Throwable t) {
152                           log.error("Failed to cancel offset read future", t);
153                       }
154                   }
155                   offsetReadFutures.clear();
156               }
157           }
158       }
159   }
