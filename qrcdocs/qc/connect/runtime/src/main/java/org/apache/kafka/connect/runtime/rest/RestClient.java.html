48   public class RestClient {

50       private static final ObjectMapper JSON_SERDE = new ObjectMapper();

52       /**
53        * Sends HTTP request to remote REST server
54        *
55        * @param url             HTTP connection will be established with this url.
56        * @param method          HTTP method ("GET", "POST", "PUT", etc.)
57        * @param headers         HTTP headers from REST endpoint
58        * @param requestBodyData Object to serialize as JSON and send in the request body.
59        * @param responseFormat  Expected format of the response to the HTTP request.
60        * @param <T>             The type of the deserialized response to the HTTP request.
61        * @return The deserialized response to the HTTP request, or null if no data is expected.
62        */
63       public static <T> HttpResponse<T> httpRequest(String url, String method, HttpHeaders headers, Object requestBodyData,
64                                                     TypeReference<T> responseFormat, WorkerConfig config) {
65           return httpRequest(url, method, headers, requestBodyData, responseFormat, config, null, null);
66       }

68       /**
69        * Sends HTTP request to remote REST server
70        *
71        * @param url                       HTTP connection will be established with this url.
72        * @param method                    HTTP method ("GET", "POST", "PUT", etc.)
73        * @param headers                   HTTP headers from REST endpoint
74        * @param requestBodyData           Object to serialize as JSON and send in the request body.
75        * @param responseFormat            Expected format of the response to the HTTP request.
76        * @param <T>                       The type of the deserialized response to the HTTP request.
77        * @param sessionKey                The key to sign the request with (intended for internal requests only);
78        *                                  may be null if the request doesn't need to be signed
79        * @param requestSignatureAlgorithm The algorithm to sign the request with (intended for internal requests only);
80        *                                  may be null if the request doesn't need to be signed
81        * @return The deserialized response to the HTTP request, or null if no data is expected.
82        */
83       public static <T> HttpResponse<T> httpRequest(String url, String method, HttpHeaders headers, Object requestBodyData,
84                                                     TypeReference<T> responseFormat, WorkerConfig config,
85                                                     SecretKey sessionKey, String requestSignatureAlgorithm) {
86           HttpClient client;
87   
88           if (url.startsWith("https://")) {
89               client = new HttpClient(SSLUtils.createClientSideSslContextFactory(config));
90           } else {
91               client = new HttpClient();
92           }
93   
94           client.setFollowRedirects(false);
95   
96           try {
97               client.start();
98           } catch (Exception e) {
99               log.error("Failed to start RestClient: ", e);
100               throw new ConnectRestException(Response.Status.INTERNAL_SERVER_ERROR, "Failed to start RestClient: " + e.getMessage(), e);
101           }
102   
103           try {
104               String serializedBody = requestBodyData == null ? null : JSON_SERDE.writeValueAsString(requestBodyData);
105               log.trace("Sending {} with input {} to {}", method, serializedBody, url);
106   
107               Request req = client.newRequest(url);
108               req.method(method);
109               req.accept("application/json");
110               req.agent("kafka-connect");
111               addHeadersToRequest(headers, req);
112   
113               if (serializedBody != null) {
114                   req.content(new StringContentProvider(serializedBody, StandardCharsets.UTF_8), "application/json");
115               }
116   
117               if (sessionKey != null && requestSignatureAlgorithm != null) {
118                   InternalRequestSignature.addToRequest(
119                       sessionKey,
120                       serializedBody != null ? serializedBody.getBytes(StandardCharsets.UTF_8) : null,
121                       requestSignatureAlgorithm,
122                       req
123                   );
124               }
125   
126               ContentResponse res = req.send();
127   
128               int responseCode = res.getStatus();
129               log.debug("Request's response code: {}", responseCode);
130               if (responseCode == HttpStatus.NO_CONTENT_204) {
131                   return new HttpResponse<>(responseCode, convertHttpFieldsToMap(res.getHeaders()), null);
132               } else if (responseCode >= 400) {
133                   ErrorMessage errorMessage = JSON_SERDE.readValue(res.getContentAsString(), ErrorMessage.class);
134                   throw new ConnectRestException(responseCode, errorMessage.errorCode(), errorMessage.message());
135               } else if (responseCode >= 200 && responseCode < 300) {
136                   T result = JSON_SERDE.readValue(res.getContentAsString(), responseFormat);
137                   return new HttpResponse<>(responseCode, convertHttpFieldsToMap(res.getHeaders()), result);
138               } else {
139                   throw new ConnectRestException(Response.Status.INTERNAL_SERVER_ERROR,
140                           Response.Status.INTERNAL_SERVER_ERROR.getStatusCode(),
141                           "Unexpected status code when handling forwarded request: " + responseCode);
142               }
143           } catch (IOException | InterruptedException | TimeoutException | ExecutionException e) {
144               log.error("IO error forwarding REST request: ", e);
145               throw new ConnectRestException(Response.Status.INTERNAL_SERVER_ERROR, "IO Error trying to forward REST request: " + e.getMessage(), e);
146           } catch (Throwable t) {
147               log.error("Error forwarding REST request", t);
148               throw new ConnectRestException(Response.Status.INTERNAL_SERVER_ERROR, "Error trying to forward REST request: " + t.getMessage(), t);
149           } finally {
150               try {
151                   client.stop();
152               } catch (Exception e) {
153                   log.error("Failed to stop HTTP client", e);
154               }
155           }
156       }

159       /**
160        * Extract headers from REST call and add to client request
161        * @param headers         Headers from REST endpoint
162        * @param req             The client request to modify
163        */
164       private static void addHeadersToRequest(HttpHeaders headers, Request req) {
165           if (headers != null) {
166               String credentialAuthorization = headers.getHeaderString(HttpHeaders.AUTHORIZATION);
167               if (credentialAuthorization != null) {
168                   req.header(HttpHeaders.AUTHORIZATION, credentialAuthorization);
169               }
170           }
171       }
172   
173       /**
174        * Convert response parameters from Jetty format (HttpFields)
175        * @param httpFields
176        * @return
177        */
178       private static Map<String, String> convertHttpFieldsToMap(HttpFields httpFields) {
179           Map<String, String> headers = new HashMap<>();
180   
181           if (httpFields == null || httpFields.size() == 0)
182               return headers;
183   
184           for (HttpField field : httpFields) {
185               headers.put(field.getName(), field.getValue());
186           }
187   
188           return headers;
189       }

191       public static class HttpResponse<T> {
192           private final int status;
193           private final Map<String, String> headers;
194           private final T body;
195   
196           public HttpResponse(int status, Map<String, String> headers, T body) {
197               this.status = status;
198               this.headers = headers;
199               this.body = body;
200           }
201   
202           public int status() {
203               return status;
204           }
205   
206           public Map<String, String> headers() {
207               return headers;
208           }
209   
210           public T body() {
211               return body;
212           }
213       }
