17   package org.apache.kafka.connect.util;
18   
19   import org.apache.kafka.connect.runtime.WorkerConfig;
20   import org.slf4j.Logger;
21   import org.slf4j.LoggerFactory;
22   
23   import java.util.Collections;
24   import java.util.HashSet;
25   import java.util.LinkedHashMap;
26   import java.util.Map;
27   import java.util.Set;
28   
29   import static org.apache.kafka.connect.runtime.TopicCreationConfig.DEFAULT_TOPIC_CREATION_GROUP;
30   
31   /**
32    * Utility to be used by worker source tasks in order to create topics, if topic creation is
33    * enabled for source connectors at the worker and the connector configurations.
34    */
35   public class TopicCreation {
36       private static final Logger log = LoggerFactory.getLogger(TopicCreation.class);
37       private static final TopicCreation EMPTY =
38               new TopicCreation(false, null, Collections.emptyMap(), Collections.emptySet());
39   
40       private final boolean isTopicCreationEnabled;
41       private final TopicCreationGroup defaultTopicGroup;
42       private final Map<String, TopicCreationGroup> topicGroups;
43       private final Set<String> topicCache;
44   
45       protected TopicCreation(boolean isTopicCreationEnabled,
46                               TopicCreationGroup defaultTopicGroup,
47                               Map<String, TopicCreationGroup> topicGroups,
48                               Set<String> topicCache) {
49           this.isTopicCreationEnabled = isTopicCreationEnabled;
50           this.defaultTopicGroup = defaultTopicGroup;
51           this.topicGroups = topicGroups;
52           this.topicCache = topicCache;
53       }
54   
55       public static TopicCreation newTopicCreation(WorkerConfig workerConfig,
56               Map<String, TopicCreationGroup> topicGroups) {
57           if (!workerConfig.topicCreationEnable() || topicGroups == null) {
58               return EMPTY;
59           }
60           Map<String, TopicCreationGroup> groups = new LinkedHashMap<>(topicGroups);
61           groups.remove(DEFAULT_TOPIC_CREATION_GROUP);
62           return new TopicCreation(true, topicGroups.get(DEFAULT_TOPIC_CREATION_GROUP), groups, new HashSet<>());
63       }
64   
65       /**
66        * Return an instance of this utility that represents what the state of the internal data
67        * structures should be when topic creation is disabled.
68        *
69        * @return the utility when topic creation is disabled
70        */
71       public static TopicCreation empty() {
72           return EMPTY;
73       }
74   
75       /**
76        * Check whether topic creation is enabled for this utility instance. This is state is set at
77        * instantiation time and remains unchanged for the lifetime of every {@link TopicCreation}
78        * object.
79        *
80        * @return true if topic creation is enabled; false otherwise
81        */
82       public boolean isTopicCreationEnabled() {
83           return isTopicCreationEnabled;
84       }
85   
86       /**
87        * Check whether topic creation may be required for a specific topic name.
88        *
89        * @return true if topic creation is enabled and the topic name is not in the topic cache;
90        * false otherwise
91        */
92       public boolean isTopicCreationRequired(String topic) {
93           return isTopicCreationEnabled && !topicCache.contains(topic);
94       }
95   
96       /**
97        * Return the default topic creation group. This group is always defined when topic creation is
98        * enabled but is {@code null} if topic creation is disabled.
99        *
100        * @return the default topic creation group if topic creation is enabled; {@code null} otherwise
101        */
102       public TopicCreationGroup defaultTopicGroup() {
103           return defaultTopicGroup;
104       }
105   
106       /**
107        * Return the topic creation groups defined for a source connector as a map of topic creation
108        * group name to topic creation group instance. This map maintains all the optionally defined
109        * groups besides the default group which is defined for any connector when topic creation is
110        * enabled.
111        *
112        * @return the map of all the topic creation groups besides the default group; may be empty
113        * but not {@code null}
114        */
115       public Map<String, TopicCreationGroup> topicGroups() {
116           return topicGroups;
117       }
118   
119       /**
120        * Inform this utility instance that a topic has been created and its creation will no
121        * longer be required. After {@link #addTopic(String)} is called for a give {@param topic}
122        * any subsequent calls to {@link #isTopicCreationRequired} will return {@code false} for the
123        * same topic.
124        *
125        * @param topic the topic name to mark as created
126        */
127       public void addTopic(String topic) {
128           if (isTopicCreationEnabled) {
129               topicCache.add(topic);
130           }
131       }
132   
133       /**
134        * Get the first topic creation group that is configured to match the given {@param topic}
135        * name. If topic creation is enabled, any topic should match at least the default topic
136        * creation group.
137        *
138        * @param topic the topic name to match against group configurations
139        *
140        * @return the first group that matches the given topic
141        */
142       public TopicCreationGroup findFirstGroup(String topic) {
143           return topicGroups.values().stream()
144                   .filter(group -> group.matches(topic))
145                   .findFirst()
146                   .orElse(defaultTopicGroup);
147       }
148   }
