17   package org.apache.kafka.connect.storage;
18   
19   import org.apache.kafka.connect.data.ConnectSchema;
20   import org.apache.kafka.connect.data.Schema;
21   import org.apache.kafka.connect.errors.DataException;
22   
23   import java.util.Map;
24   
25   public class OffsetUtils {
26       @SuppressWarnings("unchecked")
27       public static void validateFormat(Object offsetData) {
28           if (offsetData == null)
29               return;
30   
31           if (!(offsetData instanceof Map))
32               throw new DataException("Offsets must be specified as a Map");
33           validateFormat((Map<Object, Object>) offsetData);
34       }
35   
36       public static <K, V> void validateFormat(Map<K, V> offsetData) {
37           // Both keys and values for offsets may be null. For values, this is a useful way to delete offsets or indicate
38           // that there's not usable concept of offsets in your source system.
39           if (offsetData == null)
40               return;
41   
42           for (Map.Entry<K, V> entry : offsetData.entrySet()) {
43               if (!(entry.getKey() instanceof String))
44                   throw new DataException("Offsets may only use String keys");
45   
46               Object value = entry.getValue();
47               if (value == null)
48                   continue;
49               Schema.Type schemaType = ConnectSchema.schemaType(value.getClass());
50               if (schemaType == null)
51                   throw new DataException("Offsets may only contain primitive types as values, but field " + entry.getKey() + " contains " + value.getClass());
52               if (!schemaType.isPrimitive())
53                   throw new DataException("Offsets may only contain primitive types as values, but field " + entry.getKey() + " contains " + schemaType);
54           }
55       }
56   }
