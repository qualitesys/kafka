18   package org.apache.kafka.connect.converters;
19   
20   import org.apache.kafka.common.config.ConfigDef;
21   import org.apache.kafka.connect.data.Schema;
22   import org.apache.kafka.connect.data.SchemaAndValue;
23   import org.apache.kafka.connect.errors.DataException;
24   import org.apache.kafka.connect.storage.Converter;
25   import org.apache.kafka.connect.storage.ConverterConfig;
26   import org.apache.kafka.connect.storage.HeaderConverter;
27   
28   import java.util.Map;
29   
30   /**
31    * Pass-through converter for raw byte data.
32    *
33    * This implementation currently does nothing with the topic names or header names.
34    */
35   public class ByteArrayConverter implements Converter, HeaderConverter {
36   
37       private static final ConfigDef CONFIG_DEF = ConverterConfig.newConfigDef();
38   
39       @Override
40       public ConfigDef config() {
41           return CONFIG_DEF;
42       }
43   
44       @Override
45       public void configure(Map<String, ?> configs) {
46       }
47   
48       @Override
49       public void configure(Map<String, ?> configs, boolean isKey) {
50       }
51   
52       @Override
53       public byte[] fromConnectData(String topic, Schema schema, Object value) {
54           if (schema != null && schema.type() != Schema.Type.BYTES)
55               throw new DataException("Invalid schema type for ByteArrayConverter: " + schema.type().toString());
56   
57           if (value != null && !(value instanceof byte[]))
58               throw new DataException("ByteArrayConverter is not compatible with objects of type " + value.getClass());
59   
60           return (byte[]) value;
61       }
62   
63       @Override
64       public SchemaAndValue toConnectData(String topic, byte[] value) {
65           return new SchemaAndValue(Schema.OPTIONAL_BYTES_SCHEMA, value);
66       }
67   
68       @Override
69       public byte[] fromConnectHeader(String topic, String headerKey, Schema schema, Object value) {
70           return fromConnectData(topic, schema, value);
71       }
72   
73       @Override
74       public SchemaAndValue toConnectHeader(String topic, String headerKey, byte[] value) {
75           return toConnectData(topic, value);
76       }
77   
78       @Override
79       public void close() {
80           // do nothing
81       }
82   }
