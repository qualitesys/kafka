43   /**
44    * Container for connectors which is responsible for managing their lifecycle (e.g. handling startup,
45    * shutdown, pausing, etc.). Internally, we manage the runtime state of the connector and transition according
46    * to target state changes. Note that unlike connector tasks, the connector does not really have a "pause"
47    * state which is distinct from being stopped. We therefore treat pause operations as requests to momentarily
48    * stop the connector, and resume operations as requests to restart it (without reinitialization). Connector
49    * failures, whether in initialization or after startup, are treated as fatal, which means that we will not attempt
50    * to restart this connector instance after failure. What this means from a user perspective is that you must
51    * use the /restart REST API to restart a failed task. This behavior is consistent with task failures.
52    *
53    * Note that this class is NOT thread-safe.
54    */
55   public class WorkerConnector implements Runnable {
56       private static final Logger log = LoggerFactory.getLogger(WorkerConnector.class);
57       private static final String THREAD_NAME_PREFIX = "connector-thread-";
58   
59       private enum State {
60           INIT,    // initial state before startup
61           STOPPED, // the connector has been stopped/paused.
62           STARTED, // the connector has been started/resumed.
63           FAILED,  // the connector has failed (no further transitions are possible after this state)
64       }
65   
66       private final String connName;
67       private final Map<String, String> config;
68       private final ConnectorStatus.Listener statusListener;
69       private final ClassLoader loader;
70       private final CloseableConnectorContext ctx;
71       private final Connector connector;
72       private final ConnectorMetricsGroup metrics;
73       private final AtomicReference<TargetState> pendingTargetStateChange;
74       private final AtomicReference<Callback<TargetState>> pendingStateChangeCallback;
75       private final CountDownLatch shutdownLatch;
76       private volatile boolean stopping;  // indicates whether the Worker has asked the connector to stop
77       private volatile boolean cancelled; // indicates whether the Worker has cancelled the connector (e.g. because of slow shutdown)
78   
79       private State state;
80       private final CloseableOffsetStorageReader offsetStorageReader;
81       private final ConnectorOffsetBackingStore offsetStore;

83       public WorkerConnector(String connName,
84                              Connector connector,
85                              ConnectorConfig connectorConfig,
86                              CloseableConnectorContext ctx,
87                              ConnectMetrics metrics,
88                              ConnectorStatus.Listener statusListener,
89                              CloseableOffsetStorageReader offsetStorageReader,
90                              ConnectorOffsetBackingStore offsetStore,
91                              ClassLoader loader) {
92           this.connName = connName;
93           this.config = connectorConfig.originalsStrings();
94           this.loader = loader;
95           this.ctx = ctx;
96           this.connector = connector;
97           this.state = State.INIT;
98           this.metrics = new ConnectorMetricsGroup(metrics, AbstractStatus.State.UNASSIGNED, statusListener);
99           this.statusListener = this.metrics;
100           this.offsetStorageReader = offsetStorageReader;
101           this.offsetStore = offsetStore;
102           this.pendingTargetStateChange = new AtomicReference<>();
103           this.pendingStateChangeCallback = new AtomicReference<>();
104           this.shutdownLatch = new CountDownLatch(1);
105           this.stopping = false;
106           this.cancelled = false;
107       }

110           return loader;

113       @Override
114       public void run() {
115           // Clear all MDC parameters, in case this thread is being reused
116           LoggingContext.clear();
117   
118           try (LoggingContext loggingContext = LoggingContext.forConnector(connName)) {
119               ClassLoader savedLoader = Plugins.compareAndSwapLoaders(loader);
120               String savedName = Thread.currentThread().getName();
121               try {
122                   Thread.currentThread().setName(THREAD_NAME_PREFIX + connName);
123                   doRun();
124               } finally {
125                   Thread.currentThread().setName(savedName);
126                   Plugins.compareAndSwapLoaders(savedLoader);
127               }
128           } finally {
129               // In the rare case of an exception being thrown outside the doRun() method, or an
130               // uncaught one being thrown from within it, mark the connector as shut down to avoid
131               // unnecessarily blocking and eventually timing out during awaitShutdown
132               shutdownLatch.countDown();
133           }
134       }
135   
136       void doRun() {
137           initialize();
138           while (!stopping) {
139               TargetState newTargetState;
140               Callback<TargetState> stateChangeCallback;
141               synchronized (this) {
142                   newTargetState = pendingTargetStateChange.getAndSet(null);
143                   stateChangeCallback = pendingStateChangeCallback.getAndSet(null);
144               }
145               if (newTargetState != null && !stopping) {
146                   doTransitionTo(newTargetState, stateChangeCallback);
147               }
148               synchronized (this) {
149                   if (pendingTargetStateChange.get() != null || stopping) {
150                       // An update occurred before we entered the synchronized block; no big deal,
151                       // just start the loop again until we've handled everything
152                   } else {
153                       try {
154                           wait();
155                       } catch (InterruptedException e) {
156                           // We'll pick up any potential state changes at the top of the loop
157                       }
158                   }
159               }
160           }
161           doShutdown();
162       }
163   
164       void initialize() {
165           try {
166               if (!isSourceConnector() && !isSinkConnector()) {
167                   throw new ConnectException("Connector implementations must be a subclass of either SourceConnector or SinkConnector");
168               }
169               log.debug("{} Initializing connector {}", this, connName);
170               if (isSinkConnector()) {
171                   SinkConnectorConfig.validate(config);
172                   connector.initialize(new WorkerSinkConnectorContext());
173               } else {
174                   Objects.requireNonNull(offsetStorageReader, "Offset reader cannot be null for source connectors");
175                   connector.initialize(new WorkerSourceConnectorContext(offsetStorageReader));
176               }
177           } catch (Throwable t) {
178               log.error("{} Error initializing connector", this, t);
179               onFailure(t);
180           }
181       }
182   
183       private boolean doStart() throws Throwable {
184           try {
185               switch (state) {
186                   case STARTED:
187                       return false;
188   
189                   case INIT:
190                   case STOPPED:
191                       connector.start(config);
192                       this.state = State.STARTED;
193                       return true;
194   
195                   default:
196                       throw new IllegalArgumentException("Cannot start connector in state " + state);
197               }
198           } catch (Throwable t) {
199               log.error("{} Error while starting connector", this, t);
200               onFailure(t);
201               throw t;
202           }
203       }
204   
205       private void onFailure(Throwable t) {
206           statusListener.onFailure(connName, t);
207           this.state = State.FAILED;
208       }
209   
210       private void resume() throws Throwable {
211           if (doStart())
212               statusListener.onResume(connName);
213       }
214   
215       private void start() throws Throwable {
216           if (doStart())
217               statusListener.onStartup(connName);
218       }

220       public boolean isRunning() {
221           return state == State.STARTED;
222       }

224       @SuppressWarnings("fallthrough")
225       private void pause() {
226           try {
227               switch (state) {
228                   case STOPPED:
229                       return;
230   
231                   case STARTED:
232                       connector.stop();
233                       // fall through
234   
235                   case INIT:
236                       statusListener.onPause(connName);
237                       this.state = State.STOPPED;
238                       break;
239   
240                   default:
241                       throw new IllegalArgumentException("Cannot pause connector in state " + state);
242               }
243           } catch (Throwable t) {
244               log.error("{} Error while shutting down connector", this, t);
245               statusListener.onFailure(connName, t);
246               this.state = State.FAILED;
247           }
248       }
249   
250       /**
251        * Stop this connector. This method does not block, it only triggers shutdown. Use
252        * #{@link #awaitShutdown} to block until completion.
253        */
254       public synchronized void shutdown() {
255           log.info("Scheduled shutdown for {}", this);
256           stopping = true;
257           notify();
258       }
259   
260       void doShutdown() {
261           try {
262               TargetState preEmptedState = pendingTargetStateChange.getAndSet(null);
263               Callback<TargetState> stateChangeCallback = pendingStateChangeCallback.getAndSet(null);
264               if (stateChangeCallback != null) {
265                   stateChangeCallback.onCompletion(
266                           new ConnectException(
267                                   "Could not begin changing connector state to " + preEmptedState.name()
268                                       + " as the connector has been scheduled for shutdown"),
269                           null);
270               }
271               if (state == State.STARTED)
272                   connector.stop();
273               this.state = State.STOPPED;
274               statusListener.onShutdown(connName);
275               log.info("Completed shutdown for {}", this);
276           } catch (Throwable t) {
277               log.error("{} Error while shutting down connector", this, t);
278               state = State.FAILED;
279               statusListener.onFailure(connName, t);
280           } finally {
281               Utils.closeQuietly(ctx, "connector context for " + connName);
282               Utils.closeQuietly(metrics, "connector metrics for " + connName);
283               Utils.closeQuietly(offsetStorageReader, "offset reader for " + connName);
284               Utils.closeQuietly(offsetStore::stop, "offset backing store for " + connName);
285           }
286       }
287   
288       public synchronized void cancel() {
289           // Proactively update the status of the connector to UNASSIGNED since this connector
290           // instance is being abandoned and we won't update the status on its behalf any more
291           // after this since a new instance may be started soon
292           statusListener.onShutdown(connName);
293           Utils.closeQuietly(ctx, "connector context for " + connName);
294           // Preemptively close the offset reader in case the connector is blocked on an offset read.
295           Utils.closeQuietly(offsetStorageReader, "offset reader for " + connName);
296           cancelled = true;
297       }

299       /**
300        * Wait for this connector to finish shutting down.
301        *
302        * @param timeoutMs time in milliseconds to await shutdown
303        * @return true if successful, false if the timeout was reached
304        */
305       public boolean awaitShutdown(long timeoutMs) {
306           try {
307               return shutdownLatch.await(timeoutMs, TimeUnit.MILLISECONDS);
308           } catch (InterruptedException e) {
309               return false;
310           }
311       }

313       public void transitionTo(TargetState targetState, Callback<TargetState> stateChangeCallback) {
314           Callback<TargetState> preEmptedStateChangeCallback;
315           TargetState preEmptedState;
316           synchronized (this) {
317               preEmptedStateChangeCallback = pendingStateChangeCallback.getAndSet(stateChangeCallback);
318               preEmptedState = pendingTargetStateChange.getAndSet(targetState);
319               notify();
320           }
321           if (preEmptedStateChangeCallback != null) {
322               preEmptedStateChangeCallback.onCompletion(
323                       new ConnectException(
324                               "Could not begin changing connector state to " + preEmptedState.name()
325                                   + " before another request to change state was made;"
326                                   + " the new request (which is to change the state to " + targetState.name()
327                                   + ") has pre-empted this one"),
328                       null
329               );
330           }
331       }
332   
333       void doTransitionTo(TargetState targetState, Callback<TargetState> stateChangeCallback) {
334           if (state == State.FAILED) {
335               stateChangeCallback.onCompletion(
336                       new ConnectException(this + " Cannot transition connector to " + targetState + " since it has failed"),
337                       null);
338               return;
339           }
340   
341           try {
342               doTransitionTo(targetState);
343               stateChangeCallback.onCompletion(null, targetState);
344           } catch (Throwable t) {
345               stateChangeCallback.onCompletion(
346                       new ConnectException(
347                               "Failed to transition connector " + connName + " to state " + targetState,
348                               t),
349                       null);
350           }
351       }
352   
353       private void doTransitionTo(TargetState targetState) throws Throwable {
354           log.debug("{} Transition connector to {}", this, targetState);
355           if (targetState == TargetState.PAUSED) {
356               pause();
357           } else if (targetState == TargetState.STARTED) {
358               if (state == State.INIT)
359                   start();
360               else
361                   resume();
362           } else {
363               throw new IllegalArgumentException("Unhandled target state " + targetState);
364           }
365       }

398       class ConnectorMetricsGroup implements ConnectorStatus.Listener, AutoCloseable {
399           /**
400            * Use {@link AbstractStatus.State} since it has all of the states we want,
401            * unlike {@link WorkerConnector.State}.
402            */
403           private volatile AbstractStatus.State state;
404           private final MetricGroup metricGroup;
405           private final ConnectorStatus.Listener delegate;
406   
407           public ConnectorMetricsGroup(ConnectMetrics connectMetrics, AbstractStatus.State initialState, ConnectorStatus.Listener delegate) {
408               Objects.requireNonNull(connectMetrics);
409               Objects.requireNonNull(connector);
410               Objects.requireNonNull(initialState);
411               Objects.requireNonNull(delegate);
412               this.delegate = delegate;
413               this.state = initialState;
414               ConnectMetricsRegistry registry = connectMetrics.registry();
415               this.metricGroup = connectMetrics.group(registry.connectorGroupName(),
416                       registry.connectorTagName(), connName);
417               // prevent collisions by removing any previously created metrics in this group.
418               metricGroup.close();
419   
420               metricGroup.addImmutableValueMetric(registry.connectorType, connectorType());
421               metricGroup.addImmutableValueMetric(registry.connectorClass, connector.getClass().getName());
422               metricGroup.addImmutableValueMetric(registry.connectorVersion, connector.version());
423               metricGroup.addValueMetric(registry.connectorStatus, now -> state.toString().toLowerCase(Locale.getDefault()));
424           }
425   
426           public void close() {
427               metricGroup.close();
428           }
