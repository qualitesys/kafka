17   package org.apache.kafka.connect.util;
18   
19   import org.apache.kafka.common.errors.RetriableException;
20   import org.apache.kafka.common.errors.WakeupException;
21   import org.apache.kafka.common.utils.Utils;
22   import org.apache.kafka.connect.errors.ConnectException;
23   import org.slf4j.Logger;
24   import org.slf4j.LoggerFactory;
25   
26   import java.time.Duration;
27   import java.util.Optional;
28   import java.util.concurrent.Callable;
29   import java.util.function.Supplier;
30   
31   public class RetryUtil {
32       private static final Logger log = LoggerFactory.getLogger(RetryUtil.class);
33   
34       /**
35        * The method executes the callable at least once, optionally retrying the callable if
36        * {@link org.apache.kafka.connect.errors.RetriableException} is being thrown.  If timeout is exhausted,
37        * then the last exception is wrapped with a {@link org.apache.kafka.connect.errors.ConnectException} and thrown.
38        *
39        * <p>{@code description} supplies the message that indicates the purpose of the callable since the message will
40        * be used for logging.  For example, "list offsets". If the supplier is null or the supplied string is
41        * null, {@code callable} will be used as the default string.
42        *
43        * <p>The task will be executed at least once. No retries will be performed 
44        * if {@code timeoutDuration} is 0 or negative, or if {@code timeoutDuration} is less than {@code retryBackoffMs}.
45        *
46        * <p>A {@code retryBackoffMs} that is negative or zero will result in no delays between retries.
47        *
48        * @param callable          the function to execute
49        * @param description       supplier that provides custom message for logging purpose
50        * @param timeoutDuration   timeout duration; must not be null
51        * @param retryBackoffMs    the number of milliseconds to delay upon receiving a
52        *                          {@link org.apache.kafka.connect.errors.RetriableException} before retrying again
53        * @throws ConnectException If the task exhausted all the retries
54        */
55       public static <T> T retryUntilTimeout(Callable<T> callable, Supplier<String> description, Duration timeoutDuration, long retryBackoffMs) throws Exception {
56           // if null supplier or string is provided, the message will be default to "callabe"
57           final String descriptionStr = Optional.ofNullable(description)
58                   .map(Supplier::get)
59                   .orElse("callable");
60   
61           // handling null duration
62           final long timeoutMs = Optional.ofNullable(timeoutDuration)
63                   .map(Duration::toMillis)
64                   .orElse(0L);
65   
66           if (retryBackoffMs < 0) {
67               log.debug("Assuming no retry backoff since retryBackoffMs={} is negative", retryBackoffMs);
68               retryBackoffMs = 0;
69           }
70           if (timeoutMs <= 0 || retryBackoffMs >= timeoutMs) {
71               log.debug("Executing {} only once, since timeoutMs={} is not larger than retryBackoffMs={}",
72                       descriptionStr, timeoutMs, retryBackoffMs);
73               return callable.call();
74           }
75   
76           final long end = System.currentTimeMillis() + timeoutMs;
77           int attempt = 0;
78           Throwable lastError = null;
79           do {
80               attempt++;
81               try {
82                   return callable.call();
83               } catch (RetriableException | org.apache.kafka.connect.errors.RetriableException e) {
84                   log.warn("Attempt {} to {} resulted in RetriableException; retrying automatically. " +
85                           "Reason: {}", attempt, descriptionStr, e.getMessage(), e);
86                   lastError = e;
87               } catch (WakeupException e) {
88                   lastError = e;
89               }
90   
91               if (retryBackoffMs > 0) {
92                   long millisRemaining = Math.max(0, end - System.currentTimeMillis());
93                   if (millisRemaining < retryBackoffMs) {
94                       // exit when the time remaining is less than retryBackoffMs
95                       break;
96                   }
97                   Utils.sleep(retryBackoffMs);
98               }
99           } while (System.currentTimeMillis() < end);
100   
101           throw new ConnectException("Fail to " + descriptionStr + " after " + attempt + " attempts.  Reason: " + lastError.getMessage(), lastError);
102       }
103   }
