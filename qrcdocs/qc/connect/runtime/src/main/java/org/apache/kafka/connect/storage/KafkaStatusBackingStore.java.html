67   /**
68    * StatusBackingStore implementation which uses a compacted topic for storage
69    * of connector and task status information. When a state change is observed,
70    * the new state is written to the compacted topic. The new state will not be
71    * visible until it has been read back from the topic.
72    *
73    * In spite of their names, the putSafe() methods cannot guarantee the safety
74    * of the write (since Kafka itself cannot provide such guarantees currently),
75    * but it can avoid specific unsafe conditions. In particular, we putSafe()
76    * allows writes in the following conditions:
77    *
78    * 1) It is (probably) safe to overwrite the state if there is no previous
79    *    value.
80    * 2) It is (probably) safe to overwrite the state if the previous value was
81    *    set by a worker with the same workerId.
82    * 3) It is (probably) safe to overwrite the previous state if the current
83    *    generation is higher than the previous .
84    *
85    * Basically all these conditions do is reduce the window for conflicts. They
86    * obviously cannot take into account in-flight requests.
87    *
88    */

161       @Override
162       public void configure(final WorkerConfig config) {
163           this.statusTopic = config.getString(DistributedConfig.STATUS_STORAGE_TOPIC_CONFIG);
164           if (this.statusTopic == null || this.statusTopic.trim().length() == 0)
165               throw new ConfigException("Must specify topic for connector status.");
166   
167           String clusterId = ConnectUtils.lookupKafkaClusterId(config);
168           Map<String, Object> originals = config.originals();
169           Map<String, Object> producerProps = new HashMap<>(originals);
170           producerProps.put(ProducerConfig.KEY_SERIALIZER_CLASS_CONFIG, StringSerializer.class.getName());
171           producerProps.put(ProducerConfig.VALUE_SERIALIZER_CLASS_CONFIG, ByteArraySerializer.class.getName());
172           producerProps.put(ProducerConfig.RETRIES_CONFIG, 0); // we handle retries in this class
173           // By default, Connect disables idempotent behavior for all producers, even though idempotence became
174           // default for Kafka producers. This is to ensure Connect continues to work with many Kafka broker versions, including older brokers that do not support
175           // idempotent producers or require explicit steps to enable them (e.g. adding the IDEMPOTENT_WRITE ACL to brokers older than 2.8).
176           // These settings might change when https://cwiki.apache.org/confluence/display/KAFKA/KIP-318%3A+Make+Kafka+Connect+Source+idempotent
177           // gets approved and scheduled for release.
178           producerProps.put(ProducerConfig.ENABLE_IDEMPOTENCE_CONFIG, "false"); // disable idempotence since retries is force to 0
179           ConnectUtils.addMetricsContextProperties(producerProps, config, clusterId);
180   
181           Map<String, Object> consumerProps = new HashMap<>(originals);
182           consumerProps.put(ConsumerConfig.KEY_DESERIALIZER_CLASS_CONFIG, StringDeserializer.class.getName());
183           consumerProps.put(ConsumerConfig.VALUE_DESERIALIZER_CLASS_CONFIG, ByteArrayDeserializer.class.getName());
184           ConnectUtils.addMetricsContextProperties(consumerProps, config, clusterId);
185   
186           Map<String, Object> adminProps = new HashMap<>(originals);
187           ConnectUtils.addMetricsContextProperties(adminProps, config, clusterId);
188           Supplier<TopicAdmin> adminSupplier;
189           if (topicAdminSupplier != null) {
190               adminSupplier = topicAdminSupplier;
191           } else {
192               // Create our own topic admin supplier that we'll close when we're stopped
193               ownTopicAdmin = new SharedTopicAdmin(adminProps);
194               adminSupplier = ownTopicAdmin;
195           }
196   
197           Map<String, Object> topicSettings = config instanceof DistributedConfig
198                                               ? ((DistributedConfig) config).statusStorageTopicSettings()
199                                               : Collections.emptyMap();
200           NewTopic topicDescription = TopicAdmin.defineTopic(statusTopic)
201                   .config(topicSettings) // first so that we override user-supplied settings as needed
202                   .compacted()
203                   .partitions(config.getInt(DistributedConfig.STATUS_STORAGE_PARTITIONS_CONFIG))
204                   .replicationFactor(config.getShort(DistributedConfig.STATUS_STORAGE_REPLICATION_FACTOR_CONFIG))
205                   .build();
206   
207           Callback<ConsumerRecord<String, byte[]>> readCallback = (error, record) -> read(record);
208           this.kafkaLog = createKafkaBasedLog(statusTopic, producerProps, consumerProps, readCallback, topicDescription, adminSupplier);
209       }

211       private KafkaBasedLog<String, byte[]> createKafkaBasedLog(String topic, Map<String, Object> producerProps,
212                                                                 Map<String, Object> consumerProps,
213                                                                 Callback<ConsumerRecord<String, byte[]>> consumedCallback,
214                                                                 final NewTopic topicDescription, Supplier<TopicAdmin> adminSupplier) {
215           java.util.function.Consumer<TopicAdmin> createTopics = admin -> {
216               log.debug("Creating admin client to manage Connect internal status topic");
217               // Create the topic if it doesn't exist
218               Set<String> newTopics = admin.createTopics(topicDescription);
219               if (!newTopics.contains(topic)) {
220                   // It already existed, so check that the topic cleanup policy is compact only and not delete
221                   log.debug("Using admin client to check cleanup policy of '{}' topic is '{}'", topic, TopicConfig.CLEANUP_POLICY_COMPACT);
222                   admin.verifyTopicCleanupPolicyOnlyCompact(topic,
223                           DistributedConfig.STATUS_STORAGE_TOPIC_CONFIG, "connector and task statuses");
224               }
225           };
226           return new KafkaBasedLog<>(topic, producerProps, consumerProps, adminSupplier, consumedCallback, time, createTopics);
227       }

292       private void sendTopicStatus(final String connector, final String topic, final TopicStatus status) {
293           String key = TOPIC_STATUS_PREFIX + topic + TOPIC_STATUS_SEPARATOR + connector;
294   
295           final byte[] value = serializeTopicStatus(status);
296   
297           kafkaLog.send(key, value, new org.apache.kafka.clients.producer.Callback() {
298               @Override
299               public void onCompletion(RecordMetadata metadata, Exception exception) {
300                   if (exception == null) return;
301                   // TODO: retry more gracefully and not forever
302                   if (exception instanceof RetriableException) {
303                       kafkaLog.send(key, value, this);
304                   } else {
305                       log.error("Failed to write status update", exception);
306                   }
307               }
308           });
309       }

311       private <V extends AbstractStatus<?>> void send(final String key,
312                                                    final V status,
313                                                    final CacheEntry<V> entry,
314                                                    final boolean safeWrite) {
315           final int sequence;
316           synchronized (this) {
317               this.generation = status.generation();
318               if (safeWrite && !entry.canWriteSafely(status))
319                   return;
320               sequence = entry.increment();
321           }
322   
323           final byte[] value = status.state() == ConnectorStatus.State.DESTROYED ? null : serialize(status);
324   
325           kafkaLog.send(key, value, new org.apache.kafka.clients.producer.Callback() {
326               @Override
327               public void onCompletion(RecordMetadata metadata, Exception exception) {
328                   if (exception == null) return;
329                   if (exception instanceof RetriableException) {
330                       synchronized (KafkaStatusBackingStore.this) {
331                           if (entry.isDeleted()
332                               || status.generation() != generation
333                               || (safeWrite && !entry.canWriteSafely(status, sequence)))
334                               return;
335                       }
336                       kafkaLog.send(key, value, this);
337                   } else {
338                       log.error("Failed to write status update", exception);
339                   }
340               }
341           });
342       }

362           }

532       private ConnectorTaskId parseConnectorTaskId(String key) {
533           String[] parts = key.split("-");
534           if (parts.length < 4) return null;
535   
536           try {
537               int taskNum = Integer.parseInt(parts[parts.length - 1]);
538               String connectorName = Utils.join(Arrays.copyOfRange(parts, 2, parts.length - 1), "-");
539               return new ConnectorTaskId(connectorName, taskNum);
540           } catch (NumberFormatException e) {
541               log.warn("Invalid task status key {}", key);
542               return null;
543           }
544       }

546       private void readConnectorStatus(String key, byte[] value) {
547           String connector = parseConnectorStatusKey(key);
548           if (connector.isEmpty()) {
549               log.warn("Discarding record with invalid connector status key {}", key);
550               return;
551           }
552   
553           if (value == null) {
554               log.trace("Removing status for connector {}", connector);
555               remove(connector);
556               return;
557           }
558   
559           ConnectorStatus status = parseConnectorStatus(connector, value);
560           if (status == null)
561               return;
562   
563           synchronized (this) {
564               log.trace("Received connector {} status update {}", connector, status);
565               CacheEntry<ConnectorStatus> entry = getOrAdd(connector);
566               entry.put(status);
567           }
568       }

574               return;
575           }

578               log.trace("Removing task status for {}", id);
579               remove(id);
580               return;
581           }
582   
583           TaskStatus status = parseTaskStatus(id, value);
584           if (status == null) {

587           }

596       private void readTopicStatus(String key, byte[] value) {
597           int delimiterPos = key.indexOf(':');
598           int beginPos = TOPIC_STATUS_PREFIX.length();
599           if (beginPos > delimiterPos) {
600               log.warn("Discarding record with invalid topic status key {}", key);
601               return;
602           }
603   
604           String topic = key.substring(beginPos, delimiterPos);
605           if (topic.isEmpty()) {
606               log.warn("Discarding record with invalid topic status key containing empty topic {}", key);
607               return;
608           }
609   
610           beginPos = delimiterPos + TOPIC_STATUS_SEPARATOR.length();
611           int endPos = key.length();
612           if (beginPos > endPos) {
613               log.warn("Discarding record with invalid topic status key {}", key);
614               return;
615           }
616   
617           String connector = key.substring(beginPos);
618           if (connector.isEmpty()) {
619               log.warn("Discarding record with invalid topic status key containing empty connector {}", key);
620               return;
621           }
622   
623           if (value == null) {
624               log.trace("Removing status for topic {} and connector {}", topic, connector);
625               removeTopic(topic, connector);
626               return;
627           }
628   
629           TopicStatus status = parseTopicStatus(value);
630           if (status == null) {
631               log.warn("Failed to parse topic status with key {}", key);
632               return;
633           }
634   
635           log.trace("Received topic status update {}", status);
636           topics.computeIfAbsent(connector, k -> new ConcurrentHashMap<>())
637                   .put(topic, status);
638       }

641       void read(ConsumerRecord<String, byte[]> record) {
642           String key = record.key();
643           if (key.startsWith(CONNECTOR_STATUS_PREFIX)) {
644               readConnectorStatus(key, record.value());
645           } else if (key.startsWith(TASK_STATUS_PREFIX)) {

647           } else if (key.startsWith(TOPIC_STATUS_PREFIX)) {
648               readTopicStatus(key, record.value());
649           } else {
650               log.warn("Discarding record with invalid key {}", key);
651           }
