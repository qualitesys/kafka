159       @Override
160       protected Optional<SubmittedRecords.SubmittedRecord> prepareToSendRecord(
161               SourceRecord sourceRecord,
162               ProducerRecord<byte[], byte[]> producerRecord
163       ) {
164           if (offsetBackingStore.primaryOffsetsTopic().equals(producerRecord.topic())) {
165               // This is to prevent deadlock that occurs when:
166               //     1. A task provides a record whose topic is the task's offsets topic
167               //     2. That record is dispatched to the task's producer in a transaction that remains open
168               //        at least until the worker polls the task again
169               //     3. In the subsequent call to SourceTask::poll, the task requests offsets from the worker
170               //        (which requires a read to the end of the offsets topic, and will block until any open
171               //        transactions on the topic are either committed or aborted)
172               throw new ConnectException("Source tasks may not produce to their own offsets topics when exactly-once support is enabled");
173           }
174           maybeBeginTransaction();
175           return Optional.empty();
176       }

221       @Override
222       protected void finalOffsetCommit(boolean failed) {

252       private void commitTransaction() {
253           log.debug("{} Committing offsets", this);
254   
255           long started = time.milliseconds();
256   
257           // We might have just aborted a transaction, in which case we'll have to begin a new one
258           // in order to commit offsets
259           maybeBeginTransaction();
260   
261           AtomicReference<Throwable> flushError = new AtomicReference<>();
262           if (offsetWriter.beginFlush()) {
263               // Now we can actually write the offsets to the internal topic.
264               // No need to track the flush future here since it's guaranteed to complete by the time
265               // Producer::commitTransaction completes
266               // We do have to track failures for that callback though, since they may originate from outside
267               // the producer (i.e., the offset writer or the backing offset store), and would not cause
268               // Producer::commitTransaction to fail
269               offsetWriter.doFlush((error, result) -> {
270                   if (error != null) {
271                       log.error("{} Failed to flush offsets to storage: ", ExactlyOnceWorkerSourceTask.this, error);
272                       flushError.compareAndSet(null, error);
273                   } else {
274                       log.trace("{} Finished flushing offsets to storage", ExactlyOnceWorkerSourceTask.this);
275                   }
276               });
277           }
278   
279           // Commit the transaction
280           // Blocks until all outstanding records have been sent and ack'd
281           try {
282               producer.commitTransaction();
283           } catch (Throwable t) {
284               log.error("{} Failed to commit producer transaction", ExactlyOnceWorkerSourceTask.this, t);
285               flushError.compareAndSet(null, t);
286           }
287   
288           transactionOpen = false;
289   
290           Throwable error = flushError.get();
291           if (error != null) {
292               recordCommitFailure(time.milliseconds() - started, null);
293               offsetWriter.cancelFlush();
294               throw maybeWrapProducerSendException(
295                       "Failed to flush offsets and/or records for task " + id,
296                       error
297               );
298           }
299   
300           transactionMetrics.commitTransaction();
301   
302           long durationMillis = time.milliseconds() - started;
303           recordCommitSuccess(durationMillis);
304           log.debug("{} Finished commitOffsets successfully in {} ms", this, durationMillis);
305   
306           // Synchronize in order to guarantee that writes on other threads are picked up by this one
307           synchronized (commitableRecords) {
308               commitableRecords.forEach(this::commitTaskRecord);
309               commitableRecords.clear();
310           }
311           commitSourceTask();
312       }

314       private RuntimeException maybeWrapProducerSendException(String message, Throwable error) {
315           if (isPossibleTransactionTimeoutError(error)) {
316               return wrapTransactionTimeoutError(error);
317           } else {
318               return new ConnectException(message, error);
319           }
320       }

322       private static boolean isPossibleTransactionTimeoutError(Throwable error) {
323           return error instanceof InvalidProducerEpochException
324               || error.getCause() instanceof InvalidProducerEpochException;
325       }

327       private ConnectException wrapTransactionTimeoutError(Throwable error) {
328           return new ConnectException(
329               "The task " + id + " was unable to finish writing records to Kafka before its producer transaction expired. "
330                   + "It may be necessary to reconfigure this connector in order for it to run healthily with exactly-once support. "
331                   + "Options for this include: tune the connector's producer configuration for higher throughput, "
332                   + "increase the transaction timeout for the connector's producers, "
333                   + "decrease the offset commit interval (if using interval-based transaction boundaries), "
334                   + "or use the 'poll' transaction boundary (if the connector is not already configured to use it).",
335               error
336           );
337       }

346       private abstract class TransactionBoundaryManager {
347           protected boolean shouldCommitTransactionForRecord(SourceRecord record) {
348               return false;
349           }
350   
351           protected boolean shouldCommitTransactionForBatch(long currentTimeMs) {
352               return false;
353           }
354   
355           protected boolean shouldCommitFinalTransaction() {
356               return false;
357           }
358   
359           /**
360            * Hook to signal that a new transaction cycle has been started. May be invoked
361            * multiple times if the task is paused and then resumed. It can be assumed that
362            * a new transaction is created at least every time an existing transaction is
363            * committed; this is just a hook to notify that a new transaction may have been
364            * created outside of that flow as well.
365            */
366           protected void initialize() {
367           }
368   
369           public void maybeCommitTransactionForRecord(SourceRecord record) {
370               maybeCommitTransaction(shouldCommitTransactionForRecord(record));
371           }
372   
373           public void maybeCommitTransactionForBatch() {
374               maybeCommitTransaction(shouldCommitTransactionForBatch(time.milliseconds()));
375           }
376   
377           public void maybeCommitFinalTransaction() {
378               maybeCommitTransaction(shouldCommitFinalTransaction());
379           }

381           private void maybeCommitTransaction(boolean shouldCommit) {
382               if (shouldCommit && (transactionOpen || offsetWriter.willFlush())) {
383                   try (LoggingContext loggingContext = LoggingContext.forOffsets(id)) {
384                       commitTransaction();
385                   }
386               }
387           }
388       }

390       private TransactionBoundaryManager buildTransactionManager(
391               WorkerConfig workerConfig,
392               SourceConnectorConfig sourceConfig,
393               WorkerTransactionContext transactionContext) {
394           TransactionBoundary boundary = sourceConfig.transactionBoundary();
395           switch (boundary) {
396               case POLL:
397                   return new TransactionBoundaryManager() {
398                       @Override
399                       protected boolean shouldCommitTransactionForBatch(long currentTimeMs) {
400                           return true;
401                       }
402   
403                       @Override
404                       protected boolean shouldCommitFinalTransaction() {
405                           return true;
406                       }
407                   };
408   
409               case INTERVAL:
410                   long transactionBoundaryInterval = Optional.ofNullable(sourceConfig.transactionBoundaryInterval())
411                           .orElse(workerConfig.offsetCommitInterval());
412                   return new TransactionBoundaryManager() {
413                       private final long commitInterval = transactionBoundaryInterval;
414                       private long lastCommit;
415   
416                       @Override
417                       public void initialize() {
418                           this.lastCommit = time.milliseconds();
419                       }
420   
421                       @Override
422                       protected boolean shouldCommitTransactionForBatch(long currentTimeMs) {
423                           if (time.milliseconds() >= lastCommit + commitInterval) {
424                               lastCommit = time.milliseconds();
425                               return true;
426                           } else {
427                               return false;
428                           }
429                       }
430   
431                       @Override
432                       protected  boolean shouldCommitFinalTransaction() {
433                           return true;
434                       }
435                   };
436   
437               case CONNECTOR:
438                   Objects.requireNonNull(transactionContext, "Transaction context must be provided when using connector-defined transaction boundaries");
439                   return new TransactionBoundaryManager() {
440                       @Override
441                       protected boolean shouldCommitFinalTransaction() {
442                           return shouldCommitTransactionForBatch(time.milliseconds());
443                       }
444   
445                       @Override
446                       protected boolean shouldCommitTransactionForBatch(long currentTimeMs) {
447                           if (transactionContext.shouldAbortBatch()) {
448                               log.info("Aborting transaction for batch as requested by connector");
449                               abortTransaction();
450                               // We abort the transaction, which causes all the records up to this point to be dropped, but we still want to
451                               // commit offsets so that the task doesn't see the same records all over again
452                               return true;
453                           }
454                           return transactionContext.shouldCommitBatch();
455                       }
456   
457                       @Override
458                       protected boolean shouldCommitTransactionForRecord(SourceRecord record) {
459                           if (transactionContext.shouldAbortOn(record)) {
460                               log.info("Aborting transaction for record on topic {} as requested by connector", record.topic());
461                               log.trace("Last record in aborted transaction: {}", record);
462                               abortTransaction();
463                               // We abort the transaction, which causes all the records up to this point to be dropped, but we still want to
464                               // commit offsets so that the task doesn't see the same records all over again
465                               return true;
466                           }
467                           return transactionContext.shouldCommitOn(record);
468                       }
469   
470                       private void abortTransaction() {
471                           producer.abortTransaction();
472                           transactionMetrics.abortTransaction();
473                           transactionOpen = false;
474                       }
475                   };
476               default:
477                   throw new IllegalArgumentException("Unrecognized transaction boundary: " + boundary);
478           }
479       }
