61       public static final ConfigDef.Validator REPLICATION_FACTOR_VALIDATOR = ConfigDef.LambdaValidator.with(
62           (name, value) -> validateReplicationFactor(name, (short) value),
63           () -> "Positive number not larger than the number of brokers in the Kafka cluster, or -1 to use the broker's default"
64       );
65       public static final ConfigDef.Validator PARTITIONS_VALIDATOR = ConfigDef.LambdaValidator.with(
66           (name, value) -> validatePartitions(name, (int) value),
67           () -> "Positive number, or -1 to use the broker's default"
68       );
69       @SuppressWarnings("unchecked")
70       public static final ConfigDef.Validator REGEX_VALIDATOR = ConfigDef.LambdaValidator.with(
71           (name, value) -> {
72               try {
73                   ((List<String>) value).forEach(Pattern::compile);
74               } catch (PatternSyntaxException e) {
75                   throw new ConfigException(name, value,
76                           "Syntax error in regular expression: " + e.getMessage());
77               }
78           },
79           () -> "Positive number, or -1 to use the broker's default"
80       );
81   
82       private static void validatePartitions(String configName, int factor) {
83           if (factor != TopicAdmin.NO_PARTITIONS && factor < 1) {
84               throw new ConfigException(configName, factor,
85                       "Number of partitions must be positive, or -1 to use the broker's default");
86           }
87       }
88   
89       private static void validateReplicationFactor(String configName, short factor) {
90           if (factor != TopicAdmin.NO_REPLICATION_FACTOR && factor < 1) {
91               throw new ConfigException(configName, factor,
92                       "Replication factor must be positive and not larger than the number of brokers in the Kafka cluster, or -1 to use the broker's default");
93           }
94       }

96       public static ConfigDef configDef(String group, short defaultReplicationFactor, int defaultParitionCount) {
97           int orderInGroup = 0;
98           ConfigDef configDef = new ConfigDef();
99           configDef
100                   .define(INCLUDE_REGEX_CONFIG, ConfigDef.Type.LIST, Collections.emptyList(),
101                           REGEX_VALIDATOR, ConfigDef.Importance.LOW,
102                           INCLUDE_REGEX_DOC, group, ++orderInGroup, ConfigDef.Width.LONG,
103                           "Inclusion Topic Pattern for " + group)
104                   .define(EXCLUDE_REGEX_CONFIG, ConfigDef.Type.LIST, Collections.emptyList(),
105                           REGEX_VALIDATOR, ConfigDef.Importance.LOW,
106                           EXCLUDE_REGEX_DOC, group, ++orderInGroup, ConfigDef.Width.LONG,
107                           "Exclusion Topic Pattern for " + group)
108                   .define(REPLICATION_FACTOR_CONFIG, ConfigDef.Type.SHORT,
109                           defaultReplicationFactor, REPLICATION_FACTOR_VALIDATOR,
110                           ConfigDef.Importance.LOW, REPLICATION_FACTOR_DOC, group, ++orderInGroup,
111                           ConfigDef.Width.LONG, "Replication Factor for Topics in " + group)
112                   .define(PARTITIONS_CONFIG, ConfigDef.Type.INT,
113                           defaultParitionCount, PARTITIONS_VALIDATOR,
114                           ConfigDef.Importance.LOW, PARTITIONS_DOC, group, ++orderInGroup,
115                           ConfigDef.Width.LONG, "Partition Count for Topics in " + group);
116           return configDef;
117       }
