17   package org.apache.kafka.connect.runtime;
18   
19   import org.apache.kafka.connect.runtime.AbstractStatus.State;
20   
21   import java.util.concurrent.atomic.AtomicReference;
22   
23   /**
24    * Utility class that tracks the current state and the duration of time spent in each state.
25    * This class is threadsafe.
26    */
27   public class StateTracker {
28   
29       private final AtomicReference<StateChange> lastState = new AtomicReference<>(new StateChange());
30   
31       /**
32        * Change the current state.
33        * <p>
34        * This method is synchronized to ensure that all state changes are captured correctly and in the same order.
35        * Synchronization is acceptable since it is assumed that state changes will be relatively infrequent.
36        *
37        * @param newState the current state; may not be null
38        * @param now      the current time in milliseconds
39        */
40       public synchronized void changeState(State newState, long now) {
41           // JDK8: remove synchronization by using lastState.getAndUpdate(oldState->oldState.newState(newState, now));
42           lastState.set(lastState.get().newState(newState, now));
43       }
44   
45       /**
46        * Calculate the ratio of time spent in the specified state.
47        *
48        * @param ratioState the state for which the ratio is to be calculated; may not be null
49        * @param now        the current time in milliseconds
50        * @return the ratio of time spent in the specified state to the time spent in all states
51        */
52       public double durationRatio(State ratioState, long now) {
53           return lastState.get().durationRatio(ratioState, now);
54       }
55   
56       /**
57        * Get the current state.
58        *
59        * @return the current state; may be null if no state change has been recorded
60        */
61       public State currentState() {
62           return lastState.get().state;
63       }
64   
65       /**
66        * An immutable record of the accumulated times at the most recent state change. This class is required to
67        * efficiently make {@link StateTracker} threadsafe.
68        */
69       private static final class StateChange {
70   
71           private final State state;
72           private final long startTime;
73           private final long unassignedTotalTimeMs;
74           private final long runningTotalTimeMs;
75           private final long pausedTotalTimeMs;
76           private final long failedTotalTimeMs;
77           private final long destroyedTotalTimeMs;
78           private final long restartingTotalTimeMs;
79   
80           /**
81            * The initial StateChange instance before any state has changed.
82            */
83           StateChange() {
84               this(null, 0L, 0L, 0L, 0L, 0L, 0L, 0L);
85           }
86   
87           StateChange(State state, long startTime, long unassignedTotalTimeMs, long runningTotalTimeMs,
88                               long pausedTotalTimeMs, long failedTotalTimeMs, long destroyedTotalTimeMs, long restartingTotalTimeMs) {
89               this.state = state;
90               this.startTime = startTime;
91               this.unassignedTotalTimeMs = unassignedTotalTimeMs;
92               this.runningTotalTimeMs = runningTotalTimeMs;
93               this.pausedTotalTimeMs = pausedTotalTimeMs;
94               this.failedTotalTimeMs = failedTotalTimeMs;
95               this.destroyedTotalTimeMs = destroyedTotalTimeMs;
96               this.restartingTotalTimeMs = restartingTotalTimeMs;
97           }
98   
99           /**
100            * Return a new StateChange that includes the accumulated times of this state plus the time spent in the
101            * current state.
102            *
103            * @param state the new state; may not be null
104            * @param now   the time at which the state transition occurs.
105            * @return the new StateChange, though may be this instance of the state did not actually change; never null
106            */
107           public StateChange newState(State state, long now) {
108               if (this.state == null) {
109                   return new StateChange(state, now, 0L, 0L, 0L, 0L, 0L, 0L);
110               }
111               if (state == this.state) {
112                   return this;
113               }
114               long unassignedTime = this.unassignedTotalTimeMs;
115               long runningTime = this.runningTotalTimeMs;
116               long pausedTime = this.pausedTotalTimeMs;
117               long failedTime = this.failedTotalTimeMs;
118               long destroyedTime = this.destroyedTotalTimeMs;
119               long restartingTime = this.restartingTotalTimeMs;
120               long duration = now - startTime;
121               switch (this.state) {
122                   case UNASSIGNED:
123                       unassignedTime += duration;
124                       break;
125                   case RUNNING:
126                       runningTime += duration;
127                       break;
128                   case PAUSED:
129                       pausedTime += duration;
130                       break;
131                   case FAILED:
132                       failedTime += duration;
133                       break;
134                   case DESTROYED:
135                       destroyedTime += duration;
136                       break;
137                   case RESTARTING:
138                       restartingTime += duration;
139                       break;
140               }
141               return new StateChange(state, now, unassignedTime, runningTime, pausedTime, failedTime, destroyedTime, restartingTime);
142           }
143   
144           /**
145            * Calculate the ratio of time spent in the specified state.
146            *
147            * @param ratioState the state for which the ratio is to be calculated; may not be null
148            * @param now        the current time in milliseconds
149            * @return the ratio of time spent in the specified state to the time spent in all states
150            */
151           public double durationRatio(State ratioState, long now) {
152               if (state == null) {
153                   return 0.0d;
154               }
155               long durationCurrent = now - startTime; // since last state change
156               long durationDesired = ratioState == state ? durationCurrent : 0L;
157               switch (ratioState) {
158                   case UNASSIGNED:
159                       durationDesired += unassignedTotalTimeMs;
160                       break;
161                   case RUNNING:
162                       durationDesired += runningTotalTimeMs;
163                       break;
164                   case PAUSED:
165                       durationDesired += pausedTotalTimeMs;
166                       break;
167                   case FAILED:
168                       durationDesired += failedTotalTimeMs;
169                       break;
170                   case DESTROYED:
171                       durationDesired += destroyedTotalTimeMs;
172                       break;
173                   case RESTARTING:
174                       durationDesired += restartingTotalTimeMs;
175                       break;
176               }
177               long total = durationCurrent + unassignedTotalTimeMs + runningTotalTimeMs + pausedTotalTimeMs +
178                                    failedTotalTimeMs + destroyedTotalTimeMs + restartingTotalTimeMs;
179               return total == 0.0d ? 0.0d : (double) durationDesired / total;
180           }
181       }
182   }
