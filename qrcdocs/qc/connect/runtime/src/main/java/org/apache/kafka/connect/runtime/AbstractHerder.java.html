80   /**
81    * Abstract Herder implementation which handles connector/task lifecycle tracking. Extensions
82    * must invoke the lifecycle hooks appropriately.
83    *
84    * This class takes the following approach for sending status updates to the backing store:
85    *
86    * 1) When the connector or task is starting, we overwrite the previous state blindly. This ensures that
87    *    every rebalance will reset the state of tasks to the proper state. The intuition is that there should
88    *    be less chance of write conflicts when the worker has just received its assignment and is starting tasks.
89    *    In particular, this prevents us from depending on the generation absolutely. If the group disappears
90    *    and the generation is reset, then we'll overwrite the status information with the older (and larger)
91    *    generation with the updated one. The danger of this approach is that slow starting tasks may cause the
92    *    status to be overwritten after a rebalance has completed.
93    *
94    * 2) If the connector or task fails or is shutdown, we use {@link StatusBackingStore#putSafe(ConnectorStatus)},
95    *    which provides a little more protection if the worker is no longer in the group (in which case the
96    *    task may have already been started on another worker). Obviously this is still racy. If the task has just
97    *    started on another worker, we may not have the updated status cached yet. In this case, we'll overwrite
98    *    the value which will cause the state to be inconsistent (most likely until the next rebalance). Until
99    *    we have proper producer groups with fenced groups, there is not much else we can do.
100    */
101   public abstract class AbstractHerder implements Herder, TaskStatus.Listener, ConnectorStatus.Listener {

380       /**
381        * Build the {@link RestartPlan} that describes what should and should not be restarted given the restart request
382        * and the current status of the connector and task instances.
383        *
384        * @param request the restart request; may not be null
385        * @return the restart plan, or empty if this worker has no status for the connector named in the request and therefore the
386        *         connector cannot be restarted
387        */
388       public Optional<RestartPlan> buildRestartPlan(RestartRequest request) {
389           String connectorName = request.connectorName();
390           ConnectorStatus connectorStatus = statusBackingStore.get(connectorName);
391           if (connectorStatus == null) {
392               return Optional.empty();
393           }
394   
395           // If requested, mark the connector as restarting
396           AbstractStatus.State connectorState = request.shouldRestartConnector(connectorStatus) ? AbstractStatus.State.RESTARTING : connectorStatus.state();
397           ConnectorStateInfo.ConnectorState connectorInfoState = new ConnectorStateInfo.ConnectorState(
398                   connectorState.toString(),
399                   connectorStatus.workerId(),
400                   connectorStatus.trace()
401           );
402   
403           // Collect the task states, If requested, mark the task as restarting
404           List<ConnectorStateInfo.TaskState> taskStates = statusBackingStore.getAll(connectorName)
405                   .stream()
406                   .map(taskStatus -> {
407                       AbstractStatus.State taskState = request.shouldRestartTask(taskStatus) ? AbstractStatus.State.RESTARTING : taskStatus.state();
408                       return new ConnectorStateInfo.TaskState(
409                               taskStatus.id().task(),
410                               taskState.toString(),
411                               taskStatus.workerId(),
412                               taskStatus.trace()
413                       );
414                   })
415                   .collect(Collectors.toList());
416           // Construct the response from the various states
417           Map<String, String> conf = rawConfig(connectorName);
418           ConnectorStateInfo stateInfo = new ConnectorStateInfo(
419                   connectorName,
420                   connectorInfoState,
421                   taskStates,
422                   conf == null ? ConnectorType.UNKNOWN : connectorTypeForClass(conf.get(ConnectorConfig.CONNECTOR_CLASS_CONFIG))
423           );
424           return Optional.of(new RestartPlan(request, stateInfo));
425       }

444       ConfigInfos validateConnectorConfig(Map<String, String> connectorProps, boolean doLog) {
445           if (worker.configTransformer() != null) {
446               connectorProps = worker.configTransformer().transform(connectorProps);
447           }
448           String connType = connectorProps.get(ConnectorConfig.CONNECTOR_CLASS_CONFIG);
449           if (connType == null)
450               throw new BadRequestException("Connector config " + connectorProps + " contains no connector type");
451   
452           Connector connector = getConnector(connType);
453           org.apache.kafka.connect.health.ConnectorType connectorType;
454           ClassLoader savedLoader = plugins().compareAndSwapLoaders(connector);
455           ConfigDef enrichedConfigDef;
456           Map<String, ConfigValue> validatedConnectorConfig;
457           try {
458               if (connector instanceof SourceConnector) {
459                   connectorType = org.apache.kafka.connect.health.ConnectorType.SOURCE;
460                   enrichedConfigDef = ConnectorConfig.enrich(plugins(), SourceConnectorConfig.configDef(), connectorProps, false);
461                   validatedConnectorConfig = validateSourceConnectorConfig((SourceConnector) connector, enrichedConfigDef, connectorProps);
462               } else {
463                   SinkConnectorConfig.validate(connectorProps);
464                   connectorType = org.apache.kafka.connect.health.ConnectorType.SINK;
465                   enrichedConfigDef = ConnectorConfig.enrich(plugins(), SinkConnectorConfig.configDef(), connectorProps, false);
466                   validatedConnectorConfig = validateSinkConnectorConfig((SinkConnector) connector, enrichedConfigDef, connectorProps);
467               }
468   
469               connectorProps.entrySet().stream()
470                   .filter(e -> e.getValue() == null)
471                   .map(Map.Entry::getKey)
472                   .forEach(prop ->
473                       validatedConnectorConfig.computeIfAbsent(prop, ConfigValue::new)
474                           .addErrorMessage("Null value can not be supplied as the configuration value.")

476   
477               List<ConfigValue> configValues = new ArrayList<>(validatedConnectorConfig.values());
478               Map<String, ConfigKey> configKeys = new LinkedHashMap<>(enrichedConfigDef.configKeys());
479               Set<String> allGroups = new LinkedHashSet<>(enrichedConfigDef.groups());
480   
481               // do custom connector-specific validation
482               ConfigDef configDef = connector.config();
483               if (null == configDef) {
484                   throw new BadRequestException(
485                           String.format(
486                                   "%s.config() must return a ConfigDef that is not null.",
487                                   connector.getClass().getName()
488                           )
489                   );
490               }
491               Config config = connector.validate(connectorProps);
492               if (null == config) {
493                   throw new BadRequestException(
494                           String.format(
495                                   "%s.validate() must return a Config that is not null.",
496                                   connector.getClass().getName()
497                           )
498                   );
499               }
500               configKeys.putAll(configDef.configKeys());
501               allGroups.addAll(configDef.groups());
502               configValues.addAll(config.configValues());
503               ConfigInfos configInfos =  generateResult(connType, configKeys, configValues, new ArrayList<>(allGroups));
504   
505               AbstractConfig connectorConfig = new AbstractConfig(new ConfigDef(), connectorProps, doLog);
506               String connName = connectorProps.get(ConnectorConfig.NAME_CONFIG);
507               ConfigInfos producerConfigInfos = null;
508               ConfigInfos consumerConfigInfos = null;
509               ConfigInfos adminConfigInfos = null;
510   
511               if (connectorUsesProducer(connectorType, connectorProps)) {
512                   producerConfigInfos = validateClientOverrides(
513                       connName,
514                       ConnectorConfig.CONNECTOR_CLIENT_PRODUCER_OVERRIDES_PREFIX,
515                       connectorConfig,
516                       ProducerConfig.configDef(),
517                       connector.getClass(),
518                       connectorType,
519                       ConnectorClientConfigRequest.ClientType.PRODUCER,
520                       connectorClientConfigOverridePolicy);
521               }
522               if (connectorUsesAdmin(connectorType, connectorProps)) {
523                   adminConfigInfos = validateClientOverrides(
524                       connName,
525                       ConnectorConfig.CONNECTOR_CLIENT_ADMIN_OVERRIDES_PREFIX,
526                       connectorConfig,
527                       AdminClientConfig.configDef(),
528                       connector.getClass(),
529                       connectorType,
530                       ConnectorClientConfigRequest.ClientType.ADMIN,
531                       connectorClientConfigOverridePolicy);
532               }
533               if (connectorUsesConsumer(connectorType, connectorProps)) {
534                   consumerConfigInfos = validateClientOverrides(
535                       connName,
536                       ConnectorConfig.CONNECTOR_CLIENT_CONSUMER_OVERRIDES_PREFIX,
537                       connectorConfig,
538                       ConsumerConfig.configDef(),
539                       connector.getClass(),
540                       connectorType,
541                       ConnectorClientConfigRequest.ClientType.CONSUMER,
542                       connectorClientConfigOverridePolicy);
543               }
544               return mergeConfigInfos(connType, configInfos, producerConfigInfos, consumerConfigInfos, adminConfigInfos);
545           } finally {
546               Plugins.compareAndSwapLoaders(savedLoader);
547           }
548       }

551           int errorCount = 0;

555               if (configInfos != null) {
556                   errorCount += configInfos.errorCount();
557                   configInfoList.addAll(configInfos.values());
558                   groups.addAll(configInfos.groups());

564       private static ConfigInfos validateClientOverrides(String connName,
565                                                         String prefix,
566                                                         AbstractConfig connectorConfig,
567                                                         ConfigDef configDef,
568                                                         Class<? extends Connector> connectorClass,
569                                                         org.apache.kafka.connect.health.ConnectorType connectorType,
570                                                         ConnectorClientConfigRequest.ClientType clientType,
571                                                         ConnectorClientConfigOverridePolicy connectorClientConfigOverridePolicy) {
572           int errorCount = 0;
573           List<ConfigInfo> configInfoList = new LinkedList<>();
574           Map<String, ConfigKey> configKeys = configDef.configKeys();
575           Set<String> groups = new LinkedHashSet<>();
576           Map<String, Object> clientConfigs = new HashMap<>();
577           for (Map.Entry<String, Object> rawClientConfig : connectorConfig.originalsWithPrefix(prefix).entrySet()) {
578               String configName = rawClientConfig.getKey();
579               Object rawConfigValue = rawClientConfig.getValue();
580               ConfigKey configKey = configDef.configKeys().get(configName);
581               Object parsedConfigValue = configKey != null
582                   ? ConfigDef.parseType(configName, rawConfigValue, configKey.type)
583                   : rawConfigValue;
584               clientConfigs.put(configName, parsedConfigValue);
585           }
586           ConnectorClientConfigRequest connectorClientConfigRequest = new ConnectorClientConfigRequest(
587               connName, connectorType, connectorClass, clientConfigs, clientType);
588           List<ConfigValue> configValues = connectorClientConfigOverridePolicy.validate(connectorClientConfigRequest);
589           if (configValues != null) {
590               for (ConfigValue validatedConfigValue : configValues) {
591                   ConfigKey configKey = configKeys.get(validatedConfigValue.name());
592                   ConfigKeyInfo configKeyInfo = null;
593                   if (configKey != null) {
594                       if (configKey.group != null) {
595                           groups.add(configKey.group);
596                       }
597                       configKeyInfo = convertConfigKey(configKey, prefix);
598                   }
599   
600                   ConfigValue configValue = new ConfigValue(prefix + validatedConfigValue.name(), validatedConfigValue.value(),
601                                                             validatedConfigValue.recommendedValues(), validatedConfigValue.errorMessages());
602                   if (configValue.errorMessages().size() > 0) {
603                       errorCount++;
604                   }
605                   ConfigValueInfo configValueInfo = convertConfigValue(configValue, configKey != null ? configKey.type : null);
606                   configInfoList.add(new ConfigInfo(configKeyInfo, configValueInfo));
607               }
608           }
609           return new ConfigInfos(connectorClass.toString(), errorCount, new ArrayList<>(groups), configInfoList);
610       }

613       public static ConfigInfos generateResult(String connType, Map<String, ConfigKey> configKeys, List<ConfigValue> configValues, List<String> groups) {
614           int errorCount = 0;
615           List<ConfigInfo> configInfoList = new LinkedList<>();
616   
617           Map<String, ConfigValue> configValueMap = new HashMap<>();
618           for (ConfigValue configValue: configValues) {
619               String configName = configValue.name();
620               configValueMap.put(configName, configValue);
621               if (!configKeys.containsKey(configName)) {
622                   configInfoList.add(new ConfigInfo(null, convertConfigValue(configValue, null)));
623                   errorCount += configValue.errorMessages().size();
624               }
625           }
626   
627           for (Map.Entry<String, ConfigKey> entry : configKeys.entrySet()) {
628               String configName = entry.getKey();
629               ConfigKeyInfo configKeyInfo = convertConfigKey(entry.getValue());
630               Type type = entry.getValue().type;
631               ConfigValueInfo configValueInfo = null;
632               if (configValueMap.containsKey(configName)) {
633                   ConfigValue configValue = configValueMap.get(configName);
634                   configValueInfo = convertConfigValue(configValue, type);
635                   errorCount += configValue.errorMessages().size();
636               }
637               configInfoList.add(new ConfigInfo(configKeyInfo, configValueInfo));
638           }
639           return new ConfigInfos(connType, errorCount, groups, configInfoList);
640       }

751        * Performs a reverse transformation on a set of task configs, by replacing values with variable references.
752        */
753       public static List<Map<String, String>> reverseTransform(String connName,
754                                                                ClusterConfigState configState,
755                                                                List<Map<String, String>> configs) {
756   
757           // Find the config keys in the raw connector config that have variable references
758           Map<String, String> rawConnConfig = configState.rawConnectorConfig(connName);
759           Set<String> connKeysWithVariableValues = keysWithVariableValues(rawConnConfig, ConfigTransformer.DEFAULT_PATTERN);
760   
761           List<Map<String, String>> result = new ArrayList<>();
762           for (Map<String, String> config : configs) {
763               Map<String, String> newConfig = new HashMap<>(config);
764               for (String key : connKeysWithVariableValues) {
765                   if (newConfig.containsKey(key)) {
766                       newConfig.put(key, rawConnConfig.get(key));
767                   }
768               }
769               result.add(newConfig);
770           }
771           return result;
772       }
