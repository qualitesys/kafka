17   package org.apache.kafka.connect.runtime;
18   
19   import java.util.Objects;
20   
21   import org.apache.kafka.connect.connector.Connector;
22   import org.apache.kafka.connect.connector.Task;
23   
24   /**
25    * A request to restart a connector and/or task instances.
26    * <p>The natural order is based first upon the connector name and then requested restart behaviors. 
27    * If two requests have the same connector name, then the requests are ordered based on the 
28    * probable number of tasks/connector this request is going to restart.
29    */
30   public class RestartRequest implements Comparable<RestartRequest> {
31   
32       private final String connectorName;
33       private final boolean onlyFailed;
34       private final boolean includeTasks;
35   
36       /**
37        * Create a new request to restart a connector and optionally its tasks.
38        *
39        * @param connectorName the name of the connector; may not be null
40        * @param onlyFailed    true if only failed instances should be restarted
41        * @param includeTasks  true if tasks should be restarted, or false if only the connector should be restarted
42        */
43       public RestartRequest(String connectorName, boolean onlyFailed, boolean includeTasks) {
44           this.connectorName = Objects.requireNonNull(connectorName, "Connector name may not be null");
45           this.onlyFailed = onlyFailed;
46           this.includeTasks = includeTasks;
47       }
48   
49       /**
50        * Get the name of the connector.
51        *
52        * @return the connector name; never null
53        */
54       public String connectorName() {
55           return connectorName;
56       }
57   
58       /**
59        * Determine whether only failed instances be restarted.
60        *
61        * @return true if only failed instances should be restarted, or false if all applicable instances should be restarted
62        */
63       public boolean onlyFailed() {
64           return onlyFailed;
65       }
66   
67       /**
68        * Determine whether {@link Task} instances should also be restarted in addition to the {@link Connector} instance.
69        *
70        * @return true if the connector and task instances should be restarted, or false if just the connector should be restarted
71        */
72       public boolean includeTasks() {
73           return includeTasks;
74       }
75   
76       /**
77        * Determine whether the connector with the given status is to be restarted.
78        *
79        * @param status the connector status; may not be null
80        * @return true if the connector is to be restarted, or false otherwise
81        */
82       public boolean shouldRestartConnector(ConnectorStatus status) {
83           return !onlyFailed || status.state() == AbstractStatus.State.FAILED;
84       }
85   
86       /**
87        * Determine whether only the {@link Connector} instance is to be restarted even if not failed.
88        *
89        * @return true if only the {@link Connector} instance is to be restarted even if not failed, or false otherwise
90        */
91       public boolean forceRestartConnectorOnly() {
92           return !onlyFailed() && !includeTasks();
93       }
94   
95       /**
96        * Determine whether the task instance with the given status is to be restarted.
97        *
98        * @param status the task status; may not be null
99        * @return true if the task is to be restarted, or false otherwise
100        */
101       public boolean shouldRestartTask(TaskStatus status) {
102           return includeTasks && (!onlyFailed || status.state() == AbstractStatus.State.FAILED);
103       }
104   
105       @Override
106       public int compareTo(RestartRequest o) {
107           int result = connectorName.compareTo(o.connectorName);
108           return result == 0 ? impactRank() - o.impactRank() : result;
109       }
110       //calculates an internal rank for the restart request based on the probable number of tasks/connector this request is going to restart
111       private int impactRank() {
112           if (onlyFailed && !includeTasks) { //restarts only failed connector so least impactful
113               return 0;
114           } else if (onlyFailed && includeTasks) { //restarts only failed connector and tasks
115               return 1;
116           } else if (!onlyFailed && !includeTasks) { //restart connector in any state but no tasks
117               return 2;
118           }
119           //onlyFailed==false&&includeTasks  restarts both connector and tasks in any state so highest impact
120           return 3;
121       }
122   
123       @Override
124       public boolean equals(Object o) {
125           if (this == o) {
126               return true;
127           }
128           if (o == null || getClass() != o.getClass()) {
129               return false;
130           }
131           RestartRequest that = (RestartRequest) o;
132           return onlyFailed == that.onlyFailed && includeTasks == that.includeTasks && Objects.equals(connectorName, that.connectorName);
133       }
134   
135       @Override
136       public int hashCode() {
137           return Objects.hash(connectorName, onlyFailed, includeTasks);
138       }
139   
140       @Override
141       public String toString() {
142           return "restart request for {" + "connectorName='" + connectorName + "', onlyFailed=" + onlyFailed + ", includeTasks=" + includeTasks + '}';
143       }
144   }
