212       /**
213        * Returns an enriched {@link ConfigDef} building upon the {@code ConfigDef}, using the current configuration specified in {@code props} as an input.
214        *
215        * @param baseConfigDef the base configuration definition to be enriched
216        * @param props the non parsed configuration properties
217        * @return the enriched configuration definition
218        */
219       public static ConfigDef enrich(ConfigDef baseConfigDef, Map<String, String> props, AbstractConfig defaultGroupConfig) {
220           List<Object> topicCreationGroups = new ArrayList<>();
221           Object aliases = ConfigDef.parseType(TOPIC_CREATION_GROUPS_CONFIG, props.get(TOPIC_CREATION_GROUPS_CONFIG), ConfigDef.Type.LIST);
222           if (aliases instanceof List) {
223               topicCreationGroups.addAll((List<?>) aliases);
224           }
225   
226           ConfigDef newDef = new ConfigDef(baseConfigDef);
227           String defaultGroupPrefix = TOPIC_CREATION_PREFIX + DEFAULT_TOPIC_CREATION_GROUP + ".";
228           short defaultGroupReplicationFactor = defaultGroupConfig.getShort(defaultGroupPrefix + REPLICATION_FACTOR_CONFIG);
229           int defaultGroupPartitions = defaultGroupConfig.getInt(defaultGroupPrefix + PARTITIONS_CONFIG);
230           topicCreationGroups.stream().distinct().forEach(group -> {
231               if (!(group instanceof String)) {
232                   throw new ConfigException("Item in " + TOPIC_CREATION_GROUPS_CONFIG + " property is not of type String");
233               }
234               String alias = (String) group;
235               String prefix = TOPIC_CREATION_PREFIX + alias + ".";
236               String configGroup = TOPIC_CREATION_GROUP + ": " + alias;
237               newDef.embed(prefix, configGroup, 0,
238                       TopicCreationConfig.configDef(configGroup, defaultGroupReplicationFactor, defaultGroupPartitions));
239           });
240           return newDef;
241       }

243       public SourceConnectorConfig(Plugins plugins, Map<String, String> props, boolean createTopics) {
244           super(plugins, configDef(), props);
245           if (createTopics && props.entrySet().stream().anyMatch(e -> e.getKey().startsWith(TOPIC_CREATION_PREFIX))) {
246               ConfigDef defaultConfigDef = embedDefaultGroup(configDef());
247               // This config is only used to set default values for partitions and replication
248               // factor from the default group and otherwise it remains unused
249               AbstractConfig defaultGroup = new AbstractConfig(defaultConfigDef, props, false);
250   
251               // If the user has added regex of include or exclude patterns in the default group,
252               // they should be ignored.
253               Map<String, String> propsWithoutRegexForDefaultGroup = new HashMap<>(props);
254               propsWithoutRegexForDefaultGroup.entrySet()
255                       .removeIf(e -> e.getKey().equals(DEFAULT_TOPIC_CREATION_PREFIX + INCLUDE_REGEX_CONFIG)
256                               || e.getKey().equals(DEFAULT_TOPIC_CREATION_PREFIX + EXCLUDE_REGEX_CONFIG));
257               enrichedSourceConfig = new EnrichedSourceConnectorConfig(plugins,
258                       enrich(defaultConfigDef, props, defaultGroup),
259                       propsWithoutRegexForDefaultGroup);
260           } else {
261               enrichedSourceConfig = null;
262           }
263           transactionBoundary = TransactionBoundary.fromProperty(getString(TRANSACTION_BOUNDARY_CONFIG));
264           transactionBoundaryInterval = getLong(TRANSACTION_BOUNDARY_INTERVAL_CONFIG);
265           offsetsTopic = getString(OFFSETS_TOPIC_CONFIG);
266       }

314       public Map<String, Object> topicCreationOtherConfigs(String group) {
315           if (enrichedSourceConfig == null) {
316               return Collections.emptyMap();
317           }
318           return enrichedSourceConfig.originalsWithPrefix(TOPIC_CREATION_PREFIX + group + '.').entrySet().stream()
319                   .filter(e -> {
320                       String key = e.getKey();
321                       return !(INCLUDE_REGEX_CONFIG.equals(key) || EXCLUDE_REGEX_CONFIG.equals(key)
322                               || REPLICATION_FACTOR_CONFIG.equals(key) || PARTITIONS_CONFIG.equals(key));
323                   })
324                   .collect(Collectors.toMap(Map.Entry::getKey, Map.Entry::getValue));
325       }
