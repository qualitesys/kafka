17   package org.apache.kafka.connect.file;
18   
19   import org.apache.kafka.clients.consumer.OffsetAndMetadata;
20   import org.apache.kafka.common.TopicPartition;
21   import org.apache.kafka.connect.errors.ConnectException;
22   import org.apache.kafka.connect.sink.SinkRecord;
23   import org.apache.kafka.connect.sink.SinkTask;
24   import org.slf4j.Logger;
25   import org.slf4j.LoggerFactory;
26   
27   import java.io.IOException;
28   import java.io.PrintStream;
29   import java.nio.charset.StandardCharsets;
30   import java.nio.file.Files;
31   import java.nio.file.Paths;
32   import java.nio.file.StandardOpenOption;
33   import java.util.Collection;
34   import java.util.Map;
35   
36   /**
37    * FileStreamSinkTask writes records to stdout or a file.
38    */
39   public class FileStreamSinkTask extends SinkTask {
40       private static final Logger log = LoggerFactory.getLogger(FileStreamSinkTask.class);
41   
42       private String filename;
43       private PrintStream outputStream;
44   
45       public FileStreamSinkTask() {
46       }
47   
48       // for testing
49       public FileStreamSinkTask(PrintStream outputStream) {
50           filename = null;
51           this.outputStream = outputStream;
52       }
53   
54       @Override
55       public String version() {
56           return new FileStreamSinkConnector().version();
57       }
58   
59       @Override
60       public void start(Map<String, String> props) {
61           filename = props.get(FileStreamSinkConnector.FILE_CONFIG);
62           if (filename == null) {
63               outputStream = System.out;
64           } else {
65               try {
66                   outputStream = new PrintStream(
67                       Files.newOutputStream(Paths.get(filename), StandardOpenOption.CREATE, StandardOpenOption.APPEND),
68                       false,
69                       StandardCharsets.UTF_8.name());
70               } catch (IOException e) {
71                   throw new ConnectException("Couldn't find or create file '" + filename + "' for FileStreamSinkTask", e);
72               }
73           }
74       }
75   
76       @Override
77       public void put(Collection<SinkRecord> sinkRecords) {
78           for (SinkRecord record : sinkRecords) {
79               log.trace("Writing line to {}: {}", logFilename(), record.value());
80               outputStream.println(record.value());
81           }
82       }
83   
84       @Override
85       public void flush(Map<TopicPartition, OffsetAndMetadata> offsets) {
86           log.trace("Flushing output stream for {}", logFilename());
87           outputStream.flush();
88       }
89   
90       @Override
91       public void stop() {
92           if (outputStream != null && outputStream != System.out)
93               outputStream.close();
94       }
95   
96       private String logFilename() {
97           return filename == null ? "stdout" : filename;
98       }
99   }
