276       @Override
277       public void activateOptions() {
278           // check for config parameter validity
279           Properties props = new Properties();
280           if (brokerList != null)
281               props.put(BOOTSTRAP_SERVERS_CONFIG, brokerList);
282           if (props.isEmpty())
283               throw new ConfigException("The bootstrap servers property should be specified");
284           if (topic == null)
285               throw new ConfigException("Topic must be specified by the Kafka log4j appender");
286           if (compressionType != null)
287               props.put(COMPRESSION_TYPE_CONFIG, compressionType);
288   
289           props.put(ACKS_CONFIG, Integer.toString(requiredNumAcks));
290           props.put(RETRIES_CONFIG, retries);
291           props.put(DELIVERY_TIMEOUT_MS_CONFIG, deliveryTimeoutMs);
292           props.put(LINGER_MS_CONFIG, lingerMs);
293           props.put(BATCH_SIZE_CONFIG, batchSize);
294           // Disable idempotence to avoid deadlock when the producer network thread writes a log line while interacting
295           // with the TransactionManager, see KAFKA-13761 for more information.
296           props.put(ENABLE_IDEMPOTENCE_CONFIG, false);
297   
298           if (securityProtocol != null) {
299               props.put(SECURITY_PROTOCOL_CONFIG, securityProtocol);
300           }
301   
302           if (securityProtocol != null && (securityProtocol.contains("SSL") || securityProtocol.contains("SASL"))) {
303               if (sslEngineFactoryClass != null) {
304                   props.put(SSL_ENGINE_FACTORY_CLASS_CONFIG, sslEngineFactoryClass);
305               }
306           }
307   
308           if (securityProtocol != null && securityProtocol.contains("SSL") && sslTruststoreLocation != null && sslTruststorePassword != null) {
309               props.put(SSL_TRUSTSTORE_LOCATION_CONFIG, sslTruststoreLocation);
310               props.put(SSL_TRUSTSTORE_PASSWORD_CONFIG, sslTruststorePassword);
311   
312               if (sslKeystoreType != null && sslKeystoreLocation != null &&
313                       sslKeystorePassword != null) {
314                   props.put(SSL_KEYSTORE_TYPE_CONFIG, sslKeystoreType);
315                   props.put(SSL_KEYSTORE_LOCATION_CONFIG, sslKeystoreLocation);
316                   props.put(SSL_KEYSTORE_PASSWORD_CONFIG, sslKeystorePassword);
317               }
318           }
319   
320           if (securityProtocol != null && securityProtocol.contains("SASL") && saslKerberosServiceName != null && clientJaasConfPath != null) {
321               props.put(SASL_KERBEROS_SERVICE_NAME, saslKerberosServiceName);
322               System.setProperty("java.security.auth.login.config", clientJaasConfPath);
323           }
324           if (kerb5ConfPath != null) {
325               System.setProperty("java.security.krb5.conf", kerb5ConfPath);
326           }
327           if (saslMechanism != null) {
328               props.put(SASL_MECHANISM, saslMechanism);
329           }
330           if (clientJaasConf != null) {
331               props.put(SASL_JAAS_CONFIG, clientJaasConf);
332           }
333           if (maxBlockMs != null) {
334               props.put(MAX_BLOCK_MS_CONFIG, maxBlockMs);
335           }
336   
337           props.put(KEY_SERIALIZER_CLASS_CONFIG, ByteArraySerializer.class.getName());
338           props.put(VALUE_SERIALIZER_CLASS_CONFIG, ByteArraySerializer.class.getName());
339           this.producer = getKafkaProducer(props);
340           LogLog.debug("Kafka producer connected to " + brokerList);
341           LogLog.debug("Logging for topic: " + topic);
342       }

344       protected Producer<byte[], byte[]> getKafkaProducer(Properties props) {
345           return new KafkaProducer<>(props);
346       }

348       @Override
349       protected void append(LoggingEvent event) {
350           String message = subAppend(event);
351           LogLog.debug("[" + new Date(event.getTimeStamp()) + "]" + message);
352           Future<RecordMetadata> response = producer.send(
353               new ProducerRecord<>(topic, message.getBytes(StandardCharsets.UTF_8)));
354           if (syncSend) {
355               try {
356                   response.get();
357               } catch (InterruptedException | ExecutionException ex) {
358                   if (!ignoreExceptions)
359                       throw new RuntimeException(ex);
360                   LogLog.debug("Exception while getting response", ex);
361               }
362           }
363       }
364   
365       private String subAppend(LoggingEvent event) {
366           return (this.layout == null) ? event.getRenderedMessage() : this.layout.format(event);
367       }

370       public void close() {
371           if (!this.closed) {
372               this.closed = true;
373               producer.close();
374           }
375       }
