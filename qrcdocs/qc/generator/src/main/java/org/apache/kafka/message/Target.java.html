18   package org.apache.kafka.message;
19   
20   import java.util.Collections;
21   import java.util.function.Function;
22   
23   public final class Target {
24       private final FieldSpec field;
25       private final String sourceVariable;
26       private final String humanReadableName;
27       private final Function<String, String> assignmentStatementGenerator;
28   
29       Target(FieldSpec field, String sourceVariable, String humanReadableName,
30              Function<String, String> assignmentStatementGenerator) {
31           this.field = field;
32           this.sourceVariable = sourceVariable;
33           this.humanReadableName = humanReadableName;
34           this.assignmentStatementGenerator = assignmentStatementGenerator;
35       }
36   
37       public String assignmentStatement(String rightHandSide) {
38           return assignmentStatementGenerator.apply(rightHandSide);
39       }
40   
41       public Target nonNullableCopy() {
42           FieldSpec nonNullableField = new FieldSpec(field.name(),
43               field.versionsString(),
44               field.fields(),
45               field.typeString(),
46               field.mapKey(),
47               Versions.NONE.toString(),
48               field.defaultString(),
49               field.ignorable(),
50               field.entityType(),
51               field.about(),
52               field.taggedVersionsString(),
53               field.flexibleVersionsString(),
54               field.tagInteger(),
55               field.zeroCopy());
56           return new Target(nonNullableField, sourceVariable, humanReadableName, assignmentStatementGenerator);
57       }
58   
59       public Target arrayElementTarget(Function<String, String> assignmentStatementGenerator) {
60           if (!field.type().isArray()) {
61               throw new RuntimeException("Field " + field + " is not an array.");
62           }
63           FieldType.ArrayType arrayType = (FieldType.ArrayType) field.type();
64           FieldSpec elementField = new FieldSpec(field.name() + "Element",
65                   field.versions().toString(),
66                   Collections.emptyList(),
67                   arrayType.elementType().toString(),
68                   false,
69                   Versions.NONE.toString(),
70                   "",
71                   false,
72                   EntityType.UNKNOWN,
73                   "",
74                   Versions.NONE.toString(),
75                   field.flexibleVersionsString(),
76                   null,
77                   field.zeroCopy());
78           return new Target(elementField, "_element", humanReadableName + " element",
79               assignmentStatementGenerator);
80       }
81   
82       public FieldSpec field() {
83           return field;
84       }
85   
86       public String sourceVariable() {
87           return sourceVariable;
88       }
89   
90       public String humanReadableName() {
91           return humanReadableName;
92       }
93   }
