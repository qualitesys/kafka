20   import java.util.HashMap;
21   import java.util.Iterator;
22   import java.util.Map;
23   import java.util.TreeMap;

25   /**
26    * Generates Schemas for Kafka MessageData classes.
27    */
28   final class SchemaGenerator {
29       /**
30        * Schema information for a particular message.
31        */
32       static class MessageInfo {
33           /**
34            * The versions of this message that we want to generate a schema for.
35            * This will be constrained by the valid versions for the parent objects.
36            * For example, if the parent message is valid for versions 0 and 1,
37            * we will only generate a version 0 and version 1 schema for child classes,
38            * even if their valid versions are "0+".
39            */
40           private final Versions versions;
41   
42           /**
43            * Maps versions to schema declaration code.  If the schema for a
44            * particular version is the same as that of a previous version,
45            * there will be no entry in the map for it.
46            */
47           private final TreeMap<Short, CodeBuffer> schemaForVersion;
48   
49           MessageInfo(Versions versions) {
50               this.versions = versions;
51               this.schemaForVersion = new TreeMap<>();
52           }
53       }
54   
55       /**
56        * The header file generator.  This is shared with the MessageDataGenerator
57        * instance that owns this SchemaGenerator.
58        */
59       private final HeaderGenerator headerGenerator;
60   
61       /**
62        * A registry with the structures we're generating.
63        */
64       private final StructRegistry structRegistry;
65   
66       /**
67        * Maps message names to message information.
68        */
69       private final Map<String, MessageInfo> messages;
70   
71       /**
72        * The versions that implement a KIP-482 flexible schema.
73        */
74       private Versions messageFlexibleVersions;
75   
76       SchemaGenerator(HeaderGenerator headerGenerator, StructRegistry structRegistry) {
77           this.headerGenerator = headerGenerator;
78           this.structRegistry = structRegistry;
79           this.messages = new HashMap<>();
80       }
81   
82       void generateSchemas(MessageSpec message) throws Exception {
83           this.messageFlexibleVersions = message.flexibleVersions();
84   
85           // First generate schemas for common structures so that they are
86           // available when we generate the inline structures
87           for (Iterator<StructSpec> iter = structRegistry.commonStructs(); iter.hasNext(); ) {
88               StructSpec struct = iter.next();
89               generateSchemas(struct.name(), struct, message.struct().versions());
90           }
91   
92           // Generate schemas for inline structures
93           generateSchemas(message.dataClassName(), message.struct(),
94               message.struct().versions());
95       }
96   
97       void generateSchemas(String className, StructSpec struct,
98                            Versions parentVersions) throws Exception {
99           Versions versions = parentVersions.intersect(struct.versions());
100           MessageInfo messageInfo = messages.get(className);
101           if (messageInfo != null) {
102               return;
103           }
104           messageInfo = new MessageInfo(versions);
105           messages.put(className, messageInfo);
106           // Process the leaf classes first.
107           for (FieldSpec field : struct.fields()) {
108               if (field.type().isStructArray()) {
109                   FieldType.ArrayType arrayType = (FieldType.ArrayType) field.type();
110                   generateSchemas(arrayType.elementType().toString(), structRegistry.findStruct(field), versions);
111               } else if (field.type().isStruct()) {
112                   generateSchemas(field.type().toString(), structRegistry.findStruct(field), versions);
113               }
114           }
115           CodeBuffer prev = null;
116           for (short v = versions.lowest(); v <= versions.highest(); v++) {
117               CodeBuffer cur = new CodeBuffer();
118               generateSchemaForVersion(struct, v, cur);
119               // If this schema version is different from the previous one,
120               // create a new map entry.
121               if (!cur.equals(prev)) {
122                   messageInfo.schemaForVersion.put(v, cur);
123               }
124               prev = cur;
125           }
126       }
127   
128       private void generateSchemaForVersion(StructSpec struct,
129                                             short version,
130                                             CodeBuffer buffer) throws Exception {
131           // Find the last valid field index.
132           int lastValidIndex = struct.fields().size() - 1;
133           while (true) {
134               if (lastValidIndex < 0) {
135                   break;
136               }
137               FieldSpec field = struct.fields().get(lastValidIndex);
138               if ((!field.taggedVersions().contains(version)) &&
139                       field.versions().contains(version)) {
140                   break;
141               }
142               lastValidIndex--;
143           }
144           int finalLine = lastValidIndex;
145           if (messageFlexibleVersions.contains(version)) {
146               finalLine++;
147           }
148   
149           headerGenerator.addImport(MessageGenerator.SCHEMA_CLASS);
150           buffer.printf("new Schema(%n");
151           buffer.incrementIndent();
152           for (int i = 0; i <= lastValidIndex; i++) {
153               FieldSpec field = struct.fields().get(i);
154               if ((!field.versions().contains(version)) ||
155                       field.taggedVersions().contains(version)) {
156                   continue;
157               }
158               Versions fieldFlexibleVersions =
159                   field.flexibleVersions().orElse(messageFlexibleVersions);
160               headerGenerator.addImport(MessageGenerator.FIELD_CLASS);
161               buffer.printf("new Field(\"%s\", %s, \"%s\")%s%n",
162                   field.snakeCaseName(),
163                   fieldTypeToSchemaType(field, version, fieldFlexibleVersions),
164                   field.about(),
165                   i == finalLine ? "" : ",");
166           }
167           if (messageFlexibleVersions.contains(version)) {
168               generateTaggedFieldsSchemaForVersion(struct, version, buffer);
169           }
170           buffer.decrementIndent();
171           buffer.printf(");%n");
172       }
173   
174       private void generateTaggedFieldsSchemaForVersion(StructSpec struct,
175               short version, CodeBuffer buffer) throws Exception {
176           headerGenerator.addStaticImport(MessageGenerator.TAGGED_FIELDS_SECTION_CLASS);
177   
178           // Find the last valid tagged field index.
179           int lastValidIndex = struct.fields().size() - 1;
180           while (true) {
181               if (lastValidIndex < 0) {
182                   break;
183               }
184               FieldSpec field = struct.fields().get(lastValidIndex);
185               if ((field.taggedVersions().contains(version)) &&
186                   field.versions().contains(version)) {
187                   break;
188               }
189               lastValidIndex--;
190           }
191   
192           buffer.printf("TaggedFieldsSection.of(%n");
193           buffer.incrementIndent();
194           for (int i = 0; i <= lastValidIndex; i++) {
195               FieldSpec field = struct.fields().get(i);
196               if ((!field.versions().contains(version)) ||
197                       (!field.taggedVersions().contains(version))) {
198                   continue;
199               }
200               headerGenerator.addImport(MessageGenerator.FIELD_CLASS);
201               Versions fieldFlexibleVersions =
202                   field.flexibleVersions().orElse(messageFlexibleVersions);
203               buffer.printf("%d, new Field(\"%s\", %s, \"%s\")%s%n",
204                   field.tag().get(),
205                   field.snakeCaseName(),
206                   fieldTypeToSchemaType(field, version, fieldFlexibleVersions),
207                   field.about(),
208                   i == lastValidIndex ? "" : ",");
209           }
210           buffer.decrementIndent();
211           buffer.printf(")%n");
212       }

214       private String fieldTypeToSchemaType(FieldSpec field,
215                                            short version,
216                                            Versions fieldFlexibleVersions) {
217           return fieldTypeToSchemaType(field.type(),
218               field.nullableVersions().contains(version),
219               version,
220               fieldFlexibleVersions,
221               field.zeroCopy());
222       }

224       private String fieldTypeToSchemaType(FieldType type,
225                                            boolean nullable,
226                                            short version,
227                                            Versions fieldFlexibleVersions,
228                                            boolean zeroCopy) {
229           if (type instanceof FieldType.BoolFieldType) {
230               headerGenerator.addImport(MessageGenerator.TYPE_CLASS);
231               if (nullable) {
232                   throw new RuntimeException("Type " + type + " cannot be nullable.");
233               }
234               return "Type.BOOLEAN";
235           } else if (type instanceof FieldType.Int8FieldType) {
236               headerGenerator.addImport(MessageGenerator.TYPE_CLASS);
237               if (nullable) {
238                   throw new RuntimeException("Type " + type + " cannot be nullable.");
239               }
240               return "Type.INT8";
241           } else if (type instanceof FieldType.Int16FieldType) {
242               headerGenerator.addImport(MessageGenerator.TYPE_CLASS);
243               if (nullable) {
244                   throw new RuntimeException("Type " + type + " cannot be nullable.");
245               }
246               return "Type.INT16";
247           } else if (type instanceof FieldType.Uint16FieldType) {
248               headerGenerator.addImport(MessageGenerator.TYPE_CLASS);
249               if (nullable) {
250                   throw new RuntimeException("Type " + type + " cannot be nullable.");
251               }
252               return "Type.UINT16";
253           } else if (type instanceof FieldType.Uint32FieldType) {
254               headerGenerator.addImport(MessageGenerator.TYPE_CLASS);
255               if (nullable) {
256                   throw new RuntimeException("Type " + type + " cannot be nullable.");
257               }
258               return "Type.UNSIGNED_INT32";
259           } else if (type instanceof FieldType.Int32FieldType) {
260               headerGenerator.addImport(MessageGenerator.TYPE_CLASS);
261               if (nullable) {
262                   throw new RuntimeException("Type " + type + " cannot be nullable.");
263               }
264               return "Type.INT32";
265           } else if (type instanceof FieldType.Int64FieldType) {
266               headerGenerator.addImport(MessageGenerator.TYPE_CLASS);
267               if (nullable) {
268                   throw new RuntimeException("Type " + type + " cannot be nullable.");
269               }
270               return "Type.INT64";
271           } else if (type instanceof FieldType.UUIDFieldType) {
272               headerGenerator.addImport(MessageGenerator.TYPE_CLASS);
273               if (nullable) {
274                   throw new RuntimeException("Type " + type + " cannot be nullable.");
275               }
276               return "Type.UUID";
277           } else if (type instanceof FieldType.Float64FieldType) {
278               headerGenerator.addImport(MessageGenerator.TYPE_CLASS);
279               if (nullable) {
280                   throw new RuntimeException("Type " + type + " cannot be nullable.");
281               }
282               return "Type.FLOAT64";
283           } else if (type instanceof FieldType.StringFieldType) {
284               headerGenerator.addImport(MessageGenerator.TYPE_CLASS);
285               if (fieldFlexibleVersions.contains(version)) {
286                   return nullable ? "Type.COMPACT_NULLABLE_STRING" : "Type.COMPACT_STRING";
287               } else {
288                   return nullable ? "Type.NULLABLE_STRING" : "Type.STRING";
289               }
290           } else if (type instanceof FieldType.BytesFieldType) {
291               headerGenerator.addImport(MessageGenerator.TYPE_CLASS);
292               if (fieldFlexibleVersions.contains(version)) {
293                   return nullable ? "Type.COMPACT_NULLABLE_BYTES" : "Type.COMPACT_BYTES";
294               } else {
295                   return nullable ? "Type.NULLABLE_BYTES" : "Type.BYTES";
296               }
297           } else if (type.isRecords()) {
298               headerGenerator.addImport(MessageGenerator.TYPE_CLASS);
299               if (fieldFlexibleVersions.contains(version)) {
300                   return "Type.COMPACT_RECORDS";
301               } else {
302                   return "Type.RECORDS";
303               }
304           } else if (type.isArray()) {
305               if (fieldFlexibleVersions.contains(version)) {
306                   headerGenerator.addImport(MessageGenerator.COMPACT_ARRAYOF_CLASS);
307                   FieldType.ArrayType arrayType = (FieldType.ArrayType) type;
308                   String prefix = nullable ? "CompactArrayOf.nullable" : "new CompactArrayOf";
309                   return String.format("%s(%s)", prefix,
310                           fieldTypeToSchemaType(arrayType.elementType(), false, version, fieldFlexibleVersions, false));
311   
312               } else {
313                   headerGenerator.addImport(MessageGenerator.ARRAYOF_CLASS);
314                   FieldType.ArrayType arrayType = (FieldType.ArrayType) type;
315                   String prefix = nullable ? "ArrayOf.nullable" : "new ArrayOf";
316                   return String.format("%s(%s)", prefix,
317                           fieldTypeToSchemaType(arrayType.elementType(), false, version, fieldFlexibleVersions, false));
318               }
319           } else if (type.isStruct()) {
320               if (nullable) {
321                   throw new RuntimeException("Type " + type + " cannot be nullable.");
322               }
323               return String.format("%s.SCHEMA_%d", type.toString(),
324                   floorVersion(type.toString(), version));
325           } else {
326               throw new RuntimeException("Unsupported type " + type);
327           }
328       }

330       /**
331        * Find the lowest schema version for a given class that is the same as the
332        * given version.
333        */
334       private short floorVersion(String className, short v) {
335           MessageInfo message = messages.get(className);
336           return message.schemaForVersion.floorKey(v);
337       }
338   
339       /**
340        * Write the message schema to the provided buffer.
341        *
342        * @param className     The class name.
343        * @param buffer        The destination buffer.
344        */
345       void writeSchema(String className, CodeBuffer buffer) throws Exception {
346           MessageInfo messageInfo = messages.get(className);
347           Versions versions = messageInfo.versions;
348   
349           for (short v = versions.lowest(); v <= versions.highest(); v++) {
350               CodeBuffer declaration = messageInfo.schemaForVersion.get(v);
351               if (declaration == null) {
352                   buffer.printf("public static final Schema SCHEMA_%d = SCHEMA_%d;%n", v, v - 1);
353               } else {
354                   buffer.printf("public static final Schema SCHEMA_%d =%n", v);
355                   buffer.incrementIndent();
356                   declaration.write(buffer);
357                   buffer.decrementIndent();
358               }
359               buffer.printf("%n");
360           }
361           buffer.printf("public static final Schema[] SCHEMAS = new Schema[] {%n");
362           buffer.incrementIndent();
363           for (short v = 0; v < versions.lowest(); v++) {
364               buffer.printf("null%s%n", (v == versions.highest()) ? "" : ",");
365           }
366           for (short v = versions.lowest(); v <= versions.highest(); v++) {
367               buffer.printf("SCHEMA_%d%s%n", v, (v == versions.highest()) ? "" : ",");
368           }
369           buffer.decrementIndent();
370           buffer.printf("};%n");
371           buffer.printf("%n");
372   
373           buffer.printf("public static final short LOWEST_SUPPORTED_VERSION = %d;%n", versions.lowest());
374           buffer.printf("public static final short HIGHEST_SUPPORTED_VERSION = %d;%n", versions.highest());
375           buffer.printf("%n");
376       }
