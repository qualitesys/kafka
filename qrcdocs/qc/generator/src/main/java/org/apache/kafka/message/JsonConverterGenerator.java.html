20   import java.io.BufferedWriter;
21   import java.util.Iterator;

23   /**
24    * Generates Kafka MessageData classes.
25    */
26   public final class JsonConverterGenerator implements MessageClassGenerator {
27       private final static String SUFFIX = "JsonConverter";
28       private final String packageName;
29       private final StructRegistry structRegistry;
30       private final HeaderGenerator headerGenerator;
31       private final CodeBuffer buffer;
32   
33       JsonConverterGenerator(String packageName) {
34           this.packageName = packageName;
35           this.structRegistry = new StructRegistry();
36           this.headerGenerator = new HeaderGenerator(packageName);
37           this.buffer = new CodeBuffer();
38       }
39   
40       @Override
41       public String outputName(MessageSpec spec) {
42           return spec.dataClassName() + SUFFIX;
43       }
44   
45       @Override
46       public void generateAndWrite(MessageSpec message, BufferedWriter writer)
47               throws Exception {
48           structRegistry.register(message);
49           headerGenerator.addStaticImport(String.format("%s.%s.*",
50               packageName, message.dataClassName()));
51           buffer.printf("public class %s {%n",
52               MessageGenerator.capitalizeFirst(outputName(message)));
53           buffer.incrementIndent();
54           generateConverters(message.dataClassName(), message.struct(),
55               message.validVersions());
56           for (Iterator<StructRegistry.StructInfo> iter = structRegistry.structs();
57                   iter.hasNext(); ) {
58               StructRegistry.StructInfo info = iter.next();
59               buffer.printf("%n");
60               buffer.printf("public static class %s {%n",
61                   MessageGenerator.capitalizeFirst(info.spec().name() + SUFFIX));
62               buffer.incrementIndent();
63               generateConverters(MessageGenerator.capitalizeFirst(info.spec().name()),
64                   info.spec(), info.parentVersions());
65               buffer.decrementIndent();
66               buffer.printf("}%n");
67           }
68           buffer.decrementIndent();
69           buffer.printf("}%n");
70           headerGenerator.generate();
71           headerGenerator.buffer().write(writer);
72           buffer.write(writer);
73       }

75       private void generateConverters(String name,
76                                       StructSpec spec,
77                                       Versions parentVersions) {
78           generateRead(name, spec, parentVersions);
79           generateWrite(name, spec, parentVersions);
80           generateOverloadWrite(name);
81       }

83       private void generateRead(String className,
84                                 StructSpec struct,
85                                 Versions parentVersions) {
86           headerGenerator.addImport(MessageGenerator.JSON_NODE_CLASS);
87           buffer.printf("public static %s read(JsonNode _node, short _version) {%n",
88               className);
89           buffer.incrementIndent();
90           buffer.printf("%s _object = new %s();%n", className, className);
91           VersionConditional.forVersions(struct.versions(), parentVersions).
92               allowMembershipCheckAlwaysFalse(false).
93               ifNotMember(__ -> {
94                   headerGenerator.addImport(MessageGenerator.UNSUPPORTED_VERSION_EXCEPTION_CLASS);
95                   buffer.printf("throw new UnsupportedVersionException(\"Can't read " +
96                       "version \" + _version + \" of %s\");%n", className);
97               }).
98               generate(buffer);
99           Versions curVersions = parentVersions.intersect(struct.versions());
100           for (FieldSpec field : struct.fields()) {
101               String sourceVariable = String.format("_%sNode", field.camelCaseName());
102               buffer.printf("JsonNode %s = _node.get(\"%s\");%n",
103                   sourceVariable,
104                   field.camelCaseName());
105               buffer.printf("if (%s == null) {%n", sourceVariable);
106               buffer.incrementIndent();
107               Versions mandatoryVersions = field.versions().subtract(field.taggedVersions());
108               VersionConditional.forVersions(mandatoryVersions, curVersions).
109                   ifMember(__ -> {
110                       buffer.printf("throw new RuntimeException(\"%s: unable to locate " +
111                               "field \'%s\', which is mandatory in version \" + _version);%n",
112                           className, field.camelCaseName());
113                   }).
114                   ifNotMember(__ -> {
115                       buffer.printf("_object.%s = %s;%n", field.camelCaseName(),
116                           field.fieldDefault(headerGenerator, structRegistry));
117                   }).
118                   generate(buffer);
119               buffer.decrementIndent();
120               buffer.printf("} else {%n");
121               buffer.incrementIndent();
122               VersionConditional.forVersions(struct.versions(), curVersions).
123                   ifMember(presentVersions -> {
124                       generateTargetFromJson(new Target(field,
125                               sourceVariable,
126                               className,
127                           input -> String.format("_object.%s = %s", field.camelCaseName(), input)),
128                           curVersions);
129                   }).ifNotMember(__ -> {
130                       buffer.printf("throw new RuntimeException(\"%s: field \'%s\' is not " +
131                           "supported in version \" + _version);%n",
132                           className, field.camelCaseName());
133                   }).generate(buffer);
134               buffer.decrementIndent();
135               buffer.printf("}%n");
136           }
137           buffer.printf("return _object;%n");
138           buffer.decrementIndent();
139           buffer.printf("}%n");
140       }

142       private void generateTargetFromJson(Target target, Versions curVersions) {
143           if (target.field().type() instanceof FieldType.BoolFieldType) {
144               buffer.printf("if (!%s.isBoolean()) {%n", target.sourceVariable());
145               buffer.incrementIndent();
146               buffer.printf("throw new RuntimeException(\"%s expected Boolean type, " +
147                   "but got \" + _node.getNodeType());%n", target.humanReadableName());
148               buffer.decrementIndent();
149               buffer.printf("}%n");
150               buffer.printf("%s;%n", target.assignmentStatement(
151                   target.sourceVariable() + ".asBoolean()"));
152           } else if (target.field().type() instanceof FieldType.Int8FieldType) {
153               headerGenerator.addImport(MessageGenerator.MESSAGE_UTIL_CLASS);
154               buffer.printf("%s;%n", target.assignmentStatement(
155                   String.format("MessageUtil.jsonNodeToByte(%s, \"%s\")",
156                       target.sourceVariable(), target.humanReadableName())));
157           } else if (target.field().type() instanceof FieldType.Int16FieldType) {
158               headerGenerator.addImport(MessageGenerator.MESSAGE_UTIL_CLASS);
159               buffer.printf("%s;%n", target.assignmentStatement(
160                   String.format("MessageUtil.jsonNodeToShort(%s, \"%s\")",
161                       target.sourceVariable(), target.humanReadableName())));
162           } else if (target.field().type() instanceof FieldType.Uint16FieldType) {
163               headerGenerator.addImport(MessageGenerator.MESSAGE_UTIL_CLASS);
164               buffer.printf("%s;%n", target.assignmentStatement(
165                   String.format("MessageUtil.jsonNodeToUnsignedShort(%s, \"%s\")",
166                       target.sourceVariable(), target.humanReadableName())));
167           } else if (target.field().type() instanceof FieldType.Uint32FieldType) {
168               headerGenerator.addImport(MessageGenerator.MESSAGE_UTIL_CLASS);
169               buffer.printf("%s;%n", target.assignmentStatement(
170                    String.format("MessageUtil.jsonNodeToUnsignedInt(%s, \"%s\")",
171                        target.sourceVariable(), target.humanReadableName())));
172           } else if (target.field().type() instanceof FieldType.Int32FieldType) {
173               headerGenerator.addImport(MessageGenerator.MESSAGE_UTIL_CLASS);
174               buffer.printf("%s;%n", target.assignmentStatement(
175                   String.format("MessageUtil.jsonNodeToInt(%s, \"%s\")",
176                       target.sourceVariable(), target.humanReadableName())));
177           } else if (target.field().type() instanceof FieldType.Int64FieldType) {
178               headerGenerator.addImport(MessageGenerator.MESSAGE_UTIL_CLASS);
179               buffer.printf("%s;%n", target.assignmentStatement(
180                   String.format("MessageUtil.jsonNodeToLong(%s, \"%s\")",
181                       target.sourceVariable(), target.humanReadableName())));
182           } else if (target.field().type() instanceof FieldType.UUIDFieldType) {
183               buffer.printf("if (!%s.isTextual()) {%n", target.sourceVariable());
184               buffer.incrementIndent();
185               buffer.printf("throw new RuntimeException(\"%s expected a JSON string " +
186                   "type, but got \" + _node.getNodeType());%n", target.humanReadableName());
187               buffer.decrementIndent();
188               buffer.printf("}%n");
189               headerGenerator.addImport(MessageGenerator.UUID_CLASS);
190               buffer.printf("%s;%n", target.assignmentStatement(String.format(
191                   "Uuid.fromString(%s.asText())", target.sourceVariable())));
192           } else if (target.field().type() instanceof FieldType.Float64FieldType) {
193               headerGenerator.addImport(MessageGenerator.MESSAGE_UTIL_CLASS);
194               buffer.printf("%s;%n", target.assignmentStatement(
195                   String.format("MessageUtil.jsonNodeToDouble(%s, \"%s\")",
196                       target.sourceVariable(), target.humanReadableName())));
197           } else {
198               // Handle the variable length types.  All of them are potentially
199               // nullable, so handle that here.
200               IsNullConditional.forName(target.sourceVariable()).
201                   nullableVersions(target.field().nullableVersions()).
202                   possibleVersions(curVersions).
203                   conditionalGenerator((name, negated) ->
204                       String.format("%s%s.isNull()", negated ? "!" : "", name)).
205                   ifNull(() -> {
206                       buffer.printf("%s;%n", target.assignmentStatement("null"));
207                   }).
208                   ifShouldNotBeNull(() -> {
209                       generateVariableLengthTargetFromJson(target, curVersions);
210                   }).
211                   generate(buffer);
212           }
213       }

215       private void generateVariableLengthTargetFromJson(Target target, Versions curVersions) {
216           if (target.field().type().isString()) {
217               buffer.printf("if (!%s.isTextual()) {%n", target.sourceVariable());
218               buffer.incrementIndent();
219               buffer.printf("throw new RuntimeException(\"%s expected a string " +
220                   "type, but got \" + _node.getNodeType());%n", target.humanReadableName());
221               buffer.decrementIndent();
222               buffer.printf("}%n");
223               buffer.printf("%s;%n", target.assignmentStatement(
224                   String.format("%s.asText()", target.sourceVariable())));
225           } else if (target.field().type().isBytes()) {
226               headerGenerator.addImport(MessageGenerator.MESSAGE_UTIL_CLASS);
227               if (target.field().zeroCopy()) {
228                   headerGenerator.addImport(MessageGenerator.BYTE_BUFFER_CLASS);
229                   buffer.printf("%s;%n", target.assignmentStatement(
230                       String.format("ByteBuffer.wrap(MessageUtil.jsonNodeToBinary(%s, \"%s\"))",
231                           target.sourceVariable(), target.humanReadableName())));
232               } else {
233                   buffer.printf("%s;%n", target.assignmentStatement(
234                       String.format("MessageUtil.jsonNodeToBinary(%s, \"%s\")",
235                           target.sourceVariable(), target.humanReadableName())));
236               }
237           } else if (target.field().type().isRecords()) {
238               headerGenerator.addImport(MessageGenerator.MESSAGE_UTIL_CLASS);
239               headerGenerator.addImport(MessageGenerator.BYTE_BUFFER_CLASS);
240               headerGenerator.addImport(MessageGenerator.MEMORY_RECORDS_CLASS);
241               buffer.printf("%s;%n", target.assignmentStatement(
242                   String.format("MemoryRecords.readableRecords(ByteBuffer.wrap(MessageUtil.jsonNodeToBinary(%s, \"%s\")))",
243                       target.sourceVariable(), target.humanReadableName())));
244           } else if (target.field().type().isArray()) {
245               buffer.printf("if (!%s.isArray()) {%n", target.sourceVariable());
246               buffer.incrementIndent();
247               buffer.printf("throw new RuntimeException(\"%s expected a JSON " +
248                   "array, but got \" + _node.getNodeType());%n", target.humanReadableName());
249               buffer.decrementIndent();
250               buffer.printf("}%n");
251               String type = target.field().concreteJavaType(headerGenerator, structRegistry);
252               buffer.printf("%s _collection = new %s(%s.size());%n", type, type, target.sourceVariable());
253               buffer.printf("%s;%n", target.assignmentStatement("_collection"));
254               headerGenerator.addImport(MessageGenerator.JSON_NODE_CLASS);
255               buffer.printf("for (JsonNode _element : %s) {%n", target.sourceVariable());
256               buffer.incrementIndent();
257               generateTargetFromJson(target.arrayElementTarget(
258                   input -> String.format("_collection.add(%s)", input)),
259                   curVersions);
260               buffer.decrementIndent();
261               buffer.printf("}%n");
262           } else if (target.field().type().isStruct()) {
263               buffer.printf("%s;%n", target.assignmentStatement(
264                   String.format("%s%s.read(%s, _version)",
265                   target.field().type().toString(), SUFFIX, target.sourceVariable())));
266           } else {
267               throw new RuntimeException("Unexpected type " + target.field().type());
268           }
269       }

277           buffer.printf("}%n");
278       }

280       private void generateWrite(String className,
281                                  StructSpec struct,
282                                  Versions parentVersions) {
283           headerGenerator.addImport(MessageGenerator.JSON_NODE_CLASS);
284           buffer.printf("public static JsonNode write(%s _object, short _version, boolean _serializeRecords) {%n",
285               className);
286           buffer.incrementIndent();
287           VersionConditional.forVersions(struct.versions(), parentVersions).
288               allowMembershipCheckAlwaysFalse(false).
289               ifNotMember(__ -> {
290                   headerGenerator.addImport(MessageGenerator.UNSUPPORTED_VERSION_EXCEPTION_CLASS);
291                   buffer.printf("throw new UnsupportedVersionException(\"Can't write " +
292                       "version \" + _version + \" of %s\");%n", className);
293               }).
294               generate(buffer);
295           Versions curVersions = parentVersions.intersect(struct.versions());
296           headerGenerator.addImport(MessageGenerator.OBJECT_NODE_CLASS);
297           headerGenerator.addImport(MessageGenerator.JSON_NODE_FACTORY_CLASS);
298           buffer.printf("ObjectNode _node = new ObjectNode(JsonNodeFactory.instance);%n");
299           for (FieldSpec field : struct.fields()) {
300               Target target = new Target(field,
301                   String.format("_object.%s", field.camelCaseName()),
302                   field.camelCaseName(),
303                   input -> String.format("_node.set(\"%s\", %s)", field.camelCaseName(), input));
304               VersionConditional cond = VersionConditional.forVersions(field.versions(), curVersions).
305                   ifMember(presentVersions -> {
306                       VersionConditional.forVersions(field.taggedVersions(), presentVersions).
307                           ifMember(presentAndTaggedVersions -> {
308                               field.generateNonDefaultValueCheck(headerGenerator,
309                                   structRegistry, buffer, "_object.", field.nullableVersions());
310                               buffer.incrementIndent();
311                               if (field.defaultString().equals("null")) {
312                                   // If the default was null, and we already checked that this field was not
313                                   // the default, we can omit further null checks.
314                                   generateTargetToJson(target.nonNullableCopy(), presentAndTaggedVersions);
315                               } else {
316                                   generateTargetToJson(target, presentAndTaggedVersions);
317                               }
318                               buffer.decrementIndent();
319                               buffer.printf("}%n");
320                           }).
321                           ifNotMember(presentAndNotTaggedVersions -> {
322                               generateTargetToJson(target, presentAndNotTaggedVersions);
323                           }).
324                           generate(buffer);
325                   });
326               if (!field.ignorable()) {
327                   cond.ifNotMember(__ -> {
328                       field.generateNonIgnorableFieldCheck(headerGenerator,
329                           structRegistry, "_object.", buffer);
330                   });
331               }
332               cond.generate(buffer);
333           }
334           buffer.printf("return _node;%n");
335           buffer.decrementIndent();
336           buffer.printf("}%n");
337       }

339       private void generateTargetToJson(Target target, Versions versions) {
340           if (target.field().type() instanceof FieldType.BoolFieldType) {
341               headerGenerator.addImport(MessageGenerator.BOOLEAN_NODE_CLASS);
342               buffer.printf("%s;%n", target.assignmentStatement(
343                   String.format("BooleanNode.valueOf(%s)", target.sourceVariable())));
344           } else if ((target.field().type() instanceof FieldType.Int8FieldType) ||
345                   (target.field().type() instanceof FieldType.Int16FieldType)) {
346               headerGenerator.addImport(MessageGenerator.SHORT_NODE_CLASS);
347               buffer.printf("%s;%n", target.assignmentStatement(
348                   String.format("new ShortNode(%s)", target.sourceVariable())));
349           } else if ((target.field().type() instanceof FieldType.Int32FieldType) ||
350                   (target.field().type() instanceof FieldType.Uint16FieldType)) {
351               headerGenerator.addImport(MessageGenerator.INT_NODE_CLASS);
352               buffer.printf("%s;%n", target.assignmentStatement(
353                   String.format("new IntNode(%s)", target.sourceVariable())));
354           } else if (target.field().type() instanceof FieldType.Int64FieldType ||
355                   (target.field().type() instanceof FieldType.Uint32FieldType)) {
356               headerGenerator.addImport(MessageGenerator.LONG_NODE_CLASS);
357               buffer.printf("%s;%n", target.assignmentStatement(
358                   String.format("new LongNode(%s)", target.sourceVariable())));
359           } else if (target.field().type() instanceof FieldType.UUIDFieldType) {
360               headerGenerator.addImport(MessageGenerator.TEXT_NODE_CLASS);
361               buffer.printf("%s;%n", target.assignmentStatement(
362                   String.format("new TextNode(%s.toString())", target.sourceVariable())));
363           } else if (target.field().type() instanceof FieldType.Float64FieldType) {
364               headerGenerator.addImport(MessageGenerator.DOUBLE_NODE_CLASS);
365               buffer.printf("%s;%n", target.assignmentStatement(
366                   String.format("new DoubleNode(%s)", target.sourceVariable())));
367           } else {
368               // Handle the variable length types.  All of them are potentially
369               // nullable, so handle that here.
370               IsNullConditional.forName(target.sourceVariable()).
371                   nullableVersions(target.field().nullableVersions()).
372                   possibleVersions(versions).
373                   conditionalGenerator((name, negated) ->
374                       String.format("%s %s= null", name, negated ? "!" : "=")).
375                   ifNull(() -> {
376                       headerGenerator.addImport(MessageGenerator.NULL_NODE_CLASS);
377                       buffer.printf("%s;%n", target.assignmentStatement("NullNode.instance"));
378                   }).
379                   ifShouldNotBeNull(() -> {
380                       generateVariableLengthTargetToJson(target, versions);
381                   }).
382                   generate(buffer);
383           }
384       }

386       private void generateVariableLengthTargetToJson(Target target, Versions versions) {
387           if (target.field().type().isString()) {
388               headerGenerator.addImport(MessageGenerator.TEXT_NODE_CLASS);
389               buffer.printf("%s;%n", target.assignmentStatement(
390                   String.format("new TextNode(%s)", target.sourceVariable())));
391           } else if (target.field().type().isBytes()) {
392               headerGenerator.addImport(MessageGenerator.BINARY_NODE_CLASS);
393               if (target.field().zeroCopy()) {
394                   headerGenerator.addImport(MessageGenerator.MESSAGE_UTIL_CLASS);
395                   buffer.printf("%s;%n", target.assignmentStatement(
396                       String.format("new BinaryNode(MessageUtil.byteBufferToArray(%s))",
397                           target.sourceVariable())));
398               } else {
399                   headerGenerator.addImport(MessageGenerator.ARRAYS_CLASS);
400                   buffer.printf("%s;%n", target.assignmentStatement(
401                       String.format("new BinaryNode(Arrays.copyOf(%s, %s.length))",
402                               target.sourceVariable(), target.sourceVariable())));
403               }
404           } else if (target.field().type().isRecords()) {
405               headerGenerator.addImport(MessageGenerator.BINARY_NODE_CLASS);
406               headerGenerator.addImport(MessageGenerator.INT_NODE_CLASS);
407               // KIP-673: When logging requests/responses, we do not serialize the record, instead we
408               // output its sizeInBytes, because outputting the bytes is not very useful and can be
409               // quite expensive. Otherwise, we will serialize the record.
410               buffer.printf("if (_serializeRecords) {%n");
411               buffer.incrementIndent();
