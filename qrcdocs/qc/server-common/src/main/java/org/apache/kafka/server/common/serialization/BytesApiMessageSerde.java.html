17   package org.apache.kafka.server.common.serialization;
18   
19   import org.apache.kafka.common.protocol.ApiMessage;
20   import org.apache.kafka.common.protocol.ByteBufferAccessor;
21   import org.apache.kafka.common.protocol.ObjectSerializationCache;
22   import org.apache.kafka.common.protocol.Readable;
23   import org.apache.kafka.server.common.ApiMessageAndVersion;
24   
25   import java.nio.ByteBuffer;
26   
27   /**
28    * This class provides conversion of {@code ApiMessageAndVersion} to bytes and vice versa.. This can be used as serialization protocol for any
29    * metadata records derived of {@code ApiMessage}s. It internally uses {@link AbstractApiMessageSerde} for serialization/deserialization
30    * mechanism.
31    * <p></p>
32    * Implementors need to extend this class and implement {@link #apiMessageFor(short)} method to return a respective
33    * {@code ApiMessage} for the given {@code apiKey}. This is required to deserialize the bytes to build the respective
34    * {@code ApiMessage} instance.
35    */
36   public abstract class BytesApiMessageSerde {
37   
38       private final AbstractApiMessageSerde apiMessageSerde = new AbstractApiMessageSerde() {
39           @Override
40           public ApiMessage apiMessageFor(short apiKey) {
41               return BytesApiMessageSerde.this.apiMessageFor(apiKey);
42           }
43       };
44   
45       public byte[] serialize(ApiMessageAndVersion messageAndVersion) {
46           ObjectSerializationCache cache = new ObjectSerializationCache();
47           int size = apiMessageSerde.recordSize(messageAndVersion, cache);
48           ByteBufferAccessor writable = new ByteBufferAccessor(ByteBuffer.allocate(size));
49           apiMessageSerde.write(messageAndVersion, cache, writable);
50   
51           return writable.buffer().array();
52       }
53   
54       public ApiMessageAndVersion deserialize(byte[] data) {
55           Readable readable = new ByteBufferAccessor(ByteBuffer.wrap(data));
56   
57           return apiMessageSerde.read(readable, data.length);
58       }
59   
60       /**
61        * Return {@code ApiMessage} instance for the given {@code apiKey}. This is used while deserializing the bytes
62        * payload into the respective {@code ApiMessage} in {@link #deserialize(byte[])} method.
63        *
64        * @param apiKey apiKey for which a {@code ApiMessage} to be created.
65        */
66       public abstract ApiMessage apiMessageFor(short apiKey);
67   
68   }
