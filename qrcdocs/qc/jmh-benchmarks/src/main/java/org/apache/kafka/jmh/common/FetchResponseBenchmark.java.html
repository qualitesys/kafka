24   import org.apache.kafka.common.network.Send;
25   import org.apache.kafka.common.protocol.ApiKeys;
26   import org.apache.kafka.common.protocol.Errors;
27   import org.apache.kafka.common.record.CompressionType;
28   import org.apache.kafka.common.record.MemoryRecords;
29   import org.apache.kafka.common.record.SimpleRecord;
30   import org.apache.kafka.common.requests.ByteBufferChannel;
31   import org.apache.kafka.common.requests.FetchResponse;
32   import org.apache.kafka.common.requests.ResponseHeader;
33   import org.openjdk.jmh.annotations.Benchmark;
34   import org.openjdk.jmh.annotations.BenchmarkMode;
35   import org.openjdk.jmh.annotations.Fork;
36   import org.openjdk.jmh.annotations.Level;
37   import org.openjdk.jmh.annotations.Measurement;

39   import org.openjdk.jmh.annotations.OutputTimeUnit;
40   import org.openjdk.jmh.annotations.Param;
41   import org.openjdk.jmh.annotations.Scope;
42   import org.openjdk.jmh.annotations.Setup;
43   import org.openjdk.jmh.annotations.State;
44   import org.openjdk.jmh.annotations.Warmup;
45   
46   import java.io.IOException;
47   import java.nio.charset.StandardCharsets;
48   import java.util.HashMap;
49   import java.util.LinkedHashMap;
50   import java.util.Map;
51   import java.util.UUID;
52   import java.util.concurrent.TimeUnit;
53   
54   @State(Scope.Benchmark)
55   @Fork(value = 1)
56   @Warmup(iterations = 5)
57   @Measurement(iterations = 15)
58   @BenchmarkMode(Mode.AverageTime)
59   @OutputTimeUnit(TimeUnit.NANOSECONDS)
60   public class FetchResponseBenchmark {
61       @Param({"10", "500", "1000"})
62       private int topicCount;
63   
64       @Param({"3", "10", "20"})
65       private int partitionCount;
66   
67       LinkedHashMap<TopicIdPartition, FetchResponseData.PartitionData> responseData;
68   
69       Map<String, Uuid> topicIds;
70   
71       Map<Uuid, String> topicNames;
72   
73       ResponseHeader header;
74   
75       FetchResponse fetchResponse;
76   
77       FetchResponseData fetchResponseData;
78   
79       @Setup(Level.Trial)
80       public void setup() {
81           MemoryRecords records = MemoryRecords.withRecords(CompressionType.NONE,
82                   new SimpleRecord(1000, "key1".getBytes(StandardCharsets.UTF_8), "value1".getBytes(StandardCharsets.UTF_8)),
83                   new SimpleRecord(1001, "key2".getBytes(StandardCharsets.UTF_8), "value2".getBytes(StandardCharsets.UTF_8)),
84                   new SimpleRecord(1002, "key3".getBytes(StandardCharsets.UTF_8), "value3".getBytes(StandardCharsets.UTF_8)));
85   
86           this.responseData = new LinkedHashMap<>();
87           this.topicIds = new HashMap<>();
88           this.topicNames = new HashMap<>();
89           for (int topicIdx = 0; topicIdx < topicCount; topicIdx++) {
90               String topic = UUID.randomUUID().toString();
91               Uuid id = Uuid.randomUuid();
92               topicIds.put(topic, id);
93               topicNames.put(id, topic);
94               for (int partitionId = 0; partitionId < partitionCount; partitionId++) {
95                   FetchResponseData.PartitionData partitionData = new FetchResponseData.PartitionData()
96                                   .setPartitionIndex(partitionId)
97                                   .setLastStableOffset(0)
98                                   .setLogStartOffset(0)
99                                   .setRecords(records);
100                   responseData.put(new TopicIdPartition(id, new TopicPartition(topic, partitionId)), partitionData);
101               }
102           }
103   
104           this.header = new ResponseHeader(100, ApiKeys.FETCH.responseHeaderVersion(ApiKeys.FETCH.latestVersion()));
105           this.fetchResponse = FetchResponse.of(Errors.NONE, 0, 0, responseData);
106           this.fetchResponseData = this.fetchResponse.data();
107       }
108   
109       @Benchmark
110       public int testConstructFetchResponse() {
111           FetchResponse fetchResponse = FetchResponse.of(Errors.NONE, 0, 0, responseData);
112           return fetchResponse.data().responses().size();
113       }
114   
115       @Benchmark
116       public int testPartitionMapFromData() {
117           return new FetchResponse(fetchResponseData).responseData(topicNames, ApiKeys.FETCH.latestVersion()).size();
118       }
119   
120       @Benchmark
121       public int testSerializeFetchResponse() throws IOException {
122           Send send = fetchResponse.toSend(header, ApiKeys.FETCH.latestVersion());
123           ByteBufferChannel channel = new ByteBufferChannel(send.size());
124           send.writeTo(channel);
125           return channel.buffer().limit();
126       }
127   }
