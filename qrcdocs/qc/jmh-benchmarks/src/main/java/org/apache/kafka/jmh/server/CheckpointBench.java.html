65   @Warmup(iterations = 5)
66   @Measurement(iterations = 5)
67   @Fork(3)
68   @OutputTimeUnit(TimeUnit.MILLISECONDS)
69   @State(value = Scope.Benchmark)
70   public class CheckpointBench {
71   
72       @Param({"100", "1000", "2000"})
73       public int numTopics;
74   
75       @Param({"3"})
76       public int numPartitions;
77   
78       private final String topicName = "foo";
79   
80       private Scheduler scheduler;
81   
82       private Metrics metrics;
83   
84       private MockTime time;
85   
86       private KafkaConfig brokerProperties;
87   
88       private ReplicaManager replicaManager;
89       private QuotaFactory.QuotaManagers quotaManagers;
90       private LogDirFailureChannel failureChannel;
91       private LogManager logManager;
92       private AlterPartitionManager alterPartitionManager;
93   
94   
95       @SuppressWarnings("deprecation")
96       @Setup(Level.Trial)
97       public void setup() {
98           this.scheduler = new KafkaScheduler(1, "scheduler-thread", true);
99           this.brokerProperties = KafkaConfig.fromProps(TestUtils.createBrokerConfig(
100                   0, TestUtils.MockZkConnect(), true, true, 9092, Option.empty(), Option.empty(),
101                   Option.empty(), true, false, 0, false, 0, false, 0, Option.empty(), 1, true, 1,
102                   (short) 1));
103           this.metrics = new Metrics();
104           this.time = new MockTime();
105           this.failureChannel = new LogDirFailureChannel(brokerProperties.logDirs().size());
106           final List<File> files =
107               JavaConverters.seqAsJavaList(brokerProperties.logDirs()).stream().map(File::new).collect(Collectors.toList());
108           this.logManager = TestUtils.createLogManager(JavaConverters.asScalaBuffer(files),
109                   LogConfig.apply(), new MockConfigRepository(), CleanerConfig.apply(1, 4 * 1024 * 1024L, 0.9d,
110                           1024 * 1024, 32 * 1024 * 1024,
111                           Double.MAX_VALUE, 15 * 1000, true, "MD5"), time, MetadataVersion.latest());
112           scheduler.startup();
113           final BrokerTopicStats brokerTopicStats = new BrokerTopicStats();
114           final MetadataCache metadataCache =
115                   MetadataCache.zkMetadataCache(this.brokerProperties.brokerId(), this.brokerProperties.interBrokerProtocolVersion(), BrokerFeatures.createEmpty());
116           this.quotaManagers =
117                   QuotaFactory.instantiate(this.brokerProperties,
118                           this.metrics,
119                           this.time, "");
120   
121           this.alterPartitionManager = TestUtils.createAlterIsrManager();
122           this.replicaManager = new ReplicaManagerBuilder().
123               setConfig(brokerProperties).
124               setMetrics(metrics).
125               setTime(time).
126               setScheduler(scheduler).
127               setLogManager(logManager).
128               setQuotaManagers(quotaManagers).
129               setBrokerTopicStats(brokerTopicStats).
130               setMetadataCache(metadataCache).
131               setLogDirFailureChannel(failureChannel).
132               setAlterPartitionManager(alterPartitionManager).
133               build();
134           replicaManager.startup();
135   
136           List<TopicPartition> topicPartitions = new ArrayList<>();
137           for (int topicNum = 0; topicNum < numTopics; topicNum++) {
138               final String topicName = this.topicName + "-" + topicNum;
139               for (int partitionNum = 0; partitionNum < numPartitions; partitionNum++) {
140                   topicPartitions.add(new TopicPartition(topicName, partitionNum));
141               }
142           }
143   
144           OffsetCheckpoints checkpoints = (logDir, topicPartition) -> Option.apply(0L);
145           for (TopicPartition topicPartition : topicPartitions) {
146               final Partition partition = this.replicaManager.createPartition(topicPartition);
147               partition.createLogIfNotExists(true, false, checkpoints, Option.apply(Uuid.randomUuid()));
148           }
149   
150           replicaManager.checkpointHighWatermarks();
151       }
152   
153       @TearDown(Level.Trial)
154       public void tearDown() throws Exception {
155           this.replicaManager.shutdown(false);
156           this.metrics.close();
157           this.scheduler.shutdown();
158           this.quotaManagers.shutdown();
159           for (File dir : JavaConverters.asJavaCollection(logManager.liveLogDirs())) {
160               Utils.delete(dir);
161           }
162       }
163   
164   
165       @Benchmark
166       @Threads(1)
167       public void measureCheckpointHighWatermarks() {
168           this.replicaManager.checkpointHighWatermarks();
169       }
170   
171       @Benchmark
172       @Threads(1)
173       public void measureCheckpointLogStartOffsets() {
174           this.logManager.checkpointLogStartOffsets();
175       }
176   }
