27   import java.util.List;
28   import java.util.Optional;
29   import java.util.Properties;

31   import java.util.concurrent.CountDownLatch;
32   import java.util.concurrent.ExecutionException;
33   import java.util.concurrent.TimeUnit;

35   /**
36    * This exactly once demo driver takes 3 arguments:
37    *   - partition: number of partitions for input/output topic
38    *   - instances: number of instances
39    *   - records: number of records
40    * An example argument list would be `6 3 50000`.
41    *
42    * If you are using Intellij, the above arguments should be put in the configuration's `Program Arguments`.
43    * Also recommended to set an output log file by `Edit Configuration -> Logs -> Save console
44    * output to file` to record all the log output together.
45    *
46    * The driver could be decomposed as following stages:
47    *
48    * 1. Cleanup any topic whose name conflicts with input and output topic, so that we have a clean-start.
49    *
50    * 2. Set up a producer in a separate thread to pre-populate a set of records with even number keys into
51    *    the input topic. The driver will block for the record generation to finish, so the producer
52    *    must be in synchronous sending mode.
53    *
54    * 3. Set up transactional instances in separate threads which does a consume-process-produce loop,
55    *    tailing data from input topic (See {@link ExactlyOnceMessageProcessor}). Each EOS instance will
56    *    drain all the records from either given partitions or auto assigned partitions by actively
57    *    comparing log end offset with committed offset. Each record will be processed exactly once
58    *    as dividing the key by 2, and extend the value message. The driver will block for all the record
59    *    processing to finish. The transformed record shall be written to the output topic, with
60    *    transactional guarantee.
61    *
62    * 4. Set up a read committed consumer in a separate thread to verify we have all records within
63    *    the output topic, while the message ordering on partition level is maintained.
64    *    The driver will block for the consumption of all committed records.
65    *
66    * From this demo, you could see that all the records from pre-population are processed exactly once,
67    * with strong partition level ordering guarantee.
68    *
69    * Note: please start the kafka broker and zookeeper in local first. The broker version must be >= 2.5
70    * in order to run, otherwise the app could throw
71    * {@link org.apache.kafka.common.errors.UnsupportedVersionException}.
72    */
73   public class KafkaExactlyOnceDemo {
74   
75       private static final String INPUT_TOPIC = "input-topic";
76       private static final String OUTPUT_TOPIC = "output-topic";
77   
78       public static void main(String[] args) throws InterruptedException, ExecutionException {
79           if (args.length != 3) {
80               throw new IllegalArgumentException("Should accept 3 parameters: " +
81                   "[number of partitions], [number of instances], [number of records]");
82           }
83   
84           int numPartitions = Integer.parseInt(args[0]);
85           int numInstances = Integer.parseInt(args[1]);
86           int numRecords = Integer.parseInt(args[2]);
87   
88           /* Stage 1: topic cleanup and recreation */
89           recreateTopics(numPartitions);
90   
91           CountDownLatch prePopulateLatch = new CountDownLatch(1);
92   
93           /* Stage 2: pre-populate records */
94           Producer producerThread = new Producer(INPUT_TOPIC, false, null, true, numRecords, -1, prePopulateLatch);
95           producerThread.start();
96   
97           if (!prePopulateLatch.await(5, TimeUnit.MINUTES)) {
98               throw new TimeoutException("Timeout after 5 minutes waiting for data pre-population");
99           }
100   
101           CountDownLatch transactionalCopyLatch = new CountDownLatch(numInstances);
102   
103           /* Stage 3: transactionally process all messages */
104           for (int instanceIdx = 0; instanceIdx < numInstances; instanceIdx++) {
105               ExactlyOnceMessageProcessor messageProcessor = new ExactlyOnceMessageProcessor(
106                   INPUT_TOPIC, OUTPUT_TOPIC, instanceIdx, transactionalCopyLatch);
107               messageProcessor.start();
108           }
109   
110           if (!transactionalCopyLatch.await(5, TimeUnit.MINUTES)) {
111               throw new TimeoutException("Timeout after 5 minutes waiting for transactionally message copy");
112           }
113   
114           CountDownLatch consumeLatch = new CountDownLatch(1);
115   
116           /* Stage 4: consume all processed messages to verify exactly once */
117           Consumer consumerThread = new Consumer(OUTPUT_TOPIC, "Verify-consumer", Optional.empty(), true, numRecords, consumeLatch);
118           consumerThread.start();
119   
120           if (!consumeLatch.await(5, TimeUnit.MINUTES)) {
121               throw new TimeoutException("Timeout after 5 minutes waiting for output data consumption");
122           }
123   
124           consumerThread.shutdown();
125           System.out.println("All finished!");
126       }
127   
128       private static void recreateTopics(final int numPartitions)
129           throws ExecutionException, InterruptedException {
130           Properties props = new Properties();
131           props.put(ConsumerConfig.BOOTSTRAP_SERVERS_CONFIG,
132               KafkaProperties.KAFKA_SERVER_URL + ":" + KafkaProperties.KAFKA_SERVER_PORT);
133   
134           Admin adminClient = Admin.create(props);
135   
136           List<String> topicsToDelete = Arrays.asList(INPUT_TOPIC, OUTPUT_TOPIC);
137   
138           deleteTopic(adminClient, topicsToDelete);
139   
140           // Check topic existence in a retry loop
141           while (true) {
142               System.out.println("Making sure the topics are deleted successfully: " + topicsToDelete);
143   
144               Set<String> listedTopics = adminClient.listTopics().names().get();
145               System.out.println("Current list of topics: " + listedTopics);
146   
147               boolean hasTopicInfo = false;
148               for (String listedTopic : listedTopics) {
149                   if (topicsToDelete.contains(listedTopic)) {
150                       hasTopicInfo = true;
151                       break;
152                   }
153               }
154               if (!hasTopicInfo) {
155                   break;
156               }
157               Thread.sleep(1000);
158           }
159   
160           // Create topics in a retry loop
161           while (true) {
162               final short replicationFactor = 1;
163               final List<NewTopic> newTopics = Arrays.asList(
164                   new NewTopic(INPUT_TOPIC, numPartitions, replicationFactor),
165                   new NewTopic(OUTPUT_TOPIC, numPartitions, replicationFactor));
166               try {
167                   adminClient.createTopics(newTopics).all().get();
168                   System.out.println("Created new topics: " + newTopics);
169                   break;
170               } catch (ExecutionException e) {
171                   if (!(e.getCause() instanceof TopicExistsException)) {
172                       throw e;
173                   }
174                   System.out.println("Metadata of the old topics are not cleared yet...");
175   
176                   deleteTopic(adminClient, topicsToDelete);
177   
178                   Thread.sleep(1000);
179               }
180           }
181       }
182   
183       private static void deleteTopic(final Admin adminClient, final List<String> topicsToDelete)
184           throws InterruptedException, ExecutionException {
185           try {
186               adminClient.deleteTopics(topicsToDelete).all().get();
187           } catch (ExecutionException e) {
188               if (!(e.getCause() instanceof UnknownTopicOrPartitionException)) {
189                   throw e;
190               }
191               System.out.println("Encountered exception during topic deletion: " + e.getCause());
192           }
193           System.out.println("Deleted old topics: " + topicsToDelete);
194       }
195   }
