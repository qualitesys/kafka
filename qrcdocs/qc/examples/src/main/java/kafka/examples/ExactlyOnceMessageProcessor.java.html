31   import java.time.Duration;
32   import java.util.ArrayList;
33   import java.util.Collection;
34   import java.util.Collections;
35   import java.util.HashMap;

41   /**
42    * A demo class for how to write a customized EOS app. It takes a consume-process-produce loop.
43    * Important configurations and APIs are commented.
44    */
45   public class ExactlyOnceMessageProcessor extends Thread {
46   
47       private static final boolean READ_COMMITTED = true;
48   
49       private final String inputTopic;
50       private final String outputTopic;
51       private final String transactionalId;
52       private final String groupInstanceId;
53   
54       private final KafkaProducer<Integer, String> producer;
55       private final KafkaConsumer<Integer, String> consumer;
56   
57       private final CountDownLatch latch;
58   
59       public ExactlyOnceMessageProcessor(final String inputTopic,
60                                          final String outputTopic,
61                                          final int instanceIdx,
62                                          final CountDownLatch latch) {
63           this.inputTopic = inputTopic;
64           this.outputTopic = outputTopic;
65           this.transactionalId = "Processor-" + instanceIdx;
66           // It is recommended to have a relatively short txn timeout in order to clear pending offsets faster.
67           final int transactionTimeoutMs = 10000;
68           // A unique transactional.id must be provided in order to properly use EOS.
69           producer = new Producer(outputTopic, true, transactionalId, true, -1, transactionTimeoutMs, null).get();
70           // Consumer must be in read_committed mode, which means it won't be able to read uncommitted data.
71           // Consumer could optionally configure groupInstanceId to avoid unnecessary rebalances.
72           this.groupInstanceId = "Txn-consumer-" + instanceIdx;
73           consumer = new Consumer(inputTopic, "Eos-consumer",
74               Optional.of(groupInstanceId), READ_COMMITTED, -1, null).get();
75           this.latch = latch;
76       }
77   
78       @Override
79       public void run() {
80           // Init transactions call should always happen first in order to clear zombie transactions from previous generation.
81           producer.initTransactions();
82   
83           final AtomicLong messageRemaining = new AtomicLong(Long.MAX_VALUE);
84   
85           consumer.subscribe(Collections.singleton(inputTopic), new ConsumerRebalanceListener() {
86               @Override
87               public void onPartitionsRevoked(Collection<TopicPartition> partitions) {
88                   printWithTxnId("Revoked partition assignment to kick-off rebalancing: " + partitions);
89               }
90   
91               @Override
92               public void onPartitionsAssigned(Collection<TopicPartition> partitions) {
93                   printWithTxnId("Received partition assignment after rebalancing: " + partitions);
94                   messageRemaining.set(messagesRemaining(consumer));
95               }
96           });
97   
98           int messageProcessed = 0;
99           while (messageRemaining.get() > 0) {
100               try {
101                   ConsumerRecords<Integer, String> records = consumer.poll(Duration.ofMillis(200));
102                   if (records.count() > 0) {
103                       // Begin a new transaction session.
104                       producer.beginTransaction();
105                       for (ConsumerRecord<Integer, String> record : records) {
106                           // Process the record and send to downstream.
107                           ProducerRecord<Integer, String> customizedRecord = transform(record);
108                           producer.send(customizedRecord);
109                       }
110   
111                       Map<TopicPartition, OffsetAndMetadata> offsets = consumerOffsets();
112   
113                       // Checkpoint the progress by sending offsets to group coordinator broker.
114                       // Note that this API is only available for broker >= 2.5.
115                       producer.sendOffsetsToTransaction(offsets, consumer.groupMetadata());
116   
117                       // Finish the transaction. All sent records should be visible for consumption now.
118                       producer.commitTransaction();
119                       messageProcessed += records.count();
120                   }
121               } catch (ProducerFencedException e) {
122                   throw new KafkaException(String.format("The transactional.id %s has been claimed by another process", transactionalId));
123               } catch (FencedInstanceIdException e) {
124                   throw new KafkaException(String.format("The group.instance.id %s has been claimed by another process", groupInstanceId));
125               } catch (KafkaException e) {
126                   // If we have not been fenced, try to abort the transaction and continue. This will raise immediately
127                   // if the producer has hit a fatal error.
128                   producer.abortTransaction();
129   
130                   // The consumer fetch position needs to be restored to the committed offset
131                   // before the transaction started.
132                   resetToLastCommittedPositions(consumer);
133               }
134   
135               messageRemaining.set(messagesRemaining(consumer));
136               printWithTxnId("Message remaining: " + messageRemaining);
137           }
138   
139           printWithTxnId("Finished processing " + messageProcessed + " records");
140           latch.countDown();
141       }
142   
143       private Map<TopicPartition, OffsetAndMetadata> consumerOffsets() {
144           Map<TopicPartition, OffsetAndMetadata> offsets = new HashMap<>();
145           for (TopicPartition topicPartition : consumer.assignment()) {
146               offsets.put(topicPartition, new OffsetAndMetadata(consumer.position(topicPartition), null));
147           }
148           return offsets;
149       }
150   
151       private void printWithTxnId(final String message) {
152           System.out.println(transactionalId + ": " + message);
153       }
154   
155       private ProducerRecord<Integer, String> transform(final ConsumerRecord<Integer, String> record) {
156           printWithTxnId("Transformed record (" + record.key() + "," + record.value() + ")");
157           return new ProducerRecord<>(outputTopic, record.key() / 2, "Transformed_" + record.value());
158       }
159   
160       private long messagesRemaining(final KafkaConsumer<Integer, String> consumer) {
161           final Map<TopicPartition, Long> fullEndOffsets = consumer.endOffsets(new ArrayList<>(consumer.assignment()));
162           // If we couldn't detect any end offset, that means we are still not able to fetch offsets.
163           if (fullEndOffsets.isEmpty()) {
164               return Long.MAX_VALUE;
165           }
166   
167           return consumer.assignment().stream().mapToLong(partition -> {
168               long currentPosition = consumer.position(partition);
169               printWithTxnId("Processing partition " + partition + " with full offsets " + fullEndOffsets);
170               if (fullEndOffsets.containsKey(partition)) {
171                   return fullEndOffsets.get(partition) - currentPosition;
172               }
173               return 0;
174           }).sum();
175       }
176   
177       private static void resetToLastCommittedPositions(KafkaConsumer<Integer, String> consumer) {
178           final Map<TopicPartition, OffsetAndMetadata> committed = consumer.committed(consumer.assignment());
179           consumer.assignment().forEach(tp -> {
180               OffsetAndMetadata offsetAndMetadata = committed.get(tp);
181               if (offsetAndMetadata != null)
182                   consumer.seek(tp, offsetAndMetadata.offset());
183               else
184                   consumer.seekToBeginning(Collections.singleton(tp));
185           });
186       }
187   }
