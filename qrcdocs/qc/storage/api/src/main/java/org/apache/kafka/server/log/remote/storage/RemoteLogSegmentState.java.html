17   package org.apache.kafka.server.log.remote.storage;
18   
19   import org.apache.kafka.common.annotation.InterfaceStability;
20   
21   import java.util.Arrays;
22   import java.util.Collections;
23   import java.util.Map;
24   import java.util.Objects;
25   import java.util.function.Function;
26   import java.util.stream.Collectors;
27   
28   /**
29    * This enum indicates the state of the remote log segment. This will be based on the action executed on this
30    * segment by the remote log service implementation.
31    * <p>
32    * It goes through the below state transitions. Self transition is treated as valid. This allows updating with the
33    * same state in case of retries and failover.
34    * <p>
35    * <pre>
36    * +---------------------+            +----------------------+
37    * |COPY_SEGMENT_STARTED |----------&gt; |COPY_SEGMENT_FINISHED |
38    * +-------------------+-+            +--+-------------------+
39    *                     |                 |
40    *                     |                 |
41    *                     v                 v
42    *                  +--+-----------------+-+
43    *                  |DELETE_SEGMENT_STARTED|
44    *                  +-----------+----------+
45    *                              |
46    *                              |
47    *                              v
48    *                  +-----------+-----------+
49    *                  |DELETE_SEGMENT_FINISHED|
50    *                  +-----------------------+
51    * </pre>
52    */
53   @InterfaceStability.Evolving
54   public enum RemoteLogSegmentState {
55   
56       /**
57        * This state indicates that the segment copying to remote storage is started but not yet finished.
58        */
59       COPY_SEGMENT_STARTED((byte) 0),
60   
61       /**
62        * This state indicates that the segment copying to remote storage is finished.
63        */
64       COPY_SEGMENT_FINISHED((byte) 1),
65   
66       /**
67        * This state indicates that the segment deletion is started but not yet finished.
68        */
69       DELETE_SEGMENT_STARTED((byte) 2),
70   
71       /**
72        * This state indicates that the segment is deleted successfully.
73        */
74       DELETE_SEGMENT_FINISHED((byte) 3);
75   
76       private static final Map<Byte, RemoteLogSegmentState> STATE_TYPES = Collections.unmodifiableMap(
77               Arrays.stream(values()).collect(Collectors.toMap(RemoteLogSegmentState::id, Function.identity())));
78   
79       private final byte id;
80   
81       RemoteLogSegmentState(byte id) {
82           this.id = id;
83       }
84   
85       public byte id() {
86           return id;
87       }
88   
89       public static RemoteLogSegmentState forId(byte id) {
90           return STATE_TYPES.get(id);
91       }
92   
93       public static boolean isValidTransition(RemoteLogSegmentState srcState, RemoteLogSegmentState targetState) {
94           Objects.requireNonNull(targetState, "targetState can not be null");
95   
96           if (srcState == null) {
97               // If the source state is null, check the target state as the initial state viz COPY_SEGMENT_STARTED
98               // This ensures simplicity here as we don't have to define one more type to represent the state 'null' like
99               // COPY_SEGMENT_NOT_STARTED, have the null check by the caller and pass that state.
100               return targetState == COPY_SEGMENT_STARTED;
101           } else if (srcState == targetState) {
102               // Self transition is treated as valid. This is to maintain the idempotency for the state in case of retries
103               // or failover.
104               return true;
105           } else if (srcState == COPY_SEGMENT_STARTED) {
106               return targetState == COPY_SEGMENT_FINISHED || targetState == DELETE_SEGMENT_STARTED;
107           } else if (srcState == COPY_SEGMENT_FINISHED) {
108               return targetState == DELETE_SEGMENT_STARTED;
109           } else if (srcState == DELETE_SEGMENT_STARTED) {
110               return targetState == DELETE_SEGMENT_FINISHED;
111           } else {
112               return false;
113           }
114       }
115   }
