37   /**
38    * This class manages the consumer thread viz {@link ConsumerTask} that polls messages from the assigned metadata topic partitions.
39    * It also provides a way to wait until the given record is received by the consumer before it is timed out with an interval of
40    * {@link TopicBasedRemoteLogMetadataManagerConfig#consumeWaitMs()}.
41    */
42   public class ConsumerManager implements Closeable {
43   
44       public static final String COMMITTED_OFFSETS_FILE_NAME = "_rlmm_committed_offsets";
45   
46       private static final Logger log = LoggerFactory.getLogger(ConsumerManager.class);
47       private static final long CONSUME_RECHECK_INTERVAL_MS = 50L;
48   
49       private final TopicBasedRemoteLogMetadataManagerConfig rlmmConfig;
50       private final Time time;
51       private final ConsumerTask consumerTask;
52       private final Thread consumerTaskThread;
53   
54       public ConsumerManager(TopicBasedRemoteLogMetadataManagerConfig rlmmConfig,
55                              RemotePartitionMetadataEventHandler remotePartitionMetadataEventHandler,
56                              RemoteLogMetadataTopicPartitioner topicPartitioner,
57                              Time time) {
58           this.rlmmConfig = rlmmConfig;
59           this.time = time;
60   
61           //Create a task to consume messages and submit the respective events to RemotePartitionMetadataEventHandler.
62           KafkaConsumer<byte[], byte[]> consumer = new KafkaConsumer<>(rlmmConfig.consumerProperties());
63           Path committedOffsetsPath = new File(rlmmConfig.logDir(), COMMITTED_OFFSETS_FILE_NAME).toPath();
64           consumerTask = new ConsumerTask(consumer, remotePartitionMetadataEventHandler, topicPartitioner, committedOffsetsPath, time, 60_000L);
65           consumerTaskThread = KafkaThread.nonDaemon("RLMMConsumerTask", consumerTask);
66       }
67   
68       public void startConsumerThread() {
69           try {
70               // Start a thread to continuously consume records from topic partitions.
71               consumerTaskThread.start();
72               log.info("RLMM Consumer task thread is started");
73           } catch (Exception e) {
74               throw new KafkaException("Error encountered while initializing and scheduling ConsumerTask thread", e);
75           }
76       }
77   
78       /**
79        * Waits if necessary for the consumption to reach the offset of the given {@code recordMetadata}.
80        *
81        * @param recordMetadata record metadata to be checked for consumption.
82        * @throws TimeoutException if this method execution did not complete with in the wait time configured with
83        *                          property {@code TopicBasedRemoteLogMetadataManagerConfig#REMOTE_LOG_METADATA_CONSUME_WAIT_MS_PROP}.
84        */
85       public void waitTillConsumptionCatchesUp(RecordMetadata recordMetadata) throws TimeoutException {
86           waitTillConsumptionCatchesUp(recordMetadata, rlmmConfig.consumeWaitMs());
87       }
88   
89       /**
90        * Waits if necessary for the consumption to reach the offset of the given {@code recordMetadata}.
91        *
92        * @param recordMetadata record metadata to be checked for consumption.
93        * @param timeoutMs      wait timeout in milli seconds
94        * @throws TimeoutException if this method execution did not complete with in the given {@code timeoutMs}.
95        */
96       public void waitTillConsumptionCatchesUp(RecordMetadata recordMetadata,
97                                                long timeoutMs) throws TimeoutException {
98           final int partition = recordMetadata.partition();
99           final long consumeCheckIntervalMs = Math.min(CONSUME_RECHECK_INTERVAL_MS, timeoutMs);
100   
101           // If the current assignment does not have the subscription for this partition then return immediately.
102           if (!consumerTask.isPartitionAssigned(partition)) {
103               throw new KafkaException("This consumer is not subscribed to the target partition " + partition + " on which message is produced.");
104           }
105   
106           final long offset = recordMetadata.offset();
107           long startTimeMs = time.milliseconds();
108           while (true) {
109               long receivedOffset = consumerTask.receivedOffsetForPartition(partition).orElse(-1L);
110               if (receivedOffset >= offset) {
111                   return;
112               }
113   
114               log.debug("Committed offset [{}] for partition [{}], but the target offset: [{}],  Sleeping for [{}] to retry again",
115                         offset, partition, receivedOffset, consumeCheckIntervalMs);
116   
117               if (time.milliseconds() - startTimeMs > timeoutMs) {
118                   log.warn("Committed offset for partition:[{}] is : [{}], but the target offset: [{}] ",
119                            partition, receivedOffset, offset);
120                   throw new TimeoutException("Timed out in catching up with the expected offset by consumer.");
121               }
122   
123               time.sleep(consumeCheckIntervalMs);
124           }
125       }
126   
127       @Override
128       public void close() throws IOException {
129           // Consumer task will close the task and it internally closes all the resources including the consumer.
130           Utils.closeQuietly(consumerTask, "ConsumerTask");
131   
132           // Wait until the consumer thread finishes.
133           try {
134               consumerTaskThread.join();
135           } catch (Exception e) {
136               log.error("Encountered error while waiting for consumerTaskThread to finish.", e);
137           }
138       }
139   
140       public void addAssignmentsForPartitions(Set<TopicIdPartition> partitions) {
141           consumerTask.addAssignmentsForPartitions(partitions);
142       }
143   
144       public void removeAssignmentsForPartitions(Set<TopicIdPartition> partitions) {
145           consumerTask.removeAssignmentsForPartitions(partitions);
146       }
147   
148       public Optional<Long> receivedOffsetForPartition(int metadataPartition) {
149           return consumerTask.receivedOffsetForPartition(metadataPartition);
150       }
151   }
