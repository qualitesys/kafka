36   /**
37    * This class provides an in-memory cache of remote log segment metadata. This maintains the lineage of segments
38    * with respect to leader epochs.
39    * <p>
40    * Remote log segment can go through the state transitions as mentioned in {@link RemoteLogSegmentState}.
41    * <p>
42    * This class will have all the segments which did not reach terminal state viz DELETE_SEGMENT_FINISHED. That means,any
43    * segment reaching the terminal state will get cleared from this instance.
44    * This class provides different methods to fetch segment metadata like {@link #remoteLogSegmentMetadata(int, long)},
45    * {@link #highestOffsetForEpoch(int)}, {@link #listRemoteLogSegments(int)}, {@link #listAllRemoteLogSegments()}. Those
46    * methods have different semantics to fetch the segment based on its state.
47    * <p>
48    * <ul>
49    * <li>
50    * {@link RemoteLogSegmentState#COPY_SEGMENT_STARTED}:
51    * <br>
52    * Segment in this state indicates it is not yet copied successfully. So, these segments will not be
53    * accessible for reads but these are considered for cleanups when a partition is deleted.
54    * </li>
55    * <li>
56    * {@link RemoteLogSegmentState#COPY_SEGMENT_FINISHED}:
57    * <br>
58    * Segment in this state indicates it is successfully copied and it is available for reads. So, these segments
59    * will be accessible for reads. But this should be available for any cleanup activity like deleting segments by the
60    * caller of this class.
61    * </li>
62    * <li>
63    * {@link RemoteLogSegmentState#DELETE_SEGMENT_STARTED}:
64    * Segment in this state indicates it is getting deleted. That means, it is not available for reads. But it should be
65    * available for any cleanup activity like deleting segments by the caller of this class.
66    * </li>
67    * <li>
68    * {@link RemoteLogSegmentState#DELETE_SEGMENT_FINISHED}:
69    * Segment in this state indicate it is already deleted. That means, it is not available for any activity including
70    * reads or cleanup activity. This cache will clear entries containing this state.
71    * </li>
72    * </ul>
73    *
74    * <p>
75    *  The below table summarizes whether the segment with the respective state are available for the given methods.
76    * <pre>
77    * +---------------------------------+----------------------+------------------------+-------------------------+-------------------------+
78    * |  Method / SegmentState          | COPY_SEGMENT_STARTED | COPY_SEGMENT_FINISHED  | DELETE_SEGMENT_STARTED  | DELETE_SEGMENT_FINISHED |
79    * |---------------------------------+----------------------+------------------------+-------------------------+-------------------------|
80    * | remoteLogSegmentMetadata        |        No            |           Yes          |          No             |           No            |
81    * | (int leaderEpoch, long offset)  |                      |                        |                         |                         |
82    * |---------------------------------+----------------------+------------------------+-------------------------+-------------------------|
83    * | listRemoteLogSegments           |        Yes           |           Yes          |          Yes            |           No            |
84    * | (int leaderEpoch)               |                      |                        |                         |                         |
85    * |---------------------------------+----------------------+------------------------+-------------------------+-------------------------|
86    * | listAllRemoteLogSegments()      |        Yes           |           Yes          |          Yes            |           No            |
87    * |                                 |                      |                        |                         |                         |
88    * +---------------------------------+----------------------+------------------------+-------------------------+-------------------------+
89    * </pre>
90    * </p>
91    * <p></p>
92    */
93   public class RemoteLogMetadataCache {
94   
95       private static final Logger log = LoggerFactory.getLogger(RemoteLogMetadataCache.class);
96   
97       // It contains all the segment-id to metadata mappings which did not reach the terminal state viz DELETE_SEGMENT_FINISHED.
98       protected final ConcurrentMap<RemoteLogSegmentId, RemoteLogSegmentMetadata> idToSegmentMetadata
99               = new ConcurrentHashMap<>();
100   
101       // It contains leader epoch to the respective entry containing the state.
102       // TODO We are not clearing the entry for epoch when RemoteLogLeaderEpochState becomes empty. This will be addressed
103       // later. We will look into it when we integrate these APIs along with RemoteLogManager changes.
104       // https://issues.apache.org/jira/browse/KAFKA-12641
105       protected final ConcurrentMap<Integer, RemoteLogLeaderEpochState> leaderEpochEntries = new ConcurrentHashMap<>();
106   
107       /**
108        * Returns {@link RemoteLogSegmentMetadata} if it exists for the given leader-epoch containing the offset and with
109        * {@link RemoteLogSegmentState#COPY_SEGMENT_FINISHED} state, else returns {@link Optional#empty()}.
110        *
111        * @param leaderEpoch leader epoch for the given offset
112        * @param offset      offset
113        * @return the requested remote log segment metadata if it exists.
114        */
115       public Optional<RemoteLogSegmentMetadata> remoteLogSegmentMetadata(int leaderEpoch, long offset) {
116           RemoteLogLeaderEpochState remoteLogLeaderEpochState = leaderEpochEntries.get(leaderEpoch);
117   
118           if (remoteLogLeaderEpochState == null) {
119               return Optional.empty();
120           }
121   
122           // Look for floor entry as the given offset may exist in this entry.
123           RemoteLogSegmentId remoteLogSegmentId = remoteLogLeaderEpochState.floorEntry(offset);
124           if (remoteLogSegmentId == null) {
125               // If the offset is lower than the minimum offset available in metadata then return empty.
126               return Optional.empty();
127           }
128   
129           RemoteLogSegmentMetadata metadata = idToSegmentMetadata.get(remoteLogSegmentId);
130           // Check whether the given offset with leaderEpoch exists in this segment.
131           // Check for epoch's offset boundaries with in this segment.
132           //      1. Get the next epoch's start offset -1 if exists
133           //      2. If no next epoch exists, then segment end offset can be considered as epoch's relative end offset.
134           Map.Entry<Integer, Long> nextEntry = metadata.segmentLeaderEpochs().higherEntry(leaderEpoch);
135           long epochEndOffset = (nextEntry != null) ? nextEntry.getValue() - 1 : metadata.endOffset();
136   
137           // Return empty when target offset > epoch's end offset.
138           return offset > epochEndOffset ? Optional.empty() : Optional.of(metadata);
139       }

141       public void updateRemoteLogSegmentMetadata(RemoteLogSegmentMetadataUpdate metadataUpdate)
142               throws RemoteResourceNotFoundException {
143           log.debug("Updating remote log segment metadata: [{}]", metadataUpdate);
144           Objects.requireNonNull(metadataUpdate, "metadataUpdate can not be null");
145   
146           RemoteLogSegmentState targetState = metadataUpdate.state();
147           RemoteLogSegmentId remoteLogSegmentId = metadataUpdate.remoteLogSegmentId();
148           RemoteLogSegmentMetadata existingMetadata = idToSegmentMetadata.get(remoteLogSegmentId);
149           if (existingMetadata == null) {
150               throw new RemoteResourceNotFoundException("No remote log segment metadata found for :" +
151                                                         remoteLogSegmentId);
152           }
153   
154           // Check the state transition.
155           checkStateTransition(existingMetadata.state(), targetState);
156   
157           switch (targetState) {
158               case COPY_SEGMENT_STARTED:
159                   // Callers should use addCopyInProgressSegment to add RemoteLogSegmentMetadata with state as
160                   // RemoteLogSegmentState.COPY_SEGMENT_STARTED.
161                   throw new IllegalArgumentException("metadataUpdate: " + metadataUpdate + " with state " + RemoteLogSegmentState.COPY_SEGMENT_STARTED +
162                                                      " can not be updated");
163               case COPY_SEGMENT_FINISHED:
164                   handleSegmentWithCopySegmentFinishedState(existingMetadata.createWithUpdates(metadataUpdate));
165                   break;
166               case DELETE_SEGMENT_STARTED:
167                   handleSegmentWithDeleteSegmentStartedState(existingMetadata.createWithUpdates(metadataUpdate));
168                   break;
169               case DELETE_SEGMENT_FINISHED:
170                   handleSegmentWithDeleteSegmentFinishedState(existingMetadata.createWithUpdates(metadataUpdate));
171                   break;
172               default:
173                   throw new IllegalArgumentException("Metadata with the state " + targetState + " is not supported");
174           }
175       }
176   
177       protected final void handleSegmentWithCopySegmentFinishedState(RemoteLogSegmentMetadata remoteLogSegmentMetadata) {
178           doHandleSegmentStateTransitionForLeaderEpochs(remoteLogSegmentMetadata,
179                                                         (leaderEpoch, remoteLogLeaderEpochState, startOffset, segmentId) -> {
180                                                             long leaderEpochEndOffset = highestOffsetForEpoch(leaderEpoch,
181                                                                                                               remoteLogSegmentMetadata);
182                                                             remoteLogLeaderEpochState.handleSegmentWithCopySegmentFinishedState(startOffset,
183                                                                                                                                 segmentId,
184                                                                                                                                 leaderEpochEndOffset);
185                                                         });
186   
187           // Put the entry with the updated metadata.
188           idToSegmentMetadata.put(remoteLogSegmentMetadata.remoteLogSegmentId(), remoteLogSegmentMetadata);
189       }
190   
191       protected final void handleSegmentWithDeleteSegmentStartedState(RemoteLogSegmentMetadata remoteLogSegmentMetadata) {
192           log.debug("Cleaning up the state for : [{}]", remoteLogSegmentMetadata);
193   
194           doHandleSegmentStateTransitionForLeaderEpochs(remoteLogSegmentMetadata,
195                                                         (leaderEpoch, remoteLogLeaderEpochState, startOffset, segmentId) ->
196                                                                 remoteLogLeaderEpochState.handleSegmentWithDeleteSegmentStartedState(startOffset, segmentId));
197   
198           // Put the entry with the updated metadata.
199           idToSegmentMetadata.put(remoteLogSegmentMetadata.remoteLogSegmentId(), remoteLogSegmentMetadata);
200       }
201   
202       private void handleSegmentWithDeleteSegmentFinishedState(RemoteLogSegmentMetadata remoteLogSegmentMetadata) {
203           log.debug("Removing the entry as it reached the terminal state: [{}]", remoteLogSegmentMetadata);
204   
205           doHandleSegmentStateTransitionForLeaderEpochs(remoteLogSegmentMetadata,
206                                                         (leaderEpoch, remoteLogLeaderEpochState, startOffset, segmentId) ->
207                                                                 remoteLogLeaderEpochState.handleSegmentWithDeleteSegmentFinishedState(segmentId));
208   
209           // Remove the segment's id to metadata mapping because this segment is considered as deleted and it cleared all
210           // the state of this segment in the cache.
211           idToSegmentMetadata.remove(remoteLogSegmentMetadata.remoteLogSegmentId());
212       }
213   
214       private void doHandleSegmentStateTransitionForLeaderEpochs(RemoteLogSegmentMetadata remoteLogSegmentMetadata,
215                                                                  RemoteLogLeaderEpochState.Action action) {
216           RemoteLogSegmentId remoteLogSegmentId = remoteLogSegmentMetadata.remoteLogSegmentId();
217           Map<Integer, Long> leaderEpochToOffset = remoteLogSegmentMetadata.segmentLeaderEpochs();
218   
219           // Go through all the leader epochs and apply the given action.
220           for (Map.Entry<Integer, Long> entry : leaderEpochToOffset.entrySet()) {
221               Integer leaderEpoch = entry.getKey();
222               Long startOffset = entry.getValue();
223               // leaderEpochEntries will be empty when resorting the metadata from snapshot.
224               RemoteLogLeaderEpochState remoteLogLeaderEpochState = leaderEpochEntries.computeIfAbsent(
225                       leaderEpoch, x -> new RemoteLogLeaderEpochState());
226               action.accept(leaderEpoch, remoteLogLeaderEpochState, startOffset, remoteLogSegmentId);
227           }
228       }

230       private static long highestOffsetForEpoch(Integer leaderEpoch, RemoteLogSegmentMetadata segmentMetadata) {
231           // Compute the highest offset for the leader epoch with in the segment
232           NavigableMap<Integer, Long> epochToOffset = segmentMetadata.segmentLeaderEpochs();
233           Map.Entry<Integer, Long> nextEntry = epochToOffset.higherEntry(leaderEpoch);
234   
235           return nextEntry != null ? nextEntry.getValue() - 1 : segmentMetadata.endOffset();
236       }

238       /**
239        * Returns all the segments stored in this cache.
240        *
241        * @return
242        */
243       public Iterator<RemoteLogSegmentMetadata> listAllRemoteLogSegments() {
244           // Return all the segments including unreferenced metadata.
245           return Collections.unmodifiableCollection(idToSegmentMetadata.values()).iterator();
246       }

248       /**
249        * Returns all the segments mapped to the leader epoch that exist in this cache sorted by {@link RemoteLogSegmentMetadata#startOffset()}.
250        *
251        * @param leaderEpoch leader epoch.
252        */
253       public Iterator<RemoteLogSegmentMetadata> listRemoteLogSegments(int leaderEpoch)
254               throws RemoteResourceNotFoundException {
255           RemoteLogLeaderEpochState remoteLogLeaderEpochState = leaderEpochEntries.get(leaderEpoch);
256           if (remoteLogLeaderEpochState == null) {
257               return Collections.emptyIterator();
258           }
259   
260           return remoteLogLeaderEpochState.listAllRemoteLogSegments(idToSegmentMetadata);
261       }
262   
263       /**
264        * Returns the highest offset of a segment for the given leader epoch if exists, else it returns empty. The segments
265        * that have reached the {@link RemoteLogSegmentState#COPY_SEGMENT_FINISHED} or later states are considered here.
266        *
267        * @param leaderEpoch leader epoch
268        */
269       public Optional<Long> highestOffsetForEpoch(int leaderEpoch) {
270           RemoteLogLeaderEpochState entry = leaderEpochEntries.get(leaderEpoch);
271           return entry != null ? Optional.ofNullable(entry.highestLogOffset()) : Optional.empty();
272       }
273   
274       /**
275        * This method tracks the given remote segment as not yet available for reads. It does not add the segment
276        * leader epoch offset mapping until this segment reaches COPY_SEGMENT_FINISHED state.
277        *
278        * @param remoteLogSegmentMetadata RemoteLogSegmentMetadata instance
279        */
280       public void addCopyInProgressSegment(RemoteLogSegmentMetadata remoteLogSegmentMetadata) {
281           log.debug("Adding to in-progress state: [{}]", remoteLogSegmentMetadata);
282           Objects.requireNonNull(remoteLogSegmentMetadata, "remoteLogSegmentMetadata can not be null");
283   
284           // This method is allowed only to add remote log segment with the initial state(which is RemoteLogSegmentState.COPY_SEGMENT_STARTED)
285           // but not to update the existing remote log segment metadata.
286           if (remoteLogSegmentMetadata.state() != RemoteLogSegmentState.COPY_SEGMENT_STARTED) {
287               throw new IllegalArgumentException(
288                       "Given remoteLogSegmentMetadata:" + remoteLogSegmentMetadata + " should have state as " + RemoteLogSegmentState.COPY_SEGMENT_STARTED
289                       + " but it contains state as: " + remoteLogSegmentMetadata.state());
290           }
291   
292           RemoteLogSegmentId remoteLogSegmentId = remoteLogSegmentMetadata.remoteLogSegmentId();
293           RemoteLogSegmentMetadata existingMetadata = idToSegmentMetadata.get(remoteLogSegmentId);
294           checkStateTransition(existingMetadata != null ? existingMetadata.state() : null,
295                   remoteLogSegmentMetadata.state());
296   
297           for (Integer epoch : remoteLogSegmentMetadata.segmentLeaderEpochs().keySet()) {
298               leaderEpochEntries.computeIfAbsent(epoch, leaderEpoch -> new RemoteLogLeaderEpochState())
299                       .handleSegmentWithCopySegmentStartedState(remoteLogSegmentId);
300           }
301   
302           idToSegmentMetadata.put(remoteLogSegmentId, remoteLogSegmentMetadata);
303       }
304   
305       private void checkStateTransition(RemoteLogSegmentState existingState, RemoteLogSegmentState targetState) {
306           if (!RemoteLogSegmentState.isValidTransition(existingState, targetState)) {
307               throw new IllegalStateException(
308                       "Current state: " + existingState + " can not be transitioned to target state: " + targetState);
309           }
310       }
