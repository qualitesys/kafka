27   import java.util.Iterator;
28   import java.util.Map;
29   import java.util.NavigableMap;

31   import java.util.concurrent.ConcurrentHashMap;
32   import java.util.concurrent.ConcurrentSkipListMap;

34   /**
35    * This class represents the in-memory state of segments associated with a leader epoch. This includes the mapping of offset to
36    * segment ids and unreferenced segments which are not mapped to any offset but they exist in remote storage.
37    * <p>
38    * This is used by {@link RemoteLogMetadataCache} to track the segments for each leader epoch.
39    */
40   class RemoteLogLeaderEpochState {
41   
42       // It contains offset to segment ids mapping with the segment state as COPY_SEGMENT_FINISHED.
43       private final NavigableMap<Long, RemoteLogSegmentId> offsetToId = new ConcurrentSkipListMap<>();
44   
45       /**
46        * It represents unreferenced segments for this leader epoch. It contains the segments still in COPY_SEGMENT_STARTED
47        * and DELETE_SEGMENT_STARTED state or these have been replaced by callers with other segments having the same
48        * start offset for the leader epoch. These will be returned by {@link RemoteLogMetadataCache#listAllRemoteLogSegments()}
49        * and {@link RemoteLogMetadataCache#listRemoteLogSegments(int leaderEpoch)} so that callers can clean them up if
50        * they still exist. These will be cleaned from the cache once they reach DELETE_SEGMENT_FINISHED state.
51        */
52       private final Set<RemoteLogSegmentId> unreferencedSegmentIds = ConcurrentHashMap.newKeySet();
53   
54       // It represents the highest log offset of the segments that reached the COPY_SEGMENT_FINISHED state.
55       private volatile Long highestLogOffset;
56   
57       /**
58        * Returns all the segments associated with this leader epoch sorted by start offset in ascending order.
59        *
60        * @param idToSegmentMetadata mapping of id to segment metadata. This will be used to get RemoteLogSegmentMetadata
61        *                            for an id to be used for sorting.
62        */
63       Iterator<RemoteLogSegmentMetadata> listAllRemoteLogSegments(Map<RemoteLogSegmentId, RemoteLogSegmentMetadata> idToSegmentMetadata)
64               throws RemoteResourceNotFoundException {
65           // Return all the segments including unreferenced metadata.
66           int size = offsetToId.size() + unreferencedSegmentIds.size();
67           if (size == 0) {
68               return Collections.emptyIterator();
69           }
70   
71           ArrayList<RemoteLogSegmentMetadata> metadataList = new ArrayList<>(size);
72           collectConvertedIdToMetadata(offsetToId.values(), idToSegmentMetadata, metadataList);
73   
74           if (!unreferencedSegmentIds.isEmpty()) {
75               collectConvertedIdToMetadata(unreferencedSegmentIds, idToSegmentMetadata, metadataList);
76   
77               // Sort only when unreferenced entries exist as they are already sorted in offsetToId.
78               metadataList.sort(Comparator.comparingLong(RemoteLogSegmentMetadata::startOffset));
79           }
80   
81           return metadataList.iterator();
82       }
83   
84       private void collectConvertedIdToMetadata(Collection<RemoteLogSegmentId> segmentIds,
85                                                 Map<RemoteLogSegmentId, RemoteLogSegmentMetadata> idToSegmentMetadata,
86                                                 Collection<RemoteLogSegmentMetadata> result) throws RemoteResourceNotFoundException {
87           for (RemoteLogSegmentId id : segmentIds) {
88               RemoteLogSegmentMetadata metadata = idToSegmentMetadata.get(id);
89               if (metadata == null) {
90                   throw new RemoteResourceNotFoundException("No remote log segment metadata found for :" + id);
91               }
92               result.add(metadata);
93           }
94       }
95   
96       void handleSegmentWithCopySegmentStartedState(RemoteLogSegmentId remoteLogSegmentId) {
97           // Add this to unreferenced set of segments for the respective leader epoch.
98           unreferencedSegmentIds.add(remoteLogSegmentId);
99       }
100   
101       void handleSegmentWithCopySegmentFinishedState(Long startOffset, RemoteLogSegmentId remoteLogSegmentId,
102                                                      Long leaderEpochEndOffset) {
103           // Add the segment epochs mapping as the segment is copied successfully.
104           RemoteLogSegmentId oldEntry = offsetToId.put(startOffset, remoteLogSegmentId);
105   
106           // Remove the metadata from unreferenced entries as it is successfully copied and added to the offset mapping.
107           unreferencedSegmentIds.remove(remoteLogSegmentId);
108   
109           // Add the old entry to unreferenced entries as the mapping is removed for the old entry.
110           if (oldEntry != null) {
111               unreferencedSegmentIds.add(oldEntry);
112           }
113   
114           // Update the highest offset entry for this leader epoch as we added a new mapping.
115           if (highestLogOffset == null || leaderEpochEndOffset > highestLogOffset) {
116               highestLogOffset = leaderEpochEndOffset;
117           }
118       }
119   
120       void handleSegmentWithDeleteSegmentStartedState(Long startOffset, RemoteLogSegmentId remoteLogSegmentId) {
121           // Remove the offset mappings as this segment is getting deleted.
122           offsetToId.remove(startOffset, remoteLogSegmentId);
123   
124           // Add this entry to unreferenced set for the leader epoch as it is being deleted.
125           // This allows any retries of deletion as these are returned from listAllSegments and listSegments(leaderEpoch).
126           unreferencedSegmentIds.add(remoteLogSegmentId);
127       }
128   
129       void handleSegmentWithDeleteSegmentFinishedState(RemoteLogSegmentId remoteLogSegmentId) {
130           // It completely removes the tracking of this segment as it is considered as deleted.
131           unreferencedSegmentIds.remove(remoteLogSegmentId);
132       }
133   
134       Long highestLogOffset() {
135           return highestLogOffset;
136       }
137   
138       /**
139        * Returns the RemoteLogSegmentId of a segment for the given offset, if there exists a mapping associated with
140        * the greatest offset less than or equal to the given offset, or null if there is no such mapping.
141        *
142        * @param offset offset
143        */
144       RemoteLogSegmentId floorEntry(long offset) {
145           Map.Entry<Long, RemoteLogSegmentId> entry = offsetToId.floorEntry(offset);
146   
147           return entry == null ? null : entry.getValue();
148       }
149   
150       Collection<RemoteLogSegmentId> unreferencedSegmentIds() {
151           return Collections.unmodifiableCollection(unreferencedSegmentIds);
152       }
153   
154       Collection<RemoteLogSegmentId> referencedSegmentIds() {
155           return Collections.unmodifiableCollection(offsetToId.values());
156       }

158       /**
159        * Action interface to act on remote log segment transition for the given {@link RemoteLogLeaderEpochState}.
160        */
161       @FunctionalInterface
162       interface Action {
163   
164           /**
165            * Performs this operation with the given {@code remoteLogLeaderEpochState}.
166            *
167            * @param leaderEpoch               leader epoch value
168            * @param remoteLogLeaderEpochState In-memory state of the segments for a leader epoch.
169            * @param startOffset               start offset of the segment.
170            * @param segmentId                 segment id.
171            */
172           void accept(int leaderEpoch,
173                       RemoteLogLeaderEpochState remoteLogLeaderEpochState,
174                       long startOffset,
175                       RemoteLogSegmentId segmentId);
176       }
