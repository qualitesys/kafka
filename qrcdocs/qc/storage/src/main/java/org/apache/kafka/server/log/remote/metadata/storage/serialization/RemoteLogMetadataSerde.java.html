17   package org.apache.kafka.server.log.remote.metadata.storage.serialization;
18   
19   import org.apache.kafka.common.protocol.ApiMessage;
20   import org.apache.kafka.server.common.ApiMessageAndVersion;
21   import org.apache.kafka.server.common.serialization.BytesApiMessageSerde;
22   import org.apache.kafka.server.log.remote.metadata.storage.RemoteLogSegmentMetadataSnapshot;
23   import org.apache.kafka.server.log.remote.metadata.storage.generated.MetadataRecordType;
24   import org.apache.kafka.server.log.remote.metadata.storage.generated.RemoteLogSegmentMetadataRecord;
25   import org.apache.kafka.server.log.remote.metadata.storage.generated.RemoteLogSegmentMetadataSnapshotRecord;
26   import org.apache.kafka.server.log.remote.metadata.storage.generated.RemoteLogSegmentMetadataUpdateRecord;
27   import org.apache.kafka.server.log.remote.metadata.storage.generated.RemotePartitionDeleteMetadataRecord;
28   import org.apache.kafka.server.log.remote.storage.RemoteLogMetadata;
29   import org.apache.kafka.server.log.remote.storage.RemoteLogSegmentMetadata;
30   import org.apache.kafka.server.log.remote.storage.RemoteLogSegmentMetadataUpdate;
31   import org.apache.kafka.server.log.remote.storage.RemotePartitionDeleteMetadata;
32   
33   import java.util.HashMap;
34   import java.util.Map;
35   
36   /**
37    * This class provides serialization and deserialization for {@link RemoteLogMetadata}. This is the root serde
38    * for the messages that are stored in internal remote log metadata topic.
39    */
40   public class RemoteLogMetadataSerde {
41       private static final short REMOTE_LOG_SEGMENT_METADATA_API_KEY = new RemoteLogSegmentMetadataRecord().apiKey();
42       private static final short REMOTE_LOG_SEGMENT_METADATA_UPDATE_API_KEY = new RemoteLogSegmentMetadataUpdateRecord().apiKey();
43       private static final short REMOTE_PARTITION_DELETE_API_KEY = new RemotePartitionDeleteMetadataRecord().apiKey();
44       private static final short REMOTE_LOG_SEGMENT_METADATA_SNAPSHOT_API_KEY = new RemoteLogSegmentMetadataSnapshotRecord().apiKey();
45   
46       private final Map<String, Short> remoteLogStorageClassToApiKey;
47       private final Map<Short, RemoteLogMetadataTransform> keyToTransform;
48       private final BytesApiMessageSerde bytesApiMessageSerde;
49   
50       public RemoteLogMetadataSerde() {
51           remoteLogStorageClassToApiKey = createRemoteLogStorageClassToApiKeyMap();
52           keyToTransform = createRemoteLogMetadataTransforms();
53           bytesApiMessageSerde = new BytesApiMessageSerde() {
54               @Override
55               public ApiMessage apiMessageFor(short apiKey) {
56                   return newApiMessage(apiKey);
57               }
58           };
59       }
60   
61       protected ApiMessage newApiMessage(short apiKey) {
62           return MetadataRecordType.fromId(apiKey).newMetadataRecord();
63       }
64   
65       protected Map<Short, RemoteLogMetadataTransform> createRemoteLogMetadataTransforms() {
66           Map<Short, RemoteLogMetadataTransform> map = new HashMap<>();
67           map.put(REMOTE_LOG_SEGMENT_METADATA_API_KEY, new RemoteLogSegmentMetadataTransform());
68           map.put(REMOTE_LOG_SEGMENT_METADATA_UPDATE_API_KEY, new RemoteLogSegmentMetadataUpdateTransform());
69           map.put(REMOTE_PARTITION_DELETE_API_KEY, new RemotePartitionDeleteMetadataTransform());
70           map.put(REMOTE_LOG_SEGMENT_METADATA_SNAPSHOT_API_KEY, new RemoteLogSegmentMetadataSnapshotTransform());
71           return map;
72       }
73   
74       protected Map<String, Short> createRemoteLogStorageClassToApiKeyMap() {
75           Map<String, Short> map = new HashMap<>();
76           map.put(RemoteLogSegmentMetadata.class.getName(), REMOTE_LOG_SEGMENT_METADATA_API_KEY);
77           map.put(RemoteLogSegmentMetadataUpdate.class.getName(), REMOTE_LOG_SEGMENT_METADATA_UPDATE_API_KEY);
78           map.put(RemotePartitionDeleteMetadata.class.getName(), REMOTE_PARTITION_DELETE_API_KEY);
79           map.put(RemoteLogSegmentMetadataSnapshot.class.getName(), REMOTE_LOG_SEGMENT_METADATA_SNAPSHOT_API_KEY);
80           return map;
81       }
82   
83       public byte[] serialize(RemoteLogMetadata remoteLogMetadata) {
84           Short apiKey = remoteLogStorageClassToApiKey.get(remoteLogMetadata.getClass().getName());
85           if (apiKey == null) {
86               throw new IllegalArgumentException("ApiKey for given RemoteStorageMetadata class: " + remoteLogMetadata.getClass()
87                                                          + " does not exist.");
88           }
89   
90           @SuppressWarnings("unchecked")
91           ApiMessageAndVersion apiMessageAndVersion = remoteLogMetadataTransform(apiKey).toApiMessageAndVersion(remoteLogMetadata);
92   
93           return bytesApiMessageSerde.serialize(apiMessageAndVersion);
94       }
95   
96       public RemoteLogMetadata deserialize(byte[] data) {
97           ApiMessageAndVersion apiMessageAndVersion = bytesApiMessageSerde.deserialize(data);
98   
99           return remoteLogMetadataTransform(apiMessageAndVersion.message().apiKey()).fromApiMessageAndVersion(apiMessageAndVersion);
100       }
101   
102       private RemoteLogMetadataTransform remoteLogMetadataTransform(short apiKey) {
103           RemoteLogMetadataTransform metadataTransform = keyToTransform.get(apiKey);
104           if (metadataTransform == null) {
105               throw new IllegalArgumentException("RemoteLogMetadataTransform for apikey: " + apiKey + " does not exist.");
106           }
107   
108           return metadataTransform;
109       }
