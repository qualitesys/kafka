63   /**
64    * {@link StreamsResetter} resets the processing state of a Kafka Streams application so that, for example,
65    * you can reprocess its input from scratch.
66    * <p>
67    * <strong>This class is not part of public API. For backward compatibility,
68    * use the provided script in "bin/" instead of calling this class directly from your code.</strong>
69    * <p>
70    * Resetting the processing state of an application includes the following actions:
71    * <ol>
72    * <li>setting the application's consumer offsets for input and internal topics to zero</li>
73    * <li>skip over all intermediate user topics (i.e., "seekToEnd" for consumers of intermediate topics)</li>
74    * <li>deleting any topics created internally by Kafka Streams for this application</li>
75    * </ol>
76    * <p>
77    * Do only use this tool if <strong>no</strong> application instance is running.
78    * Otherwise, the application will get into an invalid state and crash or produce wrong results.
79    * <p>
80    * If you run multiple application instances, running this tool once is sufficient.
81    * However, you need to call {@code KafkaStreams#cleanUp()} before re-starting any instance
82    * (to clean local state store directory).
83    * Otherwise, your application is in an invalid state.
84    * <p>
85    * User output topics will not be deleted or modified by this tool.
86    * If downstream applications consume intermediate or output topics,
87    * it is the user's responsibility to adjust those applications manually if required.
88    */

143       public int run(final String[] args,
144                      final Properties config) {
145           int exitCode;
146   
147           Admin adminClient = null;
148           try {
149               parseArguments(args);
150   
151               final boolean dryRun = options.has(dryRunOption);
152   
153               final String groupId = options.valueOf(applicationIdOption);
154               final Properties properties = new Properties();
155               if (options.has(commandConfigOption)) {
156                   properties.putAll(Utils.loadProps(options.valueOf(commandConfigOption)));
157               }
158               properties.put(ConsumerConfig.BOOTSTRAP_SERVERS_CONFIG, options.valueOf(bootstrapServerOption));
159   
160               adminClient = Admin.create(properties);
161               maybeDeleteActiveConsumers(groupId, adminClient);
162   
163               allTopics.clear();
164               allTopics.addAll(adminClient.listTopics().names().get(60, TimeUnit.SECONDS));
165   
166               if (dryRun) {
167                   System.out.println("----Dry run displays the actions which will be performed when running Streams Reset Tool----");
168               }
169   
170               final HashMap<Object, Object> consumerConfig = new HashMap<>(config);
171               consumerConfig.putAll(properties);
172               exitCode = maybeResetInputAndSeekToEndIntermediateTopicOffsets(consumerConfig, dryRun);
173               exitCode |= maybeDeleteInternalTopics(adminClient, dryRun);
174           } catch (final Throwable e) {
175               exitCode = EXIT_CODE_ERROR;
176               System.err.println("ERROR: " + e);
177               e.printStackTrace(System.err);
178           } finally {
179               if (adminClient != null) {
180                   adminClient.close(Duration.ofSeconds(60));
181               }
182           }
183   
184           return exitCode;
185       }

187       private void maybeDeleteActiveConsumers(final String groupId,
188                                               final Admin adminClient)
189           throws ExecutionException, InterruptedException {
190   
191           final DescribeConsumerGroupsResult describeResult = adminClient.describeConsumerGroups(
192               Collections.singleton(groupId),
193               new DescribeConsumerGroupsOptions().timeoutMs(10 * 1000));
194           final List<MemberDescription> members =
195               new ArrayList<>(describeResult.describedGroups().get(groupId).get().members());
196           if (!members.isEmpty()) {
197               if (options.has(forceOption)) {
198                   System.out.println("Force deleting all active members in the group: " + groupId);
199                   adminClient.removeMembersFromConsumerGroup(groupId, new RemoveMembersFromConsumerGroupOptions()).all().get();
200               } else {
201                   throw new IllegalStateException("Consumer group '" + groupId + "' is still active "
202                           + "and has following members: " + members + ". "
203                           + "Make sure to stop all running application instances before running the reset tool."
204                           + " You can use option '--force' to remove active members from the group.");
205               }
206           }
207       }

312       private int maybeResetInputAndSeekToEndIntermediateTopicOffsets(final Map<Object, Object> consumerConfig,
313                                                                       final boolean dryRun)
314           throws IOException, ParseException {
315   
316           final List<String> inputTopics = options.valuesOf(inputTopicsOption);
317           final List<String> intermediateTopics = options.valuesOf(intermediateTopicsOption);
318           int topicNotFound = EXIT_CODE_SUCCESS;
319   
320           final List<String> notFoundInputTopics = new ArrayList<>();
321           final List<String> notFoundIntermediateTopics = new ArrayList<>();
322   
323           final String groupId = options.valueOf(applicationIdOption);
324   
325           if (inputTopics.size() == 0 && intermediateTopics.size() == 0) {
326               System.out.println("No input or intermediate topics specified. Skipping seek.");
327               return EXIT_CODE_SUCCESS;
328           }
329   
330           if (inputTopics.size() != 0) {
331               System.out.println("Reset-offsets for input topics " + inputTopics);
332           }
333           if (intermediateTopics.size() != 0) {
334               System.out.println("Seek-to-end for intermediate topics " + intermediateTopics);
335           }
336   
337           final Set<String> topicsToSubscribe = new HashSet<>(inputTopics.size() + intermediateTopics.size());
338   
339           for (final String topic : inputTopics) {
340               if (!allTopics.contains(topic)) {
341                   notFoundInputTopics.add(topic);
342               } else {
343                   topicsToSubscribe.add(topic);
344               }
345           }
346           for (final String topic : intermediateTopics) {
347               if (!allTopics.contains(topic)) {
348                   notFoundIntermediateTopics.add(topic);
349               } else {
350                   topicsToSubscribe.add(topic);
351               }
352           }
353   
354           if (!notFoundInputTopics.isEmpty()) {
355               System.out.println("Following input topics are not found, skipping them");
356               for (final String topic : notFoundInputTopics) {
357                   System.out.println("Topic: " + topic);
358               }
359               topicNotFound = EXIT_CODE_ERROR;
360           }
361   
362           if (!notFoundIntermediateTopics.isEmpty()) {
363               System.out.println("Following intermediate topics are not found, skipping them");
364               for (final String topic : notFoundIntermediateTopics) {
365                   System.out.println("Topic:" + topic);
366               }
367               topicNotFound = EXIT_CODE_ERROR;
368           }
369   
370           // Return early if there are no topics to reset (the consumer will raise an error if we
371           // try to poll with an empty subscription)
372           if (topicsToSubscribe.isEmpty()) {
373               return topicNotFound;
374           }
375   
376           final Properties config = new Properties();
377           config.putAll(consumerConfig);
378           config.setProperty(ConsumerConfig.GROUP_ID_CONFIG, groupId);
379           config.setProperty(ConsumerConfig.ENABLE_AUTO_COMMIT_CONFIG, "false");
380   
381           try (final KafkaConsumer<byte[], byte[]> client =
382                    new KafkaConsumer<>(config, new ByteArrayDeserializer(), new ByteArrayDeserializer())) {
383   
384               final Collection<TopicPartition> partitions = topicsToSubscribe.stream().map(client::partitionsFor)
385                       .flatMap(Collection::stream)
386                       .map(info -> new TopicPartition(info.topic(), info.partition()))
387                       .collect(Collectors.toList());
388               client.assign(partitions);
389   
390               final Set<TopicPartition> inputTopicPartitions = new HashSet<>();
391               final Set<TopicPartition> intermediateTopicPartitions = new HashSet<>();
392   
393               for (final TopicPartition p : partitions) {
394                   final String topic = p.topic();
395                   if (isInputTopic(topic)) {
396                       inputTopicPartitions.add(p);
397                   } else if (isIntermediateTopic(topic)) {
398                       intermediateTopicPartitions.add(p);
399                   } else {
400                       System.err.println("Skipping invalid partition: " + p);
401                   }
402               }
403   
404               maybeReset(groupId, client, inputTopicPartitions);
405   
406               maybeSeekToEnd(groupId, client, intermediateTopicPartitions);
407   
408               if (!dryRun) {
409                   for (final TopicPartition p : partitions) {
410                       client.position(p);
411                   }
412                   client.commitSync();
413               }
414           } catch (final IOException | ParseException e) {
415               System.err.println("ERROR: Resetting offsets failed.");
416               throw e;
417           }
418           System.out.println("Done.");
419           return topicNotFound;
420       }

570       private Map<TopicPartition, Long> parseResetPlan(final String resetPlanCsv) throws ParseException {
571           final Map<TopicPartition, Long> topicPartitionAndOffset = new HashMap<>();
572           if (resetPlanCsv == null || resetPlanCsv.isEmpty()) {
573               throw new ParseException("Error parsing reset plan CSV file. It is empty,", 0);
574           }
575   
576           final String[] resetPlanCsvParts = resetPlanCsv.split("\n");
577   
578           for (final String line : resetPlanCsvParts) {
579               final String[] lineParts = line.split(",");
580               if (lineParts.length != 3) {
581                   throw new ParseException("Reset plan CSV file is not following the format `TOPIC,PARTITION,OFFSET`.", 0);
582               }
583               final String topic = lineParts[0];
584               final int partition = Integer.parseInt(lineParts[1]);
585               final long offset = Long.parseLong(lineParts[2]);
586               final TopicPartition topicPartition = new TopicPartition(topic, partition);
587               topicPartitionAndOffset.put(topicPartition, offset);
588           }
589   
590           return topicPartitionAndOffset;
591       }

593       private Map<TopicPartition, Long> checkOffsetRange(final Map<TopicPartition, Long> inputTopicPartitionsAndOffset,
594                                                          final Map<TopicPartition, Long> beginningOffsets,
595                                                          final Map<TopicPartition, Long> endOffsets) {
596           final Map<TopicPartition, Long> validatedTopicPartitionsOffsets = new HashMap<>();
597           for (final Map.Entry<TopicPartition, Long> topicPartitionAndOffset : inputTopicPartitionsAndOffset.entrySet()) {
598               final long endOffset = endOffsets.get(topicPartitionAndOffset.getKey());
599               final long offset = topicPartitionAndOffset.getValue();
600               if (offset < endOffset) {
601                   final long beginningOffset = beginningOffsets.get(topicPartitionAndOffset.getKey());
602                   if (offset > beginningOffset) {
603                       validatedTopicPartitionsOffsets.put(topicPartitionAndOffset.getKey(), offset);
604                   } else {
605                       System.out.println("New offset (" + offset + ") is lower than earliest offset. Value will be set to " + beginningOffset);
606                       validatedTopicPartitionsOffsets.put(topicPartitionAndOffset.getKey(), beginningOffset);
607                   }
608               } else {
609                   System.out.println("New offset (" + offset + ") is higher than latest offset. Value will be set to " + endOffset);
610                   validatedTopicPartitionsOffsets.put(topicPartitionAndOffset.getKey(), endOffset);
611               }
612           }
613           return validatedTopicPartitionsOffsets;
614       }

624       private int maybeDeleteInternalTopics(final Admin adminClient, final boolean dryRun) {
625           final List<String> inferredInternalTopics = allTopics.stream()
626                   .filter(this::isInferredInternalTopic)
627                   .collect(Collectors.toList());
628           final List<String> specifiedInternalTopics = options.valuesOf(internalTopicsOption);
629           final List<String> topicsToDelete;
630   
631           if (!specifiedInternalTopics.isEmpty()) {
632               if (!inferredInternalTopics.containsAll(specifiedInternalTopics)) {
633                   throw new IllegalArgumentException("Invalid topic specified in the "
634                           + "--internal-topics option. "
635                           + "Ensure that the topics specified are all internal topics. "
636                           + "Do a dry run without the --internal-topics option to see the "
637                           + "list of all internal topics that can be deleted.");
638               }
639   
640               topicsToDelete = specifiedInternalTopics;
641               System.out.println("Deleting specified internal topics " + topicsToDelete);
642           } else {
643               topicsToDelete = inferredInternalTopics;
644               System.out.println("Deleting inferred internal topics " + topicsToDelete);
645           }
646   
647           if (!dryRun) {
648               doDelete(topicsToDelete, adminClient);
649           }
650   
651           System.out.println("Done.");
652           return EXIT_CODE_SUCCESS;
653       }

676       private boolean isInferredInternalTopic(final String topicName) {
677           // Specified input/intermediate topics might be named like internal topics (by chance).
678           // Even is this is not expected in general, we need to exclude those topics here
679           // and don't consider them as internal topics even if they follow the same naming schema.
680           // Cf. https://issues.apache.org/jira/browse/KAFKA-7930
681           return !isInputTopic(topicName) && !isIntermediateTopic(topicName) && topicName.startsWith(options.valueOf(applicationIdOption) + "-")
682                  && matchesInternalTopicFormat(topicName);
683       }

687           return topicName.endsWith("-changelog") || topicName.endsWith("-repartition")
688                  || topicName.endsWith("-subscription-registration-topic")
689                  || topicName.endsWith("-subscription-response-topic")
