18   package org.apache.kafka.trogdor.workload;
19   
20   import com.fasterxml.jackson.annotation.JsonCreator;
21   import com.fasterxml.jackson.annotation.JsonProperty;
22   import com.fasterxml.jackson.databind.JsonNode;
23   
24   import com.fasterxml.jackson.databind.node.NullNode;
25   import org.apache.kafka.trogdor.task.TaskController;
26   import org.apache.kafka.trogdor.task.TaskSpec;
27   import org.apache.kafka.trogdor.task.TaskWorker;
28   
29   import java.util.ArrayList;
30   import java.util.Collections;
31   import java.util.List;
32   import java.util.Optional;
33   
34   /**
35    * ExternalCommandSpec describes a task that executes Trogdor tasks with the command.
36    *
37    * An example uses the python runner to execute the ProduceBenchSpec task.
38    *
39    * #{@code
40    *   {
41    *      "class": "org.apache.kafka.trogdor.workload.ExternalCommandSpec",
42    *      "command": ["python", "/path/to/trogdor/python/runner"],
43    *      "durationMs": 10000000,
44    *      "producerNode": "node0",
45    *      "workload": {
46    *        "class": "org.apache.kafka.trogdor.workload.ProduceBenchSpec",
47    *        "bootstrapServers": "localhost:9092",
48    *        "targetMessagesPerSec": 10,
49    *        "maxMessages": 100,
50    *        "activeTopics": {
51    *          "foo[1-3]": {
52    *            "numPartitions": 3,
53    *            "replicationFactor": 1
54    *          }
55    *        },
56    *        "inactiveTopics": {
57    *          "foo[4-5]": {
58    *            "numPartitions": 3,
59    *            "replicationFactor": 1
60    *          }
61    *        }
62    *     }
63    *   }
64    */
65   public class ExternalCommandSpec extends TaskSpec {
66       private final String commandNode;
67       private final List<String> command;
68       private final JsonNode workload;
69       private final Optional<Integer> shutdownGracePeriodMs;
70   
71       @JsonCreator
72       public ExternalCommandSpec(
73               @JsonProperty("startMs") long startMs,
74               @JsonProperty("durationMs") long durationMs,
75               @JsonProperty("commandNode") String commandNode,
76               @JsonProperty("command") List<String> command,
77               @JsonProperty("workload") JsonNode workload,
78               @JsonProperty("shutdownGracePeriodMs") Optional<Integer> shutdownGracePeriodMs) {
79           super(startMs, durationMs);
80           this.commandNode = (commandNode == null) ? "" : commandNode;
81           this.command = (command == null) ? Collections.unmodifiableList(new ArrayList<String>()) : command;
82           this.workload = (workload == null) ? NullNode.instance : workload;
83           this.shutdownGracePeriodMs = shutdownGracePeriodMs;
84       }
85   
86       @JsonProperty
87       public String commandNode() {
88           return commandNode;
89       }
90   
91       @JsonProperty
92       public List<String> command() {
93           return command;
94       }
95   
96       @JsonProperty
97       public JsonNode workload() {
98           return workload;
99       }
100   
101       @JsonProperty
102       public Optional<Integer> shutdownGracePeriodMs() {
103           return shutdownGracePeriodMs;
104       }
105   
106       @Override
107       public TaskController newController(String id) {
108           return topology -> Collections.singleton(commandNode);
109       }
110   
111       @Override
112       public TaskWorker newTaskWorker(String id) {
113           return new ExternalCommandWorker(id, this);
114       }
115   }
