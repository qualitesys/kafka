27   /**
28    * A histogram that can easily find the average, median etc of a large number of samples in a
29    * restricted domain.
30    */
31   public class Histogram {
32       private final int[] counts;
33   
34       private final Logger log = LoggerFactory.getLogger(Histogram.class);
35   
36       public Histogram(int maxValue) {
37           this.counts = new int[maxValue + 1];
38       }
39   
40       /**
41        * Add a new value to the histogram.
42        *
43        * Note that the value will be clipped to the maximum value available in the Histogram instance.
44        * So if the histogram has 100 buckets, inserting 101 will increment the last bucket.
45        */
46       public void add(int value) {
47           if (value < 0) {
48               throw new RuntimeException("invalid negative value.");
49           }
50           if (value >= counts.length) {
51               value = counts.length - 1;
52           }
53           synchronized (this) {
54               int curCount = counts[value];
55               if (curCount < Integer.MAX_VALUE) {
56                   counts[value] = counts[value] + 1;
57               }
58           }
59       }
60   
61       /**
62        * Add a new value to the histogram.
63        *
64        * Note that the value will be clipped to the maximum value available in the Histogram instance.
65        * This method is provided for convenience, but handles the same numeric range as the method which
66        * takes an int.
67        */
68       public void add(long value) {
69           if (value > Integer.MAX_VALUE) {
70               add(Integer.MAX_VALUE);
71           } else if (value < Integer.MIN_VALUE) {
72               add(Integer.MIN_VALUE);
73           } else {
74               add((int) value);
75           }
76       }

78       public static class Summary {
79           /**
80            * The total number of samples.
81            */
82           private final long numSamples;
83   
84           /**
85            * The average of all samples.
86            */
87           private final float average;
88   
89           /**
90            * Percentile information.
91            *
92            * percentile(fraction=0.99) will have a value which is greater than or equal to 99%
93            * of the samples.  percentile(fraction=0.5) is the median sample.  And so forth.
94            */
95           private final List<PercentileSummary> percentiles;
96   
97           Summary(long numSamples, float average, List<PercentileSummary> percentiles) {
98               this.numSamples = numSamples;
99               this.average = average;
100               this.percentiles = percentiles;
101           }
102   
103           public long numSamples() {
104               return numSamples;
105           }
106   
107           public float average() {
108               return average;
109           }
110   
111           public List<PercentileSummary> percentiles() {
112               return percentiles;
113           }
114       }

116       /**
117        * Information about a percentile.
118        */
119       public static class PercentileSummary {
120           /**
121            * The fraction of samples which are less than or equal to the value of this percentile.
122            */
123           private final float fraction;
124   
125           /**
126            * The value of this percentile.
127            */
128           private final int value;
129   
130           PercentileSummary(float fraction, int value) {
131               this.fraction = fraction;
132               this.value = value;
133           }
134   
135           public float fraction() {
136               return fraction;
137           }
138   
139           public int value() {
140               return value;
141           }
142       }

144       public Summary summarize() {
145           return summarize(new float[0]);
146       }

148       public Summary summarize(float[] percentiles) {
149           int[] countsCopy = new int[counts.length];
150           synchronized (this) {
151               System.arraycopy(counts, 0, countsCopy, 0, counts.length);
152           }
153           // Verify that the percentiles array is sorted and positive.
154           float prev = 0f;
155           for (int i = 0; i < percentiles.length; i++) {
156               if (percentiles[i] < prev) {
157                   throw new RuntimeException("Invalid percentiles fraction array.  Bad element " +
158                       percentiles[i] + ".  The array must be sorted and non-negative.");
159               }
160               if (percentiles[i] > 1.0f) {
161                   throw new RuntimeException("Invalid percentiles fraction array.  Bad element " +
162                       percentiles[i] + ".  Elements must be less than or equal to 1.");
163               }
164           }
165           // Find out how many total samples we have, and what the average is.
166           long numSamples = 0;
167           float total = 0f;
168           for (int i = 0; i < countsCopy.length; i++) {
169               long count = countsCopy[i];
170               numSamples = numSamples + count;
171               total = total + (i * count);
172           }
173           float average = (numSamples == 0) ? 0.0f : (total / numSamples);
174   
175           List<PercentileSummary> percentileSummaries =
176               summarizePercentiles(countsCopy, percentiles, numSamples);
177           return new Summary(numSamples, average, percentileSummaries);
178       }
179   
180       private List<PercentileSummary> summarizePercentiles(int[] countsCopy, float[] percentiles,
181                                                            long numSamples) {
182           if (percentiles.length == 0) {
183               return Collections.emptyList();
184           }
185           List<PercentileSummary> summaries = new ArrayList<>(percentiles.length);
186           int i = 0, j = 0;
187           long seen = 0, next = (long) (numSamples * percentiles[0]);
188           while (true) {
189               if (i == countsCopy.length - 1) {
190                   for (; j < percentiles.length; j++) {
191                       summaries.add(new PercentileSummary(percentiles[j], i));
192                   }
193                   return summaries;
194               }
195               seen += countsCopy[i];
196               while (seen >= next) {
197                   summaries.add(new PercentileSummary(percentiles[j], i));
198                   j++;
199                   if (j == percentiles.length) {
200                       return summaries;
201                   }
202                   next = (long) (numSamples * percentiles[j]);
203               }
204               i++;
205           }
206       }
