40   /**
41    * Uses the linux utility <pre>tc</pre> (traffic controller) to degrade performance on a specified network device
42    */
43   public class DegradedNetworkFaultWorker implements TaskWorker {
44   
45       private static final Logger log = LoggerFactory.getLogger(DegradedNetworkFaultWorker.class);
46   
47       private final String id;
48       private final Map<String, DegradedNetworkFaultSpec.NodeDegradeSpec> nodeSpecs;
49       private WorkerStatusTracker status;
50   
51       public DegradedNetworkFaultWorker(String id, Map<String, DegradedNetworkFaultSpec.NodeDegradeSpec> nodeSpecs) {
52           this.id = id;
53           this.nodeSpecs = nodeSpecs;
54       }
55   
56       @Override
57       public void start(Platform platform, WorkerStatusTracker status, KafkaFutureImpl<String> haltFuture) throws Exception {
58           log.info("Activating DegradedNetworkFaultWorker {}.", id);
59           this.status = status;
60           this.status.update(new TextNode("enabling traffic control " + id));
61           Node curNode = platform.curNode();
62           DegradedNetworkFaultSpec.NodeDegradeSpec nodeSpec = nodeSpecs.get(curNode.name());
63           if (nodeSpec != null) {
64               for (String device : devicesForSpec(nodeSpec)) {
65                   if (nodeSpec.latencyMs() < 0 || nodeSpec.rateLimitKbit() < 0) {
66                       throw new RuntimeException("Expected non-negative values for latencyMs and rateLimitKbit, but got " + nodeSpec);
67                   } else {
68                       enableTrafficControl(platform, device, nodeSpec.latencyMs(), nodeSpec.rateLimitKbit());
69                   }
70               }
71           }
72           this.status.update(new TextNode("enabled traffic control " + id));
73       }
74   
75       @Override
76       public void stop(Platform platform) throws Exception {
77           log.info("Deactivating DegradedNetworkFaultWorker {}.", id);
78           this.status.update(new TextNode("disabling traffic control " + id));
79           Node curNode = platform.curNode();
80           DegradedNetworkFaultSpec.NodeDegradeSpec nodeSpec = nodeSpecs.get(curNode.name());
81           if (nodeSpec != null) {
82               for (String device : devicesForSpec(nodeSpec)) {
83                   disableTrafficControl(platform, device);
84               }
85           }
86           this.status.update(new TextNode("disabled traffic control " + id));
87       }
88   
89       private Set<String> devicesForSpec(DegradedNetworkFaultSpec.NodeDegradeSpec nodeSpec) throws Exception {
90           Set<String> devices = new HashSet<>();
91           if (nodeSpec.networkDevice().isEmpty()) {
92               for (NetworkInterface networkInterface : Collections.list(NetworkInterface.getNetworkInterfaces())) {
93                   if (!networkInterface.isLoopback()) {
94                       devices.add(networkInterface.getName());
95                   }
96               }
97           } else {
98               devices.add(nodeSpec.networkDevice());
99           }
100           return devices;
101       }
102   
103       /**
104        * Constructs the appropriate "tc" commands to apply latency and rate limiting, if they are non zero.
105        */
106       private void enableTrafficControl(Platform platform, String networkDevice, int delayMs, int rateLimitKbps) throws IOException {
107           if (delayMs > 0) {
108               int deviationMs = Math.max(1, (int) Math.sqrt(delayMs));
109               List<String> delay = new ArrayList<>();
110               rootHandler(networkDevice, delay::add);
111               netemDelay(delayMs, deviationMs, delay::add);
112               platform.runCommand(delay.toArray(new String[0]));
113   
114               if (rateLimitKbps > 0) {
115                   List<String> rate = new ArrayList<>();
116                   childHandler(networkDevice, rate::add);
117                   tbfRate(rateLimitKbps, rate::add);
118                   platform.runCommand(rate.toArray(new String[0]));
119               }
120           } else if (rateLimitKbps > 0) {
121               List<String> rate = new ArrayList<>();
122               rootHandler(networkDevice, rate::add);
123               tbfRate(rateLimitKbps, rate::add);
124               platform.runCommand(rate.toArray(new String[0]));
125           } else {
126               log.warn("Not applying any rate limiting or latency");
127           }
128       }
129   
130       /**
131        * Construct the first part of a "tc" command to define a qdisc root handler for the given network interface
132        */
133       private void rootHandler(String networkDevice, Consumer<String> consumer) {
134           Stream.of("sudo", "tc", "qdisc", "add", "dev", networkDevice, "root", "handle", "1:0").forEach(consumer);
135       }
136   
137       /**
138        * Construct the first part of a "tc" command to define a qdisc child handler for the given interface. This can
139        * only be used if a root handler has been appropriately defined first (as in {@link #rootHandler}).
140        */
141       private void childHandler(String networkDevice, Consumer<String> consumer) {
142           Stream.of("sudo", "tc", "qdisc", "add", "dev", networkDevice, "parent", "1:1", "handle", "10:").forEach(consumer);
143       }
144   
145       /**
146        * Construct the second part of a "tc" command that defines a netem (Network Emulator) filter that will apply some
147        * amount of latency with a small amount of deviation. The distribution of the latency deviation follows a so-called
148        * Pareto-normal distribution. This is the formal name for the 80/20 rule, which might better represent real-world
149        * patterns.
150        */
151       private void netemDelay(int delayMs, int deviationMs, Consumer<String> consumer) {
152           Stream.of("netem", "delay", String.format("%dms", delayMs), String.format("%dms", deviationMs),
153                   "distribution", "paretonormal").forEach(consumer);
154       }
155   
156       /**
157        * Construct the second part of a "tc" command that defines a tbf (token buffer filter) that will rate limit the
158        * packets going through a qdisc.
159        */
160       private void tbfRate(int rateLimitKbit, Consumer<String> consumer) {
161           Stream.of("tbf", "rate", String.format("%dkbit", rateLimitKbit), "burst", "1mbit", "latency", "500ms").forEach(consumer);
162       }
163   
164       /**
165        * Delete any previously defined qdisc for the given network interface.
166        * @throws IOException
167        */
168       private void disableTrafficControl(Platform platform, String networkDevice) throws IOException {
169           platform.runCommand(new String[] {
170               "sudo", "tc", "qdisc", "del", "dev", networkDevice, "root"
171           });
172       }
173   }
