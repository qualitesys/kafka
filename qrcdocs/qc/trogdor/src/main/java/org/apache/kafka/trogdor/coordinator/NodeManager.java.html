18   /*
19    * So, when a task comes in, it happens via createTask (the RPC backend).
20    * This starts a CreateTask on the main state change thread, and waits for it.
21    * That task checks the main task hash map, and returns back the existing task spec
22    * if there is something there.  If there is nothing there, it creates
23    * something new, and returns null.
24    * It also schedules a RunTask some time in the future on the main state change thread.
25    * We save the future from this in case we need to cancel it later, in a StopTask.
26    * If we can't create the TaskController for the task, we transition to DONE with an
27    * appropriate error message.
28    *
29    * RunTask actually starts the task which was created earlier.  This could
30    * happen an arbitrary amount of time after task creation (it is based on the
31    * task spec).  RunTask must operate only on PENDING tasks... if the task has been
32    * stopped, then we have nothing to do here.
33    * RunTask asks the TaskController for a list of all the names of nodes
34    * affected by this task.
35    * If this list contains nodes we don't know about, or zero nodes, we
36    * transition directly to DONE state with an appropriate error set.
37    * RunTask schedules CreateWorker Callables on all the affected worker nodes.
38    * These callables run in the context of the relevant NodeManager.
39    *
40    * CreateWorker calls the RPC of the same name for the agent.
41    * There is some complexity here due to retries.
42    */
43

71   /**
72    * The NodeManager handles communicating with a specific agent node.
73    * Each NodeManager has its own ExecutorService which runs in a dedicated thread.
74    */
75   public final class NodeManager {
76       private static final Logger log = LoggerFactory.getLogger(NodeManager.class);

78       /**
79        * The normal amount of seconds between heartbeats sent to the agent.
80        */
81       private static final long HEARTBEAT_DELAY_MS = 1000L;

83       class ManagedWorker {
84           private final long workerId;
85           private final String taskId;
86           private final TaskSpec spec;
87           private boolean shouldRun;
88           private WorkerState state;
89   
90           ManagedWorker(long workerId, String taskId, TaskSpec spec,
91                         boolean shouldRun, WorkerState state) {
92               this.workerId = workerId;
93               this.taskId = taskId;
94               this.spec = spec;
95               this.shouldRun = shouldRun;
96               this.state = state;
97           }
98   
99           void tryCreate() {
100               try {
101                   client.createWorker(new CreateWorkerRequest(workerId, taskId, spec));
102               } catch (Throwable e) {
103                   log.error("{}: error creating worker {}.", node.name(), this, e);
104               }
105           }
106   
107           void tryStop() {
108               try {
109                   client.stopWorker(new StopWorkerRequest(workerId));
110               } catch (Throwable e) {
111                   log.error("{}: error stopping worker {}.", node.name(), this, e);
112               }
113           }
114   
115           @Override
116           public String toString() {
117               return String.format("%s_%d", taskId, workerId);
118           }

146       /**
147        * The heartbeat runnable.
148        */
149       private final NodeHeartbeat heartbeat;
150   
151       /**
152        * A future which can be used to cancel the periodic hearbeat task.
153        */
154       private ScheduledFuture<?> heartbeatFuture;
155   
156       NodeManager(Node node, TaskManager taskManager) {
157           this.node = node;
158           this.taskManager = taskManager;
159           this.client = new AgentClient.Builder().
160               maxTries(1).
161               target(node.hostname(), Node.Util.getTrogdorAgentPort(node)).
162               build();
163           this.workers = new HashMap<>();
164           this.executor = Executors.newSingleThreadScheduledExecutor(
165               ThreadUtils.createThreadFactory("NodeManager(" + node.name() + ")",
166                   false));
167           this.heartbeat = new NodeHeartbeat();
168           rescheduleNextHeartbeat(HEARTBEAT_DELAY_MS);
169       }
170   
171       /**
172        * Reschedule the heartbeat runnable.
173        *
174        * @param initialDelayMs        The initial delay to use.
175        */
176       void rescheduleNextHeartbeat(long initialDelayMs) {
177           if (this.heartbeatFuture != null) {
178               this.heartbeatFuture.cancel(false);
179           }
180           this.heartbeatFuture = this.executor.scheduleAtFixedRate(heartbeat,
181               initialDelayMs, HEARTBEAT_DELAY_MS, TimeUnit.MILLISECONDS);
182       }

184       /**
185        * The heartbeat runnable.
186        */
187       class NodeHeartbeat implements Runnable {
188           @Override
189           public void run() {
190               rescheduleNextHeartbeat(HEARTBEAT_DELAY_MS);
191               try {
192                   AgentStatusResponse agentStatus = null;
193                   try {
194                       agentStatus = client.status();
195                   } catch (ConnectException e) {
196                       log.error("{}: failed to get agent status: ConnectException {}", node.name(), e.getMessage());
197                       return;
198                   } catch (Exception e) {
199                       log.error("{}: failed to get agent status", node.name(), e);
200                       // TODO: eventually think about putting tasks into a bad state as a result of
201                       // agents going down?
202                       return;
203                   }
204                   if (log.isTraceEnabled()) {
205                       log.trace("{}: got heartbeat status {}", node.name(), agentStatus);
206                   }
207                   handleMissingWorkers(agentStatus);
208                   handlePresentWorkers(agentStatus);
209               } catch (Throwable e) {
210                   log.error("{}: Unhandled exception in NodeHeartbeatRunnable", node.name(), e);
211               }
212           }
213   
214           /**
215            * Identify workers which we think should be running but do not appear in the agent's response.
216            * We need to send startWorker requests for those
217            */
218           private void handleMissingWorkers(AgentStatusResponse agentStatus) {
219               for (Map.Entry<Long, ManagedWorker> entry : workers.entrySet()) {
220                   Long workerId = entry.getKey();
221                   if (!agentStatus.workers().containsKey(workerId)) {
222                       ManagedWorker worker = entry.getValue();
223                       if (worker.shouldRun) {
224                           worker.tryCreate();
225                       }
226                   }
227               }
228           }
229   
230           private void handlePresentWorkers(AgentStatusResponse agentStatus) {
231               for (Map.Entry<Long, WorkerState> entry : agentStatus.workers().entrySet()) {
232                   long workerId = entry.getKey();
233                   WorkerState state = entry.getValue();
234                   ManagedWorker worker = workers.get(workerId);
235                   if (worker == null) {
236                       // Identify tasks which are running, but which we don't know about.
237                       // Add these to the NodeManager as tasks that should not be running.
238                       log.warn("{}: scheduling unknown worker with ID {} for stopping.", node.name(), workerId);
239                       workers.put(workerId, new ManagedWorker(workerId, state.taskId(),
240                           state.spec(), false, state));
241                   } else {
242                       // Handle workers which need to be stopped.
243                       if (state instanceof WorkerStarting || state instanceof WorkerRunning) {
244                           if (!worker.shouldRun) {
245                               worker.tryStop();
246                           }
247                       }
248                       // Notify the TaskManager if the worker state has changed.
249                       if (worker.state.equals(state)) {
250                           log.debug("{}: worker state is still {}", node.name(), worker.state);
251                       } else {
252                           log.info("{}: worker state changed from {} to {}", node.name(), worker.state, state);
253                           if (state instanceof WorkerDone || state instanceof WorkerStopping)
254                               worker.shouldRun = false;
255                           worker.state = state;
256                           taskManager.updateWorkerState(node.name(), worker.workerId, state);
257                       }
258                   }
259               }
260           }
261       }

264        * Create a new worker.
265        *
266        * @param workerId              The new worker id.
267        * @param taskId                The new task id.

269        */

274       /**
275        * Starts a worker.
276        */
277       class CreateWorker implements Callable<Void> {
278           private final long workerId;
279           private final String taskId;
280           private final TaskSpec spec;
281   
282           CreateWorker(long workerId, String taskId, TaskSpec spec) {
283               this.workerId = workerId;
284               this.taskId = taskId;
285               this.spec = spec;
286           }
287   
288           @Override
289           public Void call() throws Exception {
290               ManagedWorker worker = workers.get(workerId);
291               if (worker != null) {
292                   log.error("{}: there is already a worker {} with ID {}.",
293                       node.name(), worker, workerId);
294                   return null;
295               }
296               worker = new ManagedWorker(workerId, taskId, spec, true, new WorkerReceiving(taskId, spec));
297               log.info("{}: scheduling worker {} to start.", node.name(), worker);
298               workers.put(workerId, worker);
299               rescheduleNextHeartbeat(0);
300               return null;
301           }
302       }

309       public void stopWorker(long workerId) {

313       /**
314        * Stops a worker.
315        */
316       class StopWorker implements Callable<Void> {
317           private final long workerId;
318   
319           StopWorker(long workerId) {
320               this.workerId = workerId;
321           }
322   
323           @Override
324           public Void call() throws Exception {
325               ManagedWorker worker = workers.get(workerId);
326               if (worker == null) {
327                   log.error("{}: unable to locate worker to stop with ID {}.", node.name(), workerId);
328                   return null;
329               }
330               if (!worker.shouldRun) {
331                   log.error("{}: Worker {} is already scheduled to stop.",
332                       node.name(), worker);
333                   return null;
334               }
335               log.info("{}: scheduling worker {} to stop.", node.name(), worker);
336               worker.shouldRun = false;
337               rescheduleNextHeartbeat(0);
338               return null;
339           }
340       }

347       public void destroyWorker(long workerId) {

352        * Destroys a worker.

354       class DestroyWorker implements Callable<Void> {
355           private final long workerId;
356   
357           DestroyWorker(long workerId) {
358               this.workerId = workerId;
359           }
360   
361           @Override
362           public Void call() throws Exception {
363               ManagedWorker worker = workers.remove(workerId);
364               if (worker == null) {
365                   log.error("{}: unable to locate worker to destroy with ID {}.", node.name(), workerId);
366                   return null;
367               }
368               rescheduleNextHeartbeat(0);
369               return null;
370           }
371       }
372   
373       public void beginShutdown(boolean stopNode) {
374           executor.shutdownNow();
375           if (stopNode) {
376               try {
377                   client.invokeShutdown();
378               } catch (Exception e) {
379                   log.error("{}: Failed to send shutdown request", node.name(), e);
380               }
381           }
382       }
383   
384       public void waitForShutdown() throws InterruptedException {
385           executor.awaitTermination(1, TimeUnit.DAYS);
386       }
