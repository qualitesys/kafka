127       class Prepare implements Runnable {
128           @Override
129           public void run() {
130               try {
131                   if (spec.targetMessagesPerSec() <= 0) {
132                       throw new ConfigException("Can't have targetMessagesPerSec <= 0.");
133                   }
134                   Map<String, NewTopic> newTopics = new HashMap<>();
135                   HashSet<TopicPartition> active = new HashSet<>();
136                   for (Map.Entry<String, PartitionsSpec> entry :
137                       spec.activeTopics().materialize().entrySet()) {
138                       String topicName = entry.getKey();
139                       PartitionsSpec partSpec = entry.getValue();
140                       newTopics.put(topicName, partSpec.newTopic(topicName));
141                       for (Integer partitionNumber : partSpec.partitionNumbers()) {
142                           active.add(new TopicPartition(topicName, partitionNumber));
143                       }
144                   }
145                   if (active.isEmpty()) {
146                       throw new RuntimeException("You must specify at least one active topic.");
147                   }
148                   status.update(new TextNode("Creating " + newTopics.keySet().size() + " topic(s)"));
149                   WorkerUtils.createTopics(log, spec.bootstrapServers(), spec.commonClientConf(),
150                       spec.adminClientConf(), newTopics, false);
151                   status.update(new TextNode("Created " + newTopics.keySet().size() + " topic(s)"));
152                   toSendTracker = new ToSendTracker(spec.maxMessages());
153                   toReceiveTracker = new ToReceiveTracker();
154                   executor.submit(new ProducerRunnable(active));
155                   executor.submit(new ConsumerRunnable(active));
156                   executor.submit(new StatusUpdater());
157                   executor.scheduleWithFixedDelay(
158                       new StatusUpdater(), 30, 30, TimeUnit.SECONDS);
159               } catch (Throwable e) {
160                   WorkerUtils.abort(log, "Prepare", e, doneFuture);
161               }
162           }
163       }

175       private static class ToSendTracker {
176           private final long maxMessages;
177           private final List<Long> failed = new ArrayList<>();
178           private long frontier = 0;
179   
180           ToSendTracker(long maxMessages) {
181               this.maxMessages = maxMessages;
182           }
183   
184           synchronized void addFailed(long index) {
185               failed.add(index);
186           }
187   
188           synchronized long frontier() {
189               return frontier;
190           }
191   
192           synchronized ToSendTrackerResult next() {
193               if (failed.isEmpty()) {
194                   if (frontier >= maxMessages) {
195                       return null;
196                   } else {
197                       return new ToSendTrackerResult(frontier++, true);
198                   }
199               } else {
200                   return new ToSendTrackerResult(failed.remove(0), false);
201               }
202           }
203       }

205       class ProducerRunnable implements Runnable {
206           private final HashSet<TopicPartition> partitions;
207           private final Throttle throttle;
208   
209           ProducerRunnable(HashSet<TopicPartition> partitions) {
210               this.partitions = partitions;
211               Properties props = new Properties();
212               props.put(ProducerConfig.BOOTSTRAP_SERVERS_CONFIG, spec.bootstrapServers());
213               props.put(ProducerConfig.BATCH_SIZE_CONFIG, 16 * 1024);
214               props.put(ProducerConfig.BUFFER_MEMORY_CONFIG, 4 * 16 * 1024L);
215               props.put(ProducerConfig.MAX_BLOCK_MS_CONFIG, 1000L);
216               props.put(ProducerConfig.CLIENT_ID_CONFIG, "producer." + id);
217               props.put(ProducerConfig.ACKS_CONFIG, "all");
218               props.put(ProducerConfig.REQUEST_TIMEOUT_MS_CONFIG, 105000);
219               // user may over-write the defaults with common client config and producer config
220               WorkerUtils.addConfigsToProperties(props, spec.commonClientConf(), spec.producerConf());
221               producer = new KafkaProducer<>(props, new ByteArraySerializer(),
222                   new ByteArraySerializer());
223               int perPeriod = WorkerUtils.
224                   perSecToPerPeriod(spec.targetMessagesPerSec(), THROTTLE_PERIOD_MS);
225               this.throttle = new Throttle(perPeriod, THROTTLE_PERIOD_MS);
226           }
227   
228           @Override
229           public void run() {
230               long messagesSent = 0;
231               long uniqueMessagesSent = 0;
232               log.debug("{}: Starting RoundTripWorker#ProducerRunnable.", id);
233               try {
234                   Iterator<TopicPartition> iter = partitions.iterator();
235                   while (true) {
236                       final ToSendTrackerResult result = toSendTracker.next();
237                       if (result == null) {
238                           break;
239                       }
240                       throttle.increment();
241                       final long messageIndex = result.index;
242                       if (result.firstSend) {
243                           toReceiveTracker.addPending(messageIndex);
244                           uniqueMessagesSent++;
245                       }
246                       messagesSent++;
247                       if (!iter.hasNext()) {
248                           iter = partitions.iterator();
249                       }
250                       TopicPartition partition = iter.next();
251                       // we explicitly specify generator position based on message index
252                       ProducerRecord<byte[], byte[]> record = new ProducerRecord<>(partition.topic(),
253                           partition.partition(), KEY_GENERATOR.generate(messageIndex),
254                           spec.valueGenerator().generate(messageIndex));
255                       producer.send(record, (metadata, exception) -> {
256                           if (exception == null) {
257                               lock.lock();
258                               try {
259                                   unackedSends -= 1;
260                                   if (unackedSends <= 0)
261                                       unackedSendsAreZero.signalAll();
262                               } finally {
263                                   lock.unlock();
264                               }
265                           } else {
266                               log.info("{}: Got exception when sending message {}: {}",
267                                   id, messageIndex, exception.getMessage());
268                               toSendTracker.addFailed(messageIndex);
269                           }
270                       });
271                   }
272               } catch (Throwable e) {
273                   WorkerUtils.abort(log, "ProducerRunnable", e, doneFuture);
274               } finally {
275                   lock.lock();
276                   try {
277                       log.info("{}: ProducerRunnable is exiting.  messagesSent={}; uniqueMessagesSent={}; " +
278                                       "ackedSends={}/{}.", id, messagesSent, uniqueMessagesSent,
279                               spec.maxMessages() - unackedSends, spec.maxMessages());
280                   } finally {
281                       lock.unlock();
282                   }
283               }
284           }
285       }

287       private class ToReceiveTracker {
288           private final TreeSet<Long> pending = new TreeSet<>();
289   
290           private long totalReceived = 0;
291   
292           synchronized void addPending(long messageIndex) {
293               pending.add(messageIndex);
294           }
295   
296           synchronized boolean removePending(long messageIndex) {
297               if (pending.remove(messageIndex)) {
298                   totalReceived++;
299                   return true;
300               } else {
301                   return false;
302               }
303           }
304   
305           synchronized long totalReceived() {
306               return totalReceived;
307           }
308   
309           void log() {
310               long numToReceive;
311               List<Long> list = new ArrayList<>(LOG_NUM_MESSAGES);
312               synchronized (this) {
313                   numToReceive = pending.size();
314                   for (Iterator<Long> iter = pending.iterator();
315                           iter.hasNext() && (list.size() < LOG_NUM_MESSAGES); ) {
316                       Long i = iter.next();
317                       list.add(i);
318                   }
319               }
320               log.info("{}: consumer waiting for {} message(s), starting with: {}",
321                   id, numToReceive, Utils.join(list, ", "));
322           }
323       }

325       class ConsumerRunnable implements Runnable {
326           private final Properties props;
327   
328           ConsumerRunnable(HashSet<TopicPartition> partitions) {
329               this.props = new Properties();
330               props.put(ConsumerConfig.BOOTSTRAP_SERVERS_CONFIG, spec.bootstrapServers());
331               props.put(ConsumerConfig.CLIENT_ID_CONFIG, "consumer." + id);
332               props.put(ConsumerConfig.GROUP_ID_CONFIG, "round-trip-consumer-group-" + id);
333               props.put(ConsumerConfig.AUTO_OFFSET_RESET_CONFIG, "earliest");
334               props.put(ConsumerConfig.REQUEST_TIMEOUT_MS_CONFIG, 105000);
335               props.put(ConsumerConfig.MAX_POLL_INTERVAL_MS_CONFIG, 100000);
336               // user may over-write the defaults with common client config and consumer config
337               WorkerUtils.addConfigsToProperties(props, spec.commonClientConf(), spec.consumerConf());
338               consumer = new KafkaConsumer<>(props, new ByteArrayDeserializer(),
339                   new ByteArrayDeserializer());
340               consumer.assign(partitions);
341           }
342   
343           @Override
344           public void run() {
345               long uniqueMessagesReceived = 0;
346               long messagesReceived = 0;
347               long pollInvoked = 0;
348               log.debug("{}: Starting RoundTripWorker#ConsumerRunnable.", id);
349               try {
350                   long lastLogTimeMs = Time.SYSTEM.milliseconds();
351                   while (true) {
352                       try {
353                           pollInvoked++;
354                           ConsumerRecords<byte[], byte[]> records = consumer.poll(Duration.ofMillis(50));
355                           for (Iterator<ConsumerRecord<byte[], byte[]>> iter = records.iterator(); iter.hasNext(); ) {
356                               ConsumerRecord<byte[], byte[]> record = iter.next();
357                               int messageIndex = ByteBuffer.wrap(record.key()).order(ByteOrder.LITTLE_ENDIAN).getInt();
358                               messagesReceived++;
359                               if (toReceiveTracker.removePending(messageIndex)) {
360                                   uniqueMessagesReceived++;
361                                   if (uniqueMessagesReceived >= spec.maxMessages()) {
362                                       lock.lock();
363                                       try {
364                                           log.info("{}: Consumer received the full count of {} unique messages.  " +
365                                               "Waiting for all {} sends to be acked...", id, spec.maxMessages(), unackedSends);
366                                           while (unackedSends > 0)
367                                               unackedSendsAreZero.await();
368                                       } finally {
369                                           lock.unlock();
370                                       }
371   
372                                       log.info("{}: all sends have been acked.", id);
373                                       new StatusUpdater().update();
374                                       doneFuture.complete("");
375                                       return;
376                                   }
377                               }
378                           }
379                           long curTimeMs = Time.SYSTEM.milliseconds();
380                           if (curTimeMs > lastLogTimeMs + LOG_INTERVAL_MS) {
381                               toReceiveTracker.log();
382                               lastLogTimeMs = curTimeMs;
383                           }
384                       } catch (WakeupException e) {
385                           log.debug("{}: Consumer got WakeupException", id, e);
386                       } catch (TimeoutException e) {
387                           log.debug("{}: Consumer got TimeoutException", id, e);
388                       }
389                   }
390               } catch (Throwable e) {
391                   WorkerUtils.abort(log, "ConsumerRunnable", e, doneFuture);
392               } finally {
393                   log.info("{}: ConsumerRunnable is exiting.  Invoked poll {} time(s).  " +
394                       "messagesReceived = {}; uniqueMessagesReceived = {}.",
395                       id, pollInvoked, messagesReceived, uniqueMessagesReceived);
396               }
397           }
398       }

401           @Override

440       @Override
441       public void stop(Platform platform) throws Exception {
442           if (!running.compareAndSet(true, false)) {
443               throw new IllegalStateException("RoundTripWorker is not running.");
444           }
445           log.info("{}: Deactivating RoundTripWorker.", id);
446           doneFuture.complete("");
447           executor.shutdownNow();
448           executor.awaitTermination(1, TimeUnit.DAYS);
449           Utils.closeQuietly(consumer, "consumer");
450           Utils.closeQuietly(producer, "producer");
451           this.consumer = null;
452           this.producer = null;
453           this.unackedSends = null;
454           this.executor = null;
455           this.doneFuture = null;
456           log.info("{}: Deactivated RoundTripWorker.", id);
457       }
