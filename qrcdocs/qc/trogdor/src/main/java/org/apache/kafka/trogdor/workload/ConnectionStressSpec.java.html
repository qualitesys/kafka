18   package org.apache.kafka.trogdor.workload;
19   
20   import com.fasterxml.jackson.annotation.JsonCreator;
21   import com.fasterxml.jackson.annotation.JsonProperty;
22   import org.apache.kafka.trogdor.task.TaskController;
23   import org.apache.kafka.trogdor.task.TaskSpec;
24   import org.apache.kafka.trogdor.task.TaskWorker;
25   
26   import java.util.ArrayList;
27   import java.util.Collections;
28   import java.util.List;
29   import java.util.Map;
30   import java.util.TreeSet;
31   
32   /**
33    * The specification for a task which connects and disconnects many times a
34    * second to stress the broker.
35    */
36   public class ConnectionStressSpec extends TaskSpec {
37       private final List<String> clientNodes;
38       private final String bootstrapServers;
39       private final Map<String, String> commonClientConf;
40       private final int targetConnectionsPerSec;
41       private final int numThreads;
42       private final ConnectionStressAction action;
43   
44       enum ConnectionStressAction {
45           CONNECT,
46           FETCH_METADATA
47       }
48   
49       @JsonCreator
50       public ConnectionStressSpec(@JsonProperty("startMs") long startMs,
51               @JsonProperty("durationMs") long durationMs,
52               @JsonProperty("clientNode") List<String> clientNodes,
53               @JsonProperty("bootstrapServers") String bootstrapServers,
54               @JsonProperty("commonClientConf") Map<String, String> commonClientConf,
55               @JsonProperty("targetConnectionsPerSec") int targetConnectionsPerSec,
56               @JsonProperty("numThreads") int numThreads,
57               @JsonProperty("action") ConnectionStressAction action) {
58           super(startMs, durationMs);
59           this.clientNodes = (clientNodes == null) ? Collections.emptyList() :
60               Collections.unmodifiableList(new ArrayList<>(clientNodes));
61           this.bootstrapServers = (bootstrapServers == null) ? "" : bootstrapServers;
62           this.commonClientConf = configOrEmptyMap(commonClientConf);
63           this.targetConnectionsPerSec = targetConnectionsPerSec;
64           this.numThreads = numThreads < 1 ? 1 : numThreads;
65           this.action = (action == null) ? ConnectionStressAction.CONNECT : action;
66       }
67   
68       @JsonProperty
69       public List<String> clientNode() {
70           return clientNodes;
71       }
72   
73       @JsonProperty
74       public String bootstrapServers() {
75           return bootstrapServers;
76       }
77   
78       @JsonProperty
79       public Map<String, String> commonClientConf() {
80           return commonClientConf;
81       }
82   
83       @JsonProperty
84       public int targetConnectionsPerSec() {
85           return targetConnectionsPerSec;
86       }
87   
88       @JsonProperty
89       public int numThreads() {
90           return numThreads;
91       }
92   
93       @JsonProperty
94       public ConnectionStressAction action() {
95           return action;
96       }
97   
98       public TaskController newController(String id) {
99           return topology -> new TreeSet<>(clientNodes);
100       }
101   
102       @Override
103       public TaskWorker newTaskWorker(String id) {
104           return new ConnectionStressWorker(id, this);
105       }
106   }
