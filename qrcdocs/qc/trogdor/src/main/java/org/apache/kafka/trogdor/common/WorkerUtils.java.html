58        *
59        * @param log               The logger to use.
60        * @param what              The component that had the exception.
61        * @param exception         The exception.
62        * @param doneFuture        The TaskWorker's doneFuture
63        * @throws KafkaException   A wrapped version of the exception.
64        */
65       public static void abort(Logger log, String what, Throwable exception,
66               KafkaFutureImpl<String> doneFuture) throws KafkaException {
67           log.warn("{} caught an exception", what, exception);
68           if (exception.getMessage() == null || exception.getMessage().isEmpty()) {
69               doneFuture.complete(exception.getClass().getCanonicalName());
70           } else {
71               doneFuture.complete(exception.getMessage());
72           }
73           throw new KafkaException(exception);
74       }

77        * Convert a rate expressed per second to a rate expressed per the given period.
78        *
79        * @param perSec            The per-second rate.
80        * @param periodMs          The new period to use.
81        * @return                  The rate per period.  This will never be less than 1.
82        */
83       public static int perSecToPerPeriod(float perSec, long periodMs) {
84           float period = ((float) periodMs) / 1000.0f;
85           float perPeriod = perSec * period;
86           perPeriod = Math.max(1.0f, perPeriod);
87           return (int) perPeriod;
88       }

113       /**
114        * Create some Kafka topics.
115        *
116        * @param log               The logger to use.
117        * @param bootstrapServers  The bootstrap server list.
118        * @param commonClientConf  Common client config
119        * @param adminClientConf   AdminClient config. This config has precedence over fields in
120        *                          common client config.
121        * @param topics            Maps topic names to partition assignments.
122        * @param failOnExisting    If true, the method will throw TopicExistsException if one or
123        *                          more topics already exist. Otherwise, the existing topics are

125        *                          partitions of an existing topic does not match the requested
126        *                          number of partitions, the method throws RuntimeException.
127        */
128       public static void createTopics(
129           Logger log, String bootstrapServers, Map<String, String> commonClientConf,
130           Map<String, String> adminClientConf,
131           Map<String, NewTopic> topics, boolean failOnExisting) throws Throwable {
132           // this method wraps the call to createTopics() that takes admin client, so that we can
133           // unit test the functionality with MockAdminClient. The exception is caught and
134           // re-thrown so that admin client is closed when the method returns.
135           try (Admin adminClient
136                    = createAdminClient(bootstrapServers, commonClientConf, adminClientConf)) {
137               createTopics(log, adminClient, topics, failOnExisting);
138           } catch (Exception e) {
139               log.warn("Failed to create or verify topics {}", topics, e);
140               throw e;
141           }
142       }

144       /**
145        * The actual create topics functionality is separated into this method and called from the
146        * above method to be able to unit test with mock adminClient.
147        * @throws TopicExistsException if the specified topic already exists.
148        * @throws UnknownTopicOrPartitionException if topic creation was issued but failed to verify if it was created.
149        * @throws Throwable if creation of one or more topics fails (except for the cases above).
150        */
151       static void createTopics(
152           Logger log, Admin adminClient,
153           Map<String, NewTopic> topics, boolean failOnExisting) throws Throwable {
154           if (topics.isEmpty()) {
155               log.warn("Request to create topics has an empty topic list.");
156               return;
157           }
158   
159           Collection<String> topicsExists = createTopics(log, adminClient, topics.values());
160           if (!topicsExists.isEmpty()) {
161               if (failOnExisting) {
162                   log.warn("Topic(s) {} already exist.", topicsExists);
163                   throw new TopicExistsException("One or more topics already exist.");
164               } else {
165                   verifyTopics(log, adminClient, topicsExists, topics, 3, 2500);
166               }
167           }
168       }

170       /**
171        * Creates Kafka topics and returns a list of topics that already exist
172        * @param log             The logger to use
173        * @param adminClient     AdminClient
174        * @param topics          List of topics to create
175        * @return                Collection of topics names that already exist.
176        * @throws Throwable if creation of one or more topics fails (except for topic exists case).
177        */
178       private static Collection<String> createTopics(Logger log, Admin adminClient,
179                                                      Collection<NewTopic> topics) throws Throwable {
180           long startMs = Time.SYSTEM.milliseconds();
181           int tries = 0;
182           List<String> existingTopics = new ArrayList<>();
183   
184           Map<String, NewTopic> newTopics = new HashMap<>();
185           for (NewTopic newTopic : topics) {
186               newTopics.put(newTopic.name(), newTopic);
187           }
188           List<String> topicsToCreate = new ArrayList<>(newTopics.keySet());
189           while (true) {
190               log.info("Attempting to create {} topics (try {})...", topicsToCreate.size(), ++tries);
191               Map<String, Future<Void>> creations = new HashMap<>();
192               while (!topicsToCreate.isEmpty()) {
193                   List<NewTopic> newTopicsBatch = new ArrayList<>();
194                   for (int i = 0; (i < MAX_CREATE_TOPICS_BATCH_SIZE) &&
195                                   !topicsToCreate.isEmpty(); i++) {
196                       String topicName = topicsToCreate.remove(0);
197                       newTopicsBatch.add(newTopics.get(topicName));
198                   }
199                   creations.putAll(adminClient.createTopics(newTopicsBatch).values());
200               }
201               // We retry cases where the topic creation failed with a
202               // timeout.  This is a workaround for KAFKA-6368.
203               for (Map.Entry<String, Future<Void>> entry : creations.entrySet()) {
204                   String topicName = entry.getKey();
205                   Future<Void> future = entry.getValue();
206                   try {
207                       future.get();
208                       log.debug("Successfully created {}.", topicName);
209                   } catch (Exception e) {
210                       if ((e.getCause() instanceof TimeoutException)
211                           || (e.getCause() instanceof NotEnoughReplicasException)) {
212                           log.warn("Attempt to create topic `{}` failed: {}", topicName,
213                                    e.getCause().getMessage());
214                           topicsToCreate.add(topicName);
215                       } else if (e.getCause() instanceof TopicExistsException) {
216                           log.info("Topic {} already exists.", topicName);
217                           existingTopics.add(topicName);
218                       } else {
219                           log.warn("Failed to create {}", topicName, e.getCause());
220                           throw e.getCause();
221                       }
222                   }
223               }
224               if (topicsToCreate.isEmpty()) {
225                   break;
226               }
227               if (Time.SYSTEM.milliseconds() > startMs + CREATE_TOPICS_CALL_TIMEOUT) {
228                   String str = "Unable to create topic(s): " +
229                                Utils.join(topicsToCreate, ", ") + "after " + tries + " attempt(s)";
230                   log.warn(str);
231                   throw new TimeoutException(str);
232               }
233           }
234           return existingTopics;
235       }

237       /**
238        * Verifies that topics in 'topicsToVerify' list have the same number of partitions as
239        * described in 'topicsInfo'
240        * @param log                The logger to use
241        * @param adminClient        AdminClient
242        * @param topicsToVerify     List of topics to verify
243        * @param topicsInfo         Map of topic name to topic description, which includes topics in
244        *                           'topicsToVerify' list.
245        * @param retryCount         The number of times to retry the fetching of the topics
246        * @param retryBackoffMs     The amount of time, in milliseconds, to wait in between retries
247        * @throws UnknownTopicOrPartitionException If at least one topic contained in 'topicsInfo'
248        * does not exist after retrying.
249        * @throws RuntimeException  If one or more topics have different number of partitions than
250        * described in 'topicsInfo'
251        */
252       static void verifyTopics(
253           Logger log, Admin adminClient,
254           Collection<String> topicsToVerify, Map<String, NewTopic> topicsInfo, int retryCount, long retryBackoffMs) throws Throwable {
255   
256           Map<String, TopicDescription> topicDescriptionMap = topicDescriptions(topicsToVerify, adminClient,
257                   retryCount, retryBackoffMs);
258   
259           for (TopicDescription desc: topicDescriptionMap.values()) {
260               // map will always contain the topic since all topics in 'topicsExists' are in given
261               // 'topics' map
262               int partitions = topicsInfo.get(desc.name()).numPartitions();
263               if (partitions != CreateTopicsRequest.NO_NUM_PARTITIONS && desc.partitions().size() != partitions) {
264                   String str = "Topic '" + desc.name() + "' exists, but has "
265                                + desc.partitions().size() + " partitions, while requested "
266                                + " number of partitions is " + partitions;
267                   log.warn(str);
268                   throw new RuntimeException(str);
269               }
270           }
271       }

273       private static Map<String, TopicDescription> topicDescriptions(Collection<String> topicsToVerify,
274                                                                      Admin adminClient,
275                                                                      int retryCount, long retryBackoffMs)
276               throws ExecutionException, InterruptedException {
277           UnknownTopicOrPartitionException lastException = null;
278           for (int i = 0; i < retryCount; i++) {
279               try {
280                   DescribeTopicsResult topicsResult = adminClient.describeTopics(
281                           topicsToVerify, new DescribeTopicsOptions().timeoutMs(ADMIN_REQUEST_TIMEOUT));
282                   return topicsResult.allTopicNames().get();
283               } catch (ExecutionException exception) {
284                   if (exception.getCause() instanceof UnknownTopicOrPartitionException) {
285                       lastException = (UnknownTopicOrPartitionException) exception.getCause();
286                       Thread.sleep(retryBackoffMs);
287                   } else {
288                       throw exception;
289                   }
290               }
291           }
292           throw lastException;
293       }

295       /**
296        * Returns list of existing, not internal, topics/partitions that match given pattern and
297        * where partitions are in range [startPartition, endPartition]
298        * @param adminClient     AdminClient
299        * @param topicRegex      Topic regular expression to match
300        * @return                List of topic names
301        * @throws Throwable      If failed to get list of existing topics
302        */
303       static Collection<TopicPartition> getMatchingTopicPartitions(
304           Admin adminClient, String topicRegex, int startPartition, int endPartition)
305           throws Throwable {
306           final Pattern topicNamePattern = Pattern.compile(topicRegex);
307   
308           // first get list of matching topics
309           List<String> matchedTopics = new ArrayList<>();
310           ListTopicsResult res = adminClient.listTopics(
311               new ListTopicsOptions().timeoutMs(ADMIN_REQUEST_TIMEOUT));
312           Map<String, TopicListing> topicListingMap = res.namesToListings().get();
313           for (Map.Entry<String, TopicListing> topicListingEntry: topicListingMap.entrySet()) {
314               if (!topicListingEntry.getValue().isInternal()
315                   && topicNamePattern.matcher(topicListingEntry.getKey()).matches()) {
316                   matchedTopics.add(topicListingEntry.getKey());
317               }
318           }
319   
320           // create a list of topic/partitions
321           List<TopicPartition> out = new ArrayList<>();
322           DescribeTopicsResult topicsResult = adminClient.describeTopics(
323               matchedTopics, new DescribeTopicsOptions().timeoutMs(ADMIN_REQUEST_TIMEOUT));
324           Map<String, TopicDescription> topicDescriptionMap = topicsResult.allTopicNames().get();
325           for (TopicDescription desc: topicDescriptionMap.values()) {
326               List<TopicPartitionInfo> partitions = desc.partitions();
327               for (TopicPartitionInfo info: partitions) {
328                   if ((info.partition() >= startPartition) && (info.partition() <= endPartition)) {
329                       out.add(new TopicPartition(desc.name(), info.partition()));
330                   }
331               }
332           }
333           return out;
334       }

336       private static Admin createAdminClient(
337           String bootstrapServers,
338           Map<String, String> commonClientConf, Map<String, String> adminClientConf) {
339           Properties props = new Properties();
340           props.put(AdminClientConfig.BOOTSTRAP_SERVERS_CONFIG, bootstrapServers);
341           props.put(AdminClientConfig.REQUEST_TIMEOUT_MS_CONFIG, ADMIN_REQUEST_TIMEOUT);
342           // first add common client config, and then admin client config to properties, possibly
343           // over-writing default or common properties.
344           addConfigsToProperties(props, commonClientConf, adminClientConf);
345           return Admin.create(props);
346       }
