17   package org.apache.kafka.trogdor.workload;
18   
19   import com.fasterxml.jackson.annotation.JsonCreator;
20   import com.fasterxml.jackson.annotation.JsonProperty;
21   
22   /**
23    * A PayloadGenerator which always generates a constant payload.
24    */
25   public class ConstantPayloadGenerator implements PayloadGenerator {
26       private final int size;
27       private final byte[] value;
28   
29       @JsonCreator
30       public ConstantPayloadGenerator(@JsonProperty("size") int size,
31                                       @JsonProperty("value") byte[] value) {
32           this.size = size;
33           this.value = (value == null || value.length == 0) ? new byte[size] : value;
34       }
35   
36       @JsonProperty
37       public int size() {
38           return size;
39       }
40   
41       @JsonProperty
42       public byte[] value() {
43           return value;
44       }
45   
46       @Override
47       public byte[] generate(long position) {
48           byte[] next = new byte[size];
49           for (int i = 0; i < next.length; i += value.length) {
50               System.arraycopy(value, 0, next, i, Math.min(next.length - i, value.length));
51           }
52           return next;
53       }
54   }
