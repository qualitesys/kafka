18   package org.apache.kafka.trogdor.workload;
19   
20   import com.fasterxml.jackson.annotation.JsonCreator;
21   import com.fasterxml.jackson.annotation.JsonProperty;
22   import org.apache.kafka.common.utils.Time;
23   
24   import java.nio.ByteOrder;
25   import java.util.Random;
26   import java.nio.ByteBuffer;
27   
28   /**
29    * A PayloadGenerator which generates a timestamped uniform random payload.
30    *
31    * This generator generates pseudo-random payloads that can be reproduced from run to run.
32    * The guarantees are the same as those of java.util.Random.
33    *
34    * The timestamp used for this class is in milliseconds since epoch, encoded directly to the first several bytes of the
35    * payload.
36    *
37    * This should be used in conjunction with TimestampRecordProcessor in the Consumer to measure true end-to-end latency
38    * of a system.
39    *
40    * `size` - The size in bytes of each message.
41    * `seed` - Used to initialize Random() to remove some non-determinism.
42    *
43    * Here is an example spec:
44    *
45    * {
46    *    "type": "timestampRandom",
47    *    "size": 512
48    * }
49    *
50    * This will generate a 512-byte random message with the first several bytes encoded with the timestamp.
51    */
52   public class TimestampRandomPayloadGenerator implements PayloadGenerator {
53       private final int size;
54       private final long seed;
55   
56       private final byte[] randomBytes;
57       private final ByteBuffer buffer;
58   
59       private final Random random = new Random();
60   
61       @JsonCreator
62       public TimestampRandomPayloadGenerator(@JsonProperty("size") int size,
63                                              @JsonProperty("seed") long seed) {
64           this.size = size;
65           this.seed = seed;
66           if (size < Long.BYTES) {
67               throw new RuntimeException("The size of the payload must be greater than or equal to " + Long.BYTES + ".");
68           }
69           random.setSeed(seed);
70           this.randomBytes = new byte[size - Long.BYTES];
71           buffer = ByteBuffer.allocate(Long.BYTES);
72           buffer.order(ByteOrder.LITTLE_ENDIAN);
73       }
74   
75       @JsonProperty
76       public int size() {
77           return size;
78       }
79   
80       @JsonProperty
81       public long seed() {
82           return seed;
83       }
84   
85       @Override
86       public synchronized byte[] generate(long position) {
87           // Generate out of order to prevent inclusion of random number generation in latency numbers.
88           byte[] result = new byte[size];
89           if (randomBytes.length > 0) {
90               random.setSeed(seed + position);
91               random.nextBytes(randomBytes);
92               System.arraycopy(randomBytes, 0, result, Long.BYTES, randomBytes.length);
93           }
94   
95           // Do the timestamp generation as the very last task.
96           buffer.clear();
97           buffer.putLong(Time.SYSTEM.milliseconds());
98           buffer.rewind();
99           System.arraycopy(buffer.array(), 0, result, 0, Long.BYTES);
100           return result;
101       }
102   }
