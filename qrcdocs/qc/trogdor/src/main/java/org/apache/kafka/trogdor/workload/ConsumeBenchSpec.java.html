40   /**
41    * The specification for a benchmark that consumer messages from a set of topic/partitions.
42    *
43    * If a consumer group is not given to the specification, a random one will be generated and
44    *  used to track offsets/subscribe to topics.
45    *
46    * This specification uses a specific way to represent a topic partition via its "activeTopics" field.
47    * The notation for that is topic_name:partition_number (e.g "foo:1" represents partition-1 of topic "foo")
48    * Note that a topic name cannot have more than one colon.
49    *
50    * The "activeTopics" field also supports ranges that get expanded. See #{@link StringExpander}.
51    *
52    * There now exists a clever and succinct way to represent multiple partitions of multiple topics.
53    * Example:
54    * Given "activeTopics": ["foo[1-3]:[1-3]"], "foo[1-3]:[1-3]" will get
55    * expanded to [foo1:1, foo1:2, foo1:3, foo2:1, ..., foo3:3].
56    * This represents all partitions 1-3 for the three topics foo1, foo2 and foo3.
57    *
58    * If there is at least one topic:partition pair, the consumer will be manually assigned partitions via
59    * #{@link org.apache.kafka.clients.consumer.KafkaConsumer#assign(Collection)}.
60    * Note that in this case the consumer will fetch and assign all partitions for a topic if no partition is given for it (e.g ["foo:1", "bar"])
61    *
62    * If there are no topic:partition pairs given, the consumer will subscribe to the topics via
63    * #{@link org.apache.kafka.clients.consumer.KafkaConsumer#subscribe(Collection)}.
64    * It will be assigned partitions dynamically from the consumer group.
65    *
66    * This specification supports the spawning of multiple consumers in the single Trogdor worker agent.
67    * The "threadsPerWorker" field denotes how many consumers should be spawned for this spec.
68    * It is worth noting that the "targetMessagesPerSec", "maxMessages" and "activeTopics" fields apply for every consumer individually.
69    *
70    * If a consumer group is not specified, every consumer is assigned a different, random group. When specified, all consumers use the same group.
71    * Since no two consumers in the same group can be assigned the same partition,
72    * explicitly specifying partitions in "activeTopics" when there are multiple "threadsPerWorker"
73    * and a particular "consumerGroup" will result in an #{@link ConfigException}, aborting the task.
74    *
75    * The "recordProcessor" field allows the specification of tasks to run on records that are consumed.  This is run
76    * immediately after the messages are polled.  See the `RecordProcessor` interface for more information.
77    *
78    * An example JSON representation which will result in a consumer that is part of the consumer group "cg" and
79    * subscribed to topics foo1, foo2, foo3 and bar.
80    * #{@code
81    *    {
82    *        "class": "org.apache.kafka.trogdor.workload.ConsumeBenchSpec",
83    *        "durationMs": 10000000,
84    *        "consumerNode": "node0",
85    *        "bootstrapServers": "localhost:9092",
86    *        "maxMessages": 100,
87    *        "consumerGroup": "cg",
88    *        "activeTopics": ["foo[1-3]", "bar"]
89    *    }
90    * }
91    */
92   public class ConsumeBenchSpec extends TaskSpec {

200       /**
201        * Materializes a list of topic names (optionally with ranges) into a map of the topics and their partitions
202        *
203        * Example:
204        * ['foo[1-3]', 'foobar:2', 'bar[1-2]:[1-2]'] => {'foo1': [], 'foo2': [], 'foo3': [], 'foobar': [2],
205        *                                                'bar1': [1, 2], 'bar2': [1, 2] }
206        */
207       Map<String, List<TopicPartition>> materializeTopics() {
208           Map<String, List<TopicPartition>> partitionsByTopics = new HashMap<>();
209   
210           for (String rawTopicName : this.activeTopics) {
211               Set<String> expandedNames = expandTopicName(rawTopicName);
212               if (!expandedNames.iterator().next().matches(VALID_EXPANDED_TOPIC_NAME_PATTERN))
213                   throw new IllegalArgumentException(String.format("Expanded topic name %s is invalid", rawTopicName));
214   
215               for (String topicName : expandedNames) {
216                   TopicPartition partition = null;
217                   if (topicName.contains(":")) {
218                       String[] topicAndPartition = topicName.split(":");
219                       topicName = topicAndPartition[0];
220                       partition = new TopicPartition(topicName, Integer.parseInt(topicAndPartition[1]));
221                   }
222                   if (!partitionsByTopics.containsKey(topicName)) {
223                       partitionsByTopics.put(topicName, new ArrayList<>());
224                   }
225                   if (partition != null) {
226                       partitionsByTopics.get(topicName).add(partition);
227                   }
228               }
229           }
230   
231           return partitionsByTopics;
232       }

234       /**
235        * Expands a topic name until there are no more ranges in it
236        */
237       private Set<String> expandTopicName(String topicName) {
238           Set<String> expandedNames = StringExpander.expand(topicName);
239           if (expandedNames.size() == 1) {
240               return expandedNames;
241           }
242   
243           Set<String> newNames = new HashSet<>();
244           for (String name : expandedNames) {
245               newNames.addAll(expandTopicName(name));
246           }
247           return newNames;
248       }
