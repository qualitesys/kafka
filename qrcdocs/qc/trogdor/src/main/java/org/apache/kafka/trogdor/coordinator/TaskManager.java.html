65   /**
66    * The TaskManager is responsible for managing tasks inside the Trogdor coordinator.
67    *
68    * The task manager has a single thread, managed by the executor.  We start, stop,
69    * and handle state changes to tasks by adding requests to the executor queue.
70    * Because the executor is single threaded, no locks are needed when accessing
71    * TaskManager data structures.
72    *
73    * The TaskManager maintains a state machine for each task.  Tasks begin in the
74    * PENDING state, waiting for their designated start time to arrive.
75    * When their time arrives, they transition to the RUNNING state.  In this state,
76    * the NodeManager will start them, and monitor them.
77    *
78    * The TaskManager does not handle communication with the agents.  This is handled
79    * by the NodeManagers.  There is one NodeManager per node being managed.
80    * See {org.apache.kafka.trogdor.coordinator.NodeManager} for details.
81    */
82   public final class TaskManager {
83       private static final Logger log = LoggerFactory.getLogger(TaskManager.class);

148       class ManagedTask {
149           /**
150            * The task id.
151            */
152           final private String id;
153   
154           /**
155            * The original task specification as submitted when the task was created.
156            */
157           final private TaskSpec originalSpec;
158   
159           /**
160            * The effective task specification.
161            * The start time will be adjusted to reflect the time when the task was submitted.
162            */
163           final private TaskSpec spec;
164   
165           /**
166            * The task controller.
167            */
168           final private TaskController controller;
169   
170           /**
171            * The task state.
172            */
173           private TaskStateType state;
174   
175           /**
176            * The time when the task was started, or -1 if the task has not been started.
177            */
178           private long startedMs = -1;
179   
180           /**
181            * The time when the task was finished, or -1 if the task has not been finished.
182            */
183           private long doneMs = -1;
184   
185           /**
186            * True if the task was cancelled by a stop request.
187            */
188           boolean cancelled = false;
189   
190           /**
191            * If there is a task start scheduled, this is a future which can
192            * be used to cancel it.
193            */
194           private Future<?> startFuture = null;
195   
196           /**
197            * Maps node names to worker IDs.
198            */
199           public TreeMap<String, Long> workerIds = new TreeMap<>();
200   
201           /**
202            * If this is non-empty, a message describing how this task failed.
203            */
204           private String error = "";
205   
206           ManagedTask(String id, TaskSpec originalSpec, TaskSpec spec,
207                       TaskController controller, TaskStateType state) {
208               this.id = id;
209               this.originalSpec = originalSpec;
210               this.spec = spec;
211               this.controller = controller;
212               this.state = state;
213           }
214   
215           void clearStartFuture() {
216               if (startFuture != null) {
217                   startFuture.cancel(false);
218                   startFuture = null;
219               }
220           }
221   
222           long startDelayMs(long now) {
223               if (now > spec.startMs()) {
224                   return 0;
225               }
226               return spec.startMs() - now;
227           }
228   
229           TreeSet<String> findNodeNames() {
230               Set<String> nodeNames = controller.targetNodes(platform.topology());
231               TreeSet<String> validNodeNames = new TreeSet<>();
232               TreeSet<String> nonExistentNodeNames = new TreeSet<>();
233               for (String nodeName : nodeNames) {
234                   if (nodeManagers.containsKey(nodeName)) {
235                       validNodeNames.add(nodeName);
236                   } else {
237                       nonExistentNodeNames.add(nodeName);
238                   }
239               }
240               if (!nonExistentNodeNames.isEmpty()) {
241                   throw new KafkaException("Unknown node names: " +
242                           Utils.join(nonExistentNodeNames, ", "));
243               }
244               if (validNodeNames.isEmpty()) {
245                   throw new KafkaException("No node names specified.");
246               }
247               return validNodeNames;
248           }
249   
250           void maybeSetError(String newError) {
251               if (error.isEmpty()) {
252                   error = newError;
253               }
254           }
255   
256           TaskState taskState() {
257               switch (state) {
258                   case PENDING:
259                       return new TaskPending(spec);
260                   case RUNNING:
261                       return new TaskRunning(spec, startedMs, getCombinedStatus());
262                   case STOPPING:
263                       return new TaskStopping(spec, startedMs, getCombinedStatus());
264                   case DONE:
265                       return new TaskDone(spec, startedMs, doneMs, error, cancelled, getCombinedStatus());
266               }
267               throw new RuntimeException("unreachable");
268           }
269   
270           private JsonNode getCombinedStatus() {
271               if (workerIds.size() == 1) {
272                   return workerStates.get(workerIds.values().iterator().next()).status();
273               } else {
274                   ObjectNode objectNode = new ObjectNode(JsonNodeFactory.instance);
275                   for (Map.Entry<String, Long> entry : workerIds.entrySet()) {
276                       String nodeName = entry.getKey();
277                       Long workerId = entry.getValue();
278                       WorkerState state = workerStates.get(workerId);
279                       JsonNode node = state.status();
280                       if (node != null) {
281                           objectNode.set(nodeName, node);
282                       }
283                   }
284                   return objectNode;
285               }
286           }
287   
288           TreeMap<String, Long> activeWorkerIds() {
289               TreeMap<String, Long> activeWorkerIds = new TreeMap<>();
290               for (Map.Entry<String, Long> entry : workerIds.entrySet()) {
291                   WorkerState workerState = workerStates.get(entry.getValue());
292                   if (!workerState.done()) {
293                       activeWorkerIds.put(entry.getKey(), entry.getValue());
294                   }
295               }
296               return activeWorkerIds;
297           }
298       }

317       /**
318        * Handles a request to create a new task.  Processed by the state change thread.
319        */
320       class CreateTask implements Callable<Void> {
321           private final String id;
322           private final TaskSpec originalSpec;
323           private final TaskSpec spec;
324   
325           CreateTask(String id, TaskSpec spec) throws JsonProcessingException {
326               this.id = id;
327               this.originalSpec = spec;
328               ObjectNode node = JsonUtil.JSON_SERDE.valueToTree(originalSpec);
329               node.set("startMs", new LongNode(Math.max(time.milliseconds(), originalSpec.startMs())));
330               this.spec = JsonUtil.JSON_SERDE.treeToValue(node, TaskSpec.class);
331           }
332   
333           @Override
334           public Void call() throws Exception {
335               if (id.isEmpty()) {
336                   throw new InvalidRequestException("Invalid empty ID in createTask request.");
337               }
338               ManagedTask task = tasks.get(id);
339               if (task != null) {
340                   if (!task.originalSpec.equals(originalSpec)) {
341                       throw new RequestConflictException("Task ID " + id + " already " +
342                           "exists, and has a different spec " + task.originalSpec);
343                   }
344                   log.info("Task {} already exists with spec {}", id, originalSpec);
345                   return null;
346               }
347               TaskController controller = null;
348               String failure = null;
349               try {
350                   controller = spec.newController(id);
351               } catch (Throwable t) {
352                   failure = "Failed to create TaskController: " + t.getMessage();
353               }
354               if (failure != null) {
355                   log.info("Failed to create a new task {} with spec {}: {}",
356                       id, spec, failure);
357                   task = new ManagedTask(id, originalSpec, spec, null, TaskStateType.DONE);
358                   task.doneMs = time.milliseconds();
359                   task.maybeSetError(failure);
360                   tasks.put(id, task);
361                   return null;
362               }
363               task = new ManagedTask(id, originalSpec, spec, controller, TaskStateType.PENDING);
364               tasks.put(id, task);
365               long delayMs = task.startDelayMs(time.milliseconds());
366               task.startFuture = scheduler.schedule(executor, new RunTask(task), delayMs);
367               log.info("Created a new task {} with spec {}, scheduled to start {} ms from now.",
368                       id, spec, delayMs);
369               return null;
370           }
371       }
372   
373       /**
374        * Handles starting a task.  Processed by the state change thread.
375        */
376       class RunTask implements Callable<Void> {
377           private final ManagedTask task;
378   
379           RunTask(ManagedTask task) {
380               this.task = task;
381           }
382   
383           @Override
384           public Void call() throws Exception {
385               task.clearStartFuture();
386               if (task.state != TaskStateType.PENDING) {
387                   log.info("Can't start task {}, because it is already in state {}.",
388                       task.id, task.state);
389                   return null;
390               }
391               TreeSet<String> nodeNames;
392               try {
393                   nodeNames = task.findNodeNames();
394               } catch (Exception e) {
395                   log.error("Unable to find nodes for task {}", task.id, e);
396                   task.doneMs = time.milliseconds();
397                   task.state = TaskStateType.DONE;
398                   task.maybeSetError("Unable to find nodes for task: " + e.getMessage());
399                   return null;
400               }
401               log.info("Running task {} on node(s): {}", task.id, Utils.join(nodeNames, ", "));
402               task.state = TaskStateType.RUNNING;
403               task.startedMs = time.milliseconds();
404               for (String workerName : nodeNames) {
405                   long workerId = nextWorkerId++;
406                   task.workerIds.put(workerName, workerId);
407                   workerStates.put(workerId, new WorkerReceiving(task.id, task.spec));
408                   nodeManagers.get(workerName).createWorker(workerId, task.id, task.spec);
409               }
410               return null;
411           }
412       }

428       /**
429        * Handles cancelling a task.  Processed by the state change thread.
430        */
431       class CancelTask implements Callable<Void> {
432           private final String id;
433   
434           CancelTask(String id) {
435               this.id = id;
436           }
437   
438           @Override
439           public Void call() throws Exception {
440               if (id.isEmpty()) {
441                   throw new InvalidRequestException("Invalid empty ID in stopTask request.");
442               }
443               ManagedTask task = tasks.get(id);
444               if (task == null) {
445                   log.info("Can't cancel non-existent task {}.", id);
446                   return null;
447               }
448               switch (task.state) {
449                   case PENDING:
450                       task.cancelled = true;
451                       task.clearStartFuture();
452                       task.doneMs = time.milliseconds();
453                       task.state = TaskStateType.DONE;
454                       log.info("Stopped pending task {}.", id);
455                       break;
456                   case RUNNING:
457                       task.cancelled = true;
458                       TreeMap<String, Long> activeWorkerIds = task.activeWorkerIds();
459                       if (activeWorkerIds.isEmpty()) {
460                           if (task.error.isEmpty()) {
461                               log.info("Task {} is now complete with no errors.", id);
462                           } else {
463                               log.info("Task {} is now complete with error: {}", id, task.error);
464                           }
465                           task.doneMs = time.milliseconds();
466                           task.state = TaskStateType.DONE;
467                       } else {
468                           for (Map.Entry<String, Long> entry : activeWorkerIds.entrySet()) {
469                               nodeManagers.get(entry.getKey()).stopWorker(entry.getValue());
470                           }
471                           log.info("Cancelling task {} with worker(s) {}",
472                               id, Utils.mkString(activeWorkerIds, "", "", " = ", ", "));
473                           task.state = TaskStateType.STOPPING;
474                       }
475                       break;
476                   case STOPPING:
477                       log.info("Can't cancel task {} because it is already stopping.", id);
478                       break;
479                   case DONE:
480                       log.info("Can't cancel task {} because it is already done.", id);
481                       break;
482               }
483               return null;
484           }
485       }

496       /**
497        * Handles destroying a task.  Processed by the state change thread.
498        */
499       class DestroyTask implements Callable<Void> {
500           private final String id;

502           DestroyTask(String id) {

539       /**
540        * Updates the state of a worker.  Process by the state change thread.
541        */
542       class UpdateWorkerState implements Callable<Void> {
543           private final String nodeName;
544           private final long workerId;
545           private final WorkerState nextState;
546   
547           UpdateWorkerState(String nodeName, long workerId, WorkerState nextState) {
548               this.nodeName = nodeName;
549               this.workerId = workerId;
550               this.nextState = nextState;
551           }
552   
553           @Override
554           public Void call() throws Exception {
555               try {
556                   WorkerState prevState = workerStates.get(workerId);
557                   if (prevState == null) {
558                       throw new RuntimeException("Unable to find workerId " + workerId);
559                   }
560                   ManagedTask task = tasks.get(prevState.taskId());
561                   if (task == null) {
562                       throw new RuntimeException("Unable to find taskId " + prevState.taskId());
563                   }
564                   log.debug("Task {}: Updating worker state for {} on {} from {} to {}.",
565                       task.id, workerId, nodeName, prevState, nextState);
566                   workerStates.put(workerId, nextState);
567                   if (nextState.done() && (!prevState.done())) {
568                       handleWorkerCompletion(task, nodeName, (WorkerDone) nextState);
569                   }
570               } catch (Exception e) {
571                   log.error("Error updating worker state for {} on {}.  Stopping worker.",
572                       workerId, nodeName, e);
573                   nodeManagers.get(nodeName).stopWorker(workerId);
574               }
575               return null;
576           }
577       }
578   
579       /**
580        * Handle a worker being completed.
581        *
582        * @param task      The task that owns the worker.
583        * @param nodeName  The name of the node on which the worker is running.
584        * @param state     The worker state.
585        */
586       private void handleWorkerCompletion(ManagedTask task, String nodeName, WorkerDone state) {
587           if (state.error().isEmpty()) {
588               log.info("{}: Worker {} finished with status '{}'",
589                   nodeName, task.id, JsonUtil.toJsonString(state.status()));
590           } else {
591               log.warn("{}: Worker {} finished with error '{}' and status '{}'",
592                   nodeName, task.id, state.error(), JsonUtil.toJsonString(state.status()));
593               task.maybeSetError(state.error());
594           }
595           TreeMap<String, Long> activeWorkerIds = task.activeWorkerIds();
596           if (activeWorkerIds.isEmpty()) {
597               task.doneMs = time.milliseconds();
598               task.state = TaskStateType.DONE;
599               log.info("{}: Task {} is now complete on {} with error: {}",
600                   nodeName, task.id, Utils.join(task.workerIds.keySet(), ", "),
601                   task.error.isEmpty() ? "(none)" : task.error);
602           } else if ((task.state == TaskStateType.RUNNING) && (!task.error.isEmpty())) {
603               log.info("{}: task {} stopped with error {}.  Stopping worker(s): {}",
604                   nodeName, task.id, task.error, Utils.mkString(activeWorkerIds, "{", "}", ": ", ", "));
605               task.state = TaskStateType.STOPPING;
606               for (Map.Entry<String, Long> entry : activeWorkerIds.entrySet()) {
607                   nodeManagers.get(entry.getKey()).stopWorker(entry.getValue());
608               }
609           }
610       }
