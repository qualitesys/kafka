28   /**
29    * Defines a cluster topology
30    */
31   public interface Platform {
32       class Config {
33           public static final String TROGDOR_AGENT_PORT = "trogdor.agent.port";
34   
35           public static final String TROGDOR_COORDINATOR_PORT = "trogdor.coordinator.port";
36   
37           public static final String TROGDOR_COORDINATOR_HEARTBEAT_MS =
38               "trogdor.coordinator.heartbeat.ms";
39   
40           public static final int TROGDOR_COORDINATOR_HEARTBEAT_MS_DEFAULT = 60000;
41   
42           public static Platform parse(String curNodeName, String path) throws Exception {
43               JsonNode root = JsonUtil.JSON_SERDE.readTree(new File(path));
44               JsonNode platformNode = root.get("platform");
45               if (platformNode == null) {
46                   throw new RuntimeException("Expected to find a 'platform' field " +
47                       "in the root JSON configuration object");
48               }
49               String platformName = platformNode.textValue();
50               return Utils.newParameterizedInstance(platformName,
51                   String.class, curNodeName,
52                   Scheduler.class, Scheduler.SYSTEM,
53                   JsonNode.class, root);
54           }
55       }
56   
57       /**
58        * Get name for this platform.
59        */
60       String name();
61   
62       /**
63        * Get the current node.
64        */
65       Node curNode();
66   
67       /**
68        * Get the cluster topology.
69        */
70       Topology topology();
71   
72       /**
73        * Get the scheduler to use.
74        */
75       Scheduler scheduler();
76   
77       /**
78        * Run a command on this local node.
79        *
80        * Throws an exception if the command could not be run, or if the
81        * command returned a non-zero error status.
82        *
83        * @param command   The command
84        *
85        * @return          The command output.
86        */
87       String runCommand(String[] command) throws IOException;
88   }
