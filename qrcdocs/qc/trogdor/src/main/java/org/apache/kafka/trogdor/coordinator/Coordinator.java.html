22   import net.sourceforge.argparse4j.inf.ArgumentParserException;

40   import org.slf4j.LoggerFactory;
41   
42   import java.util.concurrent.ThreadLocalRandom;

44   import static net.sourceforge.argparse4j.impl.Arguments.store;

46   /**
47    * The Trogdor coordinator.
48    *
49    * The coordinator manages the agent processes in the cluster. 
50    */
51   public final class Coordinator {
52       private static final Logger log = LoggerFactory.getLogger(Coordinator.class);
53   
54       public static final int DEFAULT_PORT = 8889;
55   
56       /**
57        * The start time of the Coordinator in milliseconds.
58        */
59       private final long startTimeMs;
60   
61       /**
62        * The task manager.
63        */
64       private final TaskManager taskManager;
65   
66       /**
67        * The REST server.
68        */
69       private final JsonRestServer restServer;
70   
71       private final Time time;
72   
73       /**
74        * Create a new Coordinator.
75        *
76        * @param platform      The platform object to use.
77        * @param scheduler     The scheduler to use for this Coordinator.
78        * @param restServer    The REST server to use.
79        * @param resource      The AgentRestResource to use.
80        */
81       public Coordinator(Platform platform, Scheduler scheduler, JsonRestServer restServer,
82                          CoordinatorRestResource resource, long firstWorkerId) {
83           this.time = scheduler.time();
84           this.startTimeMs = time.milliseconds();
85           this.taskManager = new TaskManager(platform, scheduler, firstWorkerId);
86           this.restServer = restServer;
87           resource.setCoordinator(this);
88       }
89   
90       public int port() {
91           return this.restServer.port();
92       }
93   
94       public CoordinatorStatusResponse status() throws Exception {
95           return new CoordinatorStatusResponse(startTimeMs);
96       }
97   
98       public UptimeResponse uptime() {
99           return new UptimeResponse(startTimeMs, time.milliseconds());
100       }
101   
102       public void createTask(CreateTaskRequest request) throws Throwable {
103           taskManager.createTask(request.id(), request.spec());
104       }
105   
106       public void stopTask(StopTaskRequest request) throws Throwable {
107           taskManager.stopTask(request.id());
108       }
109   
110       public void destroyTask(DestroyTaskRequest request) throws Throwable {
111           taskManager.destroyTask(request.id());
112       }
113   
114       public TasksResponse tasks(TasksRequest request) throws Exception {
115           return taskManager.tasks(request);
116       }
117   
118       public TaskState task(TaskRequest request) throws Exception {
119           return taskManager.task(request);
120       }
121   
122       public void beginShutdown(boolean stopAgents) throws Exception {
123           restServer.beginShutdown();
124           taskManager.beginShutdown(stopAgents);
125       }
126   
127       public void waitForShutdown() throws Exception {
128           restServer.waitForShutdown();
129           taskManager.waitForShutdown();
130       }
131   
132       public static void main(String[] args) throws Exception {
133           ArgumentParser parser = ArgumentParsers
134               .newArgumentParser("trogdor-coordinator")
135               .defaultHelp(true)
136               .description("The Trogdor fault injection coordinator");
137           parser.addArgument("--coordinator.config", "-c")
138               .action(store())
139               .required(true)
140               .type(String.class)
141               .dest("config")
142               .metavar("CONFIG")
143               .help("The configuration file to use.");
144           parser.addArgument("--node-name", "-n")
145               .action(store())
146               .required(true)
147               .type(String.class)
148               .dest("node_name")
149               .metavar("NODE_NAME")
150               .help("The name of this node.");
151           Namespace res = null;
152           try {
153               res = parser.parseArgs(args);
154           } catch (ArgumentParserException e) {
155               if (args.length == 0) {
156                   parser.printHelp();
157                   Exit.exit(0);
158               } else {
159                   parser.handleError(e);
160                   Exit.exit(1);
161               }
162           }
163           String configPath = res.getString("config");
164           String nodeName = res.getString("node_name");
165   
166           Platform platform = Platform.Config.parse(nodeName, configPath);
167           JsonRestServer restServer = new JsonRestServer(
168               Node.Util.getTrogdorCoordinatorPort(platform.curNode()));
169           CoordinatorRestResource resource = new CoordinatorRestResource();
170           log.info("Starting coordinator process.");
171           final Coordinator coordinator = new Coordinator(platform, Scheduler.SYSTEM,
172               restServer, resource, ThreadLocalRandom.current().nextLong(0, Long.MAX_VALUE / 2));
173           restServer.start(resource);
174           Exit.addShutdownHook("coordinator-shutdown-hook", () -> {
175               log.warn("Running coordinator shutdown hook.");
176               try {
177                   coordinator.beginShutdown(false);
178                   coordinator.waitForShutdown();
179               } catch (Exception e) {
180                   log.error("Got exception while running coordinator shutdown hook.", e);
181               }
182           });
183           coordinator.waitForShutdown();
184       }
185   };
