50   /**
51    * The Trogdor agent.
52    *
53    * The agent process runs tasks.
54    */
55   public final class Agent {
56       private static final Logger log = LoggerFactory.getLogger(Agent.class);
57   
58       /**
59        * The default Agent port.
60        */
61       public static final int DEFAULT_PORT = 8888;
62   
63       /**
64        * The workerId to use in exec mode.
65        */
66       private static final long EXEC_WORKER_ID = 1;
67   
68       /**
69        * The taskId to use in exec mode.
70        */
71       private static final String EXEC_TASK_ID = "task0";
72   
73       /**
74        * The platform object to use for this agent.
75        */
76       private final Platform platform;
77   
78       /**
79        * The time at which this server was started.
80        */
81       private final long serverStartMs;
82   
83       /**
84        * The WorkerManager.
85        */
86       private final WorkerManager workerManager;
87   
88       /**
89        * The REST server.
90        */
91       private final JsonRestServer restServer;
92   
93       private final Time time;
94   
95       /**
96        * Create a new Agent.
97        *
98        * @param platform      The platform object to use.
99        * @param scheduler     The scheduler to use for this Agent.
100        * @param restServer    The REST server to use.
101        * @param resource      The AgentRestResource to use.
102        */
103       public Agent(Platform platform, Scheduler scheduler,
104                    JsonRestServer restServer, AgentRestResource resource) {
105           this.platform = platform;
106           this.time = scheduler.time();
107           this.serverStartMs = time.milliseconds();
108           this.workerManager = new WorkerManager(platform, scheduler);
109           this.restServer = restServer;
110           resource.setAgent(this);
111       }
112   
113       public int port() {
114           return this.restServer.port();
115       }
116   
117       public void beginShutdown() throws Exception {
118           restServer.beginShutdown();
119           workerManager.beginShutdown();
120       }
121   
122       public void waitForShutdown() throws Exception {
123           restServer.waitForShutdown();
124           workerManager.waitForShutdown();
125       }
126   
127       public AgentStatusResponse status() throws Exception {
128           return new AgentStatusResponse(serverStartMs, workerManager.workerStates());
129       }
130   
131       public UptimeResponse uptime() {
132           return new UptimeResponse(serverStartMs, time.milliseconds());
133       }
134   
135       public void createWorker(CreateWorkerRequest req) throws Throwable {
136           workerManager.createWorker(req.workerId(), req.taskId(), req.spec());
137       }
138   
139       public void stopWorker(StopWorkerRequest req) throws Throwable {
140           workerManager.stopWorker(req.workerId(), false);
141       }
142   
143       public void destroyWorker(DestroyWorkerRequest req) throws Throwable {
144           workerManager.stopWorker(req.workerId(), true);
145       }
146   
147       /**
148        * Rebase the task spec time so that it is not earlier than the current time.
149        * This is only needed for tasks passed in with --exec.  Normally, the
150        * controller rebases the task spec time.
151        */
152       TaskSpec rebaseTaskSpecTime(TaskSpec spec) throws Exception {
153           ObjectNode node = JsonUtil.JSON_SERDE.valueToTree(spec);
154           node.set("startMs", new LongNode(Math.max(time.milliseconds(), spec.startMs())));
155           return JsonUtil.JSON_SERDE.treeToValue(node, TaskSpec.class);
156       }
157   
158       /**
159        * Start a task on the agent, and block until it completes.
160        *
161        * @param spec          The task specifiction.
162        * @param out           The output stream to print to.
163        *
164        * @return              True if the task run successfully; false otherwise.
165        */
166       boolean exec(TaskSpec spec, PrintStream out) throws Exception {
167           TaskController controller = null;
168           try {
169               controller = spec.newController(EXEC_TASK_ID);
170           } catch (Exception e) {
171               out.println("Unable to create the task controller.");
172               e.printStackTrace(out);
173               return false;
174           }
175           Set<String> nodes = controller.targetNodes(platform.topology());
176           if (!nodes.contains(platform.curNode().name())) {
177               out.println("This task is not configured to run on this node.  It runs on node(s): " +
178                   Utils.join(nodes, ", ") + ", whereas this node is " +
179                   platform.curNode().name());
180               return false;
181           }
182           KafkaFuture<String> future = null;
183           try {
184               future = workerManager.createWorker(EXEC_WORKER_ID, EXEC_TASK_ID, spec);
185           } catch (Throwable e) {
186               out.println("createWorker failed");
187               e.printStackTrace(out);
188               return false;
189           }
190           out.println("Waiting for completion of task:" + JsonUtil.toPrettyJsonString(spec));
191           String error = future.get();
192           if (error == null || error.isEmpty()) {
193               out.println("Task succeeded with status " +
194                   JsonUtil.toPrettyJsonString(workerManager.workerStates().get(EXEC_WORKER_ID).status()));
195               return true;
196           } else {
197               out.println("Task failed with status " +
198                   JsonUtil.toPrettyJsonString(workerManager.workerStates().get(EXEC_WORKER_ID).status()) +
199                   " and error " + error);
200               return false;
201           }
202       }
203   
204       public static void main(String[] args) throws Exception {
205           ArgumentParser parser = ArgumentParsers
206               .newArgumentParser("trogdor-agent")
207               .defaultHelp(true)
208               .description("The Trogdor fault injection agent");
209           parser.addArgument("--agent.config", "-c")
210               .action(store())
211               .required(true)
212               .type(String.class)
213               .dest("config")
214               .metavar("CONFIG")
215               .help("The configuration file to use.");
216           parser.addArgument("--node-name", "-n")
217               .action(store())
218               .required(true)
219               .type(String.class)
220               .dest("node_name")
221               .metavar("NODE_NAME")
222               .help("The name of this node.");
223           parser.addArgument("--exec", "-e")
224               .action(store())
225               .type(String.class)
226               .dest("task_spec")
227               .metavar("TASK_SPEC")
228               .help("Execute a single task spec and then exit.  The argument is the task spec to load when starting up, or a path to it.");
229           Namespace res = null;
230           try {
231               res = parser.parseArgs(args);
232           } catch (ArgumentParserException e) {
233               if (args.length == 0) {
234                   parser.printHelp();
235                   Exit.exit(0);
236               } else {
237                   parser.handleError(e);
238                   Exit.exit(1);
239               }
240           }
241           String configPath = res.getString("config");
242           String nodeName = res.getString("node_name");
243           String taskSpec = res.getString("task_spec");
244   
245           Platform platform = Platform.Config.parse(nodeName, configPath);
246           JsonRestServer restServer =
247               new JsonRestServer(Node.Util.getTrogdorAgentPort(platform.curNode()));
248           AgentRestResource resource = new AgentRestResource();
249           log.info("Starting agent process.");
250           final Agent agent = new Agent(platform, Scheduler.SYSTEM, restServer, resource);
251           restServer.start(resource);
252           Exit.addShutdownHook("agent-shutdown-hook", () -> {
253               log.warn("Running agent shutdown hook.");
254               try {
255                   agent.beginShutdown();
256                   agent.waitForShutdown();
257               } catch (Exception e) {
258                   log.error("Got exception while running agent shutdown hook.", e);
259               }
260           });
261           if (taskSpec != null) {
262               TaskSpec spec = null;
263               try {
264                   spec = JsonUtil.objectFromCommandLineArgument(taskSpec, TaskSpec.class);
265               } catch (Exception e) {
266                   System.out.println("Unable to parse the supplied task spec.");
267                   e.printStackTrace();
268                   Exit.exit(1);
269               }
270               TaskSpec effectiveSpec = agent.rebaseTaskSpecTime(spec);
271               Exit.exit(agent.exec(effectiveSpec, System.out) ? 0 : 1);
272           }
273           agent.waitForShutdown();
274       }
275   }
