53   /**
54    * Embedded server for the REST API that provides the control plane for Trogdor.
55    */
56   public class JsonRestServer {
57       private static final Logger log = LoggerFactory.getLogger(JsonRestServer.class);
58   
59       private static final long GRACEFUL_SHUTDOWN_TIMEOUT_MS = 100;
60   
61       private final ScheduledExecutorService shutdownExecutor;
62   
63       private final Server jettyServer;
64   
65       private final ServerConnector connector;
66   
67       /**
68        * Create a REST server for this herder using the specified configs.
69        *
70        * @param port              The port number to use for the REST server, or
71        *                          0 to use a random port.
72        */
73       public JsonRestServer(int port) {
74           this.shutdownExecutor = Executors.newSingleThreadScheduledExecutor(
75               ThreadUtils.createThreadFactory("JsonRestServerCleanupExecutor", false));
76           this.jettyServer = new Server();
77           this.connector = new ServerConnector(jettyServer);
78           if (port > 0) {
79               connector.setPort(port);
80           }
81           jettyServer.setConnectors(new Connector[]{connector});
82       }
83   
84       /**
85        * Start the JsonRestServer.
86        *
87        * @param resources         The path handling resources to register.
88        */
89       public void start(Object... resources) {
90           log.info("Starting REST server");
91           ResourceConfig resourceConfig = new ResourceConfig();
92           resourceConfig.register(new JacksonJsonProvider(JsonUtil.JSON_SERDE));
93           for (Object resource : resources) {
94               resourceConfig.register(resource);
95               log.info("Registered resource {}", resource);
96           }
97           resourceConfig.register(RestExceptionMapper.class);
98           ServletContainer servletContainer = new ServletContainer(resourceConfig);
99           ServletHolder servletHolder = new ServletHolder(servletContainer);
100           ServletContextHandler context = new ServletContextHandler(ServletContextHandler.SESSIONS);
101           context.setContextPath("/");
102           context.addServlet(servletHolder, "/*");
103   
104           RequestLogHandler requestLogHandler = new RequestLogHandler();
105           Slf4jRequestLogWriter slf4jRequestLogWriter = new Slf4jRequestLogWriter();
106           slf4jRequestLogWriter.setLoggerName(JsonRestServer.class.getCanonicalName());
107           CustomRequestLog requestLog = new CustomRequestLog(slf4jRequestLogWriter, CustomRequestLog.EXTENDED_NCSA_FORMAT + " %{ms}T");
108           requestLogHandler.setRequestLog(requestLog);
109   
110           HandlerCollection handlers = new HandlerCollection();
111           handlers.setHandlers(new Handler[]{context, new DefaultHandler(), requestLogHandler});
112           StatisticsHandler statsHandler = new StatisticsHandler();
113           statsHandler.setHandler(handlers);
114           jettyServer.setHandler(statsHandler);
115           /* Needed for graceful shutdown as per `setStopTimeout` documentation */
116           jettyServer.setStopTimeout(GRACEFUL_SHUTDOWN_TIMEOUT_MS);
117           jettyServer.setStopAtShutdown(true);
118   
119           try {
120               jettyServer.start();
121           } catch (Exception e) {
122               throw new RuntimeException("Unable to start REST server", e);
123           }
124           log.info("REST server listening at " + jettyServer.getURI());
125       }
126   
127       public int port() {
128           return connector.getLocalPort();
129       }
130   
131       /**
132        * Initiate shutdown, but do not wait for it to complete.
133        */
134       public void beginShutdown() {
135           if (!shutdownExecutor.isShutdown()) {
136               shutdownExecutor.submit((Callable<Void>) () -> {
137                   try {
138                       log.info("Stopping REST server");
139                       jettyServer.stop();
140                       jettyServer.join();
141                       log.info("REST server stopped");
142                   } catch (Exception e) {
143                       log.error("Unable to stop REST server", e);
144                   } finally {
145                       jettyServer.destroy();
146                   }
147                   shutdownExecutor.shutdown();
148                   return null;
149               });
150           }
151       }
152   
153       /**
154        * Wait for shutdown to complete.  May be called prior to beginShutdown.
155        */
156       public void waitForShutdown() throws InterruptedException {
157           while (!shutdownExecutor.isShutdown()) {
158               shutdownExecutor.awaitTermination(1, TimeUnit.DAYS);
159           }
160       }

162       /**
163        * Make an HTTP request.
164        *
165        * @param logger            The logger to use.
166        * @param url               HTTP connection will be established with this url.
167        * @param method            HTTP method ("GET", "POST", "PUT", etc.)
168        * @param requestBodyData   Object to serialize as JSON and send in the request body.
169        * @param responseFormat    Expected format of the response to the HTTP request.
170        * @param <T>               The type of the deserialized response to the HTTP request.
171        * @return The deserialized response to the HTTP request, or null if no data is expected.
172        */
173       public static <T> HttpResponse<T> httpRequest(Logger logger, String url, String method,
174               Object requestBodyData, TypeReference<T> responseFormat) throws IOException {
175           HttpURLConnection connection = null;
176           try {
177               String serializedBody = requestBodyData == null ? null :
178                   JsonUtil.JSON_SERDE.writeValueAsString(requestBodyData);
179               logger.debug("Sending {} with input {} to {}", method, serializedBody, url);
180               connection = (HttpURLConnection) new URL(url).openConnection();
181               connection.setRequestMethod(method);
182               connection.setRequestProperty("User-Agent", "kafka");
183               connection.setRequestProperty("Accept", "application/json");
184   
185               // connection.getResponseCode() implicitly calls getInputStream, so always set
186               // this to true.
187               connection.setDoInput(true);
188   
189               connection.setUseCaches(false);
190   
191               if (requestBodyData != null) {
192                   connection.setRequestProperty("Content-Type", "application/json");
193                   connection.setDoOutput(true);
194   
195                   OutputStream os = connection.getOutputStream();
196                   os.write(serializedBody.getBytes(StandardCharsets.UTF_8));
197                   os.flush();
198                   os.close();
199               }
200   
201               int responseCode = connection.getResponseCode();
202               if (responseCode == HttpURLConnection.HTTP_NO_CONTENT) {
203                   return new HttpResponse<>(null, new ErrorResponse(responseCode, connection.getResponseMessage()));
204               } else if ((responseCode >= 200) && (responseCode < 300)) {
205                   InputStream is = connection.getInputStream();
206                   T result = JsonUtil.JSON_SERDE.readValue(is, responseFormat);
207                   is.close();
208                   return new HttpResponse<>(result, null);
209               } else {
210                   // If the resposne code was not in the 200s, we assume that this is an error
211                   // response.
212                   InputStream es = connection.getErrorStream();
213                   if (es == null) {
214                       // Handle the case where HttpURLConnection#getErrorStream returns null.
215                       return new HttpResponse<>(null, new ErrorResponse(responseCode, ""));
216                   }
217                   // Try to read the error response JSON.
218                   ErrorResponse error = JsonUtil.JSON_SERDE.readValue(es, ErrorResponse.class);
219                   es.close();
220                   return new HttpResponse<>(null, error);
221               }
222           } finally {
223               if (connection != null) {
224                   connection.disconnect();
225               }
226           }
227       }

229       /**
230        * Make an HTTP request with retries.
231        *
232        * @param url               HTTP connection will be established with this url.
233        * @param method            HTTP method ("GET", "POST", "PUT", etc.)
234        * @param requestBodyData   Object to serialize as JSON and send in the request body.
235        * @param responseFormat    Expected format of the response to the HTTP request.
236        * @param <T>               The type of the deserialized response to the HTTP request.
237        * @return The deserialized response to the HTTP request, or null if no data is expected.
238        */
239       public static <T> HttpResponse<T> httpRequest(String url, String method, Object requestBodyData,
240                                                     TypeReference<T> responseFormat, int maxTries)
241               throws IOException, InterruptedException {
242           return httpRequest(log, url, method, requestBodyData, responseFormat, maxTries);
243       }
244   
245       /**
246        * Make an HTTP request with retries.
247        *
248        * @param logger            The logger to use.
249        * @param url               HTTP connection will be established with this url.
250        * @param method            HTTP method ("GET", "POST", "PUT", etc.)
251        * @param requestBodyData   Object to serialize as JSON and send in the request body.
252        * @param responseFormat    Expected format of the response to the HTTP request.
253        * @param <T>               The type of the deserialized response to the HTTP request.
254        * @return The deserialized response to the HTTP request, or null if no data is expected.
255        */
256       public static <T> HttpResponse<T> httpRequest(Logger logger, String url, String method,
257               Object requestBodyData, TypeReference<T> responseFormat, int maxTries)
258               throws IOException, InterruptedException {
259           IOException exc = null;
260           for (int tries = 0; tries < maxTries; tries++) {
261               if (tries > 0) {
262                   Thread.sleep(tries > 1 ? 10 : 2);
263               }
264               try {
265                   return httpRequest(logger, url, method, requestBodyData, responseFormat);
266               } catch (IOException e) {
267                   logger.info("{} {}: error: {}", method, url, e.getMessage());
268                   exc = e;
269               }
270           }
271           throw exc;
272       }

274       public static class HttpResponse<T> {
275           private final T body;
276           private final ErrorResponse error;
277   
278           HttpResponse(T body, ErrorResponse error) {
279               this.body = body;
280               this.error = error;
281           }
282   
283           public T body() throws Exception {
284               if (error != null) {
285                   throw RestExceptionMapper.toException(error.code(), error.message());
286               }
287               return body;
288           }
289   
290           public ErrorResponse error() {
291               return error;
292           }
293       }
