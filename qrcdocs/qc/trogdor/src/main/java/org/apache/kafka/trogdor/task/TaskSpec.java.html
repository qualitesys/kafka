18   package org.apache.kafka.trogdor.task;
19   
20   import com.fasterxml.jackson.annotation.JsonProperty;
21   import com.fasterxml.jackson.annotation.JsonTypeInfo;
22   import org.apache.kafka.trogdor.common.JsonUtil;
23   
24   import java.util.Collections;
25   import java.util.Map;
26   import java.util.Objects;
27   
28   
29   /**
30    * The specification for a task. This should be immutable and suitable for serializing and sending over the wire.
31    */
32   @JsonTypeInfo(use = JsonTypeInfo.Id.CLASS,
33                 include = JsonTypeInfo.As.PROPERTY,
34                 property = "class")
35   public abstract class TaskSpec {
36       /**
37        * The maximum task duration.
38        *
39        * We cap the task duration at this value to avoid worrying about 64-bit overflow or floating
40        * point rounding.  (Objects serialized as JSON canonically contain only floating point numbers,
41        * because JavaScript did not support integers.)
42        */
43       public final static long MAX_TASK_DURATION_MS = 1000000000000000L;
44   
45       /**
46        * When the time should start in milliseconds.
47        */
48       private final long startMs;
49   
50       /**
51        * How long the task should run in milliseconds.
52        */
53       private final long durationMs;
54   
55       protected TaskSpec(@JsonProperty("startMs") long startMs,
56               @JsonProperty("durationMs") long durationMs) {
57           this.startMs = startMs;
58           this.durationMs = Math.max(0, Math.min(durationMs, MAX_TASK_DURATION_MS));
59       }
60   
61       /**
62        * Get the target start time of this task in ms.
63        */
64       @JsonProperty
65       public final long startMs() {
66           return startMs;
67       }
68   
69       /**
70        * Get the deadline time of this task in ms
71        */
72       public final long endMs() {
73           return startMs + durationMs;
74       }
75   
76       /**
77        * Get the duration of this task in ms.
78        */
79       @JsonProperty
80       public final long durationMs() {
81           return durationMs;
82       }
83   
84       /**
85        * Hydrate this task on the coordinator.
86        *
87        * @param id        The task id.
88        */
89       public abstract TaskController newController(String id);
90   
91       /**
92        * Hydrate this task on the agent.
93        *
94        * @param id        The worker id.
95        */
96       public abstract TaskWorker newTaskWorker(String id);
97   
98       @Override
99       public final boolean equals(Object o) {
100           if (this == o) return true;
101           if (o == null || getClass() != o.getClass()) return false;
102           return toString().equals(o.toString());
103       }
104   
105       @Override
106       public final int hashCode() {
107           return Objects.hashCode(toString());
108       }
109   
110       @Override
111       public String toString() {
112           return JsonUtil.toJsonString(this);
113       }
114   
115       protected Map<String, String> configOrEmptyMap(Map<String, String> config) {
116           return (config == null) ? Collections.<String, String>emptyMap() : config;
117       }
118   }
