18   package org.apache.kafka.trogdor.rest;
19   
20   import com.fasterxml.jackson.annotation.JsonCreator;
21   import com.fasterxml.jackson.annotation.JsonProperty;
22   
23   import java.util.Collection;
24   import java.util.Collections;
25   import java.util.HashSet;
26   import java.util.Optional;
27   import java.util.Set;
28   
29   /**
30    * The request to /coordinator/tasks
31    */
32   public class TasksRequest extends Message {
33       /**
34        * The task IDs to list.
35        * An empty set of task IDs indicates that we should list all task IDs.
36        */
37       private final Set<String> taskIds;
38   
39       /**
40        * If this is non-zero, only tasks with a startMs at or after this time will be listed.
41        */
42       private final long firstStartMs;
43   
44       /**
45        * If this is non-zero, only tasks with a startMs at or before this time will be listed.
46        */
47       private final long lastStartMs;
48   
49       /**
50        * If this is non-zero, only tasks with an endMs at or after this time will be listed.
51        */
52       private final long firstEndMs;
53   
54       /**
55        * If this is non-zero, only tasks with an endMs at or before this time will be listed.
56        */
57       private final long lastEndMs;
58   
59       /**
60        * The desired state of the tasks.
61        * An empty string will match all states.
62        */
63       private final Optional<TaskStateType> state;
64   
65       @JsonCreator
66       public TasksRequest(@JsonProperty("taskIds") Collection<String> taskIds,
67               @JsonProperty("firstStartMs") long firstStartMs,
68               @JsonProperty("lastStartMs") long lastStartMs,
69               @JsonProperty("firstEndMs") long firstEndMs,
70               @JsonProperty("lastEndMs") long lastEndMs,
71               @JsonProperty("state") Optional<TaskStateType> state) {
72           this.taskIds = Collections.unmodifiableSet((taskIds == null) ?
73               new HashSet<String>() : new HashSet<>(taskIds));
74           this.firstStartMs = Math.max(0, firstStartMs);
75           this.lastStartMs = Math.max(0, lastStartMs);
76           this.firstEndMs = Math.max(0, firstEndMs);
77           this.lastEndMs = Math.max(0, lastEndMs);
78           this.state = state == null ? Optional.empty() : state;
79       }
80   
81       @JsonProperty
82       public Collection<String> taskIds() {
83           return taskIds;
84       }
85   
86       @JsonProperty
87       public long firstStartMs() {
88           return firstStartMs;
89       }
90   
91       @JsonProperty
92       public long lastStartMs() {
93           return lastStartMs;
94       }
95   
96       @JsonProperty
97       public long firstEndMs() {
98           return firstEndMs;
99       }
100   
101       @JsonProperty
102       public long lastEndMs() {
103           return lastEndMs;
104       }
105   
106       @JsonProperty
107       public Optional<TaskStateType> state() {
108           return state;
109       }
110   
111       /**
112        * Determine if this TaskRequest should return a particular task.
113        *
114        * @param taskId    The task ID.
115        * @param startMs   The task start time, or -1 if the task hasn't started.
116        * @param endMs     The task end time, or -1 if the task hasn't ended.
117        * @return          True if information about the task should be returned.
118        */
119       public boolean matches(String taskId, long startMs, long endMs, TaskStateType state) {
120           if ((!taskIds.isEmpty()) && (!taskIds.contains(taskId))) {
121               return false;
122           }
123           if ((firstStartMs > 0) && (startMs < firstStartMs)) {
124               return false;
125           }
126           if ((lastStartMs > 0) && ((startMs < 0) || (startMs > lastStartMs))) {
127               return false;
128           }
129           if ((firstEndMs > 0) && (endMs < firstEndMs)) {
130               return false;
131           }
132           if ((lastEndMs > 0) && ((endMs < 0) || (endMs > lastEndMs))) {
133               return false;
134           }
135   
136           if (this.state.isPresent() && !this.state.get().equals(state)) {
137               return false;
138           }
139   
140           return true;
141       }
142   }
