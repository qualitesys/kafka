78               if (logContext == null) logContext = new LogContext();
79               if (snapshotRegistry == null) snapshotRegistry = new SnapshotRegistry(logContext);

126       ControllerResult<Map<String, ApiError>> updateFeatures(
127           Map<String, Short> updates,
128           Map<String, FeatureUpdate.UpgradeType> upgradeTypes,
129           Map<Integer, Map<String, VersionRange>> brokerFeatures,
130           boolean validateOnly
131       ) {
132           TreeMap<String, ApiError> results = new TreeMap<>();
133           List<ApiMessageAndVersion> records = new ArrayList<>();
134           for (Entry<String, Short> entry : updates.entrySet()) {
135               results.put(entry.getKey(), updateFeature(entry.getKey(), entry.getValue(),
136                   upgradeTypes.getOrDefault(entry.getKey(), FeatureUpdate.UpgradeType.UPGRADE), brokerFeatures, records));
137           }
138   
139           if (validateOnly) {
140               return ControllerResult.of(Collections.emptyList(), results);
141           } else {
142               return ControllerResult.atomicOf(records, results);
143           }
144       }

150       private ApiError updateFeature(
151           String featureName,
152           short newVersion,
153           FeatureUpdate.UpgradeType upgradeType,
154           Map<Integer, Map<String, VersionRange>> brokersAndFeatures,
155           List<ApiMessageAndVersion> records
156       ) {
157           if (upgradeType.equals(FeatureUpdate.UpgradeType.UNKNOWN)) {
158               return invalidUpdateVersion(featureName, newVersion,
159                   "The controller does not support the given upgrade type.");
160           }
161   
162           final Short currentVersion;
163           if (featureName.equals(MetadataVersion.FEATURE_NAME)) {
164               currentVersion = metadataVersion.get().featureLevel();
165           } else {
166               currentVersion = finalizedVersions.get(featureName);
167           }
168   
169           if (newVersion < 0) {
170               return invalidUpdateVersion(featureName, newVersion,
171                   "A feature version cannot be less than 0.");
172           }
173   
174           Optional<String> reasonNotSupported = quorumFeatures.reasonNotSupported(featureName, newVersion);
175           if (reasonNotSupported.isPresent()) {
176               return invalidUpdateVersion(featureName, newVersion, reasonNotSupported.get());
177           }
178   
179           for (Entry<Integer, Map<String, VersionRange>> brokerEntry : brokersAndFeatures.entrySet()) {
180               VersionRange brokerRange = brokerEntry.getValue().get(featureName);
181               if (brokerRange == null) {
182                   return invalidUpdateVersion(featureName, newVersion,
183                       "Broker " + brokerEntry.getKey() + " does not support this feature.");
184               } else if (!brokerRange.contains(newVersion)) {
185                   return invalidUpdateVersion(featureName, newVersion,
186                       "Broker " + brokerEntry.getKey() + " does not support the given " +
187                       "version. It supports " + brokerRange.min() + " to " + brokerRange.max() + ".");
188               }
189           }
190   
191           if (currentVersion != null && newVersion < currentVersion) {
192               if (upgradeType.equals(FeatureUpdate.UpgradeType.UPGRADE)) {
193                   return invalidUpdateVersion(featureName, newVersion,
194                       "Can't downgrade the version of this feature without setting the " +
195                       "upgrade type to either safe or unsafe downgrade.");
196               }
197           }
198   
199           if (featureName.equals(MetadataVersion.FEATURE_NAME)) {
200               // Perform additional checks if we're updating metadata.version
201               return updateMetadataVersion(newVersion, upgradeType.equals(FeatureUpdate.UpgradeType.UNSAFE_DOWNGRADE), records::add);
202           } else {
203               records.add(new ApiMessageAndVersion(
204                   new FeatureLevelRecord()
205                       .setName(featureName)
206                       .setFeatureLevel(newVersion),
207                   FEATURE_LEVEL_RECORD.highestSupportedVersion()));
208               return ApiError.NONE;
209           }
210       }

212       private ApiError invalidUpdateVersion(String feature, short version, String message) {
213           String errorMessage = String.format("Invalid update version %d for feature %s. %s", version, feature, message);
214           log.debug(errorMessage);
215           return new ApiError(Errors.INVALID_UPDATE_VERSION, errorMessage);
216       }

218       /**
219        * Perform some additional validation for metadata.version updates.
220        */
221       private ApiError updateMetadataVersion(
222           short newVersionLevel,
223           boolean allowUnsafeDowngrade,
224           Consumer<ApiMessageAndVersion> recordConsumer
225       ) {
226           MetadataVersion currentVersion = metadataVersion();
227           final MetadataVersion newVersion;
228           try {
229               newVersion = MetadataVersion.fromFeatureLevel(newVersionLevel);
230           } catch (IllegalArgumentException e) {
231               return invalidMetadataVersion(newVersionLevel, "Unknown metadata.version.");
232           }
233   
234           if (newVersion.isLessThan(currentVersion)) {
235               // This is a downgrade
236               boolean metadataChanged = MetadataVersion.checkIfMetadataChanged(currentVersion, newVersion);
237               if (!metadataChanged) {
238                   log.info("Downgrading metadata.version from {} to {}.", currentVersion, newVersion);
239               } else if (allowUnsafeDowngrade) {
240                   log.info("Downgrading metadata.version unsafely from {} to {}.", currentVersion, newVersion);
241               } else {
242                   return invalidMetadataVersion(newVersionLevel, "Refusing to perform the requested " +
243                           "downgrade because it might delete metadata information. Retry using " +
244                           "UNSAFE_DOWNGRADE if you want to force the downgrade to proceed.");
245               }
246           } else {
247               log.info("Upgrading metadata.version from {} to {}.", currentVersion, newVersion);
248           }
249   
250           recordConsumer.accept(new ApiMessageAndVersion(
251               new FeatureLevelRecord()
252                   .setName(MetadataVersion.FEATURE_NAME)
253                   .setFeatureLevel(newVersionLevel), FEATURE_LEVEL_RECORD.lowestSupportedVersion()));
254           return ApiError.NONE;
255       }

257       private ApiError invalidMetadataVersion(short version, String message) {
258           String errorMessage = String.format("Invalid metadata.version %d. %s", version, message);
259           log.error(errorMessage);
260           return new ApiError(Errors.INVALID_UPDATE_VERSION, errorMessage);
261       }

263       FinalizedControllerFeatures finalizedFeatures(long epoch) {
264           Map<String, Short> features = new HashMap<>();
265           features.put(MetadataVersion.FEATURE_NAME, metadataVersion.get(epoch).featureLevel());
266           for (Entry<String, Short> entry : finalizedVersions.entrySet(epoch)) {
267               features.put(entry.getKey(), entry.getValue());
268           }
269           return new FinalizedControllerFeatures(features, epoch);
270       }

275       boolean sawMetadataVersion() {
276           return this.sawMetadataVersion.get();

279       public void replay(FeatureLevelRecord record) {
280           VersionRange range = quorumFeatures.localSupportedFeature(record.name());
281           if (!range.contains(record.featureLevel())) {
282               throw new RuntimeException("Tried to apply FeatureLevelRecord " + record + ", but this controller only " +
283                   "supports versions " + range);
284           }
285           if (record.name().equals(MetadataVersion.FEATURE_NAME)) {
286               log.info("Setting metadata.version to {}", record.featureLevel());
287               metadataVersion.set(MetadataVersion.fromFeatureLevel(record.featureLevel()));
288               sawMetadataVersion.set(true);
289           } else {
290               if (record.featureLevel() == 0) {
291                   log.info("Removing feature {}", record.name());
292                   finalizedVersions.remove(record.name());
293               } else {
294                   log.info("Setting feature {} to {}", record.name(), record.featureLevel());
295                   finalizedVersions.put(record.name(), record.featureLevel());
296               }
297           }

300       class FeatureControlIterator implements Iterator<List<ApiMessageAndVersion>> {
301           private final Iterator<Entry<String, Short>> iterator;
302           private final MetadataVersion metadataVersion;
303           private boolean wroteVersion = false;
304   
305           FeatureControlIterator(long epoch) {
306               this.iterator = finalizedVersions.entrySet(epoch).iterator();
307               this.metadataVersion = FeatureControlManager.this.metadataVersion.get(epoch);
308           }
309   
310           @Override
311           public boolean hasNext() {
312               return !wroteVersion || iterator.hasNext();
313           }
314   
315           @Override
316           public List<ApiMessageAndVersion> next() {
317               // Write the metadata.version first
318               if (!wroteVersion) {
319                   wroteVersion = true;
320                   return Collections.singletonList(new ApiMessageAndVersion(new FeatureLevelRecord()
321                       .setName(MetadataVersion.FEATURE_NAME)
322                       .setFeatureLevel(metadataVersion.featureLevel()), FEATURE_LEVEL_RECORD.lowestSupportedVersion()));
323               }
324               // Then write the rest of the features
325               if (!hasNext()) throw new NoSuchElementException();
326               Entry<String, Short> entry = iterator.next();
327               return Collections.singletonList(new ApiMessageAndVersion(new FeatureLevelRecord()
328                   .setName(entry.getKey())
329                   .setFeatureLevel(entry.getValue()), FEATURE_LEVEL_RECORD.highestSupportedVersion()));
330           }
331       }
332   
333       FeatureControlIterator iterator(long epoch) {
334           return new FeatureControlIterator(epoch);
335       }
