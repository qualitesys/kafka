18   package org.apache.kafka.controller;
19   
20   import org.apache.kafka.common.errors.UnknownServerException;
21   import org.apache.kafka.common.metadata.ProducerIdsRecord;
22   import org.apache.kafka.server.common.ApiMessageAndVersion;
23   import org.apache.kafka.server.common.ProducerIdsBlock;
24   import org.apache.kafka.timeline.SnapshotRegistry;
25   import org.apache.kafka.timeline.TimelineLong;
26   import org.apache.kafka.timeline.TimelineObject;
27   
28   import java.util.ArrayList;
29   import java.util.Collections;
30   import java.util.Iterator;
31   import java.util.List;
32   
33   
34   public class ProducerIdControlManager {
35   
36       private final ClusterControlManager clusterControlManager;
37       private final TimelineObject<ProducerIdsBlock> nextProducerBlock;
38       private final TimelineLong brokerEpoch;
39   
40       ProducerIdControlManager(ClusterControlManager clusterControlManager, SnapshotRegistry snapshotRegistry) {
41           this.clusterControlManager = clusterControlManager;
42           this.nextProducerBlock = new TimelineObject<>(snapshotRegistry, ProducerIdsBlock.EMPTY);
43           this.brokerEpoch = new TimelineLong(snapshotRegistry);
44       }
45   
46       ControllerResult<ProducerIdsBlock> generateNextProducerId(int brokerId, long brokerEpoch) {
47           clusterControlManager.checkBrokerEpoch(brokerId, brokerEpoch);
48   
49           long firstProducerIdInBlock = nextProducerBlock.get().firstProducerId();
50           if (firstProducerIdInBlock > Long.MAX_VALUE - ProducerIdsBlock.PRODUCER_ID_BLOCK_SIZE) {
51               throw new UnknownServerException("Exhausted all producerIds as the next block's end producerId " +
52                   "has exceeded the int64 type limit");
53           }
54   
55           ProducerIdsBlock block = new ProducerIdsBlock(brokerId, firstProducerIdInBlock, ProducerIdsBlock.PRODUCER_ID_BLOCK_SIZE);
56           long newNextProducerId = block.nextBlockFirstId();
57   
58           ProducerIdsRecord record = new ProducerIdsRecord()
59               .setNextProducerId(newNextProducerId)
60               .setBrokerId(brokerId)
61               .setBrokerEpoch(brokerEpoch);
62           return ControllerResult.of(Collections.singletonList(new ApiMessageAndVersion(record, (short) 0)), block);
63       }
64   
65       void replay(ProducerIdsRecord record) {
66           long currentNextProducerId = nextProducerBlock.get().firstProducerId();
67           if (record.nextProducerId() <= currentNextProducerId) {
68               throw new RuntimeException("Next Producer ID from replayed record (" + record.nextProducerId() + ")" +
69                   " is not greater than current next Producer ID (" + currentNextProducerId + ")");
70           } else {
71               nextProducerBlock.set(new ProducerIdsBlock(record.brokerId(), record.nextProducerId(), ProducerIdsBlock.PRODUCER_ID_BLOCK_SIZE));
72               brokerEpoch.set(record.brokerEpoch());
73           }
74       }
75   
76       Iterator<List<ApiMessageAndVersion>> iterator(long epoch) {
77           List<ApiMessageAndVersion> records = new ArrayList<>(1);
78   
79           ProducerIdsBlock producerIdBlock = nextProducerBlock.get(epoch);
80           if (producerIdBlock.firstProducerId() > 0) {
81               records.add(new ApiMessageAndVersion(
82                   new ProducerIdsRecord()
83                       .setNextProducerId(producerIdBlock.firstProducerId())
84                       .setBrokerId(producerIdBlock.assignedBrokerId())
85                       .setBrokerEpoch(brokerEpoch.get(epoch)),
86                   (short) 0));
87           }
88           return Collections.singleton(records).iterator();
89       }
90   }
