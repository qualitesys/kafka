18   package org.apache.kafka.controller;
19   
20   import org.apache.kafka.common.errors.InvalidReplicaAssignmentException;
21   import org.apache.kafka.metadata.PartitionRegistration;
22   import org.apache.kafka.metadata.Replicas;
23   
24   import java.util.ArrayList;
25   import java.util.Set;
26   import java.util.List;
27   import java.util.Objects;
28   
29   
30   class PartitionReassignmentRevert {
31       private final List<Integer> replicas;
32       private final List<Integer> isr;
33       private final boolean unclean;
34   
35       PartitionReassignmentRevert(PartitionRegistration registration) {
36           // Figure out the replica list and ISR that we will have after reverting the
37           // reassignment. In general, we want to take out any replica that the reassignment
38           // was adding, but keep the ones the reassignment was removing. (But see the
39           // special case below.)
40           Set<Integer> adding = Replicas.toSet(registration.addingReplicas);
41           this.replicas = new ArrayList<>(registration.replicas.length);
42           this.isr = new ArrayList<>(registration.isr.length);
43           for (int i = 0; i < registration.isr.length; i++) {
44               int replica = registration.isr[i];
45               if (!adding.contains(replica)) {
46                   this.isr.add(replica);
47               }
48           }
49           for (int replica : registration.replicas) {
50               if (!adding.contains(replica)) {
51                   this.replicas.add(replica);
52               }
53           }
54           if (isr.isEmpty()) {
55               // In the special case that all the replicas that are in the ISR are also
56               // contained in addingReplicas, we choose the first remaining replica and add
57               // it to the ISR. This is considered an unclean leader election. Therefore,
58               // calling code must check that unclean leader election is enabled before
59               // accepting the new ISR.
60               if (this.replicas.isEmpty()) {
61                   // This should not be reachable, since it would require a partition
62                   // starting with an empty replica set prior to the reassignment we are
63                   // trying to revert.
64                   throw new InvalidReplicaAssignmentException("Invalid replica " +
65                       "assignment: addingReplicas contains all replicas.");
66               }
67               isr.add(replicas.get(0));
68               this.unclean = true;
69           } else {
70               this.unclean = false;
71           }
72       }
73   
74       List<Integer> replicas() {
75           return replicas;
76       }
77   
78       List<Integer> isr() {
79           return isr;
80       }
81   
82       boolean unclean() {
83           return unclean;
84       }
85   
86       @Override
87       public int hashCode() {
88           return Objects.hash(replicas, isr);
89       }
90   
91       @Override
92       public boolean equals(Object o) {
93           if (!(o instanceof PartitionReassignmentRevert)) return false;
94           PartitionReassignmentRevert other = (PartitionReassignmentRevert) o;
95           return replicas.equals(other.replicas) &&
96               isr.equals(other.isr);
97       }
98   
99       @Override
100       public String toString() {
101           return "PartitionReassignmentRevert(" +
102               "replicas=" + replicas + ", " +
103               "isr=" + isr + ")";
104       }
105   }
