42   /**
43    * The BrokerHeartbeatManager manages all the soft state associated with broker heartbeats.
44    * Soft state is state which does not appear in the metadata log.  This state includes
45    * things like the last time each broker sent us a heartbeat, and whether the broker is
46    * trying to perform a controlled shutdown.
47    *
48    * Only the active controller has a BrokerHeartbeatManager, since only the active
49    * controller handles broker heartbeats.  Standby controllers will create a heartbeat
50    * manager as part of the process of activating.  This design minimizes the size of the
51    * metadata partition by excluding heartbeats from it.  However, it does mean that after
52    * a controller failover, we may take some extra time to fence brokers, since the new
53    * active controller does not know when the last heartbeats were received from each.
54    */
55   public class BrokerHeartbeatManager {
56       static class BrokerHeartbeatState {
57           /**
58            * The broker ID.
59            */
60           private final int id;
61   
62           /**
63            * The last time we received a heartbeat from this broker, in monotonic nanoseconds.
64            * When this field is updated, we also may have to update the broker's position in
65            * the unfenced list.
66            */
67           long lastContactNs;
68   
69           /**
70            * The last metadata offset which this broker reported.  When this field is updated,
71            * we may also have to update the broker's position in the active set.
72            */
73           long metadataOffset;
74   
75           /**
76            * The offset at which the broker should complete its controlled shutdown, or -1
77            * if the broker is not performing a controlled shutdown.  When this field is
78            * updated, we also have to update the broker's position in the shuttingDown set.
79            */
80           private long controlledShutDownOffset;
81   
82           /**
83            * The previous entry in the unfenced list, or null if the broker is not in that list.
84            */
85           private BrokerHeartbeatState prev;
86   
87           /**
88            * The next entry in the unfenced list, or null if the broker is not in that list.
89            */
90           private BrokerHeartbeatState next;
91   
92           BrokerHeartbeatState(int id) {
93               this.id = id;
94               this.lastContactNs = 0;
95               this.prev = null;
96               this.next = null;
97               this.metadataOffset = -1;
98               this.controlledShutDownOffset = -1;
99           }
100   
101           /**
102            * Returns the broker ID.
103            */
104           int id() {
105               return id;
106           }
107   
108           /**
109            * Returns true only if the broker is fenced.
110            */
111           boolean fenced() {
112               return prev == null;
113           }
114   
115           /**
116            * Returns true only if the broker is in controlled shutdown state.
117            */
118           boolean shuttingDown() {
119               return controlledShutDownOffset >= 0;
120           }

123       static class MetadataOffsetComparator implements Comparator<BrokerHeartbeatState> {
124           static final MetadataOffsetComparator INSTANCE = new MetadataOffsetComparator();
125   
126           @Override
127           public int compare(BrokerHeartbeatState a, BrokerHeartbeatState b) {
128               if (a.metadataOffset < b.metadataOffset) {
129                   return -1;
130               } else if (a.metadataOffset > b.metadataOffset) {
131                   return 1;
132               } else if (a.id < b.id) {
133                   return -1;
134               } else if (a.id > b.id) {
135                   return 1;
136               } else {
137                   return 0;
138               }
139           }

142       static class BrokerHeartbeatStateList {
143           /**
144            * The head of the list of unfenced brokers.  The list is sorted in ascending order
145            * of last contact time.
146            */
147           private final BrokerHeartbeatState head;
148   
149           BrokerHeartbeatStateList() {
150               this.head = new BrokerHeartbeatState(-1);
151               head.prev = head;
152               head.next = head;
153           }
154   
155           /**
156            * Return the head of the list, or null if the list is empty.
157            */
158           BrokerHeartbeatState first() {
159               BrokerHeartbeatState result = head.next;
160               return result == head ? null : result;
161           }
162   
163           /**
164            * Add the broker to the list. We start looking for a place to put it at the end
165            * of the list.
166            */
167           void add(BrokerHeartbeatState broker) {
168               BrokerHeartbeatState cur = head.prev;
169               while (true) {
170                   if (cur == head || cur.lastContactNs <= broker.lastContactNs) {
171                       broker.next = cur.next;
172                       cur.next.prev = broker;
173                       broker.prev = cur;
174                       cur.next = broker;
175                       break;
176                   }
177                   cur = cur.prev;
178               }
179           }
180   
181           /**
182            * Remove a broker from the list.
183            */
184           void remove(BrokerHeartbeatState broker) {
185               if (broker.next == null) {
186                   throw new RuntimeException(broker + " is not in the  list.");
187               }
188               broker.prev.next = broker.next;
189               broker.next.prev = broker.prev;
190               broker.prev = null;
191               broker.next = null;
192           }
193   
194           BrokerHeartbeatStateIterator iterator() {
195               return new BrokerHeartbeatStateIterator(head);
196           }
197       }

332        *
333        * @param broker        The broker to check.
334        *

336        */
337       private boolean hasValidSession(BrokerHeartbeatState broker) {
338           if (broker.fenced()) {
339               return false;
340           } else {
341               return broker.lastContactNs + sessionTimeoutNs >= time.nanoseconds();
342           }
343       }

345       /**
346        * Register this broker if we haven't already, and make sure its fencing state is
347        * correct.
348        *
349        * @param brokerId          The broker ID.
350        * @param fenced            True only if the broker is currently fenced.
351        */
352       void register(int brokerId, boolean fenced) {
353           BrokerHeartbeatState broker = brokers.get(brokerId);
354           if (broker == null) {
355               touch(brokerId, fenced, -1);
356           } else if (broker.fenced() != fenced) {
357               touch(brokerId, fenced, broker.metadataOffset);
358           }
359       }
360   
361       /**
362        * Update broker state, including lastContactNs.
363        *
364        * @param brokerId          The broker ID.
365        * @param fenced            True only if the broker is currently fenced.
366        * @param metadataOffset    The latest metadata offset of the broker.
367        */
368       void touch(int brokerId, boolean fenced, long metadataOffset) {
369           BrokerHeartbeatState broker = brokers.get(brokerId);
370           if (broker == null) {
371               broker = new BrokerHeartbeatState(brokerId);
372               brokers.put(brokerId, broker);
373           } else {
374               // Remove the broker from the unfenced list and/or the active set. Its
375               // position in either of those data structures depends on values we are
376               // changing here. We will re-add it if necessary at the end of this function.
377               untrack(broker);
378           }
379           broker.lastContactNs = time.nanoseconds();
380           broker.metadataOffset = metadataOffset;
381           if (fenced) {
382               // If a broker is fenced, it leaves controlled shutdown.  On its next heartbeat,
383               // it will shut down immediately.
384               broker.controlledShutDownOffset = -1;
385           } else {
386               unfenced.add(broker);
387               if (!broker.shuttingDown()) {
388                   active.add(broker);
389               }
390           }
391       }

403        * Mark a broker as being in the controlled shutdown state.
404        *
405        * @param brokerId                  The broker id.

408       void updateControlledShutdownOffset(int brokerId, long controlledShutDownOffset) {
409           BrokerHeartbeatState broker = brokers.get(brokerId);
410           if (broker == null) {

417           broker.controlledShutDownOffset = controlledShutDownOffset;

419               brokerId, controlledShutDownOffset);

428           if (broker == null) {

435       /**
436        * Check if the oldest broker to have hearbeated has already violated the
437        * sessionTimeoutNs timeout and needs to be fenced.
438        *
439        * @return      An Optional broker node id.
440        */
441       Optional<Integer> findOneStaleBroker() {
442           BrokerHeartbeatStateIterator iterator = unfenced.iterator();
443           if (iterator.hasNext()) {
444               BrokerHeartbeatState broker = iterator.next();
445               // The unfenced list is sorted on last contact time from each
446               // broker. If the first broker is not stale, then none is.
447               if (!hasValidSession(broker)) {
448                   return Optional.of(broker.id);
449               }
450           }
451           return Optional.empty();
452       }

461       static class UsableBrokerIterator implements Iterator<UsableBroker> {
462           private final Iterator<BrokerHeartbeatState> iterator;
463           private final Function<Integer, Optional<String>> idToRack;
464           private UsableBroker next;
465   
466           UsableBrokerIterator(Iterator<BrokerHeartbeatState> iterator,
467                                Function<Integer, Optional<String>> idToRack) {
468               this.iterator = iterator;
469               this.idToRack = idToRack;
470               this.next = null;
471           }
472   
473           @Override
474           public boolean hasNext() {
475               if (next != null) {
476                   return true;
477               }
478               BrokerHeartbeatState result;
479               do {
480                   if (!iterator.hasNext()) {
481                       return false;
482                   }
483                   result = iterator.next();
484               } while (result.shuttingDown());
485               Optional<String> rack = idToRack.apply(result.id());
486               next = new UsableBroker(result.id(), rack, result.fenced());
487               return true;
488           }
489   
490           @Override
491           public UsableBroker next() {
492               if (!hasNext()) {
493                   throw new NoSuchElementException();
494               }
495               UsableBroker result = next;
496               next = null;
497               return result;
498           }
499       }

511       /**
512        * Calculate the next broker state for a broker that just sent a heartbeat request.
513        *
514        * @param brokerId              The broker id.
515        * @param request               The incoming heartbeat request.
516        * @param lastCommittedOffset   The last committed offset of the quorum controller.
517        * @param hasLeaderships        A callback which evaluates to true if the broker leads
518        *                              at least one partition.
519        *
520        * @return                      The current and next broker states.
521        */
522       BrokerControlStates calculateNextBrokerState(int brokerId,
523                                                    BrokerHeartbeatRequestData request,
524                                                    long lastCommittedOffset,
525                                                    Supplier<Boolean> hasLeaderships) {
526           BrokerHeartbeatState broker = brokers.getOrDefault(brokerId,
527               new BrokerHeartbeatState(brokerId));
528           BrokerControlState currentState = currentBrokerState(broker);
529           switch (currentState) {
530               case FENCED:
531                   if (request.wantShutDown()) {
532                       log.info("Fenced broker {} has requested and been granted an immediate " +
533                           "shutdown.", brokerId);
534                       return new BrokerControlStates(currentState, SHUTDOWN_NOW);
535                   } else if (!request.wantFence()) {
536                       if (request.currentMetadataOffset() >= lastCommittedOffset) {
537                           log.info("The request from broker {} to unfence has been granted " +
538                                   "because it has caught up with the last committed metadata " +
539                                   "offset {}.", brokerId, lastCommittedOffset);
540                           return new BrokerControlStates(currentState, UNFENCED);
541                       } else {
542                           if (log.isDebugEnabled()) {
543                               log.debug("The request from broker {} to unfence cannot yet " +
544                                   "be granted because it has not caught up with the last " +
545                                   "committed metadata offset {}. It is still at offset {}.",
546                                   brokerId, lastCommittedOffset, request.currentMetadataOffset());
547                           }
548                           return new BrokerControlStates(currentState, FENCED);
549                       }
550                   }
551                   return new BrokerControlStates(currentState, FENCED);
552   
553               case UNFENCED:
554                   if (request.wantFence()) {
555                       if (request.wantShutDown()) {
556                           log.info("Unfenced broker {} has requested and been granted an " +
557                               "immediate shutdown.", brokerId);
558                           return new BrokerControlStates(currentState, SHUTDOWN_NOW);
559                       } else {
560                           log.info("Unfenced broker {} has requested and been granted " +
561                               "fencing", brokerId);
562                           return new BrokerControlStates(currentState, FENCED);
563                       }
564                   } else if (request.wantShutDown()) {
565                       if (hasLeaderships.get()) {
566                           log.info("Unfenced broker {} has requested and been granted a " +
567                               "controlled shutdown.", brokerId);
568                           return new BrokerControlStates(currentState, CONTROLLED_SHUTDOWN);
569                       } else {
570                           log.info("Unfenced broker {} has requested and been granted an " +
571                               "immediate shutdown.", brokerId);
572                           return new BrokerControlStates(currentState, SHUTDOWN_NOW);
573                       }
574                   }
575                   return new BrokerControlStates(currentState, UNFENCED);
576   
577               case CONTROLLED_SHUTDOWN:
578                   if (hasLeaderships.get()) {
579                       log.debug("Broker {} is in controlled shutdown state, but can not " +
580                           "shut down because more leaders still need to be moved.", brokerId);
581                       return new BrokerControlStates(currentState, CONTROLLED_SHUTDOWN);
582                   }
583                   long lowestActiveOffset = lowestActiveOffset();
584                   if (broker.controlledShutDownOffset <= lowestActiveOffset) {
585                       log.info("The request from broker {} to shut down has been granted " +
586                           "since the lowest active offset {} is now greater than the " +
587                           "broker's controlled shutdown offset {}.", brokerId,
588                           lowestActiveOffset, broker.controlledShutDownOffset);
589                       return new BrokerControlStates(currentState, SHUTDOWN_NOW);
590                   }
591                   log.debug("The request from broker {} to shut down can not yet be granted " +
592                       "because the lowest active offset {} is not greater than the broker's " +
593                       "shutdown offset {}.", brokerId, lowestActiveOffset,
594                       broker.controlledShutDownOffset);
595                   return new BrokerControlStates(currentState, CONTROLLED_SHUTDOWN);
596   
597               default:
598                   return new BrokerControlStates(currentState, SHUTDOWN_NOW);
599           }
600       }
601   }
