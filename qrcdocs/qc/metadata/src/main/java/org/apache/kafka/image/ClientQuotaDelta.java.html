18   package org.apache.kafka.image;
19   
20   import org.apache.kafka.common.metadata.ClientQuotaRecord;
21   
22   import java.util.HashMap;
23   import java.util.Map;
24   import java.util.OptionalDouble;
25   
26   import static java.util.Map.Entry;
27   
28   
29   public final class ClientQuotaDelta {
30       private final ClientQuotaImage image;
31       private final Map<String, OptionalDouble> changes = new HashMap<>();
32   
33       public ClientQuotaDelta(ClientQuotaImage image) {
34           this.image = image;
35       }
36   
37       public Map<String, OptionalDouble> changes() {
38           return changes;
39       }
40   
41       public void finishSnapshot() {
42           for (String key : image.quotas().keySet()) {
43               if (!changes.containsKey(key)) {
44                   // If a quota from the image did not appear in the snapshot, mark it as removed.
45                   changes.put(key, OptionalDouble.empty());
46               }
47           }
48       }
49   
50       public void replay(ClientQuotaRecord record) {
51           if (record.remove()) {
52               changes.put(record.key(), OptionalDouble.empty());
53           } else {
54               changes.put(record.key(), OptionalDouble.of(record.value()));
55           }
56       }
57   
58       public ClientQuotaImage apply() {
59           Map<String, Double> newQuotas = new HashMap<>(image.quotas().size());
60           for (Entry<String, Double> entry : image.quotas().entrySet()) {
61               OptionalDouble change = changes.get(entry.getKey());
62               if (change == null) {
63                   newQuotas.put(entry.getKey(), entry.getValue());
64               } else if (change.isPresent()) {
65                   newQuotas.put(entry.getKey(), change.getAsDouble());
66               }
67           }
68           for (Entry<String, OptionalDouble> entry : changes.entrySet()) {
69               if (!newQuotas.containsKey(entry.getKey())) {
70                   if (entry.getValue().isPresent()) {
71                       newQuotas.put(entry.getKey(), entry.getValue().getAsDouble());
72                   }
73               }
74           }
75           return new ClientQuotaImage(newQuotas);
76       }
77   }
