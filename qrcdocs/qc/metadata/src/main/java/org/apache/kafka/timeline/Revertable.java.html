20   /**
21    * An API which all snapshot data structures implement, indicating that their contents
22    * can be reverted to a point in time.
23    */
24   interface Revertable {
25       /**
26        * Revert to the target epoch.
27        *
28        * @param targetEpoch   The epoch to revert to.
29        * @param delta         The delta associated with this epoch for this object.
30        */
31       void executeRevert(long targetEpoch, Delta delta);
32   
33       /**
34        * Reverts to the initial value.
35        */
36       void reset();
37   }
