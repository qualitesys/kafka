252       /**
253        * Authorize an action based on the current set of ACLs.
254        *
255        * In order to know whether to allow or deny the action, we need to examine the ACLs
256        * that apply to it. If any DENY ACLs match, the operation is denied, no matter how
257        * many ALLOW ACLs match. If neither ALLOW nor DENY ACLs match, we return the default
258        * result. In general it makes more sense to configure the default result to be
259        * DENY, but some people (and unit tests) configure it as ALLOW.
260        */
261       public AuthorizationResult authorize(
262           AuthorizableRequestContext requestContext,
263           Action action
264       ) {
265           KafkaPrincipal principal = baseKafkaPrincipal(requestContext);
266           final MatchingRule rule;
267   
268           // Superusers are authorized to do anything.
269           if (superUsers.contains(principal.toString())) {
270               rule = SuperUserRule.INSTANCE;
271           } else if (!loadingComplete) {
272               throw new AuthorizerNotReadyException();
273           } else {
274               MatchingAclRule aclRule = findAclRule(
275                   matchingPrincipals(requestContext),
276                   requestContext.clientAddress().getHostAddress(),
277                   action
278               );
279   
280               if (aclRule != null) {
281                   rule = aclRule;
282               } else {
283                   // If nothing matched, we return the default result.
284                   rule = defaultRule;
285               }
286           }
287   
288           logAuditMessage(principal, requestContext, action, rule);
289           return rule.result();
290       }

292       private String buildAuditMessage(
293           KafkaPrincipal principal,
294           AuthorizableRequestContext context,
295           Action action,
296           MatchingRule rule
297       ) {
298           StringBuilder bldr = new StringBuilder();
299           bldr.append("Principal = ").append(principal);
300           bldr.append(" is ").append(rule.result() == ALLOWED ? "Allowed" : "Denied");
301           bldr.append(" operation = ").append(action.operation());
302           bldr.append(" from host = ").append(context.clientAddress().getHostAddress());
303           bldr.append(" on resource = ");
304           appendResourcePattern(action.resourcePattern(), bldr);
305           bldr.append(" for request = ").append(ApiKeys.forId(context.requestType()).name);
306           bldr.append(" with resourceRefCount = ").append(action.resourceReferenceCount());
307           bldr.append(" based on rule ").append(rule);
308           return bldr.toString();
309       }

311       private void appendResourcePattern(ResourcePattern resourcePattern, StringBuilder bldr) {
312           bldr.append(SecurityUtils.resourceTypeName(resourcePattern.resourceType()))
313               .append(":")
314               .append(resourcePattern.patternType())
315               .append(":")
316               .append(resourcePattern.name());
317       }

319       private void logAuditMessage(
320           KafkaPrincipal principal,
321           AuthorizableRequestContext requestContext,
322           Action action,
323           MatchingRule rule
324       ) {
325           switch (rule.result()) {
326               case ALLOWED:
327                   // logIfAllowed is true if access is granted to the resource as a result of this authorization.
328                   // In this case, log at debug level. If false, no access is actually granted, the result is used
329                   // only to determine authorized operations. So log only at trace level.
330                   if (action.logIfAllowed() && auditLog.isDebugEnabled()) {
331                       auditLog.debug(buildAuditMessage(principal, requestContext, action, rule));
332                   } else if (auditLog.isTraceEnabled()) {
333                       auditLog.trace(buildAuditMessage(principal, requestContext, action, rule));
334                   }
335                   return;
336   
337               case DENIED:
338                   // logIfDenied is true if access to the resource was explicitly requested. Since this is an attempt
339                   // to access unauthorized resources, log at info level. If false, this is either a request to determine
340                   // authorized operations or a filter (e.g for regex subscriptions) to filter out authorized resources.
341                   // In this case, log only at trace level.
342                   if (action.logIfDenied()) {
343                       auditLog.info(buildAuditMessage(principal, requestContext, action, rule));
344                   } else if (auditLog.isTraceEnabled()) {
345                       auditLog.trace(buildAuditMessage(principal, requestContext, action, rule));
346                   }
347           }
348       }

350       private MatchingAclRule findAclRule(
351           Set<KafkaPrincipal> matchingPrincipals,
352           String host,
353           Action action
354       ) {
355           // This code relies on the ordering of StandardAcl within the NavigableMap.
356           // Entries are sorted by resource type first, then REVERSE resource name.
357           // Therefore, we can find all the applicable ACLs by starting at
358           // (resource_type, resource_name) and stepping forwards until we reach an ACL with
359           // a resource name which is not a prefix of the current one.
360           //
361           // For example, when trying to authorize a TOPIC resource named foobar, we would
362           // start at element 2, and continue on to 3 and 4 following map:
363           //
364           // 1. rs=TOPIC rn=gar pt=PREFIX
365           // 2. rs=TOPIC rn=foobar pt=PREFIX
366           // 3. rs=TOPIC rn=foob pt=LITERAL
367           // 4. rs=TOPIC rn=foo pt=PREFIX
368           // 5. rs=TOPIC rn=eeee pt=LITERAL
369           //
370           // Once we reached element 5, we would stop scanning.
371           MatchingAclBuilder matchingAclBuilder = new MatchingAclBuilder();
372           StandardAcl exemplar = new StandardAcl(
373               action.resourcePattern().resourceType(),
374               action.resourcePattern().name(),
375               PatternType.UNKNOWN, // Note that the UNKNOWN value sorts before all others.
376               "",
377               "",
378               AclOperation.UNKNOWN,
379               AclPermissionType.UNKNOWN);
380           checkSection(action, exemplar, matchingPrincipals, host, matchingAclBuilder);
381           if (matchingAclBuilder.foundDeny()) {
382               return matchingAclBuilder.build();
383           }
384   
385           // In addition to ACLs for this specific resource name, there can also be wildcard
386           // ACLs that match any resource name. These are stored as type = LITERAL,
387           // name = "*". We search these next.
388           exemplar = new StandardAcl(
389               action.resourcePattern().resourceType(),
390               WILDCARD,
391               LITERAL,
392               "",
393               "",
394               AclOperation.UNKNOWN,
395               AclPermissionType.UNKNOWN);
396           checkSection(action, exemplar, matchingPrincipals, host, matchingAclBuilder);
397           return matchingAclBuilder.build();
398       }

400       private void checkSection(
401           Action action,
402           StandardAcl exemplar,
403           Set<KafkaPrincipal> matchingPrincipals,
404           String host,
405           MatchingAclBuilder matchingAclBuilder
406       ) {
407           NavigableSet<StandardAcl> tailSet = aclsByResource.tailSet(exemplar, true);
408           String resourceName = action.resourcePattern().name();
409           for (Iterator<StandardAcl> iterator = tailSet.iterator();
410                iterator.hasNext(); ) {
411               StandardAcl acl = iterator.next();
412               if (!acl.resourceType().equals(action.resourcePattern().resourceType())) {
413                   // We've stepped outside the section for the resource type we care about and
414                   // should stop scanning.
415                   break;
416               }
417               if (resourceName.startsWith(acl.resourceName())) {
418                   if (acl.patternType() == LITERAL && !resourceName.equals(acl.resourceName())) {
419                       // This is a literal ACL whose name is a prefix of the resource name, but
420                       // which doesn't match it exactly. We should skip over this ACL, but keep
421                       // scanning in case there are any relevant PREFIX ACLs.
422                       continue;
423                   }
424               } else if (!(acl.resourceName().equals(WILDCARD) && acl.patternType() == LITERAL)) {
425                   // If the ACL resource name is NOT a prefix of the current resource name,
426                   // and we're not dealing with the special case of a wildcard ACL, we've
427                   // stepped outside of the section we care about and should stop scanning.
428                   break;
429               }
430               AuthorizationResult result = findResult(action, matchingPrincipals, host, acl);
431               if (ALLOWED == result) {
432                   matchingAclBuilder.allowAcl = acl;
433               } else if (DENIED == result) {
434                   matchingAclBuilder.denyAcl = acl;
435                   return;
436               }
437           }
438       }

470       static Set<KafkaPrincipal> matchingPrincipals(AuthorizableRequestContext context) {
471           KafkaPrincipal sessionPrincipal = context.principal();
472           KafkaPrincipal basePrincipal = sessionPrincipal.getClass().equals(KafkaPrincipal.class)
473               ? sessionPrincipal
474               : new KafkaPrincipal(sessionPrincipal.getPrincipalType(), sessionPrincipal.getName());
475           return Utils.mkSet(basePrincipal, WILDCARD_KAFKA_PRINCIPAL);
476       }

478       /**
479        * Determine what the result of applying an ACL to the given action and request
480        * context should be. Note that this function assumes that the resource name matches;
481        * the resource name is not checked here.
482        *
483        * @param action             The input action.
484        * @param matchingPrincipals The set of input matching principals
485        * @param host               The input host.
486        * @param acl                The input ACL.
487        * @return                   null if the ACL does not match. The authorization result
488        *                           otherwise.
489        */
490       static AuthorizationResult findResult(Action action,
491                                             Set<KafkaPrincipal> matchingPrincipals,
492                                             String host,
493                                             StandardAcl acl) {
494           // Check if the principal matches. If it doesn't, return no result (null).
495           if (!matchingPrincipals.contains(acl.kafkaPrincipal())) {
496               return null;
497           }
498           // Check if the host matches. If it doesn't, return no result (null).
499           if (!acl.host().equals(WILDCARD) && !acl.host().equals(host)) {
500               return null;
501           }
502           // Check if the operation field matches. Here we hit a slight complication.
503           // ACLs for various operations (READ, WRITE, DELETE, ALTER), "imply" the presence
504           // of DESCRIBE, even if it isn't explictly stated. A similar rule applies to
505           // DESCRIBE_CONFIGS.
506           //
507           // But this rule only applies to ALLOW ACLs. So for example, a DENY ACL for READ
508           // on a resource does not DENY describe for that resource.
509           if (acl.operation() != ALL) {
510               if (acl.permissionType().equals(ALLOW)) {
511                   switch (action.operation()) {
512                       case DESCRIBE:
513                           if (!IMPLIES_DESCRIBE.contains(acl.operation())) return null;
514                           break;
515                       case DESCRIBE_CONFIGS:
516                           if (!IMPLIES_DESCRIBE_CONFIGS.contains(acl.operation())) return null;
517                           break;
518                       default:
519                           if (action.operation() != acl.operation()) {
520                               return null;
521                           }
522                           break;
523                   }
524               } else if (action.operation() != acl.operation()) {
525                   return null;
526               }
527           }
528   
529           return acl.permissionType().equals(ALLOW) ? ALLOWED : DENIED;
530       }
