39   /**
40    * PartitionChangeBuilder handles changing partition registrations.
41    */
42   public class PartitionChangeBuilder {
43       private static final Logger log = LoggerFactory.getLogger(PartitionChangeBuilder.class);
44   
45       public static boolean changeRecordIsNoOp(PartitionChangeRecord record) {
46           if (record.isr() != null) return false;
47           if (record.leader() != NO_LEADER_CHANGE) return false;
48           if (record.replicas() != null) return false;
49           if (record.removingReplicas() != null) return false;
50           if (record.addingReplicas() != null) return false;
51           if (record.leaderRecoveryState() != LeaderRecoveryState.NO_CHANGE) return false;
52           return true;
53       }
54   
55       /**
56        * Election types.
57        */
58       public enum Election {
59           /**
60            * Perform leader election to keep the partition online. Elect the preferred replica if it is in the ISR.
61            */
62           PREFERRED,
63           /**
64            * Perform leader election from the ISR to keep the partition online.
65            */
66           ONLINE,
67           /**
68            * Prefer replicas in the ISR but keep the partition online even if it requires picking a leader that is not in the ISR.
69            */
70           UNCLEAN
71       }

132       // VisibleForTesting
133       static class ElectionResult {
134           final int node;
135           final boolean unclean;
136   
137           private ElectionResult(int node, boolean unclean) {
138               this.node = node;
139               this.unclean = unclean;
140           }
141       }
142   
143       // VisibleForTesting
144       /**
145        * Perform leader election based on the partition state and leader election type.
146        *
147        * See documentation for the Election type to see more details on the election types supported.
148        */
149       ElectionResult electLeader() {
150           if (election == Election.PREFERRED) {
151               return electPreferredLeader();
152           }
153   
154           return electAnyLeader();
155       }
156   
157       /**
158        * Assumes that the election type is Election.PREFERRED
159        */
160       private ElectionResult electPreferredLeader() {
161           int preferredReplica = targetReplicas.get(0);
162           if (isValidNewLeader(preferredReplica)) {
163               return new ElectionResult(preferredReplica, false);
164           }
165   
166           if (isValidNewLeader(partition.leader)) {
167               // Don't consider a new leader since the current leader meets all the constraints
168               return new ElectionResult(partition.leader, false);
169           }
170   
171           Optional<Integer> onlineLeader = targetReplicas.stream()
172               .skip(1)
173               .filter(this::isValidNewLeader)
174               .findFirst();
175           if (onlineLeader.isPresent()) {
176               return new ElectionResult(onlineLeader.get(), false);
177           }
178   
179           return new ElectionResult(NO_LEADER, false);
180       }
181   
182       /**
183        * Assumes that the election type is either Election.ONLINE or Election.UNCLEAN
184        */
185       private ElectionResult electAnyLeader() {
186           if (isValidNewLeader(partition.leader)) {
187               // Don't consider a new leader since the current leader meets all the constraints
188               return new ElectionResult(partition.leader, false);
189           }
190   
191           Optional<Integer> onlineLeader = targetReplicas.stream()
192               .filter(this::isValidNewLeader)
193               .findFirst();
194           if (onlineLeader.isPresent()) {
195               return new ElectionResult(onlineLeader.get(), false);
196           }
197   
198           if (election == Election.UNCLEAN) {
199               // Attempt unclean leader election
200               Optional<Integer> uncleanLeader = targetReplicas.stream()
201                   .filter(replica -> isAcceptableLeader.apply(replica))
202                   .findFirst();
203               if (uncleanLeader.isPresent()) {
204                   return new ElectionResult(uncleanLeader.get(), true);
205               }
206           }
207   
208           return new ElectionResult(NO_LEADER, false);
209       }
210   
211       private boolean isValidNewLeader(int replica) {
212           return targetIsr.contains(replica) && isAcceptableLeader.apply(replica);
213       }
214   
215       private void tryElection(PartitionChangeRecord record) {
216           ElectionResult electionResult = electLeader();
217           if (electionResult.node != partition.leader) {
218               log.debug(
219                   "Setting new leader for topicId {}, partition {} to {} using {} election",
220                   topicId,
221                   partitionId,
222                   electionResult.node,
223                   electionResult.unclean ? "an unclean" : "a clean"
224               );
225               record.setLeader(electionResult.node);
226               if (electionResult.unclean) {
227                   // If the election was unclean, we have to forcibly set the ISR to just the
228                   // new leader. This can result in data loss!
229                   record.setIsr(Collections.singletonList(electionResult.node));
230                   if (partition.leaderRecoveryState != LeaderRecoveryState.RECOVERING &&
231                       isLeaderRecoverySupported) {
232                       // And mark the leader recovery state as RECOVERING
233                       record.setLeaderRecoveryState(LeaderRecoveryState.RECOVERING.value());
234                   }
235               }
236           } else {
237               log.debug("Failed to find a new leader with current state: {}", this);
238           }
239       }
240   
241       /**
242        * Trigger a leader epoch bump if one is needed.
243        *
244        * We need to bump the leader epoch if:
245        * 1. The leader changed, or
246        * 2. The new ISR does not contain all the nodes that the old ISR did, or
247        * 3. The new replica list does not contain all the nodes that the old replica list did.
248        *
249        * Changes that do NOT fall in any of these categories will increase the partition epoch, but
250        * not the leader epoch. Note that if the leader epoch increases, the partition epoch will
251        * always increase as well; there is no case where the partition epoch increases more slowly
252        * than the leader epoch.
253        *
254        * If the PartitionChangeRecord sets the leader field to something other than
255        * NO_LEADER_CHANGE, a leader epoch bump will automatically occur. That takes care of
256        * case 1. In this function, we check for cases 2 and 3, and handle them by manually
257        * setting record.leader to the current leader.
258        */
259       void triggerLeaderEpochBumpIfNeeded(PartitionChangeRecord record) {
260           if (record.leader() == NO_LEADER_CHANGE) {
261               if (!Replicas.contains(targetIsr, partition.isr) ||
262                       !Replicas.contains(targetReplicas, partition.replicas)) {
263                   record.setLeader(partition.leader);
264               }
265           }
266       }
267   
268       private void completeReassignmentIfNeeded() {
269           // Check if there is a reassignment to complete.
270           if (targetRemoving.isEmpty() && targetAdding.isEmpty()) return;
271   
272           List<Integer> newTargetIsr = targetIsr;
273           List<Integer> newTargetReplicas = targetReplicas;
274           if (!targetRemoving.isEmpty()) {
275               newTargetIsr = new ArrayList<>(targetIsr.size());
276               for (int replica : targetIsr) {
277                   if (!targetRemoving.contains(replica)) {
278                       newTargetIsr.add(replica);
279                   }
280               }
281               if (newTargetIsr.isEmpty()) return;
282               newTargetReplicas = new ArrayList<>(targetReplicas.size());
283               for (int replica : targetReplicas) {
284                   if (!targetRemoving.contains(replica)) {
285                       newTargetReplicas.add(replica);
286                   }
287               }
288               if (newTargetReplicas.isEmpty()) return;
289           }
290           for (int replica : targetAdding) {
291               if (!newTargetIsr.contains(replica)) return;
292           }
293           targetIsr = newTargetIsr;
294           targetReplicas = newTargetReplicas;
295           targetRemoving = Collections.emptyList();
296           targetAdding = Collections.emptyList();
297       }
298   
299       public Optional<ApiMessageAndVersion> build() {
300           PartitionChangeRecord record = new PartitionChangeRecord().
301               setTopicId(topicId).
302               setPartitionId(partitionId);
303   
304           completeReassignmentIfNeeded();
305   
306           tryElection(record);
307   
308           triggerLeaderEpochBumpIfNeeded(record);
309   
310           if (record.isr() == null && !targetIsr.isEmpty() && !targetIsr.equals(Replicas.toList(partition.isr))) {
311               // Set the new ISR if it is different from the current ISR and unclean leader election didn't already set it.
312               record.setIsr(targetIsr);
313           }
314           if (!targetReplicas.isEmpty() && !targetReplicas.equals(Replicas.toList(partition.replicas))) {
315               record.setReplicas(targetReplicas);
316           }
317           if (!targetRemoving.equals(Replicas.toList(partition.removingReplicas))) {
318               record.setRemovingReplicas(targetRemoving);
319           }
320           if (!targetAdding.equals(Replicas.toList(partition.addingReplicas))) {
321               record.setAddingReplicas(targetAdding);
322           }
323           if (targetLeaderRecoveryState != partition.leaderRecoveryState) {
324               record.setLeaderRecoveryState(targetLeaderRecoveryState.value());
325           }
326   
327           if (changeRecordIsNoOp(record)) {
328               return Optional.empty();
329           } else {
330               return Optional.of(new ApiMessageAndVersion(record,
331                   PARTITION_CHANGE_RECORD.highestSupportedVersion()));
332           }
333       }
