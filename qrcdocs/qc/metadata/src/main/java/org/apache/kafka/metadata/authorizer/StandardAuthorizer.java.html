47   /**
48    * The standard authorizer which is used in KRaft-based clusters if no other authorizer is
49    * configured.
50    */
51   public class StandardAuthorizer implements ClusterMetadataAuthorizer {
52       public final static String SUPER_USERS_CONFIG = "super.users";
53   
54       public final static String ALLOW_EVERYONE_IF_NO_ACL_IS_FOUND_CONFIG = "allow.everyone.if.no.acl.found";
55   
56       /**
57        * A future which is completed once we have loaded up to the initial high water mark.
58        */
59       private final CompletableFuture<Void> initialLoadFuture = new CompletableFuture<>();
60   
61       /**
62        * The current data. Can be read without a lock. Must be written while holding the object lock.
63        */
64       private volatile StandardAuthorizerData data = StandardAuthorizerData.createEmpty();
65   
66       @Override
67       public synchronized void setAclMutator(AclMutator aclMutator) {
68           this.data = data.copyWithNewAclMutator(aclMutator);
69       }
70   
71       @Override
72       public AclMutator aclMutatorOrException() {
73           AclMutator aclMutator = data.aclMutator;
74           if (aclMutator == null) {
75               throw new NotControllerException("The current node is not the active controller.");
76           }
77           return aclMutator;
78       }
79   
80       @Override
81       public synchronized void completeInitialLoad() {
82           data = data.copyWithNewLoadingComplete(true);
83           data.log.info("Completed initial ACL load process.");
84           initialLoadFuture.complete(null);
85       }
86   
87       // Visible for testing
88       public CompletableFuture<Void> initialLoadFuture() {
89           return initialLoadFuture;
90       }
91   
92       @Override
93       public void completeInitialLoad(Exception e) {
94           data.log.error("Failed to complete initial ACL load process.", e);
95           initialLoadFuture.completeExceptionally(e);
96       }
97   
98       @Override
99       public void addAcl(Uuid id, StandardAcl acl) {
100           data.addAcl(id, acl);
101       }
102   
103       @Override
104       public void removeAcl(Uuid id) {
105           data.removeAcl(id);
106       }
107   
108       @Override
109       public synchronized void loadSnapshot(Map<Uuid, StandardAcl> acls) {
110           data = data.copyWithNewAcls(acls.entrySet());
111       }
112   
113       @Override
114       public Map<Endpoint, ? extends CompletionStage<Void>> start(
115               AuthorizerServerInfo serverInfo) {
116           Map<Endpoint, CompletableFuture<Void>> result = new HashMap<>();
117           for (Endpoint endpoint : serverInfo.endpoints()) {
118               if (serverInfo.earlyStartListeners().contains(
119                       endpoint.listenerName().orElseGet(() -> ""))) {
120                   result.put(endpoint, CompletableFuture.completedFuture(null));
121               } else {
122                   result.put(endpoint, initialLoadFuture);
123               }
124           }
125           return result;
126       }
127   
128       @Override
129       public List<AuthorizationResult> authorize(
130               AuthorizableRequestContext requestContext,
131               List<Action> actions) {
132           StandardAuthorizerData curData = data;
133           List<AuthorizationResult> results = new ArrayList<>(actions.size());
134           for (Action action: actions) {
135               AuthorizationResult result = curData.authorize(requestContext, action);
136               results.add(result);
137           }
138           return results;
139       }
140   
141       @Override
142       public Iterable<AclBinding> acls(AclBindingFilter filter) {
143           return data.acls(filter);
144       }
145   
146       @Override
147       public int aclCount() {
148           return data.aclCount();
149       }
150   
151       @Override
152       public void close() throws IOException {
153           // Complete the initialLoadFuture, if it hasn't been completed already.
154           initialLoadFuture.completeExceptionally(new TimeoutException("The authorizer was " +
155               "closed before the initial load could complete."));
156       }
157   
158       @Override
159       public synchronized void configure(Map<String, ?> configs) {
160           Set<String> superUsers = getConfiguredSuperUsers(configs);
161           AuthorizationResult defaultResult = getDefaultResult(configs);
162           int nodeId;
163           try {
164               nodeId = Integer.parseInt(configs.get("node.id").toString());
165           } catch (Exception e) {
166               nodeId = -1;
167           }
168           this.data = data.copyWithNewConfig(nodeId, superUsers, defaultResult);
169           this.data.log.info("set super.users={}, default result={}", String.join(",", superUsers), defaultResult);
170       }
171   
172       // VisibleForTesting
173       Set<String> superUsers()  {
174           return new HashSet<>(data.superUsers());
175       }
176   
177       AuthorizationResult defaultResult() {
178           return data.defaultResult();
179       }
180   
181       static Set<String> getConfiguredSuperUsers(Map<String, ?> configs) {
182           Object configValue = configs.get(SUPER_USERS_CONFIG);
183           if (configValue == null) return Collections.emptySet();
184           String[] values = configValue.toString().split(";");
185           Set<String> result = new HashSet<>();
186           for (String value : values) {
187               String v = value.trim();
188               if (!v.isEmpty()) {
189                   SecurityUtils.parseKafkaPrincipal(v);
190                   result.add(v);
191               }
192           }
193           return result;
194       }
195   
196       static AuthorizationResult getDefaultResult(Map<String, ?> configs) {
197           Object configValue = configs.get(ALLOW_EVERYONE_IF_NO_ACL_IS_FOUND_CONFIG);
198           if (configValue == null) return DENIED;
199           return Boolean.valueOf(configValue.toString()) ? ALLOWED : DENIED;
200       }
