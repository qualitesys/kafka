26   /**
27    * SnapshottableHashTable implements a hash table that supports creating point-in-time
28    * snapshots.  Each snapshot is immutable once it is created; the past cannot be changed.
29    * We handle divergences between the current state and historical state by copying a
30    * reference to elements that have been deleted or overwritten into the most recent
31    * snapshot tier.
32    *
33    * Note that there are no keys in SnapshottableHashTable, only values.  So it more similar
34    * to a hash set than a hash map.  The subclasses implement full-featured maps and sets
35    * using this class as a building block.
36    *
37    * Each snapshot tier contains a size and a hash table.  The size reflects the size at
38    * the time the snapshot was taken.  Note that, as an optimization, snapshot tiers will
39    * be null if they don't contain anything.  So for example, if snapshot 20 of Object O
40    * contains the same entries as snapshot 10 of that object, the snapshot 20 tier for
41    * object O will be null.
42    *
43    * The current tier's data is stored in the fields inherited from BaseHashTable.  It
44    * would be conceptually simpler to have a separate BaseHashTable object, but since Java
45    * doesn't have value types, subclassing is the only way to avoid another pointer
46    * indirection and the associated extra memory cost.
47    *
48    * Note that each element in the hash table contains a start epoch, and a value.  The
49    * start epoch is there to identify when the object was first inserted.  This in turn
50    * determines which snapshots it is a member of.
51    *
52    * In order to retrieve an object from snapshot E, we start by checking to see if the
53    * object exists in the "current" hash tier.  If it does, and its startEpoch extends back
54    * to E, we return that object.  Otherwise, we check all the snapshot tiers, starting
55    * with E, and ending with the most recent snapshot, to see if the object is there.
56    * As an optimization, if we encounter the object in a snapshot tier but its epoch is too
57    * new, we know that its value at epoch E must be null, so we can return that immediately.
58    *
59    * The class hierarchy looks like this:
60    *
61    *        Revertable       BaseHashTable
62    *              ↑              ↑
63    *           SnapshottableHashTable → SnapshotRegistry → Snapshot
64    *               ↑             ↑
65    *   TimelineHashSet       TimelineHashMap
66    *
67    * BaseHashTable is a simple hash table that uses separate chaining.  The interface is
68    * pretty bare-bones since this class is not intended to be used directly by end-users.
69    *
70    * This class, SnapshottableHashTable, has the logic for snapshotting and iterating over
71    * snapshots.  This is the core of the snapshotted hash table code and handles the
72    * tiering.
73    *
74    * TimelineHashSet and TimelineHashMap are mostly wrappers around this
75    * SnapshottableHashTable class.  They implement standard Java APIs for Set and Map,
76    * respectively.  There's a fair amount of boilerplate for this, but it's necessary so
77    * that timeline data structures can be used while writing idiomatic Java code.
78    * The accessor APIs have two versions -- one that looks at the current state, and one
79    * that looks at a historical snapshotted state.  Mutation APIs only ever mutate the
80    * current state.
81    *
82    * One very important feature of SnapshottableHashTable is that we support iterating
83    * over a snapshot even while changes are being made to the current state.  See the
84    * Javadoc for the iterator for more information about how this is accomplished.
85    *
86    * All of these classes require external synchronization, and don't support null keys or
87    * values.
88    */
89   class SnapshottableHashTable<T extends SnapshottableHashTable.ElementWithStartEpoch>
90           extends BaseHashTable<T> implements Revertable {

102       static class HashTier<T extends SnapshottableHashTable.ElementWithStartEpoch> implements Delta {
103           private final int size;
104           private BaseHashTable<T> deltaTable;
105   
106           HashTier(int size) {
107               this.size = size;
108           }

110           @SuppressWarnings("unchecked")
111           @Override
112           public void mergeFrom(long epoch, Delta source) {
113               HashTier<T> other = (HashTier<T>) source;
114               List<T> list = new ArrayList<>();
115               Object[] otherElements = other.deltaTable.baseElements();
116               for (int slot = 0; slot < otherElements.length; slot++) {
117                   BaseHashTable.unpackSlot(list, otherElements, slot);
118                   for (T element : list) {
119                       // When merging in a later hash tier, we want to keep only the elements
120                       // that were present at our epoch.
121                       if (element.startEpoch() <= epoch) {
122                           deltaTable.baseAddOrReplace(element);
123                       }
124                   }
125               }
126           }
127       }

129       /**
130        * Iterate over the values that currently exist in the hash table.
131        *
132        * You can use this iterator even if you are making changes to the map.
133        * The changes may or may not be visible while you are iterating.
134        */
135       class CurrentIterator implements Iterator<T> {
136           private final Object[] topTier;
137           private final List<T> ready;
138           private int slot;
139           private T lastReturned;
140   
141           CurrentIterator(Object[] topTier) {
142               this.topTier = topTier;
143               this.ready = new ArrayList<>();
144               this.slot = 0;
145               this.lastReturned = null;
146           }
147   
148           @Override
149           public boolean hasNext() {
150               while (ready.isEmpty()) {
151                   if (slot == topTier.length) {
152                       return false;
153                   }
154                   BaseHashTable.unpackSlot(ready, topTier, slot);
155                   slot++;
156               }
157               return true;
158           }
159   
160           @Override
161           public T next() {
162               if (!hasNext()) {
163                   throw new NoSuchElementException();
164               }
165               lastReturned = ready.remove(ready.size() - 1);
166               return lastReturned;
167           }
168   
169           @Override
170           public void remove() {
171               if (lastReturned == null) {
172                   throw new UnsupportedOperationException("remove");
173               }
174               snapshottableRemove(lastReturned);
175               lastReturned = null;
176           }
177       }

179       /**
180        * Iterate over the values that existed in the hash table during a specific snapshot.
181        *
182        * You can use this iterator even if you are making changes to the map.
183        * The snapshot is immutable and will always show up the same.
184        */
185       class HistoricalIterator implements Iterator<T> {
186           private final Object[] topTier;
187           private final Snapshot snapshot;
188           private final List<T> temp;
189           private final List<T> ready;
190           private int slot;
191   
192           HistoricalIterator(Object[] topTier, Snapshot snapshot) {
193               this.topTier = topTier;
194               this.snapshot = snapshot;
195               this.temp = new ArrayList<>();
196               this.ready = new ArrayList<>();
197               this.slot = 0;
198           }
199   
200           @Override
201           public boolean hasNext() {
202               while (ready.isEmpty()) {
203                   if (slot == topTier.length) {
204                       return false;
205                   }
206                   BaseHashTable.unpackSlot(temp, topTier, slot);
207                   for (T object : temp) {
208                       if (object.startEpoch() <= snapshot.epoch()) {
209                           ready.add(object);
210                       }
211                   }
212                   temp.clear();
213   
214                   /*
215                    * As we iterate over the SnapshottableHashTable, elements may move from
216                    * the top tier into the snapshot tiers.  This would happen if something
217                    * were deleted in the top tier, for example, but still retained in the
218                    * snapshot.
219                    *
220                    * We don't want to return any elements twice, though.  Therefore, we
221                    * iterate over the top tier and the snapshot tier at the
222                    * same time.  The key to understanding how this works is realizing that
223                    * both hash tables use the same hash function, but simply choose a
224                    * different number of significant bits based on their size.
225                    * So if the top tier has size 4 and the snapshot tier has size 2, we have
226                    * the following mapping:
227                    *
228                    * Elements that would be in slot 0 or 1 in the top tier can only be in
229                    * slot 0 in the snapshot tier.
230                    * Elements that would be in slot 2 or 3 in the top tier can only be in
231                    * slot 1 in the snapshot tier.
232                    *
233                    * Therefore, we can do something like this:
234                    * 1. check slot 0 in the top tier and slot 0 in the snapshot tier.
235                    * 2. check slot 1 in the top tier and slot 0 in the snapshot tier.
236                    * 3. check slot 2 in the top tier and slot 1 in the snapshot tier.
237                    * 4. check slot 3 in the top tier and slot 1 in the snapshot tier.
238                    *
239                    * If elements move from the top tier to the snapshot tier, then
240                    * we'll still find them and report them exactly once.
241                    *
242                    * Note that while I used 4 and 2 as example sizes here, the same pattern
243                    * holds for different powers of two.  The "snapshot slot" of an element
244                    * will be the top few bits of the top tier slot of that element.
245                    */
246                   Iterator<Snapshot> iterator = snapshotRegistry.iterator(snapshot);
247                   while (iterator.hasNext()) {
248                       Snapshot curSnapshot = iterator.next();
249                       HashTier<T> tier = curSnapshot.getDelta(SnapshottableHashTable.this);
250                       if (tier != null && tier.deltaTable != null) {
251                           BaseHashTable<T> deltaTable = tier.deltaTable;
252                           int shift = Integer.numberOfLeadingZeros(deltaTable.baseElements().length) -
253                               Integer.numberOfLeadingZeros(topTier.length);
254                           int tierSlot = slot >>> shift;
255                           BaseHashTable.unpackSlot(temp, deltaTable.baseElements(), tierSlot);
256                           for (T object : temp) {
257                               if (BaseHashTable.findSlot(object, topTier.length) == slot) {
258                                   ready.add(object);
259                               }
260                           }
261                           temp.clear();
262                       }
263                   }
264                   slot++;
265               }
266               return true;
267           }
268   
269           @Override
270           public T next() {
271               if (!hasNext()) {
272                   throw new NoSuchElementException();
273               }
274               return ready.remove(ready.size() - 1);
275           }
276       }

286       int snapshottableSize(long epoch) {
287           if (epoch == LATEST_EPOCH) {
288               return baseSize();
289           } else {
290               Iterator<Snapshot> iterator = snapshotRegistry.iterator(epoch);
291               while (iterator.hasNext()) {
292                   Snapshot snapshot = iterator.next();
293                   HashTier<T> tier = snapshot.getDelta(SnapshottableHashTable.this);
294                   if (tier != null) {
295                       return tier.size;
296                   }
297               }
298               return baseSize();
299           }
300       }

302       T snapshottableGet(Object key, long epoch) {
303           T result = baseGet(key);
304           if (result != null && result.startEpoch() <= epoch) {
305               return result;
306           }
307           if (epoch == LATEST_EPOCH) {
308               return null;
309           }
310           Iterator<Snapshot> iterator = snapshotRegistry.iterator(epoch);
311           while (iterator.hasNext()) {
312               Snapshot snapshot = iterator.next();
313               HashTier<T> tier = snapshot.getDelta(SnapshottableHashTable.this);
314               if (tier != null && tier.deltaTable != null) {
315                   result = tier.deltaTable.baseGet(key);
316                   if (result != null) {
317                       if (result.startEpoch() <= epoch) {
318                           return result;
319                       } else {
320                           return null;
321                       }
322                   }
323               }
324           }
325           return null;
326       }

328       boolean snapshottableAddUnlessPresent(T object) {
329           T prev = baseGet(object);
330           if (prev != null) {
331               return false;
332           }
333           object.setStartEpoch(snapshotRegistry.latestEpoch() + 1);
334           int prevSize = baseSize();
335           baseAddOrReplace(object);
336           updateTierData(prevSize);
337           return true;
338       }
339   
340       T snapshottableAddOrReplace(T object) {
341           object.setStartEpoch(snapshotRegistry.latestEpoch() + 1);
342           int prevSize = baseSize();
343           T prev = baseAddOrReplace(object);
344           if (prev == null) {
345               updateTierData(prevSize);
346           } else {
347               updateTierData(prev, prevSize);
348           }
349           return prev;
350       }

352       T snapshottableRemove(Object object) {
353           T prev = baseRemove(object);
354           if (prev == null) {
355               return null;
356           } else {
357               updateTierData(prev, baseSize() + 1);
358               return prev;
359           }

362       private void updateTierData(int prevSize) {
363           Iterator<Snapshot> iterator = snapshotRegistry.reverseIterator();
364           if (iterator.hasNext()) {
365               Snapshot snapshot = iterator.next();
366               HashTier<T> tier = snapshot.getDelta(SnapshottableHashTable.this);
367               if (tier == null) {
368                   tier = new HashTier<>(prevSize);
369                   snapshot.setDelta(SnapshottableHashTable.this, tier);
370               }
371           }
372       }
373   
374       private void updateTierData(T prev, int prevSize) {
375           Iterator<Snapshot> iterator = snapshotRegistry.reverseIterator();
376           if (iterator.hasNext()) {
377               Snapshot snapshot = iterator.next();
378               // If the previous element was present in the most recent snapshot, add it to
379               // that snapshot's hash tier.
380               if (prev.startEpoch() <= snapshot.epoch()) {
381                   HashTier<T> tier = snapshot.getDelta(SnapshottableHashTable.this);
382                   if (tier == null) {
383                       tier = new HashTier<>(prevSize);
384                       snapshot.setDelta(SnapshottableHashTable.this, tier);
385                   }
386                   if (tier.deltaTable == null) {
387                       tier.deltaTable = new BaseHashTable<>(1);
388                   }
389                   tier.deltaTable.baseAddOrReplace(prev);
390               }
391           }
392       }

394       Iterator<T> snapshottableIterator(long epoch) {
395           if (epoch == LATEST_EPOCH) {
396               return new CurrentIterator(baseElements());
397           } else {
398               return new HistoricalIterator(baseElements(), snapshotRegistry.getSnapshot(epoch));
399           }

402       String snapshottableToDebugString() {
403           StringBuilder bld = new StringBuilder();
404           bld.append(String.format("SnapshottableHashTable{%n"));
405           bld.append("top tier: ");
406           bld.append(baseToDebugString());
407           bld.append(String.format(",%nsnapshot tiers: [%n"));
408           String prefix = "";
409           for (Iterator<Snapshot> iter = snapshotRegistry.iterator(); iter.hasNext(); ) {
410               Snapshot snapshot = iter.next();
411               bld.append(prefix);
412               bld.append("epoch ").append(snapshot.epoch()).append(": ");
413               HashTier<T> tier = snapshot.getDelta(this);
414               if (tier == null) {
415                   bld.append("null");
416               } else {
417                   bld.append("HashTier{");
418                   bld.append("size=").append(tier.size);
419                   bld.append(", deltaTable=");
420                   if (tier.deltaTable == null) {
421                       bld.append("null");
422                   } else {
423                       bld.append(tier.deltaTable.baseToDebugString());
424                   }
425                   bld.append("}");
426               }
427               bld.append(String.format("%n"));
428           }
429           bld.append(String.format("]}%n"));
430           return bld.toString();
431       }

433       @SuppressWarnings("unchecked")
434       @Override
435       public void executeRevert(long targetEpoch, Delta delta) {
436           HashTier<T> tier = (HashTier<T>) delta;
437           Iterator<T> iter = snapshottableIterator(LATEST_EPOCH);
438           while (iter.hasNext()) {
439               T element = iter.next();
440               if (element.startEpoch() > targetEpoch) {
441                   iter.remove();
442               }
443           }
444           BaseHashTable<T> deltaTable = tier.deltaTable;
445           if (deltaTable != null) {
446               List<T> out = new ArrayList<>();
447               for (int i = 0; i < deltaTable.baseElements().length; i++) {
448                   BaseHashTable.unpackSlot(out, deltaTable.baseElements(), i);
449                   for (T value : out) {
450                       baseAddOrReplace(value);
451                   }
452                   out.clear();
