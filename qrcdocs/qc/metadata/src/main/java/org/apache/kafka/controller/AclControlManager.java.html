54   /**
55    * The AclControlManager manages any ACLs that are stored in the __cluster_metadata topic.
56    * If the ACLs are stored externally (such as in ZooKeeper) then there will be nothing for
57    * this manager to do, and the authorizer field will always be Optional.empty.
58    *
59    * Because the Authorizer is being concurrently used by other threads, we need to be
60    * careful about snapshots. We don't want the Authorizer to act based on partial state
61    * during the loading process. Therefore, unlike most of the other managers,
62    * AclControlManager needs to receive callbacks when we start loading a snapshot and when
63    * we finish. The prepareForSnapshotLoad callback clears the authorizer field, preventing
64    * any changes from affecting the authorizer until completeSnapshotLoad is called.
65    * Note that the Authorizer's start() method will block until the first snapshot load has
66    * completed, which is another reason the prepare / complete callbacks are needed.
67    */
68   public class AclControlManager {
69       private final TimelineHashMap<Uuid, StandardAcl> idToAcl;
70       private final TimelineHashSet<StandardAcl> existingAcls;
71       private final Optional<ClusterMetadataAuthorizer> authorizer;
72   
73       AclControlManager(SnapshotRegistry snapshotRegistry,
74                         Optional<ClusterMetadataAuthorizer> authorizer) {
75           this.idToAcl = new TimelineHashMap<>(snapshotRegistry, 0);
76           this.existingAcls = new TimelineHashSet<>(snapshotRegistry, 0);
77           this.authorizer = authorizer;
78       }

80       ControllerResult<List<AclCreateResult>> createAcls(List<AclBinding> acls) {
81           List<AclCreateResult> results = new ArrayList<>(acls.size());
82           List<ApiMessageAndVersion> records = new ArrayList<>(acls.size());
83           for (AclBinding acl : acls) {
84               try {
85                   validateNewAcl(acl);
86               } catch (Throwable t) {
87                   ApiException e = (t instanceof ApiException) ? (ApiException) t :
88                       new UnknownServerException("Unknown error while trying to create ACL", t);
89                   results.add(new AclCreateResult(e));
90                   continue;
91               }
92               StandardAcl standardAcl = StandardAcl.fromAclBinding(acl);
93               if (existingAcls.add(standardAcl)) {
94                   StandardAclWithId standardAclWithId = new StandardAclWithId(newAclId(), standardAcl);
95                   idToAcl.put(standardAclWithId.id(), standardAcl);
96                   records.add(new ApiMessageAndVersion(standardAclWithId.toRecord(), (short) 0));
97               }
98               results.add(AclCreateResult.SUCCESS);
99           }
100           return new ControllerResult<>(records, results, true);
101       }
102   
103       Uuid newAclId() {
104           Uuid uuid;
105           do {
106               uuid = Uuid.randomUuid();
107           } while (idToAcl.containsKey(uuid));
108           return uuid;
109       }

111       static void validateNewAcl(AclBinding binding) {
112           switch (binding.pattern().resourceType()) {
113               case UNKNOWN:
114               case ANY:
115                   throw new InvalidRequestException("Invalid resourceType " +
116                       binding.pattern().resourceType());
117               default:
118                   break;
119           }
120           switch (binding.pattern().patternType()) {
121               case LITERAL:
122               case PREFIXED:
123                   break;
124               default:
125                   throw new InvalidRequestException("Invalid patternType " +
126                       binding.pattern().patternType());
127           }
128           switch (binding.entry().operation()) {
129               case UNKNOWN:
130               case ANY:
131                   throw new InvalidRequestException("Invalid operation " +
132                       binding.entry().operation());
133               default:
134                   break;
135           }
136           switch (binding.entry().permissionType()) {
137               case DENY:
138               case ALLOW:
139                   break;
140               default:
141                   throw new InvalidRequestException("Invalid permissionType " +
142                       binding.entry().permissionType());
143           }
144       }

146       ControllerResult<List<AclDeleteResult>> deleteAcls(List<AclBindingFilter> filters) {
147           List<AclDeleteResult> results = new ArrayList<>();
148           Set<ApiMessageAndVersion> records = new HashSet<>();
149           for (AclBindingFilter filter : filters) {
150               try {
151                   validateFilter(filter);
152                   AclDeleteResult result = deleteAclsForFilter(filter, records);
153                   results.add(result);
154               } catch (Throwable e) {
155                   results.add(new AclDeleteResult(ApiError.fromThrowable(e).exception()));
156               }
157           }
158           return ControllerResult.atomicOf(records.stream().collect(Collectors.toList()), results);
159       }
160   
161       AclDeleteResult deleteAclsForFilter(AclBindingFilter filter,
162                                           Set<ApiMessageAndVersion> records) {
163           List<AclBindingDeleteResult> deleted = new ArrayList<>();
164           for (Entry<Uuid, StandardAcl> entry : idToAcl.entrySet()) {
165               Uuid id = entry.getKey();
166               StandardAcl acl = entry.getValue();
167               AclBinding binding = acl.toBinding();
168               if (filter.matches(binding)) {
169                   deleted.add(new AclBindingDeleteResult(binding));
170                   records.add(new ApiMessageAndVersion(
171                       new RemoveAccessControlEntryRecord().setId(id), (short) 0));
172               }
173           }
174           return new AclDeleteResult(deleted);
175       }
176   
177       static void validateFilter(AclBindingFilter filter) {
178           if (filter.patternFilter().isUnknown()) {
179               throw new InvalidRequestException("Unknown patternFilter.");
180           }
181           if (filter.entryFilter().isUnknown()) {
182               throw new InvalidRequestException("Unknown entryFilter.");
183           }
184       }

186       public void replay(AccessControlEntryRecord record,
187                          Optional<OffsetAndEpoch> snapshotId) {
188           StandardAclWithId aclWithId = StandardAclWithId.fromRecord(record);
189           idToAcl.put(aclWithId.id(), aclWithId.acl());
190           existingAcls.add(aclWithId.acl());
191           if (!snapshotId.isPresent()) {
192               authorizer.ifPresent(a -> {
193                   a.addAcl(aclWithId.id(), aclWithId.acl());
194               });
195           }
196       }
197   
198       public void replay(RemoveAccessControlEntryRecord record,
199                          Optional<OffsetAndEpoch> snapshotId) {
200           StandardAcl acl = idToAcl.remove(record.id());
201           if (acl == null) {
202               throw new RuntimeException("Unable to replay " + record + ": no acl with " +
203                   "that ID found.");
204           }
205           if (!existingAcls.remove(acl)) {
206               throw new RuntimeException("Unable to replay " + record + " for " + acl +
207                   ": acl not found " + "in existingAcls.");
208           }
209           if (!snapshotId.isPresent()) {
210               authorizer.ifPresent(a -> {
211                   a.removeAcl(record.id());
212               });
213           }
214       }

216       Map<Uuid, StandardAcl> idToAcl() {
217           return Collections.unmodifiableMap(idToAcl);
218       }
219   
220       Iterator<List<ApiMessageAndVersion>> iterator(long epoch) {
221           Iterator<Entry<Uuid, StandardAcl>> iterator = idToAcl.entrySet(epoch).iterator();
222           return new StandardAclRecordIterator(new Iterator<StandardAclWithId>() {
223               @Override
224               public boolean hasNext() {
225                   return iterator.hasNext();
226               }
227   
228               @Override
229               public StandardAclWithId next() {
230                   Entry<Uuid, StandardAcl> entry = iterator.next();
231                   return new StandardAclWithId(entry.getKey(), entry.getValue());
232               }
233           });
234       }
