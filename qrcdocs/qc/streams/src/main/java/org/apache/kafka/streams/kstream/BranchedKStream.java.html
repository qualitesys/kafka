18   
19   import java.util.Map;

21   /**
22    * Branches the records in the original stream based on the predicates supplied for the branch definitions.
23    * <p>
24    * Branches are defined with {@link BranchedKStream#branch(Predicate, Branched)} or
25    * {@link BranchedKStream#defaultBranch(Branched)} methods. Each record is evaluated against the {@code predicate}
26    * supplied via {@link Branched} parameters, and is routed to the first branch for which its respective predicate
27    * evaluates to {@code true}. If a record does not match any predicates, it will be routed to the default branch,
28    * or dropped if no default branch is created.
29    * <p>
30    * Each branch (which is a {@link KStream} instance) then can be processed either by
31    * a {@link java.util.function.Function} or a {@link java.util.function.Consumer} provided via a {@link Branched}
32    * parameter. If certain conditions are met, it also can be accessed from the {@link Map} returned by an optional
33    * {@link BranchedKStream#defaultBranch(Branched)} or {@link BranchedKStream#noDefaultBranch()} method call
34    * (see <a href="#examples">usage examples</a>).
35    * <p>
36    * The branching happens on a first-match basis: A record in the original stream is assigned to the corresponding result
37    * stream for the first predicate that evaluates to {@code true}, and is assigned to this stream only. If you need
38    * to route a record to multiple streams, you can apply multiple {@link KStream#filter(Predicate)} operators
39    * to the same {@link KStream} instance, one for each predicate, instead of branching.
40    * <p>
41    * The process of routing the records to different branches is a stateless record-by-record operation.
42    *
43    * <h2><a name="maprules">Rules of forming the resulting map</a></h2>
44    * The keys of the {@code Map<String, KStream<K, V>>} entries returned by {@link BranchedKStream#defaultBranch(Branched)} or
45    * {@link BranchedKStream#noDefaultBranch()} are defined by the following rules:
46    * <ul>
47    *     <li>If {@link Named} parameter was provided for {@link KStream#split(Named)}, its value is used as
48    *     a prefix for each key. By default, no prefix is used
49    *     <li>If a branch name is provided in {@link BranchedKStream#branch(Predicate, Branched)} via the
50    *     {@link Branched} parameter, its value is appended to the prefix to form the {@code Map} key
51    *     <li>If a name is not provided for the branch, then the key defaults to {@code prefix + position} of the branch
52    *     as a decimal number, starting from {@code "1"}
53    *     <li>If a name is not provided for the {@link BranchedKStream#defaultBranch()}, then the key defaults
54    *     to {@code prefix + "0"}
55    * </ul>
56    * The values of the respective {@code Map<Stream, KStream<K, V>>} entries are formed as following:
57    * <ul>
58    *     <li>If no chain function or consumer is provided in {@link BranchedKStream#branch(Predicate, Branched)} via
59    *     the {@link Branched} parameter, then the branch itself is added to the {@code Map}
60    *     <li>If chain function is provided and it returns a non-null value for a given branch, then the value
61    *     is the result returned by this function
62    *     <li>If a chain function returns {@code null} for a given branch, then no entry is added to the map
63    *     <li>If a consumer is provided for a given branch, then no entry is added to the map
64    * </ul>
65    * For example:
66    * <pre> {@code
67    * Map<String, KStream<..., ...>> result =
68    *   source.split(Named.as("foo-"))
69    *     .branch(predicate1, Branched.as("bar"))                    // "foo-bar"
70    *     .branch(predicate2, Branched.withConsumer(ks->ks.to("A"))  // no entry: a Consumer is provided
71    *     .branch(predicate3, Branched.withFunction(ks->null))       // no entry: chain function returns null
72    *     .branch(predicate4, Branched.withFunction(ks->ks))         // "foo-4": chain function returns non-null value
73    *     .branch(predicate5)                                        // "foo-5": name defaults to the branch position
74    *     .defaultBranch()                                           // "foo-0": "0" is the default name for the default branch
75    * }</pre>
76    *
77    * <h2><a name="examples">Usage examples</a></h2>
78    *
79    * <h3>Direct Branch Consuming</h3>
80    * In many cases we do not need to have a single scope for all the branches, each branch being processed completely
81    * independently from others. Then we can use 'consuming' lambdas or method references in {@link Branched} parameter:
82    *
83    * <pre> {@code
84    * source.split()
85    *     .branch(predicate1, Branched.withConsumer(ks -> ks.to("A")))
86    *     .branch(predicate2, Branched.withConsumer(ks -> ks.to("B")))
87    *     .defaultBranch(Branched.withConsumer(ks->ks.to("C")));
88    * }</pre>
89    *
90    * <h3>Collecting branches in a single scope</h3>
91    * In other cases we want to combine branches again after splitting. The map returned by
92    * {@link BranchedKStream#defaultBranch()} or {@link BranchedKStream#noDefaultBranch()} methods provides
93    * access to all the branches in the same scope:
94    *
95    * <pre> {@code
96    * Map<String, KStream<String, String>> branches = source.split(Named.as("split-"))
97    *     .branch((key, value) -> value == null, Branched.withFunction(s -> s.mapValues(v->"NULL"), "null")
98    *     .defaultBranch(Branched.as("non-null"));
99    *
100    * KStream<String, String> merged = branches.get("split-non-null").merge(branches.get("split-null"));
101    * }</pre>
102    *
103    * <h3>Dynamic branching</h3>
104    * There is also a case when we might need to create branches dynamically, e. g. one per enum value:
105    *
106    * <pre> {@code
107    * BranchedKStream branched = stream.split();
108    * for (RecordType recordType : RecordType.values())
109    *     branched.branch((k, v) -> v.getRecType() == recordType,
110    *         Branched.withConsumer(recordType::processRecords));
111    * }</pre>
112    *
113    * @param <K> Type of keys
114    * @param <V> Type of values
115    * @see KStream
116    */
117   public interface BranchedKStream<K, V> {
118       /**
119        * Define a branch for records that match the predicate.
120        *
121        * @param predicate A {@link Predicate} instance, against which each record will be evaluated.
122        *                  If this predicate returns {@code true} for a given record, the record will be
123        *                  routed to the current branch and will not be evaluated against the predicates
124        *                  for the remaining branches.
125        * @return {@code this} to facilitate method chaining
126        */
127       BranchedKStream<K, V> branch(Predicate<? super K, ? super V> predicate);
128   
129       /**
130        * Define a branch for records that match the predicate.
131        *
132        * @param predicate A {@link Predicate} instance, against which each record will be evaluated.
133        *                  If this predicate returns {@code true} for a given record, the record will be
134        *                  routed to the current branch and will not be evaluated against the predicates
135        *                  for the remaining branches.
136        * @param branched  A {@link Branched} parameter, that allows to define a branch name, an in-place
137        *                  branch consumer or branch mapper (see <a href="#examples">code examples</a>
138        *                  for {@link BranchedKStream})
139        * @return {@code this} to facilitate method chaining
140        */
141       BranchedKStream<K, V> branch(Predicate<? super K, ? super V> predicate, Branched<K, V> branched);
142   
143       /**
144        * Finalize the construction of branches and defines the default branch for the messages not intercepted
145        * by other branches. Calling {@code defaultBranch} or {@link #noDefaultBranch()} is optional.
146        *
147        * @return {@link Map} of named branches. For rules of forming the resulting map, see {@code BranchedKStream}
148        * <a href="#maprules">description</a>.
149        */
150       Map<String, KStream<K, V>> defaultBranch();
151   
152       /**
153        * Finalize the construction of branches and defines the default branch for the messages not intercepted
154        * by other branches. Calling {@code defaultBranch} or {@link #noDefaultBranch()} is optional.
155        *
156        * @param branched A {@link Branched} parameter, that allows to define a branch name, an in-place
157        *                 branch consumer or branch mapper (see <a href="#examples">code examples</a>
158        *                 for {@link BranchedKStream})
159        * @return {@link Map} of named branches. For rules of forming the resulting map, see {@link BranchedKStream}
160        * <a href="#maprules">description</a>.
161        */
162       Map<String, KStream<K, V>> defaultBranch(Branched<K, V> branched);
163   
164       /**
165        * Finalize the construction of branches without forming a default branch.  Calling {@code #noDefaultBranch()}
166        * or {@link #defaultBranch()} is optional.
167        *
168        * @return {@link Map} of named branches. For rules of forming the resulting map, see {@link BranchedKStream}
169        * <a href="#maprules">description</a>.
170        */
171       Map<String, KStream<K, V>> noDefaultBranch();
172   }
