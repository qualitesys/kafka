30   /**
31    * A session based window specification used for aggregating events into sessions.
32    * <p>
33    * Sessions represent a period of activity separated by a defined gap of inactivity.
34    * Any events processed that fall within the inactivity gap of any existing sessions are merged into the existing sessions.
35    * If the event falls outside of the session gap then a new session will be created.
36    * <p>
37    * For example, if we have a session gap of 5 and the following data arrives:
38    * <pre>
39    * +--------------------------------------+
40    * |    key    |    value    |    time    |
41    * +-----------+-------------+------------+
42    * |    A      |     1       |     10     |
43    * +-----------+-------------+------------+
44    * |    A      |     2       |     12     |
45    * +-----------+-------------+------------+
46    * |    A      |     3       |     20     |
47    * +-----------+-------------+------------+
48    * </pre>
49    * We'd have 2 sessions for key A.
50    * One starting from time 10 and ending at time 12 and another starting and ending at time 20.
51    * The length of the session is driven by the timestamps of the data within the session.
52    * Thus, session windows are no fixed-size windows (c.f. {@link TimeWindows} and {@link JoinWindows}).
53    * <p>
54    * If we then received another record:
55    * <pre>
56    * +--------------------------------------+
57    * |    key    |    value    |    time    |
58    * +-----------+-------------+------------+
59    * |    A      |     4       |     16     |
60    * +-----------+-------------+------------+
61    * </pre>
62    * The previous 2 sessions would be merged into a single session with start time 10 and end time 20.
63    * The aggregate value for this session would be the result of aggregating all 4 values.
64    * <p>
65    * For time semantics, see {@link TimestampExtractor}.
66    *
67    * @see TimeWindows
68    * @see UnlimitedWindows
69    * @see JoinWindows
70    * @see KGroupedStream#windowedBy(SessionWindows)
71    * @see TimestampExtractor
72    */
73   public final class SessionWindows {
74   
75       private final long gapMs;
76   
77       private final long graceMs;
78   
79       // flag to check if the grace is already set via ofInactivityGapAndGrace or ofInactivityGapWithNoGrace
80       private final boolean hasSetGrace;
81   
82       private SessionWindows(final long gapMs, final long graceMs, final boolean hasSetGrace) {
83           this.gapMs = gapMs;
84           this.graceMs = graceMs;
85           this.hasSetGrace = hasSetGrace;
86   
87           if (gapMs <= 0) {
88               throw new IllegalArgumentException("Gap time cannot be zero or negative.");
89           }
90   
91           if (graceMs < 0) {
92               throw new IllegalArgumentException("Grace period must not be negative.");
93           }
94       }
95   
96       /**
97        * Creates a new window specification with the specified inactivity gap.
98        * <p>
99        * Note that new events may change the boundaries of session windows, so aggressive
100        * close times can lead to surprising results in which an out-of-order event is rejected and then
101        * a subsequent event moves the window boundary forward.
102        * <p>
103        * CAUTION: Using this method implicitly sets the grace period to zero, which means that any out-of-order
104        * records arriving after the window ends are considered late and will be dropped.
105        *
106        * @param inactivityGap the gap of inactivity between sessions
107        * @return a window definition with the window size and no grace period. Note that this means out-of-order records arriving after the window end will be dropped
108        * @throws IllegalArgumentException if {@code inactivityGap} is zero or negative or can't be represented as {@code long milliseconds}
109        */
110       public static SessionWindows ofInactivityGapWithNoGrace(final Duration inactivityGap) {
111           return ofInactivityGapAndGrace(inactivityGap, ofMillis(NO_GRACE_PERIOD));
112       }
113   
114       /**
115        * Creates a new window specification with the specified inactivity gap.
116        * <p>
117        * Note that new events may change the boundaries of session windows, so aggressive
118        * close times can lead to surprising results in which an out-of-order event is rejected and then
119        * a subsequent event moves the window boundary forward.
120        * <p>
121        * Using this method explicitly sets the grace period to the duration specified by {@code afterWindowEnd}, which
122        * means that only out-of-order records arriving more than the grace period after the window end will be dropped.
123        * The window close, after which any incoming records are considered late and will be rejected, is defined as
124        * {@code windowEnd + afterWindowEnd}
125        *
126        * @param inactivityGap the gap of inactivity between sessions
127        * @param afterWindowEnd The grace period to admit out-of-order events to a window.
128        * @return A SessionWindows object with the specified inactivity gap and grace period
129        * @throws IllegalArgumentException if {@code inactivityGap} is zero or negative or can't be represented as {@code long milliseconds}
130        *                                  if {@code afterWindowEnd} is negative or can't be represented as {@code long milliseconds}
131        */
132       public static SessionWindows ofInactivityGapAndGrace(final Duration inactivityGap, final Duration afterWindowEnd) {
133           final String inactivityGapMsgPrefix = prepareMillisCheckFailMsgPrefix(inactivityGap, "inactivityGap");
134           final long inactivityGapMs = validateMillisecondDuration(inactivityGap, inactivityGapMsgPrefix);
135   
136           final String afterWindowEndMsgPrefix = prepareMillisCheckFailMsgPrefix(afterWindowEnd, "afterWindowEnd");
137           final long afterWindowEndMs = validateMillisecondDuration(afterWindowEnd, afterWindowEndMsgPrefix);
138   
139           return new SessionWindows(inactivityGapMs, afterWindowEndMs, true);
140       }
141   
142       /**
143        * Create a new window specification with the specified inactivity gap.
144        *
145        * @param inactivityGap the gap of inactivity between sessions
146        * @return a new window specification without specifying a grace period (default to 24 hours minus {@code inactivityGap})
147        * @throws IllegalArgumentException if {@code inactivityGap} is zero or negative or can't be represented as {@code long milliseconds}
148        * @deprecated since 3.0. Use {@link #ofInactivityGapWithNoGrace(Duration)} instead
149        */
150       @Deprecated
151       public static SessionWindows with(final Duration inactivityGap) {
152           final String msgPrefix = prepareMillisCheckFailMsgPrefix(inactivityGap, "inactivityGap");
153           final long inactivityGapMs = validateMillisecondDuration(inactivityGap, msgPrefix);
154   
155           return new SessionWindows(inactivityGapMs, Math.max(DEPRECATED_DEFAULT_24_HR_GRACE_PERIOD - inactivityGapMs, 0), false);
156       }
157   
158       /**
159        * Reject out-of-order events that arrive more than {@code afterWindowEnd}
160        * after the end of its window.
161        * <p>
162        * Note that new events may change the boundaries of session windows, so aggressive
163        * close times can lead to surprising results in which an out-of-order event is rejected and then
164        * a subsequent event moves the window boundary forward.
165        *
166        * @param afterWindowEnd The grace period to admit out-of-order events to a window.
167        * @return this updated builder
168        * @throws IllegalArgumentException if the {@code afterWindowEnd} is negative or can't be represented as {@code long milliseconds}
169        * @throws IllegalStateException if {@link #grace(Duration)} is called after {@link #ofInactivityGapAndGrace(Duration, Duration)} or {@link #ofInactivityGapWithNoGrace(Duration)}
170        * @deprecated since 3.0. Use {@link #ofInactivityGapAndGrace(Duration, Duration)} instead
171        */
172       @Deprecated
173       public SessionWindows grace(final Duration afterWindowEnd) throws IllegalArgumentException {
174           if (this.hasSetGrace) {
175               throw new IllegalStateException(
176                   "Cannot call grace() after setting grace value via ofInactivityGapAndGrace or ofInactivityGapWithNoGrace.");
177           }
178   
179           final String msgPrefix = prepareMillisCheckFailMsgPrefix(afterWindowEnd, "afterWindowEnd");
180           final long afterWindowEndMs = validateMillisecondDuration(afterWindowEnd, msgPrefix);
181   
182           return new SessionWindows(gapMs, afterWindowEndMs, false);
183       }
184   
185       public long gracePeriodMs() {
186           return graceMs;
187       }
188   
189       /**
190        * Return the specified gap for the session windows in milliseconds.
191        *
192        * @return the inactivity gap of the specified windows
193        */
194       public long inactivityGap() {
195           return gapMs;
196       }
197   
198       @Override
199       public boolean equals(final Object o) {
200           if (this == o) {
201               return true;
202           }
203           if (o == null || getClass() != o.getClass()) {
204               return false;
205           }
206           final SessionWindows that = (SessionWindows) o;
207           return gapMs == that.gapMs &&
208               graceMs == that.graceMs;
209       }
210   
211       @Override
212       public int hashCode() {
213           return Objects.hash(gapMs, graceMs);
214       }
215   
216       @Override
217       public String toString() {
218           return "SessionWindows{" +
219               "gapMs=" + gapMs +
220               ", graceMs=" + graceMs +
221               '}';
222       }
223   }
