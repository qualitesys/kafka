157       public void put(final Bytes key, final byte[] value, final long windowStartTimestamp) {
158           removeExpiredSegments();
159           observedStreamTime = Math.max(observedStreamTime, windowStartTimestamp);
160   
161           if (windowStartTimestamp <= observedStreamTime - retentionPeriod) {
162               expiredRecordSensor.record(1.0d, ProcessorContextUtils.currentSystemTime(context));
163               LOG.warn("Skipping record for expired segment.");
164           } else {
165               if (value != null) {
166                   maybeUpdateSeqnumForDups();
167                   final Bytes keyBytes = retainDuplicates ? wrapForDups(key, seqnum) : key;
168                   segmentMap.computeIfAbsent(windowStartTimestamp, t -> new ConcurrentSkipListMap<>());
169                   segmentMap.get(windowStartTimestamp).put(keyBytes, value);
170               } else if (!retainDuplicates) {
171                   // Skip if value is null and duplicates are allowed since this delete is a no-op
172                   segmentMap.computeIfPresent(windowStartTimestamp, (t, kvMap) -> {
173                       kvMap.remove(key);
174                       if (kvMap.isEmpty()) {
175                           segmentMap.remove(windowStartTimestamp);
176                       }
177                       return kvMap;
178                   });
179               }
180           }

183       }

478       private static abstract class InMemoryWindowStoreIteratorWrapper {
479

481           private KeyValue<Bytes, byte[]> next;
482           private long currentTime;
483

485           private final Bytes keyFrom;
486           private final Bytes keyTo;
487           private final Iterator<Map.Entry<Long, ConcurrentNavigableMap<Bytes, byte[]>>> segmentIterator;
488           private final ClosingCallback callback;
489           private final boolean retainDuplicates;
490           private final boolean forward;
491   
492           InMemoryWindowStoreIteratorWrapper(final Bytes keyFrom,
493                                              final Bytes keyTo,
494                                              final Iterator<Map.Entry<Long, ConcurrentNavigableMap<Bytes, byte[]>>> segmentIterator,
495                                              final ClosingCallback callback,
496                                              final boolean retainDuplicates,
497                                              final boolean forward) {
498               this.keyFrom = keyFrom;
499               this.keyTo = keyTo;
500               allKeys = (keyFrom == null) && (keyTo == null);
501               this.retainDuplicates = retainDuplicates;
502               this.forward = forward;
503   
504               this.segmentIterator = segmentIterator;
505               this.callback = callback;
506               recordIterator = segmentIterator == null ? null : setRecordIterator();
507           }

509           public boolean hasNext() {
510               if (next != null) {
511                   return true;
512               }
513               if (recordIterator == null || (!recordIterator.hasNext() && !segmentIterator.hasNext())) {
514                   return false;
515               }
516   
517               next = getNext();
518               if (next == null) {
519                   return false;
520               }
521   
522               if (allKeys || !retainDuplicates) {
523                   return true;
524               }
525   
526               final Bytes key = getKey(next.key);
527               if (isKeyWithinRange(key)) {
528                   return true;
529               } else {
530                   next = null;
531                   return hasNext();

535           private boolean isKeyWithinRange(final Bytes key) {
536               // split all cases for readability and avoid BooleanExpressionComplexity checkstyle warning
537               if (keyFrom == null && keyTo == null) {
538                   // fetch all
539                   return true;
540               } else if (keyFrom == null) {
541                   // start from the beginning
542                   return key.compareTo(getKey(keyTo)) <= 0;
543               } else if (keyTo == null) {
544                   // end to the last
545                   return key.compareTo(getKey(keyFrom)) >= 0;
546               } else {
547                   // key is within the range
548                   return key.compareTo(getKey(keyFrom)) >= 0 && key.compareTo(getKey(keyTo)) <= 0;
549               }
550           }

553               next = null;

558           // getNext is only called when either recordIterator or segmentIterator has a next
559           // Note this does not guarantee a next record exists as the next segments may not contain any keys in range
560           protected KeyValue<Bytes, byte[]> getNext() {
561               while (!recordIterator.hasNext()) {
562                   recordIterator = setRecordIterator();
563                   if (recordIterator == null) {
564                       return null;
565                   }
566               }
567               final Map.Entry<Bytes, byte[]> nextRecord = recordIterator.next();
568               return new KeyValue<>(nextRecord.getKey(), nextRecord.getValue());
569           }
570   
571           // Resets recordIterator to point to the next segment and returns null if there are no more segments
572           // Note it may not actually point to anything if no keys in range exist in the next segment
573           Iterator<Map.Entry<Bytes, byte[]>> setRecordIterator() {
574               if (!segmentIterator.hasNext()) {
575                   return null;
576               }
577   
578               final Map.Entry<Long, ConcurrentNavigableMap<Bytes, byte[]>> currentSegment = segmentIterator.next();
579               currentTime = currentSegment.getKey();
580   
581               final ConcurrentNavigableMap<Bytes, byte[]> subMap;
582               if (allKeys) { // keyFrom == null && keyTo == null
583                   subMap = currentSegment.getValue();
584               } else if (keyFrom == null) {
585                   subMap = currentSegment.getValue().headMap(keyTo, true);
586               } else if (keyTo == null) {
587                   subMap = currentSegment.getValue().tailMap(keyFrom, true);
588               } else {
589                   subMap = currentSegment.getValue().subMap(keyFrom, true, keyTo, true);
590               }
591   
592               if (forward) {
593                   return subMap.entrySet().iterator();
594               } else {
595                   return subMap.descendingMap().entrySet().iterator();
596               }
597           }

600               return currentTime;
601           }
