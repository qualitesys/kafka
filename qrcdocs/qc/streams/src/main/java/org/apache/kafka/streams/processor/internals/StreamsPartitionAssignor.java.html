293       /*
294        * This assigns tasks to consumer clients in the following steps.
295        *
296        * 0. decode the subscriptions to assemble the metadata for each client and check for version probing
297        *
298        * 1. check all repartition source topics and use internal topic manager to make sure
299        *    they have been created with the right number of partitions. Also verify and/or create
300        *    any changelog topics with the correct number of partitions.
301        *
302        * 2. use the partition grouper to generate tasks along with their assigned partitions, then use
303        *    the configured TaskAssignor to construct the mapping of tasks to clients.
304        *
305        * 3. construct the global mapping of host to partitions to enable query routing.
306        *
307        * 4. within each client, assign tasks to consumer clients.
308        */
309       @Override
310       public GroupAssignment assign(final Cluster metadata, final GroupSubscription groupSubscription) {
311           final Map<String, Subscription> subscriptions = groupSubscription.groupSubscription();
312   
313           // ---------------- Step Zero ---------------- //
314   
315           // construct the client metadata from the decoded subscription info
316   
317           final Map<UUID, ClientMetadata> clientMetadataMap = new HashMap<>();
318           final Set<TopicPartition> allOwnedPartitions = new HashSet<>();
319   
320           int minReceivedMetadataVersion = LATEST_SUPPORTED_VERSION;
321           int minSupportedMetadataVersion = LATEST_SUPPORTED_VERSION;
322   
323           boolean shutdownRequested = false;
324           boolean assignmentErrorFound = false;
325           int futureMetadataVersion = UNKNOWN;
326           for (final Map.Entry<String, Subscription> entry : subscriptions.entrySet()) {
327               final String consumerId = entry.getKey();
328               final Subscription subscription = entry.getValue();
329               final SubscriptionInfo info = SubscriptionInfo.decode(subscription.userData());
330               final int usedVersion = info.version();
331               if (info.errorCode() == AssignorError.SHUTDOWN_REQUESTED.code()) {
332                   shutdownRequested = true;
333               }
334   
335               minReceivedMetadataVersion = updateMinReceivedVersion(usedVersion, minReceivedMetadataVersion);
336               minSupportedMetadataVersion = updateMinSupportedVersion(info.latestSupportedVersion(), minSupportedMetadataVersion);
337   
338               final UUID processId;
339               if (usedVersion > LATEST_SUPPORTED_VERSION) {
340                   futureMetadataVersion = usedVersion;
341                   processId = FUTURE_ID;
342                   if (!clientMetadataMap.containsKey(FUTURE_ID)) {
343                       clientMetadataMap.put(FUTURE_ID, new ClientMetadata(null, Collections.emptyMap()));
344                   }
345               } else {
346                   processId = info.processId();
347               }
348   
349               ClientMetadata clientMetadata = clientMetadataMap.get(processId);
350   
351               // create the new client metadata if necessary
352               if (clientMetadata == null) {
353                   clientMetadata = new ClientMetadata(info.userEndPoint(), info.clientTags());
354                   clientMetadataMap.put(info.processId(), clientMetadata);
355               }
356   
357               // add the consumer and any info in its subscription to the client
358               clientMetadata.addConsumer(consumerId, subscription.ownedPartitions());
359               final int prevSize = allOwnedPartitions.size();
360               allOwnedPartitions.addAll(subscription.ownedPartitions());
361               if (allOwnedPartitions.size() < prevSize + subscription.ownedPartitions().size()) {
362                   assignmentErrorFound = true;
363               }
364               clientMetadata.addPreviousTasksAndOffsetSums(consumerId, info.taskOffsetSums());
365           }
366   
367           if (assignmentErrorFound) {
368               log.warn("The previous assignment contains a partition more than once. " +
369                   "\t Mapping: {}", subscriptions);
370           }
371   
372           try {
373               final boolean versionProbing =
374                   checkMetadataVersions(minReceivedMetadataVersion, minSupportedMetadataVersion, futureMetadataVersion);
375   
376               log.debug("Constructed client metadata {} from the member subscriptions.", clientMetadataMap);
377   
378               // ---------------- Step One ---------------- //
379   
380               if (shutdownRequested) {
381                   return new GroupAssignment(errorAssignment(clientMetadataMap, AssignorError.SHUTDOWN_REQUESTED.code()));
382               }
383   
384               // parse the topology to determine the repartition source topics,
385               // making sure they are created with the number of partitions as
386               // the maximum of the depending sub-topologies source topics' number of partitions
387               final RepartitionTopics repartitionTopics = prepareRepartitionTopics(metadata);
388               final Map<TopicPartition, PartitionInfo> allRepartitionTopicPartitions = repartitionTopics.topicPartitionsInfo();
389   
390               final Cluster fullMetadata = metadata.withPartitions(allRepartitionTopicPartitions);
391               log.debug("Created repartition topics {} from the parsed topology.", allRepartitionTopicPartitions.values());
392   
393               // ---------------- Step Two ---------------- //
394   
395               // construct the assignment of tasks to clients
396   
397               final Map<Subtopology, TopicsInfo> topicGroups =
398                   taskManager.topologyMetadata().subtopologyTopicsInfoMapExcluding(repartitionTopics.topologiesWithMissingInputTopics());
399   
400               final Set<String> allSourceTopics = new HashSet<>();
401               final Map<Subtopology, Set<String>> sourceTopicsByGroup = new HashMap<>();
402               for (final Map.Entry<Subtopology, TopicsInfo> entry : topicGroups.entrySet()) {
403                   allSourceTopics.addAll(entry.getValue().sourceTopics);
404                   sourceTopicsByGroup.put(entry.getKey(), entry.getValue().sourceTopics);
405               }
406   
407               // get the tasks as partition groups from the partition grouper
408               final Map<TaskId, Set<TopicPartition>> partitionsForTask =
409                   partitionGrouper.partitionGroups(sourceTopicsByGroup, fullMetadata);
410   
411               final Set<TaskId> statefulTasks = new HashSet<>();
412   
413               final boolean probingRebalanceNeeded = assignTasksToClients(fullMetadata, allSourceTopics, topicGroups, clientMetadataMap, partitionsForTask, statefulTasks);
414   
415               // ---------------- Step Three ---------------- //
416   
417               // construct the global partition assignment per host map
418   
419               final Map<HostInfo, Set<TopicPartition>> partitionsByHost = new HashMap<>();
420               final Map<HostInfo, Set<TopicPartition>> standbyPartitionsByHost = new HashMap<>();
421               if (minReceivedMetadataVersion >= 2) {
422                   populatePartitionsByHostMaps(partitionsByHost, standbyPartitionsByHost, partitionsForTask, clientMetadataMap);
423               }
424               streamsMetadataState.onChange(partitionsByHost, standbyPartitionsByHost, fullMetadata);
425   
426               // ---------------- Step Four ---------------- //
427   
428               // compute the assignment of tasks to threads within each client and build the final group assignment
429   
430               final Map<String, Assignment> assignment = computeNewAssignment(
431                   statefulTasks,
432                   clientMetadataMap,
433                   partitionsForTask,
434                   partitionsByHost,
435                   standbyPartitionsByHost,
436                   allOwnedPartitions,
437                   minReceivedMetadataVersion,
438                   minSupportedMetadataVersion,
439                   versionProbing,
440                   probingRebalanceNeeded
441               );
442   
443               return new GroupAssignment(assignment);
444           } catch (final MissingSourceTopicException e) {
