35   import static org.apache.kafka.streams.processor.internals.Task.State.CLOSED;
36   import static org.apache.kafka.streams.processor.internals.Task.State.CREATED;
37   
38   public abstract class AbstractTask implements Task {
39       private final static long NO_DEADLINE = -1L;
40   
41       private Task.State state = CREATED;
42       private long deadlineMs = NO_DEADLINE;
43   
44       protected Set<TopicPartition> inputPartitions;
45       protected final Logger log;
46       protected final LogContext logContext;
47       protected final String logPrefix;
48   
49       /**
50        * If the checkpoint has not been loaded from the file yet (null), then we should not overwrite the checkpoint;
51        * If the checkpoint has been loaded from the file and has never been re-written (empty map), then we should re-write the checkpoint;
52        * If the checkpoint has been loaded from the file but has not been updated since, then we do not need to checkpoint;
53        * If the checkpoint has been loaded from the file and has been updated since, then we could overwrite the checkpoint;
54        */
55       protected Map<TopicPartition, Long> offsetSnapshotSinceLastFlush = null;
56   
57       protected final TaskId id;
58       protected final ProcessorTopology topology;
59       protected final StateDirectory stateDirectory;
60       protected final ProcessorStateManager stateMgr;
61       private final long taskTimeoutMs;
62   
63       AbstractTask(final TaskId id,
64                    final ProcessorTopology topology,
65                    final StateDirectory stateDirectory,
66                    final ProcessorStateManager stateMgr,
67                    final Set<TopicPartition> inputPartitions,
68                    final long taskTimeoutMs,
69                    final String taskType,
70                    final Class<? extends AbstractTask> clazz) {
71           this.id = id;
72           this.stateMgr = stateMgr;
73           this.topology = topology;
74           this.inputPartitions = inputPartitions;
75           this.stateDirectory = stateDirectory;
76           this.taskTimeoutMs = taskTimeoutMs;
77   
78           final String threadIdPrefix = String.format("stream-thread [%s] ", Thread.currentThread().getName());
79           logPrefix = threadIdPrefix + String.format("%s [%s] ", taskType, id);
80           logContext = new LogContext(logPrefix);
81           log = logContext.logger(clazz);
82       }
83   
84       /**
85        * The following exceptions maybe thrown from the state manager flushing call
86        *
87        * @throws TaskMigratedException recoverable error sending changelog records that would cause the task to be removed
88        * @throws StreamsException fatal error when flushing the state store, for example sending changelog records failed
89        *                          or flushing state store get IO errors; such error should cause the thread to die
90        */
91       protected void maybeWriteCheckpoint(final boolean enforceCheckpoint) {
92           final Map<TopicPartition, Long> offsetSnapshot = stateMgr.changelogOffsets();
93           if (StateManagerUtil.checkpointNeeded(enforceCheckpoint, offsetSnapshotSinceLastFlush, offsetSnapshot)) {
94               // the state's current offset would be used to checkpoint
95               stateMgr.flush();
96               stateMgr.checkpoint();
97               offsetSnapshotSinceLastFlush = new HashMap<>(offsetSnapshot);
98           }
99       }
100

102       @Override
103       public TaskId id() {
104           return id;
105       }
106   
107       @Override
108       public Set<TopicPartition> inputPartitions() {
109           return inputPartitions;
110       }
111   
112       @Override
113       public Collection<TopicPartition> changelogPartitions() {
114           return stateMgr.changelogPartitions();
115       }
116   
117       @Override
118       public void markChangelogAsCorrupted(final Collection<TopicPartition> partitions) {
119           stateMgr.markChangelogAsCorrupted(partitions);
120       }
121   
122       @Override
123       public StateStore getStore(final String name) {
124           return stateMgr.getStore(name);
125       }
126   
127       @Override
128       public final Task.State state() {
129           return state;
130       }
131   
132       @Override
133       public void revive() {
134           if (state == CLOSED) {
135               clearTaskTimeout();
136               transitionTo(CREATED);
137           } else {
138               throw new IllegalStateException("Illegal state " + state() + " while reviving task " + id);
139           }
140       }
141   
142       final void transitionTo(final Task.State newState) {
143           final State oldState = state();
144   
145           if (oldState.isValidTransition(newState)) {
146               state = newState;
147           } else {
148               throw new IllegalStateException("Invalid transition from " + oldState + " to " + newState);
149           }
150       }
151   
152       @Override
153       public void updateInputPartitions(final Set<TopicPartition> topicPartitions, final Map<String, List<String>> allTopologyNodesToSourceTopics) {
154           this.inputPartitions = topicPartitions;
155           topology.updateSourceTopics(allTopologyNodesToSourceTopics);
156       }
157   
158       @Override
159       public void maybeInitTaskTimeoutOrThrow(final long currentWallClockMs,
160                                               final Exception cause) {
161           if (deadlineMs == NO_DEADLINE) {
162               deadlineMs = currentWallClockMs + taskTimeoutMs;
163           } else if (currentWallClockMs > deadlineMs) {
164               final String errorMessage = String.format(
165                   "Task %s did not make progress within %d ms. Adjust `%s` if needed.",
166                   id,
167                   currentWallClockMs - deadlineMs + taskTimeoutMs,
168                   StreamsConfig.TASK_TIMEOUT_MS_CONFIG
169               );
170   
171               if (cause != null) {
172                   throw new StreamsException(new TimeoutException(errorMessage, cause), id);
173               } else {
174                   throw new StreamsException(new TimeoutException(errorMessage), id);
175               }
176           }
177   
178           if (cause != null) {
179               log.debug(
180                   String.format(
181                       "Task did not make progress. Remaining time to deadline %d; retrying.",
182                       deadlineMs - currentWallClockMs
183                   ),
184                   cause
185               );
186           } else {
187               log.debug(
188                   "Task did not make progress. Remaining time to deadline {}; retrying.",
189                   deadlineMs - currentWallClockMs
190               );
191           }
192   
193       }
194   
195       @Override
196       public void clearTaskTimeout() {
197           if (deadlineMs != NO_DEADLINE) {
198               log.debug("Clearing task timeout.");
199               deadlineMs = NO_DEADLINE;
200           }
201       }
202   }
