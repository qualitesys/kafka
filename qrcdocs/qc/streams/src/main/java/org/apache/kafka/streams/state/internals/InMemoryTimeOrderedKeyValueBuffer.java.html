314       private void restoreBatch(final Collection<ConsumerRecord<byte[], byte[]>> batch) {
315           for (final ConsumerRecord<byte[], byte[]> record : batch) {
316               if (record.partition() != partition) {
317                   throw new IllegalStateException(
318                       String.format(
319                           "record partition [%d] is being restored by the wrong suppress partition [%d]",
320                           record.partition(),
321                           partition
322                       )
323                   );
324               }
325               final Bytes key = Bytes.wrap(record.key());
326               if (record.value() == null) {
327                   // This was a tombstone. Delete the record.
328                   final BufferKey bufferKey = index.remove(key);
329                   if (bufferKey != null) {
330                       final BufferValue removed = sortedMap.remove(bufferKey);
331                       if (removed != null) {
332                           memBufferSize -= computeRecordSize(bufferKey.key(), removed);
333                       }
334                       if (bufferKey.time() == minTimestamp) {
335                           minTimestamp = sortedMap.isEmpty() ? Long.MAX_VALUE : sortedMap.firstKey().time();
336                       }
337                   }
338               } else {
339                   final Header versionHeader = record.headers().lastHeader("v");
340                   if (versionHeader == null) {
341                       // Version 0:
342                       // value:
343                       //  - buffer time
344                       //  - old value
345                       //  - new value
346                       final byte[] previousBufferedValue = index.containsKey(key)
347                           ? internalPriorValueForBuffered(key)
348                           : null;
349                       final DeserializationResult deserializationResult = deserializeV0(record, key, previousBufferedValue);
350                       cleanPut(deserializationResult.time(), deserializationResult.key(), deserializationResult.bufferValue());
351                   } else if (Arrays.equals(versionHeader.value(), V_3_CHANGELOG_HEADER_VALUE)) {
352                       // Version 3:
353                       // value:
354                       //  - record context
355                       //  - prior value
356                       //  - old value
357                       //  - new value
358                       //  - buffer time
359                       final DeserializationResult deserializationResult = deserializeV3(record, key);
360                       cleanPut(deserializationResult.time(), deserializationResult.key(), deserializationResult.bufferValue());
361   
362                   } else if (Arrays.equals(versionHeader.value(), V_2_CHANGELOG_HEADER_VALUE)) {
363                       // Version 2:
364                       // value:
365                       //  - record context
366                       //  - old value
367                       //  - new value
368                       //  - prior value
369                       //  - buffer time
370                       // NOTE: 2.4.0, 2.4.1, and 2.5.0 actually encode Version 3 formatted data,
371                       // but still set the Version 2 flag, so to deserialize, we have to duck type.
372                       final DeserializationResult deserializationResult = duckTypeV2(record, key);
373                       cleanPut(deserializationResult.time(), deserializationResult.key(), deserializationResult.bufferValue());
374                   } else if (Arrays.equals(versionHeader.value(), V_1_CHANGELOG_HEADER_VALUE)) {
375                       // Version 1:
376                       // value:
377                       //  - buffer time
378                       //  - record context
379                       //  - old value
380                       //  - new value
381                       final byte[] previousBufferedValue = index.containsKey(key)
382                           ? internalPriorValueForBuffered(key)
383                           : null;
384                       final DeserializationResult deserializationResult = deserializeV1(record, key, previousBufferedValue);
385                       cleanPut(deserializationResult.time(), deserializationResult.key(), deserializationResult.bufferValue());
386                   } else {
387                       throw new IllegalArgumentException("Restoring apparently invalid changelog record: " + record);
388                   }
389               }
390           }
391           updateBufferMetrics();
392       }

395       @Override
396       public void evictWhile(final Supplier<Boolean> predicate,
397                              final Consumer<Eviction<K, V>> callback) {
398           final Iterator<Map.Entry<BufferKey, BufferValue>> delegate = sortedMap.entrySet().iterator();
399           int evictions = 0;
400   
401           if (predicate.get()) {
402               Map.Entry<BufferKey, BufferValue> next = null;
403               if (delegate.hasNext()) {
404                   next = delegate.next();
405               }
406   
407               // predicate being true means we read one record, call the callback, and then remove it
408               while (next != null && predicate.get()) {
409                   if (next.getKey().time() != minTimestamp) {
410                       throw new IllegalStateException(
411                           "minTimestamp [" + minTimestamp + "] did not match the actual min timestamp [" +
412                               next.getKey().time() + "]"
413                       );
414                   }
415                   final K key = keySerde.deserializer().deserialize(changelogTopic, next.getKey().key().get());
416                   final BufferValue bufferValue = next.getValue();
417                   final Change<V> value = valueSerde.deserializeParts(
418                       changelogTopic,
419                       new Change<>(bufferValue.newValue(), bufferValue.oldValue())
420                   );
421                   callback.accept(new Eviction<>(key, value, bufferValue.context()));
422   
423                   delegate.remove();
424                   index.remove(next.getKey().key());
425   
426                   dirtyKeys.add(next.getKey().key());
427   
428                   memBufferSize -= computeRecordSize(next.getKey().key(), bufferValue);
429   
430                   // peek at the next record so we can update the minTimestamp
431                   if (delegate.hasNext()) {
432                       next = delegate.next();
433                       minTimestamp = next == null ? Long.MAX_VALUE : next.getKey().time();
434                   } else {
435                       next = null;
436                       minTimestamp = Long.MAX_VALUE;
437                   }
438   
439                   evictions++;
440               }
441           }
442           if (evictions > 0) {
443               updateBufferMetrics();
444           }
445       }

447       @Override
448       public Maybe<ValueAndTimestamp<V>> priorValueForBuffered(final K key) {
449           final Bytes serializedKey = Bytes.wrap(keySerde.serializer().serialize(changelogTopic, key));
450           if (index.containsKey(serializedKey)) {
451               final byte[] serializedValue = internalPriorValueForBuffered(serializedKey);
452   
453               final V deserializedValue = valueSerde.innerSerde().deserializer().deserialize(
454                   changelogTopic,
455                   serializedValue
456               );
457   
458               // it's unfortunately not possible to know this, unless we materialize the suppressed result, since our only
459               // knowledge of the prior value is what the upstream processor sends us as the "old value" when we first
460               // buffer something.
461               return Maybe.defined(ValueAndTimestamp.make(deserializedValue, RecordQueue.UNKNOWN));
462           } else {
463               return Maybe.undefined();
464           }
465       }

467       private byte[] internalPriorValueForBuffered(final Bytes key) {
468           final BufferKey bufferKey = index.get(key);
469           if (bufferKey == null) {
470               throw new NoSuchElementException("Key [" + key + "] is not in the buffer.");
471           } else {
472               final BufferValue bufferValue = sortedMap.get(bufferKey);
473               return bufferValue.priorValue();
474           }

477       @Override
478       public void put(final long time,
479                       final Record<K, Change<V>> record,
480                       final ProcessorRecordContext recordContext) {
481           requireNonNull(record.value(), "value cannot be null");
482           requireNonNull(recordContext, "recordContext cannot be null");
483   
484           final Bytes serializedKey = Bytes.wrap(keySerde.serializer().serialize(changelogTopic, record.key()));
485           final Change<byte[]> serialChange = valueSerde.serializeParts(changelogTopic, record.value());
486   
487           final BufferValue buffered = getBuffered(serializedKey);
488           final byte[] serializedPriorValue;
489           if (buffered == null) {
490               serializedPriorValue = serialChange.oldValue;
491           } else {
492               serializedPriorValue = buffered.priorValue();
493           }
494   
495           cleanPut(
496               time,
497               serializedKey,
498               new BufferValue(serializedPriorValue, serialChange.oldValue, serialChange.newValue, recordContext)
499           );
500           dirtyKeys.add(serializedKey);
501           updateBufferMetrics();
502       }
503   
504       private BufferValue getBuffered(final Bytes key) {
505           final BufferKey bufferKey = index.get(key);
506           return bufferKey == null ? null : sortedMap.get(bufferKey);
507       }
508   
509       private void cleanPut(final long time, final Bytes key, final BufferValue value) {
510           // non-resetting semantics:
511           // if there was a previous version of the same record,
512           // then insert the new record in the same place in the priority queue
513   
514           final BufferKey previousKey = index.get(key);
515           if (previousKey == null) {
516               final BufferKey nextKey = new BufferKey(time, key);
517               index.put(key, nextKey);
518               sortedMap.put(nextKey, value);
519               minTimestamp = Math.min(minTimestamp, time);
520               memBufferSize += computeRecordSize(key, value);
521           } else {
522               final BufferValue removedValue = sortedMap.put(previousKey, value);
523               memBufferSize =
524                   memBufferSize
525                       + computeRecordSize(key, value)
526                       - (removedValue == null ? 0 : computeRecordSize(key, removedValue));
527           }
528       }

545       private static long computeRecordSize(final Bytes key, final BufferValue value) {
546           long size = 0L;
547           size += 8; // buffer time
548           size += key.get().length;
549           if (value != null) {
550               size += value.residentMemorySizeEstimate();
551           }
552           return size;
553       }
