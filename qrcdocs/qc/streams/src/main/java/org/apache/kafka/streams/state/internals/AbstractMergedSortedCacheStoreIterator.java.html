17   package org.apache.kafka.streams.state.internals;
18   
19   import org.apache.kafka.common.utils.Bytes;
20   import org.apache.kafka.streams.KeyValue;
21   import org.apache.kafka.streams.state.KeyValueIterator;
22   
23   import java.util.NoSuchElementException;
24   
25   /**
26    * Merges two iterators. Assumes each of them is sorted by key
27    *
28    * @param <K>
29    * @param <V>
30    */
31   abstract class AbstractMergedSortedCacheStoreIterator<K, KS, V, VS> implements KeyValueIterator<K, V> {
32       private final PeekingKeyValueIterator<Bytes, LRUCacheEntry> cacheIterator;
33       private final KeyValueIterator<KS, VS> storeIterator;
34       private final boolean forward;
35   
36       AbstractMergedSortedCacheStoreIterator(final PeekingKeyValueIterator<Bytes, LRUCacheEntry> cacheIterator,
37                                              final KeyValueIterator<KS, VS> storeIterator,
38                                              final boolean forward) {
39           this.cacheIterator = cacheIterator;
40           this.storeIterator = storeIterator;
41           this.forward = forward;
42       }
43   
44       abstract int compare(final Bytes cacheKey, final KS storeKey);
45   
46       abstract K deserializeStoreKey(final KS key);
47   
48       abstract KeyValue<K, V> deserializeStorePair(final KeyValue<KS, VS> pair);
49   
50       abstract K deserializeCacheKey(final Bytes cacheKey);
51   
52       abstract V deserializeCacheValue(final LRUCacheEntry cacheEntry);
53   
54       private boolean isDeletedCacheEntry(final KeyValue<Bytes, LRUCacheEntry> nextFromCache) {
55           return nextFromCache.value.value() == null;
56       }
57   
58       @Override
59       public boolean hasNext() {
60           // skip over items deleted from cache, and corresponding store items if they have the same key
61           while (cacheIterator.hasNext() && isDeletedCacheEntry(cacheIterator.peekNext())) {
62               if (storeIterator.hasNext()) {
63                   final KS nextStoreKey = storeIterator.peekNextKey();
64                   // advance the store iterator if the key is the same as the deleted cache key
65                   if (compare(cacheIterator.peekNextKey(), nextStoreKey) == 0) {
66                       storeIterator.next();
67                   }
68               }
69               cacheIterator.next();
70           }
71   
72           return cacheIterator.hasNext() || storeIterator.hasNext();
73       }
74   
75       @Override
76       public KeyValue<K, V> next() {
77           if (!hasNext()) {
78               throw new NoSuchElementException();
79           }
80   
81           final Bytes nextCacheKey = cacheIterator.hasNext() ? cacheIterator.peekNextKey() : null;
82           final KS nextStoreKey = storeIterator.hasNext() ? storeIterator.peekNextKey() : null;
83   
84           if (nextCacheKey == null) {
85               return nextStoreValue(nextStoreKey);
86           }
87   
88           if (nextStoreKey == null) {
89               return nextCacheValue(nextCacheKey);
90           }
91   
92           final int comparison = compare(nextCacheKey, nextStoreKey);
93           return chooseNextValue(nextCacheKey, nextStoreKey, comparison);
94       }
95   
96       private KeyValue<K, V> chooseNextValue(final Bytes nextCacheKey,
97                                              final KS nextStoreKey,
98                                              final int comparison) {
99           if (forward) {
100               if (comparison > 0) {
101                   return nextStoreValue(nextStoreKey);
102               } else if (comparison < 0) {
103                   return nextCacheValue(nextCacheKey);
104               } else {
105                   // skip the same keyed element
106                   storeIterator.next();
107                   return nextCacheValue(nextCacheKey);
108               }
109           } else {
110               if (comparison < 0) {
111                   return nextStoreValue(nextStoreKey);
112               } else if (comparison > 0) {
113                   return nextCacheValue(nextCacheKey);
114               } else {
115                   // skip the same keyed element
116                   storeIterator.next();
117                   return nextCacheValue(nextCacheKey);
118               }
119           }
120       }
121   
122       private KeyValue<K, V> nextStoreValue(final KS nextStoreKey) {
123           final KeyValue<KS, VS> next = storeIterator.next();
124   
125           if (!next.key.equals(nextStoreKey)) {
126               throw new IllegalStateException("Next record key is not the peeked key value; this should not happen");
127           }
128   
129           return deserializeStorePair(next);
130       }
131   
132       private KeyValue<K, V> nextCacheValue(final Bytes nextCacheKey) {
133           final KeyValue<Bytes, LRUCacheEntry> next = cacheIterator.next();
134   
135           if (!next.key.equals(nextCacheKey)) {
136               throw new IllegalStateException("Next record key is not the peeked key value; this should not happen");
137           }
138   
139           return KeyValue.pair(deserializeCacheKey(next.key), deserializeCacheValue(next.value));
140       }
141   
142       @Override
143       public K peekNextKey() {
144           if (!hasNext()) {
145               throw new NoSuchElementException();
146           }
147   
148           final Bytes nextCacheKey = cacheIterator.hasNext() ? cacheIterator.peekNextKey() : null;
149           final KS nextStoreKey = storeIterator.hasNext() ? storeIterator.peekNextKey() : null;
150   
151           if (nextCacheKey == null) {
152               return deserializeStoreKey(nextStoreKey);
153           }
154   
155           if (nextStoreKey == null) {
156               return deserializeCacheKey(nextCacheKey);
157           }
158   
159           final int comparison = compare(nextCacheKey, nextStoreKey);
160           return chooseNextKey(nextCacheKey, nextStoreKey, comparison);
161       }
162   
163       private K chooseNextKey(final Bytes nextCacheKey,
164                               final KS nextStoreKey,
165                               final int comparison) {
166           if (forward) {
167               if (comparison > 0) {
168                   return deserializeStoreKey(nextStoreKey);
169               } else if (comparison < 0) {
170                   return deserializeCacheKey(nextCacheKey);
171               } else {
172                   // skip the same keyed element
173                   storeIterator.next();
174                   return deserializeCacheKey(nextCacheKey);
175               }
176           } else {
177               if (comparison < 0) {
178                   return deserializeStoreKey(nextStoreKey);
179               } else if (comparison > 0) {
180                   return deserializeCacheKey(nextCacheKey);
181               } else {
182                   // skip the same keyed element
183                   storeIterator.next();
184                   return deserializeCacheKey(nextCacheKey);
185               }
186           }
187       }
188   
189       @Override
190       public void close() {
191           cacheIterator.close();
192           storeIterator.close();
193       }
194   }
