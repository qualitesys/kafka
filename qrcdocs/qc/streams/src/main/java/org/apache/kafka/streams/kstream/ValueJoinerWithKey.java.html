20   /**
21    * The {@code ValueJoinerWithKey} interface for joining two values into a new value of arbitrary type.
22    * This interface provides access to a read-only key that the user should not modify as this would lead to
23    * undefined behavior
24    * This is a stateless operation, i.e, {@link #apply(Object, Object, Object)} is invoked individually for each joining
25    * record-pair of a {@link KStream}-{@link KStream}, {@link KStream}-{@link KTable}, or {@link KTable}-{@link KTable}
26    * join.
27    *
28    * @param <K1> key value type
29    * @param <V1> first value type
30    * @param <V2> second value type
31    * @param <VR> joined value type
32    * @see KStream#join(KStream, ValueJoinerWithKey, JoinWindows)
33    * @see KStream#join(KStream, ValueJoinerWithKey, JoinWindows, StreamJoined)
34    * @see KStream#leftJoin(KStream, ValueJoinerWithKey, JoinWindows)
35    * @see KStream#leftJoin(KStream, ValueJoinerWithKey, JoinWindows, StreamJoined)
36    * @see KStream#outerJoin(KStream, ValueJoinerWithKey, JoinWindows)
37    * @see KStream#outerJoin(KStream, ValueJoinerWithKey, JoinWindows, StreamJoined)
38    * @see KStream#join(KTable, ValueJoinerWithKey)
39    * @see KStream#join(KTable, ValueJoinerWithKey, Joined)
40    * @see KStream#leftJoin(KTable, ValueJoinerWithKey)
41    * @see KStream#leftJoin(KTable, ValueJoinerWithKey, Joined)
42    * @see KStream#join(GlobalKTable, KeyValueMapper, ValueJoinerWithKey) 
43    * @see KStream#join(GlobalKTable, KeyValueMapper, ValueJoinerWithKey, Named)
44    * @see KStream#leftJoin(GlobalKTable, KeyValueMapper, ValueJoinerWithKey)
45    * @see KStream#leftJoin(GlobalKTable, KeyValueMapper, ValueJoinerWithKey, Named)
46    */
47   public interface ValueJoinerWithKey<K1, V1, V2, VR> {
48   
49       /**
50        * Return a joined value consisting of {@code readOnlyKey}, {@code value1} and {@code value2}.
51        *
52        * @param readOnlyKey the key
53        * @param value1 the first value for joining
54        * @param value2 the second value for joining
55        * @return the joined value
56        */
57       VR apply(final K1 readOnlyKey, final V1 value1, final V2 value2);
58   }
