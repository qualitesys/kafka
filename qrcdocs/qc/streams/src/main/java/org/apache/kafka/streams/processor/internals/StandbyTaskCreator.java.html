40   class StandbyTaskCreator {
41       private final TopologyMetadata topologyMetadata;
42       private final StreamsConfig applicationConfig;
43       private final StreamsMetricsImpl streamsMetrics;
44       private final StateDirectory stateDirectory;
45       private final ChangelogReader storeChangelogReader;
46       private final ThreadCache dummyCache;
47       private final Logger log;
48       private final Sensor createTaskSensor;
49   
50       // tasks may be assigned for a NamedTopology that is not yet known by this host, and saved for later creation
51       private final Map<TaskId, Set<TopicPartition>> unknownTasksToBeCreated = new HashMap<>();
52   
53       StandbyTaskCreator(final TopologyMetadata topologyMetadata,
54                          final StreamsConfig applicationConfig,
55                          final StreamsMetricsImpl streamsMetrics,
56                          final StateDirectory stateDirectory,
57                          final ChangelogReader storeChangelogReader,
58                          final String threadId,
59                          final Logger log) {
60           this.topologyMetadata = topologyMetadata;
61           this.applicationConfig = applicationConfig;
62           this.streamsMetrics = streamsMetrics;
63           this.stateDirectory = stateDirectory;
64           this.storeChangelogReader = storeChangelogReader;
65           this.log = log;
66   
67           createTaskSensor = ThreadMetrics.createTaskSensor(threadId, streamsMetrics);
68   
69           dummyCache = new ThreadCache(
70               new LogContext(String.format("stream-thread [%s] ", Thread.currentThread().getName())),
71               0,
72               streamsMetrics
73           );
74       }
75   
76       void removeRevokedUnknownTasks(final Set<TaskId> assignedTasks) {
77           unknownTasksToBeCreated.keySet().retainAll(assignedTasks);
78       }
79   
80       Map<TaskId, Set<TopicPartition>> uncreatedTasksForTopologies(final Set<String> currentTopologies) {
81           return filterMap(unknownTasksToBeCreated, t -> currentTopologies.contains(t.getKey().topologyName()));
82       }

84       // TODO: change return type to `StandbyTask`
85       Collection<Task> createTasks(final Map<TaskId, Set<TopicPartition>> tasksToBeCreated) {
86           // TODO: change type to `StandbyTask`
87           final List<Task> createdTasks = new ArrayList<>();
88           final Map<TaskId, Set<TopicPartition>>  newUnknownTasks = new HashMap<>();
89   
90           for (final Map.Entry<TaskId, Set<TopicPartition>> newTaskAndPartitions : tasksToBeCreated.entrySet()) {
91               final TaskId taskId = newTaskAndPartitions.getKey();
92               final Set<TopicPartition> partitions = newTaskAndPartitions.getValue();
93   
94               // task belongs to a named topology that hasn't been added yet, wait until it has to create this
95               if (taskId.topologyName() != null && !topologyMetadata.namedTopologiesView().contains(taskId.topologyName())) {
96                   newUnknownTasks.put(taskId, partitions);
97                   continue;
98               }
99   
100               final ProcessorTopology topology = topologyMetadata.buildSubtopology(taskId);
101   
102               if (topology.hasStateWithChangelogs()) {
103                   final ProcessorStateManager stateManager = new ProcessorStateManager(
104                       taskId,
105                       Task.TaskType.STANDBY,
106                       eosEnabled(applicationConfig),
107                       getLogContext(taskId),
108                       stateDirectory,
109                       storeChangelogReader,
110                       topology.storeToChangelogTopic(),
111                       partitions
112                   );
113   
114                   final InternalProcessorContext context = new ProcessorContextImpl(
115                       taskId,
116                       applicationConfig,
117                       stateManager,
118                       streamsMetrics,
119                       dummyCache
120                   );
121   
122                   createdTasks.add(createStandbyTask(taskId, partitions, topology, stateManager, context));
123               } else {
124                   log.trace(
125                       "Skipped standby task {} with assigned partitions {} " +
126                           "since it does not have any state stores to materialize",
127                       taskId, partitions
128                   );
129               }
130               unknownTasksToBeCreated.remove(taskId);
131           }
132           if (!newUnknownTasks.isEmpty()) {
133               log.info("Delaying creation of tasks not yet known by this instance: {}", newUnknownTasks.keySet());
134               unknownTasksToBeCreated.putAll(newUnknownTasks);
135           }
136           return createdTasks;
137       }

139       StandbyTask createStandbyTaskFromActive(final StreamTask streamTask,
140                                               final Set<TopicPartition> inputPartitions) {
141           final InternalProcessorContext context = streamTask.processorContext();
142           final ProcessorStateManager stateManager = streamTask.stateMgr;
143   
144           streamTask.closeCleanAndRecycleState();
145           stateManager.transitionTaskType(TaskType.STANDBY, getLogContext(streamTask.id()));
146   
147           return createStandbyTask(
148               streamTask.id(),
149               inputPartitions,
150               topologyMetadata.buildSubtopology(streamTask.id),
151               stateManager,
152               context
153           );
154       }
155   
156       StandbyTask createStandbyTask(final TaskId taskId,
157                                     final Set<TopicPartition> inputPartitions,
158                                     final ProcessorTopology topology,
159                                     final ProcessorStateManager stateManager,
160                                     final InternalProcessorContext context) {
161           final StandbyTask task = new StandbyTask(
162               taskId,
163               inputPartitions,
164               topology,
165               topologyMetadata.getTaskConfigFor(taskId),
166               streamsMetrics,
167               stateManager,
168               stateDirectory,
169               dummyCache,
170               context
171           );
172   
173           log.trace("Created task {} with assigned partitions {}", taskId, inputPartitions);
174           createTaskSensor.record();
175           return task;
176       }
177   
178       private LogContext getLogContext(final TaskId taskId) {
179           final String threadIdPrefix = String.format("stream-thread [%s] ", Thread.currentThread().getName());
180           final String logPrefix = threadIdPrefix + String.format("%s [%s] ", "standby-task", taskId);
181           return new LogContext(logPrefix);
182       }
