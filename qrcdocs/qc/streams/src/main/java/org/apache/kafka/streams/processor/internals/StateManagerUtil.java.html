39   /**
40    * Shared functions to handle state store registration and cleanup between
41    * active and standby tasks.
42    */
43   final class StateManagerUtil {
44       static final String CHECKPOINT_FILE_NAME = ".checkpoint";
45       static final long OFFSET_DELTA_THRESHOLD_FOR_CHECKPOINT = 10_000L;
46   
47       private StateManagerUtil() {}
48   
49       static RecordConverter converterForStore(final StateStore store) {
50           return isTimestamped(store) ? rawValueToTimestampedValue() : identity();
51       }

53       static boolean checkpointNeeded(final boolean enforceCheckpoint,
54                                       final Map<TopicPartition, Long> oldOffsetSnapshot,
55                                       final Map<TopicPartition, Long> newOffsetSnapshot) {
56           // we should always have the old snapshot post completing the register state stores;
57           // if it is null it means the registration is not done and hence we should not overwrite the checkpoint
58           if (oldOffsetSnapshot == null) {
59               return false;
60           }
61   
62           if (enforceCheckpoint)
63               return true;
64   
65           // we can checkpoint if the difference between the current and the previous snapshot is large enough
66           long totalOffsetDelta = 0L;
67           for (final Map.Entry<TopicPartition, Long> entry : newOffsetSnapshot.entrySet()) {
68               totalOffsetDelta += entry.getValue() - oldOffsetSnapshot.getOrDefault(entry.getKey(), 0L);
69           }
70   
71           // when enforcing checkpoint is required, we should overwrite the checkpoint if it is different from the old one;
72           // otherwise, we only overwrite the checkpoint if it is largely different from the old one
73           return totalOffsetDelta > OFFSET_DELTA_THRESHOLD_FOR_CHECKPOINT;
74       }
75   
76       /**
77        * @throws StreamsException If the store's changelog does not contain the partition
78        */
79       static void registerStateStores(final Logger log,
80                                       final String logPrefix,
81                                       final ProcessorTopology topology,
82                                       final ProcessorStateManager stateMgr,
83                                       final StateDirectory stateDirectory,
84                                       final InternalProcessorContext processorContext) {
85           if (topology.stateStores().isEmpty()) {
86               return;
87           }
88   
89           final TaskId id = stateMgr.taskId();
90           if (!stateDirectory.lock(id)) {
91               throw new LockException(String.format("%sFailed to lock the state directory for task %s", logPrefix, id));
92           }
93           log.debug("Acquired state directory lock");
94   
95           final boolean storeDirsEmpty = stateDirectory.directoryForTaskIsEmpty(id);
96   
97           stateMgr.registerStateStores(topology.stateStores(), processorContext);
98           log.debug("Registered state stores");
99   
100           // We should only load checkpoint AFTER the corresponding state directory lock has been acquired and
101           // the state stores have been registered; we should not try to load at the state manager construction time.
102           // See https://issues.apache.org/jira/browse/KAFKA-8574
103           stateMgr.initializeStoreOffsetsFromCheckpoint(storeDirsEmpty);
104           log.debug("Initialized state stores");
105       }

107       /**
108        * @throws ProcessorStateException if there is an error while closing the state manager
109        */
110       static void closeStateManager(final Logger log,
111                                     final String logPrefix,
112                                     final boolean closeClean,
113                                     final boolean eosEnabled,
114                                     final ProcessorStateManager stateMgr,
115                                     final StateDirectory stateDirectory,
116                                     final TaskType taskType) {
117           // if EOS is enabled, wipe out the whole state store for unclean close since it is now invalid
118           final boolean wipeStateStore = !closeClean && eosEnabled;
119   
120           final TaskId id = stateMgr.taskId();
121           log.trace("Closing state manager for {} task {}", taskType, id);
122   
123           final AtomicReference<ProcessorStateException> firstException = new AtomicReference<>(null);
124           try {
125               if (stateDirectory.lock(id)) {
126                   try {
127                       stateMgr.close();
128                   } catch (final ProcessorStateException e) {
129                       firstException.compareAndSet(null, e);
130                   } finally {
131                       try {
132                           if (wipeStateStore) {
133                               log.debug("Wiping state stores for {} task {}", taskType, id);
134                               // we can just delete the whole dir of the task, including the state store images and the checkpoint files,
135                               // and then we write an empty checkpoint file indicating that the previous close is graceful and we just
136                               // need to re-bootstrap the restoration from the beginning
137                               Utils.delete(stateMgr.baseDir());
138                           }
139                       } finally {
140                           stateDirectory.unlock(id);
141                       }
142                   }
143               }
144           } catch (final IOException e) {
145               final ProcessorStateException exception = new ProcessorStateException(
146                   String.format("%sFatal error while trying to close the state manager for task %s", logPrefix, id), e
147               );
148               firstException.compareAndSet(null, exception);
149           }
150   
151           final ProcessorStateException exception = firstException.get();
152           if (exception != null) {
153               throw exception;
154           }
155       }

157       /**
158        *  Parse the task directory name (of the form topicGroupId_partition) and construct the TaskId with the
159        *  optional namedTopology (may be null)
160        *
161        *  @throws TaskIdFormatException if the taskIdStr is not a valid {@link TaskId}
162        */
163       static TaskId parseTaskDirectoryName(final String taskIdStr, final String namedTopology) {
164           final int index = taskIdStr.indexOf('_');
165           if (index <= 0 || index + 1 >= taskIdStr.length()) {
166               throw new TaskIdFormatException(taskIdStr);
167           }
168   
169           try {
170               final int topicGroupId = Integer.parseInt(taskIdStr.substring(0, index));
171               final int partition = Integer.parseInt(taskIdStr.substring(index + 1));
172   
173               return new TaskId(topicGroupId, partition, namedTopology);
174           } catch (final Exception e) {
175               throw new TaskIdFormatException(taskIdStr);
176           }
177       }
178   
179       /**
180        * @return The string representation of the subtopology and partition metadata, ie the task id string without
181        *         the named topology, which defines the innermost task directory name of this task's state
182        */
183       static String toTaskDirString(final TaskId taskId) {
184           return taskId.subtopology() + "_" + taskId.partition();
185       }
