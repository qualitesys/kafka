33   /**
34    * The task ID representation composed as subtopology (aka topicGroupId) plus the assigned partition ID.
35    */
36   public class TaskId implements Comparable<TaskId> {
37   
38       private static final Logger LOG = LoggerFactory.getLogger(TaskId.class);
39   
40       public static final String NAMED_TOPOLOGY_DELIMITER = "__";
41   
42       /** The ID of the subtopology, aka topicGroupId. */
43       @Deprecated
44       public final int topicGroupId;
45       /** The ID of the partition. */
46       @Deprecated
47       public final int partition;
48   
49       /** The namedTopology that this task belongs to, or null if it does not belong to one */
50       private final String topologyName;
51   
52       public TaskId(final int topicGroupId, final int partition) {
53           this(topicGroupId, partition, null);
54       }
55   
56       public TaskId(final int topicGroupId, final int partition, final String topologyName) {
57           this.topicGroupId = topicGroupId;
58           this.partition = partition;
59           if (topologyName != null && topologyName.length() == 0) {
60               LOG.warn("Empty string passed in for task's namedTopology, since NamedTopology name cannot be empty, we "
61                            + "assume this task does not belong to a NamedTopology and downgrade this to null");
62               this.topologyName = null;
63           } else {
64               this.topologyName = topologyName;
65           }
66       }
67   
68       public int subtopology() {
69           return topicGroupId;
70       }
71   
72       public int partition() {
73           return partition;
74       }
75   
76       /**
77        * Experimental feature -- will return null
78        */
79       public String topologyName() {
80           return topologyName;
81       }
82   
83       @Override
84       public String toString() {
85           return topologyName != null ? topologyName + NAMED_TOPOLOGY_DELIMITER + topicGroupId + "_" + partition : topicGroupId + "_" + partition;
86       }
87   
88       /**
89        * @throws TaskIdFormatException if the taskIdStr is not a valid {@link TaskId}
90        */
91       public static TaskId parse(final String taskIdStr) {
92           try {
93               final int namedTopologyDelimiterIndex = taskIdStr.indexOf(NAMED_TOPOLOGY_DELIMITER);
94               // If there is no copy of the NamedTopology delimiter, this task has no named topology and only one `_` char
95               if (namedTopologyDelimiterIndex < 0) {
96                   final int index = taskIdStr.indexOf('_');
97   
98                   final int topicGroupId = Integer.parseInt(taskIdStr.substring(0, index));
99                   final int partition = Integer.parseInt(taskIdStr.substring(index + 1));
100   
101                   return new TaskId(topicGroupId, partition);
102               } else {
103                   final int topicGroupIdIndex = namedTopologyDelimiterIndex + 2;
104                   final int subtopologyPartitionDelimiterIndex = taskIdStr.indexOf('_', topicGroupIdIndex);
105   
106                   final String namedTopology = taskIdStr.substring(0, namedTopologyDelimiterIndex);
107                   final int topicGroupId = Integer.parseInt(taskIdStr.substring(topicGroupIdIndex, subtopologyPartitionDelimiterIndex));
108                   final int partition = Integer.parseInt(taskIdStr.substring(subtopologyPartitionDelimiterIndex + 1));
109   
110                   return new TaskId(topicGroupId, partition, namedTopology);
111               }
112           } catch (final Exception e) {
113               throw new TaskIdFormatException(taskIdStr);
114           }
115       }
116   
117       /**
118        * @throws IOException if cannot write to output stream
119        * @deprecated since 3.0, for internal use, will be removed
120        */
121       @Deprecated
122       public void writeTo(final DataOutputStream out, final int version) throws IOException {
123           writeTaskIdTo(this, out, version);
124       }
125   
126       /**
127        * @throws IOException if cannot read from input stream
128        * @deprecated since 3.0, for internal use, will be removed
129        */
130       @Deprecated
131       public static TaskId readFrom(final DataInputStream in, final int version) throws IOException {
132           return readTaskIdFrom(in, version);
133       }
134   
135       /**
136        * @deprecated since 3.0, for internal use, will be removed
137        */
138       @Deprecated
139       public void writeTo(final ByteBuffer buf, final int version) {
140           writeTaskIdTo(this, buf, version);
141       }
142   
143       /**
144        * @deprecated since 3.0, for internal use, will be removed
145        */
146       @Deprecated
147       public static TaskId readFrom(final ByteBuffer buf, final int version) {
148           return readTaskIdFrom(buf, version);
149       }
150   
151       @Override
152       public boolean equals(final Object o) {
153           if (this == o) {
154               return true;
155           }
156           if (o == null || getClass() != o.getClass()) {
157               return false;
158           }
159           final TaskId taskId = (TaskId) o;
160   
161           if (topicGroupId != taskId.topicGroupId || partition != taskId.partition) {
162               return false;
163           }
164   
165           if (topologyName != null && taskId.topologyName != null) {
166               return topologyName.equals(taskId.topologyName);
167           } else {
168               return topologyName == null && taskId.topologyName == null;
169           }
170       }
171   
172       @Override
173       public int hashCode() {
174           return Objects.hash(topicGroupId, partition, topologyName);
175       }
176   
177       @Override
178       public int compareTo(final TaskId other) {
179           if (topologyName != null && other.topologyName != null) {
180               final int comparingNamedTopologies = topologyName.compareTo(other.topologyName);
181               if (comparingNamedTopologies != 0) {
182                   return comparingNamedTopologies;
183               }
184           } else if (topologyName != null || other.topologyName != null) {
185               LOG.error("Tried to compare this = {} with other = {}, but only one had a valid named topology", this, other);
186               throw new IllegalStateException("Can't compare a TaskId with a namedTopology to one without");
187           }
188           final int comparingTopicGroupId = Integer.compare(this.topicGroupId, other.topicGroupId);
189           return comparingTopicGroupId != 0 ? comparingTopicGroupId : Integer.compare(this.partition, other.partition);
190       }
191   }
