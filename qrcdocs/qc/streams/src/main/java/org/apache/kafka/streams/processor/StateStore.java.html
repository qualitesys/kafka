17   package org.apache.kafka.streams.processor;
18   
19   import org.apache.kafka.common.annotation.InterfaceStability.Evolving;
20   import org.apache.kafka.streams.errors.StreamsException;
21   import org.apache.kafka.streams.processor.api.ProcessorContext;
22   import org.apache.kafka.streams.processor.internals.StoreToProcessorContextAdapter;
23   import org.apache.kafka.streams.query.FailureReason;
24   import org.apache.kafka.streams.query.Position;
25   import org.apache.kafka.streams.query.PositionBound;
26   import org.apache.kafka.streams.query.Query;
27   import org.apache.kafka.streams.query.QueryConfig;
28   import org.apache.kafka.streams.query.QueryResult;
29   
30   /**
31    * A storage engine for managing state maintained by a stream processor.
32    * <p>
33    * If the store is implemented as a persistent store, it <em>must</em> use the store name as directory name and write
34    * all data into this store directory.
35    * The store directory must be created with the state directory.
36    * The state directory can be obtained via {@link ProcessorContext#stateDir() #stateDir()} using the
37    * {@link ProcessorContext} provided via {@link #init(StateStoreContext, StateStore) init(...)}.
38    * <p>
39    * Using nested store directories within the state directory isolates different state stores.
40    * If a state store would write into the state directory directly, it might conflict with others state stores and thus,
41    * data might get corrupted and/or Streams might fail with an error.
42    * Furthermore, Kafka Streams relies on using the store name as store directory name to perform internal cleanup tasks.
43    * <p>
44    * This interface does not specify any query capabilities, which, of course,
45    * would be query engine specific. Instead, it just specifies the minimum
46    * functionality required to reload a storage engine from its changelog as well
47    * as basic lifecycle management.
48    */
49   public interface StateStore {
50   
51       /**
52        * The name of this store.
53        * @return the storage name
54        */
55       String name();
56   
57       /**
58        * Initializes this state store.
59        * <p>
60        * The implementation of this function must register the root store in the context via the
61        * {@link org.apache.kafka.streams.processor.ProcessorContext#register(StateStore, StateRestoreCallback)} function,
62        * where the first {@link StateStore} parameter should always be the passed-in {@code root} object, and
63        * the second parameter should be an object of user's implementation
64        * of the {@link StateRestoreCallback} interface used for restoring the state store from the changelog.
65        * <p>
66        * Note that if the state store engine itself supports bulk writes, users can implement another
67        * interface {@link BatchingStateRestoreCallback} which extends {@link StateRestoreCallback} to
68        * let users implement bulk-load restoration logic instead of restoring one record at a time.
69        * <p>
70        * This method is not called if {@link StateStore#init(StateStoreContext, StateStore)}
71        * is implemented.
72        *
73        * @throws IllegalStateException If store gets registered after initialized is already finished
74        * @throws StreamsException if the store's change log does not contain the partition
75        * @deprecated Since 2.7.0. Callers should invoke {@link #init(StateStoreContext, StateStore)} instead.
76        *             Implementers may choose to implement this method for backward compatibility or to throw an
77        *             informative exception instead.
78        */
79       @Deprecated
80       void init(org.apache.kafka.streams.processor.ProcessorContext context, StateStore root);
81   
82       /**
83        * Initializes this state store.
84        * <p>
85        * The implementation of this function must register the root store in the context via the
86        * {@link StateStoreContext#register(StateStore, StateRestoreCallback, CommitCallback)} function, where the
87        * first {@link StateStore} parameter should always be the passed-in {@code root} object, and
88        * the second parameter should be an object of user's implementation
89        * of the {@link StateRestoreCallback} interface used for restoring the state store from the changelog.
90        * <p>
91        * Note that if the state store engine itself supports bulk writes, users can implement another
92        * interface {@link BatchingStateRestoreCallback} which extends {@link StateRestoreCallback} to
93        * let users implement bulk-load restoration logic instead of restoring one record at a time.
94        *
95        * @throws IllegalStateException If store gets registered after initialized is already finished
96        * @throws StreamsException if the store's change log does not contain the partition
97        */
98       default void init(final StateStoreContext context, final StateStore root) {
99           init(StoreToProcessorContextAdapter.adapt(context), root);
100       }
101   
102       /**
103        * Flush any cached data
104        */
105       void flush();
106   
107       /**
108        * Close the storage engine.
109        * Note that this function needs to be idempotent since it may be called
110        * several times on the same state store.
111        * <p>
112        * Users only need to implement this function but should NEVER need to call this api explicitly
113        * as it will be called by the library automatically when necessary
114        */
115       void close();
116   
117       /**
118        * Return if the storage is persistent or not.
119        *
120        * @return  {@code true} if the storage is persistent&mdash;{@code false} otherwise
121        */
122       boolean persistent();
123   
124       /**
125        * Is this store open for reading and writing
126        * @return {@code true} if the store is open
127        */
128       boolean isOpen();
129   
130       /**
131        * Execute a query. Returns a QueryResult containing either result data or
132        * a failure.
133        * <p>
134        * If the store doesn't know how to handle the given query, the result
135        * shall be a {@link FailureReason#UNKNOWN_QUERY_TYPE}.
136        * If the store couldn't satisfy the given position bound, the result
137        * shall be a {@link FailureReason#NOT_UP_TO_BOUND}.
138        * <p>
139        * Note to store implementers: if your store does not support position tracking,
140        * you can correctly respond {@link FailureReason#NOT_UP_TO_BOUND} if the argument is
141        * anything but {@link PositionBound#unbounded()}. Be sure to explain in the failure message
142        * that bounded positions are not supported.
143        * <p>
144        * @param query The query to execute
145        * @param positionBound The position the store must be at or past
146        * @param config Per query configuration parameters, such as whether the store should collect detailed execution
147        * info for the query
148        * @param <R> The result type
149        */
150       @Evolving
151       default <R> QueryResult<R> query(
152           final Query<R> query,
153           final PositionBound positionBound,
154           final QueryConfig config) {
155           // If a store doesn't implement a query handler, then all queries are unknown.
156           return QueryResult.forUnknownQueryType(query, this);
157       }
158   
159       /**
160        * Returns the position the state store is at with respect to the input topic/partitions
161        */
162       @Evolving
163       default Position getPosition() {
164           throw new UnsupportedOperationException(
165               "getPosition is not implemented by this StateStore (" + getClass() + ")"
166           );
167       }
168   }
