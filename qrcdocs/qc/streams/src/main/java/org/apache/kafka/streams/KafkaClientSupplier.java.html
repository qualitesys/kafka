17   package org.apache.kafka.streams;
18   
19   import org.apache.kafka.clients.admin.Admin;
20   import org.apache.kafka.clients.consumer.Consumer;
21   import org.apache.kafka.clients.producer.Producer;
22   import org.apache.kafka.streams.kstream.GlobalKTable;
23   import org.apache.kafka.streams.processor.StateStore;
24   
25   import java.util.Map;
26   
27   /**
28    * {@code KafkaClientSupplier} can be used to provide custom Kafka clients to a {@link KafkaStreams} instance.
29    *
30    * @see KafkaStreams#KafkaStreams(Topology, java.util.Properties, KafkaClientSupplier)
31    */
32   public interface KafkaClientSupplier {
33       /**
34        * Create an {@link Admin} which is used for internal topic management.
35        *
36        * @param config Supplied by the {@link java.util.Properties} given to the {@link KafkaStreams}
37        * @return an instance of {@link Admin}
38        */
39       default Admin getAdmin(final Map<String, Object> config) {
40           throw new UnsupportedOperationException("Implementations of KafkaClientSupplier should implement the getAdmin() method.");
41       }
42   
43       /**
44        * Create a {@link Producer} which is used to write records to sink topics.
45        *
46        * @param config {@link StreamsConfig#getProducerConfigs(String) producer config} which is supplied by the
47        *               {@link java.util.Properties} given to the {@link KafkaStreams} instance
48        * @return an instance of Kafka producer
49        */
50       Producer<byte[], byte[]> getProducer(final Map<String, Object> config);
51   
52       /**
53        * Create a {@link Consumer} which is used to read records of source topics.
54        *
55        * @param config {@link StreamsConfig#getMainConsumerConfigs(String, String, int) consumer config} which is
56        *               supplied by the {@link java.util.Properties} given to the {@link KafkaStreams} instance
57        * @return an instance of Kafka consumer
58        */
59       Consumer<byte[], byte[]> getConsumer(final Map<String, Object> config);
60   
61       /**
62        * Create a {@link Consumer} which is used to read records to restore {@link StateStore}s.
63        *
64        * @param config {@link StreamsConfig#getRestoreConsumerConfigs(String) restore consumer config} which is supplied
65        *               by the {@link java.util.Properties} given to the {@link KafkaStreams}
66        * @return an instance of Kafka consumer
67        */
68       Consumer<byte[], byte[]> getRestoreConsumer(final Map<String, Object> config);
69   
70       /**
71        * Create a {@link Consumer} which is used to consume records for {@link GlobalKTable}.
72        *
73        * @param config {@link StreamsConfig#getGlobalConsumerConfigs(String) global consumer config} which is supplied
74        *               by the {@link java.util.Properties} given to the {@link KafkaStreams}
75        * @return an instance of Kafka consumer
76        */
77       Consumer<byte[], byte[]> getGlobalConsumer(final Map<String, Object> config);
78   }
