61       static {
62           // Just statically check to make sure that the generated code always stays in sync with the overall protocol
63           final int subscriptionInfoLatestVersion = SubscriptionInfoData.SCHEMAS.length - 1;
64           if (subscriptionInfoLatestVersion != LATEST_SUPPORTED_VERSION) {
65               throw new IllegalArgumentException(
66                   "streams/src/main/resources/common/message/SubscriptionInfo.json needs to be updated to match the " +
67                       "latest assignment protocol version. SubscriptionInfo only supports up to  ["
68                       + subscriptionInfoLatestVersion + "] but needs to support up to [" + LATEST_SUPPORTED_VERSION + "].");
69           }

72       private static void validateVersions(final int version, final int latestSupportedVersion) {
73           if (latestSupportedVersion == UNKNOWN && (version < 1 || version > 2)) {
74               throw new IllegalArgumentException(
75                   "Only versions 1 and 2 are expected to use an UNKNOWN (-1) latest supported version. " +
76                       "Got " + version + "."
77               );
78           } else if (latestSupportedVersion != UNKNOWN && (version < 1 || version > latestSupportedVersion)) {
79               throw new IllegalArgumentException(
80                   "version must be between 1 and " + latestSupportedVersion + "; was: " + version
81               );
82           }
83       }

85       public SubscriptionInfo(final int version,
86                               final int latestSupportedVersion,
87                               final UUID processId,
88                               final String userEndPoint,
89                               final Map<TaskId, Long> taskOffsetSums,
90                               final byte uniqueField,
91                               final int errorCode,
92                               final Map<String, String> clientTags) {
93           validateVersions(version, latestSupportedVersion);
94           final SubscriptionInfoData data = new SubscriptionInfoData();
95           data.setVersion(version);
96           data.setProcessId(new Uuid(processId.getMostSignificantBits(),
97                   processId.getLeastSignificantBits()));
98   
99           if (version >= 2) {
100               data.setUserEndPoint(userEndPoint == null
101                       ? new byte[0]
102                       : userEndPoint.getBytes(StandardCharsets.UTF_8));
103           }
104           if (version >= 3) {
105               data.setLatestSupportedVersion(latestSupportedVersion);
106           }
107           if (version >= 8) {
108               data.setUniqueField(uniqueField);
109           }
110           if (version >= 9) {
111               data.setErrorCode(errorCode);
112           }
113           if (version >= 11) {
114               data.setClientTags(buildClientTagsFromMap(clientTags));
115           }
116   
117           this.data = data;
118   
119           if (version >= MIN_NAMED_TOPOLOGY_VERSION) {
120               setTaskOffsetSumDataWithNamedTopologiesFromTaskOffsetSumMap(taskOffsetSums);
121           } else if (version >= MIN_VERSION_OFFSET_SUM_SUBSCRIPTION) {
122               setTaskOffsetSumDataFromTaskOffsetSumMap(taskOffsetSums);
123           } else {
124               setPrevAndStandbySetsFromParsedTaskOffsetSumMap(taskOffsetSums);
125           }
126       }

129           validateVersions(subscriptionInfoData.version(), subscriptionInfoData.latestSupportedVersion());
130           this.data = subscriptionInfoData;
131       }

147       private List<ClientTag> buildClientTagsFromMap(final Map<String, String> clientTags) {
148           return clientTags.entrySet().stream()
149               .map(clientTagEntry -> {
150                   final ClientTag clientTag = new ClientTag();
151                   clientTag.setKey(clientTagEntry.getKey().getBytes(StandardCharsets.UTF_8));
152                   clientTag.setValue(clientTagEntry.getValue().getBytes(StandardCharsets.UTF_8));
153                   return clientTag;
154               })
155               .collect(Collectors.toList());
156       }

158       // For version > MIN_NAMED_TOPOLOGY_VERSION
159       private void setTaskOffsetSumDataWithNamedTopologiesFromTaskOffsetSumMap(final Map<TaskId, Long> taskOffsetSums) {
160           data.setTaskOffsetSums(taskOffsetSums.entrySet().stream().map(t -> {
161               final SubscriptionInfoData.TaskOffsetSum taskOffsetSum = new SubscriptionInfoData.TaskOffsetSum();
162               final TaskId task = t.getKey();
163               taskOffsetSum.setTopicGroupId(task.subtopology());
164               taskOffsetSum.setPartition(task.partition());
165               taskOffsetSum.setNamedTopology(task.topologyName());
166               taskOffsetSum.setOffsetSum(t.getValue());
167               return taskOffsetSum;
168           }).collect(Collectors.toList()));
169       }
170   
171       // For MIN_NAMED_TOPOLOGY_VERSION > version > MIN_VERSION_OFFSET_SUM_SUBSCRIPTION
172       private void setTaskOffsetSumDataFromTaskOffsetSumMap(final Map<TaskId, Long> taskOffsetSums) {
173           final Map<Integer, List<SubscriptionInfoData.PartitionToOffsetSum>> topicGroupIdToPartitionOffsetSum = new HashMap<>();
174           for (final Map.Entry<TaskId, Long> taskEntry : taskOffsetSums.entrySet()) {
175               final TaskId task = taskEntry.getKey();
176               if (task.topologyName() != null) {
177                   throw new TaskAssignmentException("Named topologies are not compatible with older protocol versions");
178               }
179               topicGroupIdToPartitionOffsetSum.computeIfAbsent(task.subtopology(), t -> new ArrayList<>()).add(
180                   new SubscriptionInfoData.PartitionToOffsetSum()
181                       .setPartition(task.partition())
182                       .setOffsetSum(taskEntry.getValue()));
183           }
184   
185           data.setTaskOffsetSums(topicGroupIdToPartitionOffsetSum.entrySet().stream().map(t -> {
186               final SubscriptionInfoData.TaskOffsetSum taskOffsetSum = new SubscriptionInfoData.TaskOffsetSum();
187               taskOffsetSum.setTopicGroupId(t.getKey());
188               taskOffsetSum.setPartitionToOffsetSum(t.getValue());
189               return taskOffsetSum;
190           }).collect(Collectors.toList()));
191       }
192   
193       // For MIN_VERSION_OFFSET_SUM_SUBSCRIPTION > version
194       private void setPrevAndStandbySetsFromParsedTaskOffsetSumMap(final Map<TaskId, Long> taskOffsetSums) {
195           final Set<TaskId> prevTasks = new HashSet<>();
196           final Set<TaskId> standbyTasks = new HashSet<>();
197   
198           for (final Map.Entry<TaskId, Long> taskOffsetSum : taskOffsetSums.entrySet()) {
199               if (taskOffsetSum.getKey().topologyName() != null) {
200                   throw new TaskAssignmentException("Named topologies are not compatible with older protocol versions");
201               }
202               if (taskOffsetSum.getValue() == Task.LATEST_OFFSET) {
203                   prevTasks.add(taskOffsetSum.getKey());
204               } else {
205                   standbyTasks.add(taskOffsetSum.getKey());
206               }
207           }
208   
209           data.setPrevTasks(prevTasks.stream().map(t -> {
210               final SubscriptionInfoData.TaskId taskId = new SubscriptionInfoData.TaskId();
211               taskId.setTopicGroupId(t.subtopology());
212               taskId.setPartition(t.partition());
213               return taskId;
214           }).collect(Collectors.toList()));
215           data.setStandbyTasks(standbyTasks.stream().map(t -> {
216               final SubscriptionInfoData.TaskId taskId = new SubscriptionInfoData.TaskId();
217               taskId.setTopicGroupId(t.subtopology());
218               taskId.setPartition(t.partition());
219               return taskId;
220           }).collect(Collectors.toList()));
221       }

235       public Set<TaskId> prevTasks() {
236           if (prevTasksCache == null) {
237               if (data.version() >= MIN_VERSION_OFFSET_SUM_SUBSCRIPTION) {
238                   prevTasksCache = getActiveTasksFromTaskOffsetSumMap(taskOffsetSums());
239               } else {
240                   prevTasksCache = Collections.unmodifiableSet(
241                       data.prevTasks()
242                           .stream()
243                           .map(t -> new TaskId(t.topicGroupId(), t.partition()))
244                           .collect(Collectors.toSet())
245                   );
246               }
247           }
248           return prevTasksCache;
249       }

251       public Set<TaskId> standbyTasks() {
252           if (standbyTasksCache == null) {
253               if (data.version() >= MIN_VERSION_OFFSET_SUM_SUBSCRIPTION) {
254                   standbyTasksCache = getStandbyTasksFromTaskOffsetSumMap(taskOffsetSums());
255               } else {
256                   standbyTasksCache = Collections.unmodifiableSet(
257                       data.standbyTasks()
258                           .stream()
259                           .map(t -> new TaskId(t.topicGroupId(), t.partition()))
260                           .collect(Collectors.toSet())
261                   );
262               }
263           }
264           return standbyTasksCache;
265       }

267       public Map<TaskId, Long> taskOffsetSums() {
268           if (taskOffsetSumsCache == null) {
269               taskOffsetSumsCache = new HashMap<>();
270               if (data.version() >= MIN_VERSION_OFFSET_SUM_SUBSCRIPTION) {
271                   for (final TaskOffsetSum taskOffsetSum : data.taskOffsetSums()) {
272                       if (data.version() >= MIN_NAMED_TOPOLOGY_VERSION) {
273                           taskOffsetSumsCache.put(
274                               new TaskId(taskOffsetSum.topicGroupId(),
275                                          taskOffsetSum.partition(),
276                                          taskOffsetSum.namedTopology()),
277                               taskOffsetSum.offsetSum());
278                       } else {
279                           for (final PartitionToOffsetSum partitionOffsetSum : taskOffsetSum.partitionToOffsetSum()) {
280                               taskOffsetSumsCache.put(
281                                   new TaskId(taskOffsetSum.topicGroupId(),
282                                              partitionOffsetSum.partition()),
283                                   partitionOffsetSum.offsetSum()
284                               );
285                           }
286                       }
287                   }
288               } else {
289                   prevTasks().forEach(taskId -> taskOffsetSumsCache.put(taskId, Task.LATEST_OFFSET));
290                   standbyTasks().forEach(taskId -> taskOffsetSumsCache.put(taskId, UNKNOWN_OFFSET_SUM));
291               }
292           }
293           return taskOffsetSumsCache;
294       }

310       private static Set<TaskId> taskOffsetSumMapToTaskSet(final Map<TaskId, Long> taskOffsetSums,
311                                                            final boolean getActiveTasks) {
312           return taskOffsetSums.entrySet().stream()
313                      .filter(t -> getActiveTasks == (t.getValue() == Task.LATEST_OFFSET))
314                      .map(Map.Entry::getKey)
315                      .collect(Collectors.toSet());
316       }

318       /**
319        * @throws TaskAssignmentException if method fails to encode the data
320        */
321       public ByteBuffer encode() {
322           if (data.version() > LATEST_SUPPORTED_VERSION) {
323               throw new IllegalStateException(
324                   "Should never try to encode a SubscriptionInfo with version [" +
325                       data.version() + "] > LATEST_SUPPORTED_VERSION [" + LATEST_SUPPORTED_VERSION + "]"
326               );
327           } else return MessageUtil.toByteBuffer(data, (short) data.version());
328       }

330       /**
331        * @throws TaskAssignmentException if method fails to decode the data
332        */
333       public static SubscriptionInfo decode(final ByteBuffer data) {
334           data.rewind();
335           final int version = data.getInt();
336   
337           if (version > LATEST_SUPPORTED_VERSION) {
338               // in this special case, we only rely on the version and latest version,
339               //
340               final int latestSupportedVersion = data.getInt();
341               final SubscriptionInfoData subscriptionInfoData = new SubscriptionInfoData();
342               subscriptionInfoData.setVersion(version);
343               subscriptionInfoData.setLatestSupportedVersion(latestSupportedVersion);
344               LOG.info("Unable to decode subscription data: used version: {}; latest supported version: {}",
345                   version,
346                   latestSupportedVersion
347               );
348               return new SubscriptionInfo(subscriptionInfoData);
349           } else {
350               data.rewind();
351               final ByteBufferAccessor accessor = new ByteBufferAccessor(data);
352               final SubscriptionInfoData subscriptionInfoData = new SubscriptionInfoData(accessor, (short) version);
353               return new SubscriptionInfo(subscriptionInfoData);
354           }
