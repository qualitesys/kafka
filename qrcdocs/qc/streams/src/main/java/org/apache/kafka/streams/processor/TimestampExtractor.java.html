17   package org.apache.kafka.streams.processor;
18   
19   import org.apache.kafka.clients.consumer.ConsumerRecord;
20   import org.apache.kafka.streams.kstream.KTable;
21   
22   /**
23    * An interface that allows the Kafka Streams framework to extract a timestamp from an instance of {@link ConsumerRecord}.
24    * The extracted timestamp is defined as milliseconds.
25    */
26   public interface TimestampExtractor {
27   
28       /**
29        * Extracts a timestamp from a record. The timestamp must be positive to be considered a valid timestamp.
30        * Returning a negative timestamp will cause the record not to be processed but rather silently skipped.
31        * In case the record contains a negative timestamp and this is considered a fatal error for the application,
32        * throwing a {@link RuntimeException} instead of returning the timestamp is a valid option too.
33        * For this case, Streams will stop processing and shut down to allow you investigate in the root cause of the
34        * negative timestamp.
35        * <p>
36        * The timestamp extractor implementation must be stateless.
37        * <p>
38        * The extracted timestamp MUST represent the milliseconds since midnight, January 1, 1970 UTC.
39        * <p>
40        * It is important to note that this timestamp may become the message timestamp for any messages sent to changelogs
41        * updated by {@link KTable}s and joins.
42        * The message timestamp is used for log retention and log rolling, so using nonsensical values may result in
43        * excessive log rolling and therefore broker performance degradation.
44        *
45        *
46        * @param record a data record
47        * @param partitionTime the highest extracted valid timestamp of the current record's partition˙ (could be -1 if unknown)
48        * @return the timestamp of the record
49        */
50       long extract(ConsumerRecord<Object, Object> record, long partitionTime);
51   }
