35   public class MemoryNavigableLRUCache extends MemoryLRUCache {
36   
37       private static final Logger LOG = LoggerFactory.getLogger(MemoryNavigableLRUCache.class);
38   
39       public MemoryNavigableLRUCache(final String name, final int maxCacheSize) {
40           super(name, maxCacheSize);
41       }
42   
43       @Override
44       public KeyValueIterator<Bytes, byte[]> range(final Bytes from, final Bytes to) {
45           if (Objects.nonNull(from) && Objects.nonNull(to) && from.compareTo(to) > 0) {
46               LOG.warn("Returning empty iterator for fetch with invalid key range: from > to. " +
47                       "This may be due to range arguments set in the wrong order, " +
48                       "or serdes that don't preserve ordering when lexicographically comparing the serialized bytes. " +
49                       "Note that the built-in numerical serdes do not follow this for negative numbers");
50               return KeyValueIterators.emptyIterator();
51           } else {
52               final TreeMap<Bytes, byte[]> treeMap = toTreeMap();
53               final Iterator<Bytes> keys = getIterator(treeMap, from, to, true);
54               return new DelegatingPeekingKeyValueIterator<>(name(),
55                       new MemoryNavigableLRUCache.CacheIterator(keys, treeMap));
56           }
57       }
58   
59       @Override
60       public KeyValueIterator<Bytes, byte[]> reverseRange(final Bytes from, final Bytes to) {
61           if (Objects.nonNull(from) && Objects.nonNull(to) && from.compareTo(to) > 0) {
62               LOG.warn("Returning empty iterator for fetch with invalid key range: from > to. " +
63                       "This may be due to range arguments set in the wrong order, " +
64                       "or serdes that don't preserve ordering when lexicographically comparing the serialized bytes. " +
65                       "Note that the built-in numerical serdes do not follow this for negative numbers");
66               return KeyValueIterators.emptyIterator();
67           } else {
68               final TreeMap<Bytes, byte[]> treeMap = toTreeMap();
69               final Iterator<Bytes> keys = getIterator(treeMap, from, to, false);
70               return new DelegatingPeekingKeyValueIterator<>(name(),
71                       new MemoryNavigableLRUCache.CacheIterator(keys, treeMap));
72           }
73       }
74   
75       private Iterator<Bytes> getIterator(final TreeMap<Bytes, byte[]> treeMap, final Bytes from, final Bytes to, final boolean forward) {
76           if (from == null && to == null) {
77               return forward ? treeMap.navigableKeySet().iterator() : treeMap.navigableKeySet().descendingIterator();
78           } else if (from == null) {
79               return forward ? treeMap.navigableKeySet().headSet(to, true).iterator() : treeMap.navigableKeySet().headSet(to, true).descendingIterator();
80           } else if (to == null) {
81               return forward ? treeMap.navigableKeySet().tailSet(from, true).iterator() : treeMap.navigableKeySet().tailSet(from, true).descendingIterator();
82           } else {
83               return forward ? treeMap.navigableKeySet().subSet(from, true, to, true).iterator() : treeMap.navigableKeySet().subSet(from, true, to, true).descendingIterator();
84           }
85       }
86   
87       @Override
88       public <PS extends Serializer<P>, P> KeyValueIterator<Bytes, byte[]> prefixScan(final P prefix, final PS prefixKeySerializer) {
89   
90           final Bytes from = Bytes.wrap(prefixKeySerializer.serialize(null, prefix));
91           final Bytes to = Bytes.increment(from);
92   
93           final TreeMap<Bytes, byte[]> treeMap = toTreeMap();
94   
95           return new DelegatingPeekingKeyValueIterator<>(
96               name(),
97               new MemoryNavigableLRUCache.CacheIterator(treeMap.subMap(from, true, to, false).keySet().iterator(), treeMap)
98           );
99       }
100   
101       @Override
102       public  KeyValueIterator<Bytes, byte[]> all() {
103           return range(null, null);
104       }
105   
106       @Override
107       public  KeyValueIterator<Bytes, byte[]> reverseAll() {
108           return reverseRange(null, null);
109       }

111       private synchronized TreeMap<Bytes, byte[]> toTreeMap() {
112           return new TreeMap<>(this.map);
113       }

118           final Query<R> query,

123               query,

125               config,
126               this,
127               getPosition(),
128               context
129           );
130       }

133       private static class CacheIterator implements KeyValueIterator<Bytes, byte[]> {
134           private final Iterator<Bytes> keys;
135           private final Map<Bytes, byte[]> entries;
136   
137           private CacheIterator(final Iterator<Bytes> keys, final Map<Bytes, byte[]> entries) {
138               this.keys = keys;
139               this.entries = entries;
140           }
141   
142           @Override
143           public boolean hasNext() {
144               return keys.hasNext();
145           }
146   
147           @Override
148           public KeyValue<Bytes, byte[]> next() {
149               final Bytes lastKey = keys.next();
150               return new KeyValue<>(lastKey, entries.get(lastKey));
151           }
152   
153           @Override
154           public void close() {
155               // do nothing
156           }
157   
158           @Override
159           public Bytes peekNextKey() {
160               throw new UnsupportedOperationException("peekNextKey not supported");
161           }
162       }
