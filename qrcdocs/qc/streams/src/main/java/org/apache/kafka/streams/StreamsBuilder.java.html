51   /**
52    * {@code StreamsBuilder} provide the high-level Kafka Streams DSL to specify a Kafka Streams topology.
53    *
54    * <p>
55    * It is a requirement that the processing logic ({@link Topology}) be defined in a deterministic way,
56    * as in, the order in which all operators are added must be predictable and the same across all application
57    * instances.
58    * Topologies are only identical if all operators are added in the same order.
59    * If different {@link KafkaStreams} instances of the same application build different topologies the result may be
60    * incompatible runtime code and unexpected results or errors
61    *
62    * @see Topology
63    * @see KStream
64    * @see KTable
65    * @see GlobalKTable
66    */

221       /**
222        * Create a {@link KTable} for the specified topic.
223        * The {@code "auto.offset.reset"} strategy, {@link TimestampExtractor}, key and value deserializers
224        * are defined by the options in {@link Consumed} are used.
225        * Input {@link KeyValue records} with {@code null} key will be dropped.
226        * <p>
227        * Note that the specified input topic must be partitioned by key.
228        * If this is not the case the returned {@link KTable} will be corrupted.
229        * <p>
230        * The resulting {@link KTable} will be materialized in a local {@link KeyValueStore} using the given
231        * {@code Materialized} instance.
232        * An internal changelog topic is created by default. Because the source topic can
233        * be used for recovery, you can avoid creating the changelog topic by setting
234        * the {@code "topology.optimization"} to {@code "all"} in the {@link StreamsConfig}.
235        * <p>
236        * You should only specify serdes in the {@link Consumed} instance as these will also be used to overwrite the
237        * serdes in {@link Materialized}, i.e.,
238        * <pre> {@code
239        * streamBuilder.table(topic, Consumed.with(Serde.String(), Serde.String()), Materialized.<String, String, KeyValueStore<Bytes, byte[]>as(storeName))
240        * }
241        * </pre>
242        * To query the local {@link ReadOnlyKeyValueStore} it must be obtained via
243        * {@link KafkaStreams#store(StoreQueryParameters) KafkaStreams#store(...)}:
244        * <pre>{@code
245        * KafkaStreams streams = ...
246        * ReadOnlyKeyValueStore<K, ValueAndTimestamp<V>> localStore = streams.store(queryableStoreName, QueryableStoreTypes.<K, ValueAndTimestamp<V>>timestampedKeyValueStore());
247        * K key = "some-key";
248        * ValueAndTimestamp<V> valueForKey = localStore.get(key); // key must be local (application state is shared over all running Kafka Streams instances)
249        * }</pre>
250        * For non-local keys, a custom RPC mechanism must be implemented using {@link KafkaStreams#metadataForAllStreamsClients()} to
251        * query the value of the key on a parallel running instance of your Kafka Streams application.
252        *
253        * @param topic              the topic name; cannot be {@code null}
254        * @param consumed           the instance of {@link Consumed} used to define optional parameters; cannot be {@code null}
255        * @param materialized       the instance of {@link Materialized} used to materialize a state store; cannot be {@code null}
256        * @return a {@link KTable} for the specified topic
257        */
258       public synchronized <K, V> KTable<K, V> table(final String topic,
259                                                     final Consumed<K, V> consumed,
260                                                     final Materialized<K, V, KeyValueStore<Bytes, byte[]>> materialized) {
261           Objects.requireNonNull(topic, "topic can't be null");
262           Objects.requireNonNull(consumed, "consumed can't be null");
263           Objects.requireNonNull(materialized, "materialized can't be null");
264           final ConsumedInternal<K, V> consumedInternal = new ConsumedInternal<>(consumed);
265           materialized.withKeySerde(consumedInternal.keySerde()).withValueSerde(consumedInternal.valueSerde());
266   
267           final MaterializedInternal<K, V, KeyValueStore<Bytes, byte[]>> materializedInternal =
268               new MaterializedInternal<>(materialized, internalStreamsBuilder, topic + "-");
269   
270           return internalStreamsBuilder.table(topic, consumedInternal, materializedInternal);
271       }
