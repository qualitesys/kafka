17   package org.apache.kafka.streams.kstream;
18   
19   import org.apache.kafka.streams.errors.TopologyException;
20   
21   import java.util.Objects;
22   
23   public class Named implements NamedOperation<Named> {
24   
25       private static final int MAX_NAME_LENGTH = 249;
26   
27       protected String name;
28   
29       protected Named(final Named named) {
30           this(Objects.requireNonNull(named, "named can't be null").name);
31       }
32   
33       protected Named(final String name) {
34           this.name = name;
35           if (name != null) {
36               validate(name);
37           }
38       }
39   
40       /**
41        * Create a Named instance with provided name.
42        *
43        * @param name  the processor name to be used. If {@code null} a default processor name will be generated.
44        * @return      A new {@link Named} instance configured with name
45        *
46        * @throws TopologyException if an invalid name is specified; valid characters are ASCII alphanumerics, '.', '_' and '-'.
47        */
48       public static Named as(final String name) {
49           Objects.requireNonNull(name, "name can't be null");
50           return new Named(name);
51       }
52   
53       @Override
54       public Named withName(final String name) {
55           return new Named(name);
56       }
57   
58       protected static void validate(final String name) {
59           if (name.isEmpty())
60               throw new TopologyException("Name is illegal, it can't be empty");
61           if (name.equals(".") || name.equals(".."))
62               throw new TopologyException("Name cannot be \".\" or \"..\"");
63           if (name.length() > MAX_NAME_LENGTH)
64               throw new TopologyException("Name is illegal, it can't be longer than " + MAX_NAME_LENGTH +
65                       " characters, name: " + name);
66           if (!containsValidPattern(name))
67               throw new TopologyException("Name \"" + name + "\" is illegal, it contains a character other than " +
68                       "ASCII alphanumerics, '.', '_' and '-'");
69       }
70   
71       /**
72        * Valid characters for Kafka topics are the ASCII alphanumerics, '.', '_', and '-'
73        */
74       private static boolean containsValidPattern(final String topic) {
75           for (int i = 0; i < topic.length(); ++i) {
76               final char c = topic.charAt(i);
77   
78               // We don't use Character.isLetterOrDigit(c) because it's slower
79               final boolean validLetterOrDigit = (c >= 'a' && c <= 'z') || (c >= '0' && c <= '9') || (c >= 'A' && c <= 'Z');
80               final boolean validChar = validLetterOrDigit || c == '.' || c == '_' || c == '-';
81               if (!validChar) {
82                   return false;
83               }
84           }
85           return true;
86       }
87   }
