18   package org.apache.kafka.streams.processor;
19   
20   import org.apache.kafka.common.TopicPartition;
21   
22   /**
23    * Class for listening to various states of the restoration process of a StateStore.
24    *
25    * When calling {@link org.apache.kafka.streams.KafkaStreams#setGlobalStateRestoreListener(StateRestoreListener)}
26    * the passed instance is expected to be stateless since the {@code StateRestoreListener} is shared
27    * across all {@link org.apache.kafka.streams.processor.internals.StreamThread} instances.
28    *
29    * Users desiring stateful operations will need to provide synchronization internally in
30    * the {@code StateRestorerListener} implementation.
31    *
32    * Note that this listener is only registered at the per-client level and users can base on the {@code storeName}
33    * parameter to define specific monitoring for different {@link StateStore}s. There is another
34    * {@link StateRestoreCallback} interface which is registered via the
35    * {@link StateStoreContext#register(StateStore, StateRestoreCallback, CommitCallback)}
36    * function per-store, and it is used to apply the fetched changelog records into the local state store during restoration.
37    * These two interfaces serve different restoration purposes and users should not try to implement both of them in a single
38    * class during state store registration.
39    *
40    * Incremental updates are exposed so users can estimate how much progress has been made.
41    */
42   public interface StateRestoreListener {
43   
44       /**
45        * Method called at the very beginning of {@link StateStore} restoration.
46        *
47        * @param topicPartition the TopicPartition containing the values to restore
48        * @param storeName      the name of the store undergoing restoration
49        * @param startingOffset the starting offset of the entire restoration process for this TopicPartition
50        * @param endingOffset   the exclusive ending offset of the entire restoration process for this TopicPartition
51        */
52       void onRestoreStart(final TopicPartition topicPartition,
53                           final String storeName,
54                           final long startingOffset,
55                           final long endingOffset);
56   
57       /**
58        * Method called after restoring a batch of records.  In this case the maximum size of the batch is whatever
59        * the value of the MAX_POLL_RECORDS is set to.
60        *
61        * This method is called after restoring each batch and it is advised to keep processing to a minimum.
62        * Any heavy processing will hold up recovering the next batch, hence slowing down the restore process as a
63        * whole.
64        *
65        * If you need to do any extended processing or connecting to an external service consider doing so asynchronously.
66        *
67        * @param topicPartition the TopicPartition containing the values to restore
68        * @param storeName the name of the store undergoing restoration
69        * @param batchEndOffset the inclusive ending offset for the current restored batch for this TopicPartition
70        * @param numRestored the total number of records restored in this batch for this TopicPartition
71        */
72       void onBatchRestored(final TopicPartition topicPartition,
73                            final String storeName,
74                            final long batchEndOffset,
75                            final long numRestored);
76   
77       /**
78        * Method called when restoring the {@link StateStore} is complete.
79        *
80        * @param topicPartition the TopicPartition containing the values to restore
81        * @param storeName the name of the store just restored
82        * @param totalRestored the total number of records restored for this TopicPartition
83        */
84       void onRestoreEnd(final TopicPartition topicPartition,
85                         final String storeName,
86                         final long totalRestored);
87   
88   }
