19   import org.apache.kafka.common.serialization.Deserializer;
20   import org.apache.kafka.common.serialization.Serde;

24   public class WindowedSerdes {
25   
26       static public class TimeWindowedSerde<T> extends Serdes.WrapperSerde<Windowed<T>> {
27           // Default constructor needed for reflection object creation
28           public TimeWindowedSerde() {
29               super(new TimeWindowedSerializer<>(), new TimeWindowedDeserializer<>());
30           }
31   
32           @Deprecated
33           public TimeWindowedSerde(final Serde<T> inner) {
34               super(new TimeWindowedSerializer<>(inner.serializer()), new TimeWindowedDeserializer<>(inner.deserializer()));
35           }
36   
37           // This constructor can be used for serialize/deserialize a windowed topic
38           public TimeWindowedSerde(final Serde<T> inner, final long windowSize) {
39               super(new TimeWindowedSerializer<>(inner.serializer()), new TimeWindowedDeserializer<>(inner.deserializer(), windowSize));
40           }
41   
42           // Helper method for users to specify whether the input topic is a changelog topic for deserializing the key properly.
43           public TimeWindowedSerde<T> forChangelog(final boolean isChangelogTopic) {
44               final TimeWindowedDeserializer deserializer = (TimeWindowedDeserializer) this.deserializer();
45               deserializer.setIsChangelogTopic(isChangelogTopic);
46               return this;
47           }
48       }
49   
50       static public class SessionWindowedSerde<T> extends Serdes.WrapperSerde<Windowed<T>> {
51           // Default constructor needed for reflection object creation
52           public SessionWindowedSerde() {
53               super(new SessionWindowedSerializer<>(), new SessionWindowedDeserializer<>());
54           }
55   
56           public SessionWindowedSerde(final Serde<T> inner) {
57               super(new SessionWindowedSerializer<>(inner.serializer()), new SessionWindowedDeserializer<>(inner.deserializer()));
58           }
59       }
60   
61       /**
62        * Construct a {@code TimeWindowedSerde} object for the specified inner class type.
63        */
64       @Deprecated
65       static public <T> Serde<Windowed<T>> timeWindowedSerdeFrom(final Class<T> type) {
66           return new TimeWindowedSerde<>(Serdes.serdeFrom(type));
67       }
68   
69       /**
70        * Construct a {@code TimeWindowedSerde} object to deserialize changelog topic
71        * for the specified inner class type and window size.
72        */
73       static public <T> Serde<Windowed<T>> timeWindowedSerdeFrom(final Class<T> type, final long windowSize) {
74           return new TimeWindowedSerde<>(Serdes.serdeFrom(type), windowSize);
75       }
76   
77       /**
78        * Construct a {@code SessionWindowedSerde} object for the specified inner class type.
79        */
80       static public <T> Serde<Windowed<T>> sessionWindowedSerdeFrom(final Class<T> type) {
81           return new SessionWindowedSerde<>(Serdes.serdeFrom(type));
82       }

84       static void verifyInnerSerializerNotNull(final Serializer inner,
85                                                final Serializer wrapper) {
86           if (inner == null) {
87               throw new NullPointerException("Inner serializer is `null`. " +
88                   "User code must use constructor `" + wrapper.getClass().getSimpleName() + "(final Serializer<T> inner)` " +
89                   "instead of the no-arg constructor.");
90           }
91       }
92   
93       static void verifyInnerDeserializerNotNull(final Deserializer inner,
94                                                  final Deserializer wrapper) {
95           if (inner == null) {
96               throw new NullPointerException("Inner deserializer is `null`. " +
97                   "User code must use constructor `" + wrapper.getClass().getSimpleName() + "(final Deserializer<T> inner)` " +
98                   "instead of the no-arg constructor.");
99           }
100       }
