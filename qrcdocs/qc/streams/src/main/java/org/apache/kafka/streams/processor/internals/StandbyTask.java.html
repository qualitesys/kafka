38   /**
39    * A StandbyTask
40    */
41   public class StandbyTask extends AbstractTask implements Task {
42       private final Sensor closeTaskSensor;
43       private final boolean eosEnabled;
44       private final InternalProcessorContext processorContext;
45       private final StreamsMetricsImpl streamsMetrics;
46   
47       /**
48        * @param id              the ID of this task
49        * @param inputPartitions input topic partitions, used for thread metadata only
50        * @param topology        the instance of {@link ProcessorTopology}
51        * @param config          the {@link StreamsConfig} specified by the user
52        * @param streamsMetrics  the {@link StreamsMetrics} created by the thread
53        * @param stateMgr        the {@link ProcessorStateManager} for this task
54        * @param stateDirectory  the {@link StateDirectory} created by the thread
55        */
56       StandbyTask(final TaskId id,
57                   final Set<TopicPartition> inputPartitions,
58                   final ProcessorTopology topology,
59                   final TaskConfig config,
60                   final StreamsMetricsImpl streamsMetrics,
61                   final ProcessorStateManager stateMgr,
62                   final StateDirectory stateDirectory,
63                   final ThreadCache cache,
64                   final InternalProcessorContext processorContext) {
65           super(
66               id,
67               topology,
68               stateDirectory,
69               stateMgr,
70               inputPartitions,
71               config.taskTimeoutMs,
72               "standby-task",
73               StandbyTask.class
74           );
75           this.processorContext = processorContext;
76           this.streamsMetrics = streamsMetrics;
77           processorContext.transitionToStandby(cache);
78   
79           closeTaskSensor = ThreadMetrics.closeTaskSensor(Thread.currentThread().getName(), streamsMetrics);
80           this.eosEnabled = config.eosEnabled;
81       }
82   
83       @Override
84       public boolean isActive() {
85           return false;
86       }
87   
88       /**
89        * @throws StreamsException fatal error, should close the thread
90        */
91       @Override
92       public void initializeIfNeeded() {
93           if (state() == State.CREATED) {
94               StateManagerUtil.registerStateStores(log, logPrefix, topology, stateMgr, stateDirectory, processorContext);
95   
96               // with and without EOS we would check for checkpointing at each commit during running,
97               // and the file may be deleted in which case we should checkpoint immediately,
98               // therefore we initialize the snapshot as empty
99               offsetSnapshotSinceLastFlush = Collections.emptyMap();
100   
101               // no topology needs initialized, we can transit to RUNNING
102               // right after registered the stores
103               transitionTo(State.RESTORING);
104               transitionTo(State.RUNNING);
105   
106               processorContext.initialize();
107   
108               log.info("Initialized");
109           } else if (state() == State.RESTORING) {
110               throw new IllegalStateException("Illegal state " + state() + " while initializing standby task " + id);
111           }
112       }
113   
114       @Override
115       public void completeRestoration(final java.util.function.Consumer<Set<TopicPartition>> offsetResetter) {
116           throw new IllegalStateException("Standby task " + id + " should never be completing restoration");
117       }
118   
119       @Override
120       public void suspend() {
121           switch (state()) {
122               case CREATED:
123                   log.info("Suspended created");
124                   transitionTo(State.SUSPENDED);
125   
126                   break;
127   
128               case RUNNING:
129                   log.info("Suspended running");
130                   transitionTo(State.SUSPENDED);
131   
132                   break;
133   
134               case SUSPENDED:
135                   log.info("Skip suspending since state is {}", state());
136   
137                   break;
138   
139               case RESTORING:
140               case CLOSED:
141                   throw new IllegalStateException("Illegal state " + state() + " while suspending standby task " + id);
142   
143               default:
144                   throw new IllegalStateException("Unknown state " + state() + " while suspending standby task " + id);
145           }
146       }
147   
148       @Override
149       public void resume() {
150           if (state() == State.RESTORING) {
151               throw new IllegalStateException("Illegal state " + state() + " while resuming standby task " + id);
152           }
153           log.trace("No-op resume with state {}", state());
154       }
155   
156       /**
157        * Flush stores before a commit; the following exceptions maybe thrown from the state manager flushing call
158        *
159        * @throws TaskMigratedException recoverable error sending changelog records that would cause the task to be removed
160        * @throws StreamsException fatal error when flushing the state store, for example sending changelog records failed
161        *                          or flushing state store get IO errors; such error should cause the thread to die
162        */
163       @Override
164       public Map<TopicPartition, OffsetAndMetadata> prepareCommit() {
165           switch (state()) {
166               case CREATED:
167                   log.debug("Skipped preparing created task for commit");
168   
169                   break;
170   
171               case RUNNING:
172               case SUSPENDED:
173                   // do not need to flush state store caches in pre-commit since nothing would be sent for standby tasks
174                   log.debug("Prepared {} task for committing", state());
175   
176                   break;
177   
178               default:
179                   throw new IllegalStateException("Illegal state " + state() + " while preparing standby task " + id + " for committing ");
180           }
181   
182           return Collections.emptyMap();
183       }
184   
185       @Override
186       public void postCommit(final boolean enforceCheckpoint) {
187           switch (state()) {
188               case CREATED:
189                   // We should never write a checkpoint for a CREATED task as we may overwrite an existing checkpoint
190                   // with empty uninitialized offsets
191                   log.debug("Skipped writing checkpoint for created task");
192   
193                   break;
194   
195               case RUNNING:
196               case SUSPENDED:
197                   maybeWriteCheckpoint(enforceCheckpoint);
198   
199                   log.debug("Finalized commit for {} task", state());
200   
201                   break;
202   
203               default:
204                   throw new IllegalStateException("Illegal state " + state() + " while post committing standby task " + id);
205           }
206       }

208       @Override
209       public void closeClean() {
210           streamsMetrics.removeAllTaskLevelSensors(Thread.currentThread().getName(), id.toString());
211           close(true);
212           log.info("Closed clean");
213       }

215       @Override
216       public void closeDirty() {
217           streamsMetrics.removeAllTaskLevelSensors(Thread.currentThread().getName(), id.toString());
218           close(false);
219           log.info("Closed dirty");
220       }
221   
222       @Override
223       public void closeCleanAndRecycleState() {
224           streamsMetrics.removeAllTaskLevelSensors(Thread.currentThread().getName(), id.toString());
225           if (state() == State.SUSPENDED) {
226               stateMgr.recycle();
227           } else {
228               throw new IllegalStateException("Illegal state " + state() + " while closing standby task " + id);
229           }
230   
231           closeTaskSensor.record();
232           transitionTo(State.CLOSED);
233   
234           log.info("Closed clean and recycled state");
235       }
236   
237       private void close(final boolean clean) {
238           switch (state()) {
239               case SUSPENDED:
240                   TaskManager.executeAndMaybeSwallow(
241                       clean,
242                       () -> StateManagerUtil.closeStateManager(
243                           log,
244                           logPrefix,
245                           clean,
246                           eosEnabled,
247                           stateMgr,
248                           stateDirectory,
249                           TaskType.STANDBY
250                       ),
251                       "state manager close",
252                       log
253                   );
254   
255                   break;
256   
257               case CLOSED:
258                   log.trace("Skip closing since state is {}", state());
259                   return;
260   
261               case CREATED:
262               case RESTORING: // a StandbyTask is never in RESTORING state
263               case RUNNING:
264                   throw new IllegalStateException("Illegal state " + state() + " while closing standby task " + id);
265   
266               default:
267                   throw new IllegalStateException("Unknown state " + state() + " while closing standby task " + id);
268           }
269   
270           closeTaskSensor.record();
271           transitionTo(State.CLOSED);
272       }

274       @Override
275       public boolean commitNeeded() {
276           // for standby tasks committing is the same as checkpointing,
277           // so we only need to commit if we want to checkpoint
278           return StateManagerUtil.checkpointNeeded(false, offsetSnapshotSinceLastFlush, stateMgr.changelogOffsets());
279       }

335           if (topology != null) {
336               sb.append(indent).append(topology.toString(indent + "\t"));
337           }
338

340       }
