17   package org.apache.kafka.streams.kstream;
18   
19   import java.time.Duration;
20   import org.apache.kafka.streams.KeyValue;
21   import org.apache.kafka.streams.processor.ProcessorContext;
22   import org.apache.kafka.streams.processor.PunctuationType;
23   import org.apache.kafka.streams.processor.Punctuator;
24   import org.apache.kafka.streams.processor.StateStore;
25   import org.apache.kafka.streams.processor.To;
26   
27   /**
28    * The {@code Transformer} interface is for stateful mapping of an input record to zero, one, or multiple new output
29    * records (both key and value type can be altered arbitrarily).
30    * This is a stateful record-by-record operation, i.e, {@link #transform(Object, Object)} is invoked individually for
31    * each record of a stream and can access and modify a state that is available beyond a single call of
32    * {@link #transform(Object, Object)} (cf. {@link KeyValueMapper} for stateless record transformation).
33    * Additionally, this {@code Transformer} can {@link ProcessorContext#schedule(Duration, PunctuationType, Punctuator) schedule}
34    * a method to be {@link Punctuator#punctuate(long) called periodically} with the provided context.
35    * <p>
36    * Use {@link TransformerSupplier} to provide new instances of {@code Transformer} to Kafka Stream's runtime.
37    * <p>
38    * If only a record's value should be modified {@link ValueTransformer} can be used.
39    *
40    * @param <K> key type
41    * @param <V> value type
42    * @param <R> {@link KeyValue} return type (both key and value type can be set
43    *            arbitrarily)
44    * @see TransformerSupplier
45    * @see KStream#transform(TransformerSupplier, String...)
46    * @see ValueTransformer
47    * @see KStream#map(KeyValueMapper)
48    * @see KStream#flatMap(KeyValueMapper)
49    */
50   public interface Transformer<K, V, R> {
51   
52       /**
53        * Initialize this transformer.
54        * This is called once per instance when the topology gets initialized.
55        * When the framework is done with the transformer, {@link #close()} will be called on it; the
56        * framework may later re-use the transformer by calling {@link #init(ProcessorContext)} again.
57        * <p>
58        * The provided {@link ProcessorContext context} can be used to access topology and record meta data, to
59        * {@link ProcessorContext#schedule(Duration, PunctuationType, Punctuator) schedule} a method to be
60        * {@link Punctuator#punctuate(long) called periodically} and to access attached {@link StateStore}s.
61        * <p>
62        * Note, that {@link ProcessorContext} is updated in the background with the current record's meta data.
63        * Thus, it only contains valid record meta data when accessed within {@link #transform(Object, Object)}.
64        *
65        * @param context the context
66        */
67       void init(final ProcessorContext context);
68   
69       /**
70        * Transform the record with the given key and value.
71        * Additionally, any {@link StateStore state} that is {@link KStream#transform(TransformerSupplier, String...)
72        * attached} to this operator can be accessed and modified
73        * arbitrarily (cf. {@link ProcessorContext#getStateStore(String)}).
74        * <p>
75        * If only one record should be forward downstream, {@code transform} can return a new {@link KeyValue}. If
76        * more than one output record should be forwarded downstream, {@link ProcessorContext#forward(Object, Object)}
77        * and {@link ProcessorContext#forward(Object, Object, To)} can be used.
78        * If no record should be forwarded downstream, {@code transform} can return {@code null}.
79        *
80        * Note that returning a new {@link KeyValue} is merely for convenience. The same can be achieved by using
81        * {@link ProcessorContext#forward(Object, Object)} and returning {@code null}.
82        *
83        * @param key the key for the record
84        * @param value the value for the record
85        * @return new {@link KeyValue} pair&mdash;if {@code null} no key-value pair will
86        * be forwarded to down stream
87        */
88       R transform(final K key, final V value);
89   
90       /**
91        * Close this transformer and clean up any resources. The framework may
92        * later re-use this transformer by calling {@link #init(ProcessorContext)} on it again.
93        * <p>
94        * To generate new {@link KeyValue} pairs {@link ProcessorContext#forward(Object, Object)} and
95        * {@link ProcessorContext#forward(Object, Object, To)} can be used.
96        */
97       void close();
98   
99   }
