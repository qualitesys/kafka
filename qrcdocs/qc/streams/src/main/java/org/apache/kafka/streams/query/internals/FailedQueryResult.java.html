17   package org.apache.kafka.streams.query.internals;
18   
19   
20   import org.apache.kafka.streams.query.FailureReason;
21   import org.apache.kafka.streams.query.QueryResult;
22   import org.apache.kafka.streams.query.StateQueryRequest;
23   
24   /**
25    * Container for a single partition's result when executing a {@link StateQueryRequest}.
26    *
27    * @param <R> The result type of the query.
28    */
29   public final class FailedQueryResult<R>
30       extends AbstractQueryResult<R>
31       implements QueryResult<R> {
32   
33       private final FailureReason failureReason;
34       private final String failure;
35   
36       public FailedQueryResult(final FailureReason failureReason, final String failure) {
37           this.failureReason = failureReason;
38           this.failure = failure;
39       }
40   
41       /**
42        * True iff the query was successfully executed. The response is available in {@link
43        * this#getResult()}.
44        */
45       public boolean isSuccess() {
46           return false;
47       }
48   
49   
50       /**
51        * True iff the query execution failed. More information about the failure is available in
52        * {@link this#getFailureReason()} and {@link this#getFailureMessage()}.
53        */
54       public boolean isFailure() {
55           return true;
56       }
57   
58       /**
59        * If this partition failed to execute the query, returns the reason.
60        *
61        * @throws IllegalArgumentException if this is not a failed result.
62        */
63       public FailureReason getFailureReason() {
64           return failureReason;
65       }
66   
67       /**
68        * If this partition failed to execute the query, returns the failure message.
69        *
70        * @throws IllegalArgumentException if this is not a failed result.
71        */
72       public String getFailureMessage() {
73           return failure;
74       }
75   
76       /**
77        * Returns the result of executing the query on one partition. The result type is determined by
78        * the query. Note: queries may choose to return {@code null} for a successful query, so {@link
79        * this#isSuccess()} and {@link this#isFailure()} must be used to determine whether the query
80        * was successful of failed on this partition.
81        *
82        * @throws IllegalArgumentException if this is not a successful query.
83        */
84       public R getResult() {
85           throw new IllegalArgumentException(
86               "Cannot get result for failed query. Failure is " + failureReason.name() + ": "
87                   + failure);
88       }
89   
90       @Override
91       public String toString() {
92           return "FailedQueryResult{" +
93               "failureReason=" + failureReason +
94               ", failure='" + failure + '\'' +
95               ", executionInfo=" + getExecutionInfo() +
96               ", position=" + getPosition() +
97               '}';
98       }
99   }
