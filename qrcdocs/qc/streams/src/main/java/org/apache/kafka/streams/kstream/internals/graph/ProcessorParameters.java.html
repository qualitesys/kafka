18   package org.apache.kafka.streams.kstream.internals.graph;
19   
20   import org.apache.kafka.streams.kstream.internals.KTableKTableJoinMerger;
21   import org.apache.kafka.streams.kstream.internals.KTableProcessorSupplier;
22   import org.apache.kafka.streams.kstream.internals.KTableSource;
23   import org.apache.kafka.streams.processor.api.FixedKeyProcessorSupplier;
24   import org.apache.kafka.streams.processor.api.ProcessorSupplier;
25   import org.apache.kafka.streams.processor.internals.InternalTopologyBuilder;
26   import org.apache.kafka.streams.processor.internals.ProcessorAdapter;
27   import org.apache.kafka.streams.state.StoreBuilder;
28   
29   /**
30    * Class used to represent a {@link ProcessorSupplier} or {@link FixedKeyProcessorSupplier} and the name
31    * used to register it with the {@link org.apache.kafka.streams.processor.internals.InternalTopologyBuilder}
32    *
33    * Used by the Join nodes as there are several parameters, this abstraction helps
34    * keep the number of arguments more reasonable.
35    *
36    * @see ProcessorSupplier
37    * @see FixedKeyProcessorSupplier
38    */
39   public class ProcessorParameters<KIn, VIn, KOut, VOut> {
40   
41       // During the transition to KIP-478, we capture arguments passed from the old API to simplify
42       // the performance of casts that we still need to perform. This will eventually be removed.
43       @SuppressWarnings("deprecation") // Old PAPI. Needs to be migrated.
44       private final org.apache.kafka.streams.processor.ProcessorSupplier<KIn, VIn> oldProcessorSupplier;
45       private final ProcessorSupplier<KIn, VIn, KOut, VOut> processorSupplier;
46       private final FixedKeyProcessorSupplier<KIn, VIn, VOut> fixedKeyProcessorSupplier;
47       private final String processorName;
48   
49       @SuppressWarnings("deprecation") // Old PAPI compatibility.
50       public ProcessorParameters(final org.apache.kafka.streams.processor.ProcessorSupplier<KIn, VIn> processorSupplier,
51                                  final String processorName) {
52           oldProcessorSupplier = processorSupplier;
53           this.processorSupplier = () -> ProcessorAdapter.adapt(processorSupplier.get());
54           fixedKeyProcessorSupplier = null;
55           this.processorName = processorName;
56       }
57   
58       public ProcessorParameters(final ProcessorSupplier<KIn, VIn, KOut, VOut> processorSupplier,
59                                  final String processorName) {
60           oldProcessorSupplier = null;
61           this.processorSupplier = processorSupplier;
62           fixedKeyProcessorSupplier = null;
63           this.processorName = processorName;
64       }
65   
66       public ProcessorParameters(final FixedKeyProcessorSupplier<KIn, VIn, VOut> processorSupplier,
67                                  final String processorName) {
68           oldProcessorSupplier = null;
69           this.processorSupplier = null;
70           fixedKeyProcessorSupplier = processorSupplier;
71           this.processorName = processorName;
72       }
73   
74       public ProcessorSupplier<KIn, VIn, KOut, VOut> processorSupplier() {
75           return processorSupplier;
76       }
77   
78       public FixedKeyProcessorSupplier<KIn, VIn, VOut> fixedKeyProcessorSupplier() {
79           return fixedKeyProcessorSupplier;
80       }
81   
82       public void addProcessorTo(final InternalTopologyBuilder topologyBuilder, final String[] parentNodeNames) {
83           if (processorSupplier != null) {
84               topologyBuilder.addProcessor(processorName, processorSupplier, parentNodeNames);
85               if (processorSupplier.stores() != null) {
86                   for (final StoreBuilder<?> storeBuilder : processorSupplier.stores()) {
87                       topologyBuilder.addStateStore(storeBuilder, processorName);
88                   }
89               }
90           }
91   
92           if (fixedKeyProcessorSupplier != null) {
93               topologyBuilder.addProcessor(processorName, fixedKeyProcessorSupplier, parentNodeNames);
94               if (fixedKeyProcessorSupplier.stores() != null) {
95                   for (final StoreBuilder<?> storeBuilder : fixedKeyProcessorSupplier.stores()) {
96                       topologyBuilder.addStateStore(storeBuilder, processorName);
97                   }
98               }
99           }
100   
101           // temporary hack until KIP-478 is fully implemented
102           // Old PAPI. Needs to be migrated.
103           if (oldProcessorSupplier != null && oldProcessorSupplier.stores() != null) {
104               for (final StoreBuilder<?> storeBuilder : oldProcessorSupplier.stores()) {
105                   topologyBuilder.addStateStore(storeBuilder, processorName);
106               }
107           }
108       }
109   
110       @SuppressWarnings("unchecked")
111       KTableSource<KIn, VIn> kTableSourceSupplier() {
112           return processorSupplier instanceof KTableSource ? (KTableSource<KIn, VIn>) processorSupplier : null;
113       }
114   
115       @SuppressWarnings("unchecked")
116       <KR, VR> KTableProcessorSupplier<KIn, VIn, KR, VR> kTableProcessorSupplier() {
117           // This cast always works because KTableProcessorSupplier hasn't been converted yet.
118           return (KTableProcessorSupplier<KIn, VIn, KR, VR>) processorSupplier;
119       }
120   
121       @SuppressWarnings("unchecked")
122       KTableKTableJoinMerger<KIn, VIn> kTableKTableJoinMergerProcessorSupplier() {
123           return (KTableKTableJoinMerger<KIn, VIn>) processorSupplier;
124       }
125   
126       public String processorName() {
127           return processorName;
128       }
129   
130       @Override
131       public String toString() {
132           return "ProcessorParameters{" +
133               "processor class=" + processorSupplier.get().getClass() +
134               ", processor name='" + processorName + '\'' +
135               '}';
136       }
137   }
