17   package org.apache.kafka.streams.kstream.internals;
18   
19   import org.apache.kafka.common.serialization.Serde;
20   import org.apache.kafka.streams.kstream.Materialized;
21   import org.apache.kafka.streams.processor.StateStore;
22   import org.apache.kafka.streams.TopologyConfig;
23   import org.apache.kafka.streams.state.StoreSupplier;
24   
25   import java.time.Duration;
26   import java.util.Map;
27   
28   public class MaterializedInternal<K, V, S extends StateStore> extends Materialized<K, V, S> {
29   
30       private final boolean queryable;
31   
32       public MaterializedInternal(final Materialized<K, V, S> materialized) {
33           this(materialized, null, null);
34       }
35   
36       public MaterializedInternal(final Materialized<K, V, S> materialized,
37                                   final InternalNameProvider nameProvider,
38                                   final String generatedStorePrefix) {
39           super(materialized);
40   
41           // if storeName is not provided, the corresponding KTable would never be queryable;
42           // but we still need to provide an internal name for it in case we materialize.
43           queryable = storeName() != null;
44           if (!queryable && nameProvider != null) {
45               storeName = nameProvider.newStoreName(generatedStorePrefix);
46           }
47   
48           // if store type is not configured during creating Materialized, then try to get the topologyConfigs from nameProvider
49           // otherwise, set to default rocksDB
50           if (storeType == null) {
51               storeType = StoreType.ROCKS_DB;
52               if (nameProvider instanceof InternalStreamsBuilder) {
53                   final TopologyConfig topologyConfig = ((InternalStreamsBuilder) nameProvider).internalTopologyBuilder.topologyConfigs();
54                   if (topologyConfig != null) {
55                       storeType = topologyConfig.parseStoreType();
56                   }
57               }
58           }
59       }
60   
61       public String queryableStoreName() {
62           return queryable ? storeName() : null;
63       }
64   
65       public String storeName() {
66           if (storeSupplier != null) {
67               return storeSupplier.name();
68           }
69           return storeName;
70       }
71   
72       public StoreType storeType() {
73           return storeType;
74       }
75   
76       public StoreSupplier<S> storeSupplier() {
77           return storeSupplier;
78       }
79   
80       public Serde<K> keySerde() {
81           return keySerde;
82       }
83   
84       public Serde<V> valueSerde() {
85           return valueSerde;
86       }
87   
88       public boolean loggingEnabled() {
89           return loggingEnabled;
90       }
91   
92       Map<String, String> logConfig() {
93           return topicConfig;
94       }
95   
96       public boolean cachingEnabled() {
97           return cachingEnabled;
98       }
99   
100       Duration retention() {
101           return retention;
102       }
103   }
