385       @SuppressWarnings("unchecked")
386       @Override
387       public <R> QueryResult<R> query(final Query<R> query,
388                                       final PositionBound positionBound,
389                                       final QueryConfig config) {
390           final long start = time.nanoseconds();
391           final QueryResult<R> result;
392   
393           final QueryHandler handler = queryHandlers.get(query.getClass());
394           if (handler == null) {
395               result = wrapped().query(query, positionBound, config);
396               if (config.isCollectExecutionInfo()) {
397                   result.addExecutionInfo(
398                       "Handled in " + getClass() + " in " + (time.nanoseconds() - start) + "ns");
399               }
400           } else {
401               result = (QueryResult<R>) handler.apply(
402                   query,
403                   positionBound,
404                   config,
405                   this
406               );
407               if (config.isCollectExecutionInfo()) {
408                   result.addExecutionInfo(
409                       "Handled in " + getClass() + " with serdes "
410                           + serdes + " in " + (time.nanoseconds() - start) + "ns");
411               }
412           }
413           return result;
414       }

416       @SuppressWarnings("unchecked")
417       private <R> QueryResult<R> runRangeQuery(final Query<R> query,
418                                                final PositionBound positionBound,
419                                                final QueryConfig config) {
420           final QueryResult<R> result;
421           final WindowRangeQuery<K, V> typedQuery = (WindowRangeQuery<K, V>) query;
422           if (typedQuery.getKey().isPresent()) {
423               final WindowRangeQuery<Bytes, byte[]> rawKeyQuery =
424                   WindowRangeQuery.withKey(
425                       Bytes.wrap(serdes.rawKey(typedQuery.getKey().get()))
426                   );
427               final QueryResult<KeyValueIterator<Windowed<Bytes>, byte[]>> rawResult =
428                   wrapped().query(rawKeyQuery, positionBound, config);
429               if (rawResult.isSuccess()) {
430                   final MeteredWindowedKeyValueIterator<K, V> typedResult =
431                       new MeteredWindowedKeyValueIterator<>(
432                           rawResult.getResult(),
433                           fetchSensor,
434                           streamsMetrics,
435                           serdes::keyFrom,
436                           StoreQueryUtils.getDeserializeValue(serdes, wrapped()),
437                           time
438                       );
439                   final QueryResult<MeteredWindowedKeyValueIterator<K, V>> typedQueryResult =
440                       InternalQueryResultUtil.copyAndSubstituteDeserializedResult(rawResult, typedResult);
441                   result = (QueryResult<R>) typedQueryResult;
442               } else {
443                   // the generic type doesn't matter, since failed queries have no result set.
444                   result = (QueryResult<R>) rawResult;
445               }
446           } else {
447   
448               result = QueryResult.forFailure(
449                   FailureReason.UNKNOWN_QUERY_TYPE,
450                   "This store (" + getClass() + ") doesn't know how to"
451                       + " execute the given query (" + query + ") because"
452                       + " SessionStores only support WindowRangeQuery.withKey."
453                       + " Contact the store maintainer if you need support"
454                       + " for a new query type."
455               );
456           }
457           return result;
458       }

464       private void maybeRecordE2ELatency() {
465           // Context is null if the provided context isn't an implementation of InternalProcessorContext.
466           // In that case, we _can't_ get the current timestamp, so we don't record anything.
467           if (e2eLatencySensor.shouldRecord() && context != null) {
468               final long currentTime = time.milliseconds();
469               final long e2eLatency =  currentTime - context.timestamp();
470               e2eLatencySensor.record(e2eLatency, currentTime);
471           }
472       }
