17   package org.apache.kafka.streams.state;
18   
19   import org.apache.kafka.common.serialization.Deserializer;
20   import org.apache.kafka.common.serialization.Serde;
21   import org.apache.kafka.common.serialization.Serdes;
22   import org.apache.kafka.common.serialization.Serializer;
23   import org.apache.kafka.streams.errors.StreamsException;
24   import org.apache.kafka.streams.state.internals.ValueAndTimestampSerializer;
25   
26   import java.util.Objects;
27   
28   /**
29    * Factory for creating serializers / deserializers for state stores in Kafka Streams.
30    *
31    * @param <K> key type of serde
32    * @param <V> value type of serde
33    */
34   public final class StateSerdes<K, V> {
35   
36       public static final int TIMESTAMP_SIZE = 8;
37       public static final int BOOLEAN_SIZE = 1;
38   
39       /**
40        * Create a new instance of {@link StateSerdes} for the given state name and key-/value-type classes.
41        *
42        * @param topic      the topic name
43        * @param keyClass   the class of the key type
44        * @param valueClass the class of the value type
45        * @param <K>        the key type
46        * @param <V>        the value type
47        * @return a new instance of {@link StateSerdes}
48        */
49       public static <K, V> StateSerdes<K, V> withBuiltinTypes(
50           final String topic,
51           final Class<K> keyClass,
52           final Class<V> valueClass) {
53           return new StateSerdes<>(topic, Serdes.serdeFrom(keyClass), Serdes.serdeFrom(valueClass));
54       }
55   
56       private final String topic;
57       private final Serde<K> keySerde;
58       private final Serde<V> valueSerde;
59   
60       /**
61        * Create a context for serialization using the specified serializers and deserializers which
62        * <em>must</em> match the key and value types used as parameters for this object; the state changelog topic
63        * is provided to bind this serde factory to, so that future calls for serialize / deserialize do not
64        * need to provide the topic name any more.
65        *
66        * @param topic         the topic name
67        * @param keySerde      the serde for keys; cannot be null
68        * @param valueSerde    the serde for values; cannot be null
69        * @throws IllegalArgumentException if key or value serde is null
70        */
71       public StateSerdes(final String topic,
72                          final Serde<K> keySerde,
73                          final Serde<V> valueSerde) {
74           Objects.requireNonNull(topic, "topic cannot be null");
75           Objects.requireNonNull(keySerde, "key serde cannot be null");
76           Objects.requireNonNull(valueSerde, "value serde cannot be null");
77   
78           this.topic = topic;
79           this.keySerde = keySerde;
80           this.valueSerde = valueSerde;
81       }
82   
83       /**
84        * Return the key serde.
85        *
86        * @return the key serde
87        */
88       public Serde<K> keySerde() {
89           return keySerde;
90       }
91   
92       /**
93        * Return the value serde.
94        *
95        * @return the value serde
96        */
97       public Serde<V> valueSerde() {
98           return valueSerde;
99       }
100   
101       /**
102        * Return the key deserializer.
103        *
104        * @return the key deserializer
105        */
106       public Deserializer<K> keyDeserializer() {
107           return keySerde.deserializer();
108       }
109   
110       /**
111        * Return the key serializer.
112        *
113        * @return the key serializer
114        */
115       public Serializer<K> keySerializer() {
116           return keySerde.serializer();
117       }
118   
119       /**
120        * Return the value deserializer.
121        *
122        * @return the value deserializer
123        */
124       public Deserializer<V> valueDeserializer() {
125           return valueSerde.deserializer();
126       }
127   
128       /**
129        * Return the value serializer.
130        *
131        * @return the value serializer
132        */
133       public Serializer<V> valueSerializer() {
134           return valueSerde.serializer();
135       }
136   
137       /**
138        * Return the topic.
139        *
140        * @return the topic
141        */
142       public String topic() {
143           return topic;
144       }
145   
146       /**
147        * Deserialize the key from raw bytes.
148        *
149        * @param rawKey  the key as raw bytes
150        * @return        the key as typed object
151        */
152       public K keyFrom(final byte[] rawKey) {
153           return keySerde.deserializer().deserialize(topic, rawKey);
154       }
155   
156       /**
157        * Deserialize the value from raw bytes.
158        *
159        * @param rawValue  the value as raw bytes
160        * @return          the value as typed object
161        */
162       public V valueFrom(final byte[] rawValue) {
163           return valueSerde.deserializer().deserialize(topic, rawValue);
164       }
165   
166       /**
167        * Serialize the given key.
168        *
169        * @param key  the key to be serialized
170        * @return     the serialized key
171        */
172       public byte[] rawKey(final K key) {
173           try {
174               return keySerde.serializer().serialize(topic, key);
175           } catch (final ClassCastException e) {
176               final String keyClass = key == null ? "unknown because key is null" : key.getClass().getName();
177               throw new StreamsException(
178                       String.format("A serializer (%s) is not compatible to the actual key type " +
179                                       "(key type: %s). Change the default Serdes in StreamConfig or " +
180                                       "provide correct Serdes via method parameters.",
181                               keySerializer().getClass().getName(),
182                               keyClass),
183                       e);
184           }
185       }
186   
187       /**
188        * Serialize the given value.
189        *
190        * @param value  the value to be serialized
191        * @return       the serialized value
192        */
193       public byte[] rawValue(final V value) {
194           try {
195               return valueSerde.serializer().serialize(topic, value);
196           } catch (final ClassCastException e) {
197               final String valueClass;
198               final Class<? extends Serializer> serializerClass;
199               if (valueSerializer() instanceof ValueAndTimestampSerializer) {
200                   serializerClass = ((ValueAndTimestampSerializer) valueSerializer()).valueSerializer.getClass();
201                   valueClass = value == null ? "unknown because value is null" : ((ValueAndTimestamp) value).value().getClass().getName();
202               } else {
203                   serializerClass = valueSerializer().getClass();
204                   valueClass = value == null ? "unknown because value is null" : value.getClass().getName();
205               }
206               throw new StreamsException(
207                       String.format("A serializer (%s) is not compatible to the actual value type " +
208                                       "(value type: %s). Change the default Serdes in StreamConfig or " +
209                                       "provide correct Serdes via method parameters.",
210                               serializerClass.getName(),
211                               valueClass),
212                       e);
213           }
214       }
215   }
