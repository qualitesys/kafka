17   package org.apache.kafka.streams.kstream.internals;
18   
19   import org.apache.kafka.streams.kstream.Windowed;
20   import org.apache.kafka.streams.processor.StateStore;
21   import org.apache.kafka.streams.processor.api.ProcessorContext;
22   import org.apache.kafka.streams.processor.api.Record;
23   import org.apache.kafka.streams.state.internals.CacheFlushListener;
24   import org.apache.kafka.streams.state.internals.WrappedStateStore;
25   
26   /**
27    * This class is used to determine if a processor should forward values to child nodes.
28    * Forwarding by this class only occurs when caching is not enabled. If caching is enabled,
29    * forwarding occurs in the flush listener when the cached store flushes.
30    *
31    * @param <K>
32    * @param <V>
33    */
34   class SessionTupleForwarder<K, V> {
35       private final ProcessorContext<Windowed<K>, Change<V>> context;
36       private final boolean sendOldValues;
37       private final boolean cachingEnabled;
38   
39       @SuppressWarnings("unchecked")
40       SessionTupleForwarder(final StateStore store,
41                             final ProcessorContext<Windowed<K>, Change<V>> context,
42                             final CacheFlushListener<Windowed<K>, V> flushListener,
43                             final boolean sendOldValues) {
44           this.context = context;
45           this.sendOldValues = sendOldValues;
46           cachingEnabled = ((WrappedStateStore) store).setFlushListener(flushListener, sendOldValues);
47       }
48   
49       public void maybeForward(final Record<Windowed<K>, Change<V>> record) {
50           if (!cachingEnabled) {
51               context.forward(
52                   record.withValue(new Change<>(record.value().newValue, sendOldValues ? record.value().oldValue : null))
53                       .withTimestamp(record.key() != null ? record.key().window().end() : record.timestamp()));
54           }
55       }
56   }
