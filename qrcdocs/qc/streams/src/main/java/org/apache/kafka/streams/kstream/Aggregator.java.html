20   /**
21    * The {@code Aggregator} interface for aggregating values of the given key.
22    * This is a generalization of {@link Reducer} and allows to have different types for input value and aggregation
23    * result.
24    * {@code Aggregator} is used in combination with {@link Initializer} that provides an initial aggregation value.
25    * <p>
26    * {@code Aggregator} can be used to implement aggregation functions like count.
27   
28    * @param <K> key type
29    * @param <V> input value type
30    * @param <VA> aggregate value type
31    * @see Initializer
32    * @see KGroupedStream#aggregate(Initializer, Aggregator)
33    * @see KGroupedStream#aggregate(Initializer, Aggregator, Materialized)
34    * @see TimeWindowedKStream#aggregate(Initializer, Aggregator)
35    * @see TimeWindowedKStream#aggregate(Initializer, Aggregator, Materialized)
36    * @see SessionWindowedKStream#aggregate(Initializer, Aggregator, Merger)
37    * @see SessionWindowedKStream#aggregate(Initializer, Aggregator, Merger, Materialized)
38    * @see Reducer
39    */
40   public interface Aggregator<K, V, VA> {
41   
42       /**
43        * Compute a new aggregate from the key and value of a record and the current aggregate of the same key.
44        *
45        * @param key       the key of the record
46        * @param value     the value of the record
47        * @param aggregate the current aggregate value
48        * @return the new aggregate value
49        */
50       VA apply(final K key, final V value, final VA aggregate);
51   }
