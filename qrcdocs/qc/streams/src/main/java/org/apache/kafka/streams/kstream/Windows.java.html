19   import org.apache.kafka.streams.processor.TimestampExtractor;
20   
21   import java.util.Map;
22   
23   /**
24    * The window specification for fixed size windows that is used to define window boundaries and grace period.
25    * <p>
26    * Grace period defines how long to wait on out-of-order events. That is, windows will continue to accept new records until {@code stream_time >= window_end + grace_period}.
27    * Records that arrive after the grace period passed are considered <em>late</em> and will not be processed but are dropped.
28    * <p>
29    * Warning: It may be unsafe to use objects of this class in set- or map-like collections,
30    * since the equals and hashCode methods depend on mutable fields.
31    *
32    * @param <W> type of the window instance
33    * @see TimeWindows
34    * @see UnlimitedWindows
35    * @see JoinWindows
36    * @see SessionWindows
37    * @see TimestampExtractor
38    */
39   public abstract class Windows<W extends Window> {
40   
41       /**
42        * By default grace period is 24 hours for all windows in other words we allow out-of-order data for up to a day
43        * This behavior is now deprecated and additional details are available in the motivation for the KIP
44        * Check out <a href="https://cwiki.apache.org/confluence/x/Ho2NCg">KIP-633</a> for more details
45        */
46       protected static final long DEPRECATED_DEFAULT_24_HR_GRACE_PERIOD = 24 * 60 * 60 * 1000L;
47   
48       /**
49        * This constant is used as the specified grace period where we do not have any grace periods instead of magic constants
50        */
51       protected static final long NO_GRACE_PERIOD = 0L;
52   
53       protected Windows() {}
54   
55       /**
56        * Create all windows that contain the provided timestamp, indexed by non-negative window start timestamps.
57        *
58        * @param timestamp the timestamp window should get created for
59        * @return a map of {@code windowStartTimestamp -> Window} entries
60        */
61       public abstract Map<Long, W> windowsFor(final long timestamp);
62   
63       /**
64        * Return the size of the specified windows in milliseconds.
65        *
66        * @return the size of the specified windows
67        */
68       public abstract long size();
69   
70       /**
71        * Return the window grace period (the time to admit
72        * out-of-order events after the end of the window.)
73        *
74        * Delay is defined as (stream_time - record_timestamp).
75        */
76       public abstract long gracePeriodMs();
77   }
