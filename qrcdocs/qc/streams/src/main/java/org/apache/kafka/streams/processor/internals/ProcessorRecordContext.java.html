78       public long residentMemorySizeEstimate() {
79           long size = 0;
80           size += Long.BYTES; // value.context.timestamp
81           size += Long.BYTES; // value.context.offset
82           if (topic != null) {
83               size += topic.toCharArray().length;
84           }
85           size += Integer.BYTES; // partition
86           for (final Header header : headers) {
87               size += header.key().toCharArray().length;
88               final byte[] value = header.value();
89               if (value != null) {
90                   size += value.length;
91               }
92           }
93           return size;
94       }

96       public byte[] serialize() {
97           final byte[] topicBytes = topic.getBytes(UTF_8);
98           final byte[][] headerKeysBytes;
99           final byte[][] headerValuesBytes;
100   
101           int size = 0;
102           size += Long.BYTES; // value.context.timestamp
103           size += Long.BYTES; // value.context.offset
104           size += Integer.BYTES; // size of topic
105           size += topicBytes.length;
106           size += Integer.BYTES; // partition
107           size += Integer.BYTES; // number of headers
108   
109           final Header[] headers = this.headers.toArray();
110           headerKeysBytes = new byte[headers.length][];
111           headerValuesBytes = new byte[headers.length][];
112   
113           for (int i = 0; i < headers.length; i++) {
114               size += 2 * Integer.BYTES; // sizes of key and value
115   
116               final byte[] keyBytes = headers[i].key().getBytes(UTF_8);
117               size += keyBytes.length;
118               final byte[] valueBytes = headers[i].value();
119               if (valueBytes != null) {
120                   size += valueBytes.length;
121               }
122   
123               headerKeysBytes[i] = keyBytes;
124               headerValuesBytes[i] = valueBytes;
125           }
126   
127           final ByteBuffer buffer = ByteBuffer.allocate(size);
128           buffer.putLong(timestamp);
129           buffer.putLong(offset);
130   
131           // not handling the null condition because we believe topic will never be null in cases where we serialize
132           buffer.putInt(topicBytes.length);
133           buffer.put(topicBytes);
134   
135           buffer.putInt(partition);
136           buffer.putInt(headerKeysBytes.length);
137           for (int i = 0; i < headerKeysBytes.length; i++) {
138               buffer.putInt(headerKeysBytes[i].length);
139               buffer.put(headerKeysBytes[i]);
140   
141               if (headerValuesBytes[i] != null) {
142                   buffer.putInt(headerValuesBytes[i].length);
143                   buffer.put(headerValuesBytes[i]);
144               } else {
145                   buffer.putInt(-1);
146               }
147           }
148   
149           return buffer.array();
150       }
151   
152       public static ProcessorRecordContext deserialize(final ByteBuffer buffer) {
153           final long timestamp = buffer.getLong();
154           final long offset = buffer.getLong();
155           final String topic;
156           {
157               // we believe the topic will never be null when we serialize
158               final byte[] topicBytes = requireNonNull(getNullableSizePrefixedArray(buffer));
159               topic = new String(topicBytes, UTF_8);
160           }
161           final int partition = buffer.getInt();
162           final int headerCount = buffer.getInt();
163           final Headers headers;
164           if (headerCount == -1) { // keep for backward compatibilty
165               headers = new RecordHeaders();
166           } else {
167               final Header[] headerArr = new Header[headerCount];
168               for (int i = 0; i < headerCount; i++) {
169                   final byte[] keyBytes = requireNonNull(getNullableSizePrefixedArray(buffer));
170                   final byte[] valueBytes = getNullableSizePrefixedArray(buffer);
171                   headerArr[i] = new RecordHeader(new String(keyBytes, UTF_8), valueBytes);
172               }
173               headers = new RecordHeaders(headerArr);
174           }
175   
176           return new ProcessorRecordContext(timestamp, offset, partition, topic, headers);
177       }

185               return false;

188           return timestamp == that.timestamp &&
189               offset == that.offset &&
190               partition == that.partition &&
191               Objects.equals(topic, that.topic) &&
192               Objects.equals(headers, that.headers);
193       }
