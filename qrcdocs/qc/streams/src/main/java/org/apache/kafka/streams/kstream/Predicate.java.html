19   import org.apache.kafka.streams.KeyValue;

21   /**
22    * The {@code Predicate} interface represents a predicate (boolean-valued function) of a {@link KeyValue} pair.
23    * This is a stateless record-by-record operation, i.e, {@link #test(Object, Object)} is invoked individually for each
24    * record of a stream.
25    *
26    * @param <K> key type
27    * @param <V> value type
28    * @see KStream#filter(Predicate)
29    * @see KStream#filterNot(Predicate)
30    * @see BranchedKStream#branch(Predicate)
31    * @see KTable#filter(Predicate)
32    * @see KTable#filterNot(Predicate)
33    */
34   public interface Predicate<K, V> {
35   
36       /**
37        * Test if the record with the given key and value satisfies the predicate.
38        *
39        * @param key   the key of the record
40        * @param value the value of the record
41        * @return {@code true} if the {@link KeyValue} pair satisfies the predicate&mdash;{@code false} otherwise
42        */
43       boolean test(final K key, final V value);
44   }
