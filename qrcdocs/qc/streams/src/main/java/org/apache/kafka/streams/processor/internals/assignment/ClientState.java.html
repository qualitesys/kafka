306       private void maybeFilterUnknownPrevTasksAndPartitions(final Map<TopicPartition, TaskId> taskForPartitionMap,
307                                                             final boolean hasNamedTopologies) {
308           // If this application uses named topologies, then it's possible for members to report tasks
309           // or partitions in their subscription that belong to a named topology that the group leader
310           // doesn't currently recognize, eg because it was just removed
311           if (hasNamedTopologies) {
312               ownedPartitions.keySet().retainAll(taskForPartitionMap.keySet());
313               previousActiveTasks.taskIds().retainAll(taskForPartitionMap.values());
314               previousStandbyTasks.taskIds().retainAll(taskForPartitionMap.values());
315           }

318       /**
319        * Compute the lag for each stateful task, including tasks this client did not previously have.
320        */
321       public void computeTaskLags(final UUID uuid, final Map<TaskId, Long> allTaskEndOffsetSums) {
322           if (!taskLagTotals.isEmpty()) {
323               throw new IllegalStateException("Already computed task lags for this client.");
324           }
325   
326           for (final Map.Entry<TaskId, Long> taskEntry : allTaskEndOffsetSums.entrySet()) {
327               final TaskId task = taskEntry.getKey();
328               final Long endOffsetSum = taskEntry.getValue();
329               final Long offsetSum = taskOffsetSums.getOrDefault(task, 0L);
330   
331               if (offsetSum == Task.LATEST_OFFSET) {
332                   taskLagTotals.put(task, Task.LATEST_OFFSET);
333               } else if (offsetSum == UNKNOWN_OFFSET_SUM) {
334                   taskLagTotals.put(task, UNKNOWN_OFFSET_SUM);
335               } else if (endOffsetSum < offsetSum) {
336                   LOG.warn("Task " + task + " had endOffsetSum=" + endOffsetSum + " smaller than offsetSum=" +
337                                offsetSum + " on member " + uuid + ". This probably means the task is corrupted," +
338                                " which in turn indicates that it will need to restore from scratch if it gets assigned." +
339                                " The assignor will de-prioritize returning this task to this member in the hopes that" +
340                                " some other member may be able to re-use its state.");
341                   taskLagTotals.put(task, endOffsetSum);
342               } else {
343                   taskLagTotals.put(task, endOffsetSum - offsetSum);
344               }
345           }
346       }
347   
348       /**
349        * Returns the total lag across all logged stores in the task. Equal to the end offset sum if this client
350        * did not have any state for this task on disk.
351        *
352        * @return end offset sum - offset sum
353        *          Task.LATEST_OFFSET if this was previously an active running task on this client
354        */
355       public long lagFor(final TaskId task) {
356           final Long totalLag = taskLagTotals.get(task);
357           if (totalLag == null) {
358               throw new IllegalStateException("Tried to lookup lag for unknown task " + task);
359           }
360           return totalLag;
361       }

363       /**
364        * @return the previous tasks assigned to this consumer ordered by lag, filtered for any tasks that don't exist in this assignment
365        */
366       public SortedSet<TaskId> prevTasksByLag(final String consumer) {
367           final SortedSet<TaskId> prevTasksByLag = new TreeSet<>(comparingLong(this::lagFor).thenComparing(TaskId::compareTo));
368           for (final TaskId task : prevOwnedStatefulTasksByConsumer(consumer)) {
369               if (taskLagTotals.containsKey(task)) {
370                   prevTasksByLag.add(task);
371               } else {
372                   LOG.debug("Skipping previous task {} since it's not part of the current assignment", task);
373               }
374           }
375           return prevTasksByLag;
376       }

390       boolean hasMoreAvailableCapacityThan(final ClientState other) {
391           if (capacity <= 0) {
392               throw new IllegalStateException("Capacity of this ClientState must be greater than 0.");
393           }
394   
395           if (other.capacity <= 0) {
396               throw new IllegalStateException("Capacity of other ClientState must be greater than 0");
397           }
398   
399           final double otherLoad = (double) other.assignedTaskCount() / other.capacity;
400           final double thisLoad = (double) assignedTaskCount() / capacity;
401   
402           if (thisLoad < otherLoad) {
403               return true;
404           } else if (thisLoad > otherLoad) {
405               return false;
406           } else {
407               return capacity > other.capacity;
408           }
409       }

438       private void initializePrevActiveTasksFromOwnedPartitions(final Map<TopicPartition, TaskId> taskForPartitionMap) {
439           // there are three cases where we need to construct some or all of the prevTasks from the ownedPartitions:
440           // 1) COOPERATIVE clients on version 2.4-2.5 do not encode active tasks at all and rely on ownedPartitions
441           // 2) future client during version probing, when we can't decode the future subscription info's prev tasks
442           // 3) stateless tasks are not encoded in the task lags, and must be figured out from the ownedPartitions
443           for (final Map.Entry<TopicPartition, String> partitionEntry : ownedPartitions.entrySet()) {
444               final TopicPartition tp = partitionEntry.getKey();
445               final TaskId task = taskForPartitionMap.get(tp);
446               if (task != null) {
447                   addPreviousActiveTask(task);
448                   previousActiveTasks.consumerToTaskIds().computeIfAbsent(partitionEntry.getValue(), k -> new HashSet<>()).add(task);
449               } else {
450                   LOG.error("No task found for topic partition {}", tp);
451               }
452           }

455       private void initializeRemainingPrevTasksFromTaskOffsetSums() {
456           final Set<TaskId> previousActiveTaskIds = previousActiveTasks.taskIds();
457           if (previousActiveTaskIds.isEmpty() && !ownedPartitions.isEmpty()) {
458               LOG.error("Tried to process tasks in offset sum map before processing tasks from ownedPartitions = {}", ownedPartitions);
459               throw new IllegalStateException("Must initialize prevActiveTasks from ownedPartitions before initializing remaining tasks.");
460           }
461           for (final Map.Entry<TaskId, Long> taskEntry : taskOffsetSums.entrySet()) {
462               final TaskId task = taskEntry.getKey();
463               if (!previousActiveTaskIds.contains(task)) {
464                   final long offsetSum = taskEntry.getValue();
465                   if (offsetSum == Task.LATEST_OFFSET) {
466                       addPreviousActiveTask(task);
467                   } else {
468                       addPreviousStandbyTask(task);
469                   }
470               }
471           }
