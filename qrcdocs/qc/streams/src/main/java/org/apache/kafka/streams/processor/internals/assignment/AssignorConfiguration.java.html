50       public AssignorConfiguration(final Map<String, ?> configs) {
51           // NOTE: If you add a new config to pass through to here, be sure to test it in a real
52           // application. Since we filter out some configurations, we may have to explicitly copy
53           // them over when we construct the Consumer.
54           streamsConfig = new ClientUtils.QuietStreamsConfig(configs);
55           internalConfigs = configs;
56   
57           // Setting the logger with the passed in client thread name
58           logPrefix = String.format("stream-thread [%s] ", streamsConfig.getString(CommonClientConfigs.CLIENT_ID_CONFIG));
59           final LogContext logContext = new LogContext(logPrefix);
60           log = logContext.logger(getClass());
61   
62           {
63               final Object o = configs.get(InternalConfig.REFERENCE_CONTAINER_PARTITION_ASSIGNOR);
64               if (o == null) {
65                   final KafkaException fatalException = new KafkaException("ReferenceContainer is not specified");
66                   log.error(fatalException.getMessage(), fatalException);
67                   throw fatalException;
68               }
69   
70               if (!(o instanceof ReferenceContainer)) {
71                   final KafkaException fatalException = new KafkaException(
72                       String.format("%s is not an instance of %s", o.getClass().getName(), ReferenceContainer.class.getName())
73                   );
74                   log.error(fatalException.getMessage(), fatalException);
75                   throw fatalException;
76               }
77   
78               referenceContainer = (ReferenceContainer) o;
79           }
80   
81           {
82               final String o = (String) configs.get(INTERNAL_TASK_ASSIGNOR_CLASS);
83               if (o == null) {
84                   taskAssignorClass = HighAvailabilityTaskAssignor.class.getName();
85               } else {
86                   taskAssignorClass = o;
87               }
88           }
89       }

95       public RebalanceProtocol rebalanceProtocol() {
96           final String upgradeFrom = streamsConfig.getString(StreamsConfig.UPGRADE_FROM_CONFIG);
97           if (upgradeFrom != null) {
98               switch (upgradeFrom) {
99                   case StreamsConfig.UPGRADE_FROM_0100:
100                   case StreamsConfig.UPGRADE_FROM_0101:
101                   case StreamsConfig.UPGRADE_FROM_0102:
102                   case StreamsConfig.UPGRADE_FROM_0110:
103                   case StreamsConfig.UPGRADE_FROM_10:
104                   case StreamsConfig.UPGRADE_FROM_11:
105                   case StreamsConfig.UPGRADE_FROM_20:
106                   case StreamsConfig.UPGRADE_FROM_21:
107                   case StreamsConfig.UPGRADE_FROM_22:
108                   case StreamsConfig.UPGRADE_FROM_23:
109                       // ATTENTION: The following log messages is used for verification in system test
110                       // streams/streams_cooperative_rebalance_upgrade_test.py::StreamsCooperativeRebalanceUpgradeTest.test_upgrade_to_cooperative_rebalance
111                       // If you change it, please do also change the system test accordingly and
112                       // verify whether the test passes.
113                       log.info("Eager rebalancing protocol is enabled now for upgrade from {}.x", upgradeFrom);
114                       log.warn("The eager rebalancing protocol is deprecated and will stop being supported in a future release." +
115                           " Please be prepared to remove the 'upgrade.from' config soon.");
116                       return RebalanceProtocol.EAGER;
117                   case StreamsConfig.UPGRADE_FROM_24:
118                   case StreamsConfig.UPGRADE_FROM_25:
119                   case StreamsConfig.UPGRADE_FROM_26:
120                   case StreamsConfig.UPGRADE_FROM_27:
121                   case StreamsConfig.UPGRADE_FROM_28:
122                   case StreamsConfig.UPGRADE_FROM_30:
123                   case StreamsConfig.UPGRADE_FROM_31:
124                   case StreamsConfig.UPGRADE_FROM_32:
125                       // This config is for explicitly sending FK response to a requested partition
126                       // and should not affect the rebalance protocol
127                       break;
128                   default:
129                       throw new IllegalArgumentException("Unknown configuration value for parameter 'upgrade.from': " + upgradeFrom);
130               }
131           }
132           // ATTENTION: The following log messages is used for verification in system test
133           // streams/streams_cooperative_rebalance_upgrade_test.py::StreamsCooperativeRebalanceUpgradeTest.test_upgrade_to_cooperative_rebalance
134           // If you change it, please do also change the system test accordingly and
135           // verify whether the test passes.
136           log.info("Cooperative rebalancing protocol is enabled now");
137           return RebalanceProtocol.COOPERATIVE;
138       }

144       public int configuredMetadataVersion(final int priorVersion) {
145           final String upgradeFrom = streamsConfig.getString(StreamsConfig.UPGRADE_FROM_CONFIG);
146           if (upgradeFrom != null) {
147               switch (upgradeFrom) {
148                   case StreamsConfig.UPGRADE_FROM_0100:
149                       log.info(
150                           "Downgrading metadata version from {} to 1 for upgrade from 0.10.0.x.",
151                           LATEST_SUPPORTED_VERSION
152                       );
153                       return 1;
154                   case StreamsConfig.UPGRADE_FROM_0101:
155                   case StreamsConfig.UPGRADE_FROM_0102:
156                   case StreamsConfig.UPGRADE_FROM_0110:
157                   case StreamsConfig.UPGRADE_FROM_10:
158                   case StreamsConfig.UPGRADE_FROM_11:
159                       log.info(
160                           "Downgrading metadata version from {} to 2 for upgrade from {}.x.",
161                           LATEST_SUPPORTED_VERSION,
162                           upgradeFrom
163                       );
164                       return 2;
165                   case StreamsConfig.UPGRADE_FROM_20:
166                   case StreamsConfig.UPGRADE_FROM_21:
167                   case StreamsConfig.UPGRADE_FROM_22:
168                   case StreamsConfig.UPGRADE_FROM_23:
169                       // These configs are for cooperative rebalancing and should not affect the metadata version
170                       break;
171                   case StreamsConfig.UPGRADE_FROM_24:
172                   case StreamsConfig.UPGRADE_FROM_25:
173                   case StreamsConfig.UPGRADE_FROM_26:
174                   case StreamsConfig.UPGRADE_FROM_27:
175                   case StreamsConfig.UPGRADE_FROM_28:
176                   case StreamsConfig.UPGRADE_FROM_30:
177                   case StreamsConfig.UPGRADE_FROM_31:
178                   case StreamsConfig.UPGRADE_FROM_32:
179                       // This config is for explicitly sending FK response to a requested partition
180                       // and should not affect the metadata version
181                       break;
182                   default:
183                       throw new IllegalArgumentException(
184                           "Unknown configuration value for parameter 'upgrade.from': " + upgradeFrom
185                       );
186               }
187           }
188           return priorVersion;
189       }

191       public String userEndPoint() {
192           final String configuredUserEndpoint = streamsConfig.getString(StreamsConfig.APPLICATION_SERVER_CONFIG);
193           if (configuredUserEndpoint != null && !configuredUserEndpoint.isEmpty()) {
194               try {
195                   final String host = getHost(configuredUserEndpoint);
196                   final Integer port = getPort(configuredUserEndpoint);
197   
198                   if (host == null || port == null) {
199                       throw new ConfigException(
200                           String.format(
201                               "%s Config %s isn't in the correct format. Expected a host:port pair but received %s",
202                               logPrefix, StreamsConfig.APPLICATION_SERVER_CONFIG, configuredUserEndpoint
203                           )
204                       );
205                   }
206               } catch (final NumberFormatException nfe) {
207                   throw new ConfigException(
208                       String.format("%s Invalid port supplied in %s for config %s: %s",
209                                     logPrefix, configuredUserEndpoint, StreamsConfig.APPLICATION_SERVER_CONFIG, nfe)
210                   );
211               }
212               return configuredUserEndpoint;
213           } else {
214               return null;
215           }
216       }

241       public AssignmentListener assignmentListener() {
242           final Object o = internalConfigs.get(InternalConfig.ASSIGNMENT_LISTENER);
243           if (o == null) {
244               return stable -> { };
245           }
246   
247           if (!(o instanceof AssignmentListener)) {
248               final KafkaException fatalException = new KafkaException(
249                   String.format("%s is not an instance of %s", o.getClass().getName(), AssignmentListener.class.getName())
250               );
251               log.error(fatalException.getMessage(), fatalException);
252               throw fatalException;
253           }
254   
255           return (AssignmentListener) o;
256       }

262       public static class AssignmentConfigs {
263           public final long acceptableRecoveryLag;
264           public final int maxWarmupReplicas;

266           public final long probingRebalanceIntervalMs;
267           public final List<String> rackAwareAssignmentTags;
268   
269           private AssignmentConfigs(final StreamsConfig configs) {
270               acceptableRecoveryLag = configs.getLong(StreamsConfig.ACCEPTABLE_RECOVERY_LAG_CONFIG);
271               maxWarmupReplicas = configs.getInt(StreamsConfig.MAX_WARMUP_REPLICAS_CONFIG);
272               numStandbyReplicas = configs.getInt(StreamsConfig.NUM_STANDBY_REPLICAS_CONFIG);
273               probingRebalanceIntervalMs = configs.getLong(StreamsConfig.PROBING_REBALANCE_INTERVAL_MS_CONFIG);
274               rackAwareAssignmentTags = configs.getList(StreamsConfig.RACK_AWARE_ASSIGNMENT_TAGS_CONFIG);
275           }
276   
277           AssignmentConfigs(final Long acceptableRecoveryLag,
278                             final Integer maxWarmupReplicas,
279                             final Integer numStandbyReplicas,
280                             final Long probingRebalanceIntervalMs,
281                             final List<String> rackAwareAssignmentTags) {
282               this.acceptableRecoveryLag = validated(StreamsConfig.ACCEPTABLE_RECOVERY_LAG_CONFIG, acceptableRecoveryLag);
283               this.maxWarmupReplicas = validated(StreamsConfig.MAX_WARMUP_REPLICAS_CONFIG, maxWarmupReplicas);
284               this.numStandbyReplicas = validated(StreamsConfig.NUM_STANDBY_REPLICAS_CONFIG, numStandbyReplicas);
285               this.probingRebalanceIntervalMs = validated(StreamsConfig.PROBING_REBALANCE_INTERVAL_MS_CONFIG, probingRebalanceIntervalMs);
286               this.rackAwareAssignmentTags = validated(StreamsConfig.RACK_AWARE_ASSIGNMENT_TAGS_CONFIG, rackAwareAssignmentTags);
287           }
288   
289           private static <T> T validated(final String configKey, final T value) {
290               final ConfigDef.Validator validator = StreamsConfig.configDef().configKeys().get(configKey).validator;
291               if (validator != null) {
292                   validator.ensureValid(configKey, value);
293               }
294               return value;
295           }
