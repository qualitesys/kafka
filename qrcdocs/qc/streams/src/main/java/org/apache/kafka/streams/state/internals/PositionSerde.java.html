17   package org.apache.kafka.streams.state.internals;
18   
19   import org.apache.kafka.streams.query.Position;
20   
21   import java.nio.ByteBuffer;
22   import java.nio.charset.StandardCharsets;
23   import java.util.ArrayList;
24   import java.util.HashMap;
25   import java.util.Map;
26   import java.util.Map.Entry;
27   
28   public final class PositionSerde {
29   
30       // uninstantiable utility class
31       public PositionSerde() {
32       }
33   
34       public static Position deserialize(final ByteBuffer buffer) {
35           final byte version = buffer.get();
36   
37           switch (version) {
38               case (byte) 0: // actual position, v0
39                   final int nTopics = buffer.getInt();
40                   final Map<String, Map<Integer, Long>> position = new HashMap<>(nTopics);
41                   for (int i = 0; i < nTopics; i++) {
42                       final int topicNameLength = buffer.getInt();
43                       final byte[] topicNameBytes = new byte[topicNameLength];
44                       buffer.get(topicNameBytes);
45                       final String topic = new String(topicNameBytes, StandardCharsets.UTF_8);
46   
47                       final int numPairs = buffer.getInt();
48                       final Map<Integer, Long> partitionOffsets = new HashMap<>(numPairs);
49                       for (int j = 0; j < numPairs; j++) {
50                           partitionOffsets.put(buffer.getInt(), buffer.getLong());
51                       }
52                       position.put(topic, partitionOffsets);
53                   }
54                   return Position.fromMap(position);
55               default:
56                   throw new IllegalArgumentException(
57                           "Unknown version " + version + " when deserializing Position"
58                   );
59           }
60       }
61   
62       public static ByteBuffer serialize(final Position position) {
63           final byte version = (byte) 0;
64   
65           int arraySize = Byte.SIZE; // version
66   
67           final int nTopics = position.getTopics().size();
68           arraySize += Integer.SIZE;
69   
70           final ArrayList<String> entries =
71                   new ArrayList<>(position.getTopics());
72           final byte[][] topics = new byte[entries.size()][];
73   
74           for (int i = 0; i < nTopics; i++) {
75               final String topic = entries.get(i);
76               final byte[] topicBytes = topic.getBytes(StandardCharsets.UTF_8);
77               topics[i] = topicBytes;
78               arraySize += Integer.SIZE; // topic name length
79               arraySize += topicBytes.length; // topic name itself
80   
81               final Map<Integer, Long> partitionOffsets = position.getPartitionPositions(topic);
82               arraySize += Integer.SIZE; // Number of PartitionOffset pairs
83               arraySize += (Integer.SIZE + Long.SIZE)
84                       * partitionOffsets.size(); // partitionOffsets themselves
85           }
86   
87           final ByteBuffer buffer = ByteBuffer.allocate(arraySize);
88           buffer.put(version);
89   
90           buffer.putInt(nTopics);
91           for (int i = 0; i < nTopics; i++) {
92               buffer.putInt(topics[i].length);
93               buffer.put(topics[i]);
94   
95               final String topic = entries.get(i);
96               final Map<Integer, Long> partitionOffsets = position.getPartitionPositions(topic);
97               buffer.putInt(partitionOffsets.size());
98               for (final Entry<Integer, Long> partitionOffset : partitionOffsets.entrySet()) {
99                   buffer.putInt(partitionOffset.getKey());
100                   buffer.putLong(partitionOffset.getValue());
101               }
102           }
103   
104           buffer.flip();
105           return buffer;
106       }
107   }
