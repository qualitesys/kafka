17   package org.apache.kafka.streams;
18   
19   import org.apache.kafka.streams.state.QueryableStoreType;
20   
21   import java.util.Objects;
22   
23   /**
24    * {@code StoreQueryParameters} allows you to pass a variety of parameters when fetching a store for interactive query.
25    */
26   public class StoreQueryParameters<T> {
27   
28       private final Integer partition;
29       private final boolean staleStores;
30       private final String storeName;
31       private final QueryableStoreType<T> queryableStoreType;
32   
33       protected StoreQueryParameters(final String storeName, final QueryableStoreType<T>  queryableStoreType, final Integer partition, final boolean staleStores) {
34           this.storeName = storeName;
35           this.queryableStoreType = queryableStoreType;
36           this.partition = partition;
37           this.staleStores = staleStores;
38       }
39   
40       public static <T> StoreQueryParameters<T> fromNameAndType(final String storeName,
41                                                                 final QueryableStoreType<T>  queryableStoreType) {
42           return new StoreQueryParameters<>(storeName, queryableStoreType, null, false);
43       }
44   
45       /**
46        * Set a specific partition that should be queried exclusively.
47        *
48        * @param partition   The specific integer partition to be fetched from the stores list by using {@link StoreQueryParameters}.
49        *
50        * @return StoreQueryParameters a new {@code StoreQueryParameters} instance configured with the specified partition
51        */
52       public StoreQueryParameters<T> withPartition(final Integer partition) {
53           return new StoreQueryParameters<>(storeName, queryableStoreType, partition, staleStores);
54       }
55   
56       /**
57        * Enable querying of stale state stores, i.e., allow to query active tasks during restore as well as standby tasks.
58        *
59        * @return StoreQueryParameters a new {@code StoreQueryParameters} instance configured with serving from stale stores enabled
60        */
61       public StoreQueryParameters<T> enableStaleStores() {
62           return new StoreQueryParameters<>(storeName, queryableStoreType, partition, true);
63       }
64   
65       /**
66        * Get the name of the state store that should be queried.
67        *
68        * @return String state store name
69        */
70       public String storeName() {
71           return storeName;
72       }
73   
74       /**
75        * Get the queryable store type for which key is queried by the user.
76        *
77        * @return QueryableStoreType type of queryable store
78        */
79       public QueryableStoreType<T> queryableStoreType() {
80           return queryableStoreType;
81       }
82   
83       /**
84        * Get the store partition that will be queried.
85        * If the method returns {@code null}, it would mean that no specific partition has been requested,
86        * so all the local partitions for the store will be queried.
87        *
88        * @return Integer partition
89        */
90       public Integer partition() {
91           return partition;
92       }
93   
94       /**
95        * Get the flag staleStores. If {@code true}, include standbys and recovering stores along with running stores.
96        *
97        * @return boolean staleStores
98        */
99       public boolean staleStoresEnabled() {
100           return staleStores;
101       }
102   
103       @Override
104       public boolean equals(final Object obj) {
105           if (!(obj instanceof StoreQueryParameters)) {
106               return false;
107           }
108           final StoreQueryParameters storeQueryParameters = (StoreQueryParameters) obj;
109           return Objects.equals(storeQueryParameters.partition, partition)
110                   && Objects.equals(storeQueryParameters.staleStores, staleStores)
111                   && Objects.equals(storeQueryParameters.storeName, storeName)
112                   && Objects.equals(storeQueryParameters.queryableStoreType, queryableStoreType);
113       }
114   
115       @Override
116       public String toString() {
117           return "StoreQueryParameters {" +
118                   "partition=" + partition +
119                   ", staleStores=" + staleStores +
120                   ", storeName=" + storeName +
121                   ", queryableStoreType=" + queryableStoreType +
122                   '}';
123       }
124   
125       @Override
126       public int hashCode() {
127           return Objects.hash(partition, staleStores, storeName, queryableStoreType);
128       }
129   }
