51   public class DefaultStateUpdater implements StateUpdater {
52   
53       private final static String BUG_ERROR_MESSAGE = "This indicates a bug. " +
54           "Please report at https://issues.apache.org/jira/projects/KAFKA/issues or to the dev-mailing list (https://kafka.apache.org/contact).";

56       private class StateUpdaterThread extends Thread {
57   
58           private final ChangelogReader changelogReader;
59           private final AtomicBoolean isRunning = new AtomicBoolean(true);
60           private final Consumer<Set<TopicPartition>> offsetResetter;
61           private final Map<TaskId, Task> updatingTasks = new ConcurrentHashMap<>();
62           private final Logger log;
63   
64           public StateUpdaterThread(final String name,
65                                     final ChangelogReader changelogReader,
66                                     final Consumer<Set<TopicPartition>> offsetResetter) {
67               super(name);
68               this.changelogReader = changelogReader;
69               this.offsetResetter = offsetResetter;
70   
71               final String logPrefix = String.format("%s ", name);
72               final LogContext logContext = new LogContext(logPrefix);
73               log = logContext.logger(DefaultStateUpdater.class);
74           }

76           public Collection<Task> getUpdatingTasks() {
77               return updatingTasks.values();
78           }

80           public Collection<StandbyTask> getUpdatingStandbyTasks() {
81               return updatingTasks.values().stream()
82                   .filter(t -> !t.isActive())
83                   .map(t -> (StandbyTask) t)
84                   .collect(Collectors.toList());
85           }
86   
87           public boolean onlyStandbyTasksLeft() {
88               return !updatingTasks.isEmpty() && updatingTasks.values().stream().allMatch(t -> !t.isActive());
89           }
90   
91           @Override
92           public void run() {
93               log.info("State updater thread started");
94               try {
95                   while (isRunning.get()) {
96                       try {
97                           runOnce();
98                       } catch (final InterruptedException interruptedException) {
99                           return;
100                       }
101                   }
102               } catch (final RuntimeException anyOtherException) {
103                   handleRuntimeException(anyOtherException);
104               } finally {
105                   clear();
106                   shutdownGate.countDown();
107                   log.info("State updater thread shutdown");
108               }
109           }
110   
111           private void runOnce() throws InterruptedException {
112               performActionsOnTasks();
113               restoreTasks();
114               waitIfAllChangelogsCompletelyRead();
115           }
116   
117           private void performActionsOnTasks() {
118               tasksAndActionsLock.lock();
119               try {
120                   for (final TaskAndAction taskAndAction : getTasksAndActions()) {
121                       final Action action = taskAndAction.getAction();
122                       switch (action) {
123                           case ADD:
124                               addTask(taskAndAction.getTask());
125                               break;
126                           case REMOVE:
127                               removeTask(taskAndAction.getTaskId());
128                               break;
129                       }
130                   }
131               } finally {
132                   tasksAndActionsLock.unlock();
133               }
134           }
135   
136           private void restoreTasks() {
137               try {
138                   changelogReader.restore(updatingTasks);
139               } catch (final TaskCorruptedException taskCorruptedException) {
140                   handleTaskCorruptedException(taskCorruptedException);
141               } catch (final StreamsException streamsException) {
142                   handleStreamsException(streamsException);
143               }
144               final Set<TopicPartition> completedChangelogs = changelogReader.completedChangelogs();
145               final List<Task> activeTasks = updatingTasks.values().stream().filter(Task::isActive).collect(Collectors.toList());
146               for (final Task task : activeTasks) {
147                   maybeCompleteRestoration((StreamTask) task, completedChangelogs);
148               }
149           }
150   
151           private void handleRuntimeException(final RuntimeException runtimeException) {
152               log.error("An unexpected error occurred within the state updater thread: " + runtimeException);
153               final ExceptionAndTasks exceptionAndTasks = new ExceptionAndTasks(new HashSet<>(updatingTasks.values()), runtimeException);
154               updatingTasks.clear();
155               exceptionsAndFailedTasks.add(exceptionAndTasks);
156               isRunning.set(false);
157           }
158   
159           private void handleTaskCorruptedException(final TaskCorruptedException taskCorruptedException) {
160               log.info("Encountered task corrupted exception: ", taskCorruptedException);
161               final Set<TaskId> corruptedTaskIds = taskCorruptedException.corruptedTasks();
162               final Set<Task> corruptedTasks = new HashSet<>();
163               for (final TaskId taskId : corruptedTaskIds) {
164                   final Task corruptedTask = updatingTasks.remove(taskId);
165                   if (corruptedTask == null) {
166                       throw new IllegalStateException("Task " + taskId + " is corrupted but is not updating. " + BUG_ERROR_MESSAGE);
167                   }
168                   corruptedTasks.add(corruptedTask);
169               }
170               exceptionsAndFailedTasks.add(new ExceptionAndTasks(corruptedTasks, taskCorruptedException));
171           }
172   
173           private void handleStreamsException(final StreamsException streamsException) {
174               log.info("Encountered streams exception: ", streamsException);
175               final ExceptionAndTasks exceptionAndTasks;
176               if (streamsException.taskId().isPresent()) {
177                   exceptionAndTasks = handleStreamsExceptionWithTask(streamsException);
178               } else {
179                   exceptionAndTasks = handleStreamsExceptionWithoutTask(streamsException);
180               }
181               exceptionsAndFailedTasks.add(exceptionAndTasks);
182           }
183   
184           private ExceptionAndTasks handleStreamsExceptionWithTask(final StreamsException streamsException) {
185               final TaskId failedTaskId = streamsException.taskId().get();
186               if (!updatingTasks.containsKey(failedTaskId)) {
187                   throw new IllegalStateException("Task " + failedTaskId + " failed but is not updating. " + BUG_ERROR_MESSAGE);
188               }
189               final Set<Task> failedTask = new HashSet<>();
190               failedTask.add(updatingTasks.get(failedTaskId));
191               updatingTasks.remove(failedTaskId);
192               return new ExceptionAndTasks(failedTask, streamsException);
193           }
194   
195           private ExceptionAndTasks handleStreamsExceptionWithoutTask(final StreamsException streamsException) {
196               final ExceptionAndTasks exceptionAndTasks = new ExceptionAndTasks(new HashSet<>(updatingTasks.values()), streamsException);
197               updatingTasks.clear();
198               return exceptionAndTasks;
199           }

201           private void waitIfAllChangelogsCompletelyRead() throws InterruptedException {
202               if (isRunning.get() && changelogReader.allChangelogsCompleted()) {
203                   tasksAndActionsLock.lock();
204                   try {
205                       while (tasksAndActions.isEmpty()) {
206                           tasksAndActionsCondition.await();
207                       }
208                   } finally {
209                       tasksAndActionsLock.unlock();
210                   }
211               }
212           }
213   
214           private void clear() {
215               tasksAndActionsLock.lock();
216               restoredActiveTasksLock.lock();
217               try {
218                   tasksAndActions.clear();
219                   restoredActiveTasks.clear();
220               } finally {
221                   restoredActiveTasksLock.unlock();
222                   tasksAndActionsLock.unlock();
223               }
224               changelogReader.clear();
225               updatingTasks.clear();
226           }
227   
228           private List<TaskAndAction> getTasksAndActions() {
229               final List<TaskAndAction> tasksAndActionsToProcess = new ArrayList<>(tasksAndActions);
230               tasksAndActions.clear();
231               return tasksAndActionsToProcess;
232           }

234           private void addTask(final Task task) {
235               if (isStateless(task)) {
236                   addTaskToRestoredTasks((StreamTask) task);
237                   log.debug("Stateless active task " + task.id() + " was added to the restored tasks of the state updater");
238               } else {
239                   updatingTasks.put(task.id(), task);
240                   if (task.isActive()) {
241                       log.debug("Stateful active task " + task.id() + " was added to the updating tasks of the state updater");
242                       changelogReader.enforceRestoreActive();
243                   } else {
244                       log.debug("Standby task " + task.id() + " was added to the updating tasks of the state updater");
245                       if (updatingTasks.size() == 1) {
246                           changelogReader.transitToUpdateStandby();
247                       }
248                   }
249               }
250           }
251   
252           private void removeTask(final TaskId taskId) {
253               final Task task = updatingTasks.remove(taskId);
254               if (task != null) {
255                   final Collection<TopicPartition> changelogPartitions = task.changelogPartitions();
256                   changelogReader.unregister(changelogPartitions);
257                   removedTasks.add(task);
258                   log.debug((task.isActive() ? "Active" : "Standby")
259                       + " task " + task.id() + " was removed from the updating tasks and added to the removed tasks.");
260               } else {
261                   log.debug("Task " + taskId + " was not removed since it is not updating.");
262               }
263           }
264   
265           private boolean isStateless(final Task task) {
266               return task.changelogPartitions().isEmpty() && task.isActive();
267           }
268   
269           private void maybeCompleteRestoration(final StreamTask task,
270                                                 final Set<TopicPartition> restoredChangelogs) {
271               final Collection<TopicPartition> taskChangelogPartitions = task.changelogPartitions();
272               if (restoredChangelogs.containsAll(taskChangelogPartitions)) {
273                   task.completeRestoration(offsetResetter);
274                   log.debug("Stateful active task " + task.id() + " completed restoration");
275                   addTaskToRestoredTasks(task);
276                   updatingTasks.remove(task.id());
277                   if (onlyStandbyTasksLeft()) {
278                       changelogReader.transitToUpdateStandby();
279                   }
280               }
281           }
282   
283           private void addTaskToRestoredTasks(final StreamTask task) {
284               restoredActiveTasksLock.lock();
285               try {
286                   restoredActiveTasks.add(task);
287                   log.debug("Active task " + task.id() + " was added to the restored tasks");
288                   restoredActiveTasksCondition.signalAll();
289               } finally {
290                   restoredActiveTasksLock.unlock();
291               }
292           }
293       }

295       private final Time time;
296       private final ChangelogReader changelogReader;
297       private final Consumer<Set<TopicPartition>> offsetResetter;
298       private final Queue<TaskAndAction> tasksAndActions = new LinkedList<>();
299       private final Lock tasksAndActionsLock = new ReentrantLock();
300       private final Condition tasksAndActionsCondition = tasksAndActionsLock.newCondition();
301       private final Queue<StreamTask> restoredActiveTasks = new LinkedList<>();
302       private final Lock restoredActiveTasksLock = new ReentrantLock();
303       private final Condition restoredActiveTasksCondition = restoredActiveTasksLock.newCondition();
304       private final BlockingQueue<ExceptionAndTasks> exceptionsAndFailedTasks = new LinkedBlockingQueue<>();
305       private final BlockingQueue<Task> removedTasks = new LinkedBlockingQueue<>();
306       private CountDownLatch shutdownGate;
307   
308       private StateUpdaterThread stateUpdaterThread = null;
309   
310       public DefaultStateUpdater(final ChangelogReader changelogReader,
311                                  final Consumer<Set<TopicPartition>> offsetResetter,
312                                  final Time time) {
313           this.changelogReader = changelogReader;
314           this.offsetResetter = offsetResetter;
315           this.time = time;
316       }
317   
318       @Override
319       public void add(final Task task) {
320           if (stateUpdaterThread == null) {
321               stateUpdaterThread = new StateUpdaterThread("state-updater", changelogReader, offsetResetter);
322               stateUpdaterThread.start();
323               shutdownGate = new CountDownLatch(1);
324           }
325   
326           verifyStateFor(task);
327   
328           tasksAndActionsLock.lock();
329           try {
330               tasksAndActions.add(TaskAndAction.createAddTask(task));
331               tasksAndActionsCondition.signalAll();
332           } finally {
333               tasksAndActionsLock.unlock();
334           }
335       }
336   
337       private void verifyStateFor(final Task task) {
338           if (task.isActive() && task.state() != State.RESTORING) {
339               throw new IllegalStateException("Active task " + task.id() + " is not in state RESTORING. " + BUG_ERROR_MESSAGE);
340           }
341           if (!task.isActive() && task.state() != State.RUNNING) {
342               throw new IllegalStateException("Standby task " + task.id() + " is not in state RUNNING. " + BUG_ERROR_MESSAGE);
343           }
344       }
345   
346       @Override
347       public void remove(final TaskId taskId) {
348           tasksAndActionsLock.lock();
349           try {
350               tasksAndActions.add(TaskAndAction.createRemoveTask(taskId));
351               tasksAndActionsCondition.signalAll();
352           } finally {
353               tasksAndActionsLock.unlock();
354           }
355       }
356   
357       @Override
358       public Set<StreamTask> drainRestoredActiveTasks(final Duration timeout) {
359           final long timeoutMs = timeout.toMillis();
360           final long startTime = time.milliseconds();
361           final long deadline = startTime + timeoutMs;
362           long now = startTime;
363           final Set<StreamTask> result = new HashSet<>();
364           try {
365               while (now <= deadline && result.isEmpty()) {
366                   restoredActiveTasksLock.lock();
367                   try {
368                       while (restoredActiveTasks.isEmpty() && now <= deadline) {
369                           final boolean elapsed = restoredActiveTasksCondition.await(deadline - now, TimeUnit.MILLISECONDS);
370                           now = time.milliseconds();
371                       }
372                       result.addAll(restoredActiveTasks);
373                       restoredActiveTasks.clear();
374                   } finally {
375                       restoredActiveTasksLock.unlock();
376                   }
377                   now = time.milliseconds();
378               }
379               return result;
380           } catch (final InterruptedException e) {
381               // ignore
382           }
383           return result;
384       }

388           final List<Task> result = new ArrayList<>();
389           removedTasks.drainTo(result);
390           return new HashSet<>(result);
391       }

396           exceptionsAndFailedTasks.drainTo(result);
397           return result;
398       }

425       @Override
426       public void shutdown(final Duration timeout) {
427           if (stateUpdaterThread != null) {
428               stateUpdaterThread.isRunning.set(false);
429               stateUpdaterThread.interrupt();
430               try {
431                   if (!shutdownGate.await(timeout.toMillis(), TimeUnit.MILLISECONDS)) {
432                       throw new StreamsException("State updater thread did not shutdown within the timeout");
433                   }
434                   stateUpdaterThread = null;
435               } catch (final InterruptedException ignored) {
436               }
437           }
438       }
