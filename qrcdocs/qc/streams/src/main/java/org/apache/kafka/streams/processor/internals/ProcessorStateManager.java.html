56   /**
57    * ProcessorStateManager is the source of truth for the current offset for each state store,
58    * which is either the read offset during restoring, or the written offset during normal processing.
59    *
60    * The offset is initialized as null when the state store is registered, and then it can be updated by
61    * loading checkpoint file, restore state stores, or passing from the record collector's written offsets.
62    *
63    * When checkpointing, if the offset is not null it would be written to the file.
64    *
65    * The manager is also responsible for restoring state stores via their registered restore callback,
66    * which is used for both updating standby tasks as well as restoring active tasks.
67    */
68   public class ProcessorStateManager implements StateManager {

230       // package-private for test only
231       void initializeStoreOffsetsFromCheckpoint(final boolean storeDirIsEmpty) {
232           try {
233               final Map<TopicPartition, Long> loadedCheckpoints = checkpointFile.read();
234   
235               log.trace("Loaded offsets from the checkpoint file: {}", loadedCheckpoints);
236   
237               for (final StateStoreMetadata store : stores.values()) {
238                   if (store.corrupted) {
239                       log.error("Tried to initialize store offsets for corrupted store {}", store);
240                       throw new IllegalStateException("Should not initialize offsets for a corrupted task");
241                   }
242   
243                   if (store.changelogPartition == null) {
244                       log.info("State store {} is not logged and hence would not be restored", store.stateStore.name());
245                   } else if (!store.stateStore.persistent()) {
246                       log.info("Initializing to the starting offset for changelog {} of in-memory state store {}",
247                                store.changelogPartition, store.stateStore.name());
248                   } else if (store.offset() == null) {
249                       if (loadedCheckpoints.containsKey(store.changelogPartition)) {
250                           final Long offset = changelogOffsetFromCheckpointedOffset(loadedCheckpoints.remove(store.changelogPartition));
251                           store.setOffset(offset);
252   
253                           log.debug("State store {} initialized from checkpoint with offset {} at changelog {}",
254                                     store.stateStore.name(), store.offset, store.changelogPartition);
255                       } else {
256                           // with EOS, if the previous run did not shutdown gracefully, we may lost the checkpoint file
257                           // and hence we are uncertain that the current local state only contains committed data;
258                           // in that case we need to treat it as a task-corrupted exception
259                           if (eosEnabled && !storeDirIsEmpty) {
260                               log.warn("State store {} did not find checkpoint offsets while stores are not empty, " +
261                                   "since under EOS it has the risk of getting uncommitted data in stores we have to " +
262                                   "treat it as a task corruption error and wipe out the local state of task {} " +
263                                   "before re-bootstrapping", store.stateStore.name(), taskId);
264   
265                               throw new TaskCorruptedException(Collections.singleton(taskId));
266                           } else {
267                               log.info("State store {} did not find checkpoint offset, hence would " +
268                                   "default to the starting offset at changelog {}",
269                                   store.stateStore.name(), store.changelogPartition);
270                           }
271                       }
272                   }  else {
273                       loadedCheckpoints.remove(store.changelogPartition);
274                       log.debug("Skipping re-initialization of offset from checkpoint for recycled store {}",
275                                 store.stateStore.name());
276                   }
277               }
278   
279               if (!loadedCheckpoints.isEmpty()) {
280                   log.warn("Some loaded checkpoint offsets cannot find their corresponding state stores: {}", loadedCheckpoints);
281               }
282   
283               if (eosEnabled) {
284                   checkpointFile.delete();
285               }
286           } catch (final TaskCorruptedException e) {
287               throw e;
288           } catch (final IOException | RuntimeException e) {
289               // both IOException or runtime exception like number parsing can throw
290               throw new ProcessorStateException(format("%sError loading and deleting checkpoint file when creating the state manager",
291                   logPrefix), e);
292           }
293       }

316       @Override
317       public void registerStore(final StateStore store,
318                                 final StateRestoreCallback stateRestoreCallback,
319                                 final CommitCallback commitCallback) {
320           final String storeName = store.name();
321   
322           // TODO (KAFKA-12887): we should not trigger user's exception handler for illegal-argument but always
323           // fail-crash; in this case we would not need to immediately close the state store before throwing
324           if (CHECKPOINT_FILE_NAME.equals(storeName)) {
325               store.close();
326               throw new IllegalArgumentException(format("%sIllegal store name: %s, which collides with the pre-defined " +
327                   "checkpoint file name", logPrefix, storeName));
328           }
329   
330           if (stores.containsKey(storeName)) {
331               store.close();
332               throw new IllegalArgumentException(format("%sStore %s has already been registered.", logPrefix, storeName));
333           }
334   
335           if (stateRestoreCallback instanceof StateRestoreListener) {
336               log.warn("The registered state restore callback is also implementing the state restore listener interface, " +
337                       "which is not expected and would be ignored");
338           }
339   
340           final StateStoreMetadata storeMetadata = isLoggingEnabled(storeName) ?
341               new StateStoreMetadata(
342                   store,
343                   getStorePartition(storeName),
344                   stateRestoreCallback,
345                   commitCallback,
346                   converterForStore(store)) :
347               new StateStoreMetadata(store, commitCallback);
348   
349           // register the store first, so that if later an exception is thrown then eventually while we call `close`
350           // on the state manager this state store would be closed as well
351           stores.put(storeName, storeMetadata);
352   
353           maybeRegisterStoreWithChangelogReader(storeName);
354   
355           log.debug("Registered state store {} to its state manager", storeName);
356       }

388           final Map<TopicPartition, Long> changelogOffsets = new HashMap<>();
389           for (final StateStoreMetadata storeMetadata : stores.values()) {
390               if (storeMetadata.changelogPartition != null) {

425       // used by the changelog reader only
426       void restore(final StateStoreMetadata storeMetadata, final List<ConsumerRecord<byte[], byte[]>> restoreRecords) {
427           if (!stores.containsValue(storeMetadata)) {
428               throw new IllegalStateException("Restoring " + storeMetadata + " which is not registered in this state manager, " +
429                   "this should not happen.");
430           }
431   
432           if (!restoreRecords.isEmpty()) {
433               // restore states from changelog records and update the snapshot offset as the batch end record's offset
434               final Long batchEndOffset = restoreRecords.get(restoreRecords.size() - 1).offset();
435               final RecordBatchingStateRestoreCallback restoreCallback = adapt(storeMetadata.restoreCallback);
436               final List<ConsumerRecord<byte[], byte[]>> convertedRecords = restoreRecords.stream()
437                   .map(storeMetadata.recordConverter::convert)
438                   .collect(Collectors.toList());
439   
440               try {
441                   restoreCallback.restoreBatch(convertedRecords);
442               } catch (final RuntimeException e) {
443                   throw new ProcessorStateException(
444                       format("%sException caught while trying to restore state from %s", logPrefix, storeMetadata.changelogPartition),
445                       e
446                   );
447               }
448   
449               storeMetadata.setOffset(batchEndOffset);

454        * @throws TaskMigratedException recoverable error sending changelog records that would cause the task to be removed
455        * @throws StreamsException fatal error when flushing the state store, for example sending changelog records failed
456        *                          or flushing state store get IO errors; such error should cause the thread to die
457        */
458       @Override
459       public void flush() {
460           RuntimeException firstException = null;
461           // attempting to flush the stores
462           if (!stores.isEmpty()) {
463               log.debug("Flushing all stores registered in the state manager: {}", stores);
464               for (final StateStoreMetadata metadata : stores.values()) {
465                   final StateStore store = metadata.stateStore;
466                   log.trace("Flushing store {}", store.name());
467                   try {
468                       store.flush();
469                   } catch (final RuntimeException exception) {
470                       if (firstException == null) {
471                           // do NOT wrap the error if it is actually caused by Streams itself
472                           if (exception instanceof StreamsException)
473                               firstException = exception;
474                           else
475                               firstException = new ProcessorStateException(
476                                   format("%sFailed to flush state store %s", logPrefix, store.name()), exception);
477                       }
478                       log.error("Failed to flush state store {}: ", store.name(), exception);
479                   }
480               }
481           }
482   
483           if (firstException != null) {
484               throw firstException;
485           }
486       }

491           if (!stores.isEmpty()) {
492               log.debug("Flushing all store caches registered in the state manager: {}", stores);
493               for (final StateStoreMetadata metadata : stores.values()) {
494                   final StateStore store = metadata.stateStore;
495

497                       // buffer should be flushed to send all records to changelog

505                       if (firstException == null) {

509                           } else {
510                               firstException = new ProcessorStateException(

512                                   exception
513                               );
514                           }
515                       }
516                       log.error("Failed to flush cache of store {}: ", store.name(), exception);
517                   }
518               }
519           }
520   
521           if (firstException != null) {
522               throw firstException;
523           }
524       }

557                   }
558               }
559

561           }
562

606       @Override
607       public void checkpoint() {
608           // checkpoint those stores that are only logged and persistent to the checkpoint file
609           final Map<TopicPartition, Long> checkpointingOffsets = new HashMap<>();
610           for (final StateStoreMetadata storeMetadata : stores.values()) {
611               if (storeMetadata.commitCallback != null && !storeMetadata.corrupted) {
612                   try {
613                       storeMetadata.commitCallback.onCommit();
614                   } catch (final IOException e) {
615                       throw new ProcessorStateException(
616                               format("%sException caught while trying to checkpoint store, " +
617                                       "changelog partition %s", logPrefix, storeMetadata.changelogPartition),
618                               e
619                       );
620                   }
621               }
622   
623               // store is logged, persistent, not corrupted, and has a valid current offset
624               if (storeMetadata.changelogPartition != null &&
625                   storeMetadata.stateStore.persistent() &&
626                   !storeMetadata.corrupted) {
627   
628                   final long checkpointableOffset = checkpointableOffsetFromChangelogOffset(storeMetadata.offset);
629                   checkpointingOffsets.put(storeMetadata.changelogPartition, checkpointableOffset);
630               }
631           }
632   
633           log.debug("Writing checkpoint: {}", checkpointingOffsets);
634           try {
635               checkpointFile.write(checkpointingOffsets);
636           } catch (final IOException e) {
637               log.warn("Failed to write offset checkpoint file to [{}]." +
638                   " This may occur if OS cleaned the state.dir in case when it located in ${java.io.tmpdir} directory." +
639                   " This may also occur due to running multiple instances on the same machine using the same state dir." +
640                   " Changing the location of state.dir may resolve the problem.",
641                   checkpointFile, e);
642           }
643       }
