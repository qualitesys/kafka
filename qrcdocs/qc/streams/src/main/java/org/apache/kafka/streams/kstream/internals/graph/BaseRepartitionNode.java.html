26   public abstract class BaseRepartitionNode<K, V> extends GraphNode {
27   
28       protected final Serde<K> keySerde;
29       protected final Serde<V> valueSerde;
30       protected final String sinkName;
31       protected final String sourceName;
32       protected final String repartitionTopic;
33       protected final ProcessorParameters<K, V, ?, ?> processorParameters;
34       protected final StreamPartitioner<K, V> partitioner;
35       protected final InternalTopicProperties internalTopicProperties;
36   
37       BaseRepartitionNode(final String nodeName,
38                           final String sourceName,
39                           final ProcessorParameters<K, V, ?, ?> processorParameters,
40                           final Serde<K> keySerde,
41                           final Serde<V> valueSerde,
42                           final String sinkName,
43                           final String repartitionTopic,
44                           final StreamPartitioner<K, V> partitioner,
45                           final InternalTopicProperties internalTopicProperties) {
46   
47           super(nodeName);
48   
49           this.keySerde = keySerde;
50           this.valueSerde = valueSerde;
51           this.sinkName = sinkName;
52           this.sourceName = sourceName;
53           this.repartitionTopic = repartitionTopic;
54           this.processorParameters = processorParameters;
55           this.partitioner = partitioner;
56           this.internalTopicProperties = internalTopicProperties;
57       }
58   
59       Serializer<V> valueSerializer() {
60           return valueSerde != null ? valueSerde.serializer() : null;
61       }
62   
63       Deserializer<V> valueDeserializer() {
64           return valueSerde != null ? valueSerde.deserializer() : null;
65       }
66   
67       Serializer<K> keySerializer() {
68           return keySerde != null ? keySerde.serializer() : null;
69       }
70   
71       Deserializer<K> keyDeserializer() {
72           return keySerde != null ? keySerde.deserializer() : null;
73       }
74   
75       @Override
76       public String toString() {
77           return "BaseRepartitionNode{" +
78                  "keySerde=" + keySerde +
79                  ", valueSerde=" + valueSerde +
80                  ", sinkName='" + sinkName + '\'' +
81                  ", sourceName='" + sourceName + '\'' +
82                  ", repartitionTopic='" + repartitionTopic + '\'' +
83                  ", processorParameters=" + processorParameters + '\'' +
84                  ", partitioner=" + partitioner +
85                  ", internalTopicProperties=" + internalTopicProperties +
86                  "} " + super.toString();
87       }

96           protected String repartitionTopic;
97           protected StreamPartitioner<K, V> partitioner;
98           protected InternalTopicProperties internalTopicProperties = InternalTopicProperties.empty();
99

108           }
109

111               this.valueSerde = valueSerde;
112               return this;
113           }
114   
115           public BaseRepartitionNodeBuilder<K, V, T> withSinkName(final String sinkName) {

117               return this;
118           }
119   
120           public BaseRepartitionNodeBuilder<K, V, T> withSourceName(final String sourceName) {

123           }
124   
125           public BaseRepartitionNodeBuilder<K, V, T> withRepartitionTopic(final String repartitionTopic) {

138           }
139

142               return this;
143           }
