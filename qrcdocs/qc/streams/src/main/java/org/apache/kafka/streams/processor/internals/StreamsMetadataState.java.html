105        * Find all of the {@link StreamsMetadata}s for a given storeName
106        *
107        * @param storeName the storeName to find metadata for
108        * @return A collection of {@link StreamsMetadata} that have the provided storeName
109        */
110       public synchronized Collection<StreamsMetadata> getAllMetadataForStore(final String storeName) {
111           Objects.requireNonNull(storeName, "storeName cannot be null");
112           if (topologyMetadata.hasNamedTopologies()) {
113               throw new IllegalArgumentException("Cannot invoke the getAllMetadataForStore(storeName) method when"
114                                                      + "using named topologies, please use the overload that accepts"
115                                                      + "a topologyName parameter to identify the correct store");
116           }
117   
118           if (!isInitialized()) {
119               return Collections.emptyList();
120           }
121   
122           if (globalStores.contains(storeName)) {
123               return allMetadata;
124           }
125   
126           final Collection<String> sourceTopics = topologyMetadata.sourceTopicsForStore(storeName, null);
127           if (sourceTopics.isEmpty()) {
128               return Collections.emptyList();
129           }
130   
131           final ArrayList<StreamsMetadata> results = new ArrayList<>();
132           for (final StreamsMetadata metadata : allMetadata) {
133               if (metadata.stateStoreNames().contains(storeName) || metadata.standbyStateStoreNames().contains(storeName)) {
134                   results.add(metadata);
135               }
136           }
137           return results;
138       }

141        * Find all of the {@link StreamsMetadata}s for a given storeName in the given topology
142        *
143        * @param storeName the storeName to find metadata for
144        * @param topologyName the storeName to find metadata for
145        * @return A collection of {@link StreamsMetadata} that have the provided storeName
146        */
147       public synchronized Collection<StreamsMetadata> getAllMetadataForStore(final String storeName, final String topologyName) {
148           Objects.requireNonNull(storeName, "storeName cannot be null");
149           Objects.requireNonNull(topologyName, "topologyName cannot be null");
150   
151           if (!isInitialized()) {
152               return Collections.emptyList();
153           }
154   
155           final Collection<String> sourceTopics = topologyMetadata.sourceTopicsForStore(storeName, topologyName);
156           if (sourceTopics.isEmpty()) {
157               return Collections.emptyList();
158           }
159   
160           final ArrayList<StreamsMetadata> results = new ArrayList<>();
161           for (final StreamsMetadata metadata : allMetadata) {
162               final String metadataTopologyName = ((StreamsMetadataImpl) metadata).topologyName();
163               if (metadataTopologyName != null && metadataTopologyName.equals(topologyName)
164                   && metadata.stateStoreNames().contains(storeName) || metadata.standbyStateStoreNames().contains(storeName)) {
165                   results.add(metadata);
166               }
167           }
168           return results;
169       }

233        * Find the {@link KeyQueryMetadata}s for a given storeName and key
234        *
235        * Note: the key may not exist in the {@link StateStore},this method provides a way of finding which
236        * {@link StreamsMetadata} it would exist on.
237        *
238        * @param storeName   Name of the store
239        * @param key         Key to use
240        * @param partitioner partitioner to use to find correct partition for key

242        * @return The {@link KeyQueryMetadata} for the storeName and key or {@link KeyQueryMetadata#NOT_AVAILABLE}
243        * if streams is (re-)initializing, or {@code null} if no matching metadata could be found.
244        */
245       public synchronized <K> KeyQueryMetadata getKeyQueryMetadataForKey(final String storeName,
246                                                                          final K key,
247                                                                          final StreamPartitioner<? super K, ?> partitioner) {
248           Objects.requireNonNull(storeName, "storeName can't be null");
249           Objects.requireNonNull(key, "key can't be null");
250           Objects.requireNonNull(partitioner, "partitioner can't be null");
251           if (topologyMetadata.hasNamedTopologies()) {
252               throw new IllegalArgumentException("Cannot invoke the getKeyQueryMetadataForKey(storeName, key, partitioner)"
253                                                      + "method when using named topologies, please use the overload that"
254                                                      + "accepts a topologyName parameter to identify the correct store");
255           }
256   
257           if (!isInitialized()) {
258               return KeyQueryMetadata.NOT_AVAILABLE;
259           }
260   
261           if (globalStores.contains(storeName)) {
262               // global stores are on every node. if we don't have the host info
263               // for this host then just pick the first metadata
264               if (thisHost.equals(UNKNOWN_HOST)) {
265                   return new KeyQueryMetadata(allMetadata.get(0).hostInfo(), Collections.emptySet(), -1);
266               }
267               return new KeyQueryMetadata(localMetadata.get().hostInfo(), Collections.emptySet(), -1);
268           }
269   
270           final SourceTopicsInfo sourceTopicsInfo = getSourceTopicsInfo(storeName);
271           if (sourceTopicsInfo == null) {
272               return null;
273           }
274           return getKeyQueryMetadataForKey(storeName, key, partitioner, sourceTopicsInfo);
275       }

316       private boolean hasPartitionsForAnyTopics(final List<String> topicNames, final Set<TopicPartition> partitionForHost) {
317           for (final TopicPartition topicPartition : partitionForHost) {
318               if (topicNames.contains(topicPartition.topic())) {
319                   return true;
320               }
321           }
322           return false;
323       }

325       private Set<String> getStoresOnHost(final Map<String, List<String>> storeToSourceTopics,
326                                           final Set<TopicPartition> sourceTopicPartitions) {
327           final Set<String> storesOnHost = new HashSet<>();
328           for (final Map.Entry<String, List<String>> storeTopicEntry : storeToSourceTopics.entrySet()) {
329               final List<String> topicsForStore = storeTopicEntry.getValue();
330               if (hasPartitionsForAnyTopics(topicsForStore, sourceTopicPartitions)) {
331                   storesOnHost.add(storeTopicEntry.getKey());
332               }
333           }
334           return storesOnHost;
335       }

337       private void rebuildMetadata(final Map<HostInfo, Set<TopicPartition>> activePartitionHostMap,
338                                    final Map<HostInfo, Set<TopicPartition>> standbyPartitionHostMap) {
339           if (activePartitionHostMap.isEmpty() && standbyPartitionHostMap.isEmpty()) {
340               allMetadata = Collections.emptyList();
341               localMetadata.set(new StreamsMetadataImpl(
342                   thisHost,
343                   Collections.emptySet(),
344                   Collections.emptySet(),
345                   Collections.emptySet(),
346                   Collections.emptySet()
347               ));
348               return;
349           }
350   
351           allMetadata = topologyMetadata.hasNamedTopologies() ?
352               rebuildMetadataForNamedTopologies(activePartitionHostMap, standbyPartitionHostMap) :
353               rebuildMetadataForSingleTopology(activePartitionHostMap, standbyPartitionHostMap);
354       }

356       private List<StreamsMetadata> rebuildMetadataForNamedTopologies(final Map<HostInfo, Set<TopicPartition>> activePartitionHostMap,
357                                                                       final Map<HostInfo, Set<TopicPartition>> standbyPartitionHostMap) {
358           final List<StreamsMetadata> rebuiltMetadata = new ArrayList<>();
359           Stream.concat(activePartitionHostMap.keySet().stream(), standbyPartitionHostMap.keySet().stream())
360               .distinct()
361               .sorted(Comparator.comparing(HostInfo::host).thenComparingInt(HostInfo::port))
362               .forEach(hostInfo -> {
363                   for (final String topologyName : topologyMetadata.namedTopologiesView()) {
364                       final Map<String, List<String>> storeToSourceTopics =
365                           topologyMetadata.stateStoreNameToSourceTopicsForTopology(topologyName);
366   
367                       final Set<TopicPartition> activePartitionsOnHost = new HashSet<>();
368                       final Set<String> activeStoresOnHost = new HashSet<>();
369                       if (activePartitionHostMap.containsKey(hostInfo)) {
370                           // filter out partitions for topics that are not connected to this topology
371                           activePartitionsOnHost.addAll(
372                               activePartitionHostMap.get(hostInfo).stream()
373                                   .filter(tp -> topologyMetadata.fullSourceTopicNamesForTopology(topologyName).contains(tp.topic()))
374                                   .collect(Collectors.toSet())
375                           );
376                           activeStoresOnHost.addAll(getStoresOnHost(storeToSourceTopics, activePartitionsOnHost));
377                       }
378   
379                       // TODO KAFKA-13281: when we add support for global stores with named topologies we will
380                       //  need to add the global stores to the activeStoresOnHost set
381   
382                       final Set<TopicPartition> standbyPartitionsOnHost = new HashSet<>();
383                       final Set<String> standbyStoresOnHost = new HashSet<>();
384                       if (standbyPartitionHostMap.containsKey(hostInfo)) {
385                           standbyPartitionsOnHost.addAll(
386                               standbyPartitionHostMap.get(hostInfo).stream()
387                                   .filter(tp -> topologyMetadata.fullSourceTopicNamesForTopology(topologyName).contains(tp.topic()))
388                                   .collect(Collectors.toSet()));
389                           standbyStoresOnHost.addAll(getStoresOnHost(storeToSourceTopics, standbyPartitionsOnHost));
390                       }
391   
392                       if (!(activeStoresOnHost.isEmpty() && activePartitionsOnHost.isEmpty() && standbyStoresOnHost.isEmpty() && standbyPartitionsOnHost.isEmpty())) {
393                           final StreamsMetadata metadata = new StreamsMetadataImpl(
394                               hostInfo,
395                               activeStoresOnHost,
396                               activePartitionsOnHost,
397                               standbyStoresOnHost,
398                               standbyPartitionsOnHost,
399                               topologyName
400                           );
401   
402                           rebuiltMetadata.add(metadata);
403                           if (hostInfo.equals(thisHost)) {
404                               localMetadata.set(metadata);
405                           }
406                       } else {
407                           log.debug("Host {} has no tasks for topology {} at the moment, this metadata will not be built", hostInfo, topologyName);
408                       }
409                   }
410   
411                   // Construct metadata across all topologies on this host for the `localMetadata` field
412                   final Map<String, List<String>> storeToSourceTopics = topologyMetadata.stateStoreNameToSourceTopics();
413                   final Set<TopicPartition> localActivePartitions = activePartitionHostMap.get(thisHost);
414                   final Set<TopicPartition> localStandbyPartitions = standbyPartitionHostMap.get(thisHost);
415   
416                   localMetadata.set(
417                       new StreamsMetadataImpl(thisHost,
418                                               getStoresOnHost(storeToSourceTopics, localActivePartitions),
419                                               localActivePartitions,
420                                               getStoresOnHost(storeToSourceTopics, localStandbyPartitions),
421                                               localStandbyPartitions)
422                   );
423               });
424           return rebuiltMetadata;
425       }

427       private List<StreamsMetadata> rebuildMetadataForSingleTopology(final Map<HostInfo, Set<TopicPartition>> activePartitionHostMap,
428                                                                      final Map<HostInfo, Set<TopicPartition>> standbyPartitionHostMap) {
429           final List<StreamsMetadata> rebuiltMetadata = new ArrayList<>();
430           final Map<String, List<String>> storeToSourceTopics = topologyMetadata.stateStoreNameToSourceTopics();
