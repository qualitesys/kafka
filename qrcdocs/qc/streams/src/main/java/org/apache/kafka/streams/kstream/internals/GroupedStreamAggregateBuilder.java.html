34   class GroupedStreamAggregateBuilder<K, V> {
35   
36       private final InternalStreamsBuilder builder;
37       private final Serde<K> keySerde;
38       private final Serde<V> valueSerde;
39       private final boolean repartitionRequired;
40       private final String userProvidedRepartitionTopicName;
41       private final Set<String> subTopologySourceNodes;
42       private final String name;
43       private final GraphNode graphNode;
44       private GraphNode repartitionNode;
45   
46       final Initializer<Long> countInitializer = () -> 0L;
47   
48       final Aggregator<K, V, Long> countAggregator = (aggKey, value, aggregate) -> aggregate + 1;
49   
50       final Initializer<V> reduceInitializer = () -> null;
51   
52       GroupedStreamAggregateBuilder(final InternalStreamsBuilder builder,
53                                     final GroupedInternal<K, V> groupedInternal,
54                                     final boolean repartitionRequired,
55                                     final Set<String> subTopologySourceNodes,
56                                     final String name,
57                                     final GraphNode graphNode) {
58   
59           this.builder = builder;
60           this.keySerde = groupedInternal.keySerde();
61           this.valueSerde = groupedInternal.valueSerde();
62           this.repartitionRequired = repartitionRequired;
63           this.subTopologySourceNodes = subTopologySourceNodes;
64           this.name = name;
65           this.graphNode = graphNode;
66           this.userProvidedRepartitionTopicName = groupedInternal.name();
67       }
68   
69       <KR, VR> KTable<KR, VR> build(final NamedInternal functionName,
70                                     final StoreBuilder<?> storeBuilder,
71                                     final KStreamAggProcessorSupplier<K, V, KR, VR> aggregateSupplier,
72                                     final String queryableStoreName,
73                                     final Serde<KR> keySerde,
74                                     final Serde<VR> valueSerde) {
75           assert queryableStoreName == null || queryableStoreName.equals(storeBuilder.name());
76   
77           final String aggFunctionName = functionName.name();
78   
79           String sourceName = this.name;
80           GraphNode parentNode = graphNode;
81   
82           if (repartitionRequired) {
83               final OptimizableRepartitionNodeBuilder<K, V> repartitionNodeBuilder = optimizableRepartitionNodeBuilder();
84               final String repartitionTopicPrefix = userProvidedRepartitionTopicName != null ? userProvidedRepartitionTopicName : storeBuilder.name();
85               sourceName = createRepartitionSource(repartitionTopicPrefix, repartitionNodeBuilder);
86   
87               // First time through we need to create a repartition node.
88               // Any subsequent calls to GroupedStreamAggregateBuilder#build we check if
89               // the user has provided a name for the repartition topic, is so we re-use
90               // the existing repartition node, otherwise we create a new one.
91               if (repartitionNode == null || userProvidedRepartitionTopicName == null) {
92                   repartitionNode = repartitionNodeBuilder.build();
93               }
94   
95               builder.addGraphNode(parentNode, repartitionNode);
96               parentNode = repartitionNode;
97           }
98   
99           final StatefulProcessorNode<K, V> statefulProcessorNode =
100               new StatefulProcessorNode<>(
101                   aggFunctionName,
102                   new ProcessorParameters<>(aggregateSupplier, aggFunctionName),
103                   storeBuilder
104               );
105   
106           builder.addGraphNode(parentNode, statefulProcessorNode);
107   
108           return new KTableImpl<>(aggFunctionName,
109                                   keySerde,
110                                   valueSerde,
111                                   sourceName.equals(this.name) ? subTopologySourceNodes : Collections.singleton(sourceName),
112                                   queryableStoreName,
113                                   aggregateSupplier,
114                                   statefulProcessorNode,
115                                   builder);
116   
117       }
118   
119       /**
120        * @return the new sourceName of the repartitioned source
121        */
122       private String createRepartitionSource(final String repartitionTopicNamePrefix,
123                                              final OptimizableRepartitionNodeBuilder<K, V> optimizableRepartitionNodeBuilder) {
124   
125           return KStreamImpl.createRepartitionedSource(builder,
126                                                        keySerde,
127                                                        valueSerde,
128                                                        repartitionTopicNamePrefix,
129                                                        null,
130                                                        optimizableRepartitionNodeBuilder);
131   
132       }
