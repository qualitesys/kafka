29   /**
30    * Provides a set of {@link StoreBuilder}s that will be automatically added to the topology and connected to the
31    * associated processor.
32    * <p>
33    * Implementing this interface is recommended when the associated processor wants to encapsulate its usage of its state
34    * stores, rather than exposing them to the user building the topology.
35    * <p>
36    * In the event that separate but related processors may want to share the same store, different {@link ConnectedStoreProvider}s
37    * may provide the same instance of {@link StoreBuilder}, as shown below.
38    * <pre>{@code
39    * class StateSharingProcessors {
40    *     StoreBuilder<KeyValueStore<String, String>> storeBuilder =
41    *         Stores.keyValueStoreBuilder(Stores.persistentKeyValueStore("myStore"), Serdes.String(), Serdes.String());
42    *
43    *     class SupplierA implements ProcessorSupplier<String, Integer> {
44    *         Processor<String, Integer> get() {
45    *             return new Processor() {
46    *                 private StateStore store;
47    *
48    *                 void init(ProcessorContext context) {
49    *                     this.store = context.getStateStore("myStore");
50    *                 }
51    *
52    *                 void process(String key, Integer value) {
53    *                     // can access this.store
54    *                 }
55    *
56    *                 void close() {
57    *                     // can access this.store
58    *                 }
59    *             }
60    *         }
61    *
62    *         Set<StoreBuilder<?>> stores() {
63    *             return Collections.singleton(storeBuilder);
64    *         }
65    *     }
66    *
67    *     class SupplierB implements ProcessorSupplier<String, String> {
68    *         Processor<String, String> get() {
69    *             return new Processor() {
70    *                 private StateStore store;
71    *
72    *                 void init(ProcessorContext context) {
73    *                     this.store = context.getStateStore("myStore");
74    *                 }
75    *
76    *                 void process(String key, String value) {
77    *                     // can access this.store
78    *                 }
79    *
80    *                 void close() {
81    *                     // can access this.store
82    *                 }
83    *             }
84    *         }
85    *
86    *         Set<StoreBuilder<?>> stores() {
87    *             return Collections.singleton(storeBuilder);
88    *         }
89    *     }
90    * }
91    * }</pre>
92    *
93    * @see Topology#addProcessor(String, org.apache.kafka.streams.processor.api.ProcessorSupplier, String...)
94    * @see KStream#process(org.apache.kafka.streams.processor.api.ProcessorSupplier, String...)
95    * @see KStream#process(org.apache.kafka.streams.processor.api.ProcessorSupplier, Named, String...)
96    * @see KStream#transform(TransformerSupplier, String...)
97    * @see KStream#transform(TransformerSupplier, Named, String...)
98    * @see KStream#transformValues(ValueTransformerSupplier, String...)
99    * @see KStream#transformValues(ValueTransformerSupplier, Named, String...)
100    * @see KStream#transformValues(ValueTransformerWithKeySupplier, String...)
101    * @see KStream#transformValues(ValueTransformerWithKeySupplier, Named, String...)
102    * @see KStream#flatTransform(TransformerSupplier, String...)
103    * @see KStream#flatTransform(TransformerSupplier, Named, String...)
104    * @see KStream#flatTransformValues(ValueTransformerSupplier, String...)
105    * @see KStream#flatTransformValues(ValueTransformerSupplier, Named, String...)
106    * @see KStream#flatTransformValues(ValueTransformerWithKeySupplier, String...)
107    * @see KStream#flatTransformValues(ValueTransformerWithKeySupplier, Named, String...)
108    */
109   public interface ConnectedStoreProvider {
110   
111       /**
112        * @return the state stores to be connected and added, or null if no stores should be automatically connected and added.
113        */
114       default Set<StoreBuilder<?>> stores() {
115           return null;
116       }
117   }
