115   /**
116    * A Kafka client that allows for performing continuous computation on input coming from one or more input topics and
117    * sends output to zero, one, or more output topics.
118    * <p>
119    * The computational logic can be specified either by using the {@link Topology} to define a DAG topology of
120    * {@link org.apache.kafka.streams.processor.api.Processor}s or by using the {@link StreamsBuilder} which provides the high-level DSL to define
121    * transformations.
122    * <p>
123    * One {@code KafkaStreams} instance can contain one or more threads specified in the configs for the processing work.
124    * <p>
125    * A {@code KafkaStreams} instance can co-ordinate with any other instances with the same
126    * {@link StreamsConfig#APPLICATION_ID_CONFIG application ID} (whether in the same process, on other processes on this
127    * machine, or on remote machines) as a single (possibly distributed) stream processing application.
128    * These instances will divide up the work based on the assignment of the input topic partitions so that all partitions
129    * are being consumed.
130    * If instances are added or fail, all (remaining) instances will rebalance the partition assignment among themselves
131    * to balance processing load and ensure that all input topic partitions are processed.
132    * <p>
133    * Internally a {@code KafkaStreams} instance contains a normal {@link KafkaProducer} and {@link KafkaConsumer} instance
134    * that is used for reading input and writing output.
135    * <p>
136    * A simple example might look like this:
137    * <pre>{@code
138    * Properties props = new Properties();
139    * props.put(StreamsConfig.APPLICATION_ID_CONFIG, "my-stream-processing-application");
140    * props.put(StreamsConfig.BOOTSTRAP_SERVERS_CONFIG, "localhost:9092");
141    * props.put(StreamsConfig.DEFAULT_KEY_SERDE_CLASS_CONFIG, Serdes.String().getClass());
142    * props.put(StreamsConfig.DEFAULT_VALUE_SERDE_CLASS_CONFIG, Serdes.String().getClass());
143    *
144    * StreamsBuilder builder = new StreamsBuilder();
145    * builder.<String, String>stream("my-input-topic").mapValues(value -> String.valueOf(value.length())).to("my-output-topic");
146    *
147    * KafkaStreams streams = new KafkaStreams(builder.build(), props);
148    * streams.start();
149    * }</pre>
150    *
151    * @see org.apache.kafka.streams.StreamsBuilder
152    * @see org.apache.kafka.streams.Topology
153    */

203        *         |       +-----+--------+
204        *         |             |
205        *         |             v
206        *         |       +----+--+------+
207        *         |       | Re-          |
208        *         +&lt;----- | Balancing (1)| --------&gt;+
209        *         |       +-----+-+------+          |
210        *         |             | ^                 |
211        *         |             v |                 |
212        *         |       +--------------+          v
213        *         |       | Running (2)  | --------&gt;+
214        *         |       +------+-------+          |
215        *         |              |                  |
216        *         |              v                  |
217        *         |       +------+-------+     +----+-------+
218        *         +-----&gt; | Pending      |     | Pending    |
219        *                 | Shutdown (3) |     | Error (5)  |
220        *                 +------+-------+     +-----+------+

283       private boolean waitOnState(final State targetState, final long waitMs) {
284           final long begin = time.milliseconds();
285           synchronized (stateLock) {
286               boolean interrupted = false;
287               long elapsedMs = 0L;
288               try {
289                   while (state != targetState) {
290                       if (waitMs > elapsedMs) {
291                           final long remainingMs = waitMs - elapsedMs;
292                           try {
293                               stateLock.wait(remainingMs);
294                           } catch (final InterruptedException e) {
295                               interrupted = true;
296                           }
297                       } else {
298                           log.debug("Cannot transit to {} within {}ms", targetState, waitMs);
299                           return false;
300                       }
301                       elapsedMs = time.milliseconds() - begin;
302                   }
303               } finally {
304                   // Make sure to restore the interruption status before returning.
305                   // We do not always own the current thread that executes this method, i.e., we do not know the
306                   // interruption policy of the thread. The least we can do is restore the interruption status before
307                   // the current thread exits this method.
308                   if (interrupted) {
309                       Thread.currentThread().interrupt();
310                   }
311               }
312               return true;
313           }
314       }

316       /**
317        * Sets the state
318        * @param newState New state
319        */
320       private boolean setState(final State newState) {
321           final State oldState;
322   
323           synchronized (stateLock) {
324               oldState = state;
325   
326               if (state == State.PENDING_SHUTDOWN && newState != State.NOT_RUNNING) {
327                   // when the state is already in PENDING_SHUTDOWN, all other transitions than NOT_RUNNING (due to thread dying) will be
328                   // refused but we do not throw exception here, to allow appropriate error handling
329                   return false;
330               } else if (state == State.NOT_RUNNING && (newState == State.PENDING_SHUTDOWN || newState == State.NOT_RUNNING)) {
331                   // when the state is already in NOT_RUNNING, its transition to PENDING_SHUTDOWN or NOT_RUNNING (due to consecutive close calls)
332                   // will be refused but we do not throw exception here, to allow idempotent close calls
333                   return false;
334               } else if (state == State.REBALANCING && newState == State.REBALANCING) {
335                   // when the state is already in REBALANCING, it should not transit to REBALANCING again
336                   return false;
337               } else if (state == State.ERROR && (newState == State.PENDING_ERROR || newState == State.ERROR)) {
338                   // when the state is already in ERROR, its transition to PENDING_ERROR or ERROR (due to consecutive close calls)
339                   return false;
340               } else if (state == State.PENDING_ERROR && newState != State.ERROR) {
341                   // when the state is already in PENDING_ERROR, all other transitions than ERROR (due to thread dying) will be
342                   // refused but we do not throw exception here, to allow appropriate error handling
343                   return false;
344               } else if (!state.isValidTransition(newState)) {
345                   throw new IllegalStateException("Stream-client " + clientId + ": Unexpected state transition from " + oldState + " to " + newState);
346               } else {
347                   log.info("State transition from {} to {}", oldState, newState);
348               }
349               state = newState;
350               stateLock.notifyAll();
351           }
352   
353           // we need to call the user customized state listener outside the state lock to avoid potential deadlocks
354           if (stateListener != null) {
355               stateListener.onChange(newState, oldState);
356           }
357   
358           return true;
359       }

534       private void handleStreamsUncaughtException(final Throwable throwable,
535                                                   final StreamsUncaughtExceptionHandler streamsUncaughtExceptionHandler,
536                                                   final boolean skipThreadReplacement) {
537           final StreamsUncaughtExceptionHandler.StreamThreadExceptionResponse action = getActionForThrowable(throwable, streamsUncaughtExceptionHandler);
538           if (oldHandler) {
539               log.warn("Stream's new uncaught exception handler is set as well as the deprecated old handler." +
540                       "The old handler will be ignored as long as a new handler is set.");
541           }
542           switch (action) {
543               case REPLACE_THREAD:
544                   if (!skipThreadReplacement) {
545                       log.error("Replacing thread in the streams uncaught exception handler", throwable);
546                       replaceStreamThread(throwable);
547                   } else {
548                       log.debug("Skipping thread replacement for recoverable error");
549                   }
550                   break;
551               case SHUTDOWN_CLIENT:
552                   log.error("Encountered the following exception during processing " +
553                           "and Kafka Streams opted to " + action + "." +
554                           " The streams client is going to shut down now. ", throwable);
555                   closeToError();
556                   break;
557               case SHUTDOWN_APPLICATION:
558                   if (getNumLiveStreamThreads() == 1) {
559                       log.warn("Attempt to shut down the application requires adding a thread to communicate the shutdown. No processing will be done on this thread");
560                       addStreamThread();
561                   }
562                   if (throwable instanceof Error) {
563                       log.error("This option requires running threads to shut down the application." +
564                               "but the uncaught exception was an Error, which means this runtime is no " +
565                               "longer in a well-defined state. Attempting to send the shutdown command anyway.", throwable);
566                   }
567                   if (Thread.currentThread().equals(globalStreamThread) && getNumLiveStreamThreads() == 0) {
568                       log.error("Exception in global thread caused the application to attempt to shutdown." +
569                               " This action will succeed only if there is at least one StreamThread running on this client." +
570                               " Currently there are no running threads so will now close the client.");
571                       closeToError();
572                       break;
573                   }
574                   processStreamThread(thread -> thread.sendShutdownRequest(AssignorError.SHUTDOWN_REQUESTED));
575                   log.error("Encountered the following exception during processing " +
576                           "and sent shutdown request for the entire application.", throwable);
577                   break;
578           }
579       }
