44           @Override
45           public Bytes upperRange(final Bytes key, final long to) {
46               if (key == null) {
47                   // Put next prefix instead of null so that we can start from right prefix
48                   // when scanning backwards
49                   final byte nextPrefix = TIME_FIRST_PREFIX + 1;
50                   return Bytes.wrap(ByteBuffer.allocate(PREFIX_SIZE).put(nextPrefix).array());
51               }
52               return Bytes.wrap(ByteBuffer.allocate(PREFIX_SIZE + 2 * TIMESTAMP_SIZE + key.get().length)
53                   .put(TIME_FIRST_PREFIX)
54                   // the end timestamp can be as large as possible as long as it's larger than start time
55                   .putLong(Long.MAX_VALUE)
56                   // this is the start timestamp
57                   .putLong(to)
58                   .put(key.get())
59                   .array());
60           }

62           @Override
63           public Bytes lowerRange(final Bytes key, final long from) {
64               if (key == null) {
65                   return Bytes.wrap(ByteBuffer.allocate(PREFIX_SIZE + TIMESTAMP_SIZE)
66                       .put(TIME_FIRST_PREFIX)
67                       .putLong(from)
68                       .array());
69               }
70   
71               return Bytes.wrap(ByteBuffer.allocate(PREFIX_SIZE + 2 * TIMESTAMP_SIZE + key.get().length)
72                   .put(TIME_FIRST_PREFIX)
73                   .putLong(from)
74                   .putLong(0L)
75                   .put(key.get())
76                   .array());
77           }

106           @Override
107           public HasNextCondition hasNextCondition(final Bytes binaryKeyFrom,
108               final Bytes binaryKeyTo, final long from, final long to, final boolean forward) {
109               return iterator -> {
110                   while (iterator.hasNext()) {
111                       final Bytes bytes = iterator.peekNextKey();
112                       final byte prefix = extractPrefix(bytes.get());
113   
114                       if (prefix != TIME_FIRST_PREFIX) {
115                           return false;
116                       }
117   
118                       final Windowed<Bytes> windowedKey = from(bytes);
119                       final long endTime = windowedKey.window().end();
120                       final long startTime = windowedKey.window().start();
121   
122                       // We can return false directly here since keys are sorted by end time and if
123                       // we get time smaller than `from`, there won't be time within range.
124                       if (!forward && endTime < from) {
125                           return false;
126                       }
127   
128                       if ((binaryKeyFrom == null || windowedKey.key().compareTo(binaryKeyFrom) >= 0)
129                           && (binaryKeyTo == null || windowedKey.key().compareTo(binaryKeyTo) <= 0)
130                           && endTime >= from && startTime <= to) {
131                           return true;
132                       }
133                       iterator.next();
134                   }
135                   return false;
136               };
137           }

219           public static byte[] extractWindowBytesFromNonPrefixSessionKey(final byte[] binaryKey) {
220               final ByteBuffer buffer = ByteBuffer.allocate(PREFIX_SIZE + binaryKey.length).put(TIME_FIRST_PREFIX);
221               // Put timestamp
222               buffer.put(binaryKey, binaryKey.length - 2 * TIMESTAMP_SIZE, 2 * TIMESTAMP_SIZE);

224   
225               return buffer.array();
226           }

365           private static Bytes wrapPrefix(final Bytes noPrefixKey, final boolean upperRange) {
366               // Need to scan from prefix even key is null
367               if (noPrefixKey == null) {
368                   final byte prefix = upperRange ? KEY_FIRST_PREFIX + 1 : KEY_FIRST_PREFIX;
369                   final byte[] ret = ByteBuffer.allocate(PREFIX_SIZE)
370                       .put(prefix)
371                       .array();
372                   return Bytes.wrap(ret);
373               }
374               final byte[] ret = ByteBuffer.allocate(PREFIX_SIZE + noPrefixKey.get().length)
375                   .put(KEY_FIRST_PREFIX)
376                   .put(noPrefixKey.get())
377                   .array();
378               return Bytes.wrap(ret);
379           }
