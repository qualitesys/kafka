173       public void executeTopologyUpdatesAndBumpThreadVersion(final Consumer<Set<String>> handleTopologyAdditions,
174                                                              final Consumer<Set<String>> handleTopologyRemovals) {
175           try {

178               handleTopologyAdditions.accept(namedTopologiesView());
179               handleTopologyRemovals.accept(namedTopologiesView());

186       public void maybeNotifyTopologyVersionListeners() {
187           try {
188               lock();
189               final long minThreadVersion = getMinimumThreadVersion();
190               final Iterator<TopologyVersionListener> iterator = version.activeTopologyUpdateListeners.listIterator();
191               TopologyVersionListener topologyVersionListener;
192               while (iterator.hasNext()) {
193                   topologyVersionListener = iterator.next();
194                   final long topologyVersionWaitersVersion = topologyVersionListener.topologyVersion;
195                   if (minThreadVersion >= topologyVersionWaitersVersion) {
196                       topologyVersionListener.future.complete(null);
197                       iterator.remove();
198                       log.info("All threads are now on topology version {}", topologyVersionListener.topologyVersion);
199                   }
200               }
201           } finally {
202               unlock();
203           }
204       }

240        * Adds the topology and registers a future that listens for all threads on the older version to see the update
241        */
242       public void registerAndBuildNewTopology(final KafkaFutureImpl<Void> future, final InternalTopologyBuilder newTopologyBuilder) {
243           try {
244               lock();
245               buildAndVerifyTopology(newTopologyBuilder);
246               log.info("New NamedTopology {} passed validation and will be added, old topology version is {}", newTopologyBuilder.topologyName(), version.topologyVersion.get());
247               version.topologyVersion.incrementAndGet();
248               version.activeTopologyUpdateListeners.add(new TopologyVersionListener(topologyVersion(), future));
249               builders.put(newTopologyBuilder.topologyName(), newTopologyBuilder);
250               wakeupThreads();
251               log.info("Added NamedTopology {} and updated topology version to {}", newTopologyBuilder.topologyName(), version.topologyVersion.get());
252           } catch (final Throwable throwable) {
253               log.error("Failed to add NamedTopology {}, please retry the operation.", newTopologyBuilder.topologyName());
254               future.completeExceptionally(throwable);
255           } finally {
256               unlock();
257           }
258       }

263       public KafkaFuture<Void> unregisterTopology(final KafkaFutureImpl<Void> removeTopologyFuture,
264                                                   final String topologyName) {
265           try {
266               lock();
267               log.info("Beginning removal of NamedTopology {}, old topology version is {}", topologyName, version.topologyVersion.get());
268               version.topologyVersion.incrementAndGet();
269               version.activeTopologyUpdateListeners.add(new TopologyVersionListener(topologyVersion(), removeTopologyFuture));
270               final InternalTopologyBuilder removedBuilder = builders.remove(topologyName);
271               removedBuilder.fullSourceTopicNames().forEach(allInputTopics::remove);
272               removedBuilder.allSourcePatternStrings().forEach(allInputTopics::remove);
273               log.info("Finished removing NamedTopology {}, topology version was updated to {}", topologyName, version.topologyVersion.get());
274           } catch (final Throwable throwable) {
275               log.error("Failed to remove NamedTopology {}, please retry.", topologyName);
276               removeTopologyFuture.completeExceptionally(throwable);
277           } finally {
278               unlock();
279           }
280           return removeTopologyFuture;
281       }

292       private void buildAndVerifyTopology(final InternalTopologyBuilder builder) {
293           builder.rewriteTopology(config);
294           builder.buildTopology();
295   
296           final Set<String> allInputTopicsCopy = new HashSet<>(allInputTopics);
297   
298           // As we go, check each topology for overlap in the set of input topics/patterns
299           final int numInputTopics = allInputTopicsCopy.size();
300           final List<String> inputTopics = builder.fullSourceTopicNames();
301           final Collection<String> inputPatterns = builder.allSourcePatternStrings();
302   
303           final Set<String> newInputTopics = new HashSet<>(inputTopics);
304           newInputTopics.addAll(inputPatterns);
305   
306           final int numNewInputTopics = newInputTopics.size();
307           allInputTopicsCopy.addAll(newInputTopics);
308   
309           if (allInputTopicsCopy.size() != numInputTopics + numNewInputTopics) {
310               inputTopics.retainAll(allInputTopicsCopy);
311               inputPatterns.retainAll(allInputTopicsCopy);
312               log.error("Tried to add the NamedTopology {} but it had overlap with other input topics {} or patterns {}",
313                         builder.topologyName(), inputTopics, inputPatterns);
314               throw new TopologyException("Named Topologies may not subscribe to the same input topics or patterns");
315           }
316   
317           final ProcessorTopology globalTopology = builder.buildGlobalStateTopology();
318           if (globalTopology != null) {
319               if (builder.topologyName() != null) {
320                   throw new IllegalStateException("Global state stores are not supported with Named Topologies");
321               } else if (this.globalTopology != null) {
322                   throw new TopologyException("Topology builder had global state, but global topology has already been set");
323               } else {
324                   this.globalTopology = globalTopology;
325                   globalStateStores.putAll(builder.globalStateStores());
326               }
327           }
328           allInputTopics.addAll(newInputTopics);
329       }

331       public int getNumStreamThreads(final StreamsConfig config) {
332           final int configuredNumStreamThreads = config.getInt(StreamsConfig.NUM_STREAM_THREADS_CONFIG);
333   
334           // If there are named topologies but some are empty, this indicates a bug in user code
335           if (hasNamedTopologies()) {
336               if (hasNoLocalTopology()) {
337                   log.error("Detected a named topology with no input topics, a named topology may not be empty.");
338                   throw new TopologyException("Topology has no stream threads and no global threads, " +
339                                                   "must subscribe to at least one source topic or pattern.");
340               }
341           } else {
342               // If both the global and non-global topologies are empty, this indicates a bug in user code
343               if (hasNoLocalTopology() && !hasGlobalTopology()) {
344                   log.error("Topology with no input topics will create no stream threads and no global thread.");
345                   throw new TopologyException("Topology has no stream threads and no global threads, " +
346                                                   "must subscribe to at least one source topic or global table.");
347               }
348           }
349   
350           // Lastly we check for an empty non-global topology and override the threads to zero if set otherwise
351           if (configuredNumStreamThreads != 0 && hasNoLocalTopology()) {
352               log.info("Overriding number of StreamThreads to zero for global-only topology");
353               return 0;
354           }
355   
356           return configuredNumStreamThreads;
357       }
