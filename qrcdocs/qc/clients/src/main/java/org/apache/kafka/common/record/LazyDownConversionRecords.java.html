27   /**
28    * Encapsulation for holding records that require down-conversion in a lazy, chunked manner (KIP-283). See
29    * {@link LazyDownConversionRecordsSend} for the actual chunked send implementation.
30    */
31   public class LazyDownConversionRecords implements BaseRecords {
32       private final TopicPartition topicPartition;
33       private final Records records;
34       private final byte toMagic;
35       private final long firstOffset;
36       private ConvertedRecords firstConvertedBatch;
37       private final int sizeInBytes;
38       private final Time time;
39   
40       /**
41        * @param topicPartition The topic-partition to which records belong
42        * @param records Records to lazily down-convert
43        * @param toMagic Magic version to down-convert to
44        * @param firstOffset The starting offset for down-converted records. This only impacts some cases. See
45        *                    {@link RecordsUtil#downConvert(Iterable, byte, long, Time)} for an explanation.
46        * @param time The time instance to use
47        *
48        * @throws org.apache.kafka.common.errors.UnsupportedCompressionTypeException If the first batch to down-convert
49        *    has a compression type which we do not support down-conversion for.
50        */
51       public LazyDownConversionRecords(TopicPartition topicPartition, Records records, byte toMagic, long firstOffset, Time time) {
52           this.topicPartition = Objects.requireNonNull(topicPartition);
53           this.records = Objects.requireNonNull(records);
54           this.toMagic = toMagic;
55           this.firstOffset = firstOffset;
56           this.time = Objects.requireNonNull(time);
57   
58           // To make progress, kafka consumers require at least one full record batch per partition, i.e. we need to
59           // ensure we can accommodate one full batch of down-converted messages. We achieve this by having `sizeInBytes`
60           // factor in the size of the first down-converted batch and we return at least that many bytes.
61           java.util.Iterator<ConvertedRecords<?>> it = iterator(0);
62           if (it.hasNext()) {
63               firstConvertedBatch = it.next();
64               sizeInBytes = Math.max(records.sizeInBytes(), firstConvertedBatch.records().sizeInBytes());
65           } else {
66               // If there are messages before down-conversion and no messages after down-conversion,
67               // make sure we are able to send at least an overflow message to the consumer so that it can throw
68               // a RecordTooLargeException. Typically, the consumer would need to increase the fetch size in such cases.
69               // If there are no messages before down-conversion, we return an empty record batch.
70               firstConvertedBatch = null;
71               sizeInBytes = records.batches().iterator().hasNext() ? LazyDownConversionRecordsSend.MIN_OVERFLOW_MESSAGE_LENGTH : 0;
72           }
73       }
74   
75       @Override
76       public int sizeInBytes() {
77           return sizeInBytes;
78       }
79   
80       @Override
81       public LazyDownConversionRecordsSend toSend() {
82           return new LazyDownConversionRecordsSend(this);
83       }
84   
85       public TopicPartition topicPartition() {
86           return topicPartition;
87       }
88   
89       @Override
90       public boolean equals(Object o) {
91           if (o instanceof LazyDownConversionRecords) {
92               LazyDownConversionRecords that = (LazyDownConversionRecords) o;
93               return toMagic == that.toMagic &&
94                       firstOffset == that.firstOffset &&
95                       topicPartition.equals(that.topicPartition) &&
96                       records.equals(that.records);
97           }
98           return false;
99       }
100   
101       @Override
102       public int hashCode() {
103           int result = toMagic;
104           result = 31 * result + Long.hashCode(firstOffset);
105           result = 31 * result + topicPartition.hashCode();
106           result = 31 * result + records.hashCode();
107           return result;
108       }
109   
110       @Override
111       public String toString() {
112           return "LazyDownConversionRecords(size=" + sizeInBytes +
113                   ", underlying=" + records +
114                   ", toMagic=" + toMagic +
115                   ", firstOffset=" + firstOffset +
116                   ")";
117       }

119       public java.util.Iterator<ConvertedRecords<?>> iterator(long maximumReadSize) {
120           // We typically expect only one iterator instance to be created, so null out the first converted batch after
121           // first use to make it available for GC.
122           ConvertedRecords firstBatch = firstConvertedBatch;
123           firstConvertedBatch = null;
124           return new Iterator(records, maximumReadSize, firstBatch);
125       }
126   
127       /**
128        * Implementation for being able to iterate over down-converted records. Goal of this implementation is to keep
129        * it as memory-efficient as possible by not having to maintain all down-converted records in-memory. Maintains
130        * a view into batches of down-converted records.
131        */
132       private class Iterator extends AbstractIterator<ConvertedRecords<?>> {
133           private final AbstractIterator<? extends RecordBatch> batchIterator;
134           private final long maximumReadSize;
135           private ConvertedRecords firstConvertedBatch;
136   
137           /**
138            * @param recordsToDownConvert Records that require down-conversion
139            * @param maximumReadSize Maximum possible size of underlying records that will be down-converted in each call to
140            *                        {@link #makeNext()}. This is a soft limit as {@link #makeNext()} will always convert
141            *                        and return at least one full message batch.
142            */
143           private Iterator(Records recordsToDownConvert, long maximumReadSize, ConvertedRecords<?> firstConvertedBatch) {
144               this.batchIterator = recordsToDownConvert.batchIterator();
145               this.maximumReadSize = maximumReadSize;
146               this.firstConvertedBatch = firstConvertedBatch;
147               // If we already have the first down-converted batch, advance the underlying records iterator to next batch
148               if (firstConvertedBatch != null)
149                   this.batchIterator.next();
150           }
151   
152           /**
153            * Make next set of down-converted records
154            * @return Down-converted records
155            */
156           @Override
157           protected ConvertedRecords makeNext() {
158               // If we have cached the first down-converted batch, return that now
159               if (firstConvertedBatch != null) {
160                   ConvertedRecords convertedBatch = firstConvertedBatch;
161                   firstConvertedBatch = null;
162                   return convertedBatch;
163               }
164   
165               while (batchIterator.hasNext()) {
166                   final List<RecordBatch> batches = new ArrayList<>();
167                   boolean isFirstBatch = true;
168                   long sizeSoFar = 0;
169   
170                   // Figure out batches we should down-convert based on the size constraints
171                   while (batchIterator.hasNext() &&
172                           (isFirstBatch || (batchIterator.peek().sizeInBytes() + sizeSoFar) <= maximumReadSize)) {
173                       RecordBatch currentBatch = batchIterator.next();
174                       batches.add(currentBatch);
175                       sizeSoFar += currentBatch.sizeInBytes();
176                       isFirstBatch = false;
177                   }
178   
179                   ConvertedRecords convertedRecords = RecordsUtil.downConvert(batches, toMagic, firstOffset, time);
180                   // During conversion, it is possible that we drop certain batches because they do not have an equivalent
181                   // representation in the message format we want to convert to. For example, V0 and V1 message formats
182                   // have no notion of transaction markers which were introduced in V2 so they get dropped during conversion.
183                   // We return converted records only when we have at least one valid batch of messages after conversion.
184                   if (convertedRecords.records().sizeInBytes() > 0)
185                       return convertedRecords;
186               }
187               return allDone();
188           }
189       }
190   }
