17   package org.apache.kafka.common.network;
18   
19   /*
20    * Transport layer for underlying communication.
21    * At very basic level it is wrapper around SocketChannel and can be used as substitute for SocketChannel
22    * and other network Channel implementations.
23    * As NetworkClient replaces BlockingChannel and other implementations we will be using KafkaChannel as
24    * a network I/O channel.
25    */
26   
27   import org.apache.kafka.common.errors.AuthenticationException;
28   
29   import java.io.IOException;
30   import java.nio.channels.ScatteringByteChannel;
31   import java.nio.channels.SelectionKey;
32   import java.nio.channels.SocketChannel;
33   import java.security.Principal;
34   
35   public interface TransportLayer extends ScatteringByteChannel, TransferableChannel {
36   
37       /**
38        * Returns true if the channel has handshake and authentication done.
39        */
40       boolean ready();
41   
42       /**
43        * Finishes the process of connecting a socket channel.
44        */
45       boolean finishConnect() throws IOException;
46   
47       /**
48        * disconnect socketChannel
49        */
50       void disconnect();
51   
52       /**
53        * Tells whether or not this channel's network socket is connected.
54        */
55       boolean isConnected();
56   
57       /**
58        * returns underlying socketChannel
59        */
60       SocketChannel socketChannel();
61   
62       /**
63        * Get the underlying selection key
64        */
65       SelectionKey selectionKey();
66   
67       /**
68        * This a no-op for the non-secure PLAINTEXT implementation. For SSL, this performs
69        * SSL handshake. The SSL handshake includes client authentication if configured using
70        * {@link org.apache.kafka.common.config.SslConfigs#SSL_CLIENT_AUTH_CONFIG}.
71        * @throws AuthenticationException if handshake fails due to an {@link javax.net.ssl.SSLException}.
72        * @throws IOException if read or write fails with an I/O error.
73       */
74       void handshake() throws AuthenticationException, IOException;
75   
76       /**
77        * Returns `SSLSession.getPeerPrincipal()` if this is an SslTransportLayer and there is an authenticated peer,
78        * `KafkaPrincipal.ANONYMOUS` is returned otherwise.
79        */
80       Principal peerPrincipal() throws IOException;
81   
82       void addInterestOps(int ops);
83   
84       void removeInterestOps(int ops);
85   
86       boolean isMute();
87   
88       /**
89        * @return true if channel has bytes to be read in any intermediate buffers
90        * which may be processed without reading additional data from the network.
91        */
92       boolean hasBytesBuffered();
93   }
