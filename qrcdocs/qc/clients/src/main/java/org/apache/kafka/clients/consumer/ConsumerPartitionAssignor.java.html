33   /**
34    * This interface is used to define custom partition assignment for use in
35    * {@link org.apache.kafka.clients.consumer.KafkaConsumer}. Members of the consumer group subscribe
36    * to the topics they are interested in and forward their subscriptions to a Kafka broker serving
37    * as the group coordinator. The coordinator selects one member to perform the group assignment and
38    * propagates the subscriptions of all members to it. Then {@link #assign(Cluster, GroupSubscription)} is called
39    * to perform the assignment and the results are forwarded back to each respective members
40    *
41    * In some cases, it is useful to forward additional metadata to the assignor in order to make
42    * assignment decisions. For this, you can override {@link #subscriptionUserData(Set)} and provide custom
43    * userData in the returned Subscription. For example, to have a rack-aware assignor, an implementation
44    * can use this user data to forward the rackId belonging to each member.
45    */
46   public interface ConsumerPartitionAssignor {
47   
48       /**
49        * Return serialized data that will be included in the {@link Subscription} sent to the leader
50        * and can be leveraged in {@link #assign(Cluster, GroupSubscription)} ((e.g. local host/rack information)
51        *
52        * @param topics Topics subscribed to through {@link org.apache.kafka.clients.consumer.KafkaConsumer#subscribe(java.util.Collection)}
53        *               and variants
54        * @return nullable subscription user data
55        */
56       default ByteBuffer subscriptionUserData(Set<String> topics) {
57           return null;
58       }

62        * @param metadata Current topic/broker metadata known by consumer
63        * @param groupSubscription Subscriptions from all members including metadata provided through {@link #subscriptionUserData(Set)}
64        * @return A map from the members to their respective assignments. This should have one entry

72        * @param metadata Additional metadata on the consumer (optional)
73        */
74       default void onAssignment(Assignment assignment, ConsumerGroupMetadata metadata) {
75       }

220       /**
221        * The rebalance protocol defines partition assignment and revocation semantics. The purpose is to establish a
222        * consistent set of rules that all consumers in a group follow in order to transfer ownership of a partition.
223        * {@link ConsumerPartitionAssignor} implementors can claim supporting one or more rebalance protocols via the
224        * {@link ConsumerPartitionAssignor#supportedProtocols()}, and it is their responsibility to respect the rules
225        * of those protocols in their {@link ConsumerPartitionAssignor#assign(Cluster, GroupSubscription)} implementations.
226        * Failures to follow the rules of the supported protocols would lead to runtime error or undefined behavior.
227        *
228        * The {@link RebalanceProtocol#EAGER} rebalance protocol requires a consumer to always revoke all its owned
229        * partitions before participating in a rebalance event. It therefore allows a complete reshuffling of the assignment.
230        *
231        * {@link RebalanceProtocol#COOPERATIVE} rebalance protocol allows a consumer to retain its currently owned
232        * partitions before participating in a rebalance event. The assignor should not reassign any owned partitions
233        * immediately, but instead may indicate consumers the need for partition revocation so that the revoked
234        * partitions can be reassigned to other consumers in the next rebalance event. This is designed for sticky assignment
235        * logic which attempts to minimize partition reassignment with cooperative adjustments.
236        */
237       enum RebalanceProtocol {
238           EAGER((byte) 0), COOPERATIVE((byte) 1);
239   
240           private final byte id;
241   
242           RebalanceProtocol(byte id) {
243               this.id = id;
244           }
245   
246           public byte id() {
247               return id;
248           }
249   
250           public static RebalanceProtocol forId(byte id) {
251               switch (id) {
252                   case 0:
253                       return EAGER;
254                   case 1:
255                       return COOPERATIVE;
256                   default:
257                       throw new IllegalArgumentException("Unknown rebalance protocol id: " + id);
258               }
259           }
260       }

262       /**
263        * Get a list of configured instances of {@link org.apache.kafka.clients.consumer.ConsumerPartitionAssignor}
264        * based on the class names/types specified by {@link org.apache.kafka.clients.consumer.ConsumerConfig#PARTITION_ASSIGNMENT_STRATEGY_CONFIG}
265        */
266       static List<ConsumerPartitionAssignor> getAssignorInstances(List<String> assignorClasses, Map<String, Object> configs) {
267           List<ConsumerPartitionAssignor> assignors = new ArrayList<>();
268           // a map to store assignor name -> assignor class name
269           Map<String, String> assignorNameMap = new HashMap<>();
270   
271           if (assignorClasses == null)
272               return assignors;
273   
274           for (Object klass : assignorClasses) {
275               // first try to get the class if passed in as a string
276               if (klass instanceof String) {
277                   try {
278                       klass = Class.forName((String) klass, true, Utils.getContextOrKafkaClassLoader());
279                   } catch (ClassNotFoundException classNotFound) {
280                       throw new KafkaException(klass + " ClassNotFoundException exception occurred", classNotFound);
281                   }
282               }
283   
284               if (klass instanceof Class<?>) {
285                   Object assignor = Utils.newInstance((Class<?>) klass);
286                   if (assignor instanceof Configurable)
287                       ((Configurable) assignor).configure(configs);
288   
289                   if (assignor instanceof ConsumerPartitionAssignor) {
290                       String assignorName = ((ConsumerPartitionAssignor) assignor).name();
291                       if (assignorNameMap.containsKey(assignorName)) {
292                           throw new KafkaException("The assignor name: '" + assignorName + "' is used in more than one assignor: " +
293                               assignorNameMap.get(assignorName) + ", " + assignor.getClass().getName());
294                       }
295                       assignorNameMap.put(assignorName, assignor.getClass().getName());
296                       assignors.add((ConsumerPartitionAssignor) assignor);
297                   } else {
298                       throw new KafkaException(klass + " is not an instance of " + ConsumerPartitionAssignor.class.getName());
299                   }
300               } else {
301                   throw new KafkaException("List contains element of type " + klass.getClass().getName() + ", expected String or Class");
302               }
303           }
304           return assignors;
305       }
