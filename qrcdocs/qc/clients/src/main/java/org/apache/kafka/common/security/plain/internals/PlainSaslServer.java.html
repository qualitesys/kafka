18   
19   import java.nio.charset.StandardCharsets;
20   import java.util.ArrayList;
21   import java.util.Arrays;
22   import java.util.List;
23   import java.util.Map;
24   
25   import javax.security.auth.callback.Callback;
26   import javax.security.auth.callback.CallbackHandler;
27   import javax.security.auth.callback.NameCallback;
28   import javax.security.sasl.Sasl;
29   import javax.security.sasl.SaslException;
30   import javax.security.sasl.SaslServer;
31   import javax.security.sasl.SaslServerFactory;
32   
33   import org.apache.kafka.common.errors.SaslAuthenticationException;
34   import org.apache.kafka.common.security.plain.PlainAuthenticateCallback;
35   
36   /**
37    * Simple SaslServer implementation for SASL/PLAIN. In order to make this implementation
38    * fully pluggable, authentication of username/password is fully contained within the
39    * server implementation.
40    * <p>
41    * Valid users with passwords are specified in the Jaas configuration file. Each user
42    * is specified with user_<username> as key and <password> as value. This is consistent
43    * with Zookeeper Digest-MD5 implementation.
44    * <p>
45    * To avoid storing clear passwords on disk or to integrate with external authentication
46    * servers in production systems, this module can be replaced with a different implementation.
47    *
48    */
49   public class PlainSaslServer implements SaslServer {
50   
51       public static final String PLAIN_MECHANISM = "PLAIN";
52   
53       private final CallbackHandler callbackHandler;
54       private boolean complete;
55       private String authorizationId;
56   
57       public PlainSaslServer(CallbackHandler callbackHandler) {
58           this.callbackHandler = callbackHandler;
59       }
60   
61       /**
62        * @throws SaslAuthenticationException if username/password combination is invalid or if the requested
63        *         authorization id is not the same as username.
64        * <p>
65        * <b>Note:</b> This method may throw {@link SaslAuthenticationException} to provide custom error messages
66        * to clients. But care should be taken to avoid including any information in the exception message that
67        * should not be leaked to unauthenticated clients. It may be safer to throw {@link SaslException} in
68        * some cases so that a standard error message is returned to clients.
69        * </p>
70        */
71       @Override
72       public byte[] evaluateResponse(byte[] responseBytes) throws SaslAuthenticationException {
73           /*
74            * Message format (from https://tools.ietf.org/html/rfc4616):
75            *
76            * message   = [authzid] UTF8NUL authcid UTF8NUL passwd
77            * authcid   = 1*SAFE ; MUST accept up to 255 octets
78            * authzid   = 1*SAFE ; MUST accept up to 255 octets
79            * passwd    = 1*SAFE ; MUST accept up to 255 octets
80            * UTF8NUL   = %x00 ; UTF-8 encoded NUL character
81            *
82            * SAFE      = UTF1 / UTF2 / UTF3 / UTF4
83            *                ;; any UTF-8 encoded Unicode character except NUL
84            */
85   
86           String response = new String(responseBytes, StandardCharsets.UTF_8);
87           List<String> tokens = extractTokens(response);
88           String authorizationIdFromClient = tokens.get(0);
89           String username = tokens.get(1);
90           String password = tokens.get(2);
91   
92           if (username.isEmpty()) {
93               throw new SaslAuthenticationException("Authentication failed: username not specified");
94           }
95           if (password.isEmpty()) {
96               throw new SaslAuthenticationException("Authentication failed: password not specified");
97           }
98   
99           NameCallback nameCallback = new NameCallback("username", username);
100           PlainAuthenticateCallback authenticateCallback = new PlainAuthenticateCallback(password.toCharArray());
101           try {
102               callbackHandler.handle(new Callback[]{nameCallback, authenticateCallback});
103           } catch (Throwable e) {
104               throw new SaslAuthenticationException("Authentication failed: credentials for user could not be verified", e);
105           }
106           if (!authenticateCallback.authenticated())
107               throw new SaslAuthenticationException("Authentication failed: Invalid username or password");
108           if (!authorizationIdFromClient.isEmpty() && !authorizationIdFromClient.equals(username))
109               throw new SaslAuthenticationException("Authentication failed: Client requested an authorization id that is different from username");
110   
111           this.authorizationId = username;
112   
113           complete = true;
114           return new byte[0];
115       }
116   
117       private List<String> extractTokens(String string) {
118           List<String> tokens = new ArrayList<>();
119           int startIndex = 0;
120           for (int i = 0; i < 4; ++i) {
121               int endIndex = string.indexOf("\u0000", startIndex);
122               if (endIndex == -1) {
123                   tokens.add(string.substring(startIndex));
124                   break;
125               }
126               tokens.add(string.substring(startIndex, endIndex));
127               startIndex = endIndex + 1;
128           }
129   
130           if (tokens.size() != 3)
131               throw new SaslAuthenticationException("Invalid SASL/PLAIN response: expected 3 tokens, got " +
132                   tokens.size());
133   
134           return tokens;
135       }
136   
137       @Override
138       public String getAuthorizationID() {
139           if (!complete)
140               throw new IllegalStateException("Authentication exchange has not completed");
141           return authorizationId;
142       }
143   
144       @Override
145       public String getMechanismName() {
146           return PLAIN_MECHANISM;
147       }
148   
149       @Override
150       public Object getNegotiatedProperty(String propName) {
151           if (!complete)
152               throw new IllegalStateException("Authentication exchange has not completed");
153           return null;
154       }
155   
156       @Override
157       public boolean isComplete() {
158           return complete;
159       }
160   
161       @Override
162       public byte[] unwrap(byte[] incoming, int offset, int len) {
163           if (!complete)
164               throw new IllegalStateException("Authentication exchange has not completed");
165           return Arrays.copyOfRange(incoming, offset, offset + len);
166       }
167   
168       @Override
169       public byte[] wrap(byte[] outgoing, int offset, int len) {
170           if (!complete)
171               throw new IllegalStateException("Authentication exchange has not completed");
172           return Arrays.copyOfRange(outgoing, offset, offset + len);
173       }
174   
175       @Override
176       public void dispose() {
177       }
178   
179       public static class PlainSaslServerFactory implements SaslServerFactory {
180   
181           @Override
182           public SaslServer createSaslServer(String mechanism, String protocol, String serverName, Map<String, ?> props, CallbackHandler cbh)
183               throws SaslException {
184   
185               if (!PLAIN_MECHANISM.equals(mechanism))
186                   throw new SaslException(String.format("Mechanism \'%s\' is not supported. Only PLAIN is supported.", mechanism));
187   
188               return new PlainSaslServer(cbh);
189           }
190   
191           @Override
192           public String[] getMechanismNames(Map<String, ?> props) {
193               if (props == null) return new String[]{PLAIN_MECHANISM};
194               String noPlainText = (String) props.get(Sasl.POLICY_NOPLAINTEXT);
195               if ("true".equals(noPlainText))
196                   return new String[]{};
197               else
198                   return new String[]{PLAIN_MECHANISM};
199           }
200       }
201   }
