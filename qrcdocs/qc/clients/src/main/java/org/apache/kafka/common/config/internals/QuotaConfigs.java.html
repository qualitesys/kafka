18   package org.apache.kafka.common.config.internals;
19   
20   import org.apache.kafka.common.config.ConfigDef;
21   import org.apache.kafka.common.security.scram.internals.ScramMechanism;
22   
23   import java.util.Arrays;
24   import java.util.HashSet;
25   import java.util.Set;
26   
27   /**
28    * Define the dynamic quota configs. Note that these are not normal configurations that exist in properties files. They
29    * only exist dynamically in the controller (or ZK, depending on which mode the cluster is running).
30    */
31   public class QuotaConfigs {
32       public static final String PRODUCER_BYTE_RATE_OVERRIDE_CONFIG = "producer_byte_rate";
33       public static final String CONSUMER_BYTE_RATE_OVERRIDE_CONFIG = "consumer_byte_rate";
34       public static final String REQUEST_PERCENTAGE_OVERRIDE_CONFIG = "request_percentage";
35       public static final String CONTROLLER_MUTATION_RATE_OVERRIDE_CONFIG = "controller_mutation_rate";
36       public static final String IP_CONNECTION_RATE_OVERRIDE_CONFIG = "connection_creation_rate";
37   
38       public static final String PRODUCER_BYTE_RATE_DOC = "A rate representing the upper bound (bytes/sec) for producer traffic.";
39       public static final String CONSUMER_BYTE_RATE_DOC = "A rate representing the upper bound (bytes/sec) for consumer traffic.";
40       public static final String REQUEST_PERCENTAGE_DOC = "A percentage representing the upper bound of time spent for processing requests.";
41       public static final String CONTROLLER_MUTATION_RATE_DOC = "The rate at which mutations are accepted for the create " +
42           "topics request, the create partitions request and the delete topics request. The rate is accumulated by " +
43           "the number of partitions created or deleted.";
44       public static final String IP_CONNECTION_RATE_DOC = "An int representing the upper bound of connections accepted " +
45           "for the specified IP.";
46   
47       public static final int IP_CONNECTION_RATE_DEFAULT = Integer.MAX_VALUE;
48   
49       private static Set<String> userClientConfigNames = new HashSet<>(Arrays.asList(
50           PRODUCER_BYTE_RATE_OVERRIDE_CONFIG, CONSUMER_BYTE_RATE_OVERRIDE_CONFIG,
51           REQUEST_PERCENTAGE_OVERRIDE_CONFIG, CONTROLLER_MUTATION_RATE_OVERRIDE_CONFIG
52       ));
53   
54       private static void buildUserClientQuotaConfigDef(ConfigDef configDef) {
55           configDef.define(PRODUCER_BYTE_RATE_OVERRIDE_CONFIG, ConfigDef.Type.LONG, Long.MAX_VALUE,
56               ConfigDef.Importance.MEDIUM, PRODUCER_BYTE_RATE_DOC);
57   
58           configDef.define(CONSUMER_BYTE_RATE_OVERRIDE_CONFIG, ConfigDef.Type.LONG, Long.MAX_VALUE,
59               ConfigDef.Importance.MEDIUM, CONSUMER_BYTE_RATE_DOC);
60   
61           configDef.define(REQUEST_PERCENTAGE_OVERRIDE_CONFIG, ConfigDef.Type.DOUBLE,
62               Integer.valueOf(Integer.MAX_VALUE).doubleValue(),
63               ConfigDef.Importance.MEDIUM, REQUEST_PERCENTAGE_DOC);
64   
65           configDef.define(CONTROLLER_MUTATION_RATE_OVERRIDE_CONFIG, ConfigDef.Type.DOUBLE,
66               Integer.valueOf(Integer.MAX_VALUE).doubleValue(),
67               ConfigDef.Importance.MEDIUM, CONTROLLER_MUTATION_RATE_DOC);
68       }
69   
70       public static boolean isClientOrUserConfig(String name) {
71           return userClientConfigNames.contains(name);
72       }
73   
74       public static ConfigDef userConfigs() {
75           ConfigDef configDef = new ConfigDef();
76           ScramMechanism.mechanismNames().forEach(mechanismName -> {
77               configDef.define(mechanismName, ConfigDef.Type.STRING, null, ConfigDef.Importance.MEDIUM,
78                   "User credentials for SCRAM mechanism " + mechanismName);
79           });
80           buildUserClientQuotaConfigDef(configDef);
81           return configDef;
82       }
83   
84       public static ConfigDef clientConfigs() {
85           ConfigDef configDef = new ConfigDef();
86           buildUserClientQuotaConfigDef(configDef);
87           return configDef;
88       }
89   
90       public static ConfigDef ipConfigs() {
91           ConfigDef configDef = new ConfigDef();
92           configDef.define(IP_CONNECTION_RATE_OVERRIDE_CONFIG, ConfigDef.Type.INT, Integer.MAX_VALUE,
93               ConfigDef.Range.atLeast(0), ConfigDef.Importance.MEDIUM, IP_CONNECTION_RATE_DOC);
94           return configDef;
95       }
96   }
