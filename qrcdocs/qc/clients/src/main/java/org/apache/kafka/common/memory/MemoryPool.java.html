17   package org.apache.kafka.common.memory;
18   
19   import java.nio.ByteBuffer;
20   
21   
22   /**
23    * A common memory pool interface for non-blocking pools.
24    * Every buffer returned from {@link #tryAllocate(int)} must always be {@link #release(ByteBuffer) released}.
25    */
26   public interface MemoryPool {
27       MemoryPool NONE = new MemoryPool() {
28           @Override
29           public ByteBuffer tryAllocate(int sizeBytes) {
30               return ByteBuffer.allocate(sizeBytes);
31           }
32   
33           @Override
34           public void release(ByteBuffer previouslyAllocated) {
35               //nop
36           }
37   
38           @Override
39           public long size() {
40               return Long.MAX_VALUE;
41           }
42   
43           @Override
44           public long availableMemory() {
45               return Long.MAX_VALUE;
46           }
47   
48           @Override
49           public boolean isOutOfMemory() {
50               return false;
51           }
52   
53           @Override
54           public String toString() {
55               return "NONE";
56           }
57       };
58   
59       /**
60        * Tries to acquire a ByteBuffer of the specified size
61        * @param sizeBytes size required
62        * @return a ByteBuffer (which later needs to be release()ed), or null if no memory available.
63        *         the buffer will be of the exact size requested, even if backed by a larger chunk of memory
64        */
65       ByteBuffer tryAllocate(int sizeBytes);
66   
67       /**
68        * Returns a previously allocated buffer to the pool.
69        * @param previouslyAllocated a buffer previously returned from tryAllocate()
70        */
71       void release(ByteBuffer previouslyAllocated);
72   
73       /**
74        * Returns the total size of this pool
75        * @return total size, in bytes
76        */
77       long size();
78   
79       /**
80        * Returns the amount of memory available for allocation by this pool.
81        * NOTE: result may be negative (pools may over allocate to avoid starvation issues)
82        * @return bytes available
83        */
84       long availableMemory();
85   
86       /**
87        * Returns true if the pool cannot currently allocate any more buffers
88        * - meaning total outstanding buffers meets or exceeds pool size and
89        * some would need to be released before further allocations are possible.
90        *
91        * This is equivalent to availableMemory() <= 0
92        * @return true if out of memory
93        */
94       boolean isOutOfMemory();
95   }
