17   package org.apache.kafka.common.metrics.stats;
18   
19   import java.util.concurrent.TimeUnit;
20   import org.apache.kafka.common.metrics.MeasurableStat;
21   import org.apache.kafka.common.metrics.MetricConfig;
22   import org.apache.kafka.common.metrics.Quota;
23   
24   import static org.apache.kafka.common.metrics.internals.MetricsUtils.convert;
25   
26   /**
27    * The {@link TokenBucket} is a {@link MeasurableStat} implementing a token bucket algorithm
28    * that is usable within a {@link org.apache.kafka.common.metrics.Sensor}.
29    *
30    * The {@link Quota#bound()} defined the refill rate of the bucket while the maximum burst or
31    * the maximum number of credits of the bucket is defined by
32    * {@link MetricConfig#samples() * MetricConfig#timeWindowMs() * Quota#bound()}.
33    *
34    * The quota is considered as exhausted when the amount of remaining credits in the bucket
35    * is below zero. The enforcement is done by the {@link org.apache.kafka.common.metrics.Sensor}.
36    *
37    * Token Bucket vs Rate based Quota:
38    * The current sampled rate based quota does not cope well with bursty workloads. The issue is
39    * that a unique and large sample can hold the average above the quota until it is discarded.
40    * Practically, when this happens, one must wait until the sample is expired to bring the rate
41    * below the quota even though less time would be theoretically required. As an example, let's
42    * imagine that we have:
43    * - Quota (Q)   = 5
44    * - Samples (S) = 100
45    * - Window (W)  = 1s
46    * A burst of 560 brings the average rate (R) to 5.6 (560 / 100). The expected throttle time is
47    * computed as follow: ((R - Q / Q * S * W)) = ((5.6 - 5) / 5 * 100 * 1) = 12 secs. In practice,
48    * the average rate won't go below the quota before the burst is dropped from the samples so one
49    * must wait 100s (S * W).
50    *
51    * The token bucket relies on continuously updated amount of credits. Therefore, it does not
52    * suffers from the above issue. The same example would work as follow:
53    * - Quota (Q) = 5
54    * - Burst (B) = 5 * 1 * 100 = 500 (Q * S * W)
55    * A burst of 560 brings the amount of credits to -60. One must wait 12s (-(-60)/5) to refill the
56    * bucket to zero.
57    */
58   public class TokenBucket implements MeasurableStat {
59       private final TimeUnit unit;
60       private double tokens;
61       private long lastUpdateMs;
62   
63       public TokenBucket() {
64           this(TimeUnit.SECONDS);
65       }
66   
67       public TokenBucket(TimeUnit unit) {
68           this.unit = unit;
69           this.tokens = 0;
70           this.lastUpdateMs = 0;
71       }
72   
73       @Override
74       public double measure(final MetricConfig config, final long timeMs) {
75           if (config.quota() == null)
76               return Long.MAX_VALUE;
77           final double quota = config.quota().bound();
78           final double burst = burst(config);
79           refill(quota, burst, timeMs);
80           return this.tokens;
81       }
82   
83       @Override
84       public void record(final MetricConfig config, final double value, final long timeMs) {
85           if (config.quota() == null)
86               return;
87           final double quota = config.quota().bound();
88           final double burst = burst(config);
89           refill(quota, burst, timeMs);
90           this.tokens = Math.min(burst, this.tokens - value);
91       }
92   
93       private void refill(final double quota, final double burst, final long timeMs) {
94           this.tokens = Math.min(burst, this.tokens + quota * convert(timeMs - lastUpdateMs, unit));
95           this.lastUpdateMs = timeMs;
96       }
97   
98       private double burst(final MetricConfig config) {
99           return config.samples() * convert(config.timeWindowMs(), unit) * config.quota().bound();
100       }
101   
102       @Override
103       public String toString() {
104           return "TokenBucket(" +
105               "unit=" + unit +
106               ", tokens=" + tokens +
107               ", lastUpdateMs=" + lastUpdateMs +
108               ')';
109       }
110   }
