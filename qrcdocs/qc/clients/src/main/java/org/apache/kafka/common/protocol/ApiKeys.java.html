172       private static boolean shouldRetainsBufferReference(Schema[] requestSchemas) {
173           boolean requestRetainsBufferReference = false;
174           for (Schema requestVersionSchema : requestSchemas) {
175               if (retainsBufferReference(requestVersionSchema)) {
176                   requestRetainsBufferReference = true;
177                   break;
178               }
179           }
180           return requestRetainsBufferReference;
181       }

203       public List<Short> allVersions() {
204           List<Short> versions = new ArrayList<>(latestVersion() - oldestVersion() + 1);
205           for (short version = oldestVersion(); version <= latestVersion(); version++) {

252       private static boolean retainsBufferReference(Schema schema) {
253           final AtomicBoolean hasBuffer = new AtomicBoolean(false);
254           Schema.Visitor detector = new Schema.Visitor() {
255               @Override
256               public void visit(Type field) {
257                   if (field == BYTES || field == NULLABLE_BYTES || field == RECORDS ||
258                       field == COMPACT_BYTES || field == COMPACT_NULLABLE_BYTES)
259                       hasBuffer.set(true);
260               }
261           };
262           schema.walk(detector);
263           return hasBuffer.get();
264       }

278       private static EnumSet<ApiKeys> filterApisForListener(ApiMessageType.ListenerType listener) {
279           List<ApiKeys> controllerApis = Arrays.stream(ApiKeys.values())
280               .filter(apiKey -> apiKey.messageType.listeners().contains(listener))
281               .collect(Collectors.toList());
282           return EnumSet.copyOf(controllerApis);
283       }
