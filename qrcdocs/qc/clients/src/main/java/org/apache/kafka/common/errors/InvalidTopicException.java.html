17   package org.apache.kafka.common.errors;
18   
19   import java.util.HashSet;
20   import java.util.Set;
21   
22   
23   /**
24    * The client has attempted to perform an operation on an invalid topic.
25    * For example the topic name is too long, contains invalid characters etc.
26    * This exception is not retriable because the operation won't suddenly become valid.
27    *
28    * @see UnknownTopicOrPartitionException
29    */
30   public class InvalidTopicException extends ApiException {
31       private static final long serialVersionUID = 1L;
32   
33       private final Set<String> invalidTopics;
34   
35       public InvalidTopicException() {
36           super();
37           invalidTopics = new HashSet<>();
38       }
39   
40       public InvalidTopicException(String message, Throwable cause) {
41           super(message, cause);
42           invalidTopics = new HashSet<>();
43       }
44   
45       public InvalidTopicException(String message) {
46           super(message);
47           invalidTopics = new HashSet<>();
48       }
49   
50       public InvalidTopicException(Throwable cause) {
51           super(cause);
52           invalidTopics = new HashSet<>();
53       }
54   
55       public InvalidTopicException(Set<String> invalidTopics) {
56           super("Invalid topics: " + invalidTopics);
57           this.invalidTopics = invalidTopics;
58       }
59   
60       public InvalidTopicException(String message, Set<String> invalidTopics) {
61           super(message);
62           this.invalidTopics = invalidTopics;
63       }
64   
65       public Set<String> invalidTopics() {
66           return invalidTopics;
67       }
68   }
