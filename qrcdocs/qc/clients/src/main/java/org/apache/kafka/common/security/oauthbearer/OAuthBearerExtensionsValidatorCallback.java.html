17   package org.apache.kafka.common.security.oauthbearer;
18   
19   import org.apache.kafka.common.security.auth.SaslExtensions;
20   
21   import javax.security.auth.callback.Callback;
22   import java.util.Collections;
23   import java.util.HashMap;
24   import java.util.Map;
25   import java.util.Objects;
26   
27   import static org.apache.kafka.common.utils.CollectionUtils.subtractMap;
28   
29   /**
30    * A {@code Callback} for use by the {@code SaslServer} implementation when it
31    * needs to validate the SASL extensions for the OAUTHBEARER mechanism
32    * Callback handlers should use the {@link #valid(String)}
33    * method to communicate valid extensions back to the SASL server.
34    * Callback handlers should use the
35    * {@link #error(String, String)} method to communicate validation errors back to
36    * the SASL Server.
37    * As per RFC-7628 (https://tools.ietf.org/html/rfc7628#section-3.1), unknown extensions must be ignored by the server.
38    * The callback handler implementation should simply ignore unknown extensions,
39    * not calling {@link #error(String, String)} nor {@link #valid(String)}.
40    * Callback handlers should communicate other problems by raising an {@code IOException}.
41    * <p>
42    * The OAuth bearer token is provided in the callback for better context in extension validation.
43    * It is very important that token validation is done in its own {@link OAuthBearerValidatorCallback}
44    * irregardless of provided extensions, as they are inherently insecure.
45    */
46   public class OAuthBearerExtensionsValidatorCallback implements Callback {
47       private final OAuthBearerToken token;
48       private final SaslExtensions inputExtensions;
49       private final Map<String, String> validatedExtensions = new HashMap<>();
50       private final Map<String, String> invalidExtensions = new HashMap<>();
51   
52       public OAuthBearerExtensionsValidatorCallback(OAuthBearerToken token, SaslExtensions extensions) {
53           this.token = Objects.requireNonNull(token);
54           this.inputExtensions = Objects.requireNonNull(extensions);
55       }
56   
57       /**
58        * @return {@link OAuthBearerToken} the OAuth bearer token of the client
59        */
60       public OAuthBearerToken token() {
61           return token;
62       }
63   
64       /**
65        * @return {@link SaslExtensions} consisting of the unvalidated extension names and values that were sent by the client
66        */
67       public SaslExtensions inputExtensions() {
68           return inputExtensions;
69       }
70   
71       /**
72        * @return an unmodifiable {@link Map} consisting of the validated and recognized by the server extension names and values
73        */
74       public Map<String, String> validatedExtensions() {
75           return Collections.unmodifiableMap(validatedExtensions);
76       }
77   
78       /**
79        * @return An immutable {@link Map} consisting of the name-&gt;error messages of extensions which failed validation
80        */
81       public Map<String, String> invalidExtensions() {
82           return Collections.unmodifiableMap(invalidExtensions);
83       }
84   
85       /**
86        * @return An immutable {@link Map} consisting of the extensions that have neither been validated nor invalidated
87        */
88       public Map<String, String> ignoredExtensions() {
89           return Collections.unmodifiableMap(subtractMap(subtractMap(inputExtensions.map(), invalidExtensions), validatedExtensions));
90       }
91   
92       /**
93        * Validates a specific extension in the original {@code inputExtensions} map
94        * @param extensionName - the name of the extension which was validated
95        */
96       public void valid(String extensionName) {
97           if (!inputExtensions.map().containsKey(extensionName))
98               throw new IllegalArgumentException(String.format("Extension %s was not found in the original extensions", extensionName));
99           validatedExtensions.put(extensionName, inputExtensions.map().get(extensionName));
100       }
101       /**
102        * Set the error value for a specific extension key-value pair if validation has failed
103        *
104        * @param invalidExtensionName
105        *            the mandatory extension name which caused the validation failure
106        * @param errorMessage
107        *            error message describing why the validation failed
108        */
109       public void error(String invalidExtensionName, String errorMessage) {
110           if (Objects.requireNonNull(invalidExtensionName).isEmpty())
111               throw new IllegalArgumentException("extension name must not be empty");
112           this.invalidExtensions.put(invalidExtensionName, errorMessage);
113       }
114   }
