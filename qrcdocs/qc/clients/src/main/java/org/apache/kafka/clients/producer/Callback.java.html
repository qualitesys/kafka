19   /**
20    * A callback interface that the user can implement to allow code to execute when the request is complete. This callback
21    * will generally execute in the background I/O thread so it should be fast.
22    */
23   public interface Callback {
24   
25       /**
26        * A callback method the user can implement to provide asynchronous handling of request completion. This method will
27        * be called when the record sent to the server has been acknowledged. When exception is not null in the callback,
28        * metadata will contain the special -1 value for all fields. If topicPartition cannot be
29        * choosen, a -1 value will be assigned.
30        *
31        * @param metadata The metadata for the record that was sent (i.e. the partition and offset). An empty metadata
32        *                 with -1 value for all fields will be returned if an error occurred.
33        * @param exception The exception thrown during processing of this record. Null if no error occurred.
34        *                  Possible thrown exceptions include:
35        *
36        *                  Non-Retriable exceptions (fatal, the message will never be sent):
37        *
38        *                  InvalidTopicException
39        *                  OffsetMetadataTooLargeException
40        *                  RecordBatchTooLargeException
41        *                  RecordTooLargeException
42        *                  UnknownServerException
43        *                  UnknownProducerIdException
44        *                  InvalidProducerEpochException
45        *
46        *                  Retriable exceptions (transient, may be covered by increasing #.retries):
47        *
48        *                  CorruptRecordException
49        *                  InvalidMetadataException
50        *                  NotEnoughReplicasAfterAppendException
51        *                  NotEnoughReplicasException
52        *                  OffsetOutOfRangeException
53        *                  TimeoutException
54        *                  UnknownTopicOrPartitionException
55        */
56       void onCompletion(RecordMetadata metadata, Exception exception);
57   }
