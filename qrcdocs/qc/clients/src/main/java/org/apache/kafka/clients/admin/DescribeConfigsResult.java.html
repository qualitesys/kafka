18   package org.apache.kafka.clients.admin;
19   
20   import org.apache.kafka.common.KafkaFuture;
21   import org.apache.kafka.common.annotation.InterfaceStability;
22   import org.apache.kafka.common.config.ConfigResource;
23   
24   import java.util.Collection;
25   import java.util.HashMap;
26   import java.util.Map;
27   import java.util.concurrent.ExecutionException;
28   
29   /**
30    * The result of the {@link KafkaAdminClient#describeConfigs(Collection)} call.
31    *
32    * The API of this class is evolving, see {@link Admin} for details.
33    */
34   @InterfaceStability.Evolving
35   public class DescribeConfigsResult {
36   
37       private final Map<ConfigResource, KafkaFuture<Config>> futures;
38   
39       protected DescribeConfigsResult(Map<ConfigResource, KafkaFuture<Config>> futures) {
40           this.futures = futures;
41       }
42   
43       /**
44        * Return a map from resources to futures which can be used to check the status of the configuration for each
45        * resource.
46        */
47       public Map<ConfigResource, KafkaFuture<Config>> values() {
48           return futures;
49       }
50   
51       /**
52        * Return a future which succeeds only if all the config descriptions succeed.
53        */
54       public KafkaFuture<Map<ConfigResource, Config>> all() {
55           return KafkaFuture.allOf(futures.values().toArray(new KafkaFuture[0])).
56                   thenApply(new KafkaFuture.BaseFunction<Void, Map<ConfigResource, Config>>() {
57                       @Override
58                       public Map<ConfigResource, Config> apply(Void v) {
59                           Map<ConfigResource, Config> configs = new HashMap<>(futures.size());
60                           for (Map.Entry<ConfigResource, KafkaFuture<Config>> entry : futures.entrySet()) {
61                               try {
62                                   configs.put(entry.getKey(), entry.getValue().get());
63                               } catch (InterruptedException | ExecutionException e) {
64                                   // This should be unreachable, because allOf ensured that all the futures
65                                   // completed successfully.
66                                   throw new RuntimeException(e);
67                               }
68                           }
69                           return configs;
70                       }
71                   });
72       }
73   }
