49   /**
50    * Higher level consumer access to the network layer with basic support for request futures. This class
51    * is thread-safe, but provides no synchronization for response callbacks. This guarantees that no locks
52    * are held when they are invoked.
53    */
54   public class ConsumerNetworkClient implements Closeable {
55       private static final int MAX_POLL_TIMEOUT_MS = 5000;
56

58       // flag and the request completion queue below).
59       private final Logger log;
60       private final KafkaClient client;
61       private final UnsentRequests unsent = new UnsentRequests();
62       private final Metadata metadata;
63       private final Time time;
64       private final long retryBackoffMs;
65       private final int maxPollTimeoutMs;
66       private final int requestTimeoutMs;
67       private final AtomicBoolean wakeupDisabled = new AtomicBoolean();
68

109       /**
110        * Send a new request. Note that the request is not actually transmitted on the
111        * network until one of the {@link #poll(Timer)} variants is invoked. At this
112        * point the request will either be transmitted successfully or will fail.
113        * Use the returned future to obtain the result of the send. Note that there is no
114        * need to check for disconnects explicitly on the {@link ClientResponse} object;
115        * instead, the future will be failed with a {@link DisconnectException}.
116        *
117        * @param node The destination of the request
118        * @param requestBuilder A builder for the request payload
119        * @param requestTimeoutMs Maximum time in milliseconds to await a response before disconnecting the socket and
120        *                         cancelling the request. The request may be cancelled sooner if the socket disconnects
121        *                         for any reason.
122        * @return A future which indicates the result of the send.
123        */
124       public RequestFuture<ClientResponse> send(Node node,
125                                                 AbstractRequest.Builder<?> requestBuilder,
126                                                 int requestTimeoutMs) {
127           long now = time.milliseconds();
128           RequestFutureCompletionHandler completionHandler = new RequestFutureCompletionHandler();
129           ClientRequest clientRequest = client.newClientRequest(node.idString(), requestBuilder, now, true,
130               requestTimeoutMs, completionHandler);
131           unsent.put(node, clientRequest);
132   
133           // wakeup the client in case it is blocking in poll so that we can send the queued request
134           client.wakeup();
135           return completionHandler.future;
136       }

195        * Block indefinitely until the given request future has finished.
196        * @param future The request future to await.

239       /**
240        * Poll for any network IO.
241        * @param timer Timer bounding how long this method can block
242        * @param pollCondition Nullable blocking condition
243        * @param disableWakeup If TRUE disable triggering wake-ups
244        */
245       public void poll(Timer timer, PollCondition pollCondition, boolean disableWakeup) {
246           // there may be handlers which need to be invoked if we woke up the previous call to poll
247           firePendingCompletedRequests();
248   
249           lock.lock();
250           try {
251               // Handle async disconnects prior to attempting any sends
252               handlePendingDisconnects();
253   
254               // send all the requests we can send now
255               long pollDelayMs = trySend(timer.currentTimeMs());
256   
257               // check whether the poll is still needed by the caller. Note that if the expected completion
258               // condition becomes satisfied after the call to shouldBlock() (because of a fired completion
259               // handler), the client will be woken up.
260               if (pendingCompletion.isEmpty() && (pollCondition == null || pollCondition.shouldBlock())) {
261                   // if there are no requests in flight, do not block longer than the retry backoff
262                   long pollTimeout = Math.min(timer.remainingMs(), pollDelayMs);
263                   if (client.inFlightRequestCount() == 0)
264                       pollTimeout = Math.min(pollTimeout, retryBackoffMs);
265                   client.poll(pollTimeout, timer.currentTimeMs());
266               } else {
267                   client.poll(0, timer.currentTimeMs());
268               }
269               timer.update();
270   
271               // handle any disconnects by failing the active requests. note that disconnects must
272               // be checked immediately following poll since any subsequent call to client.ready()
273               // will reset the disconnect status
274               checkDisconnects(timer.currentTimeMs());
275               if (!disableWakeup) {
276                   // trigger wakeups after checking for disconnects so that the callbacks will be ready
277                   // to be fired on the next call to poll()
278                   maybeTriggerWakeup();
279               }
280               // throw InterruptException if this thread is interrupted
281               maybeThrowInterruptException();
282   
283               // try again to send requests since buffer space may have been
284               // cleared or a connect finished in the poll
285               trySend(timer.currentTimeMs());
286   
287               // fail requests that couldn't be sent if they have expired
288               failExpiredRequests(timer.currentTimeMs());
289   
290               // clean unsent requests collection to keep the map from growing indefinitely
291               unsent.clean();
292           } finally {
293               lock.unlock();
294           }
295   
296           // called without the lock to avoid deadlock potential if handlers need to acquire locks
297           firePendingCompletedRequests();
298   
299           metadata.maybeThrowAnyException();
300       }

309       /**
310        * Poll for network IO in best-effort only trying to transmit the ready-to-send request
311        * Do not check any pending requests or metadata errors so that no exception should ever
312        * be thrown, also no wakeups be triggered and no interrupted exception either.
313        */
314       public void transmitSends() {
315           Timer timer = time.timer(0);
316   
317           // do not try to handle any disconnects, prev request failures, metadata exception etc;
318           // just try once and return immediately
319           lock.lock();
320           try {
321               // send all the requests we can send now
322               trySend(timer.currentTimeMs());
323   
324               client.poll(0, timer.currentTimeMs());
325           } finally {
326               lock.unlock();
327           }
328       }

405       private void firePendingCompletedRequests() {
406           boolean completedRequestsFired = false;
407           for (;;) {
408               RequestFutureCompletionHandler completionHandler = pendingCompletion.poll();
409               if (completionHandler == null)
410                   break;
411   
412               completionHandler.fireCompletion();
413               completedRequestsFired = true;
414           }
415   
416           // wakeup the client in case it is blocking in poll for this future's completion
417           if (completedRequestsFired)
418               client.wakeup();
419       }

421       private void checkDisconnects(long now) {
422           // any disconnects affecting requests that have already been transmitted will be handled
423           // by NetworkClient, so we just need to check whether connections for any of the unsent
424           // requests have been disconnected; if they have, then we complete the corresponding future
425           // and set the disconnect flag in the ClientResponse
426           for (Node node : unsent.nodes()) {
427               if (client.connectionFailed(node)) {
428                   // Remove entry before invoking request callback to avoid callbacks handling
429                   // coordinator failures traversing the unsent list again.
430                   Collection<ClientRequest> requests = unsent.remove(node);
431                   for (ClientRequest request : requests) {
432                       RequestFutureCompletionHandler handler = (RequestFutureCompletionHandler) request.callback();
433                       AuthenticationException authenticationException = client.authenticationException(node);
434                       handler.onComplete(new ClientResponse(request.makeHeader(request.requestBuilder().latestAllowedVersion()),
435                               request.callback(), request.destination(), request.createdTimeMs(), now, true,
436                               null, authenticationException, null));
437                   }
438               }
439           }
440       }
441   
442       private void handlePendingDisconnects() {
443           lock.lock();
444           try {
445               while (true) {
446                   Node node = pendingDisconnects.poll();
447                   if (node == null)
448                       break;
449   
450                   failUnsentRequests(node, DisconnectException.INSTANCE);
451                   client.disconnect(node.idString());
452               }
453           } finally {
454               lock.unlock();
455           }
456       }

458       public void disconnectAsync(Node node) {
459           pendingDisconnects.offer(node);
460           client.wakeup();
461       }

463       private void failExpiredRequests(long now) {

487       long trySend(long now) {
488           long pollDelayMs = maxPollTimeoutMs;
489   
490           // send any requests that can be sent now
491           for (Node node : unsent.nodes()) {
492               Iterator<ClientRequest> iterator = unsent.requestIterator(node);
493               if (iterator.hasNext())
494                   pollDelayMs = Math.min(pollDelayMs, client.pollDelayMs(node, now));
495   
496               while (iterator.hasNext()) {
497                   ClientRequest request = iterator.next();
498                   if (client.ready(node, now)) {
499                       client.send(request, now);
500                       iterator.remove();
501                   } else {
502                       // try next node when current node is not ready
503                       break;
504                   }
505               }
506           }
507           return pollDelayMs;
508       }

581       private class RequestFutureCompletionHandler implements RequestCompletionHandler {
582           private final RequestFuture<ClientResponse> future;
583           private ClientResponse response;
584           private RuntimeException e;
585   
586           private RequestFutureCompletionHandler() {
587               this.future = new RequestFuture<>();
588           }
589   
590           public void fireCompletion() {
591               if (e != null) {
592                   future.raise(e);
593               } else if (response.authenticationException() != null) {
594                   future.raise(response.authenticationException());
595               } else if (response.wasDisconnected()) {
596                   log.debug("Cancelled request with header {} due to node {} being disconnected",
597                           response.requestHeader(), response.destination());
598                   future.raise(DisconnectException.INSTANCE);
599               } else if (response.versionMismatch() != null) {
600                   future.raise(response.versionMismatch());
601               } else {
602                   future.complete(response);
603               }
604           }
605   
606           public void onFailure(RuntimeException e) {
607               this.e = e;
608               pendingCompletion.add(this);
609           }
610   
611           @Override
612           public void onComplete(ClientResponse response) {
613               this.response = response;
614               pendingCompletion.add(this);
615           }
616       }

618       /**
619        * When invoking poll from a multi-threaded environment, it is possible that the condition that
620        * the caller is awaiting has already been satisfied prior to the invocation of poll. We therefore
621        * introduce this interface to push the condition checking as close as possible to the invocation
622        * of poll. In particular, the check will be done while holding the lock used to protect concurrent
623        * access to {@link org.apache.kafka.clients.NetworkClient}, which means implementations must be
624        * very careful about locking order if the callback must acquire additional locks.
625        */
626       public interface PollCondition {
627           /**
628            * Return whether the caller is still awaiting an IO event.
629            * @return true if so, false otherwise.
630            */
631           boolean shouldBlock();
632       }

634       /*
635        * A thread-safe helper class to hold requests per node that have not been sent yet
636        */
637       private static final class UnsentRequests {
638           private final ConcurrentMap<Node, ConcurrentLinkedQueue<ClientRequest>> unsent;
639   
640           private UnsentRequests() {
641               unsent = new ConcurrentHashMap<>();
642           }
643   
644           public void put(Node node, ClientRequest request) {
645               // the lock protects the put from a concurrent removal of the queue for the node
646               synchronized (unsent) {
647                   ConcurrentLinkedQueue<ClientRequest> requests = unsent.computeIfAbsent(node, key -> new ConcurrentLinkedQueue<>());
648                   requests.add(request);
649               }
650           }

653               ConcurrentLinkedQueue<ClientRequest> requests = unsent.get(node);
654               return requests == null ? 0 : requests.size();
655           }
656   
657           public int requestCount() {
658               int total = 0;
659               for (ConcurrentLinkedQueue<ClientRequest> requests : unsent.values())
660                   total += requests.size();
661               return total;
662           }

665               ConcurrentLinkedQueue<ClientRequest> requests = unsent.get(node);
666               return requests != null && !requests.isEmpty();
667           }

669           public boolean hasRequests() {
670               for (ConcurrentLinkedQueue<ClientRequest> requests : unsent.values())
671                   if (!requests.isEmpty())
672                       return true;

674           }

676           private Collection<ClientRequest> removeExpiredRequests(long now) {
677               List<ClientRequest> expiredRequests = new ArrayList<>();
678               for (ConcurrentLinkedQueue<ClientRequest> requests : unsent.values()) {
679                   Iterator<ClientRequest> requestIterator = requests.iterator();
680                   while (requestIterator.hasNext()) {
681                       ClientRequest request = requestIterator.next();
682                       long elapsedMs = Math.max(0, now - request.createdTimeMs());
683                       if (elapsedMs > request.requestTimeoutMs()) {
684                           expiredRequests.add(request);
685                           requestIterator.remove();
686                       } else
687                           break;
688                   }
689               }
690               return expiredRequests;
691           }

696               synchronized (unsent) {

700                       if (requests.isEmpty())

702                   }
