44       public static Builder forMagic(byte magic, ProduceRequestData data) {
45           // Message format upgrades correspond with a bump in the produce request version. Older
46           // message format versions are generally not supported by the produce request versions
47           // following the bump.
48   
49           final short minVersion;
50           final short maxVersion;
51           if (magic < RecordBatch.MAGIC_VALUE_V2) {
52               minVersion = 2;
53               maxVersion = 2;
54           } else {
55               minVersion = 3;
56               maxVersion = ApiKeys.PRODUCE.latestVersion();
57           }
58           return new Builder(minVersion, maxVersion, data);
59       }
60   
61       public static Builder forCurrentMagic(ProduceRequestData data) {
62           return forMagic(RecordBatch.CURRENT_MAGIC_VALUE, data);
63       }

85           private ProduceRequest build(short version, boolean validate) {
86               if (validate) {
87                   // Validate the given records first
88                   data.topicData().forEach(tpd ->
89                           tpd.partitionData().forEach(partitionProduceData ->
90                                   ProduceRequest.validateRecords(version, partitionProduceData.records())));
91               }
92               return new ProduceRequest(data, version);
93           }

95           @Override
96           public String toString() {
97               StringBuilder bld = new StringBuilder();
98               bld.append("(type=ProduceRequest")
99                       .append(", acks=").append(data.acks())
100                       .append(", timeout=").append(data.timeoutMs())
101                       .append(", partitionRecords=(").append(data.topicData().stream().flatMap(d -> d.partitionData().stream()).collect(Collectors.toList()))
102                       .append("), transactionalId='").append(data.transactionalId() != null ? data.transactionalId() : "")
103                       .append("'");
104               return bld.toString();
105           }

108       /**
109        * We have to copy acks, timeout, transactionalId and partitionSizes from data since data maybe reset to eliminate
110        * the reference to ByteBuffer but those metadata are still useful.
111        */
112       private final short acks;
113       private final int timeout;
114       private final String transactionalId;
115       // This is set to null by `clearPartitionRecords` to prevent unnecessary memory retention when a produce request is
116       // put in the purgatory (due to client throttling, it can take a while before the response is sent).
117       // Care should be taken in methods that use this field.
118       private volatile ProduceRequestData data;
119       // the partitionSizes is lazily initialized since it is used by server-side in production.
120       private volatile Map<TopicPartition, Integer> partitionSizes;
121   
122       public ProduceRequest(ProduceRequestData produceRequestData, short version) {
123           super(ApiKeys.PRODUCE, version);
124           this.data = produceRequestData;
125           this.acks = data.acks();
126           this.timeout = data.timeoutMs();
127           this.transactionalId = data.transactionalId();
128       }
129   
130       // visible for testing
131       Map<TopicPartition, Integer> partitionSizes() {
132           if (partitionSizes == null) {
133               // this method may be called by different thread (see the comment on data)
134               synchronized (this) {
135                   if (partitionSizes == null) {
136                       partitionSizes = new HashMap<>();
137                       data.topicData().forEach(topicData ->
138                           topicData.partitionData().forEach(partitionData ->
139                               partitionSizes.compute(new TopicPartition(topicData.name(), partitionData.index()),
140                                   (ignored, previousValue) ->
141                                       partitionData.records().sizeInBytes() + (previousValue == null ? 0 : previousValue))
142                           )
143                       );
144                   }
145               }
146           }
147           return partitionSizes;
148       }
149   
150       /**
151        * @return data or IllegalStateException if the data is removed (to prevent unnecessary memory retention).
152        */
153       @Override
154       public ProduceRequestData data() {
155           // Store it in a local variable to protect against concurrent updates
156           ProduceRequestData tmp = data;
157           if (tmp == null)
158               throw new IllegalStateException("The partition records are no longer available because clearPartitionRecords() has been invoked.");
159           return tmp;
160       }

162       @Override
163       public String toString(boolean verbose) {
164           // Use the same format as `Struct.toString()`
165           StringBuilder bld = new StringBuilder();
166           bld.append("{acks=").append(acks)
167                   .append(",timeout=").append(timeout);
168   
169           if (verbose)
170               bld.append(",partitionSizes=").append(Utils.mkString(partitionSizes(), "[", "]", "=", ","));
171           else
172               bld.append(",numPartitions=").append(partitionSizes().size());
173   
174           bld.append("}");
175           return bld.toString();
176       }

178       @Override
179       public ProduceResponse getErrorResponse(int throttleTimeMs, Throwable e) {
180           /* In case the producer doesn't actually want any response */
181           if (acks == 0) return null;
182           ApiError apiError = ApiError.fromThrowable(e);
183           ProduceResponseData data = new ProduceResponseData().setThrottleTimeMs(throttleTimeMs);
184           partitionSizes().forEach((tp, ignored) -> {
185               ProduceResponseData.TopicProduceResponse tpr = data.responses().find(tp.topic());
186               if (tpr == null) {
187                   tpr = new ProduceResponseData.TopicProduceResponse().setName(tp.topic());
188                   data.responses().add(tpr);
189               }
190               tpr.partitionResponses().add(new ProduceResponseData.PartitionProduceResponse()
191                       .setIndex(tp.partition())
192                       .setRecordErrors(Collections.emptyList())
193                       .setBaseOffset(INVALID_OFFSET)
194                       .setLogAppendTimeMs(RecordBatch.NO_TIMESTAMP)
195                       .setLogStartOffset(INVALID_OFFSET)
196                       .setErrorMessage(apiError.message())
197                       .setErrorCode(apiError.error().code()));
198           });
199           return new ProduceResponse(data);
200       }

204           Errors error = Errors.forException(e);
205           return Collections.singletonMap(error, partitionSizes().size());

226       public static void validateRecords(short version, BaseRecords baseRecords) {
227           if (version >= 3) {
228               if (baseRecords instanceof Records) {
229                   Records records = (Records) baseRecords;
230                   Iterator<? extends RecordBatch> iterator = records.batches().iterator();
231                   if (!iterator.hasNext())
232                       throw new InvalidRecordException("Produce requests with version " + version + " must have at least " +
233                               "one record batch");
234   
235                   RecordBatch entry = iterator.next();
236                   if (entry.magic() != RecordBatch.MAGIC_VALUE_V2)
237                       throw new InvalidRecordException("Produce requests with version " + version + " are only allowed to " +
238                               "contain record batches with magic version 2");
239                   if (version < 7 && entry.compressionType() == CompressionType.ZSTD) {
240                       throw new UnsupportedCompressionTypeException("Produce requests with version " + version + " are not allowed to " +
241                               "use ZStandard compression");
242                   }
243   
244                   if (iterator.hasNext())
245                       throw new InvalidRecordException("Produce requests with version " + version + " are only allowed to " +
246                               "contain exactly one record batch");
247               }
248           }
249   
250           // Note that we do not do similar validation for older versions to ensure compatibility with
251           // clients which send the wrong magic version in the wrong version of the produce request. The broker
252           // did not do this validation before, so we maintain that behavior here.
253       }

256           return new ProduceRequest(new ProduceRequestData(new ByteBufferAccessor(buffer), version), version);

259       public static byte requiredMagicForVersion(short produceRequestVersion) {
260           if (produceRequestVersion < ApiKeys.PRODUCE.oldestVersion() || produceRequestVersion > ApiKeys.PRODUCE.latestVersion())
261               throw new IllegalArgumentException("Magic value to use for produce request version " +
262                       produceRequestVersion + " is not known");
263   
264           switch (produceRequestVersion) {
265               case 0:
266               case 1:
267                   return RecordBatch.MAGIC_VALUE_V0;
268   
269               case 2:
270                   return RecordBatch.MAGIC_VALUE_V1;
271   
272               default:
273                   return RecordBatch.MAGIC_VALUE_V2;
274           }
