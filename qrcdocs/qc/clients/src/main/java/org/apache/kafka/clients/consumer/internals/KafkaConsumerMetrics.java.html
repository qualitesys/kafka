17   package org.apache.kafka.clients.consumer.internals;
18   
19   import org.apache.kafka.common.MetricName;
20   import org.apache.kafka.common.metrics.Measurable;
21   import org.apache.kafka.common.metrics.Metrics;
22   import org.apache.kafka.common.metrics.Sensor;
23   import org.apache.kafka.common.metrics.stats.Avg;
24   import org.apache.kafka.common.metrics.stats.CumulativeSum;
25   import org.apache.kafka.common.metrics.stats.Max;
26   
27   import java.util.concurrent.TimeUnit;
28   
29   public class KafkaConsumerMetrics implements AutoCloseable {
30       private final MetricName lastPollMetricName;
31       private final Sensor timeBetweenPollSensor;
32       private final Sensor pollIdleSensor;
33       private final Sensor committedSensor;
34       private final Sensor commitSyncSensor;
35       private final Metrics metrics;
36       private long lastPollMs;
37       private long pollStartMs;
38       private long timeSinceLastPollMs;
39   
40       public KafkaConsumerMetrics(Metrics metrics, String metricGrpPrefix) {
41           this.metrics = metrics;
42           String metricGroupName = metricGrpPrefix + "-metrics";
43           Measurable lastPoll = (mConfig, now) -> {
44               if (lastPollMs == 0L)
45                   // if no poll is ever triggered, just return -1.
46                   return -1d;
47               else
48                   return TimeUnit.SECONDS.convert(now - lastPollMs, TimeUnit.MILLISECONDS);
49           };
50           this.lastPollMetricName = metrics.metricName("last-poll-seconds-ago",
51                   metricGroupName, "The number of seconds since the last poll() invocation.");
52           metrics.addMetric(lastPollMetricName, lastPoll);
53   
54           this.timeBetweenPollSensor = metrics.sensor("time-between-poll");
55           this.timeBetweenPollSensor.add(metrics.metricName("time-between-poll-avg",
56                   metricGroupName,
57                   "The average delay between invocations of poll() in milliseconds."),
58                   new Avg());
59           this.timeBetweenPollSensor.add(metrics.metricName("time-between-poll-max",
60                   metricGroupName,
61                   "The max delay between invocations of poll() in milliseconds."),
62                   new Max());
63   
64           this.pollIdleSensor = metrics.sensor("poll-idle-ratio-avg");
65           this.pollIdleSensor.add(metrics.metricName("poll-idle-ratio-avg",
66                   metricGroupName,
67                   "The average fraction of time the consumer's poll() is idle as opposed to waiting for the user code to process records."),
68                   new Avg());
69   
70           this.commitSyncSensor = metrics.sensor("commit-sync-time-ns-total");
71           this.commitSyncSensor.add(
72               metrics.metricName(
73                   "commit-sync-time-ns-total",
74                   metricGroupName,
75                   "The total time the consumer has spent in commitSync in nanoseconds"
76               ),
77               new CumulativeSum()
78           );
79   
80           this.committedSensor = metrics.sensor("committed-time-ns-total");
81           this.committedSensor.add(
82               metrics.metricName(
83                   "committed-time-ns-total",
84                   metricGroupName,
85                   "The total time the consumer has spent in committed in nanoseconds"
86               ),
87               new CumulativeSum()
88           );
89       }
90   
91       public void recordPollStart(long pollStartMs) {
92           this.pollStartMs = pollStartMs;
93           this.timeSinceLastPollMs = lastPollMs != 0L ? pollStartMs - lastPollMs : 0;
94           this.timeBetweenPollSensor.record(timeSinceLastPollMs);
95           this.lastPollMs = pollStartMs;
96       }
97   
98       public void recordPollEnd(long pollEndMs) {
99           long pollTimeMs = pollEndMs - pollStartMs;
100           double pollIdleRatio = pollTimeMs * 1.0 / (pollTimeMs + timeSinceLastPollMs);
101           this.pollIdleSensor.record(pollIdleRatio);
102       }
103   
104       public void recordCommitSync(long duration) {
105           this.commitSyncSensor.record(duration);
106       }
107   
108       public void recordCommitted(long duration) {
109           this.committedSensor.record(duration);
110       }
111   
112       @Override
113       public void close() {
114           metrics.removeMetric(lastPollMetricName);
115           metrics.removeSensor(timeBetweenPollSensor.name());
116           metrics.removeSensor(pollIdleSensor.name());
117           metrics.removeSensor(commitSyncSensor.name());
118           metrics.removeSensor(committedSensor.name());
119       }
120   }
