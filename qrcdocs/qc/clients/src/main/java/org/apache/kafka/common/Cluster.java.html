110       private Cluster(String clusterId,
111                       boolean isBootstrapConfigured,
112                       Collection<Node> nodes,
113                       Collection<PartitionInfo> partitions,
114                       Set<String> unauthorizedTopics,
115                       Set<String> invalidTopics,
116                       Set<String> internalTopics,
117                       Node controller,
118                       Map<String, Uuid> topicIds) {
119           this.isBootstrapConfigured = isBootstrapConfigured;
120           this.clusterResource = new ClusterResource(clusterId);
121           // make a randomized, unmodifiable copy of the nodes
122           List<Node> copy = new ArrayList<>(nodes);
123           Collections.shuffle(copy);
124           this.nodes = Collections.unmodifiableList(copy);
125   
126           // Index the nodes for quick lookup
127           Map<Integer, Node> tmpNodesById = new HashMap<>();
128           Map<Integer, List<PartitionInfo>> tmpPartitionsByNode = new HashMap<>(nodes.size());
129           for (Node node : nodes) {
130               tmpNodesById.put(node.id(), node);
131               // Populate the map here to make it easy to add the partitions per node efficiently when iterating over
132               // the partitions
133               tmpPartitionsByNode.put(node.id(), new ArrayList<>());
134           }
135           this.nodesById = Collections.unmodifiableMap(tmpNodesById);
136   
137           // index the partition infos by topic, topic+partition, and node
138           // note that this code is performance sensitive if there are a large number of partitions so we are careful
139           // to avoid unnecessary work
140           Map<TopicPartition, PartitionInfo> tmpPartitionsByTopicPartition = new HashMap<>(partitions.size());
141           Map<String, List<PartitionInfo>> tmpPartitionsByTopic = new HashMap<>();
142           for (PartitionInfo p : partitions) {
143               tmpPartitionsByTopicPartition.put(new TopicPartition(p.topic(), p.partition()), p);
144               tmpPartitionsByTopic.computeIfAbsent(p.topic(), topic -> new ArrayList<>()).add(p);
145   
146               // The leader may not be known
147               if (p.leader() == null || p.leader().isEmpty())
148                   continue;
149   
150               // If it is known, its node information should be available
151               List<PartitionInfo> partitionsForNode = Objects.requireNonNull(tmpPartitionsByNode.get(p.leader().id()));
152               partitionsForNode.add(p);
153           }
154   
155           // Update the values of `tmpPartitionsByNode` to contain unmodifiable lists
156           for (Map.Entry<Integer, List<PartitionInfo>> entry : tmpPartitionsByNode.entrySet()) {
157               tmpPartitionsByNode.put(entry.getKey(), Collections.unmodifiableList(entry.getValue()));
158           }
159   
160           // Populate `tmpAvailablePartitionsByTopic` and update the values of `tmpPartitionsByTopic` to contain
161           // unmodifiable lists
162           Map<String, List<PartitionInfo>> tmpAvailablePartitionsByTopic = new HashMap<>(tmpPartitionsByTopic.size());
163           for (Map.Entry<String, List<PartitionInfo>> entry : tmpPartitionsByTopic.entrySet()) {
164               String topic = entry.getKey();
165               List<PartitionInfo> partitionsForTopic = Collections.unmodifiableList(entry.getValue());
166               tmpPartitionsByTopic.put(topic, partitionsForTopic);
167               // Optimise for the common case where all partitions are available
168               boolean foundUnavailablePartition = partitionsForTopic.stream().anyMatch(p -> p.leader() == null);
169               List<PartitionInfo> availablePartitionsForTopic;
170               if (foundUnavailablePartition) {
171                   availablePartitionsForTopic = new ArrayList<>(partitionsForTopic.size());
172                   for (PartitionInfo p : partitionsForTopic) {
173                       if (p.leader() != null)
174                           availablePartitionsForTopic.add(p);
175                   }
176                   availablePartitionsForTopic = Collections.unmodifiableList(availablePartitionsForTopic);
177               } else {
178                   availablePartitionsForTopic = partitionsForTopic;
179               }
180               tmpAvailablePartitionsByTopic.put(topic, availablePartitionsForTopic);
181           }
182   
183           this.partitionsByTopicPartition = Collections.unmodifiableMap(tmpPartitionsByTopicPartition);
184           this.partitionsByTopic = Collections.unmodifiableMap(tmpPartitionsByTopic);
185           this.availablePartitionsByTopic = Collections.unmodifiableMap(tmpAvailablePartitionsByTopic);
186           this.partitionsByNode = Collections.unmodifiableMap(tmpPartitionsByNode);
187           this.topicIds = Collections.unmodifiableMap(topicIds);
188           Map<Uuid, String> tmpTopicNames = new HashMap<>();
189           topicIds.forEach((key, value) -> tmpTopicNames.put(value, key));
190           this.topicNames = Collections.unmodifiableMap(tmpTopicNames);
191   
192           this.unauthorizedTopics = Collections.unmodifiableSet(unauthorizedTopics);
193           this.invalidTopics = Collections.unmodifiableSet(invalidTopics);
194           this.internalTopics = Collections.unmodifiableSet(internalTopics);
195           this.controller = controller;
196       }

206       /**
207        * Create a "bootstrap" cluster using the given list of host/ports
208        * @param addresses The addresses
209        * @return A cluster for these hosts/ports
210        */
211       public static Cluster bootstrap(List<InetSocketAddress> addresses) {
212           List<Node> nodes = new ArrayList<>();
213           int nodeId = -1;
214           for (InetSocketAddress address : addresses)
215               nodes.add(new Node(nodeId--, address.getHostString(), address.getPort()));
216           return new Cluster(null, true, nodes, new ArrayList<>(0),
217               Collections.emptySet(), Collections.emptySet(), Collections.emptySet(), null, Collections.emptyMap());
218       }

220       /**
221        * Return a copy of this cluster combined with `partitions`.
222        */
223       public Cluster withPartitions(Map<TopicPartition, PartitionInfo> partitions) {
224           Map<TopicPartition, PartitionInfo> combinedPartitions = new HashMap<>(this.partitionsByTopicPartition);
225           combinedPartitions.putAll(partitions);
226           return new Cluster(clusterResource.clusterId(), this.nodes, combinedPartitions.values(),
227                   new HashSet<>(this.unauthorizedTopics), new HashSet<>(this.invalidTopics),
228                   new HashSet<>(this.internalTopics), this.controller);
229       }

247       /**
248        * Get the node by node id if the replica for the given partition is online
249        * @param partition
250        * @param id
251        * @return the node
252        */
253       public Optional<Node> nodeIfOnline(TopicPartition partition, int id) {
254           Node node = nodeById(id);
255           if (node != null && !Arrays.asList(partition(partition).offlineReplicas()).contains(node)) {
256               return Optional.of(node);
257           } else {
258               return Optional.empty();
259           }
260       }

262       /**
263        * Get the current leader for the given topic-partition
264        * @param topicPartition The topic and partition we want to know the leader for
265        * @return The node that is the leader for this topic-partition, or null if there is currently no leader
266        */
267       public Node leaderFor(TopicPartition topicPartition) {
268           PartitionInfo info = partitionsByTopicPartition.get(topicPartition);
269           if (info == null)
270               return null;
271           else
272               return info.leader();
273       }

371       @Override
372       public boolean equals(Object o) {
373           if (this == o) return true;
374           if (o == null || getClass() != o.getClass()) return false;
375           Cluster cluster = (Cluster) o;
376           return isBootstrapConfigured == cluster.isBootstrapConfigured &&
377                   Objects.equals(nodes, cluster.nodes) &&
378                   Objects.equals(unauthorizedTopics, cluster.unauthorizedTopics) &&
379                   Objects.equals(invalidTopics, cluster.invalidTopics) &&
380                   Objects.equals(internalTopics, cluster.internalTopics) &&
381                   Objects.equals(controller, cluster.controller) &&
382                   Objects.equals(partitionsByTopicPartition, cluster.partitionsByTopicPartition) &&
383                   Objects.equals(clusterResource, cluster.clusterResource);
384       }
