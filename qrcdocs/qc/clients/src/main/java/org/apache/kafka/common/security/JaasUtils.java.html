17   package org.apache.kafka.common.security;
18   
19   import org.apache.kafka.common.KafkaException;
20   import org.slf4j.Logger;
21   import org.slf4j.LoggerFactory;
22   
23   import javax.security.auth.login.Configuration;
24   
25   public final class JaasUtils {
26       private static final Logger LOG = LoggerFactory.getLogger(JaasUtils.class);
27       public static final String JAVA_LOGIN_CONFIG_PARAM = "java.security.auth.login.config";
28   
29       public static final String SERVICE_NAME = "serviceName";
30   
31       public static final String ZK_SASL_CLIENT = "zookeeper.sasl.client";
32       public static final String ZK_LOGIN_CONTEXT_NAME_KEY = "zookeeper.sasl.clientconfig";
33   
34       private static final String DEFAULT_ZK_LOGIN_CONTEXT_NAME = "Client";
35       private static final String DEFAULT_ZK_SASL_CLIENT = "true";
36   
37       private JaasUtils() {}
38   
39       public static String zkSecuritySysConfigString() {
40           String loginConfig = System.getProperty(JAVA_LOGIN_CONFIG_PARAM);
41           String clientEnabled = System.getProperty(ZK_SASL_CLIENT, "default:" + DEFAULT_ZK_SASL_CLIENT);
42           String contextName = System.getProperty(ZK_LOGIN_CONTEXT_NAME_KEY, "default:" + DEFAULT_ZK_LOGIN_CONTEXT_NAME);
43           return "[" +
44                   JAVA_LOGIN_CONFIG_PARAM + "=" + loginConfig +
45                   ", " +
46                   ZK_SASL_CLIENT + "=" + clientEnabled +
47                   ", " +
48                   ZK_LOGIN_CONTEXT_NAME_KEY + "=" + contextName +
49                   "]";
50       }
51   
52       public static boolean isZkSaslEnabled() {
53           // Technically a client must also check if TLS mutual authentication has been configured,
54           // but we will leave that up to the client code to determine since direct connectivity to ZooKeeper
55           // has been deprecated in many clients and we don't wish to re-introduce a ZooKeeper jar dependency here.
56           boolean zkSaslEnabled = Boolean.parseBoolean(System.getProperty(ZK_SASL_CLIENT, DEFAULT_ZK_SASL_CLIENT));
57           String zkLoginContextName = System.getProperty(ZK_LOGIN_CONTEXT_NAME_KEY, DEFAULT_ZK_LOGIN_CONTEXT_NAME);
58   
59           LOG.debug("Checking login config for Zookeeper JAAS context {}", zkSecuritySysConfigString());
60   
61           boolean foundLoginConfigEntry;
62           try {
63               Configuration loginConf = Configuration.getConfiguration();
64               foundLoginConfigEntry = loginConf.getAppConfigurationEntry(zkLoginContextName) != null;
65           } catch (Exception e) {
66               throw new KafkaException("Exception while loading Zookeeper JAAS login context " +
67                       zkSecuritySysConfigString(), e);
68           }
69   
70           if (foundLoginConfigEntry && !zkSaslEnabled) {
71               LOG.error("JAAS configuration is present, but system property " +
72                           ZK_SASL_CLIENT + " is set to false, which disables " +
73                           "SASL in the ZooKeeper client");
74               throw new KafkaException("Exception while determining if ZooKeeper is secure " +
75                       zkSecuritySysConfigString());
76           }
77   
78           return foundLoginConfigEntry;
79       }
80   }
