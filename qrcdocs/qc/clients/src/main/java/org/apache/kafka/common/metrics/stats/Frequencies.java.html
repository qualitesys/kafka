23   import org.apache.kafka.common.metrics.stats.Histogram.BinScheme;
24   import org.apache.kafka.common.metrics.stats.Histogram.ConstantBinScheme;
25   
26   import java.util.ArrayList;
27   import java.util.List;
28   
29   /**
30    * A {@link CompoundStat} that represents a normalized distribution with a {@link Frequency} metric for each
31    * bucketed value. The values of the {@link Frequency} metrics specify the frequency of the center value appearing
32    * relative to the total number of values recorded.
33    * <p>
34    * For example, consider a component that records failure or success of an operation using boolean values, with
35    * one metric to capture the percentage of operations that failed another to capture the percentage of operations
36    * that succeeded.
37    * <p>
38    * This can be accomplish by created a {@link org.apache.kafka.common.metrics.Sensor Sensor} to record the values,
39    * with 0.0 for false and 1.0 for true. Then, create a single {@link Frequencies} object that has two
40    * {@link Frequency} metrics: one centered around 0.0 and another centered around 1.0. The {@link Frequencies}
41    * object is a {@link CompoundStat}, and so it can be {@link org.apache.kafka.common.metrics.Sensor#add(CompoundStat)
42    * added directly to a Sensor} so the metrics are created automatically.
43    */
44   public class Frequencies extends SampledStat implements CompoundStat {
45   
46       /**
47        * Create a Frequencies instance with metrics for the frequency of a boolean sensor that records 0.0 for
48        * false and 1.0 for true.
49        *
50        * @param falseMetricName the name of the metric capturing the frequency of failures; may be null if not needed
51        * @param trueMetricName  the name of the metric capturing the frequency of successes; may be null if not needed
52        * @return the Frequencies instance; never null
53        * @throws IllegalArgumentException if both {@code falseMetricName} and {@code trueMetricName} are null
54        */
55       public static Frequencies forBooleanValues(MetricName falseMetricName, MetricName trueMetricName) {
56           List<Frequency> frequencies = new ArrayList<>();
57           if (falseMetricName != null) {
58               frequencies.add(new Frequency(falseMetricName, 0.0));
59           }
60           if (trueMetricName != null) {
61               frequencies.add(new Frequency(trueMetricName, 1.0));
62           }
63           if (frequencies.isEmpty()) {
64               throw new IllegalArgumentException("Must specify at least one metric name");
65           }
66           Frequency[] frequencyArray = frequencies.toArray(new Frequency[0]);
67           return new Frequencies(2, 0.0, 1.0, frequencyArray);
68       }
69   
70       private final Frequency[] frequencies;
71       private final BinScheme binScheme;
72   
73       /**
74        * Create a Frequencies that captures the values in the specified range into the given number of buckets,
75        * where the buckets are centered around the minimum, maximum, and intermediate values.
76        *
77        * @param buckets     the number of buckets; must be at least 1
78        * @param min         the minimum value to be captured
79        * @param max         the maximum value to be captured
80        * @param frequencies the list of {@link Frequency} metrics, which at most should be one per bucket centered
81        *                    on the bucket's value, though not every bucket need to correspond to a metric if the
82        *                    value is not needed
83        * @throws IllegalArgumentException if any of the {@link Frequency} objects do not have a
84        *                                  {@link Frequency#centerValue() center value} within the specified range
85        */
86       public Frequencies(int buckets, double min, double max, Frequency... frequencies) {
87           super(0.0); // initial value is unused by this implementation
88           if (max < min) {
89               throw new IllegalArgumentException("The maximum value " + max
90                                                          + " must be greater than the minimum value " + min);
91           }
92           if (buckets < 1) {
93               throw new IllegalArgumentException("Must be at least 1 bucket");
94           }
95           if (buckets < frequencies.length) {
96               throw new IllegalArgumentException("More frequencies than buckets");
97           }
98           this.frequencies = frequencies;
99           for (Frequency freq : frequencies) {
100               if (min > freq.centerValue() || max < freq.centerValue()) {
101                   throw new IllegalArgumentException("The frequency centered at '" + freq.centerValue()
102                                                              + "' is not within the range [" + min + "," + max + "]");
103               }
104           }
105           double halfBucketWidth = (max - min) / (buckets - 1) / 2.0;
106           this.binScheme = new ConstantBinScheme(buckets, min - halfBucketWidth, max + halfBucketWidth);
107       }
108   
109       @Override
110       public List<NamedMeasurable> stats() {
111           List<NamedMeasurable> ms = new ArrayList<>(frequencies.length);
112           for (Frequency frequency : frequencies) {
113               final double center = frequency.centerValue();
114               ms.add(new NamedMeasurable(frequency.name(), new Measurable() {
115                   public double measure(MetricConfig config, long now) {
116                       return frequency(config, now, center);
117                   }
118               }));
119           }
120           return ms;
121       }
122   
123       /**
124        * Return the computed frequency describing the number of occurrences of the values in the bucket for the given
125        * center point, relative to the total number of occurrences in the samples.
126        *
127        * @param config      the metric configuration
128        * @param now         the current time in milliseconds
129        * @param centerValue the value corresponding to the center point of the bucket
130        * @return the frequency of the values in the bucket relative to the total number of samples
131        */
132       public double frequency(MetricConfig config, long now, double centerValue) {
133           purgeObsoleteSamples(config, now);
134           long totalCount = 0;
135           for (Sample sample : samples) {
136               totalCount += sample.eventCount;
137           }
138           if (totalCount == 0) {
139               return 0.0d;
140           }
141           // Add up all of the counts in the bin corresponding to the center value
142           float count = 0.0f;
143           int binNum = binScheme.toBin(centerValue);
144           for (Sample s : samples) {
145               HistogramSample sample = (HistogramSample) s;
146               float[] hist = sample.histogram.counts();
147               count += hist[binNum];
148           }
149           // Compute the ratio of counts to total counts
150           return count / (double) totalCount;
151       }
152   
153       double totalCount() {
154           long count = 0;
155           for (Sample sample : samples) {
156               count += sample.eventCount;
157           }
158           return count;
159       }
160   
161       @Override
162       public double combine(List<Sample> samples, MetricConfig config, long now) {
163           return totalCount();
164       }
165   
166       @Override
167       protected HistogramSample newSample(long timeMs) {
168           return new HistogramSample(binScheme, timeMs);
169       }
170   
171       @Override
172       protected void update(Sample sample, MetricConfig config, double value, long timeMs) {
173           HistogramSample hist = (HistogramSample) sample;
174           hist.histogram.record(value);
175       }
176   
177       private static class HistogramSample extends SampledStat.Sample {
178   
179           private final Histogram histogram;
180   
181           private HistogramSample(BinScheme scheme, long now) {
182               super(0.0, now);
183               histogram = new Histogram(scheme);
184           }
185   
186           @Override
187           public void reset(long now) {
188               super.reset(now);
189               histogram.clear();
190           }
191       }
