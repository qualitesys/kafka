54   /**
55    * A class for tracking the topics, partitions, and offsets for the consumer. A partition
56    * is "assigned" either directly with {@link #assignFromUser(Set)} (manual assignment)
57    * or with {@link #assignFromSubscribed(Collection)} (automatic assignment from subscription).
58    *
59    * Once assigned, the partition is not considered "fetchable" until its initial position has
60    * been set with {@link #seekValidated(TopicPartition, FetchPosition)}. Fetchable partitions track a fetch
61    * position which is used to set the offset of the next fetch, and a consumed position
62    * which is the last offset that has been returned to the user. You can suspend fetching
63    * from a partition through {@link #pause(TopicPartition)} without affecting the fetched/consumed
64    * offsets. The partition will remain unfetchable until the {@link #resume(TopicPartition)} is
65    * used. You can also query the pause state independently with {@link #isPaused(TopicPartition)}.
66    *
67    * Note that pause state as well as fetch/consumed positions are not preserved when partition
68    * assignment is changed whether directly by the user or through a group rebalance.
69    *
70    * Thread Safety: this class is thread-safe.
71    */

215       /**
216        * Change the assignment to the specified partitions provided by the user,
217        * note this is different from {@link #assignFromSubscribed(Collection)}
218        * whose input partitions are provided from the subscribed topics.
219        */
220       public synchronized boolean assignFromUser(Set<TopicPartition> partitions) {
221           setSubscriptionType(SubscriptionType.USER_ASSIGNED);
222   
223           if (this.assignment.partitionSet().equals(partitions))
224               return false;
225   
226           assignmentId++;
227   
228           // update the subscribed topics
229           Set<String> manualSubscribedTopics = new HashSet<>();
230           Map<TopicPartition, TopicPartitionState> partitionToState = new HashMap<>();
231           for (TopicPartition partition : partitions) {
232               TopicPartitionState state = assignment.stateValue(partition);
233               if (state == null)
234                   state = new TopicPartitionState();
235               partitionToState.put(partition, state);
236   
237               manualSubscribedTopics.add(partition.topic());
238           }
239   
240           this.assignment.set(partitionToState);
241           return changeSubscription(manualSubscribedTopics);
242       }

244       /**
245        * @return true if assignments matches subscription, otherwise false
246        */
247       public synchronized boolean checkAssignmentMatchedSubscription(Collection<TopicPartition> assignments) {
248           for (TopicPartition topicPartition : assignments) {
249               if (this.subscribedPattern != null) {
250                   if (!this.subscribedPattern.matcher(topicPartition.topic()).matches()) {
251                       log.info("Assigned partition {} for non-subscribed topic regex pattern; subscription pattern is {}",
252                           topicPartition,
253                           this.subscribedPattern);
254   
255                       return false;
256                   }
257               } else {
258                   if (!this.subscription.contains(topicPartition.topic())) {
259                       log.info("Assigned partition {} for non-subscribed topic; subscription is {}", topicPartition, this.subscription);
260   
261                       return false;
262                   }
263               }
264           }
265   
266           return true;
267       }

269       /**
270        * Change the assignment to the specified partitions returned from the coordinator, note this is
271        * different from {@link #assignFromUser(Set)} which directly set the assignment from user inputs.
272        */
273       public synchronized void assignFromSubscribed(Collection<TopicPartition> assignments) {
274           if (!this.hasAutoAssignedPartitions())
275               throw new IllegalArgumentException("Attempt to dynamically assign partitions while manual assignment in use");
276   
277           Map<TopicPartition, TopicPartitionState> assignedPartitionStates = new HashMap<>(assignments.size());
278           for (TopicPartition tp : assignments) {
279               TopicPartitionState state = this.assignment.stateValue(tp);
280               if (state == null)
281                   state = new TopicPartitionState();
282               assignedPartitionStates.put(tp, state);
283           }
284   
285           assignmentId++;
286           this.assignment.set(assignedPartitionStates);
287       }

338       /**
339        * Get the subscription topics for which metadata is required. For the leader, this will include
340        * the union of the subscriptions of all group members. For followers, it is just that member's
341        * subscription. This is used when querying topic metadata to detect the metadata changes which would
342        * require rebalancing. The leader fetches metadata for all topics in the group so that it
343        * can do the partition assignment (which requires at least partition counts for all topics

350           if (groupSubscription.isEmpty())
351               return subscription;

353               return groupSubscription;

356               // new subscription topics are returned.
357               Set<String> topics = new HashSet<>(groupSubscription);
358               topics.addAll(subscription);
359               return topics;
360           }
361       }

390       synchronized void maybeSeekUnvalidated(TopicPartition tp, FetchPosition position, OffsetResetStrategy requestedResetStrategy) {
391           TopicPartitionState state = assignedStateOrNull(tp);
392           if (state == null) {
393               log.debug("Skipping reset of partition {} since it is no longer assigned", tp);
394           } else if (!state.awaitingReset()) {
395               log.debug("Skipping reset of partition {} since reset is no longer needed", tp);
396           } else if (requestedResetStrategy != state.resetStrategy) {
397               log.debug("Skipping reset of partition {} since an alternative reset has been requested", tp);
398           } else {
399               log.info("Resetting offset for partition {} to position {}.", tp, position);
400               state.seekUnvalidated(position);
401           }
402       }

447       /**
448        * Enter the offset validation state if the leader for this partition is known to support a usable version of the
449        * OffsetsForLeaderEpoch API. If the leader node does not support the API, simply complete the offset validation.
450        *
451        * @param apiVersions supported API versions
452        * @param tp topic partition to validate

454        * @return true if we enter the offset validation state

457                                                                         TopicPartition tp,

478                                                                           FetchPosition requestPosition,

505                               epochEndOffset.endOffset(), Optional.of(epochEndOffset.leaderEpoch()),

514                                "broker is {}), but no reset policy is set", tp, currentPosition, divergentOffset);
515                       return Optional.of(new LogTruncation(tp, requestPosition, Optional.of(divergentOffset)));
516                   }
517               } else {
518                   state.completeValidation();
519               }
520           }

523       }
