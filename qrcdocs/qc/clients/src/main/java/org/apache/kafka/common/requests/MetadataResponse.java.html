147           List<PartitionInfo> partitions = new ArrayList<>();
148           Map<String, Uuid> topicIds = new HashMap<>();
149   
150           for (TopicMetadata metadata : topicMetadata()) {
151               if (metadata.error == Errors.NONE) {
152                   if (metadata.isInternal)
153                       internalTopics.add(metadata.topic);
154                   if (metadata.topicId() != null && !Uuid.ZERO_UUID.equals(metadata.topicId())) {
155                       topicIds.put(metadata.topic, metadata.topicId());
156                   }
157                   for (PartitionMetadata partitionMetadata : metadata.partitionMetadata) {
158                       partitions.add(toPartitionInfo(partitionMetadata, holder().brokers));
159                   }
160               }
161           }
162           return new Cluster(data.clusterId(), brokers(), partitions, topicsByError(Errors.TOPIC_AUTHORIZATION_FAILED),
163                   topicsByError(Errors.INVALID_TOPIC_EXCEPTION), internalTopics, controller(), topicIds);
164       }

260       // Prior to Kafka version 2.4 (which coincides with Metadata version 9), the broker
261       // does not propagate leader epoch information accurately while a reassignment is in
262       // progress. Relying on a stale epoch can lead to FENCED_LEADER_EPOCH errors which
263       // can prevent consumption throughout the course of a reassignment. It is safer in
264       // this case to revert to the behavior in previous protocol versions which checks
265       // leader status only.
266       private static boolean hasReliableLeaderEpochs(short version) {
267           return version >= 9;
268       }

420       private static class Holder {
421           private final Map<Integer, Node> brokers;

426               this.brokers = Collections.unmodifiableMap(createBrokers(data));
427               this.topicMetadata = createTopicMetadata(data);
428               this.controller = brokers.get(data.controllerId());

431           private Map<Integer, Node> createBrokers(MetadataResponseData data) {
432               return data.brokers().valuesList().stream().map(b -> new Node(b.nodeId(), b.host(), b.port(), b.rack()))
433                       .collect(Collectors.toMap(Node::id, Function.identity()));
434           }

436           private Collection<TopicMetadata> createTopicMetadata(MetadataResponseData data) {
437               List<TopicMetadata> topicMetadataList = new ArrayList<>();
438               for (MetadataResponseTopic topicMetadata : data.topics()) {
439                   Errors topicError = Errors.forCode(topicMetadata.errorCode());
440                   String topic = topicMetadata.name();
441                   Uuid topicId = topicMetadata.topicId();
442                   boolean isInternal = topicMetadata.isInternal();
443                   List<PartitionMetadata> partitionMetadataList = new ArrayList<>();
444   
445                   for (MetadataResponsePartition partitionMetadata : topicMetadata.partitions()) {
446                       Errors partitionError = Errors.forCode(partitionMetadata.errorCode());
447                       int partitionIndex = partitionMetadata.partitionIndex();
448   
449                       int leaderId = partitionMetadata.leaderId();
450                       Optional<Integer> leaderIdOpt = leaderId < 0 ? Optional.empty() : Optional.of(leaderId);
451   
452                       Optional<Integer> leaderEpoch = RequestUtils.getLeaderEpoch(partitionMetadata.leaderEpoch());
453                       TopicPartition topicPartition = new TopicPartition(topic, partitionIndex);
454                       partitionMetadataList.add(new PartitionMetadata(partitionError, topicPartition, leaderIdOpt,
455                               leaderEpoch, partitionMetadata.replicaNodes(), partitionMetadata.isrNodes(),
456                               partitionMetadata.offlineReplicas()));
457                   }
458   
459                   topicMetadataList.add(new TopicMetadata(topicError, topic, topicId, isInternal, partitionMetadataList,
460                           topicMetadata.topicAuthorizedOperations()));
461               }
462               return topicMetadataList;
463           }

478       // Visible for testing
479       public static MetadataResponse prepareResponse(boolean hasReliableEpoch,
480                                                      int throttleTimeMs,
481                                                      Collection<Node> brokers,
482                                                      String clusterId,
483                                                      int controllerId,
484                                                      List<MetadataResponseTopic> topics,
485                                                      int clusterAuthorizedOperations) {
486           MetadataResponseData responseData = new MetadataResponseData();
487           responseData.setThrottleTimeMs(throttleTimeMs);
488           brokers.forEach(broker ->
489               responseData.brokers().add(new MetadataResponseBroker()
490                   .setNodeId(broker.id())
491                   .setHost(broker.host())
492                   .setPort(broker.port())
493                   .setRack(broker.rack()))
494           );
495   
496           responseData.setClusterId(clusterId);
497           responseData.setControllerId(controllerId);
498           responseData.setClusterAuthorizedOperations(clusterAuthorizedOperations);
499   
500           topics.forEach(topicMetadata -> responseData.topics().add(topicMetadata));
501           return new MetadataResponse(responseData, hasReliableEpoch);
502       }
