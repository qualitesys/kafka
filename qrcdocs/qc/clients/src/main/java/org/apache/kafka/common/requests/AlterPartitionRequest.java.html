28   public class AlterPartitionRequest extends AbstractRequest {
29   
30       private final AlterPartitionRequestData data;
31   
32       public AlterPartitionRequest(AlterPartitionRequestData data, short apiVersion) {

35       }
36   
37       @Override
38       public AlterPartitionRequestData data() {

40       }
41   
42       /**
43        * Get an error response for a request with specified throttle time in the response if applicable
44        */
45       @Override
46       public AbstractResponse getErrorResponse(int throttleTimeMs, Throwable e) {
47           return new AlterPartitionResponse(new AlterPartitionResponseData()
48               .setThrottleTimeMs(throttleTimeMs)
49               .setErrorCode(Errors.forException(e).code()));

54       }

56       public static class Builder extends AbstractRequest.Builder<AlterPartitionRequest> {
57   
58           private final AlterPartitionRequestData data;
59   
60           /**
61            * Constructs a builder for AlterPartitionRequest.
62            *
63            * @param data The data to be sent. Note that because the version of the
64            *             request is not known at this time, it is expected that all
65            *             topics have a topic id and a topic name set.
66            * @param canUseTopicIds True if version 2 and above can be used.
67            */
68           public Builder(AlterPartitionRequestData data, boolean canUseTopicIds) {
69               super(
70                   ApiKeys.ALTER_PARTITION,
71                   ApiKeys.ALTER_PARTITION.oldestVersion(),
72                   // Version 1 is the maximum version that can be used without topic ids.
73                   canUseTopicIds ? ApiKeys.ALTER_PARTITION.latestVersion() : 1
74               );
75               this.data = data;
76           }
77   
78           @Override
79           public AlterPartitionRequest build(short version) {
80               return new AlterPartitionRequest(data, version);
81           }
