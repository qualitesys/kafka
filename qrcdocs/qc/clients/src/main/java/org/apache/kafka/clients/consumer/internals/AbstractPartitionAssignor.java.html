33   /**
34    * Abstract assignor implementation which does some common grunt work (in particular collecting
35    * partition counts which are always needed in assignors).
36    */
37   public abstract class AbstractPartitionAssignor implements ConsumerPartitionAssignor {
38       private static final Logger log = LoggerFactory.getLogger(AbstractPartitionAssignor.class);
39   
40       /**
41        * Perform the group assignment given the partition counts and member subscriptions
42        * @param partitionsPerTopic The number of partitions for each subscribed topic. Topics not in metadata will be excluded
43        *                           from this map.
44        * @param subscriptions Map from the member id to their respective topic subscription
45        * @return Map from each member to the list of partitions assigned to them.
46        */
47       public abstract Map<String, List<TopicPartition>> assign(Map<String, Integer> partitionsPerTopic,
48                                                                Map<String, Subscription> subscriptions);
49   
50       @Override
51       public GroupAssignment assign(Cluster metadata, GroupSubscription groupSubscription) {
52           Map<String, Subscription> subscriptions = groupSubscription.groupSubscription();
53           Set<String> allSubscribedTopics = new HashSet<>();
54           for (Map.Entry<String, Subscription> subscriptionEntry : subscriptions.entrySet())
55               allSubscribedTopics.addAll(subscriptionEntry.getValue().topics());
56   
57           Map<String, Integer> partitionsPerTopic = new HashMap<>();
58           for (String topic : allSubscribedTopics) {
59               Integer numPartitions = metadata.partitionCountForTopic(topic);
60               if (numPartitions != null && numPartitions > 0)
61                   partitionsPerTopic.put(topic, numPartitions);
62               else
63                   log.debug("Skipping assignment for topic {} since no metadata is available", topic);
64           }
65   
66           Map<String, List<TopicPartition>> rawAssignments = assign(partitionsPerTopic, subscriptions);
67   
68           // this class maintains no user data, so just wrap the results
69           Map<String, Assignment> assignments = new HashMap<>();
70           for (Map.Entry<String, List<TopicPartition>> assignmentEntry : rawAssignments.entrySet())
71               assignments.put(assignmentEntry.getKey(), new Assignment(assignmentEntry.getValue()));
72           return new GroupAssignment(assignments);
73       }
74   
75       protected static <K, V> void put(Map<K, List<V>> map, K key, V value) {
76           List<V> list = map.computeIfAbsent(key, k -> new ArrayList<>());
77           list.add(value);
78       }
79   
80       protected static List<TopicPartition> partitions(String topic, int numPartitions) {
81           List<TopicPartition> partitions = new ArrayList<>(numPartitions);
82           for (int i = 0; i < numPartitions; i++)
83               partitions.add(new TopicPartition(topic, i));
84           return partitions;
85       }

87       public static class MemberInfo implements Comparable<MemberInfo> {
88           public final String memberId;
89           public final Optional<String> groupInstanceId;
90   
91           public MemberInfo(String memberId, Optional<String> groupInstanceId) {
92               this.memberId = memberId;
93               this.groupInstanceId = groupInstanceId;
94           }
95   
96           @Override
97           public int compareTo(MemberInfo otherMemberInfo) {
98               if (this.groupInstanceId.isPresent() &&
99                       otherMemberInfo.groupInstanceId.isPresent()) {
100                   return this.groupInstanceId.get()
101                           .compareTo(otherMemberInfo.groupInstanceId.get());
102               } else if (this.groupInstanceId.isPresent()) {
103                   return -1;
104               } else if (otherMemberInfo.groupInstanceId.isPresent()) {
105                   return 1;
106               } else {
107                   return this.memberId.compareTo(otherMemberInfo.memberId);
108               }
109           }
110   
111           @Override
112           public boolean equals(Object o) {
113               return o instanceof MemberInfo && this.memberId.equals(((MemberInfo) o).memberId);
114           }
115   
116           /**
117            * We could just use member.id to be the hashcode, since it's unique
118            * across the group.
119            */
120           @Override
121           public int hashCode() {
122               return memberId.hashCode();
123           }
124   
125           @Override
126           public String toString() {
127               return "MemberInfo [member.id: " + memberId
128                       + ", group.instance.id: " + groupInstanceId.orElse("{}")
129                       + "]";
130           }
131       }
