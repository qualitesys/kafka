17   package org.apache.kafka.common.network;
18   
19   import java.io.Closeable;
20   
21   /**
22    * Metadata about a channel is provided in various places in the network stack. This
23    * registry is used as a common place to collect them.
24    */
25   public interface ChannelMetadataRegistry extends Closeable {
26   
27       /**
28        * Register information about the SSL cipher we are using.
29        *  Re-registering the information will overwrite the previous one.
30        */
31       void registerCipherInformation(CipherInformation cipherInformation);
32   
33       /**
34        * Get the currently registered cipher information.
35        */
36       CipherInformation cipherInformation();
37   
38       /**
39        * Register information about the client client we are using.
40        * Depending on the clients, the ApiVersionsRequest could be received
41        * multiple times or not at all. Re-registering the information will
42        * overwrite the previous one.
43        */
44       void registerClientInformation(ClientInformation clientInformation);
45   
46       /**
47        * Get the currently registered client information.
48        */
49       ClientInformation clientInformation();
50   
51       /**
52        * Unregister everything that has been registered and close the registry.
53        */
54       void close();
55   }
