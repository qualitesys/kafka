43   /**
44    * <p>
45    * <code>OAuthBearerLoginCallbackHandler</code> is an {@link AuthenticateCallbackHandler} that
46    * accepts {@link OAuthBearerTokenCallback} and {@link SaslExtensionsCallback} callbacks to
47    * perform the steps to request a JWT from an OAuth/OIDC provider using the
48    * <code>clientcredentials</code>. This grant type is commonly used for non-interactive
49    * "service accounts" where there is no user available to interactively supply credentials.
50    * </p>
51    *
52    * <p>
53    * The <code>OAuthBearerLoginCallbackHandler</code> is used on the client side to retrieve a JWT
54    * and the {@link OAuthBearerValidatorCallbackHandler} is used on the broker to validate the JWT
55    * that was sent to it by the client to allow access. Both the brokers and clients will need to
56    * be configured with their appropriate callback handlers and respective configuration for OAuth
57    * functionality to work.
58    * </p>
59    *
60    * <p>
61    * Note that while this callback handler class must be specified for a Kafka client that wants to
62    * use OAuth functionality, in the case of OAuth-based inter-broker communication, the callback
63    * handler must be used on the Kafka broker side as well.
64    * {@link }
65    * </p>
66    *
67    * <p>
68    * This {@link AuthenticateCallbackHandler} is enabled by specifying its class name in the Kafka
69    * configuration. For client use, specify the class name in the
70    * {@link org.apache.kafka.common.config.SaslConfigs#SASL_LOGIN_CALLBACK_HANDLER_CLASS}
71    * configuration like so:
72    *
73    * <code>
74    * sasl.login.callback.handler.class=org.apache.kafka.common.security.oauthbearer.secured.OAuthBearerLoginCallbackHandler
75    * </code>
76    * </p>
77    *
78    * <p>
79    * If using OAuth login on the broker side (for inter-broker communication), the callback handler
80    * class will be specified with a listener-based property:
81    * <code>listener.name.<listener name>.oauthbearer.sasl.login.callback.handler.class</code> like so:
82    *
83    * <code>
84    * listener.name.<listener name>.oauthbearer.sasl.login.callback.handler.class=org.apache.kafka.common.security.oauthbearer.secured.OAuthBearerLoginCallbackHandler
85    * </code>
86    * </p>
87    *
88    * <p>
89    * The Kafka configuration must also include JAAS configuration which includes the following
90    * OAuth-specific options:
91    *
92    * <ul>
93    *     <li><code>clientId</code>OAuth client ID (required)</li>
94    *     <li><code>clientSecret</code>OAuth client secret (required)</li>
95    *     <li><code>scope</code>OAuth scope (optional)</li>
96    * </ul>
97    * </p>
98    *
99    * <p>
100    * The JAAS configuration can also include any SSL options that are needed. The configuration
101    * options are the same as those specified by the configuration in
102    * {@link org.apache.kafka.common.config.SslConfigs#addClientSslSupport(ConfigDef)}.
103    * </p>
104    *
105    * <p>
106    * Here's an example of the JAAS configuration for a Kafka client:
107    *
108    * <code>
109    * sasl.jaas.config=org.apache.kafka.common.security.oauthbearer.OAuthBearerLoginModule required \
110    *   clientId="foo" \
111    *   clientSecret="bar" \
112    *   scope="baz" \
113    *   ssl.protocol="SSL" ;
114    * </code>
115    * </p>
116    *
117    * <p>
118    * The configuration option
119    * {@link org.apache.kafka.common.config.SaslConfigs#SASL_OAUTHBEARER_TOKEN_ENDPOINT_URL}
120    * is also required in order for the client to contact the OAuth/OIDC provider. For example:
121    *
122    * <code>
123    * sasl.oauthbearer.token.endpoint.url=https://example.com/oauth2/v1/token
124    * </code>
125    *
126    * Please see the OAuth/OIDC providers documentation for the token endpoint URL.
127    * </p>
128    *
129    * <p>
130    * The following is a list of all the configuration options that are available for the login
131    * callback handler:
132    *
133    * <ul>
134    *   <li>{@link org.apache.kafka.common.config.SaslConfigs#SASL_LOGIN_CALLBACK_HANDLER_CLASS}</li>
135    *   <li>{@link org.apache.kafka.common.config.SaslConfigs#SASL_LOGIN_CONNECT_TIMEOUT_MS}</li>
136    *   <li>{@link org.apache.kafka.common.config.SaslConfigs#SASL_LOGIN_READ_TIMEOUT_MS}</li>
137    *   <li>{@link org.apache.kafka.common.config.SaslConfigs#SASL_LOGIN_RETRY_BACKOFF_MS}</li>
138    *   <li>{@link org.apache.kafka.common.config.SaslConfigs#SASL_LOGIN_RETRY_BACKOFF_MAX_MS}</li>
139    *   <li>{@link org.apache.kafka.common.config.SaslConfigs#SASL_JAAS_CONFIG}</li>
140    *   <li>{@link org.apache.kafka.common.config.SaslConfigs#SASL_OAUTHBEARER_TOKEN_ENDPOINT_URL}</li>
141    *   <li>{@link org.apache.kafka.common.config.SaslConfigs#SASL_OAUTHBEARER_SCOPE_CLAIM_NAME}</li>
142    *   <li>{@link org.apache.kafka.common.config.SaslConfigs#SASL_OAUTHBEARER_SUB_CLAIM_NAME}</li>
143    * </ul>
144    * </p>
145    */
146   
147   public class OAuthBearerLoginCallbackHandler implements AuthenticateCallbackHandler {
148   
149       private static final Logger log = LoggerFactory.getLogger(OAuthBearerLoginCallbackHandler.class);
150   
151       public static final String CLIENT_ID_CONFIG = "clientId";
152       public static final String CLIENT_SECRET_CONFIG = "clientSecret";
153       public static final String SCOPE_CONFIG = "scope";
154   
155       public static final String CLIENT_ID_DOC = "The OAuth/OIDC identity provider-issued " +
156           "client ID to uniquely identify the service account to use for authentication for " +
157           "this client. The value must be paired with a corresponding " + CLIENT_SECRET_CONFIG + " " +
158           "value and is provided to the OAuth provider using the OAuth " +
159           "clientcredentials grant type.";
160   
161       public static final String CLIENT_SECRET_DOC = "The OAuth/OIDC identity provider-issued " +
162           "client secret serves a similar function as a password to the " + CLIENT_ID_CONFIG + " " +
163           "account and identifies the service account to use for authentication for " +
164           "this client. The value must be paired with a corresponding " + CLIENT_ID_CONFIG + " " +
165           "value and is provided to the OAuth provider using the OAuth " +
166           "clientcredentials grant type.";
167   
168       public static final String SCOPE_DOC = "The (optional) HTTP/HTTPS login request to the " +
169           "token endpoint (" + SASL_OAUTHBEARER_TOKEN_ENDPOINT_URL + ") may need to specify an " +
170           "OAuth \"scope\". If so, the " + SCOPE_CONFIG + " is used to provide the value to " +
171           "include with the login request.";
172   
173       private static final String EXTENSION_PREFIX = "extension_";
174   
175       private Map<String, Object> moduleOptions;
176   
177       private AccessTokenRetriever accessTokenRetriever;
178   
179       private AccessTokenValidator accessTokenValidator;
180   
181       private boolean isInitialized = false;
182   
183       @Override
184       public void configure(Map<String, ?> configs, String saslMechanism, List<AppConfigurationEntry> jaasConfigEntries) {
185           moduleOptions = JaasOptionsUtils.getOptions(saslMechanism, jaasConfigEntries);
186           AccessTokenRetriever accessTokenRetriever = AccessTokenRetrieverFactory.create(configs, saslMechanism, moduleOptions);
187           AccessTokenValidator accessTokenValidator = AccessTokenValidatorFactory.create(configs, saslMechanism);
188           init(accessTokenRetriever, accessTokenValidator);
189       }
190   
191       /*
192        * Package-visible for testing.
193        */
194   
195       void init(AccessTokenRetriever accessTokenRetriever, AccessTokenValidator accessTokenValidator) {
196           this.accessTokenRetriever = accessTokenRetriever;
197           this.accessTokenValidator = accessTokenValidator;
198   
199           try {
200               this.accessTokenRetriever.init();
201           } catch (IOException e) {
202               throw new KafkaException("The OAuth login configuration encountered an error when initializing the AccessTokenRetriever", e);
203           }
204   
205           isInitialized = true;
206       }
207   
208       /*
209        * Package-visible for testing.
210        */
211   
212       AccessTokenRetriever getAccessTokenRetriever() {
213           return accessTokenRetriever;
214       }
215   
216       @Override
217       public void close() {
218           if (accessTokenRetriever != null) {
219               try {
220                   this.accessTokenRetriever.close();
221               } catch (IOException e) {
222                   log.warn("The OAuth login configuration encountered an error when closing the AccessTokenRetriever", e);
223               }
224           }
225       }
226   
227       @Override
228       public void handle(Callback[] callbacks) throws IOException, UnsupportedCallbackException {
229           checkInitialized();
230   
231           for (Callback callback : callbacks) {
232               if (callback instanceof OAuthBearerTokenCallback) {
233                   handleTokenCallback((OAuthBearerTokenCallback) callback);
234               } else if (callback instanceof SaslExtensionsCallback) {
235                   handleExtensionsCallback((SaslExtensionsCallback) callback);
236               } else {
237                   throw new UnsupportedCallbackException(callback);
238               }
239           }
240       }
241   
242       private void handleTokenCallback(OAuthBearerTokenCallback callback) throws IOException {
243           checkInitialized();
244           String accessToken = accessTokenRetriever.retrieve();
245   
246           try {
247               OAuthBearerToken token = accessTokenValidator.validate(accessToken);
248               callback.token(token);
249           } catch (ValidateException e) {
250               log.warn(e.getMessage(), e);
251               callback.error("invalid_token", e.getMessage(), null);
252           }
253       }
254   
255       private void handleExtensionsCallback(SaslExtensionsCallback callback) {
256           checkInitialized();
257   
258           Map<String, String> extensions = new HashMap<>();
259   
260           for (Map.Entry<String, Object> configEntry : this.moduleOptions.entrySet()) {
261               String key = configEntry.getKey();
262   
263               if (!key.startsWith(EXTENSION_PREFIX))
264                   continue;
265   
266               Object valueRaw = configEntry.getValue();
267               String value;
268   
269               if (valueRaw instanceof String)
270                   value = (String) valueRaw;
271               else
272                   value = String.valueOf(valueRaw);
273   
274               extensions.put(key.substring(EXTENSION_PREFIX.length()), value);
275           }
276   
277           SaslExtensions saslExtensions = new SaslExtensions(extensions);
278   
279           try {
280               OAuthBearerClientInitialResponse.validateExtensions(saslExtensions);
281           } catch (SaslException e) {
282               throw new ConfigException(e.getMessage());
283           }
284   
285           callback.extensions(saslExtensions);
286       }
287   
288       private void checkInitialized() {
289           if (!isInitialized)
290               throw new IllegalStateException(String.format("To use %s, first call the configure or init method", getClass().getSimpleName()));
291       }
