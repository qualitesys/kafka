17   package org.apache.kafka.common;
18   
19   import java.util.Map;
20   import java.util.Objects;
21   
22   /**
23    * The <code>MetricName</code> class encapsulates a metric's name, logical group and its related attributes. It should be constructed using metrics.MetricName(...).
24    * <p>
25    * This class captures the following parameters:
26    * <pre>
27    *  <b>name</b> The name of the metric
28    *  <b>group</b> logical group name of the metrics to which this metric belongs.
29    *  <b>description</b> A human-readable description to include in the metric. This is optional.
30    *  <b>tags</b> additional key/value attributes of the metric. This is optional.
31    * </pre>
32    * group, tags parameters can be used to create unique metric names while reporting in JMX or any custom reporting.
33    * <p>
34    * Ex: standard JMX MBean can be constructed like <b>domainName:type=group,key1=val1,key2=val2</b>
35    * <p>
36    *
37    * Usage looks something like this:
38    * <pre>{@code
39    * // set up metrics:
40    *
41    * Map<String, String> metricTags = new LinkedHashMap<String, String>();
42    * metricTags.put("client-id", "producer-1");
43    * metricTags.put("topic", "topic");
44    *
45    * MetricConfig metricConfig = new MetricConfig().tags(metricTags);
46    * Metrics metrics = new Metrics(metricConfig); // this is the global repository of metrics and sensors
47    *
48    * Sensor sensor = metrics.sensor("message-sizes");
49    *
50    * MetricName metricName = metrics.metricName("message-size-avg", "producer-metrics", "average message size");
51    * sensor.add(metricName, new Avg());
52    *
53    * metricName = metrics.metricName("message-size-max", "producer-metrics");
54    * sensor.add(metricName, new Max());
55    *
56    * metricName = metrics.metricName("message-size-min", "producer-metrics", "message minimum size", "client-id", "my-client", "topic", "my-topic");
57    * sensor.add(metricName, new Min());
58    *
59    * // as messages are sent we record the sizes
60    * sensor.record(messageSize);
61    * }</pre>
62    */
63   public final class MetricName {
64   
65       private final String name;
66       private final String group;
67       private final String description;
68       private Map<String, String> tags;
69       private int hash = 0;
70   
71       /**
72        * Please create MetricName by method {@link org.apache.kafka.common.metrics.Metrics#metricName(String, String, String, Map)}
73        *
74        * @param name        The name of the metric
75        * @param group       logical group name of the metrics to which this metric belongs
76        * @param description A human-readable description to include in the metric
77        * @param tags        additional key/value attributes of the metric
78        */
79       public MetricName(String name, String group, String description, Map<String, String> tags) {
80           this.name = Objects.requireNonNull(name);
81           this.group = Objects.requireNonNull(group);
82           this.description = Objects.requireNonNull(description);
83           this.tags = Objects.requireNonNull(tags);
84       }
85   
86       public String name() {
87           return this.name;
88       }
89   
90       public String group() {
91           return this.group;
92       }
93   
94       public Map<String, String> tags() {
95           return this.tags;
96       }
97   
98       public String description() {
99           return this.description;
100       }
101   
102       @Override
103       public int hashCode() {
104           if (hash != 0)
105               return hash;
106           final int prime = 31;
107           int result = 1;
108           result = prime * result + group.hashCode();
109           result = prime * result + name.hashCode();
110           result = prime * result + tags.hashCode();
111           this.hash = result;
112           return result;
113       }
114   
115       @Override
116       public boolean equals(Object obj) {
117           if (this == obj)
118               return true;
119           if (obj == null)
120               return false;
121           if (getClass() != obj.getClass())
122               return false;
123           MetricName other = (MetricName) obj;
124           return group.equals(other.group) && name.equals(other.name) && tags.equals(other.tags);
125       }
126   
127       @Override
128       public String toString() {
129           return "MetricName [name=" + name + ", group=" + group + ", description="
130                   + description + ", tags=" + tags + "]";
131       }
132   }
