17   package org.apache.kafka.common.network;
18   
19   import org.apache.kafka.common.errors.AuthenticationException;
20   import org.apache.kafka.common.security.auth.KafkaPrincipal;
21   import org.apache.kafka.common.security.auth.KafkaPrincipalSerde;
22   
23   import java.io.Closeable;
24   import java.io.IOException;
25   import java.util.Optional;
26   
27   /**
28    * Authentication for Channel
29    */
30   public interface Authenticator extends Closeable {
31       /**
32        * Implements any authentication mechanism. Use transportLayer to read or write tokens.
33        * For security protocols PLAINTEXT and SSL, this is a no-op since no further authentication
34        * needs to be done. For SASL_PLAINTEXT and SASL_SSL, this performs the SASL authentication.
35        *
36        * @throws AuthenticationException if authentication fails due to invalid credentials or
37        *      other security configuration errors
38        * @throws IOException if read/write fails due to an I/O error
39        */
40       void authenticate() throws AuthenticationException, IOException;
41   
42       /**
43        * Perform any processing related to authentication failure. This is invoked when the channel is about to be closed
44        * because of an {@link AuthenticationException} thrown from a prior {@link #authenticate()} call.
45        * @throws IOException if read/write fails due to an I/O error
46        */
47       default void handleAuthenticationFailure() throws IOException {
48       }
49   
50       /**
51        * Returns Principal using PrincipalBuilder
52        */
53       KafkaPrincipal principal();
54   
55       /**
56        * Returns the serializer/deserializer interface for principal
57        */
58       Optional<KafkaPrincipalSerde> principalSerde();
59   
60       /**
61        * returns true if authentication is complete otherwise returns false;
62        */
63       boolean complete();
64   
65       /**
66        * Begins re-authentication. Uses transportLayer to read or write tokens as is
67        * done for {@link #authenticate()}. For security protocols PLAINTEXT and SSL,
68        * this is a no-op since re-authentication does not apply/is not supported,
69        * respectively. For SASL_PLAINTEXT and SASL_SSL, this performs a SASL
70        * authentication. Any in-flight responses from prior requests can/will be read
71        * and collected for later processing as required. There must not be partially
72        * written requests; any request queued for writing (for which zero bytes have
73        * been written) remains queued until after re-authentication succeeds.
74        * 
75        * @param reauthenticationContext
76        *            the context in which this re-authentication is occurring. This
77        *            instance is responsible for closing the previous Authenticator
78        *            returned by
79        *            {@link ReauthenticationContext#previousAuthenticator()}.
80        * @throws AuthenticationException
81        *             if authentication fails due to invalid credentials or other
82        *             security configuration errors
83        * @throws IOException
84        *             if read/write fails due to an I/O error
85        */
86       default void reauthenticate(ReauthenticationContext reauthenticationContext) throws IOException {
87           // empty
88       }
89   
90       /**
91        * Return the session expiration time, if any, otherwise null. The value is in
92        * nanoseconds as per {@code System.nanoTime()} and is therefore only useful
93        * when compared to such a value -- it's absolute value is meaningless. This
94        * value may be non-null only on the server-side. It represents the time after
95        * which, in the absence of re-authentication, the broker will close the session
96        * if it receives a request unrelated to authentication. We store nanoseconds
97        * here to avoid having to invoke the more expensive {@code milliseconds()} call
98        * on the broker for every request
99        * 
100        * @return the session expiration time, if any, otherwise null
101        */
102       default Long serverSessionExpirationTimeNanos() {
103           return null;
104       }
105   
106       /**
107        * Return the time on or after which a client should re-authenticate this
108        * session, if any, otherwise null. The value is in nanoseconds as per
109        * {@code System.nanoTime()} and is therefore only useful when compared to such
110        * a value -- it's absolute value is meaningless. This value may be non-null
111        * only on the client-side. It will be a random time between 85% and 95% of the
112        * full session lifetime to account for latency between client and server and to
113        * avoid re-authentication storms that could be caused by many sessions
114        * re-authenticating simultaneously.
115        * 
116        * @return the time on or after which a client should re-authenticate this
117        *         session, if any, otherwise null
118        */
119       default Long clientSessionReauthenticationTimeNanos() {
120           return null;
121       }
122   
123       /**
124        * Return the number of milliseconds that elapsed while re-authenticating this
125        * session from the perspective of this instance, if applicable, otherwise null.
126        * The server-side perspective will yield a lower value than the client-side
127        * perspective of the same re-authentication because the client-side observes an
128        * additional network round-trip.
129        * 
130        * @return the number of milliseconds that elapsed while re-authenticating this
131        *         session from the perspective of this instance, if applicable,
132        *         otherwise null
133        */
134       default Long reauthenticationLatencyMs() {
135           return null;
136       }
137   
138       /**
139        * Return the next (always non-null but possibly empty) client-side
140        * {@link NetworkReceive} response that arrived during re-authentication that
141        * is unrelated to re-authentication, if any. These correspond to requests sent
142        * prior to the beginning of re-authentication; the requests were made when the
143        * channel was successfully authenticated, and the responses arrived during the
144        * re-authentication process. The response returned is removed from the authenticator's
145        * queue. Responses of requests sent after completion of re-authentication are
146        * processed only when the authenticator response queue is empty.
147        * 
148        * @return the (always non-null but possibly empty) client-side
149        *         {@link NetworkReceive} response that arrived during
150        *         re-authentication that is unrelated to re-authentication, if any
151        */
152       default Optional<NetworkReceive> pollResponseReceivedDuringReauthentication() {
153           return Optional.empty();
154       }
155       
156       /**
157        * Return true if this is a server-side authenticator and the connected client
158        * has indicated that it supports re-authentication, otherwise false
159        * 
160        * @return true if this is a server-side authenticator and the connected client
161        *         has indicated that it supports re-authentication, otherwise false
162        */
163       default boolean connectedClientSupportsReauthentication() {
164           return false;
165       }
166   }
