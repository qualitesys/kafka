41   public class ListDeserializer<Inner> implements Deserializer<List<Inner>> {
42   
43       final Logger log = LoggerFactory.getLogger(ListDeserializer.class);
44   
45       private static final Map<Class<? extends Deserializer<?>>, Integer> FIXED_LENGTH_DESERIALIZERS = mkMap(
46           mkEntry(ShortDeserializer.class, Short.BYTES),
47           mkEntry(IntegerDeserializer.class, Integer.BYTES),
48           mkEntry(FloatDeserializer.class, Float.BYTES),
49           mkEntry(LongDeserializer.class, Long.BYTES),
50           mkEntry(DoubleDeserializer.class, Double.BYTES),
51           mkEntry(UUIDDeserializer.class, 36)
52       );
53   
54       private Deserializer<Inner> inner;
55       private Class<?> listClass;
56       private Integer primitiveSize;
57   
58       public ListDeserializer() {}
59   
60       public <L extends List<Inner>> ListDeserializer(Class<L> listClass, Deserializer<Inner> inner) {
61           if (listClass == null || inner == null) {
62               log.error("Could not construct ListDeserializer as not all required parameters were present -- listClass: {}, inner: {}", listClass, inner);
63               throw new IllegalArgumentException("ListDeserializer requires both \"listClass\" and \"innerDeserializer\" parameters to be provided during initialization");
64           }
65           this.listClass = listClass;
66           this.inner = inner;
67           this.primitiveSize = FIXED_LENGTH_DESERIALIZERS.get(inner.getClass());
68       }
69   
70       public Deserializer<Inner> innerDeserializer() {
71           return inner;
72       }
73   
74       @Override
75       public void configure(Map<String, ?> configs, boolean isKey) {
76           if (listClass != null || inner != null) {
77               log.error("Could not configure ListDeserializer as some parameters were already set -- listClass: {}, inner: {}", listClass, inner);
78               throw new ConfigException("List deserializer was already initialized using a non-default constructor");
79           }
80           configureListClass(configs, isKey);
81           configureInnerSerde(configs, isKey);
82       }
83   
84       private void configureListClass(Map<String, ?> configs, boolean isKey) {
85           String listTypePropertyName = isKey ? CommonClientConfigs.DEFAULT_LIST_KEY_SERDE_TYPE_CLASS : CommonClientConfigs.DEFAULT_LIST_VALUE_SERDE_TYPE_CLASS;
86           final Object listClassOrName = configs.get(listTypePropertyName);
87           if (listClassOrName == null) {
88               throw new ConfigException("Not able to determine the list class because it was neither passed via the constructor nor set in the config.");
89           }
90           try {
91               if (listClassOrName instanceof String) {
92                   listClass = Utils.loadClass((String) listClassOrName, Object.class);
93               } else if (listClassOrName instanceof Class) {
94                   listClass = (Class<?>) listClassOrName;
95               } else {
96                   throw new KafkaException("Could not determine the list class instance using \"" + listTypePropertyName + "\" property.");
97               }
98           } catch (final ClassNotFoundException e) {
99               throw new ConfigException(listTypePropertyName, listClassOrName, "Deserializer's list class \"" + listClassOrName + "\" could not be found.");
100           }
101       }
102   
103       @SuppressWarnings("unchecked")
104       private void configureInnerSerde(Map<String, ?> configs, boolean isKey) {
105           String innerSerdePropertyName = isKey ? CommonClientConfigs.DEFAULT_LIST_KEY_SERDE_INNER_CLASS : CommonClientConfigs.DEFAULT_LIST_VALUE_SERDE_INNER_CLASS;
106           final Object innerSerdeClassOrName = configs.get(innerSerdePropertyName);
107           if (innerSerdeClassOrName == null) {
108               throw new ConfigException("Not able to determine the inner serde class because it was neither passed via the constructor nor set in the config.");
109           }
110           try {
111               if (innerSerdeClassOrName instanceof String) {
112                   inner = Utils.newInstance((String) innerSerdeClassOrName, Serde.class).deserializer();
113               } else if (innerSerdeClassOrName instanceof Class) {
114                   inner = (Deserializer<Inner>) ((Serde) Utils.newInstance((Class) innerSerdeClassOrName)).deserializer();
115               } else {
116                   throw new KafkaException("Could not determine the inner serde class instance using \"" + innerSerdePropertyName + "\" property.");
117               }
118               inner.configure(configs, isKey);
119               primitiveSize = FIXED_LENGTH_DESERIALIZERS.get(inner.getClass());
120           } catch (final ClassNotFoundException e) {
121               throw new ConfigException(innerSerdePropertyName, innerSerdeClassOrName, "Deserializer's inner serde class \"" + innerSerdeClassOrName + "\" could not be found.");
122           }
123       }
124   
125       @SuppressWarnings("unchecked")
126       private List<Inner> createListInstance(int listSize) {
127           try {
128               Constructor<List<Inner>> listConstructor;
129               try {
130                   listConstructor = (Constructor<List<Inner>>) listClass.getConstructor(Integer.TYPE);
131                   return listConstructor.newInstance(listSize);
132               } catch (NoSuchMethodException e) {
133                   listConstructor = (Constructor<List<Inner>>) listClass.getConstructor();
134                   return listConstructor.newInstance();
135               }
136           } catch (InstantiationException | IllegalAccessException | NoSuchMethodException |
137                   IllegalArgumentException | InvocationTargetException e) {
138               log.error("Failed to construct list due to ", e);
139               throw new KafkaException("Could not construct a list instance of \"" + listClass.getCanonicalName() + "\"", e);
140           }
141       }

143       private SerializationStrategy parseSerializationStrategyFlag(final int serializationStrategyFlag) throws IOException {
144           if (serializationStrategyFlag < 0 || serializationStrategyFlag >= SerializationStrategy.VALUES.length) {
145               throw new SerializationException("Invalid serialization strategy flag value");
146           }
147           return SerializationStrategy.VALUES[serializationStrategyFlag];
148       }

150       private List<Integer> deserializeNullIndexList(final DataInputStream dis) throws IOException {
151           int nullIndexListSize = dis.readInt();
152           List<Integer> nullIndexList = new ArrayList<>(nullIndexListSize);
153           while (nullIndexListSize != 0) {
154               nullIndexList.add(dis.readInt());
155               nullIndexListSize--;
156           }
157           return nullIndexList;
158       }

160       @Override
161       public List<Inner> deserialize(String topic, byte[] data) {
162           if (data == null) {
163               return null;
164           }
165           try (final DataInputStream dis = new DataInputStream(new ByteArrayInputStream(data))) {
166               SerializationStrategy serStrategy = parseSerializationStrategyFlag(dis.readByte());
167               List<Integer> nullIndexList = null;
168               if (serStrategy == SerializationStrategy.CONSTANT_SIZE) {
169                   // In CONSTANT_SIZE strategy, indexes of null entries are decoded from a null index list
170                   nullIndexList = deserializeNullIndexList(dis);
171               }
172               final int size = dis.readInt();
173               List<Inner> deserializedList = createListInstance(size);
174               for (int i = 0; i < size; i++) {
175                   int entrySize = serStrategy == SerializationStrategy.CONSTANT_SIZE ? primitiveSize : dis.readInt();
176                   if (entrySize == ListSerde.NULL_ENTRY_VALUE || (nullIndexList != null && nullIndexList.contains(i))) {
177                       deserializedList.add(null);
178                       continue;
179                   }
180                   byte[] payload = new byte[entrySize];
181                   if (dis.read(payload) == -1) {
182                       log.error("Ran out of bytes in serialized list");
183                       log.trace("Deserialized list so far: {}", deserializedList); // avoid logging actual data above TRACE level since it may contain sensitive information
184                       throw new SerializationException("End of the stream was reached prematurely");
185                   }
186                   deserializedList.add(inner.deserialize(topic, payload));
187               }
188               return deserializedList;
189           } catch (IOException e) {
190               throw new KafkaException("Unable to deserialize into a List", e);
191           }
192       }

194       @Override
195       public void close() {
196           if (inner != null) {
197               inner.close();
198           }
199       }
