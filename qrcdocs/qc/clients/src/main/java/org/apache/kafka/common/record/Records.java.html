25   /**
26    * Interface for accessing the records contained in a log. The log itself is represented as a sequence of record
27    * batches (see {@link RecordBatch}).
28    *
29    * For magic versions 1 and below, each batch consists of an 8 byte offset, a 4 byte record size, and a "shallow"
30    * {@link Record record}. If the batch is not compressed, then each batch will have only the shallow record contained
31    * inside it. If it is compressed, the batch contains "deep" records, which are packed into the value field of the
32    * shallow record. To iterate over the shallow batches, use {@link Records#batches()}; for the deep records, use
33    * {@link Records#records()}. Note that the deep iterator handles both compressed and non-compressed batches:
34    * if the batch is not compressed, the shallow record is returned; otherwise, the shallow batch is decompressed and the
35    * deep records are returned.
36    *
37    * For magic version 2, every batch contains 1 or more log record, regardless of compression. You can iterate
38    * over the batches directly using {@link Records#batches()}. Records can be iterated either directly from an individual
39    * batch or through {@link Records#records()}. Just as in previous versions, iterating over the records typically involves
40    * decompression and should therefore be used with caution.
41    *
42    * See {@link MemoryRecords} for the in-memory representation and {@link FileRecords} for the on-disk representation.
43    */
44   public interface Records extends TransferableRecords {
45       int OFFSET_OFFSET = 0;
46       int OFFSET_LENGTH = 8;
47       int SIZE_OFFSET = OFFSET_OFFSET + OFFSET_LENGTH;
48       int SIZE_LENGTH = 4;
49       int LOG_OVERHEAD = SIZE_OFFSET + SIZE_LENGTH;
50   
51       // the magic offset is at the same offset for all current message formats, but the 4 bytes
52       // between the size and the magic is dependent on the version.
53       int MAGIC_OFFSET = 16;
54       int MAGIC_LENGTH = 1;
55       int HEADER_SIZE_UP_TO_MAGIC = MAGIC_OFFSET + MAGIC_LENGTH;
56   
57       /**
58        * Get the record batches. Note that the signature allows subclasses
59        * to return a more specific batch type. This enables optimizations such as in-place offset
60        * assignment (see for example {@link DefaultRecordBatch}), and partial reading of
61        * record data (see {@link FileLogInputStream.FileChannelRecordBatch#magic()}.
62        * @return An iterator over the record batches of the log
63        */
64       Iterable<? extends RecordBatch> batches();
65   
66       /**
67        * Get an iterator over the record batches. This is similar to {@link #batches()} but returns an {@link AbstractIterator}
68        * instead of {@link Iterator}, so that clients can use methods like {@link AbstractIterator#peek() peek}.
69        * @return An iterator over the record batches of the log
70        */
71       AbstractIterator<? extends RecordBatch> batchIterator();
72   
73       /**
74        * Check whether all batches in this buffer have a certain magic value.
75        * @param magic The magic value to check
76        * @return true if all record batches have a matching magic value, false otherwise
77        */
78       boolean hasMatchingMagic(byte magic);
79   
80       /**
81        * Convert all batches in this buffer to the format passed as a parameter. Note that this requires
82        * deep iteration since all of the deep records must also be converted to the desired format.
83        * @param toMagic The magic value to convert to
84        * @param firstOffset The starting offset for returned records. This only impacts some cases. See
85        *                    {@link RecordsUtil#downConvert(Iterable, byte, long, Time)} for an explanation.
86        * @param time instance used for reporting stats
87        * @return A ConvertedRecords instance which may or may not contain the same instance in its records field.
88        */
89       ConvertedRecords<? extends Records> downConvert(byte toMagic, long firstOffset, Time time);
90   
91       /**
92        * Get an iterator over the records in this log. Note that this generally requires decompression,
93        * and should therefore be used with care.
94        * @return The record iterator
95        */
96       Iterable<Record> records();
97   }
