61       private static class ExitRefresherThreadDueToIllegalStateException extends Exception {
62           private static final long serialVersionUID = -6108495378411920380L;

69       private class Refresher implements Runnable {
70           @Override
71           public void run() {
72               log.info("[Principal={}]: Expiring credential re-login thread started.", principalLogText());
73               while (true) {
74                   /*
75                    * Refresh thread's main loop. Each expiring credential lives for one iteration
76                    * of the loop. Thread will exit if the loop exits from here.
77                    */
78                   long nowMs = currentMs();
79                   Long nextRefreshMs = refreshMs(nowMs);
80                   if (nextRefreshMs == null) {
81                       loginContextFactory.refresherThreadDone();
82                       return;
83                   }
84                   // safety check motivated by KAFKA-7945,
85                   // should generally never happen except due to a bug
86                   if (nextRefreshMs.longValue() < nowMs) {
87                       log.warn("[Principal={}]: Expiring credential re-login sleep time was calculated to be in the past! Will explicitly adjust. ({})", principalLogText(),
88                               new Date(nextRefreshMs));
89                       nextRefreshMs = Long.valueOf(nowMs + 10 * 1000); // refresh in 10 seconds
90                   }
91                   log.info("[Principal={}]: Expiring credential re-login sleeping until: {}", principalLogText(),
92                           new Date(nextRefreshMs));
93                   time.sleep(nextRefreshMs - nowMs);
94                   if (Thread.currentThread().isInterrupted()) {
95                       log.info("[Principal={}]: Expiring credential re-login thread has been interrupted and will exit.",
96                               principalLogText());
97                       loginContextFactory.refresherThreadDone();
98                       return;
99                   }
100                   while (true) {
101                       /*
102                        * Perform a re-login over and over again with some intervening delay
103                        * unless/until either the refresh succeeds or we are interrupted.
104                        */
105                       try {
106                           reLogin();
107                           break; // success
108                       } catch (ExitRefresherThreadDueToIllegalStateException e) {
109                           log.error(e.getMessage(), e);
110                           loginContextFactory.refresherThreadDone();
111                           return;
112                       } catch (LoginException loginException) {
113                           log.warn(String.format(
114                                   "[Principal=%s]: LoginException during login retry; will sleep %d seconds before trying again.",
115                                   principalLogText(), DELAY_SECONDS_BEFORE_NEXT_RETRY_WHEN_RELOGIN_FAILS),
116                                   loginException);
117                           // Sleep and allow loop to run/try again unless interrupted
118                           time.sleep(DELAY_SECONDS_BEFORE_NEXT_RETRY_WHEN_RELOGIN_FAILS * 1000);
119                           if (Thread.currentThread().isInterrupted()) {
120                               log.error(
121                                       "[Principal={}]: Interrupted while trying to perform a subsequent expiring credential re-login after one or more initial re-login failures: re-login thread exiting now: {}",
122                                       principalLogText(), String.valueOf(loginException.getMessage()));
123                               loginContextFactory.refresherThreadDone();
124                               return;
125                           }
126                       }
127                   }
128               }
129           }
130       }

193       /**
194        * Performs login for each login module specified for the login context of this
195        * instance and starts the thread used to periodically re-login.
196        * <p>
197        * The synchronized keyword is not necessary because an implementation of
198        * {@link Login} will delegate to this code (e.g. OAuthBearerRefreshingLogin},
199        * and the {@code login()} method on the delegating class will itself be
200        * synchronized if necessary.
201        */
202       public LoginContext login() throws LoginException {
203           LoginContext tmpLoginContext = loginContextFactory.createLoginContext(this);
204           tmpLoginContext.login();
205           log.info("Successfully logged in.");
206           loginContext = tmpLoginContext;
207           subject = loginContext.getSubject();
208           expiringCredential = expiringCredential();
209           hasExpiringCredential = expiringCredential != null;
210           if (!hasExpiringCredential) {
211               // do not bother with re-logins.
212               log.debug("No Expiring Credential");
213               principalName = null;
214               refresherThread = null;
215               return loginContext;
216           }
217   
218           principalName = expiringCredential.principalName();
219   
220           // Check for a clock skew problem
221           long expireTimeMs = expiringCredential.expireTimeMs();
222           long nowMs = currentMs();
223           if (nowMs > expireTimeMs) {
224               log.error(
225                       "[Principal={}]: Current clock: {} is later than expiry {}. This may indicate a clock skew problem."
226                               + " Check that this host's and remote host's clocks are in sync. Not starting refresh thread."
227                               + " This process is likely unable to authenticate SASL connections (for example, it is unlikely"
228                               + " to be able to authenticate a connection with a Kafka Broker).",
229                       principalLogText(), new Date(nowMs), new Date(expireTimeMs));
230               return loginContext;
231           }
232   
233           if (log.isDebugEnabled())
234               log.debug("[Principal={}]: It is an expiring credential", principalLogText());
235   
236           /*
237            * Re-login periodically. How often is determined by the expiration date of the
238            * credential and refresh-related configuration values.
239            */
240           refresherThread = KafkaThread.daemon(String.format("kafka-expiring-relogin-thread-%s", principalName),
241                   new Refresher());
242           refresherThread.start();
243           loginContextFactory.refresherThreadStarted();
244           return loginContext;
245       }

247       public void close() {
248           if (refresherThread != null && refresherThread.isAlive()) {
249               refresherThread.interrupt();
250               try {
251                   refresherThread.join();
252               } catch (InterruptedException e) {
253                   log.warn("[Principal={}]: Interrupted while waiting for re-login thread to shutdown.",
254                           principalLogText(), e);
255                   Thread.currentThread().interrupt();
256               }
257           }

260       public abstract ExpiringCredential expiringCredential();
261   
262       /**
263        * Determine when to sleep until before performing a refresh
264        * 
265        * @param relativeToMs
266        *            the point (in terms of number of milliseconds since the epoch) at
267        *            which to perform the calculation
268        * @return null if no refresh should occur, otherwise the time to sleep until
269        *         (in terms of the number of milliseconds since the epoch) before
270        *         performing a refresh
271        */
272       private Long refreshMs(long relativeToMs) {
273           if (expiringCredential == null) {
274               /*
275                * Re-login failed because our login() invocation did not generate a credential
276                * but also did not generate an exception. Try logging in again after some delay
277                * (it seems likely to be a bug, but it doesn't hurt to keep trying to refresh).
278                */
279               long retvalNextRefreshMs = relativeToMs + DELAY_SECONDS_BEFORE_NEXT_RETRY_WHEN_RELOGIN_FAILS * 1000L;
280               log.warn("[Principal={}]: No Expiring credential found: will try again at {}", principalLogText(),
281                       new Date(retvalNextRefreshMs));
282               return retvalNextRefreshMs;
283           }
284           long expireTimeMs = expiringCredential.expireTimeMs();
285           if (relativeToMs > expireTimeMs) {
286               boolean logoutRequiredBeforeLoggingBackIn = isLogoutRequiredBeforeLoggingBackIn();
287               if (logoutRequiredBeforeLoggingBackIn) {
288                   log.error(
289                           "[Principal={}]: Current clock: {} is later than expiry {}. This may indicate a clock skew problem."
290                                   + " Check that this host's and remote host's clocks are in sync. Exiting refresh thread.",
291                           principalLogText(), new Date(relativeToMs), new Date(expireTimeMs));
292                   return null;
293               } else {
294                   /*
295                    * Since the current soon-to-expire credential isn't logged out until we have a
296                    * new credential with a refreshed lifetime, it is possible that the current
297                    * credential could expire if the re-login continually fails over and over again
298                    * making us unable to get the new credential. Therefore keep trying rather than
299                    * exiting.
300                    */
301                   long retvalNextRefreshMs = relativeToMs + DELAY_SECONDS_BEFORE_NEXT_RETRY_WHEN_RELOGIN_FAILS * 1000L;
302                   log.warn("[Principal={}]: Expiring credential already expired at {}: will try to refresh again at {}",
303                           principalLogText(), new Date(expireTimeMs), new Date(retvalNextRefreshMs));
304                   return retvalNextRefreshMs;
305               }
306           }
307           Long absoluteLastRefreshTimeMs = expiringCredential.absoluteLastRefreshTimeMs();
308           if (absoluteLastRefreshTimeMs != null && absoluteLastRefreshTimeMs.longValue() < expireTimeMs) {
309               log.warn("[Principal={}]: Expiring credential refresh thread exiting because the"
310                       + " expiring credential's current expiration time ({}) exceeds the latest possible refresh time ({})."
311                       + " This process will not be able to authenticate new SASL connections after that"
312                       + " time (for example, it will not be able to authenticate a new connection with a Kafka Broker).",
313                       principalLogText(), new Date(expireTimeMs), new Date(absoluteLastRefreshTimeMs.longValue()));
314               return null;
315           }
316           Long optionalStartTime = expiringCredential.startTimeMs();
317           long startMs = optionalStartTime != null ? optionalStartTime.longValue() : relativeToMs;
318           log.info("[Principal={}]: Expiring credential valid from {} to {}", expiringCredential.principalName(),
319                   new java.util.Date(startMs), new java.util.Date(expireTimeMs));
320   
321           double pct = expiringCredentialRefreshConfig.loginRefreshWindowFactor()
322                   + (expiringCredentialRefreshConfig.loginRefreshWindowJitter() * RNG.nextDouble());
323           /*
324            * Ignore buffer times if the credential's remaining lifetime is less than their
325            * sum.
326            */
327           long refreshMinPeriodSeconds = expiringCredentialRefreshConfig.loginRefreshMinPeriodSeconds();
328           long clientRefreshBufferSeconds = expiringCredentialRefreshConfig.loginRefreshBufferSeconds();
329           if (relativeToMs + 1000L * (refreshMinPeriodSeconds + clientRefreshBufferSeconds) > expireTimeMs) {
330               long retvalRefreshMs = relativeToMs + (long) ((expireTimeMs - relativeToMs) * pct);
331               log.warn(
332                       "[Principal={}]: Expiring credential expires at {}, so buffer times of {} and {} seconds"
333                               + " at the front and back, respectively, cannot be accommodated.  We will refresh at {}.",
334                       principalLogText(), new Date(expireTimeMs), refreshMinPeriodSeconds, clientRefreshBufferSeconds,
335                       new Date(retvalRefreshMs));
336               return retvalRefreshMs;
337           }
338           long proposedRefreshMs = startMs + (long) ((expireTimeMs - startMs) * pct);
339           // Don't let it violate the requested end buffer time
340           long beginningOfEndBufferTimeMs = expireTimeMs - clientRefreshBufferSeconds * 1000;
341           if (proposedRefreshMs > beginningOfEndBufferTimeMs) {
342               log.info(
343                       "[Principal={}]: Proposed refresh time of {} extends into the desired buffer time of {} seconds before expiration, so refresh it at the desired buffer begin point, at {}",
344                       expiringCredential.principalName(), new Date(proposedRefreshMs), clientRefreshBufferSeconds,
345                       new Date(beginningOfEndBufferTimeMs));
346               return beginningOfEndBufferTimeMs;
347           }
348           // Don't let it violate the minimum refresh period
349           long endOfMinRefreshBufferTime = relativeToMs + 1000 * refreshMinPeriodSeconds;
350           if (proposedRefreshMs < endOfMinRefreshBufferTime) {
351               log.info(
352                       "[Principal={}]: Expiring credential re-login thread time adjusted from {} to {} since the former is sooner "
353                               + "than the minimum refresh interval ({} seconds from now).",
354                       principalLogText(), new Date(proposedRefreshMs), new Date(endOfMinRefreshBufferTime),
355                       refreshMinPeriodSeconds);
356               return endOfMinRefreshBufferTime;
357           }
358           // Proposed refresh time doesn't violate any constraints
359           return proposedRefreshMs;
360       }

362       private void reLogin() throws LoginException, ExitRefresherThreadDueToIllegalStateException {
363           synchronized (mandatoryClassToSynchronizeOnPriorToRefresh) {
364               // Only perform one refresh of a particular type at a time
365               boolean logoutRequiredBeforeLoggingBackIn = isLogoutRequiredBeforeLoggingBackIn();
366               if (hasExpiringCredential && logoutRequiredBeforeLoggingBackIn) {
367                   String principalLogTextPriorToLogout = principalLogText();
368                   log.info("Initiating logout for {}", principalLogTextPriorToLogout);
369                   loginContext.logout();
370                   // Make absolutely sure we were logged out
371                   expiringCredential = expiringCredential();
372                   hasExpiringCredential = expiringCredential != null;
373                   if (hasExpiringCredential)
374                       // We can't force the removal because we don't know how to do it, so abort
375                       throw new ExitRefresherThreadDueToIllegalStateException(String.format(
376                               "Subject's private credentials still contains an instance of %s even though logout() was invoked; exiting refresh thread",
377                               expiringCredential.getClass().getName()));
378               }
379               /*
380                * Perform a login, making note of any credential that might need a logout()
381                * afterwards
382                */
383               ExpiringCredential optionalCredentialToLogout = expiringCredential;
384               LoginContext optionalLoginContextToLogout = loginContext;
385               boolean cleanLogin = false; // remember to restore the original if necessary
386               try {
387                   loginContext = loginContextFactory.createLoginContext(ExpiringCredentialRefreshingLogin.this);
388                   log.info("Initiating re-login for {}, logout() still needs to be called on a previous login = {}",
389                           principalName, optionalCredentialToLogout != null);
390                   loginContext.login();
391                   cleanLogin = true; // no need to restore the original
392                   // Perform a logout() on any original credential if necessary
393                   if (optionalCredentialToLogout != null)
394                       optionalLoginContextToLogout.logout();
395               } finally {
396                   if (!cleanLogin)
397                       // restore the original
398                       loginContext = optionalLoginContextToLogout;
399               }
400               /*
401                * Get the new credential and make sure it is not any old one that required a
402                * logout() after the login()
403                */
404               expiringCredential = expiringCredential();
405               hasExpiringCredential = expiringCredential != null;
406               if (!hasExpiringCredential) {
407                   /*
408                    * Re-login has failed because our login() invocation has not generated a
409                    * credential but has also not generated an exception. We won't exit here;
410                    * instead we will allow login retries in case we can somehow fix the issue (it
411                    * seems likely to be a bug, but it doesn't hurt to keep trying to refresh).
412                    */
413                   log.error("No Expiring Credential after a supposedly-successful re-login");
414                   principalName = null;
415               } else {
416                   if (expiringCredential == optionalCredentialToLogout)
417                       /*
418                        * The login() didn't identify a new credential; we still have the old one. We
419                        * don't know how to fix this, so abort.
420                        */
421                       throw new ExitRefresherThreadDueToIllegalStateException(String.format(
422                               "Subject's private credentials still contains the previous, soon-to-expire instance of %s even though login() followed by logout() was invoked; exiting refresh thread",
423                               expiringCredential.getClass().getName()));
424                   principalName = expiringCredential.principalName();
425                   if (log.isDebugEnabled())
426                       log.debug("[Principal={}]: It is an expiring credential after re-login as expected",
427                               principalLogText());
428               }
429           }
430       }
431   
432       private String principalLogText() {
433           return expiringCredential == null ? principalName
434                   : expiringCredential.getClass().getSimpleName() + ":" + principalName;
435       }

438           return time.milliseconds();
439       }

442           return !expiringCredentialRefreshConfig.loginRefreshReloginAllowedBeforeLogout();
443       }
