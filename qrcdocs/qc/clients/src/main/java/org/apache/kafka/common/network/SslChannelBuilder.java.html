43   public class SslChannelBuilder implements ChannelBuilder, ListenerReconfigurable {
44       private final ListenerName listenerName;
45       private final boolean isInterBrokerListener;
46       private SslFactory sslFactory;
47       private Mode mode;
48       private Map<String, ?> configs;
49       private SslPrincipalMapper sslPrincipalMapper;
50       private final Logger log;
51   
52       /**
53        * Constructs an SSL channel builder. ListenerName is provided only
54        * for server channel builder and will be null for client channel builder.
55        */
56       public SslChannelBuilder(Mode mode,
57                                ListenerName listenerName,
58                                boolean isInterBrokerListener,
59                                LogContext logContext) {
60           this.mode = mode;
61           this.listenerName = listenerName;
62           this.isInterBrokerListener = isInterBrokerListener;
63           this.log = logContext.logger(getClass());
64       }
65   
66       public void configure(Map<String, ?> configs) throws KafkaException {
67           try {
68               this.configs = configs;
69               String sslPrincipalMappingRules = (String) configs.get(BrokerSecurityConfigs.SSL_PRINCIPAL_MAPPING_RULES_CONFIG);
70               if (sslPrincipalMappingRules != null)
71                   sslPrincipalMapper = SslPrincipalMapper.fromRules(sslPrincipalMappingRules);
72               this.sslFactory = new SslFactory(mode, null, isInterBrokerListener);
73               this.sslFactory.configure(this.configs);
74           } catch (KafkaException e) {
75               throw e;
76           } catch (Exception e) {
77               throw new KafkaException(e);
78           }
79       }
80   
81       @Override
82       public Set<String> reconfigurableConfigs() {
83           return SslConfigs.RECONFIGURABLE_CONFIGS;
84       }
85   
86       @Override
87       public void validateReconfiguration(Map<String, ?> configs) {
88           sslFactory.validateReconfiguration(configs);
89       }
90   
91       @Override
92       public void reconfigure(Map<String, ?> configs) {
93           sslFactory.reconfigure(configs);
94       }
95   
96       @Override
97       public ListenerName listenerName() {
98           return listenerName;
99       }
100   
101       @Override
102       public KafkaChannel buildChannel(String id, SelectionKey key, int maxReceiveSize,
103                                        MemoryPool memoryPool, ChannelMetadataRegistry metadataRegistry) throws KafkaException {
104           try {
105               SslTransportLayer transportLayer = buildTransportLayer(sslFactory, id, key, metadataRegistry);
106               Supplier<Authenticator> authenticatorCreator = () ->
107                   new SslAuthenticator(configs, transportLayer, listenerName, sslPrincipalMapper);
108               return new KafkaChannel(id, transportLayer, authenticatorCreator, maxReceiveSize,
109                       memoryPool != null ? memoryPool : MemoryPool.NONE, metadataRegistry);
110           } catch (Exception e) {
111               log.info("Failed to create channel due to ", e);
112               throw new KafkaException(e);
113           }
114       }
115   
116       @Override
117       public void close() {
118           if (sslFactory != null) sslFactory.close();
119       }
120   
121       protected SslTransportLayer buildTransportLayer(SslFactory sslFactory, String id, SelectionKey key, ChannelMetadataRegistry metadataRegistry) throws IOException {
122           SocketChannel socketChannel = (SocketChannel) key.channel();
123           return SslTransportLayer.create(id, key, sslFactory.createSslEngine(socketChannel.socket()),
124               metadataRegistry);
125       }
126   
127       /**
128        * Note that client SSL authentication is handled in {@link SslTransportLayer}. This class is only used
129        * to transform the derived principal using a {@link KafkaPrincipalBuilder} configured by the user.
130        */
131       private static class SslAuthenticator implements Authenticator {
132           private final SslTransportLayer transportLayer;
133           private final KafkaPrincipalBuilder principalBuilder;
134           private final ListenerName listenerName;
135   
136           private SslAuthenticator(Map<String, ?> configs, SslTransportLayer transportLayer, ListenerName listenerName, SslPrincipalMapper sslPrincipalMapper) {
137               this.transportLayer = transportLayer;
138               this.principalBuilder = ChannelBuilders.createPrincipalBuilder(configs, null, sslPrincipalMapper);
139               this.listenerName = listenerName;
140           }
141           /**
142            * No-Op for plaintext authenticator
143            */
144           @Override
145           public void authenticate() {}
146   
147           /**
148            * Constructs Principal using configured principalBuilder.
149            * @return the built principal
150            */
151           @Override
152           public KafkaPrincipal principal() {
153               InetAddress clientAddress = transportLayer.socketChannel().socket().getInetAddress();
154               // listenerName should only be null in Client mode where principal() should not be called
155               if (listenerName == null)
156                   throw new IllegalStateException("Unexpected call to principal() when listenerName is null");
157               SslAuthenticationContext context = new SslAuthenticationContext(
158                       transportLayer.sslSession(),
159                       clientAddress,
160                       listenerName.value());
161               return principalBuilder.build(context);
162           }
163   
164           @Override
165           public Optional<KafkaPrincipalSerde> principalSerde() {
166               return principalBuilder instanceof KafkaPrincipalSerde ? Optional.of((KafkaPrincipalSerde) principalBuilder) : Optional.empty();
167           }
168   
169           @Override
170           public void close() throws IOException {
171               if (principalBuilder instanceof Closeable)
172                   Utils.closeQuietly((Closeable) principalBuilder, "principal builder");
173           }
174   
175           /**
176            * SslAuthenticator doesn't implement any additional authentication mechanism.
177            * @return true
178            */
179           @Override
180           public boolean complete() {
181               return true;
182           }
183       }
184   }
