17   package org.apache.kafka.common.network;
18   
19   /*
20    * Transport layer for PLAINTEXT communication
21    */
22   
23   import java.io.IOException;
24   import java.nio.ByteBuffer;
25   import java.nio.channels.FileChannel;
26   import java.nio.channels.SocketChannel;
27   import java.nio.channels.SelectionKey;
28   
29   import java.security.Principal;
30   
31   import org.apache.kafka.common.security.auth.KafkaPrincipal;
32   
33   public class PlaintextTransportLayer implements TransportLayer {
34       private final SelectionKey key;
35       private final SocketChannel socketChannel;
36       private final Principal principal = KafkaPrincipal.ANONYMOUS;
37   
38       public PlaintextTransportLayer(SelectionKey key) throws IOException {
39           this.key = key;
40           this.socketChannel = (SocketChannel) key.channel();
41       }
42   
43       @Override
44       public boolean ready() {
45           return true;
46       }
47   
48       @Override
49       public boolean finishConnect() throws IOException {
50           boolean connected = socketChannel.finishConnect();
51           if (connected)
52               key.interestOps(key.interestOps() & ~SelectionKey.OP_CONNECT | SelectionKey.OP_READ);
53           return connected;
54       }
55   
56       @Override
57       public void disconnect() {
58           key.cancel();
59       }
60   
61       @Override
62       public SocketChannel socketChannel() {
63           return socketChannel;
64       }
65   
66       @Override
67       public SelectionKey selectionKey() {
68           return key;
69       }
70   
71       @Override
72       public boolean isOpen() {
73           return socketChannel.isOpen();
74       }
75   
76       @Override
77       public boolean isConnected() {
78           return socketChannel.isConnected();
79       }
80   
81       @Override
82       public void close() throws IOException {
83           socketChannel.socket().close();
84           socketChannel.close();
85       }
86   
87       /**
88        * Performs SSL handshake hence is a no-op for the non-secure
89        * implementation
90        */
91       @Override
92       public void handshake() {}
93   
94       /**
95       * Reads a sequence of bytes from this channel into the given buffer.
96       *
97       * @param dst The buffer into which bytes are to be transferred
98       * @return The number of bytes read, possible zero or -1 if the channel has reached end-of-stream
99       * @throws IOException if some other I/O error occurs
100       */
101       @Override
102       public int read(ByteBuffer dst) throws IOException {
103           return socketChannel.read(dst);
104       }
105   
106       /**
107        * Reads a sequence of bytes from this channel into the given buffers.
108        *
109        * @param dsts - The buffers into which bytes are to be transferred.
110        * @return The number of bytes read, possibly zero, or -1 if the channel has reached end-of-stream.
111        * @throws IOException if some other I/O error occurs
112        */
113       @Override
114       public long read(ByteBuffer[] dsts) throws IOException {
115           return socketChannel.read(dsts);
116       }
117   
118       /**
119        * Reads a sequence of bytes from this channel into a subsequence of the given buffers.
120        * @param dsts - The buffers into which bytes are to be transferred
121        * @param offset - The offset within the buffer array of the first buffer into which bytes are to be transferred; must be non-negative and no larger than dsts.length.
122        * @param length - The maximum number of buffers to be accessed; must be non-negative and no larger than dsts.length - offset
123        * @return The number of bytes read, possibly zero, or -1 if the channel has reached end-of-stream.
124        * @throws IOException if some other I/O error occurs
125        */
126       @Override
127       public long read(ByteBuffer[] dsts, int offset, int length) throws IOException {
128           return socketChannel.read(dsts, offset, length);
129       }
130   
131       /**
132       * Writes a sequence of bytes to this channel from the given buffer.
133       *
134       * @param src The buffer from which bytes are to be retrieved
135       * @return The number of bytes read, possibly zero, or -1 if the channel has reached end-of-stream
136       * @throws IOException If some other I/O error occurs
137       */
138       @Override
139       public int write(ByteBuffer src) throws IOException {
140           return socketChannel.write(src);
141       }
142   
143       /**
144       * Writes a sequence of bytes to this channel from the given buffer.
145       *
146       * @param srcs The buffer from which bytes are to be retrieved
147       * @return The number of bytes read, possibly zero, or -1 if the channel has reached end-of-stream
148       * @throws IOException If some other I/O error occurs
149       */
150       @Override
151       public long write(ByteBuffer[] srcs) throws IOException {
152           return socketChannel.write(srcs);
153       }
154   
155       /**
156       * Writes a sequence of bytes to this channel from the subsequence of the given buffers.
157       *
158       * @param srcs The buffers from which bytes are to be retrieved
159       * @param offset The offset within the buffer array of the first buffer from which bytes are to be retrieved; must be non-negative and no larger than srcs.length.
160       * @param length - The maximum number of buffers to be accessed; must be non-negative and no larger than srcs.length - offset.
161       * @return returns no.of bytes written , possibly zero.
162       * @throws IOException If some other I/O error occurs
163       */
164       @Override
165       public long write(ByteBuffer[] srcs, int offset, int length) throws IOException {
166           return socketChannel.write(srcs, offset, length);
167       }
168   
169       /**
170        * always returns false as there will be not be any
171        * pending writes since we directly write to socketChannel.
172        */
173       @Override
174       public boolean hasPendingWrites() {
175           return false;
176       }
177   
178       /**
179        * Returns ANONYMOUS as Principal.
180        */
181       @Override
182       public Principal peerPrincipal() {
183           return principal;
184       }
185   
186       /**
187        * Adds the interestOps to selectionKey.
188        */
189       @Override
190       public void addInterestOps(int ops) {
191           key.interestOps(key.interestOps() | ops);
192   
193       }
194   
195       /**
196        * Removes the interestOps from selectionKey.
197        */
198       @Override
199       public void removeInterestOps(int ops) {
200           key.interestOps(key.interestOps() & ~ops);
201       }
202   
203       @Override
204       public boolean isMute() {
205           return key.isValid() && (key.interestOps() & SelectionKey.OP_READ) == 0;
206       }
207   
208       @Override
209       public boolean hasBytesBuffered() {
210           return false;
211       }
212   
213       @Override
214       public long transferFrom(FileChannel fileChannel, long position, long count) throws IOException {
215           return fileChannel.transferTo(position, count, socketChannel);
216       }
217   }
