18   package org.apache.kafka.clients;
19   
20   import org.apache.kafka.common.Node;
21   import org.apache.kafka.common.errors.DisconnectException;
22   import org.apache.kafka.common.utils.Time;
23   
24   import java.io.IOException;
25   import java.util.List;
26   
27   /**
28    * Provides additional utilities for {@link NetworkClient} (e.g. to implement blocking behaviour).
29    */
30   public final class NetworkClientUtils {
31   
32       private NetworkClientUtils() {}
33   
34       /**
35        * Checks whether the node is currently connected, first calling `client.poll` to ensure that any pending
36        * disconnects have been processed.
37        *
38        * This method can be used to check the status of a connection prior to calling the blocking version to be able
39        * to tell whether the latter completed a new connection.
40        */
41       public static boolean isReady(KafkaClient client, Node node, long currentTime) {
42           client.poll(0, currentTime);
43           return client.isReady(node, currentTime);
44       }
45   
46       /**
47        * Invokes `client.poll` to discard pending disconnects, followed by `client.ready` and 0 or more `client.poll`
48        * invocations until the connection to `node` is ready, the timeoutMs expires or the connection fails.
49        *
50        * It returns `true` if the call completes normally or `false` if the timeoutMs expires. If the connection fails,
51        * an `IOException` is thrown instead. Note that if the `NetworkClient` has been configured with a positive
52        * connection timeoutMs, it is possible for this method to raise an `IOException` for a previous connection which
53        * has recently disconnected. If authentication to the node fails, an `AuthenticationException` is thrown.
54        *
55        * This method is useful for implementing blocking behaviour on top of the non-blocking `NetworkClient`, use it with
56        * care.
57        */
58       public static boolean awaitReady(KafkaClient client, Node node, Time time, long timeoutMs) throws IOException {
59           if (timeoutMs < 0) {
60               throw new IllegalArgumentException("Timeout needs to be greater than 0");
61           }
62           long startTime = time.milliseconds();
63   
64           if (isReady(client, node, startTime) ||  client.ready(node, startTime))
65               return true;
66   
67           long attemptStartTime = time.milliseconds();
68           while (!client.isReady(node, attemptStartTime) && attemptStartTime - startTime < timeoutMs) {
69               if (client.connectionFailed(node)) {
70                   throw new IOException("Connection to " + node + " failed.");
71               }
72               long pollTimeout = timeoutMs - (attemptStartTime - startTime); // initialize in this order to avoid overflow
73               client.poll(pollTimeout, attemptStartTime);
74               if (client.authenticationException(node) != null)
75                   throw client.authenticationException(node);
76               attemptStartTime = time.milliseconds();
77           }
78           return client.isReady(node, attemptStartTime);
79       }
80   
81       /**
82        * Invokes `client.send` followed by 1 or more `client.poll` invocations until a response is received or a
83        * disconnection happens (which can happen for a number of reasons including a request timeout).
84        *
85        * In case of a disconnection, an `IOException` is thrown.
86        * If shutdown is initiated on the client during this method, an IOException is thrown.
87        *
88        * This method is useful for implementing blocking behaviour on top of the non-blocking `NetworkClient`, use it with
89        * care.
90        */
91       public static ClientResponse sendAndReceive(KafkaClient client, ClientRequest request, Time time) throws IOException {
92           try {
93               client.send(request, time.milliseconds());
94               while (client.active()) {
95                   List<ClientResponse> responses = client.poll(Long.MAX_VALUE, time.milliseconds());
96                   for (ClientResponse response : responses) {
97                       if (response.requestHeader().correlationId() == request.correlationId()) {
98                           if (response.wasDisconnected()) {
99                               throw new IOException("Connection to " + response.destination() + " was disconnected before the response was read");
100                           }
101                           if (response.versionMismatch() != null) {
102                               throw response.versionMismatch();
103                           }
104                           return response;
105                       }
106                   }
107               }
108               throw new IOException("Client was shutdown before response was read");
109           } catch (DisconnectException e) {
110               if (client.active())
111                   throw e;
112               else
113                   throw new IOException("Client was shutdown before response was read");
114   
115           }
116       }
117   }
