19   import java.util.concurrent.CancellationException;
20   import java.util.concurrent.CompletableFuture;
21   import java.util.concurrent.CompletionException;
22   import java.util.concurrent.CompletionStage;
23   import java.util.concurrent.ExecutionException;
24   import java.util.concurrent.TimeUnit;
25   import java.util.concurrent.TimeoutException;
26   
27   import org.apache.kafka.common.KafkaFuture;

29   /**
30    * A flexible future which supports call chaining and other asynchronous programming patterns.
31    */
32   public class KafkaFutureImpl<T> extends KafkaFuture<T> {
33   
34       private final KafkaCompletableFuture<T> completableFuture;
35   
36       private final boolean isDependant;
37   
38       public KafkaFutureImpl() {
39           this(false, new KafkaCompletableFuture<>());
40       }
41   
42       private KafkaFutureImpl(boolean isDependant, KafkaCompletableFuture<T> completableFuture) {
43           this.isDependant = isDependant;
44           this.completableFuture = completableFuture;
45       }
46   
47       @Override
48       public CompletionStage<T> toCompletionStage() {
49           return completableFuture;
50       }
51   
52       /**
53        * Returns a new KafkaFuture that, when this future completes normally, is executed with this
54        * futures's result as the argument to the supplied function.
55        */
56       @Override
57       public <R> KafkaFuture<R> thenApply(BaseFunction<T, R> function) {
58           CompletableFuture<R> appliedFuture = completableFuture.thenApply(value -> {
59               try {
60                   return function.apply(value);
61               } catch (Throwable t) {
62                   if (t instanceof CompletionException) {
63                       // KafkaFuture#thenApply, when the function threw CompletionException should return
64                       // an ExecutionException wrapping a CompletionException wrapping the exception thrown by the
65                       // function. CompletableFuture#thenApply will just return ExecutionException wrapping the
66                       // exception thrown by the function, so we add an extra CompletionException here to
67                       // maintain the KafkaFuture behaviour.
68                       throw new CompletionException(t);
69                   } else {
70                       throw t;
71                   }
72               }
73           });
74           return new KafkaFutureImpl<>(true, toKafkaCompletableFuture(appliedFuture));
75       }
76   
77       private static <U> KafkaCompletableFuture<U> toKafkaCompletableFuture(CompletableFuture<U> completableFuture) {
78           if (completableFuture instanceof KafkaCompletableFuture) {
79               return (KafkaCompletableFuture<U>) completableFuture;
80           } else {
81               final KafkaCompletableFuture<U> result = new KafkaCompletableFuture<>();
82               completableFuture.whenComplete((x, y) -> {
83                   if (y != null) {
84                       result.kafkaCompleteExceptionally(y);
85                   } else {
86                       result.kafkaComplete(x);
87                   }
88               });
89               return result;
90           }
91       }
92   
93       /**
94        * @see KafkaFutureImpl#thenApply(BaseFunction)
95        * @deprecated Since Kafka 3.0.
96        */
97       @Deprecated
98       @Override
99       public <R> KafkaFuture<R> thenApply(Function<T, R> function) {
100           return thenApply((BaseFunction<T, R>) function);
101       }
102   
103       @Override
104       public KafkaFuture<T> whenComplete(final BiConsumer<? super T, ? super Throwable> biConsumer) {
105           CompletableFuture<T> tCompletableFuture = completableFuture.whenComplete((java.util.function.BiConsumer<? super T, ? super Throwable>) (a, b) -> {
106               try {
107                   biConsumer.accept(a, b);
108               } catch (Throwable t) {
109                   if (t instanceof CompletionException) {
110                       throw new CompletionException(t);
111                   } else {
112                       throw t;
113                   }
114               }
115           });
116           return new KafkaFutureImpl<>(true, toKafkaCompletableFuture(tCompletableFuture));
117       }
118   
119   
120       @Override
121       public boolean complete(T newValue) {
122           return completableFuture.kafkaComplete(newValue);
123       }
124   
125       @Override
126       public boolean completeExceptionally(Throwable newException) {
127           // CompletableFuture#get() always wraps the _cause_ of a CompletionException in ExecutionException
128           // (which KafkaFuture does not) so wrap CompletionException in an extra one to avoid losing the
129           // first CompletionException in the exception chain.
130           return completableFuture.kafkaCompleteExceptionally(
131                   newException instanceof CompletionException ? new CompletionException(newException) : newException);
132       }
133   
134       /**
135        * If not already completed, completes this future with a CancellationException.  Dependent
136        * futures that have not already completed will also complete exceptionally, with a
137        * CompletionException caused by this CancellationException.
138        */
139       @Override
140       public boolean cancel(boolean mayInterruptIfRunning) {
141           return completableFuture.cancel(mayInterruptIfRunning);
142       }
143   
144       /**
145        * We need to deal with differences between KafkaFuture's historic API and the API of CompletableFuture:
146        * CompletableFuture#get() does not wrap CancellationException in ExecutionException (nor does KafkaFuture).
147        * CompletableFuture#get() always wraps the _cause_ of a CompletionException in ExecutionException
148        * (which KafkaFuture does not).
149        *
150        * The semantics for KafkaFuture are that all exceptional completions of the future (via #completeExceptionally()
151        * or exceptions from dependants) manifest as ExecutionException, as observed via both get() and getNow().
152        */
153       private void maybeThrowCancellationException(Throwable cause) {
154           if (cause instanceof CancellationException) {
155               throw (CancellationException) cause;
156           }
157       }
158   
159       /**
160        * Waits if necessary for this future to complete, and then returns its result.
161        */
162       @Override
163       public T get() throws InterruptedException, ExecutionException {
164           try {
165               return completableFuture.get();
166           } catch (ExecutionException e) {
167               maybeThrowCancellationException(e.getCause());
168               throw e;
169           }
170       }
171   
172       /**
173        * Waits if necessary for at most the given time for this future to complete, and then returns
174        * its result, if available.
175        */
176       @Override
177       public T get(long timeout, TimeUnit unit) throws InterruptedException, ExecutionException,
178               TimeoutException {
179           try {
180               return completableFuture.get(timeout, unit);
181           } catch (ExecutionException e) {
182               maybeThrowCancellationException(e.getCause());
183               throw e;
184           }
185       }
186   
187       /**
188        * Returns the result value (or throws any encountered exception) if completed, else returns
189        * the given valueIfAbsent.
190        */
191       @Override
192       public T getNow(T valueIfAbsent) throws ExecutionException {
193           try {
194               return completableFuture.getNow(valueIfAbsent);
195           } catch (CompletionException e) {
196               maybeThrowCancellationException(e.getCause());
197               // Note, unlike CompletableFuture#get() which throws ExecutionException, CompletableFuture#getNow()
198               // throws CompletionException, thus needs rewrapping to conform to KafkaFuture API,
199               // where KafkaFuture#getNow() throws ExecutionException.
200               throw new ExecutionException(e.getCause());
201           }
202       }
203   
204       /**
205        * Returns true if this CompletableFuture was cancelled before it completed normally.
206        */
207       @Override
208       public boolean isCancelled() {
209           if (isDependant) {
210               // Having isCancelled() for a dependent future just return
211               // CompletableFuture.isCancelled() would break the historical KafkaFuture behaviour because
212               // CompletableFuture#isCancelled() just checks for the exception being CancellationException
213               // whereas it will be a CompletionException wrapping a CancellationException
214               // due needing to compensate for CompletableFuture's CompletionException unwrapping
215               // shenanigans in other methods.
216               try {
217                   completableFuture.getNow(null);
218                   return false;
219               } catch (Exception e) {
220                   return e instanceof CompletionException
221                           && e.getCause() instanceof CancellationException;
222               }
223           } else {
224               return completableFuture.isCancelled();
225           }
226       }
227   
228       /**
229        * Returns true if this CompletableFuture completed exceptionally, in any way.
230        */
231       @Override
232       public boolean isCompletedExceptionally() {
233           return completableFuture.isCompletedExceptionally();
234       }
235   
236       /**
237        * Returns true if completed in any fashion: normally, exceptionally, or via cancellation.
238        */
239       @Override
240       public boolean isDone() {
241           return completableFuture.isDone();
242       }
243   
244       @Override
245       public String toString() {
246           T value = null;
247           Throwable exception = null;
248           try {
249               value = completableFuture.getNow(null);
250           } catch (CompletionException e) {
251               exception = e.getCause();
252           } catch (Exception e) {
253               exception = e;
254           }
255           return String.format("KafkaFuture{value=%s,exception=%s,done=%b}", value, exception, exception != null || value != null);
256       }
257   }
