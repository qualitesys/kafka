186       private SSLEngine createSslEngine(Mode mode, String peerHost, int peerPort, String endpointIdentification) {
187           SSLEngine sslEngine = sslContext.createSSLEngine(peerHost, peerPort);
188           if (cipherSuites != null) sslEngine.setEnabledCipherSuites(cipherSuites);
189           if (enabledProtocols != null) sslEngine.setEnabledProtocols(enabledProtocols);
190   
191           if (mode == Mode.SERVER) {
192               sslEngine.setUseClientMode(false);
193               switch (sslClientAuth) {
194                   case REQUIRED:
195                       sslEngine.setNeedClientAuth(true);
196                       break;
197                   case REQUESTED:
198                       sslEngine.setWantClientAuth(true);
199                       break;
200                   case NONE:
201                       break;
202               }
203               sslEngine.setUseClientMode(false);
204           } else {
205               sslEngine.setUseClientMode(true);
206               SSLParameters sslParams = sslEngine.getSSLParameters();
207               // SSLParameters#setEndpointIdentificationAlgorithm enables endpoint validation
208               // only in client mode. Hence, validation is enabled only for clients.
209               sslParams.setEndpointIdentificationAlgorithm(endpointIdentification);
210               sslEngine.setSSLParameters(sslParams);
211           }
212           return sslEngine;
213       }

237       private SSLContext createSSLContext(SecurityStore keystore, SecurityStore truststore) {
238           try {
239               SSLContext sslContext;
240               if (provider != null)
241                   sslContext = SSLContext.getInstance(protocol, provider);
242               else
243                   sslContext = SSLContext.getInstance(protocol);
244   
245               KeyManager[] keyManagers = null;
246               if (keystore != null || kmfAlgorithm != null) {
247                   String kmfAlgorithm = this.kmfAlgorithm != null ?
248                           this.kmfAlgorithm : KeyManagerFactory.getDefaultAlgorithm();
249                   KeyManagerFactory kmf = KeyManagerFactory.getInstance(kmfAlgorithm);
250                   if (keystore != null) {
251                       kmf.init(keystore.get(), keystore.keyPassword());
252                   } else {
253                       kmf.init(null, null);
254                   }
255                   keyManagers = kmf.getKeyManagers();
256               }
257   
258               String tmfAlgorithm = this.tmfAlgorithm != null ? this.tmfAlgorithm : TrustManagerFactory.getDefaultAlgorithm();
259               TrustManagerFactory tmf = TrustManagerFactory.getInstance(tmfAlgorithm);
260               KeyStore ts = truststore == null ? null : truststore.get();
261               tmf.init(ts);
262   
263               sslContext.init(keyManagers, tmf.getTrustManagers(), this.secureRandomImplementation);
264               log.debug("Created SSL context with keystore {}, truststore {}, provider {}.",
265                       keystore, truststore, sslContext.getProvider().getName());
266               return sslContext;
267           } catch (Exception e) {
268               throw new KafkaException(e);
269           }
270       }

272       // Visibility to override for testing
273       protected SecurityStore createKeystore(String type, String path, Password password, Password keyPassword, Password privateKey, Password certificateChain) {
274           if (privateKey != null) {
275               if (!PEM_TYPE.equals(type))
276                   throw new InvalidConfigurationException("SSL private key can be specified only for PEM, but key store type is " + type + ".");
277               else if (certificateChain == null)
278                   throw new InvalidConfigurationException("SSL private key is specified, but certificate chain is not specified.");
279               else if (path != null)
280                   throw new InvalidConfigurationException("Both SSL key store location and separate private key are specified.");
281               else if (password != null)
282                   throw new InvalidConfigurationException("SSL key store password cannot be specified with PEM format, only key password may be specified.");
283               else
284                   return new PemStore(certificateChain, privateKey, keyPassword);
285           } else if (certificateChain != null) {
286               throw new InvalidConfigurationException("SSL certificate chain is specified, but private key is not specified");
287           } else if (PEM_TYPE.equals(type) && path != null) {
288               if (password != null)
289                   throw new InvalidConfigurationException("SSL key store password cannot be specified with PEM format, only key password may be specified");
290               else
291                   return new FileBasedPemStore(path, keyPassword, true);
292           } else if (path == null && password != null) {
293               throw new InvalidConfigurationException("SSL key store is not specified, but key store password is specified.");
294           } else if (path != null && password == null) {
295               throw new InvalidConfigurationException("SSL key store is specified, but key store password is not specified.");
296           } else if (path != null && password != null) {
297               return new FileBasedStore(type, path, password, keyPassword, true);
298           } else
299               return null; // path == null, clients may use this path with brokers that don't require client auth
300       }

302       private static SecurityStore createTruststore(String type, String path, Password password, Password trustStoreCerts) {
303           if (trustStoreCerts != null) {
304               if (!PEM_TYPE.equals(type))
305                   throw new InvalidConfigurationException("SSL trust store certs can be specified only for PEM, but trust store type is " + type + ".");
306               else if (path != null)
307                   throw new InvalidConfigurationException("Both SSL trust store location and separate trust certificates are specified.");
308               else if (password != null)
309                   throw new InvalidConfigurationException("SSL trust store password cannot be specified for PEM format.");
310               else
311                   return new PemStore(trustStoreCerts);
312           } else if (PEM_TYPE.equals(type) && path != null) {
313               if (password != null)
314                   throw new InvalidConfigurationException("SSL trust store password cannot be specified for PEM format.");
315               else
316                   return new FileBasedPemStore(path, null, false);
317           } else if (path == null && password != null) {
318               throw new InvalidConfigurationException("SSL trust store is not specified, but trust store password is specified.");
319           } else if (path != null) {
320               return new FileBasedStore(type, path, password, null, false);
321           } else
322               return null;
323       }

361           /**
362            * Loads this keystore
363            * @return the keystore
364            * @throws KafkaException if the file could not be read or if the keystore could not be loaded
365            *   using the specified configs (e.g. if the password or keystore type is invalid)
366            */
367           protected KeyStore load(boolean isKeyStore) {
368               try (InputStream in = Files.newInputStream(Paths.get(path))) {
369                   KeyStore ks = KeyStore.getInstance(type);
370                   // If a password is not set access to the truststore is still available, but integrity checking is disabled.
371                   char[] passwordChars = password != null ? password.value().toCharArray() : null;
372                   ks.load(in, passwordChars);
373                   return ks;
374               } catch (GeneralSecurityException | IOException e) {
375                   throw new KafkaException("Failed to load SSL keystore " + path + " of type " + type, e);
376               }
377           }

464               } catch (Exception e) {

466               }

485           private Certificate[] certs(String pem) throws GeneralSecurityException {
486               List<byte[]> certEntries = CERTIFICATE_PARSER.pemEntries(pem);
487               if (certEntries.isEmpty())
488                   throw new InvalidConfigurationException("At least one certificate expected, but none found");
489   
490               Certificate[] certs = new Certificate[certEntries.size()];
491               for (int i = 0; i < certs.length; i++) {
492                   certs[i] = CertificateFactory.getInstance("X.509")
493                       .generateCertificate(new ByteArrayInputStream(certEntries.get(i)));
494               }
495               return certs;
496           }

498           private PrivateKey privateKey(String pem, char[] keyPassword) throws Exception {
499               List<byte[]> keyEntries = PRIVATE_KEY_PARSER.pemEntries(pem);
500               if (keyEntries.isEmpty())
501                   throw new InvalidConfigurationException("Private key not provided");
502               if (keyEntries.size() != 1)
503                   throw new InvalidConfigurationException("Expected one private key, but found " + keyEntries.size());
504   
505               byte[] keyBytes = keyEntries.get(0);
506               PKCS8EncodedKeySpec keySpec;
507               if (keyPassword == null) {
508                   keySpec = new PKCS8EncodedKeySpec(keyBytes);
509               } else {
510                   EncryptedPrivateKeyInfo keyInfo = new EncryptedPrivateKeyInfo(keyBytes);
511                   String algorithm = keyInfo.getAlgName();
512                   SecretKeyFactory keyFactory = SecretKeyFactory.getInstance(algorithm);
513                   SecretKey pbeKey = keyFactory.generateSecret(new PBEKeySpec(keyPassword));
514                   Cipher cipher = Cipher.getInstance(algorithm);
515                   cipher.init(Cipher.DECRYPT_MODE, pbeKey, keyInfo.getAlgParameters());
516                   keySpec = keyInfo.getKeySpec(cipher);
517               }
518   
519               InvalidKeySpecException firstException = null;
520               for (KeyFactory factory : KEY_FACTORIES) {
521                   try {
522                       return factory.generatePrivate(keySpec);
523                   } catch (InvalidKeySpecException e) {
524                       if (firstException == null)
525                           firstException = e;
526                   }
527               }
528               throw new InvalidConfigurationException("Private key could not be loaded", firstException);
529           }

534               } catch (Exception e) {
535                   throw new InvalidConfigurationException("Could not create key factory for algorithm " + algorithm, e);
536               }

540       /**
541        * Parser to process certificate/private key entries from PEM files
542        * Examples:
543        *   -----BEGIN CERTIFICATE-----
544        *   Base64 cert
545        *   -----END CERTIFICATE-----
546        *
547        *   -----BEGIN ENCRYPTED PRIVATE KEY-----
548        *   Base64 private key
549        *   -----END ENCRYPTED PRIVATE KEY-----
550        *   Additional data may be included before headers, so we match all entries within the PEM.
551        */
552       static class PemParser {
553           private final String name;
554           private final Pattern pattern;
555   
556           PemParser(String name) {
557               this.name = name;
558               String beginOrEndFormat = "-+%s\\s*.*%s[^-]*-+\\s+";
559               String nameIgnoreSpace = name.replace(" ", "\\s+");
560   
561               String encodingParams = "\\s*[^\\r\\n]*:[^\\r\\n]*[\\r\\n]+";
562               String base64Pattern = "([a-zA-Z0-9/+=\\s]*)";
563               String patternStr =  String.format(beginOrEndFormat, "BEGIN", nameIgnoreSpace) +
564                   String.format("(?:%s)*", encodingParams) +
565                   base64Pattern +
566                   String.format(beginOrEndFormat, "END", nameIgnoreSpace);
567               pattern = Pattern.compile(patternStr);
568           }
569   
570           private List<byte[]> pemEntries(String pem) {
571               Matcher matcher = pattern.matcher(pem + "\n"); // allow last newline to be omitted in value
572               List<byte[]>  entries = new ArrayList<>();
573               while (matcher.find()) {
574                   String base64Str = matcher.group(1).replaceAll("\\s", "");
575                   entries.add(Base64.getDecoder().decode(base64Str));
576               }
577               if (entries.isEmpty())
578                   throw new InvalidConfigurationException("No matching " + name + " entries in PEM file");
579               return entries;
580           }
581       }
