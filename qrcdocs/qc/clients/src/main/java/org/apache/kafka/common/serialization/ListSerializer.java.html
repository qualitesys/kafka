20   import java.util.Iterator;

23   import org.apache.kafka.common.config.ConfigException;
24   import org.apache.kafka.common.utils.Utils;

26   import java.io.ByteArrayOutputStream;
27   import java.io.DataOutputStream;
28   import java.io.IOException;
29   import java.util.Arrays;

32   import org.slf4j.Logger;

35   import static org.apache.kafka.common.serialization.Serdes.ListSerde.SerializationStrategy;
36   
37   public class ListSerializer<Inner> implements Serializer<List<Inner>> {
38   
39       final Logger log = LoggerFactory.getLogger(ListSerializer.class);
40   
41       private static final List<Class<? extends Serializer<?>>> FIXED_LENGTH_SERIALIZERS = Arrays.asList(
42           ShortSerializer.class,
43           IntegerSerializer.class,
44           FloatSerializer.class,
45           LongSerializer.class,
46           DoubleSerializer.class,
47           UUIDSerializer.class);
48   
49       private Serializer<Inner> inner;
50       private SerializationStrategy serStrategy;
51   
52       public ListSerializer() {}
53   
54       public ListSerializer(Serializer<Inner> inner) {
55           if (inner == null) {
56               throw new IllegalArgumentException("ListSerializer requires \"serializer\" parameter to be provided during initialization");
57           }
58           this.inner = inner;
59           this.serStrategy = FIXED_LENGTH_SERIALIZERS.contains(inner.getClass()) ? SerializationStrategy.CONSTANT_SIZE : SerializationStrategy.VARIABLE_SIZE;
60       }
61   
62       public Serializer<Inner> getInnerSerializer() {
63           return inner;
64       }
65   
66       @SuppressWarnings("unchecked")
67       @Override
68       public void configure(Map<String, ?> configs, boolean isKey) {
69           if (inner != null) {
70               log.error("Could not configure ListSerializer as the parameter has already been set -- inner: {}", inner);
71               throw new ConfigException("List serializer was already initialized using a non-default constructor");
72           }
73           final String innerSerdePropertyName = isKey ? CommonClientConfigs.DEFAULT_LIST_KEY_SERDE_INNER_CLASS : CommonClientConfigs.DEFAULT_LIST_VALUE_SERDE_INNER_CLASS;
74           final Object innerSerdeClassOrName = configs.get(innerSerdePropertyName);
75           if (innerSerdeClassOrName == null) {
76               throw new ConfigException("Not able to determine the serializer class because it was neither passed via the constructor nor set in the config.");
77           }
78           try {
79               if (innerSerdeClassOrName instanceof String) {
80                   inner = Utils.newInstance((String) innerSerdeClassOrName, Serde.class).serializer();
81               } else if (innerSerdeClassOrName instanceof Class) {
82                   inner = (Serializer<Inner>) ((Serde) Utils.newInstance((Class) innerSerdeClassOrName)).serializer();
83               } else {
84                   throw new KafkaException("Could not create a serializer class instance using \"" + innerSerdePropertyName + "\" property.");
85               }
86               inner.configure(configs, isKey);
87               serStrategy = FIXED_LENGTH_SERIALIZERS.contains(inner.getClass()) ? SerializationStrategy.CONSTANT_SIZE : SerializationStrategy.VARIABLE_SIZE;
88           } catch (final ClassNotFoundException e) {
89               throw new ConfigException(innerSerdePropertyName, innerSerdeClassOrName, "Serializer class " + innerSerdeClassOrName + " could not be found.");
90           }
91       }
92   
93       private void serializeNullIndexList(final DataOutputStream out, List<Inner> data) throws IOException {
94           int i = 0;
95           List<Integer> nullIndexList = new ArrayList<>();
96           for (Iterator<Inner> it = data.listIterator(); it.hasNext(); i++) {
97               if (it.next() == null) {
98                   nullIndexList.add(i);
99               }
100           }
101           out.writeInt(nullIndexList.size());
102           for (int nullIndex : nullIndexList) {
103               out.writeInt(nullIndex);
104           }
105       }
106   
107       @Override
108       public byte[] serialize(String topic, List<Inner> data) {
109           if (data == null) {
110               return null;
111           }
112           try (final ByteArrayOutputStream baos = new ByteArrayOutputStream();
113                final DataOutputStream out = new DataOutputStream(baos)) {
114               out.writeByte(serStrategy.ordinal()); // write serialization strategy flag
115               if (serStrategy == SerializationStrategy.CONSTANT_SIZE) {
116                   // In CONSTANT_SIZE strategy, indexes of null entries are encoded in a null index list
117                   serializeNullIndexList(out, data);
118               }
119               final int size = data.size();
120               out.writeInt(size);
121               for (Inner entry : data) {
122                   if (entry == null) {
123                       if (serStrategy == SerializationStrategy.VARIABLE_SIZE) {
124                           out.writeInt(Serdes.ListSerde.NULL_ENTRY_VALUE);
125                       }
126                   } else {
127                       final byte[] bytes = inner.serialize(topic, entry);
128                       if (serStrategy == SerializationStrategy.VARIABLE_SIZE) {
129                           out.writeInt(bytes.length);
130                       }
131                       out.write(bytes);
132                   }
133               }
134               return baos.toByteArray();
135           } catch (IOException e) {
136               log.error("Failed to serialize list due to", e);
137               log.trace("List that could not be serialized: {}", data); // avoid logging actual data above TRACE level since it may contain sensitive information
138               throw new KafkaException("Failed to serialize List", e);
139           }
140       }
141   
142       @Override
143       public void close() {
144           if (inner != null) {
145               inner.close();
146           }
147       }
