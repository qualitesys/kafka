41   public class AbortTransactionHandler extends AdminApiHandler.Batched<TopicPartition, Void> {
42       private final Logger log;
43       private final AbortTransactionSpec abortSpec;
44       private final PartitionLeaderStrategy lookupStrategy;
45   
46       public AbortTransactionHandler(
47           AbortTransactionSpec abortSpec,
48           LogContext logContext
49       ) {
50           this.abortSpec = abortSpec;
51           this.log = logContext.logger(AbortTransactionHandler.class);
52           this.lookupStrategy = new PartitionLeaderStrategy(logContext);
53       }

55       public static AdminApiFuture.SimpleAdminApiFuture<TopicPartition, Void> newFuture(
56           Set<TopicPartition> topicPartitions
57       ) {
58           return AdminApiFuture.forKeys(topicPartitions);
59       }

61       @Override
62       public String apiName() {
63           return "abortTransaction";
64       }
65   
66       @Override
67       public AdminApiLookupStrategy<TopicPartition> lookupStrategy() {
68           return lookupStrategy;
69       }

71       @Override
72       public WriteTxnMarkersRequest.Builder buildBatchedRequest(
73           int brokerId,
74           Set<TopicPartition> topicPartitions
75       ) {
76           validateTopicPartitions(topicPartitions);
77   
78           WriteTxnMarkersRequestData.WritableTxnMarker marker = new WriteTxnMarkersRequestData.WritableTxnMarker()
79               .setCoordinatorEpoch(abortSpec.coordinatorEpoch())
80               .setProducerEpoch(abortSpec.producerEpoch())
81               .setProducerId(abortSpec.producerId())
82               .setTransactionResult(false);
83   
84           marker.topics().add(new WriteTxnMarkersRequestData.WritableTxnMarkerTopic()
85               .setName(abortSpec.topicPartition().topic())
86               .setPartitionIndexes(singletonList(abortSpec.topicPartition().partition()))
87           );
88   
89           WriteTxnMarkersRequestData request = new WriteTxnMarkersRequestData();
90           request.markers().add(marker);
91   
92           return new WriteTxnMarkersRequest.Builder(request);
93       }

95       @Override
96       public ApiResult<TopicPartition, Void> handleResponse(
97           Node broker,
98           Set<TopicPartition> topicPartitions,
99           AbstractResponse abstractResponse
100       ) {
101           validateTopicPartitions(topicPartitions);
102   
103           WriteTxnMarkersResponse response = (WriteTxnMarkersResponse) abstractResponse;
104           List<WriteTxnMarkersResponseData.WritableTxnMarkerResult> markerResponses = response.data().markers();
105   
106           if (markerResponses.size() != 1 || markerResponses.get(0).producerId() != abortSpec.producerId()) {
107               return ApiResult.failed(abortSpec.topicPartition(), new KafkaException("WriteTxnMarkers response " +
108                   "included unexpected marker entries: " + markerResponses + "(expected to find exactly one " +
109                   "entry with producerId " + abortSpec.producerId() + ")"));
110           }
111   
112           WriteTxnMarkersResponseData.WritableTxnMarkerResult markerResponse = markerResponses.get(0);
113           List<WriteTxnMarkersResponseData.WritableTxnMarkerTopicResult> topicResponses = markerResponse.topics();
114   
115           if (topicResponses.size() != 1 || !topicResponses.get(0).name().equals(abortSpec.topicPartition().topic())) {
116               return ApiResult.failed(abortSpec.topicPartition(), new KafkaException("WriteTxnMarkers response " +
117                   "included unexpected topic entries: " + markerResponses + "(expected to find exactly one " +
118                   "entry with topic partition " + abortSpec.topicPartition() + ")"));
119           }
120   
121           WriteTxnMarkersResponseData.WritableTxnMarkerTopicResult topicResponse = topicResponses.get(0);
122           List<WriteTxnMarkersResponseData.WritableTxnMarkerPartitionResult> partitionResponses =
123               topicResponse.partitions();
124   
125           if (partitionResponses.size() != 1 || partitionResponses.get(0).partitionIndex() != abortSpec.topicPartition().partition()) {
126               return ApiResult.failed(abortSpec.topicPartition(), new KafkaException("WriteTxnMarkers response " +
127                   "included unexpected partition entries for topic " + abortSpec.topicPartition().topic() +
128                   ": " + markerResponses + "(expected to find exactly one entry with partition " +
129                   abortSpec.topicPartition().partition() + ")"));
130           }
131   
132           WriteTxnMarkersResponseData.WritableTxnMarkerPartitionResult partitionResponse = partitionResponses.get(0);
133           Errors error = Errors.forCode(partitionResponse.errorCode());
134   
135           if (error != Errors.NONE) {
136               return handleError(error);
137           } else {
138               return ApiResult.completed(abortSpec.topicPartition(), null);
139           }
140       }
141   
142       private ApiResult<TopicPartition, Void> handleError(Errors error) {
143           switch (error) {
144               case CLUSTER_AUTHORIZATION_FAILED:
145                   log.error("WriteTxnMarkers request for abort spec {} failed cluster authorization", abortSpec);
146                   return ApiResult.failed(abortSpec.topicPartition(), new ClusterAuthorizationException(
147                       "WriteTxnMarkers request with " + abortSpec + " failed due to cluster " +
148                           "authorization error"));
149   
150               case INVALID_PRODUCER_EPOCH:
151                   log.error("WriteTxnMarkers request for abort spec {} failed due to an invalid producer epoch",
152                       abortSpec);
153                   return ApiResult.failed(abortSpec.topicPartition(), new InvalidProducerEpochException(
154                       "WriteTxnMarkers request with " + abortSpec + " failed due an invalid producer epoch"));
155   
156               case TRANSACTION_COORDINATOR_FENCED:
157                   log.error("WriteTxnMarkers request for abort spec {} failed because the coordinator epoch is fenced",
158                       abortSpec);
159                   return ApiResult.failed(abortSpec.topicPartition(), new TransactionCoordinatorFencedException(
160                       "WriteTxnMarkers request with " + abortSpec + " failed since the provided " +
161                           "coordinator epoch " + abortSpec.coordinatorEpoch() + " has been fenced " +
162                           "by the active coordinator"));
163   
164               case NOT_LEADER_OR_FOLLOWER:
165               case REPLICA_NOT_AVAILABLE:
166               case BROKER_NOT_AVAILABLE:
167               case UNKNOWN_TOPIC_OR_PARTITION:
168                   log.debug("WriteTxnMarkers request for abort spec {} failed due to {}. Will retry after attempting to " +
169                           "find the leader again", abortSpec, error);
170                   return ApiResult.unmapped(singletonList(abortSpec.topicPartition()));
171   
172               default:
173                   log.error("WriteTxnMarkers request for abort spec {} failed due to an unexpected error {}",
174                       abortSpec, error);
175                   return ApiResult.failed(abortSpec.topicPartition(), error.exception(
176                       "WriteTxnMarkers request with " + abortSpec + " failed due to unexpected error: " + error.message()));
177           }
178       }

180       private void validateTopicPartitions(Set<TopicPartition> topicPartitions) {
181           if (!topicPartitions.equals(singleton(abortSpec.topicPartition()))) {
182               throw new IllegalArgumentException("Received unexpected topic partitions " + topicPartitions +
183                   " (expected only " + singleton(abortSpec.topicPartition()) + ")");
184           }
185       }
