36   /**
37    * The compression type to use
38    */
39   public enum CompressionType {
40       NONE(0, "none", 1.0f) {
41           @Override
42           public OutputStream wrapForOutput(ByteBufferOutputStream buffer, byte messageVersion) {
43               return buffer;
44           }
45   
46           @Override
47           public InputStream wrapForInput(ByteBuffer buffer, byte messageVersion, BufferSupplier decompressionBufferSupplier) {
48               return new ByteBufferInputStream(buffer);
49           }
50       },
51   
52       // Shipped with the JDK
53       GZIP(1, "gzip", 1.0f) {
54           @Override
55           public OutputStream wrapForOutput(ByteBufferOutputStream buffer, byte messageVersion) {
56               try {
57                   // Set input buffer (uncompressed) to 16 KB (none by default) and output buffer (compressed) to
58                   // 8 KB (0.5 KB by default) to ensure reasonable performance in cases where the caller passes a small
59                   // number of bytes to write (potentially a single byte)
60                   return new BufferedOutputStream(new GZIPOutputStream(buffer, 8 * 1024), 16 * 1024);
61               } catch (Exception e) {
62                   throw new KafkaException(e);
63               }
64           }
65   
66           @Override
67           public InputStream wrapForInput(ByteBuffer buffer, byte messageVersion, BufferSupplier decompressionBufferSupplier) {
68               try {
69                   // Set output buffer (uncompressed) to 16 KB (none by default) and input buffer (compressed) to
70                   // 8 KB (0.5 KB by default) to ensure reasonable performance in cases where the caller reads a small
71                   // number of bytes (potentially a single byte)
72                   return new BufferedInputStream(new GZIPInputStream(new ByteBufferInputStream(buffer), 8 * 1024),
73                           16 * 1024);
74               } catch (Exception e) {
75                   throw new KafkaException(e);
76               }
77           }
78       },
79   
80       // We should only load classes from a given compression library when we actually use said compression library. This
81       // is because compression libraries include native code for a set of platforms and we want to avoid errors
82       // in case the platform is not supported and the compression library is not actually used.
83       // To ensure this, we only reference compression library code from classes that are only invoked when actual usage
84       // happens.
85   
86       SNAPPY(2, "snappy", 1.0f) {
87           @Override
88           public OutputStream wrapForOutput(ByteBufferOutputStream buffer, byte messageVersion) {
89               return SnappyFactory.wrapForOutput(buffer);
90           }
91   
92           @Override
93           public InputStream wrapForInput(ByteBuffer buffer, byte messageVersion, BufferSupplier decompressionBufferSupplier) {
94               return SnappyFactory.wrapForInput(buffer);
95           }
96       },
97   
98       LZ4(3, "lz4", 1.0f) {
99           @Override
100           public OutputStream wrapForOutput(ByteBufferOutputStream buffer, byte messageVersion) {
101               try {
102                   return new KafkaLZ4BlockOutputStream(buffer, messageVersion == RecordBatch.MAGIC_VALUE_V0);
103               } catch (Throwable e) {
104                   throw new KafkaException(e);
105               }
106           }
107   
108           @Override
109           public InputStream wrapForInput(ByteBuffer inputBuffer, byte messageVersion, BufferSupplier decompressionBufferSupplier) {
110               try {
111                   return new KafkaLZ4BlockInputStream(inputBuffer, decompressionBufferSupplier,
112                                                       messageVersion == RecordBatch.MAGIC_VALUE_V0);
113               } catch (Throwable e) {
114                   throw new KafkaException(e);
115               }
116           }
117       },
118   
119       ZSTD(4, "zstd", 1.0f) {
120           @Override
121           public OutputStream wrapForOutput(ByteBufferOutputStream buffer, byte messageVersion) {
122               return ZstdFactory.wrapForOutput(buffer);
123           }
124   
125           @Override
126           public InputStream wrapForInput(ByteBuffer buffer, byte messageVersion, BufferSupplier decompressionBufferSupplier) {
127               return ZstdFactory.wrapForInput(buffer, messageVersion, decompressionBufferSupplier);
128           }
129       };

131       public final int id;
132       public final String name;
133       public final float rate;
134   
135       CompressionType(int id, String name, float rate) {
136           this.id = id;
137           this.name = name;
138           this.rate = rate;
139       }
140   
141       /**
142        * Wrap bufferStream with an OutputStream that will compress data with this CompressionType.
143        *
144        * Note: Unlike {@link #wrapForInput}, {@link #wrapForOutput} cannot take {@link ByteBuffer}s directly.
145        * Currently, {@link MemoryRecordsBuilder#writeDefaultBatchHeader()} and {@link MemoryRecordsBuilder#writeLegacyCompressedWrapperHeader()}
146        * write to the underlying buffer in the given {@link ByteBufferOutputStream} after the compressed data has been written.
147        * In the event that the buffer needs to be expanded while writing the data, access to the underlying buffer needs to be preserved.
148        */
149       public abstract OutputStream wrapForOutput(ByteBufferOutputStream bufferStream, byte messageVersion);
150   
151       /**
152        * Wrap buffer with an InputStream that will decompress data with this CompressionType.
153        *
154        * @param decompressionBufferSupplier The supplier of ByteBuffer(s) used for decompression if supported.
155        *                                    For small record batches, allocating a potentially large buffer (64 KB for LZ4)
156        *                                    will dominate the cost of decompressing and iterating over the records in the
157        *                                    batch. As such, a supplier that reuses buffers will have a significant
158        *                                    performance impact.
159        */
160       public abstract InputStream wrapForInput(ByteBuffer buffer, byte messageVersion, BufferSupplier decompressionBufferSupplier);

162       public static CompressionType forId(int id) {
163           switch (id) {
164               case 0:
165                   return NONE;
166               case 1:
167                   return GZIP;
168               case 2:
169                   return SNAPPY;
170               case 3:
171                   return LZ4;
172               case 4:
173                   return ZSTD;
174               default:
175                   throw new IllegalArgumentException("Unknown compression type id: " + id);
176           }
177       }
178   
179       public static CompressionType forName(String name) {
180           if (NONE.name.equals(name))
181               return NONE;
182           else if (GZIP.name.equals(name))
183               return GZIP;
184           else if (SNAPPY.name.equals(name))
185               return SNAPPY;
186           else if (LZ4.name.equals(name))
187               return LZ4;
188           else if (ZSTD.name.equals(name))
189               return ZSTD;
190           else
191               throw new IllegalArgumentException("Unknown compression name: " + name);
192       }
193   
194       @Override
195       public String toString() {
196           return name;
197       }
