17   package org.apache.kafka.common.metrics.stats;
18   
19   import java.util.Arrays;
20   
21   public class Histogram {
22   
23       private final BinScheme binScheme;
24       private final float[] hist;
25       private double count;
26   
27       public Histogram(BinScheme binScheme) {
28           this.hist = new float[binScheme.bins()];
29           this.count = 0.0f;
30           this.binScheme = binScheme;
31       }
32   
33       public void record(double value) {
34           this.hist[binScheme.toBin(value)] += 1.0f;
35           this.count += 1.0d;
36       }
37   
38       public double value(double quantile) {
39           if (count == 0.0d)
40               return Double.NaN;
41           if (quantile > 1.00d)
42               return Float.POSITIVE_INFINITY;
43           if (quantile < 0.00d)
44               return Float.NEGATIVE_INFINITY;
45           float sum = 0.0f;
46           float quant = (float) quantile;
47           for (int i = 0; i < this.hist.length - 1; i++) {
48               sum += this.hist[i];
49               if (sum / count > quant)
50                   return binScheme.fromBin(i);
51           }
52           return binScheme.fromBin(this.hist.length - 1);
53       }
54   
55       public float[] counts() {
56           return this.hist;
57       }
58   
59       public void clear() {
60           Arrays.fill(this.hist, 0.0f);
61           this.count = 0;
62       }
63   
64       @Override
65       public String toString() {
66           StringBuilder b = new StringBuilder("{");
67           for (int i = 0; i < this.hist.length - 1; i++) {
68               b.append(String.format("%.10f", binScheme.fromBin(i)));
69               b.append(':');
70               b.append(String.format("%.0f", this.hist[i]));
71               b.append(',');
72           }
73           b.append(Float.POSITIVE_INFINITY);
74           b.append(':');
75           b.append(String.format("%.0f", this.hist[this.hist.length - 1]));
76           b.append('}');
77           return b.toString();
78       }
79   
80       /**
81        * An algorithm for determining the bin in which a value is to be placed as well as calculating the upper end
82        * of each bin.
83        */
84       public interface BinScheme {
85   
86           /**
87            * Get the number of bins.
88            *
89            * @return the number of bins
90            */
91           int bins();
92   
93           /**
94            * Determine the 0-based bin number in which the supplied value should be placed.
95            *
96            * @param value the value
97            * @return the 0-based index of the bin
98            */
99           int toBin(double value);
100   
101           /**
102            * Determine the value at the upper range of the specified bin.
103            *
104            * @param bin the 0-based bin number
105            * @return the value at the upper end of the bin; or {@link Float#NEGATIVE_INFINITY negative infinity}
106            * if the bin number is negative or {@link Float#POSITIVE_INFINITY positive infinity} if the 0-based
107            * bin number is greater than or equal to the {@link #bins() number of bins}.
108            */
109           double fromBin(int bin);
110       }
111   
112       /**
113        * A scheme for calculating the bins where the width of each bin is a constant determined by the range of values
114        * and the number of bins.
115        */
116       public static class ConstantBinScheme implements BinScheme {
117           private static final int MIN_BIN_NUMBER = 0;
118           private final double min;
119           private final double max;
120           private final int bins;
121           private final double bucketWidth;
122           private final int maxBinNumber;
123   
124           /**
125            * Create a bin scheme with the specified number of bins that all have the same width.
126            *
127            * @param bins the number of bins; must be at least 2
128            * @param min the minimum value to be counted in the bins
129            * @param max the maximum value to be counted in the bins
130            */
131           public ConstantBinScheme(int bins, double min, double max) {
132               if (bins < 2)
133                   throw new IllegalArgumentException("Must have at least 2 bins.");
134               this.min = min;
135               this.max = max;
136               this.bins = bins;
137               this.bucketWidth = (max - min) / bins;
138               this.maxBinNumber = bins - 1;
139           }
140   
141           public int bins() {
142               return this.bins;
143           }
144   
145           public double fromBin(int b) {
146               if (b < MIN_BIN_NUMBER) {
147                   return Float.NEGATIVE_INFINITY;
148               }
149               if (b > maxBinNumber) {
150                   return Float.POSITIVE_INFINITY;
151               }
152               return min + b * bucketWidth;
153           }
154   
155           public int toBin(double x) {
156               int binNumber = (int) ((x - min) / bucketWidth);
157               if (binNumber < MIN_BIN_NUMBER) {
158                   return MIN_BIN_NUMBER;
159               }
160               return Math.min(binNumber, maxBinNumber);
161           }
162       }
163   
164       /**
165        * A scheme for calculating the bins where the width of each bin is one more than the previous bin, and therefore
166        * the bin widths are increasing at a linear rate. However, the bin widths are scaled such that the specified range
167        * of values will all fit within the bins (e.g., the upper range of the last bin is equal to the maximum value).
168        */
169       public static class LinearBinScheme implements BinScheme {
170           private final int bins;
171           private final double max;
172           private final double scale;
173   
174           /**
175            * Create a linear bin scheme with the specified number of bins and the maximum value to be counted in the bins.
176            *
177            * @param numBins the number of bins; must be at least 2
178            * @param max the maximum value to be counted in the bins
179            */
180           public LinearBinScheme(int numBins, double max) {
181               if (numBins < 2)
182                   throw new IllegalArgumentException("Must have at least 2 bins.");
183               this.bins = numBins;
184               this.max = max;
185               double denom = numBins * (numBins - 1.0) / 2.0;
186               this.scale = max / denom;
187           }
188   
189           public int bins() {
190               return this.bins;
191           }
192   
193           public double fromBin(int b) {
194               if (b > this.bins - 1) {
195                   return Float.POSITIVE_INFINITY;
196               } else if (b < 0.0000d) {
197                   return Float.NEGATIVE_INFINITY;
198               } else {
199                   return this.scale * (b * (b + 1.0)) / 2.0;
200               }
201           }
202   
203           public int toBin(double x) {
204               if (x < 0.0d) {
205                   throw new IllegalArgumentException("Values less than 0.0 not accepted.");
206               } else if (x > this.max) {
207                   return this.bins - 1;
208               } else {
209                   return (int) (-0.5 + 0.5 * Math.sqrt(1.0 + 8.0 * x / this.scale));
210               }
211           }
212       }
