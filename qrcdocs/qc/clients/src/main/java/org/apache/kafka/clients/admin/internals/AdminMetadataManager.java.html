34   /**
35    * Manages the metadata for KafkaAdminClient.
36    *
37    * This class is not thread-safe.  It is only accessed from the AdminClient
38    * service thread (which also uses the NetworkClient).
39    */
40   public class AdminMetadataManager {
41       private final Logger log;
42   
43       /**
44        * The minimum amount of time that we should wait between subsequent
45        * retries, when fetching metadata.
46        */
47       private final long refreshBackoffMs;
48   
49       /**
50        * The minimum amount of time that we should wait before triggering an
51        * automatic metadata refresh.
52        */
53       private final long metadataExpireMs;
54   
55       /**
56        * Used to update the NetworkClient metadata.
57        */
58       private final AdminMetadataUpdater updater;
59   
60       /**
61        * The current metadata state.
62        */
63       private State state = State.QUIESCENT;
64   
65       /**
66        * The time in wall-clock milliseconds when we last updated the metadata.
67        */
68       private long lastMetadataUpdateMs = 0;
69   
70       /**
71        * The time in wall-clock milliseconds when we last attempted to fetch new
72        * metadata.
73        */
74       private long lastMetadataFetchAttemptMs = 0;
75   
76       /**
77        * The current cluster information.
78        */
79       private Cluster cluster = Cluster.empty();
80   
81       /**
82        * If we got an authorization exception when we last attempted to fetch
83        * metadata, this is it; null, otherwise.
84        */
85       private AuthenticationException authException = null;
86   
87       public class AdminMetadataUpdater implements MetadataUpdater {
88           @Override
89           public List<Node> fetchNodes() {
90               return cluster.nodes();
91           }
92   
93           @Override
94           public boolean isUpdateDue(long now) {
95               return false;
96           }
97   
98           @Override
99           public long maybeUpdate(long now) {
100               return Long.MAX_VALUE;
101           }
102   
103           @Override
104           public void handleServerDisconnect(long now, String destinationId, Optional<AuthenticationException> maybeFatalException) {
105               maybeFatalException.ifPresent(AdminMetadataManager.this::updateFailed);
106               AdminMetadataManager.this.requestUpdate();
107           }
108   
109           @Override
110           public void handleFailedRequest(long now, Optional<KafkaException> maybeFatalException) {
111               // Do nothing
112           }
113   
114           @Override
115           public void handleSuccessfulResponse(RequestHeader requestHeader, long now, MetadataResponse metadataResponse) {
116               // Do nothing
117           }
118   
119           @Override
120           public void close() {
121           }
122       }
123   
124       /**
125        * The current AdminMetadataManager state.
126        */
127       enum State {
128           QUIESCENT,
129           UPDATE_REQUESTED,
130           UPDATE_PENDING
131       }
132   
133       public AdminMetadataManager(LogContext logContext, long refreshBackoffMs, long metadataExpireMs) {
134           this.log = logContext.logger(AdminMetadataManager.class);
135           this.refreshBackoffMs = refreshBackoffMs;
136           this.metadataExpireMs = metadataExpireMs;
137           this.updater = new AdminMetadataUpdater();
138       }
139   
140       public AdminMetadataUpdater updater() {
141           return updater;
142       }
143   
144       public boolean isReady() {
145           if (authException != null) {
146               log.debug("Metadata is not usable: failed to get metadata.", authException);
147               throw authException;
148           }
149           if (cluster.nodes().isEmpty()) {
150               log.trace("Metadata is not ready: bootstrap nodes have not been " +
151                   "initialized yet.");
152               return false;
153           }
154           if (cluster.isBootstrapConfigured()) {
155               log.trace("Metadata is not ready: we have not fetched metadata from " +
156                   "the bootstrap nodes yet.");
157               return false;
158           }
159           log.trace("Metadata is ready to use.");
160           return true;
161       }
162   
163       public Node controller() {
164           return cluster.controller();
165       }
166   
167       public Node nodeById(int nodeId) {
168           return cluster.nodeById(nodeId);
169       }
170   
171       public void requestUpdate() {
172           if (state == State.QUIESCENT) {
173               state = State.UPDATE_REQUESTED;
174               log.debug("Requesting metadata update.");
175           }
176       }
177   
178       public void clearController() {
179           if (cluster.controller() != null) {
180               log.trace("Clearing cached controller node {}.", cluster.controller());
181               this.cluster = new Cluster(cluster.clusterResource().clusterId(),
182                   cluster.nodes(),
183                   Collections.emptySet(),
184                   Collections.emptySet(),
185                   Collections.emptySet(),
186                   null);
187           }
188       }
189   
190       /**
191        * Determine if the AdminClient should fetch new metadata.
192        */
193       public long metadataFetchDelayMs(long now) {
194           switch (state) {
195               case QUIESCENT:
196                   // Calculate the time remaining until the next periodic update.
197                   // We want to avoid making many metadata requests in a short amount of time,
198                   // so there is a metadata refresh backoff period.
199                   return Math.max(delayBeforeNextAttemptMs(now), delayBeforeNextExpireMs(now));
200               case UPDATE_REQUESTED:
201                   // Respect the backoff, even if an update has been requested
202                   return delayBeforeNextAttemptMs(now);
203               default:
204                   // An update is already pending, so we don't need to initiate another one.
205                   return Long.MAX_VALUE;
206           }
207       }
208   
209       private long delayBeforeNextExpireMs(long now) {
210           long timeSinceUpdate = now - lastMetadataUpdateMs;
211           return Math.max(0, metadataExpireMs - timeSinceUpdate);
212       }
213   
214       private long delayBeforeNextAttemptMs(long now) {
215           long timeSinceAttempt = now - lastMetadataFetchAttemptMs;
216           return Math.max(0, refreshBackoffMs - timeSinceAttempt);
217       }
218   
219       /**
220        * Transition into the UPDATE_PENDING state.  Updates lastMetadataFetchAttemptMs.
221        */
222       public void transitionToUpdatePending(long now) {
223           this.state = State.UPDATE_PENDING;
224           this.lastMetadataFetchAttemptMs = now;
225       }
226   
227       public void updateFailed(Throwable exception) {
228           // We depend on pending calls to request another metadata update
229           this.state = State.QUIESCENT;
230   
231           if (exception instanceof AuthenticationException) {
232               log.warn("Metadata update failed due to authentication error", exception);
233               this.authException = (AuthenticationException) exception;
234           } else {
235               log.info("Metadata update failed", exception);
236           }
237       }
238   
239       /**
240        * Receive new metadata, and transition into the QUIESCENT state.
241        * Updates lastMetadataUpdateMs, cluster, and authException.
242        */
243       public void update(Cluster cluster, long now) {
244           if (cluster.isBootstrapConfigured()) {
245               log.debug("Setting bootstrap cluster metadata {}.", cluster);
246           } else {
247               log.debug("Updating cluster metadata to {}", cluster);
248               this.lastMetadataUpdateMs = now;
249           }
250   
251           this.state = State.QUIESCENT;
252           this.authException = null;
253   
254           if (!cluster.nodes().isEmpty()) {
255               this.cluster = cluster;
256           }
257       }
258   }
