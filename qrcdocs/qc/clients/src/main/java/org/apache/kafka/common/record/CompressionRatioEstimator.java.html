18   package org.apache.kafka.common.record;
19   
20   import java.util.concurrent.ConcurrentHashMap;
21   import java.util.concurrent.ConcurrentMap;
22   
23   
24   /**
25    * This class help estimate the compression ratio for each topic and compression type combination.
26    */
27   public class CompressionRatioEstimator {
28       // The constant speed to increase compression ratio when a batch compresses better than expected.
29       public static final float COMPRESSION_RATIO_IMPROVING_STEP = 0.005f;
30       // The minimum speed to decrease compression ratio when a batch compresses worse than expected.
31       public static final float COMPRESSION_RATIO_DETERIORATE_STEP = 0.05f;
32       private static final ConcurrentMap<String, float[]> COMPRESSION_RATIO = new ConcurrentHashMap<>();
33   
34       /**
35        * Update the compression ratio estimation for a topic and compression type.
36        *
37        * @param topic         the topic to update compression ratio estimation.
38        * @param type          the compression type.
39        * @param observedRatio the observed compression ratio.
40        * @return the compression ratio estimation after the update.
41        */
42       public static float updateEstimation(String topic, CompressionType type, float observedRatio) {
43           float[] compressionRatioForTopic = getAndCreateEstimationIfAbsent(topic);
44           float currentEstimation = compressionRatioForTopic[type.id];
45           synchronized (compressionRatioForTopic) {
46               if (observedRatio > currentEstimation)
47                   compressionRatioForTopic[type.id] = Math.max(currentEstimation + COMPRESSION_RATIO_DETERIORATE_STEP, observedRatio);
48               else if (observedRatio < currentEstimation) {
49                   compressionRatioForTopic[type.id] = Math.max(currentEstimation - COMPRESSION_RATIO_IMPROVING_STEP, observedRatio);
50               }
51           }
52           return compressionRatioForTopic[type.id];
53       }
54   
55       /**
56        * Get the compression ratio estimation for a topic and compression type.
57        */
58       public static float estimation(String topic, CompressionType type) {
59           float[] compressionRatioForTopic = getAndCreateEstimationIfAbsent(topic);
60           return compressionRatioForTopic[type.id];
61       }
62   
63       /**
64        * Reset the compression ratio estimation to the initial values for a topic.
65        */
66       public static void resetEstimation(String topic) {
67           float[] compressionRatioForTopic = getAndCreateEstimationIfAbsent(topic);
68           synchronized (compressionRatioForTopic) {
69               for (CompressionType type : CompressionType.values()) {
70                   compressionRatioForTopic[type.id] = type.rate;
71               }
72           }
73       }
74   
75       /**
76        * Remove the compression ratio estimation for a topic.
77        */
78       public static void removeEstimation(String topic) {
79           COMPRESSION_RATIO.remove(topic);
80       }
81   
82       /**
83        * Set the compression estimation for a topic compression type combination. This method is for unit test purpose.
84        */
85       public static void setEstimation(String topic, CompressionType type, float ratio) {
86           float[] compressionRatioForTopic = getAndCreateEstimationIfAbsent(topic);
87           synchronized (compressionRatioForTopic) {
88               compressionRatioForTopic[type.id] = ratio;
89           }
90       }
91   
92       private static float[] getAndCreateEstimationIfAbsent(String topic) {
93           float[] compressionRatioForTopic = COMPRESSION_RATIO.get(topic);
94           if (compressionRatioForTopic == null) {
95               compressionRatioForTopic = initialCompressionRatio();
96               float[] existingCompressionRatio = COMPRESSION_RATIO.putIfAbsent(topic, compressionRatioForTopic);
97               // Someone created the compression ratio array before us, use it.
98               if (existingCompressionRatio != null)
99                   return existingCompressionRatio;
100           }
101           return compressionRatioForTopic;
102       }
103   
104       private static float[] initialCompressionRatio() {
105           float[] compressionRatio = new float[CompressionType.values().length];
106           for (CompressionType type : CompressionType.values()) {
107               compressionRatio[type.id] = type.rate;
108           }
109           return compressionRatio;
110       }
111   }
