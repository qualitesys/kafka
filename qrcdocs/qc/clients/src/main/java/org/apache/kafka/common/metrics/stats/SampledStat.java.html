17   package org.apache.kafka.common.metrics.stats;
18   
19   import java.util.ArrayList;
20   import java.util.List;
21   
22   import org.apache.kafka.common.metrics.MeasurableStat;
23   import org.apache.kafka.common.metrics.MetricConfig;
24   
25   /**
26    * A SampledStat records a single scalar value measured over one or more samples. Each sample is recorded over a
27    * configurable window. The window can be defined by number of events or elapsed time (or both, if both are given the
28    * window is complete when <i>either</i> the event count or elapsed time criterion is met).
29    * <p>
30    * All the samples are combined to produce the measurement. When a window is complete the oldest sample is cleared and
31    * recycled to begin recording the next sample.
32    * 
33    * Subclasses of this class define different statistics measured using this basic pattern.
34    */
35   public abstract class SampledStat implements MeasurableStat {
36   
37       private double initialValue;
38       private int current = 0;
39       protected List<Sample> samples;
40   
41       public SampledStat(double initialValue) {
42           this.initialValue = initialValue;
43           this.samples = new ArrayList<>(2);
44       }
45   
46       @Override
47       public void record(MetricConfig config, double value, long timeMs) {
48           Sample sample = current(timeMs);
49           if (sample.isComplete(timeMs, config))
50               sample = advance(config, timeMs);
51           update(sample, config, value, timeMs);
52           sample.eventCount += 1;
53       }
54   
55       private Sample advance(MetricConfig config, long timeMs) {
56           this.current = (this.current + 1) % config.samples();
57           if (this.current >= samples.size()) {
58               Sample sample = newSample(timeMs);
59               this.samples.add(sample);
60               return sample;
61           } else {
62               Sample sample = current(timeMs);
63               sample.reset(timeMs);
64               return sample;
65           }
66       }
67   
68       protected Sample newSample(long timeMs) {
69           return new Sample(this.initialValue, timeMs);
70       }
71   
72       @Override
73       public double measure(MetricConfig config, long now) {
74           purgeObsoleteSamples(config, now);
75           return combine(this.samples, config, now);
76       }
77   
78       public Sample current(long timeMs) {
79           if (samples.size() == 0)
80               this.samples.add(newSample(timeMs));
81           return this.samples.get(this.current);
82       }
83   
84       public Sample oldest(long now) {
85           if (samples.size() == 0)
86               this.samples.add(newSample(now));
87           Sample oldest = this.samples.get(0);
88           for (int i = 1; i < this.samples.size(); i++) {
89               Sample curr = this.samples.get(i);
90               if (curr.lastWindowMs < oldest.lastWindowMs)
91                   oldest = curr;
92           }
93           return oldest;
94       }
95   
96       @Override
97       public String toString() {
98           return "SampledStat(" +
99               "initialValue=" + initialValue +
100               ", current=" + current +
101               ", samples=" + samples +
102               ')';
103       }
104   
105       protected abstract void update(Sample sample, MetricConfig config, double value, long timeMs);
106   
107       public abstract double combine(List<Sample> samples, MetricConfig config, long now);
108   
109       /* Timeout any windows that have expired in the absence of any events */
110       protected void purgeObsoleteSamples(MetricConfig config, long now) {
111           long expireAge = config.samples() * config.timeWindowMs();
112           for (Sample sample : samples) {
113               if (now - sample.lastWindowMs >= expireAge)
114                   sample.reset(now);
115           }
116       }
117   
118       protected static class Sample {
119           public double initialValue;
120           public long eventCount;
121           public long lastWindowMs;
122           public double value;
123   
124           public Sample(double initialValue, long now) {
125               this.initialValue = initialValue;
126               this.eventCount = 0;
127               this.lastWindowMs = now;
128               this.value = initialValue;
129           }
130   
131           public void reset(long now) {
132               this.eventCount = 0;
133               this.lastWindowMs = now;
134               this.value = initialValue;
135           }
136   
137           public boolean isComplete(long timeMs, MetricConfig config) {
138               return timeMs - lastWindowMs >= config.timeWindowMs() || eventCount >= config.eventWindow();
139           }
140   
141           @Override
142           public String toString() {
143               return "Sample(" +
144                   "value=" + value +
145                   ", eventCount=" + eventCount +
146                   ", lastWindowMs=" + lastWindowMs +
147                   ", initialValue=" + initialValue +
148                   ')';
149           }
150       }
151   
152   }
