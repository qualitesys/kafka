18   package org.apache.kafka.server.authorizer;
19   
20   import java.util.Collection;
21   import org.apache.kafka.common.ClusterResource;
22   import org.apache.kafka.common.Endpoint;
23   import org.apache.kafka.common.annotation.InterfaceStability;
24   
25   /**
26    * Runtime broker configuration metadata provided to authorizers during start up.
27    */
28   @InterfaceStability.Evolving
29   public interface AuthorizerServerInfo {
30   
31       /**
32        * Returns cluster metadata for the broker running this authorizer including cluster id.
33        */
34       ClusterResource clusterResource();
35   
36       /**
37        * Returns broker id. This may be a generated broker id if `broker.id` was not configured.
38        */
39       int brokerId();
40   
41       /**
42        * Returns endpoints for all listeners including the advertised host and port to which
43        * the listener is bound.
44        */
45       Collection<Endpoint> endpoints();
46   
47       /**
48        * Returns the inter-broker endpoint. This is one of the endpoints returned by {@link #endpoints()}.
49        */
50       Endpoint interBrokerEndpoint();
51   
52       /**
53        * Returns the configured early start listeners.
54        */
55       Collection<String> earlyStartListeners();
56   }
