17   package org.apache.kafka.common.memory;
18   
19   import java.nio.ByteBuffer;
20   import java.util.concurrent.atomic.AtomicLong;
21   
22   import org.apache.kafka.common.metrics.Sensor;
23   import org.apache.kafka.common.utils.Utils;
24   import org.slf4j.Logger;
25   import org.slf4j.LoggerFactory;
26   
27   
28   /**
29    * a simple pool implementation. this implementation just provides a limit on the total outstanding memory.
30    * any buffer allocated must be release()ed always otherwise memory is not marked as reclaimed (and "leak"s)
31    */
32   public class SimpleMemoryPool implements MemoryPool {
33       protected final Logger log = LoggerFactory.getLogger(getClass()); //subclass-friendly
34   
35       protected final long sizeBytes;
36       protected final boolean strict;
37       protected final AtomicLong availableMemory;
38       protected final int maxSingleAllocationSize;
39       protected final AtomicLong startOfNoMemPeriod = new AtomicLong(); //nanoseconds
40       protected volatile Sensor oomTimeSensor;
41   
42       public SimpleMemoryPool(long sizeInBytes, int maxSingleAllocationBytes, boolean strict, Sensor oomPeriodSensor) {
43           if (sizeInBytes <= 0 || maxSingleAllocationBytes <= 0 || maxSingleAllocationBytes > sizeInBytes)
44               throw new IllegalArgumentException("must provide a positive size and max single allocation size smaller than size."
45                   + "provided " + sizeInBytes + " and " + maxSingleAllocationBytes + " respectively");
46           this.sizeBytes = sizeInBytes;
47           this.strict = strict;
48           this.availableMemory = new AtomicLong(sizeInBytes);
49           this.maxSingleAllocationSize = maxSingleAllocationBytes;
50           this.oomTimeSensor = oomPeriodSensor;
51       }
52   
53       @Override
54       public ByteBuffer tryAllocate(int sizeBytes) {
55           if (sizeBytes < 1)
56               throw new IllegalArgumentException("requested size " + sizeBytes + "<=0");
57           if (sizeBytes > maxSingleAllocationSize)
58               throw new IllegalArgumentException("requested size " + sizeBytes + " is larger than maxSingleAllocationSize " + maxSingleAllocationSize);
59   
60           long available;
61           boolean success = false;
62           //in strict mode we will only allocate memory if we have at least the size required.
63           //in non-strict mode we will allocate memory if we have _any_ memory available (so available memory
64           //can dip into the negative and max allocated memory would be sizeBytes + maxSingleAllocationSize)
65           long threshold = strict ? sizeBytes : 1;
66           while ((available = availableMemory.get()) >= threshold) {
67               success = availableMemory.compareAndSet(available, available - sizeBytes);
68               if (success)
69                   break;
70           }
71   
72           if (success) {
73               maybeRecordEndOfDrySpell();
74           } else {
75               if (oomTimeSensor != null) {
76                   startOfNoMemPeriod.compareAndSet(0, System.nanoTime());
77               }
78               log.trace("refused to allocate buffer of size {}", sizeBytes);
79               return null;
80           }
81   
82           ByteBuffer allocated = ByteBuffer.allocate(sizeBytes);
83           bufferToBeReturned(allocated);
84           return allocated;
85       }
86   
87       @Override
88       public void release(ByteBuffer previouslyAllocated) {
89           if (previouslyAllocated == null)
90               throw new IllegalArgumentException("provided null buffer");
91   
92           bufferToBeReleased(previouslyAllocated);
93           availableMemory.addAndGet(previouslyAllocated.capacity());
94           maybeRecordEndOfDrySpell();
95       }
96   
97       @Override
98       public long size() {
99           return sizeBytes;
100       }
101   
102       @Override
103       public long availableMemory() {
104           return availableMemory.get();
105       }
106   
107       @Override
108       public boolean isOutOfMemory() {
109           return availableMemory.get() <= 0;
110       }
111   
112       //allows subclasses to do their own bookkeeping (and validation) _before_ memory is returned to client code.
113       protected void bufferToBeReturned(ByteBuffer justAllocated) {
114           log.trace("allocated buffer of size {} ", justAllocated.capacity());
115       }
116   
117       //allows subclasses to do their own bookkeeping (and validation) _before_ memory is marked as reclaimed.
118       protected void bufferToBeReleased(ByteBuffer justReleased) {
119           log.trace("released buffer of size {}", justReleased.capacity());
120       }
121   
122       @Override
123       public String toString() {
124           long allocated = sizeBytes - availableMemory.get();
125           return "SimpleMemoryPool{" + Utils.formatBytes(allocated) + "/" + Utils.formatBytes(sizeBytes) + " used}";
126       }
127   
128       protected void maybeRecordEndOfDrySpell() {
129           if (oomTimeSensor != null) {
130               long startOfDrySpell = startOfNoMemPeriod.getAndSet(0);
131               if (startOfDrySpell != 0) {
132                   //how long were we refusing allocation requests for
133                   oomTimeSensor.record((System.nanoTime() - startOfDrySpell) / 1000000.0); //fractional (double) millis
134               }
135           }
136       }
137   }
