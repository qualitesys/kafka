31    *

39       public static final int LZ4_MAX_HEADER_LENGTH = 19;
40       public static final int LZ4_FRAME_INCOMPRESSIBLE_MASK = 0x80000000;

61       /**
62        * Create a new {@link OutputStream} that will compress data using the LZ4 algorithm.
63        *
64        * @param out The output stream to compress
65        * @param blockSize Default: 4. The block size used during compression. 4=64kb, 5=256kb, 6=1mb, 7=4mb. All other
66        *            values will generate an exception
67        * @param blockChecksum Default: false. When true, a XXHash32 checksum is computed and appended to the stream for
68        *            every block of data
69        * @param useBrokenFlagDescriptorChecksum Default: false. When true, writes an incorrect FrameDescriptor checksum
70        *            compatible with older kafka clients.
71        * @throws IOException
72        */
73       public KafkaLZ4BlockOutputStream(OutputStream out, int blockSize, boolean blockChecksum, boolean useBrokenFlagDescriptorChecksum) throws IOException {
74           this.out = out;
75           compressor = LZ4Factory.fastestInstance().fastCompressor();
76           checksum = XXHashFactory.fastestInstance().hash32();
77           this.useBrokenFlagDescriptorChecksum = useBrokenFlagDescriptorChecksum;
78           bd = new BD(blockSize);
79           flg = new FLG(blockChecksum);
80           bufferOffset = 0;
81           maxBlockSize = bd.getBlockMaximumSize();
82           buffer = new byte[maxBlockSize];
83           compressedBuffer = new byte[compressor.maxCompressedLength(maxBlockSize)];
84           finished = false;
85           writeHeader();
86       }

92        * @param blockSize Default: 4. The block size used during compression. 4=64kb, 5=256kb, 6=1mb, 7=4mb. All other

94        * @param blockChecksum Default: false. When true, a XXHash32 checksum is computed and appended to the stream for

128       /**
129        * Check whether KafkaLZ4BlockInputStream is configured to write an
130        * incorrect Frame Descriptor checksum, which is useful for
131        * compatibility with old client implementations.
132        */
133       public boolean useBrokenFlagDescriptorChecksum() {
134           return this.useBrokenFlagDescriptorChecksum;
135       }
136   
137       /**
138        * Writes the magic number and frame descriptor to the underlying {@link OutputStream}.
139        *
140        * @throws IOException
141        */
142       private void writeHeader() throws IOException {
143           ByteUtils.writeUnsignedIntLE(buffer, 0, MAGIC);
144           bufferOffset = 4;
145           buffer[bufferOffset++] = flg.toByte();
146           buffer[bufferOffset++] = bd.toByte();
147           // TODO write uncompressed content size, update flg.validate()
148   
149           // compute checksum on all descriptor fields
150           int offset = 4;
151           int len = bufferOffset - offset;
152           if (this.useBrokenFlagDescriptorChecksum) {
153               len += offset;
154               offset = 0;
155           }
156           byte hash = (byte) ((checksum.hash(buffer, offset, len, 0) >> 8) & 0xFF);
157           buffer[bufferOffset++] = hash;
158   
159           // write out frame descriptor
160           out.write(buffer, 0, bufferOffset);
161           bufferOffset = 0;
162       }
163   
164       /**
165        * Compresses buffered data, optionally computes an XXHash32 checksum, and writes the result to the underlying
166        * {@link OutputStream}.
167        *
168        * @throws IOException
169        */
170       private void writeBlock() throws IOException {
171           if (bufferOffset == 0) {
172               return;
173           }
174   
175           int compressedLength = compressor.compress(buffer, 0, bufferOffset, compressedBuffer, 0);
176           byte[] bufferToWrite = compressedBuffer;
177           int compressMethod = 0;
178   
179           // Store block uncompressed if compressed length is greater (incompressible)
180           if (compressedLength >= bufferOffset) {
181               bufferToWrite = buffer;
182               compressedLength = bufferOffset;
183               compressMethod = LZ4_FRAME_INCOMPRESSIBLE_MASK;
184           }
185   
186           // Write content
187           ByteUtils.writeUnsignedIntLE(out, compressedLength | compressMethod);
188           out.write(bufferToWrite, 0, compressedLength);
189   
190           // Calculate and write block checksum
191           if (flg.isBlockChecksumSet()) {
192               int hash = checksum.hash(bufferToWrite, 0, compressedLength, 0);
193               ByteUtils.writeUnsignedIntLE(out, hash);
194           }
195           bufferOffset = 0;
196       }
197   
198       /**
199        * Similar to the {@link #writeBlock()} method. Writes a 0-length block (without block checksum) to signal the end
200        * of the block stream.
201        *
202        * @throws IOException
203        */
204       private void writeEndMark() throws IOException {
205           ByteUtils.writeUnsignedIntLE(out, 0);
206           // TODO implement content checksum, update flg.validate()
207       }

209       @Override
210       public void write(int b) throws IOException {
211           ensureNotFinished();
212           if (bufferOffset == maxBlockSize) {
213               writeBlock();
214           }
215           buffer[bufferOffset++] = (byte) b;
216       }
217   
218       @Override
219       public void write(byte[] b, int off, int len) throws IOException {
220           net.jpountz.util.SafeUtils.checkRange(b, off, len);
221           ensureNotFinished();
222   
223           int bufferRemainingLength = maxBlockSize - bufferOffset;
224           // while b will fill the buffer
225           while (len > bufferRemainingLength) {
226               // fill remaining space in buffer
227               System.arraycopy(b, off, buffer, bufferOffset, bufferRemainingLength);
228               bufferOffset = maxBlockSize;
229               writeBlock();
230               // compute new offset and length
231               off += bufferRemainingLength;
232               len -= bufferRemainingLength;
233               bufferRemainingLength = maxBlockSize;
234           }
235   
236           System.arraycopy(b, off, buffer, bufferOffset, len);
237           bufferOffset += len;
238       }
239   
240       @Override
241       public void flush() throws IOException {
242           if (!finished) {
243               writeBlock();
244           }
245           if (out != null) {
246               out.flush();
247           }
248       }
249   
250       /**
251        * A simple state check to ensure the stream is still open.
252        */
253       private void ensureNotFinished() {
254           if (finished) {
255               throw new IllegalStateException(CLOSED_STREAM);
256           }
257       }
258   
259       @Override
260       public void close() throws IOException {
261           try {
262               if (!finished) {
263                   // basically flush the buffer writing the last block
264                   writeBlock();
265                   // write the end block
266                   writeEndMark();
267               }
268           } finally {
269               try {
270                   if (out != null) {
271                       try (OutputStream outStream = out) {
272                           outStream.flush();
273                       }
274                   }
275               } finally {
276                   out = null;
277                   buffer = null;
278                   compressedBuffer = null;
279                   finished = true;
280               }
281           }
282       }

284       public static class FLG {
285   
286           private static final int VERSION = 1;
287   
288           private final int reserved;

303           private FLG(int reserved,
304                       int contentChecksum,

311               this.contentSize = contentSize;

318           public static FLG fromByte(byte flg) {
319               int reserved = (flg >>> 0) & 3;
320               int contentChecksum = (flg >>> 2) & 1;
321               int contentSize = (flg >>> 3) & 1;
322               int blockChecksum = (flg >>> 4) & 1;
323               int blockIndependence = (flg >>> 5) & 1;
324               int version = (flg >>> 6) & 3;
325   
326               return new FLG(reserved,
327                              contentChecksum,
328                              contentSize,
329                              blockChecksum,
330                              blockIndependence,
331                              version);
332           }
333   
334           public byte toByte() {
335               return (byte) (((reserved & 3) << 0) | ((contentChecksum & 1) << 2)
336                       | ((contentSize & 1) << 3) | ((blockChecksum & 1) << 4) | ((blockIndependence & 1) << 5) | ((version & 3) << 6));
337           }

339           private void validate() {
340               if (reserved != 0) {
341                   throw new RuntimeException("Reserved bits must be 0");
342               }
343               if (blockIndependence != 1) {
344                   throw new RuntimeException("Dependent block stream is unsupported");
345               }
346               if (version != VERSION) {
347                   throw new RuntimeException(String.format("Version %d is unsupported", version));
348               }
349           }

393           public static BD fromByte(byte bd) {
394               int reserved2 = (bd >>> 0) & 15;

401           private void validate() {
402               if (reserved2 != 0) {
403                   throw new RuntimeException("Reserved2 field must be 0");
404               }
405               if (blockSizeValue < 4 || blockSizeValue > 7) {
406                   throw new RuntimeException("Block size value must be between 4 and 7");
407               }
408               if (reserved3 != 0) {
409                   throw new RuntimeException("Reserved3 field must be 0");
410               }
411           }
412   
413           // 2^(2n+8)
414           public int getBlockMaximumSize() {
415               return 1 << ((2 * blockSizeValue) + 8);
416           }
417   
418           public byte toByte() {
419               return (byte) (((reserved2 & 15) << 0) | ((blockSizeValue & 7) << 4) | ((reserved3 & 1) << 7));
420           }
421       }
