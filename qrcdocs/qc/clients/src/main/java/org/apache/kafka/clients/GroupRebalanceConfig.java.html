17   package org.apache.kafka.clients;
18   
19   import org.apache.kafka.common.config.AbstractConfig;
20   import org.apache.kafka.common.requests.JoinGroupRequest;
21   
22   import java.util.Locale;
23   import java.util.Optional;
24   
25   /**
26    * Class to extract group rebalance related configs.
27    */
28   public class GroupRebalanceConfig {
29   
30       public enum ProtocolType {
31           CONSUMER,
32           CONNECT;
33   
34           @Override
35           public String toString() {
36               return super.toString().toLowerCase(Locale.ROOT);
37           }
38       }
39   
40       public final int sessionTimeoutMs;
41       public final int rebalanceTimeoutMs;
42       public final int heartbeatIntervalMs;
43       public final String groupId;
44       public final Optional<String> groupInstanceId;
45       public final long retryBackoffMs;
46       public final boolean leaveGroupOnClose;
47   
48       public GroupRebalanceConfig(AbstractConfig config, ProtocolType protocolType) {
49           this.sessionTimeoutMs = config.getInt(CommonClientConfigs.SESSION_TIMEOUT_MS_CONFIG);
50   
51           // Consumer and Connect use different config names for defining rebalance timeout
52           if (protocolType == ProtocolType.CONSUMER) {
53               this.rebalanceTimeoutMs = config.getInt(CommonClientConfigs.MAX_POLL_INTERVAL_MS_CONFIG);
54           } else {
55               this.rebalanceTimeoutMs = config.getInt(CommonClientConfigs.REBALANCE_TIMEOUT_MS_CONFIG);
56           }
57   
58           this.heartbeatIntervalMs = config.getInt(CommonClientConfigs.HEARTBEAT_INTERVAL_MS_CONFIG);
59           this.groupId = config.getString(CommonClientConfigs.GROUP_ID_CONFIG);
60   
61           // Static membership is only introduced in consumer API.
62           if (protocolType == ProtocolType.CONSUMER) {
63               String groupInstanceId = config.getString(CommonClientConfigs.GROUP_INSTANCE_ID_CONFIG);
64               if (groupInstanceId != null) {
65                   JoinGroupRequest.validateGroupInstanceId(groupInstanceId);
66                   this.groupInstanceId = Optional.of(groupInstanceId);
67               } else {
68                   this.groupInstanceId = Optional.empty();
69               }
70           } else {
71               this.groupInstanceId = Optional.empty();
72           }
73   
74           this.retryBackoffMs = config.getLong(CommonClientConfigs.RETRY_BACKOFF_MS_CONFIG);
75   
76           // Internal leave group config is only defined in Consumer.
77           if (protocolType == ProtocolType.CONSUMER) {
78               this.leaveGroupOnClose = config.getBoolean("internal.leave.group.on.close");
79           } else {
80               this.leaveGroupOnClose = true;
81           }
82       }
83   
84       // For testing purpose.
85       public GroupRebalanceConfig(final int sessionTimeoutMs,
86                                   final int rebalanceTimeoutMs,
87                                   final int heartbeatIntervalMs,
88                                   String groupId,
89                                   Optional<String> groupInstanceId,
90                                   long retryBackoffMs,
91                                   boolean leaveGroupOnClose) {
92           this.sessionTimeoutMs = sessionTimeoutMs;
93           this.rebalanceTimeoutMs = rebalanceTimeoutMs;
94           this.heartbeatIntervalMs = heartbeatIntervalMs;
95           this.groupId = groupId;
96           this.groupInstanceId = groupInstanceId;
97           this.retryBackoffMs = retryBackoffMs;
98           this.leaveGroupOnClose = leaveGroupOnClose;
99       }
100   }
