44   public class AlterConsumerGroupOffsetsHandler extends AdminApiHandler.Batched<CoordinatorKey, Map<TopicPartition, Errors>> {
45   
46       private final CoordinatorKey groupId;
47       private final Map<TopicPartition, OffsetAndMetadata> offsets;
48       private final Logger log;
49       private final AdminApiLookupStrategy<CoordinatorKey> lookupStrategy;
50   
51       public AlterConsumerGroupOffsetsHandler(
52           String groupId,
53           Map<TopicPartition, OffsetAndMetadata> offsets,
54           LogContext logContext
55       ) {
56           this.groupId = CoordinatorKey.byGroupId(groupId);
57           this.offsets = offsets;
58           this.log = logContext.logger(AlterConsumerGroupOffsetsHandler.class);
59           this.lookupStrategy = new CoordinatorStrategy(CoordinatorType.GROUP, logContext);
60       }
61   
62       @Override
63       public String apiName() {
64           return "offsetCommit";
65       }
66   
67       @Override
68       public AdminApiLookupStrategy<CoordinatorKey> lookupStrategy() {
69           return lookupStrategy;
70       }
71   
72       public static AdminApiFuture.SimpleAdminApiFuture<CoordinatorKey, Map<TopicPartition, Errors>> newFuture(
73           String groupId
74       ) {
75           return AdminApiFuture.forKeys(Collections.singleton(CoordinatorKey.byGroupId(groupId)));
76       }
77   
78       private void validateKeys(Set<CoordinatorKey> groupIds) {
79           if (!groupIds.equals(singleton(groupId))) {
80               throw new IllegalArgumentException("Received unexpected group ids " + groupIds +
81                   " (expected only " + singleton(groupId) + ")");
82           }
83       }
84   
85       @Override
86       public OffsetCommitRequest.Builder buildBatchedRequest(
87           int coordinatorId,
88           Set<CoordinatorKey> groupIds
89       ) {
90           validateKeys(groupIds);
91   
92           Map<String, OffsetCommitRequestTopic> offsetData = new HashMap<>();
93           offsets.forEach((topicPartition, offsetAndMetadata) -> {
94               OffsetCommitRequestTopic topic = offsetData.computeIfAbsent(
95                   topicPartition.topic(),
96                   key -> new OffsetCommitRequestTopic().setName(topicPartition.topic())
97               );
98   
99               topic.partitions().add(new OffsetCommitRequestPartition()
100                   .setCommittedOffset(offsetAndMetadata.offset())
101                   .setCommittedLeaderEpoch(offsetAndMetadata.leaderEpoch().orElse(-1))
102                   .setCommittedMetadata(offsetAndMetadata.metadata())
103                   .setPartitionIndex(topicPartition.partition()));
104           });
105   
106           OffsetCommitRequestData data = new OffsetCommitRequestData()
107               .setGroupId(groupId.idValue)
108               .setTopics(new ArrayList<>(offsetData.values()));
109   
110           return new OffsetCommitRequest.Builder(data);
111       }
112   
113       @Override
114       public ApiResult<CoordinatorKey, Map<TopicPartition, Errors>> handleResponse(
115           Node coordinator,
116           Set<CoordinatorKey> groupIds,
117           AbstractResponse abstractResponse
118       ) {
119           validateKeys(groupIds);
120   
121           final OffsetCommitResponse response = (OffsetCommitResponse) abstractResponse;
122           final Set<CoordinatorKey> groupsToUnmap = new HashSet<>();
123           final Set<CoordinatorKey> groupsToRetry = new HashSet<>();
124           final Map<TopicPartition, Errors> partitionResults = new HashMap<>();
125   
126           for (OffsetCommitResponseTopic topic : response.data().topics()) {
127               for (OffsetCommitResponsePartition partition : topic.partitions()) {
128                   TopicPartition topicPartition = new TopicPartition(topic.name(), partition.partitionIndex());
129                   Errors error = Errors.forCode(partition.errorCode());
130   
131                   if (error != Errors.NONE) {
132                       handleError(
133                           groupId,
134                           topicPartition,
135                           error,
136                           partitionResults,
137                           groupsToUnmap,
138                           groupsToRetry
139                       );
140                   } else {
141                       partitionResults.put(topicPartition, error);
142                   }
143               }
144           }
145   
146           if (groupsToUnmap.isEmpty() && groupsToRetry.isEmpty()) {
147               return ApiResult.completed(groupId, partitionResults);
148           } else {
149               return ApiResult.unmapped(new ArrayList<>(groupsToUnmap));
150           }
151       }
152   
153       private void handleError(
154           CoordinatorKey groupId,
155           TopicPartition topicPartition,
156           Errors error,
157           Map<TopicPartition, Errors> partitionResults,
158           Set<CoordinatorKey> groupsToUnmap,
159           Set<CoordinatorKey> groupsToRetry
160       ) {
161           switch (error) {
162               // If the coordinator is in the middle of loading, or rebalance is in progress, then we just need to retry.
163               case COORDINATOR_LOAD_IN_PROGRESS:
164               case REBALANCE_IN_PROGRESS:
165                   log.debug("OffsetCommit request for group id {} returned error {}. Will retry.",
166                       groupId.idValue, error);
167                   groupsToRetry.add(groupId);
168                   break;
169   
170               // If the coordinator is not available, then we unmap and retry.
171               case COORDINATOR_NOT_AVAILABLE:
172               case NOT_COORDINATOR:
173                   log.debug("OffsetCommit request for group id {} returned error {}. Will rediscover the coordinator and retry.",
174                       groupId.idValue, error);
175                   groupsToUnmap.add(groupId);
176                   break;
177   
178               // Group level errors.
179               case INVALID_GROUP_ID:
180               case INVALID_COMMIT_OFFSET_SIZE:
181               case GROUP_AUTHORIZATION_FAILED:
182                   log.debug("OffsetCommit request for group id {} failed due to error {}.",
183                       groupId.idValue, error);
184                   partitionResults.put(topicPartition, error);
185                   break;
186   
187               // TopicPartition level errors.
188               case UNKNOWN_TOPIC_OR_PARTITION:
189               case OFFSET_METADATA_TOO_LARGE:
190               case TOPIC_AUTHORIZATION_FAILED:
191                   log.debug("OffsetCommit request for group id {} and partition {} failed due" +
192                       " to error {}.", groupId.idValue, topicPartition, error);
193                   partitionResults.put(topicPartition, error);
194                   break;
195   
196               // Unexpected errors.
197               default:
198                   log.error("OffsetCommit request for group id {} and partition {} failed due" +
199                       " to unexpected error {}.", groupId.idValue, topicPartition, error);
200                   partitionResults.put(topicPartition, error);
201           }
202       }
203   }
