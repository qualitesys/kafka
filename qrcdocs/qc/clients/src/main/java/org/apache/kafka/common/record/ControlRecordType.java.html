17   package org.apache.kafka.common.record;
18   
19   import org.apache.kafka.common.InvalidRecordException;
20   import org.apache.kafka.common.protocol.types.Field;
21   import org.apache.kafka.common.protocol.types.Schema;
22   import org.apache.kafka.common.protocol.types.Struct;
23   import org.apache.kafka.common.protocol.types.Type;
24   import org.slf4j.Logger;
25   import org.slf4j.LoggerFactory;
26   
27   import java.nio.ByteBuffer;
28   
29   /**
30    * Control records specify a schema for the record key which includes a version and type:
31    *
32    * Key => Version Type
33    *   Version => Int16
34    *   Type => Int16
35    *
36    * In the future, the version can be bumped to indicate a new schema, but it must be backwards compatible
37    * with the current schema. In general, this means we can add new fields, but we cannot remove old ones.
38    *
39    * Note that control records are not considered for compaction by the log cleaner.
40    *
41    * The schema for the value field is left to the control record type to specify.
42    */
43   public enum ControlRecordType {
44       ABORT((short) 0),
45       COMMIT((short) 1),
46   
47       // Raft quorum related control messages.
48       LEADER_CHANGE((short) 2),
49       SNAPSHOT_HEADER((short) 3),
50       SNAPSHOT_FOOTER((short) 4),
51   
52       // UNKNOWN is used to indicate a control type which the client is not aware of and should be ignored
53       UNKNOWN((short) -1);
54   
55       private static final Logger log = LoggerFactory.getLogger(ControlRecordType.class);
56   
57       static final short CURRENT_CONTROL_RECORD_KEY_VERSION = 0;
58       static final int CURRENT_CONTROL_RECORD_KEY_SIZE = 4;
59       private static final Schema CONTROL_RECORD_KEY_SCHEMA_VERSION_V0 = new Schema(
60               new Field("version", Type.INT16),
61               new Field("type", Type.INT16));
62   
63       final short type;
64   
65       ControlRecordType(short type) {
66           this.type = type;
67       }
68   
69       public Struct recordKey() {
70           if (this == UNKNOWN)
71               throw new IllegalArgumentException("Cannot serialize UNKNOWN control record type");
72   
73           Struct struct = new Struct(CONTROL_RECORD_KEY_SCHEMA_VERSION_V0);
74           struct.set("version", CURRENT_CONTROL_RECORD_KEY_VERSION);
75           struct.set("type", type);
76           return struct;
77       }
78   
79       public static short parseTypeId(ByteBuffer key) {
80           if (key.remaining() < CURRENT_CONTROL_RECORD_KEY_SIZE)
81               throw new InvalidRecordException("Invalid value size found for end control record key. Must have " +
82                       "at least " + CURRENT_CONTROL_RECORD_KEY_SIZE + " bytes, but found only " + key.remaining());
83   
84           short version = key.getShort(0);
85           if (version < 0)
86               throw new InvalidRecordException("Invalid version found for control record: " + version +
87                       ". May indicate data corruption");
88   
89           if (version != CURRENT_CONTROL_RECORD_KEY_VERSION)
90               log.debug("Received unknown control record key version {}. Parsing as version {}", version,
91                       CURRENT_CONTROL_RECORD_KEY_VERSION);
92           return key.getShort(2);
93       }
94   
95       public static ControlRecordType fromTypeId(short typeId) {
96           switch (typeId) {
97               case 0:
98                   return ABORT;
99               case 1:
100                   return COMMIT;
101               case 2:
102                   return LEADER_CHANGE;
103               case 3:
104                   return SNAPSHOT_HEADER;
105               case 4:
106                   return SNAPSHOT_FOOTER;
107   
108               default:
109                   return UNKNOWN;
110           }
111       }
112   
113       public static ControlRecordType parse(ByteBuffer key) {
114           return fromTypeId(parseTypeId(key));
115       }
116   }
