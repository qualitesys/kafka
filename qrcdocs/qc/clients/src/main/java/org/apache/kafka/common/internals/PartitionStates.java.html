31   /**
32    * This class is a useful building block for doing fetch requests where topic partitions have to be rotated via
33    * round-robin to ensure fairness and some level of determinism given the existence of a limit on the fetch response
34    * size. Because the serialization of fetch requests is more efficient if all partitions for the same topic are grouped
35    * together, we do such grouping in the method `set`.
36    *
37    * As partitions are moved to the end, the same topic may be repeated more than once. In the optimal case, a single
38    * topic would "wrap around" and appear twice. However, as partitions are fetched in different orders and partition
39    * leadership changes, we will deviate from the optimal. If this turns out to be an issue in practice, we can improve
40    * it by tracking the partitions per node or calling `set` every so often.
41    *
42    * Note that this class is not thread-safe with the exception of {@link #size()} which returns the number of
43    * partitions currently tracked.
44    */
45   public class PartitionStates<S> {
46   
47       private final LinkedHashMap<TopicPartition, S> map = new LinkedHashMap<>();
48       private final Set<TopicPartition> partitionSetView = Collections.unmodifiableSet(map.keySet());
49   
50       /* the number of partitions that are currently assigned available in a thread safe manner */
51       private volatile int size = 0;
52   
53       public PartitionStates() {}
54   
55       public void moveToEnd(TopicPartition topicPartition) {
56           S state = map.remove(topicPartition);
57           if (state != null)
58               map.put(topicPartition, state);
59       }
60   
61       public void updateAndMoveToEnd(TopicPartition topicPartition, S state) {
62           map.remove(topicPartition);
63           map.put(topicPartition, state);
64           updateSize();
65       }
66   
67       public void update(TopicPartition topicPartition, S state) {
68           map.put(topicPartition, state);
69           updateSize();
70       }
71   
72       public void remove(TopicPartition topicPartition) {
73           map.remove(topicPartition);
74           updateSize();
75       }
76   
77       /**
78        * Returns an unmodifiable view of the partitions in random order.
79        * changes to this PartitionStates instance will be reflected in this view.
80        */
81       public Set<TopicPartition> partitionSet() {
82           return partitionSetView;
83       }
84   
85       public void clear() {
86           map.clear();
87           updateSize();
88       }
89   
90       public boolean contains(TopicPartition topicPartition) {
91           return map.containsKey(topicPartition);
92       }
93   
94       public Iterator<S> stateIterator() {
95           return map.values().iterator();
96       }
97   
98       public void forEach(BiConsumer<TopicPartition, S> biConsumer) {
99           map.forEach(biConsumer);
100       }
101   
102       public Map<TopicPartition, S> partitionStateMap() {
103           return Collections.unmodifiableMap(map);
104       }
105   
106       /**
107        * Returns the partition state values in order.
108        */
109       public List<S> partitionStateValues() {
110           return new ArrayList<>(map.values());
111       }
112   
113       public S stateValue(TopicPartition topicPartition) {
114           return map.get(topicPartition);
115       }
116   
117       /**
118        * Get the number of partitions that are currently being tracked. This is thread-safe.
119        */
120       public int size() {
121           return size;
122       }
123   
124       /**
125        * Update the builder to have the received map as its state (i.e. the previous state is cleared). The builder will
126        * "batch by topic", so if we have a, b and c, each with two partitions, we may end up with something like the
127        * following (the order of topics and partitions within topics is dependent on the iteration order of the received
128        * map): a0, a1, b1, b0, c0, c1.
129        */
130       public void set(Map<TopicPartition, S> partitionToState) {
131           map.clear();
132           update(partitionToState);
133           updateSize();
134       }
135   
136       private void updateSize() {
137           size = map.size();
138       }
139   
140       private void update(Map<TopicPartition, S> partitionToState) {
141           LinkedHashMap<String, List<TopicPartition>> topicToPartitions = new LinkedHashMap<>();
142           for (TopicPartition tp : partitionToState.keySet()) {
143               List<TopicPartition> partitions = topicToPartitions.computeIfAbsent(tp.topic(), k -> new ArrayList<>());
144               partitions.add(tp);
145           }
146           for (Map.Entry<String, List<TopicPartition>> entry : topicToPartitions.entrySet()) {
147               for (TopicPartition tp : entry.getValue()) {
148                   S state = partitionToState.get(tp);
149                   map.put(tp, state);
150               }
151           }
152       }
153   
154       public static class PartitionState<S> {
155           private final TopicPartition topicPartition;
156           private final S value;
157   
158           public PartitionState(TopicPartition topicPartition, S state) {
159               this.topicPartition = Objects.requireNonNull(topicPartition);
160               this.value = Objects.requireNonNull(state);
161           }
162   
163           public S value() {
164               return value;
165           }
166   
167           @Override
168           public boolean equals(Object o) {
169               if (this == o)
170                   return true;
171               if (o == null || getClass() != o.getClass())
172                   return false;
173   
174               PartitionState<?> that = (PartitionState<?>) o;
175   
176               return topicPartition.equals(that.topicPartition) && value.equals(that.value);
177           }
178   
179           @Override
180           public int hashCode() {
181               int result = topicPartition.hashCode();
182               result = 31 * result + value.hashCode();
183               return result;
184           }
185   
186           public TopicPartition topicPartition() {
187               return topicPartition;
188           }
189   
190           @Override
191           public String toString() {
192               return "PartitionState(" + topicPartition + "=" + value + ')';
193           }
194       }
