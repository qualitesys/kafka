17   package org.apache.kafka.common.replica;
18   
19   import org.apache.kafka.common.Node;
20   
21   import java.util.Comparator;
22   import java.util.Objects;
23   
24   /**
25    * View of a replica used by {@link ReplicaSelector} to determine a preferred replica.
26    */
27   public interface ReplicaView {
28   
29       /**
30        * The endpoint information for this replica (hostname, port, rack, etc)
31        */
32       Node endpoint();
33   
34       /**
35        * The log end offset for this replica
36        */
37       long logEndOffset();
38   
39       /**
40        * The number of milliseconds (if any) since the last time this replica was caught up to the high watermark.
41        * For a leader replica, this is always zero.
42        */
43       long timeSinceLastCaughtUpMs();
44   
45       /**
46        * Comparator for ReplicaView that returns in the order of "most caught up". This is used for deterministic
47        * selection of a replica when there is a tie from a selector.
48        */
49       static Comparator<ReplicaView> comparator() {
50           return Comparator.comparingLong(ReplicaView::logEndOffset)
51               .thenComparing(Comparator.comparingLong(ReplicaView::timeSinceLastCaughtUpMs).reversed())
52               .thenComparing(replicaInfo -> replicaInfo.endpoint().id());
53       }
54   
55       class DefaultReplicaView implements ReplicaView {
56           private final Node endpoint;
57           private final long logEndOffset;
58           private final long timeSinceLastCaughtUpMs;
59   
60           public DefaultReplicaView(Node endpoint, long logEndOffset, long timeSinceLastCaughtUpMs) {
61               this.endpoint = endpoint;
62               this.logEndOffset = logEndOffset;
63               this.timeSinceLastCaughtUpMs = timeSinceLastCaughtUpMs;
64           }
65   
66           @Override
67           public Node endpoint() {
68               return endpoint;
69           }
70   
71           @Override
72           public long logEndOffset() {
73               return logEndOffset;
74           }
75   
76           @Override
77           public long timeSinceLastCaughtUpMs() {
78               return timeSinceLastCaughtUpMs;
79           }
80   
81           @Override
82           public boolean equals(Object o) {
83               if (this == o) return true;
84               if (o == null || getClass() != o.getClass()) return false;
85               DefaultReplicaView that = (DefaultReplicaView) o;
86               return logEndOffset == that.logEndOffset &&
87                       Objects.equals(endpoint, that.endpoint) &&
88                       Objects.equals(timeSinceLastCaughtUpMs, that.timeSinceLastCaughtUpMs);
89           }
90   
91           @Override
92           public int hashCode() {
93               return Objects.hash(endpoint, logEndOffset, timeSinceLastCaughtUpMs);
94           }
95   
96           @Override
97           public String toString() {
98               return "DefaultReplicaView{" +
99                       "endpoint=" + endpoint +
100                       ", logEndOffset=" + logEndOffset +
101                       ", timeSinceLastCaughtUpMs=" + timeSinceLastCaughtUpMs +
102                       '}';
103           }
104       }
105   }
