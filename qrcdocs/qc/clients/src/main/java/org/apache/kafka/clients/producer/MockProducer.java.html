298       /**
299        * Adds the record to the list of sent records.
300        *
301        * @see #history()
302        */
303       @Override
304       public synchronized Future<RecordMetadata> send(ProducerRecord<K, V> record, Callback callback) {
305           if (this.closed) {
306               throw new IllegalStateException("MockProducer is already closed.");
307           }
308   
309           if (this.producerFenced) {
310               throw new KafkaException("MockProducer is fenced.", new ProducerFencedException("Fenced"));
311           }
312           if (this.sendException != null) {
313               throw this.sendException;
314           }
315   
316           int partition = 0;
317           if (!this.cluster.partitionsForTopic(record.topic()).isEmpty())
318               partition = partition(record, this.cluster);
319           else {
320               //just to throw ClassCastException if serializers are not the proper ones to serialize key/value
321               keySerializer.serialize(record.topic(), record.key());
322               valueSerializer.serialize(record.topic(), record.value());
323           }
324               
325           TopicPartition topicPartition = new TopicPartition(record.topic(), partition);
326           ProduceRequestResult result = new ProduceRequestResult(topicPartition);
327           FutureRecordMetadata future = new FutureRecordMetadata(result, 0, RecordBatch.NO_TIMESTAMP,
328                   0, 0, Time.SYSTEM);
329           long offset = nextOffset(topicPartition);
330           long baseOffset = Math.max(0, offset - Integer.MAX_VALUE);
331           int batchIndex = (int) Math.min(Integer.MAX_VALUE, offset);
332           Completion completion = new Completion(offset, new RecordMetadata(topicPartition, baseOffset, batchIndex,
333                   RecordBatch.NO_TIMESTAMP, 0, 0), result, callback, topicPartition);
334   
335           if (!this.transactionInFlight)
336               this.sent.add(record);
337           else
338               this.uncommittedSends.add(record);
339   
340           if (autoComplete)
341               completion.complete(null);
342           else
343               this.completions.addLast(completion);
344   
345           return future;
346       }

348       /**
349        * Get the next offset for this topic/partition
350        */
351       private long nextOffset(TopicPartition tp) {
352           Long offset = this.offsets.get(tp);
353           if (offset == null) {
354               this.offsets.put(tp, 1L);
355               return 0L;
356           } else {
357               Long next = offset + 1;
358               this.offsets.put(tp, next);
359               return offset;
360           }
361       }

504       /**
505        * computes partition for given record.
506        */
507       private int partition(ProducerRecord<K, V> record, Cluster cluster) {
508           Integer partition = record.partition();
509           String topic = record.topic();
510           if (partition != null) {
511               List<PartitionInfo> partitions = cluster.partitionsForTopic(topic);
512               int numPartitions = partitions.size();
513               // they have given us a partition, use it
514               if (partition < 0 || partition >= numPartitions)
515                   throw new IllegalArgumentException("Invalid partition given with record: " + partition
516                                                      + " is not in the range [0..."
517                                                      + numPartitions
518                                                      + "].");
519               return partition;
520           }
521           byte[] keyBytes = keySerializer.serialize(topic, record.headers(), record.key());
522           byte[] valueBytes = valueSerializer.serialize(topic, record.headers(), record.value());
523           return this.partitioner.partition(topic, record.key(), keyBytes, record.value(), valueBytes, cluster);
524       }
