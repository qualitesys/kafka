52   /**
53    * A {@code CallbackHandler} that recognizes {@link OAuthBearerTokenCallback}
54    * to return an unsecured OAuth 2 bearer token and {@link SaslExtensionsCallback} to return SASL extensions
55    * <p>
56    * Claims and their values on the returned token can be specified using
57    * {@code unsecuredLoginStringClaim_<claimname>},
58    * {@code unsecuredLoginNumberClaim_<claimname>}, and
59    * {@code unsecuredLoginListClaim_<claimname>} options. The first character of
60    * the value is taken as the delimiter for list claims. You may define any claim
61    * name and value except '{@code iat}' and '{@code exp}', both of which are
62    * calculated automatically.
63    * <p>
64    * <p>
65    * You can also add custom unsecured SASL extensions using
66    * {@code unsecuredLoginExtension_<extensionname>}. Extension keys and values are subject to regex validation.
67    * The extension key must also not be equal to the reserved key {@link OAuthBearerClientInitialResponse#AUTH_KEY}
68    * <p>
69    * This implementation also accepts the following options:
70    * <ul>
71    * <li>{@code unsecuredLoginPrincipalClaimName} set to a custom claim name if
72    * you wish the name of the String claim holding the principal name to be
73    * something other than '{@code sub}'.</li>
74    * <li>{@code unsecuredLoginLifetimeSeconds} set to an integer value if the
75    * token expiration is to be set to something other than the default value of
76    * 3600 seconds (which is 1 hour). The '{@code exp}' claim reflects the
77    * expiration time.</li>
78    * <li>{@code unsecuredLoginScopeClaimName} set to a custom claim name if you
79    * wish the name of the String or String List claim holding any token scope to
80    * be something other than '{@code scope}'</li>
81    * </ul>
82    * For example:
83    *
84    * <pre>
85    * KafkaClient {
86    *      org.apache.kafka.common.security.oauthbearer.OAuthBearerLoginModule Required
87    *      unsecuredLoginStringClaim_sub="thePrincipalName"
88    *      unsecuredLoginListClaim_scope="|scopeValue1|scopeValue2"
89    *      unsecuredLoginLifetimeSeconds="60"
90    *      unsecuredLoginExtension_traceId="123";
91    * };
92    * </pre>
93    *
94    * This class is the default when the SASL mechanism is OAUTHBEARER and no value
95    * is explicitly set via either the {@code sasl.login.callback.handler.class}
96    * client configuration property or the
97    * {@code listener.name.sasl_[plaintext|ssl].oauthbearer.sasl.login.callback.handler.class}
98    * broker configuration property.
99    */
100   public class OAuthBearerUnsecuredLoginCallbackHandler implements AuthenticateCallbackHandler {
101       private final Logger log = LoggerFactory.getLogger(OAuthBearerUnsecuredLoginCallbackHandler.class);
102       private static final String OPTION_PREFIX = "unsecuredLogin";
103       private static final String PRINCIPAL_CLAIM_NAME_OPTION = OPTION_PREFIX + "PrincipalClaimName";
104       private static final String LIFETIME_SECONDS_OPTION = OPTION_PREFIX + "LifetimeSeconds";
105       private static final String SCOPE_CLAIM_NAME_OPTION = OPTION_PREFIX + "ScopeClaimName";
106       private static final Set<String> RESERVED_CLAIMS = Collections
107               .unmodifiableSet(new HashSet<>(Arrays.asList("iat", "exp")));
108       private static final String DEFAULT_PRINCIPAL_CLAIM_NAME = "sub";
109       private static final String DEFAULT_LIFETIME_SECONDS_ONE_HOUR = "3600";
110       private static final String DEFAULT_SCOPE_CLAIM_NAME = "scope";
111       private static final String STRING_CLAIM_PREFIX = OPTION_PREFIX + "StringClaim_";
112       private static final String NUMBER_CLAIM_PREFIX = OPTION_PREFIX + "NumberClaim_";
113       private static final String LIST_CLAIM_PREFIX = OPTION_PREFIX + "ListClaim_";
114       private static final String EXTENSION_PREFIX = OPTION_PREFIX + "Extension_";
115       private static final String QUOTE = "\"";
116       private Time time = Time.SYSTEM;
117       private Map<String, String> moduleOptions = null;
118       private boolean configured = false;
119   
120       private static final Pattern DOUBLEQUOTE = Pattern.compile("\"", Pattern.LITERAL);
121   
122       private static final Pattern BACKSLASH = Pattern.compile("\\", Pattern.LITERAL);

143       @SuppressWarnings("unchecked")
144       @Override
145       public void configure(Map<String, ?> configs, String saslMechanism, List<AppConfigurationEntry> jaasConfigEntries) {
146           if (!OAuthBearerLoginModule.OAUTHBEARER_MECHANISM.equals(saslMechanism))
147               throw new IllegalArgumentException(String.format("Unexpected SASL mechanism: %s", saslMechanism));
148           if (Objects.requireNonNull(jaasConfigEntries).size() != 1 || jaasConfigEntries.get(0) == null)
149               throw new IllegalArgumentException(
150                       String.format("Must supply exactly 1 non-null JAAS mechanism configuration (size was %d)",
151                               jaasConfigEntries.size()));
152           this.moduleOptions = Collections.unmodifiableMap((Map<String, String>) jaasConfigEntries.get(0).getOptions());
153           configured = true;
154       }

156       @Override
157       public void handle(Callback[] callbacks) throws IOException, UnsupportedCallbackException {
158           if (!configured())
159               throw new IllegalStateException("Callback handler not configured");
160           for (Callback callback : callbacks) {
161               if (callback instanceof OAuthBearerTokenCallback)
162                   try {
163                       handleTokenCallback((OAuthBearerTokenCallback) callback);
164                   } catch (KafkaException e) {
165                       throw new IOException(e.getMessage(), e);
166                   }
167               else if (callback instanceof SaslExtensionsCallback)
168                   try {
169                       handleExtensionsCallback((SaslExtensionsCallback) callback);
170                   } catch (KafkaException e) {
171                       throw new IOException(e.getMessage(), e);
172                   }
173               else
174                   throw new UnsupportedCallbackException(callback);
175           }
176       }

183       private void handleTokenCallback(OAuthBearerTokenCallback callback) {
184           if (callback.token() != null)
185               throw new IllegalArgumentException("Callback had a token already");
186           if (moduleOptions.isEmpty()) {
187               log.debug("Token not provided, this login cannot be used to establish client connections");
188               callback.token(null);
189               return;
190           }
191           if (moduleOptions.keySet().stream().noneMatch(name -> !name.startsWith(EXTENSION_PREFIX))) {
192               throw new OAuthBearerConfigException("Extensions provided in login context without a token");
193           }
194           String principalClaimNameValue = optionValue(PRINCIPAL_CLAIM_NAME_OPTION);
195           String principalClaimName = Utils.isBlank(principalClaimNameValue) ? DEFAULT_PRINCIPAL_CLAIM_NAME : principalClaimNameValue.trim();
196           String scopeClaimNameValue = optionValue(SCOPE_CLAIM_NAME_OPTION);
197           String scopeClaimName = Utils.isBlank(scopeClaimNameValue) ? DEFAULT_SCOPE_CLAIM_NAME : scopeClaimNameValue.trim();
198           String headerJson = "{" + claimOrHeaderJsonText("alg", "none") + "}";
199           String lifetimeSecondsValueToUse = optionValue(LIFETIME_SECONDS_OPTION, DEFAULT_LIFETIME_SECONDS_ONE_HOUR);
200           String claimsJson;
201           try {
202               claimsJson = String.format("{%s,%s%s}", expClaimText(Long.parseLong(lifetimeSecondsValueToUse)),
203                       claimOrHeaderJsonText("iat", time.milliseconds() / 1000.0),
204                       commaPrependedStringNumberAndListClaimsJsonText());
205           } catch (NumberFormatException e) {
206               throw new OAuthBearerConfigException(e.getMessage());
207           }
208           try {
209               Encoder urlEncoderNoPadding = Base64.getUrlEncoder().withoutPadding();
210               OAuthBearerUnsecuredJws jws = new OAuthBearerUnsecuredJws(
211                       String.format("%s.%s.",
212                               urlEncoderNoPadding.encodeToString(headerJson.getBytes(StandardCharsets.UTF_8)),
213                               urlEncoderNoPadding.encodeToString(claimsJson.getBytes(StandardCharsets.UTF_8))),
214                       principalClaimName, scopeClaimName);
215               log.info("Retrieved token with principal {}", jws.principalName());
216               callback.token(jws);
217           } catch (OAuthBearerIllegalTokenException e) {
218               // occurs if the principal claim doesn't exist or has an empty value
219               throw new OAuthBearerConfigException(e.getMessage(), e);
220           }
221       }
222   
223       /**
224        *  Add and validate all the configured extensions.
225        *  Token keys, apart from passing regex validation, must not be equal to the reserved key {@link OAuthBearerClientInitialResponse#AUTH_KEY}
226        */
227       private void handleExtensionsCallback(SaslExtensionsCallback callback) {
228           Map<String, String> extensions = new HashMap<>();
229           for (Map.Entry<String, String> configEntry : this.moduleOptions.entrySet()) {
230               String key = configEntry.getKey();
231               if (!key.startsWith(EXTENSION_PREFIX))
232                   continue;
233   
234               extensions.put(key.substring(EXTENSION_PREFIX.length()), configEntry.getValue());
235           }
236   
237           SaslExtensions saslExtensions = new SaslExtensions(extensions);
238           try {
239               OAuthBearerClientInitialResponse.validateExtensions(saslExtensions);
240           } catch (SaslException e) {
241               throw new ConfigException(e.getMessage());
242           }
243   
244           callback.extensions(saslExtensions);
245       }

247       private String commaPrependedStringNumberAndListClaimsJsonText() throws OAuthBearerConfigException {
248           StringBuilder sb = new StringBuilder();
249           for (String key : moduleOptions.keySet()) {
250               if (key.startsWith(STRING_CLAIM_PREFIX) && key.length() > STRING_CLAIM_PREFIX.length())
251                   sb.append(',').append(claimOrHeaderJsonText(
252                           confirmNotReservedClaimName(key.substring(STRING_CLAIM_PREFIX.length())), optionValue(key)));
253               else if (key.startsWith(NUMBER_CLAIM_PREFIX) && key.length() > NUMBER_CLAIM_PREFIX.length())
254                   sb.append(',')
255                           .append(claimOrHeaderJsonText(
256                                   confirmNotReservedClaimName(key.substring(NUMBER_CLAIM_PREFIX.length())),
257                                   Double.valueOf(optionValue(key))));
258               else if (key.startsWith(LIST_CLAIM_PREFIX) && key.length() > LIST_CLAIM_PREFIX.length())
259                   sb.append(',')
260                           .append(claimOrHeaderJsonArrayText(
261                                   confirmNotReservedClaimName(key.substring(LIST_CLAIM_PREFIX.length())),
262                                   listJsonText(optionValue(key))));
263           }
264           return sb.toString();
265       }

267       private String confirmNotReservedClaimName(String claimName) throws OAuthBearerConfigException {
268           if (RESERVED_CLAIMS.contains(claimName))
269               throw new OAuthBearerConfigException(String.format("Cannot explicitly set the '%s' claim", claimName));
270           return claimName;
271       }

273       private String listJsonText(String value) {
274           if (value.isEmpty() || value.length() <= 1)
275               return "[]";
276           String delimiter;
277           String unescapedDelimiterChar = value.substring(0, 1);
278           switch (unescapedDelimiterChar) {
279               case "\\":
280               case ".":
281               case "[":
282               case "(":
283               case "{":
284               case "|":
285               case "^":
286               case "$":
287                   delimiter = "\\" + unescapedDelimiterChar;
288                   break;
289               default:
290                   delimiter = unescapedDelimiterChar;
291                   break;
292           }
293           String listText = value.substring(1);
294           String[] elements = listText.split(delimiter);
295           StringBuilder sb = new StringBuilder();
296           for (String element : elements) {
297               sb.append(sb.length() == 0 ? '[' : ',');
298               sb.append('"').append(escape(element)).append('"');
299           }
300           if (listText.startsWith(unescapedDelimiterChar) || listText.endsWith(unescapedDelimiterChar)
301                   || listText.contains(unescapedDelimiterChar + unescapedDelimiterChar))
302               sb.append(",\"\"");
303           return sb.append(']').toString();
304       }

306       private String optionValue(String key) {
307           return optionValue(key, null);
308       }

310       private String optionValue(String key, String defaultValue) {
311           String explicitValue = option(key);
312           return explicitValue != null ? explicitValue : defaultValue;
313       }
314   
315       private String option(String key) {
316           if (!configured)
317               throw new IllegalStateException("Callback handler not configured");
318           return moduleOptions.get(Objects.requireNonNull(key));
319       }

321       private String claimOrHeaderJsonText(String claimName, Number claimValue) {
322           return QUOTE + escape(claimName) + QUOTE + ":" + claimValue;
323       }
324   
325       private String claimOrHeaderJsonText(String claimName, String claimValue) {
326           return QUOTE + escape(claimName) + QUOTE + ":" + QUOTE + escape(claimValue) + QUOTE;
327       }
328   
329       private String claimOrHeaderJsonArrayText(String claimName, String escapedClaimValue) {
330           if (!escapedClaimValue.startsWith("[") || !escapedClaimValue.endsWith("]"))
331               throw new IllegalArgumentException(String.format("Illegal JSON array: %s", escapedClaimValue));
332           return QUOTE + escape(claimName) + QUOTE + ":" + escapedClaimValue;
333       }
334   
335       private String escape(String jsonStringValue) {
336           String replace1 = DOUBLEQUOTE.matcher(jsonStringValue).replaceAll(Matcher.quoteReplacement("\\\""));
337           return BACKSLASH.matcher(replace1).replaceAll(Matcher.quoteReplacement("\\\\"));
338       }
339   
340       private String expClaimText(long lifetimeSeconds) {
341           return claimOrHeaderJsonText("exp", time.milliseconds() / 1000.0 + lifetimeSeconds);
342       }
