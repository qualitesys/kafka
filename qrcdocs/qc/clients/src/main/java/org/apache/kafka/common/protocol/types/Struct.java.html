54        * Return the value of the given pre-validated field, or if the value is missing return the default value.
55        *
56        * @param field The field for which to get the default value
57        * @throws SchemaException if the field has no value and has no default.
58        */
59       private Object getFieldOrDefault(BoundField field) {
60           Object value = this.values[field.index];
61           if (value != null)
62               return value;
63           else if (field.def.hasDefaultValue)
64               return field.def.defaultValue;
65           else if (field.def.type.isNullable())
66               return null;
67           else
68               throw new SchemaException("Missing value for field '" + field.def.name + "' which has no default value.");
69       }

551       @Override
552       public String toString() {
553           StringBuilder b = new StringBuilder();
554           b.append('{');
555           for (int i = 0; i < this.values.length; i++) {
556               BoundField f = this.schema.get(i);
557               b.append(f.def.name);
558               b.append('=');
559               if (f.def.type.isArray() && this.values[i] != null) {
560                   Object[] arrayValue = (Object[]) this.values[i];
561                   b.append('[');
562                   for (int j = 0; j < arrayValue.length; j++) {
563                       b.append(arrayValue[j]);
564                       if (j < arrayValue.length - 1)
565                           b.append(',');
566                   }
567                   b.append(']');
568               } else
