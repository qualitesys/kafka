19   import org.apache.kafka.common.metrics.Sensor;
20   import org.apache.kafka.common.utils.Utils;
21   
22   import java.lang.ref.ReferenceQueue;
23   import java.lang.ref.WeakReference;
24   import java.nio.ByteBuffer;
25   import java.util.Map;
26   import java.util.concurrent.ConcurrentHashMap;
27   
28   
29   /**
30    * An extension of SimpleMemoryPool that tracks allocated buffers and logs an error when they "leak"
31    * (when they are garbage-collected without having been release()ed).
32    * THIS IMPLEMENTATION IS A DEVELOPMENT/DEBUGGING AID AND IS NOT MEANT PRO PRODUCTION USE.
33    */
34   public class GarbageCollectedMemoryPool extends SimpleMemoryPool implements AutoCloseable {
35   
36       private final ReferenceQueue<ByteBuffer> garbageCollectedBuffers = new ReferenceQueue<>();
37       //serves 2 purposes - 1st it maintains the ref objects reachable (which is a requirement for them
38       //to ever be enqueued), 2nd keeps some (small) metadata for every buffer allocated
39       private final Map<BufferReference, BufferMetadata> buffersInFlight = new ConcurrentHashMap<>();
40       private final GarbageCollectionListener gcListener = new GarbageCollectionListener();
41       private final Thread gcListenerThread;
42       private volatile boolean alive = true;
43   
44       public GarbageCollectedMemoryPool(long sizeBytes, int maxSingleAllocationSize, boolean strict, Sensor oomPeriodSensor) {
45           super(sizeBytes, maxSingleAllocationSize, strict, oomPeriodSensor);
46           this.alive = true;
47           this.gcListenerThread = new Thread(gcListener, "memory pool GC listener");
48           this.gcListenerThread.setDaemon(true); //so we dont need to worry about shutdown
49           this.gcListenerThread.start();
50       }
51   
52       @Override
53       protected void bufferToBeReturned(ByteBuffer justAllocated) {
54           BufferReference ref = new BufferReference(justAllocated, garbageCollectedBuffers);
55           BufferMetadata metadata = new BufferMetadata(justAllocated.capacity());
56           if (buffersInFlight.put(ref, metadata) != null)
57               //this is a bug. it means either 2 different co-existing buffers got
58               //the same identity or we failed to register a released/GC'ed buffer
59               throw new IllegalStateException("allocated buffer identity " + ref.hashCode + " already registered as in use?!");
60   
61           log.trace("allocated buffer of size {} and identity {}", sizeBytes, ref.hashCode);
62       }
63   
64       @Override
65       protected void bufferToBeReleased(ByteBuffer justReleased) {
66           BufferReference ref = new BufferReference(justReleased); //used ro lookup only
67           BufferMetadata metadata = buffersInFlight.remove(ref);
68           if (metadata == null)
69               //its impossible for the buffer to have already been GC'ed (because we have a hard ref to it
70               //in the function arg) so this means either a double free or not our buffer.
71               throw new IllegalArgumentException("returned buffer " + ref.hashCode + " was never allocated by this pool");
72           if (metadata.sizeBytes != justReleased.capacity()) {
73               //this is a bug
74               throw new IllegalStateException("buffer " + ref.hashCode + " has capacity " + justReleased.capacity() + " but recorded as " + metadata.sizeBytes);
75           }
76           log.trace("released buffer of size {} and identity {}", metadata.sizeBytes, ref.hashCode);
77       }
78   
79       @Override
80       public void close() {
81           alive = false;
82           gcListenerThread.interrupt();
83       }
84   
85       private class GarbageCollectionListener implements Runnable {
86           @Override
87           public void run() {
88               while (alive) {
89                   try {
90                       BufferReference ref = (BufferReference) garbageCollectedBuffers.remove(); //blocks
91                       ref.clear();
92                       //this cannot race with a release() call because an object is either reachable or not,
93                       //release() can only happen before its GC'ed, and enqueue can only happen after.
94                       //if the ref was enqueued it must then not have been released
95                       BufferMetadata metadata = buffersInFlight.remove(ref);
96   
97                       if (metadata == null) {
98                           //it can happen rarely that the buffer was release()ed properly (so no metadata) and yet
99                           //the reference object to it remains reachable for a short period of time after release()
100                           //and hence gets enqueued. this is because we keep refs in a ConcurrentHashMap which cleans
101                           //up keys lazily.
102                           continue;
103                       }
104   
105                       availableMemory.addAndGet(metadata.sizeBytes);
106                       log.error("Reclaimed buffer of size {} and identity {} that was not properly release()ed. This is a bug.", metadata.sizeBytes, ref.hashCode);
107                   } catch (InterruptedException e) {
108                       log.debug("interrupted", e);
109                       //ignore, we're a daemon thread
110                   }
111               }
112               log.info("GC listener shutting down");
113           }
114       }
115   
116       private static final class BufferMetadata {
117           private final int sizeBytes;
118   
119           private BufferMetadata(int sizeBytes) {
120               this.sizeBytes = sizeBytes;
121           }
122       }
123   
124       private static final class BufferReference extends WeakReference<ByteBuffer> {
125           private final int hashCode;
126   
127           private BufferReference(ByteBuffer referent) { //used for lookup purposes only - no queue required.
128               this(referent, null);
129           }
130   
131           private BufferReference(ByteBuffer referent, ReferenceQueue<? super ByteBuffer> q) {
132               super(referent, q);
133               hashCode = System.identityHashCode(referent);
134           }
135   
136           @Override
137           public boolean equals(Object o) {
138               if (this == o) { //this is important to find leaked buffers (by ref identity)
139                   return true;
140               }
141               if (o == null || getClass() != o.getClass()) {
142                   return false;
143               }
144               BufferReference that = (BufferReference) o;
145               if (hashCode != that.hashCode) {
146                   return false;
147               }
148               ByteBuffer thisBuf = get();
149               if (thisBuf == null) {
150                   //our buffer has already been GC'ed, yet "that" is not us. so not same buffer
151                   return false;
152               }
153               ByteBuffer thatBuf = that.get();
154               return thisBuf == thatBuf;
155           }
156   
157           @Override
158           public int hashCode() {
159               return hashCode;
160           }
161       }
162   
163       @Override
164       public String toString() {
165           long allocated = sizeBytes - availableMemory.get();
166           return "GarbageCollectedMemoryPool{" + Utils.formatBytes(allocated) + "/" + Utils.formatBytes(sizeBytes) + " used in " + buffersInFlight.size() + " buffers}";
167       }
168   }
