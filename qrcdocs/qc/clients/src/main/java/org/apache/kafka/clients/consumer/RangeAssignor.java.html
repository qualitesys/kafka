20   import org.apache.kafka.common.TopicPartition;

23   import java.util.Collections;

28   /**
29    * <p>The range assignor works on a per-topic basis. For each topic, we lay out the available partitions in numeric order
30    * and the consumers in lexicographic order. We then divide the number of partitions by the total number of
31    * consumers to determine the number of partitions to assign to each consumer. If it does not evenly
32    * divide, then the first few consumers will have one extra partition.
33    *
34    * <p>For example, suppose there are two consumers <code>C0</code> and <code>C1</code>, two topics <code>t0</code> and
35    * <code>t1</code>, and each topic has 3 partitions, resulting in partitions <code>t0p0</code>, <code>t0p1</code>,
36    * <code>t0p2</code>, <code>t1p0</code>, <code>t1p1</code>, and <code>t1p2</code>.
37    *
38    * <p>The assignment will be:
39    * <ul>
40    * <li><code>C0: [t0p0, t0p1, t1p0, t1p1]</code></li>
41    * <li><code>C1: [t0p2, t1p2]</code></li>
42    * </ul>
43    *
44    * Since the introduction of static membership, we could leverage <code>group.instance.id</code> to make the assignment behavior more sticky.
45    * For the above example, after one rolling bounce, group coordinator will attempt to assign new <code>member.id</code> towards consumers,
46    * for example <code>C0</code> -&gt; <code>C3</code> <code>C1</code> -&gt; <code>C2</code>.
47    *
48    * <p>The assignment could be completely shuffled to:
49    * <ul>
50    * <li><code>C3 (was C0): [t0p2, t1p2] (before was [t0p0, t0p1, t1p0, t1p1])</code>
51    * <li><code>C2 (was C1): [t0p0, t0p1, t1p0, t1p1] (before was [t0p2, t1p2])</code>
52    * </ul>
53    *
54    * The assignment change was caused by the change of <code>member.id</code> relative order, and
55    * can be avoided by setting the group.instance.id.
56    * Consumers will have individual instance ids <code>I1</code>, <code>I2</code>. As long as
57    * 1. Number of members remain the same across generation
58    * 2. Static members' identities persist across generation
59    * 3. Subscription pattern doesn't change for any member
60    *
61    * <p>The assignment will always be:
62    * <ul>
63    * <li><code>I0: [t0p0, t0p1, t1p0, t1p1]</code>
64    * <li><code>I1: [t0p2, t1p2]</code>
65    * </ul>
66    */
67   public class RangeAssignor extends AbstractPartitionAssignor {
68       public static final String RANGE_ASSIGNOR_NAME = "range";
69   
70       @Override
71       public String name() {
72           return RANGE_ASSIGNOR_NAME;
73       }
74   
75       private Map<String, List<MemberInfo>> consumersPerTopic(Map<String, Subscription> consumerMetadata) {
76           Map<String, List<MemberInfo>> topicToConsumers = new HashMap<>();
77           for (Map.Entry<String, Subscription> subscriptionEntry : consumerMetadata.entrySet()) {
78               String consumerId = subscriptionEntry.getKey();
79               MemberInfo memberInfo = new MemberInfo(consumerId, subscriptionEntry.getValue().groupInstanceId());
80               for (String topic : subscriptionEntry.getValue().topics()) {
81                   put(topicToConsumers, topic, memberInfo);
82               }
83           }
84           return topicToConsumers;
85       }
86   
87       @Override
88       public Map<String, List<TopicPartition>> assign(Map<String, Integer> partitionsPerTopic,
89                                                       Map<String, Subscription> subscriptions) {
90           Map<String, List<MemberInfo>> consumersPerTopic = consumersPerTopic(subscriptions);
91   
92           Map<String, List<TopicPartition>> assignment = new HashMap<>();
93           for (String memberId : subscriptions.keySet())
94               assignment.put(memberId, new ArrayList<>());
95   
96           for (Map.Entry<String, List<MemberInfo>> topicEntry : consumersPerTopic.entrySet()) {
97               String topic = topicEntry.getKey();
98               List<MemberInfo> consumersForTopic = topicEntry.getValue();
99   
100               Integer numPartitionsForTopic = partitionsPerTopic.get(topic);
101               if (numPartitionsForTopic == null)
102                   continue;
103   
104               Collections.sort(consumersForTopic);
105   
106               int numPartitionsPerConsumer = numPartitionsForTopic / consumersForTopic.size();
107               int consumersWithExtraPartition = numPartitionsForTopic % consumersForTopic.size();
108   
109               List<TopicPartition> partitions = AbstractPartitionAssignor.partitions(topic, numPartitionsForTopic);
110               for (int i = 0, n = consumersForTopic.size(); i < n; i++) {
111                   int start = numPartitionsPerConsumer * i + Math.min(i, consumersWithExtraPartition);
112                   int length = numPartitionsPerConsumer + (i + 1 > consumersWithExtraPartition ? 0 : 1);
113                   assignment.get(consumersForTopic.get(i).memberId).addAll(partitions.subList(start, start + length));
114               }
115           }
116           return assignment;
117       }
118   }
