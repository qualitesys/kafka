31   public class JoinGroupRequest extends AbstractRequest {
32   
33       public static class Builder extends AbstractRequest.Builder<JoinGroupRequest> {
34   
35           private final JoinGroupRequestData data;
36   
37           public Builder(JoinGroupRequestData data) {
38               super(ApiKeys.JOIN_GROUP);
39               this.data = data;
40           }
41   
42           @Override
43           public JoinGroupRequest build(short version) {
44               if (data.groupInstanceId() != null && version < 5) {
45                   throw new UnsupportedVersionException("The broker join group protocol version " +
46                           version + " does not support usage of config group.instance.id.");
47               }
48               return new JoinGroupRequest(data, version);
49           }
50   
51           @Override
52           public String toString() {
53               return data.toString();
54           }
55       }
56   
57       private final JoinGroupRequestData data;
58   
59       public static final String UNKNOWN_MEMBER_ID = "";
60       public static final int UNKNOWN_GENERATION_ID = -1;
61       public static final String UNKNOWN_PROTOCOL_NAME = "";
62   
63       /**
64        * Ported from class Topic in {@link org.apache.kafka.common.internals} to restrict the charset for
65        * static member id.
66        */
67       public static void validateGroupInstanceId(String id) {
68           Topic.validate(id, "Group instance id", message -> {
69               throw new InvalidConfigurationException(message);
70           });
71       }
72   
73       public JoinGroupRequest(JoinGroupRequestData data, short version) {
74           super(ApiKeys.JOIN_GROUP, version);
75           this.data = data;
76           maybeOverrideRebalanceTimeout(version);
77       }
78   
79       private void maybeOverrideRebalanceTimeout(short version) {
80           if (version == 0) {
81               // Version 0 has no rebalance timeout, so we use the session timeout
82               // to be consistent with the original behavior of the API.
83               data.setRebalanceTimeoutMs(data.sessionTimeoutMs());
84           }
85       }
86   
87       @Override
88       public JoinGroupRequestData data() {
89           return data;
90       }
91   
92       @Override
93       public AbstractResponse getErrorResponse(int throttleTimeMs, Throwable e) {
94           JoinGroupResponseData data = new JoinGroupResponseData()
95               .setThrottleTimeMs(throttleTimeMs)
96               .setErrorCode(Errors.forException(e).code())
97               .setGenerationId(UNKNOWN_GENERATION_ID)
98               .setProtocolName(UNKNOWN_PROTOCOL_NAME)
99               .setLeader(UNKNOWN_MEMBER_ID)
100               .setMemberId(UNKNOWN_MEMBER_ID)
101               .setMembers(Collections.emptyList());
102   
103           if (version() >= 7)
104               data.setProtocolName(null);
105           else
106               data.setProtocolName(UNKNOWN_PROTOCOL_NAME);
107   
108           return new JoinGroupResponse(data);
109       }
110   
111       public static JoinGroupRequest parse(ByteBuffer buffer, short version) {
112           return new JoinGroupRequest(new JoinGroupRequestData(new ByteBufferAccessor(buffer), version), version);
113       }
