44   /**
45    *
46    * Pluggable authorizer interface for Kafka brokers.
47    *
48    * Startup sequence in brokers:
49    * <ol>
50    *   <li>Broker creates authorizer instance if configured in `authorizer.class.name`.</li>
51    *   <li>Broker configures and starts authorizer instance. Authorizer implementation starts loading its metadata.</li>
52    *   <li>Broker starts SocketServer to accept connections and process requests.</li>
53    *   <li>For each listener, SocketServer waits for authorization metadata to be available in the
54    *       authorizer before accepting connections. The future returned by {@link #start(AuthorizerServerInfo)}
55    *       for each listener must return only when authorizer is ready to authorize requests on the listener.</li>
56    *   <li>Broker accepts connections. For each connection, broker performs authentication and then accepts Kafka requests.
57    *       For each request, broker invokes {@link #authorize(AuthorizableRequestContext, List)} to authorize
58    *       actions performed by the request.</li>
59    * </ol>
60    *
61    * Authorizer implementation class may optionally implement @{@link org.apache.kafka.common.Reconfigurable}
62    * to enable dynamic reconfiguration without restarting the broker.
63    * <p>
64    * <b>Threading model:</b>
65    * <ul>
66    *   <li>All authorizer operations including authorization and ACL updates must be thread-safe.</li>
67    *   <li>ACL update methods are asynchronous. Implementations with low update latency may return a
68    *       completed future using {@link java.util.concurrent.CompletableFuture#completedFuture(Object)}.
69    *       This ensures that the request will be handled synchronously by the caller without using a
70    *       purgatory to wait for the result. If ACL updates require remote communication which may block,
71    *       return a future that is completed asynchronously when the remote operation completes. This enables
72    *       the caller to process other requests on the request threads without blocking.</li>
73    *   <li>Any threads or thread pools used for processing remote operations asynchronously can be started during
74    *       {@link #start(AuthorizerServerInfo)}. These threads must be shutdown during {@link Authorizer#close()}.</li>
75    * </ul>
76    * </p>
77    */
78   @InterfaceStability.Evolving
79   public interface Authorizer extends Configurable, Closeable {
80   
81       /**
82        * Starts loading authorization metadata and returns futures that can be used to wait until
83        * metadata for authorizing requests on each listener is available. Each listener will be
84        * started only after its metadata is available and authorizer is ready to start authorizing
85        * requests on that listener.
86        *
87        * @param serverInfo Metadata for the broker including broker id and listener endpoints
88        * @return CompletionStage for each endpoint that completes when authorizer is ready to
89        *         start authorizing requests on that listener.
90        */
91       Map<Endpoint, ? extends CompletionStage<Void>> start(AuthorizerServerInfo serverInfo);
92   
93       /**
94        * Authorizes the specified action. Additional metadata for the action is specified
95        * in `requestContext`.
96        * <p>
97        * This is a synchronous API designed for use with locally cached ACLs. Since this method is invoked on the
98        * request thread while processing each request, implementations of this method should avoid time-consuming
99        * remote communication that may block request threads.
100        *
101        * @param requestContext Request context including request type, security protocol and listener name
102        * @param actions Actions being authorized including resource and operation for each action
103        * @return List of authorization results for each action in the same order as the provided actions
104        */
105       List<AuthorizationResult> authorize(AuthorizableRequestContext requestContext, List<Action> actions);
106   
107       /**
108        * Creates new ACL bindings.
109        * <p>
110        * This is an asynchronous API that enables the caller to avoid blocking during the update. Implementations of this
111        * API can return completed futures using {@link java.util.concurrent.CompletableFuture#completedFuture(Object)}
112        * to process the update synchronously on the request thread.
113        *
114        * @param requestContext Request context if the ACL is being created by a broker to handle
115        *        a client request to create ACLs. This may be null if ACLs are created directly in ZooKeeper
116        *        using AclCommand.
117        * @param aclBindings ACL bindings to create
118        *
119        * @return Create result for each ACL binding in the same order as in the input list. Each result
120        *         is returned as a CompletionStage that completes when the result is available.
121        */
122       List<? extends CompletionStage<AclCreateResult>> createAcls(AuthorizableRequestContext requestContext, List<AclBinding> aclBindings);
123   
124       /**
125        * Deletes all ACL bindings that match the provided filters.
126        * <p>
127        * This is an asynchronous API that enables the caller to avoid blocking during the update. Implementations of this
128        * API can return completed futures using {@link java.util.concurrent.CompletableFuture#completedFuture(Object)}
129        * to process the update synchronously on the request thread.
130        * <p>
131        * Refer to the authorizer implementation docs for details on concurrent update guarantees.
132        *
133        * @param requestContext Request context if the ACL is being deleted by a broker to handle
134        *        a client request to delete ACLs. This may be null if ACLs are deleted directly in ZooKeeper
135        *        using AclCommand.
136        * @param aclBindingFilters Filters to match ACL bindings that are to be deleted
137        *
138        * @return Delete result for each filter in the same order as in the input list.
139        *         Each result indicates which ACL bindings were actually deleted as well as any
140        *         bindings that matched but could not be deleted. Each result is returned as a
141        *         CompletionStage that completes when the result is available.
142        */
143       List<? extends CompletionStage<AclDeleteResult>> deleteAcls(AuthorizableRequestContext requestContext, List<AclBindingFilter> aclBindingFilters);
144   
145       /**
146        * Returns ACL bindings which match the provided filter.
147        * <p>
148        * This is a synchronous API designed for use with locally cached ACLs. This method is invoked on the request
149        * thread while processing DescribeAcls requests and should avoid time-consuming remote communication that may
150        * block request threads.
151        *
152        * @return Iterator for ACL bindings, which may be populated lazily.
153        */
154       Iterable<AclBinding> acls(AclBindingFilter filter);
155   
156       /**
157        * Get the current number of ACLs, for the purpose of metrics. Authorizers that don't implement this function
158        * will simply return -1.
159        */
160       default int aclCount() {
161           return -1;
162       }
163   
164       /**
165        * Check if the caller is authorized to perform the given ACL operation on at least one
166        * resource of the given type.
167        *
168        * Custom authorizer implementations should consider overriding this default implementation because:
169        * 1. The default implementation iterates all AclBindings multiple times, without any caching
170        *    by principal, host, operation, permission types, and resource types. More efficient
171        *    implementations may be added in custom authorizers that directly access cached entries.
172        * 2. The default implementation cannot integrate with any audit logging included in the
173        *    authorizer implementation.
174        * 3. The default implementation does not support any custom authorizer configs or other access
175        *    rules apart from ACLs.
176        *
177        * @param requestContext Request context including request resourceType, security protocol and listener name
178        * @param op             The ACL operation to check
179        * @param resourceType   The resource type to check
180        * @return               Return {@link AuthorizationResult#ALLOWED} if the caller is authorized
181        *                       to perform the given ACL operation on at least one resource of the
182        *                       given type. Return {@link AuthorizationResult#DENIED} otherwise.
183        */
184       default AuthorizationResult authorizeByResourceType(AuthorizableRequestContext requestContext, AclOperation op, ResourceType resourceType) {
185           SecurityUtils.authorizeByResourceTypeCheckArgs(op, resourceType);
186   
187           // Check a hard-coded name to ensure that super users are granted
188           // access regardless of DENY ACLs.
189           if (authorize(requestContext, Collections.singletonList(new Action(
190                   op, new ResourcePattern(resourceType, "hardcode", PatternType.LITERAL),
191                   0, true, false)))
192                   .get(0) == AuthorizationResult.ALLOWED) {
193               return AuthorizationResult.ALLOWED;
194           }
195   
196           // Filter out all the resource pattern corresponding to the RequestContext,
197           // AclOperation, and ResourceType
198           ResourcePatternFilter resourceTypeFilter = new ResourcePatternFilter(
199               resourceType, null, PatternType.ANY);
200           AclBindingFilter aclFilter = new AclBindingFilter(
201               resourceTypeFilter, AccessControlEntryFilter.ANY);
202   
203           EnumMap<PatternType, Set<String>> denyPatterns =
204               new EnumMap<PatternType, Set<String>>(PatternType.class) {{
205                   put(PatternType.LITERAL, new HashSet<>());
206                   put(PatternType.PREFIXED, new HashSet<>());
207               }};
208           EnumMap<PatternType, Set<String>> allowPatterns =
209               new EnumMap<PatternType, Set<String>>(PatternType.class) {{
210                   put(PatternType.LITERAL, new HashSet<>());
211                   put(PatternType.PREFIXED, new HashSet<>());
212               }};
213   
214           boolean hasWildCardAllow = false;
215   
216           KafkaPrincipal principal = new KafkaPrincipal(
217               requestContext.principal().getPrincipalType(),
218               requestContext.principal().getName());
219           String hostAddr = requestContext.clientAddress().getHostAddress();
220   
221           for (AclBinding binding : acls(aclFilter)) {
222               if (!binding.entry().host().equals(hostAddr) && !binding.entry().host().equals("*"))
223                   continue;
224   
225               if (!SecurityUtils.parseKafkaPrincipal(binding.entry().principal()).equals(principal)
226                       && !binding.entry().principal().equals("User:*"))
227                   continue;
228   
229               if (binding.entry().operation() != op
230                       && binding.entry().operation() != AclOperation.ALL)
231                   continue;
232   
233               if (binding.entry().permissionType() == AclPermissionType.DENY) {
234                   switch (binding.pattern().patternType()) {
235                       case LITERAL:
236                           // If wildcard deny exists, return deny directly
237                           if (binding.pattern().name().equals(ResourcePattern.WILDCARD_RESOURCE))
238                               return AuthorizationResult.DENIED;
239                           denyPatterns.get(PatternType.LITERAL).add(binding.pattern().name());
240                           break;
241                       case PREFIXED:
242                           denyPatterns.get(PatternType.PREFIXED).add(binding.pattern().name());
243                           break;
244                       default:
245                   }
246                   continue;
247               }
248   
249               if (binding.entry().permissionType() != AclPermissionType.ALLOW)
250                   continue;
251   
252               switch (binding.pattern().patternType()) {
253                   case LITERAL:
254                       if (binding.pattern().name().equals(ResourcePattern.WILDCARD_RESOURCE)) {
255                           hasWildCardAllow = true;
256                           continue;
257                       }
258                       allowPatterns.get(PatternType.LITERAL).add(binding.pattern().name());
259                       break;
260                   case PREFIXED:
261                       allowPatterns.get(PatternType.PREFIXED).add(binding.pattern().name());
262                       break;
263                   default:
264               }
265           }
266   
267           if (hasWildCardAllow) {
268               return AuthorizationResult.ALLOWED;
269           }
270   
271           // For any literal allowed, if there's no dominant literal and prefix denied, return allow.
272           // For any prefix allowed, if there's no dominant prefix denied, return allow.
273           for (Map.Entry<PatternType, Set<String>> entry : allowPatterns.entrySet()) {
274               for (String allowStr : entry.getValue()) {
275                   if (entry.getKey() == PatternType.LITERAL
276                           && denyPatterns.get(PatternType.LITERAL).contains(allowStr))
277                       continue;
278                   StringBuilder sb = new StringBuilder();
279                   boolean hasDominatedDeny = false;
280                   for (char ch : allowStr.toCharArray()) {
281                       sb.append(ch);
282                       if (denyPatterns.get(PatternType.PREFIXED).contains(sb.toString())) {
283                           hasDominatedDeny = true;
284                           break;
285                       }
286                   }
287                   if (!hasDominatedDeny)
288                       return AuthorizationResult.ALLOWED;
289               }
290           }
291   
292           return AuthorizationResult.DENIED;
293       }
294   
295   }
