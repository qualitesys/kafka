19   import java.util.Objects;
20   
21   public class CipherInformation {
22       private final String cipher;
23       private final String protocol;
24   
25       public CipherInformation(String cipher, String protocol) {
26           this.cipher = cipher == null || cipher.isEmpty()  ? "unknown" : cipher;
27           this.protocol = protocol == null || protocol.isEmpty()  ? "unknown" : protocol;
28       }
29   
30       public String cipher() {
31           return cipher;
32       }
33   
34       public String protocol() {
35           return protocol;
36       }
37   
38       @Override
39       public String toString() {
40           return "CipherInformation(cipher=" + cipher +
41               ", protocol=" + protocol + ")";
42       }
43   
44       @Override
45       public int hashCode() {
46           return Objects.hash(cipher, protocol);
47       }
48   
49       @Override
50       public boolean equals(Object o) {
51           if (o == null) {
52               return false;
53           }
54           if (!(o instanceof CipherInformation)) {
55               return false;
56           }
57           CipherInformation other = (CipherInformation) o;
58           return other.cipher.equals(cipher) &&
59               other.protocol.equals(protocol);
60       }
61   }
