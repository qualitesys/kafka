71       @Override
72       public Map<String, List<TopicPartition>> assign(Map<String, Integer> partitionsPerTopic,
73                                                       Map<String, Subscription> subscriptions) {
74           Map<String, List<TopicPartition>> consumerToOwnedPartitions = new HashMap<>();
75           Set<TopicPartition> partitionsWithMultiplePreviousOwners = new HashSet<>();
76           if (allSubscriptionsEqual(partitionsPerTopic.keySet(), subscriptions, consumerToOwnedPartitions, partitionsWithMultiplePreviousOwners)) {
77               log.debug("Detected that all consumers were subscribed to same set of topics, invoking the "
78                             + "optimized assignment algorithm");
79               partitionsTransferringOwnership = new HashMap<>();
80               return constrainedAssign(partitionsPerTopic, consumerToOwnedPartitions, partitionsWithMultiplePreviousOwners);
81           } else {
82               log.debug("Detected that not all consumers were subscribed to same set of topics, falling back to the "
83                             + "general case assignment algorithm");
84               // we must set this to null for the general case so the cooperative assignor knows to compute it from scratch
85               partitionsTransferringOwnership = null;
86               return generalAssign(partitionsPerTopic, subscriptions, consumerToOwnedPartitions);
87           }
88       }

90       /**
91        * Returns true iff all consumers have an identical subscription. Also fills out the passed in
92        * {@code consumerToOwnedPartitions} with each consumer's previously owned and still-subscribed partitions,
93        * and the {@code partitionsWithMultiplePreviousOwners} with any partitions claimed by multiple previous owners
94        */
95       private boolean allSubscriptionsEqual(Set<String> allTopics,
96                                             Map<String, Subscription> subscriptions,
97                                             Map<String, List<TopicPartition>> consumerToOwnedPartitions,
98                                             Set<TopicPartition> partitionsWithMultiplePreviousOwners) {
99           Set<String> membersOfCurrentHighestGeneration = new HashSet<>();
100           boolean isAllSubscriptionsEqual = true;
101   
102           Set<String> subscribedTopics = new HashSet<>();
103   
104           // keep track of all previously owned partitions so we can invalidate them if invalid input is
105           // detected, eg two consumers somehow claiming the same partition in the same/current generation
106           Map<TopicPartition, String> allPreviousPartitionsToOwner = new HashMap<>();
107   
108           for (Map.Entry<String, Subscription> subscriptionEntry : subscriptions.entrySet()) {
109               String consumer = subscriptionEntry.getKey();
110               Subscription subscription = subscriptionEntry.getValue();
111   
112               // initialize the subscribed topics set if this is the first subscription
113               if (subscribedTopics.isEmpty()) {
114                   subscribedTopics.addAll(subscription.topics());
115               } else if (isAllSubscriptionsEqual && !(subscription.topics().size() == subscribedTopics.size()
116                   && subscribedTopics.containsAll(subscription.topics()))) {
117                   isAllSubscriptionsEqual = false;
118               }
119   
120               MemberData memberData = memberData(subscription);
121   
122               List<TopicPartition> ownedPartitions = new ArrayList<>();
123               consumerToOwnedPartitions.put(consumer, ownedPartitions);
124   
125               // Only consider this consumer's owned partitions as valid if it is a member of the current highest
126               // generation, or it's generation is not present but we have not seen any known generation so far
127               if (memberData.generation.isPresent() && memberData.generation.get() >= maxGeneration
128                   || !memberData.generation.isPresent() && maxGeneration == DEFAULT_GENERATION) {
129   
130                   // If the current member's generation is higher, all the previously owned partitions are invalid
131                   if (memberData.generation.isPresent() && memberData.generation.get() > maxGeneration) {
132                       allPreviousPartitionsToOwner.clear();
133                       partitionsWithMultiplePreviousOwners.clear();
134                       for (String droppedOutConsumer : membersOfCurrentHighestGeneration) {
135                           consumerToOwnedPartitions.get(droppedOutConsumer).clear();
136                       }
137   
138                       membersOfCurrentHighestGeneration.clear();
139                       maxGeneration = memberData.generation.get();
140                   }
141   
142                   membersOfCurrentHighestGeneration.add(consumer);
143                   for (final TopicPartition tp : memberData.partitions) {
144                       // filter out any topics that no longer exist or aren't part of the current subscription
145                       if (allTopics.contains(tp.topic())) {
146                           String otherConsumer = allPreviousPartitionsToOwner.put(tp, consumer);
147                           if (otherConsumer == null) {
148                               // this partition is not owned by other consumer in the same generation
149                               ownedPartitions.add(tp);
150                           } else {
151                               log.error("Found multiple consumers {} and {} claiming the same TopicPartition {} in the "
152                                   + "same generation {}, this will be invalidated and removed from their previous assignment.",
153                                        consumer, otherConsumer, tp, maxGeneration);
154                               consumerToOwnedPartitions.get(otherConsumer).remove(tp);
155                               partitionsWithMultiplePreviousOwners.add(tp);
156                           }
157                       }
158                   }
159               }
160           }
161   
162           return isAllSubscriptionsEqual;
163       }

165       /**
166        * This constrainedAssign optimizes the assignment algorithm when all consumers were subscribed to same set of topics.
167        * The method includes the following steps:
168        *
169        * 1. Reassign previously owned partitions:
170        *   a. if owned less than minQuota partitions, just assign all owned partitions, and put the member into unfilled member list
171        *   b. if owned maxQuota or more, and we're still under the number of expected max capacity members, assign maxQuota partitions
172        *   c. if owned at least "minQuota" of partitions, assign minQuota partitions, and put the member into unfilled member list if
173        *     we're still under the number of expected max capacity members
174        * 2. Fill remaining members up to the expected numbers of maxQuota partitions, otherwise, to minQuota partitions
175        *
176        * @param partitionsPerTopic                   The number of partitions for each subscribed topic
177        * @param consumerToOwnedPartitions            Each consumer's previously owned and still-subscribed partitions
178        * @param partitionsWithMultiplePreviousOwners The partitions being claimed in the previous assignment of multiple consumers
179        *
180        * @return                                     Map from each member to the list of partitions assigned to them.
181        */
182       private Map<String, List<TopicPartition>> constrainedAssign(Map<String, Integer> partitionsPerTopic,
183                                                                   Map<String, List<TopicPartition>> consumerToOwnedPartitions,
184                                                                   Set<TopicPartition> partitionsWithMultiplePreviousOwners) {
185           if (log.isDebugEnabled()) {
186               log.debug("Performing constrained assign with partitionsPerTopic: {}, consumerToOwnedPartitions: {}.",
187                   partitionsPerTopic, consumerToOwnedPartitions);
188           }
189   
190           Set<TopicPartition> allRevokedPartitions = new HashSet<>();
191   
192           // the consumers which may still be assigned one or more partitions to reach expected capacity
193           List<String> unfilledMembersWithUnderMinQuotaPartitions = new LinkedList<>();
194           LinkedList<String> unfilledMembersWithExactlyMinQuotaPartitions = new LinkedList<>();
195   
196           int numberOfConsumers = consumerToOwnedPartitions.size();
197           int totalPartitionsCount = partitionsPerTopic.values().stream().reduce(0, Integer::sum);
198   
199           int minQuota = (int) Math.floor(((double) totalPartitionsCount) / numberOfConsumers);
200           int maxQuota = (int) Math.ceil(((double) totalPartitionsCount) / numberOfConsumers);
201           // the expected number of members receiving more than minQuota partitions (zero when minQuota == maxQuota)
202           int expectedNumMembersWithOverMinQuotaPartitions = totalPartitionsCount % numberOfConsumers;
203           // the current number of members receiving more than minQuota partitions (zero when minQuota == maxQuota)
204           int currentNumMembersWithOverMinQuotaPartitions = 0;
205   
206           // initialize the assignment map with an empty array of size maxQuota for all members
207           Map<String, List<TopicPartition>> assignment = new HashMap<>(
208               consumerToOwnedPartitions.keySet().stream().collect(Collectors.toMap(c -> c, c -> new ArrayList<>(maxQuota))));
209   
210           List<TopicPartition> assignedPartitions = new ArrayList<>();
211           // Reassign previously owned partitions, up to the expected number of partitions per consumer
212           for (Map.Entry<String, List<TopicPartition>> consumerEntry : consumerToOwnedPartitions.entrySet()) {
213               String consumer = consumerEntry.getKey();
214               List<TopicPartition> ownedPartitions = consumerEntry.getValue();
215   
216               List<TopicPartition> consumerAssignment = assignment.get(consumer);
217   
218               for (TopicPartition doublyClaimedPartition : partitionsWithMultiplePreviousOwners) {
219                   if (ownedPartitions.contains(doublyClaimedPartition)) {
220                       log.error("Found partition {} still claimed as owned by consumer {}, despite being claimed by multiple "
221                                    + "consumers already in the same generation. Removing it from the ownedPartitions",
222                                doublyClaimedPartition, consumer);
223                       ownedPartitions.remove(doublyClaimedPartition);
224                   }
225               }
226   
227               if (ownedPartitions.size() < minQuota) {
228                   // the expected assignment size is more than this consumer has now, so keep all the owned partitions
229                   // and put this member into the unfilled member list
230                   if (ownedPartitions.size() > 0) {
231                       consumerAssignment.addAll(ownedPartitions);
232                       assignedPartitions.addAll(ownedPartitions);
233                   }
234                   unfilledMembersWithUnderMinQuotaPartitions.add(consumer);
235               } else if (ownedPartitions.size() >= maxQuota && currentNumMembersWithOverMinQuotaPartitions < expectedNumMembersWithOverMinQuotaPartitions) {
236                   // consumer owned the "maxQuota" of partitions or more, and we're still under the number of expected members
237                   // with more than the minQuota partitions, so keep "maxQuota" of the owned partitions, and revoke the rest of the partitions
238                   currentNumMembersWithOverMinQuotaPartitions++;
239                   if (currentNumMembersWithOverMinQuotaPartitions == expectedNumMembersWithOverMinQuotaPartitions) {
240                       unfilledMembersWithExactlyMinQuotaPartitions.clear();
241                   }
242                   List<TopicPartition> maxQuotaPartitions = ownedPartitions.subList(0, maxQuota);
243                   consumerAssignment.addAll(maxQuotaPartitions);
244                   assignedPartitions.addAll(maxQuotaPartitions);
245                   allRevokedPartitions.addAll(ownedPartitions.subList(maxQuota, ownedPartitions.size()));
246               } else {
247                   // consumer owned at least "minQuota" of partitions
248                   // so keep "minQuota" of the owned partitions, and revoke the rest of the partitions
249                   List<TopicPartition> minQuotaPartitions = ownedPartitions.subList(0, minQuota);
250                   consumerAssignment.addAll(minQuotaPartitions);
251                   assignedPartitions.addAll(minQuotaPartitions);
252                   allRevokedPartitions.addAll(ownedPartitions.subList(minQuota, ownedPartitions.size()));
253                   // this consumer is potential maxQuota candidate since we're still under the number of expected members
254                   // with more than the minQuota partitions. Note, if the number of expected members with more than
255                   // the minQuota partitions is 0, it means minQuota == maxQuota, and there are no potentially unfilled
256                   if (currentNumMembersWithOverMinQuotaPartitions < expectedNumMembersWithOverMinQuotaPartitions) {
257                       unfilledMembersWithExactlyMinQuotaPartitions.add(consumer);
258                   }
259               }
260           }
261   
262           List<TopicPartition> unassignedPartitions = getUnassignedPartitions(totalPartitionsCount, partitionsPerTopic, assignedPartitions);
263   
264           if (log.isDebugEnabled()) {
265               log.debug("After reassigning previously owned partitions, unfilled members: {}, unassigned partitions: {}, " +
266                   "current assignment: {}", unfilledMembersWithUnderMinQuotaPartitions, unassignedPartitions, assignment);
267           }
268   
269           Collections.sort(unfilledMembersWithUnderMinQuotaPartitions);
270           Collections.sort(unfilledMembersWithExactlyMinQuotaPartitions);
271   
272           Iterator<String> unfilledConsumerIter = unfilledMembersWithUnderMinQuotaPartitions.iterator();
273           // Round-Robin filling remaining members up to the expected numbers of maxQuota, otherwise, to minQuota
274           for (TopicPartition unassignedPartition : unassignedPartitions) {
275               String consumer;
276               if (unfilledConsumerIter.hasNext()) {
277                   consumer = unfilledConsumerIter.next();
278               } else {
279                   if (unfilledMembersWithUnderMinQuotaPartitions.isEmpty() && unfilledMembersWithExactlyMinQuotaPartitions.isEmpty()) {
280                       // Should not enter here since we have calculated the exact number to assign to each consumer.
281                       // This indicates issues in the assignment algorithm
282                       int currentPartitionIndex = unassignedPartitions.indexOf(unassignedPartition);
283                       log.error("No more unfilled consumers to be assigned. The remaining unassigned partitions are: {}",
284                                 unassignedPartitions.subList(currentPartitionIndex, unassignedPartitions.size()));
285                       throw new IllegalStateException("No more unfilled consumers to be assigned.");
286                   } else if (unfilledMembersWithUnderMinQuotaPartitions.isEmpty()) {
287                       consumer = unfilledMembersWithExactlyMinQuotaPartitions.poll();
288                   } else {
289                       unfilledConsumerIter = unfilledMembersWithUnderMinQuotaPartitions.iterator();
290                       consumer = unfilledConsumerIter.next();
291                   }
292               }
293   
294               List<TopicPartition> consumerAssignment = assignment.get(consumer);
295               consumerAssignment.add(unassignedPartition);
296   
297               // We already assigned all possible ownedPartitions, so we know this must be newly assigned to this consumer
298               // or else the partition was actually claimed by multiple previous owners and had to be invalidated from all
299               // members claimed ownedPartitions
300               if (allRevokedPartitions.contains(unassignedPartition) || partitionsWithMultiplePreviousOwners.contains(unassignedPartition))
301                   partitionsTransferringOwnership.put(unassignedPartition, consumer);
302   
303               int currentAssignedCount = consumerAssignment.size();
304               if (currentAssignedCount == minQuota) {
305                   unfilledConsumerIter.remove();
306                   unfilledMembersWithExactlyMinQuotaPartitions.add(consumer);
307               } else if (currentAssignedCount == maxQuota) {
308                   currentNumMembersWithOverMinQuotaPartitions++;
309                   if (currentNumMembersWithOverMinQuotaPartitions == expectedNumMembersWithOverMinQuotaPartitions) {
310                       // We only start to iterate over the "potentially unfilled" members at minQuota after we've filled
311                       // all members up to at least minQuota, so once the last minQuota member reaches maxQuota, we
312                       // should be done. But in case of some algorithmic error, just log a warning and continue to
313                       // assign any remaining partitions within the assignment constraints
314                       if (unassignedPartitions.indexOf(unassignedPartition) != unassignedPartitions.size() - 1) {
315                           log.error("Filled the last member up to maxQuota but still had partitions remaining to assign, "
316                                        + "will continue but this indicates a bug in the assignment.");
317                       }
318                   }
319               }
320           }
321   
322           if (!unfilledMembersWithUnderMinQuotaPartitions.isEmpty()) {
323               // we expected all the remaining unfilled members have minQuota partitions and we're already at the expected number
324               // of members with more than the minQuota partitions. Otherwise, there must be error here.
325               if (currentNumMembersWithOverMinQuotaPartitions != expectedNumMembersWithOverMinQuotaPartitions) {
326                   log.error("Current number of members with more than the minQuota partitions: {}, is less than the expected number " +
327                       "of members with more than the minQuota partitions: {}, and no more partitions to be assigned to the remaining unfilled consumers: {}",
328                       currentNumMembersWithOverMinQuotaPartitions, expectedNumMembersWithOverMinQuotaPartitions, unfilledMembersWithUnderMinQuotaPartitions);
329                   throw new IllegalStateException("We haven't reached the expected number of members with " +
330                       "more than the minQuota partitions, but no more partitions to be assigned");
331               } else {
332                   for (String unfilledMember : unfilledMembersWithUnderMinQuotaPartitions) {
333                       int assignedPartitionsCount = assignment.get(unfilledMember).size();
334                       if (assignedPartitionsCount != minQuota) {
335                           log.error("Consumer: [{}] should have {} partitions, but got {} partitions, and no more partitions " +
336                               "to be assigned. The remaining unfilled consumers are: {}", unfilledMember, minQuota, assignedPartitionsCount, unfilledMembersWithUnderMinQuotaPartitions);
337                           throw new IllegalStateException(String.format("Consumer: [%s] doesn't reach minQuota partitions, " +
338                               "and no more partitions to be assigned", unfilledMember));
339                       } else {
340                           log.trace("skip over this unfilled member: [{}] because we've reached the expected number of " +
341                               "members with more than the minQuota partitions, and this member already have minQuota partitions", unfilledMember);
342                       }
343                   }
344               }
345           }
346   
347           log.info("Final assignment of partitions to consumers: \n{}", assignment);
348   
349           return assignment;
350       }

353       private List<TopicPartition> getAllTopicPartitions(Map<String, Integer> partitionsPerTopic,
354                                                          List<String> sortedAllTopics,
355                                                          int totalPartitionsCount) {
356           List<TopicPartition> allPartitions = new ArrayList<>(totalPartitionsCount);
357   
358           for (String topic : sortedAllTopics) {
359               int partitionCount = partitionsPerTopic.get(topic);
360               for (int i = 0; i < partitionCount; ++i) {
361                   allPartitions.add(new TopicPartition(topic, i));
362               }
363           }
364           return allPartitions;
365       }

367       /**
368        * This generalAssign algorithm guarantees the assignment that is as balanced as possible.
369        * This method includes the following steps:
370        *
371        * 1. Preserving all the existing partition assignments
372        * 2. Removing all the partition assignments that have become invalid due to the change that triggers the reassignment
373        * 3. Assigning the unassigned partitions in a way that balances out the overall assignments of partitions to consumers
374        * 4. Further balancing out the resulting assignment by finding the partitions that can be reassigned
375        *    to another consumer towards an overall more balanced assignment.
376        *
377        * @param partitionsPerTopic         The number of partitions for each subscribed topic.
378        * @param subscriptions              Map from the member id to their respective topic subscription
379        * @param currentAssignment          Each consumer's previously owned and still-subscribed partitions
380        *
381        * @return                           Map from each member to the list of partitions assigned to them.
382        */
383       private Map<String, List<TopicPartition>> generalAssign(Map<String, Integer> partitionsPerTopic,
384                                                               Map<String, Subscription> subscriptions,
385                                                               Map<String, List<TopicPartition>> currentAssignment) {
386           if (log.isDebugEnabled()) {
387               log.debug("performing general assign. partitionsPerTopic: {}, subscriptions: {}, currentAssignment: {}",
388                   partitionsPerTopic, subscriptions, currentAssignment);
389           }
390   
391           Map<TopicPartition, ConsumerGenerationPair> prevAssignment = new HashMap<>();
392           partitionMovements = new PartitionMovements();
393   
394           prepopulateCurrentAssignments(subscriptions, prevAssignment);
395   
396           // a mapping of all topics to all consumers that can be assigned to them
397           final Map<String, List<String>> topic2AllPotentialConsumers = new HashMap<>(partitionsPerTopic.keySet().size());
398           // a mapping of all consumers to all potential topics that can be assigned to them
399           final Map<String, List<String>> consumer2AllPotentialTopics = new HashMap<>(subscriptions.keySet().size());
400   
401           // initialize topic2AllPotentialConsumers and consumer2AllPotentialTopics
402           partitionsPerTopic.keySet().stream().forEach(
403               topicName -> topic2AllPotentialConsumers.put(topicName, new ArrayList<>()));
404   
405           for (Entry<String, Subscription> entry: subscriptions.entrySet()) {
406               String consumerId = entry.getKey();
407               List<String> subscribedTopics = new ArrayList<>(entry.getValue().topics().size());
408               consumer2AllPotentialTopics.put(consumerId, subscribedTopics);
409               entry.getValue().topics().stream().filter(topic -> partitionsPerTopic.get(topic) != null).forEach(topic -> {
410                   subscribedTopics.add(topic);
411                   topic2AllPotentialConsumers.get(topic).add(consumerId);
412               });
413   
414               // add this consumer to currentAssignment (with an empty topic partition assignment) if it does not already exist
415               if (!currentAssignment.containsKey(consumerId))
416                   currentAssignment.put(consumerId, new ArrayList<>());
417           }
418   
419           // a mapping of partition to current consumer
420           Map<TopicPartition, String> currentPartitionConsumer = new HashMap<>();
421           for (Map.Entry<String, List<TopicPartition>> entry: currentAssignment.entrySet())
422               for (TopicPartition topicPartition: entry.getValue())
423                   currentPartitionConsumer.put(topicPartition, entry.getKey());
424   
425           int totalPartitionsCount = partitionsPerTopic.values().stream().reduce(0, Integer::sum);
426           List<String> sortedAllTopics = new ArrayList<>(topic2AllPotentialConsumers.keySet());
427           Collections.sort(sortedAllTopics, new TopicComparator(topic2AllPotentialConsumers));
428           List<TopicPartition> sortedAllPartitions = getAllTopicPartitions(partitionsPerTopic, sortedAllTopics, totalPartitionsCount);
429   
430           // the partitions already assigned in current assignment
431           List<TopicPartition> assignedPartitions = new ArrayList<>();
432           boolean revocationRequired = false;
433           for (Iterator<Entry<String, List<TopicPartition>>> it = currentAssignment.entrySet().iterator(); it.hasNext();) {
434               Map.Entry<String, List<TopicPartition>> entry = it.next();
435               Subscription consumerSubscription = subscriptions.get(entry.getKey());
436               if (consumerSubscription == null) {
437                   // if a consumer that existed before (and had some partition assignments) is now removed, remove it from currentAssignment
438                   for (TopicPartition topicPartition: entry.getValue())
439                       currentPartitionConsumer.remove(topicPartition);
440                   it.remove();
441               } else {
442                   // otherwise (the consumer still exists)
443                   for (Iterator<TopicPartition> partitionIter = entry.getValue().iterator(); partitionIter.hasNext();) {
444                       TopicPartition partition = partitionIter.next();
445                       if (!topic2AllPotentialConsumers.containsKey(partition.topic())) {
446                           // if this topic partition of this consumer no longer exists, remove it from currentAssignment of the consumer
447                           partitionIter.remove();
448                           currentPartitionConsumer.remove(partition);
449                       } else if (!consumerSubscription.topics().contains(partition.topic())) {
450                           // because the consumer is no longer subscribed to its topic, remove it from currentAssignment of the consumer
451                           partitionIter.remove();
452                           revocationRequired = true;
453                       } else {
454                           // otherwise, remove the topic partition from those that need to be assigned only if
455                           // its current consumer is still subscribed to its topic (because it is already assigned
456                           // and we would want to preserve that assignment as much as possible)
457                           assignedPartitions.add(partition);
458                       }
459                   }
460               }
461           }
462   
463           // all partitions that needed to be assigned
464           List<TopicPartition> unassignedPartitions = getUnassignedPartitions(sortedAllPartitions, assignedPartitions, topic2AllPotentialConsumers);
465   
466           if (log.isDebugEnabled()) {
467               log.debug("unassigned Partitions: {}", unassignedPartitions);
468           }
469   
470           // at this point we have preserved all valid topic partition to consumer assignments and removed
471           // all invalid topic partitions and invalid consumers. Now we need to assign unassignedPartitions
472           // to consumers so that the topic partition assignments are as balanced as possible.
473   
474           // an ascending sorted set of consumers based on how many topic partitions are already assigned to them
475           TreeSet<String> sortedCurrentSubscriptions = new TreeSet<>(new SubscriptionComparator(currentAssignment));
476           sortedCurrentSubscriptions.addAll(currentAssignment.keySet());
477   
478           balance(currentAssignment, prevAssignment, sortedAllPartitions, unassignedPartitions, sortedCurrentSubscriptions,
479               consumer2AllPotentialTopics, topic2AllPotentialConsumers, currentPartitionConsumer, revocationRequired,
480               partitionsPerTopic, totalPartitionsCount);
481   
482           log.info("Final assignment of partitions to consumers: \n{}", currentAssignment);
483   
484           return currentAssignment;
485       }
486   
487       /**
488        * get the unassigned partition list by computing the difference set of the sortedPartitions(all partitions)
489        * and sortedAssignedPartitions. If no assigned partitions, we'll just return all sorted topic partitions.
490        * This is used in generalAssign method
491        *
492        * We loop the sortedPartition, and compare the ith element in sortedAssignedPartitions(i start from 0):
493        *   - if not equal to the ith element, add to unassignedPartitions
494        *   - if equal to the ith element, get next element from sortedAssignedPartitions
495        *
496        * @param sortedAllPartitions:          sorted all partitions
497        * @param sortedAssignedPartitions:     sorted partitions, all are included in the sortedPartitions
498        * @param topic2AllPotentialConsumers:  topics mapped to all consumers that subscribed to it
499        * @return                              partitions that aren't assigned to any current consumer
500        */
501       private List<TopicPartition> getUnassignedPartitions(List<TopicPartition> sortedAllPartitions,
502                                                            List<TopicPartition> sortedAssignedPartitions,
503                                                            Map<String, List<String>> topic2AllPotentialConsumers) {
504           if (sortedAssignedPartitions.isEmpty()) {
505               return sortedAllPartitions;
506           }
507   
508           List<TopicPartition> unassignedPartitions = new ArrayList<>();
509   
510           Collections.sort(sortedAssignedPartitions, new PartitionComparator(topic2AllPotentialConsumers));
511   
512           boolean shouldAddDirectly = false;
513           Iterator<TopicPartition> sortedAssignedPartitionsIter = sortedAssignedPartitions.iterator();
514           TopicPartition nextAssignedPartition = sortedAssignedPartitionsIter.next();
515   
516           for (TopicPartition topicPartition : sortedAllPartitions) {
517               if (shouldAddDirectly || !nextAssignedPartition.equals(topicPartition)) {
518                   unassignedPartitions.add(topicPartition);
519               } else {
520                   // this partition is in assignedPartitions, don't add to unassignedPartitions, just get next assigned partition
521                   if (sortedAssignedPartitionsIter.hasNext()) {
522                       nextAssignedPartition = sortedAssignedPartitionsIter.next();
523                   } else {
524                       // add the remaining directly since there is no more sortedAssignedPartitions
525                       shouldAddDirectly = true;
526                   }
527               }
528           }
529           return unassignedPartitions;
530       }
531   
532       /**
533        * get the unassigned partition list by computing the difference set of all sorted partitions
534        * and sortedAssignedPartitions. If no assigned partitions, we'll just return all sorted topic partitions.
535        * This is used in constrainedAssign method
536        *
537        * To compute the difference set, we use two pointers technique here:
538        *
539        * We loop through the all sorted topics, and then iterate all partitions the topic has,
540        * compared with the ith element in sortedAssignedPartitions(i starts from 0):
541        *   - if not equal to the ith element, add to unassignedPartitions
542        *   - if equal to the ith element, get next element from sortedAssignedPartitions
543        *
544        * @param totalPartitionsCount      all partitions counts in this assignment
545        * @param partitionsPerTopic        the number of partitions for each subscribed topic.
546        * @param sortedAssignedPartitions  sorted partitions, all are included in the sortedPartitions
547        * @return                          the partitions not yet assigned to any consumers
548        */
549       private List<TopicPartition> getUnassignedPartitions(int totalPartitionsCount,
550                                                            Map<String, Integer> partitionsPerTopic,
551                                                            List<TopicPartition> sortedAssignedPartitions) {
552           List<String> sortedAllTopics = new ArrayList<>(partitionsPerTopic.keySet());
553           // sort all topics first, then we can have sorted all topic partitions by adding partitions starting from 0
554           Collections.sort(sortedAllTopics);
555   
556           if (sortedAssignedPartitions.isEmpty()) {
557               // no assigned partitions means all partitions are unassigned partitions
558               return getAllTopicPartitions(partitionsPerTopic, sortedAllTopics, totalPartitionsCount);
559           }
560   
561           List<TopicPartition> unassignedPartitions = new ArrayList<>(totalPartitionsCount - sortedAssignedPartitions.size());
562   
563           Collections.sort(sortedAssignedPartitions, Comparator.comparing(TopicPartition::topic).thenComparing(TopicPartition::partition));
564   
565           boolean shouldAddDirectly = false;
566           Iterator<TopicPartition> sortedAssignedPartitionsIter = sortedAssignedPartitions.iterator();
567           TopicPartition nextAssignedPartition = sortedAssignedPartitionsIter.next();
568   
569           for (String topic : sortedAllTopics) {
570               int partitionCount = partitionsPerTopic.get(topic);
571               for (int i = 0; i < partitionCount; i++) {
572                   if (shouldAddDirectly || !(nextAssignedPartition.topic().equals(topic) && nextAssignedPartition.partition() == i)) {
573                       unassignedPartitions.add(new TopicPartition(topic, i));
574                   } else {
575                       // this partition is in assignedPartitions, don't add to unassignedPartitions, just get next assigned partition
576                       if (sortedAssignedPartitionsIter.hasNext()) {
577                           nextAssignedPartition = sortedAssignedPartitionsIter.next();
578                       } else {
579                           // add the remaining directly since there is no more sortedAssignedPartitions
580                           shouldAddDirectly = true;
581                       }
582                   }
583               }
584           }
585   
586           return unassignedPartitions;
587       }

589       /**
590        * update the prevAssignment with the partitions, consumer and generation in parameters
591        *
592        * @param partitions:       The partitions to be updated the prevAssignment
593        * @param consumer:         The consumer Id
594        * @param prevAssignment:   The assignment contains the assignment with the 2nd largest generation
595        * @param generation:       The generation of this assignment (partitions)
596        */
597       private void updatePrevAssignment(Map<TopicPartition, ConsumerGenerationPair> prevAssignment,
598                                         List<TopicPartition> partitions,
599                                         String consumer,
600                                         int generation) {
601           for (TopicPartition partition: partitions) {
602               if (prevAssignment.containsKey(partition)) {
603                   // only keep the latest previous assignment
604                   if (generation > prevAssignment.get(partition).generation) {
605                       prevAssignment.put(partition, new ConsumerGenerationPair(consumer, generation));
606                   }
607               } else {
608                   prevAssignment.put(partition, new ConsumerGenerationPair(consumer, generation));
609               }
610           }
611       }
612   
613       /**
614        * filling in the prevAssignment from the subscriptions.
615        *
616        * @param subscriptions:        Map from the member id to their respective topic subscription
617        * @param prevAssignment:       The assignment contains the assignment with the 2nd largest generation
618        */
619       private void prepopulateCurrentAssignments(Map<String, Subscription> subscriptions,
620                                                  Map<TopicPartition, ConsumerGenerationPair> prevAssignment) {
621           // we need to process subscriptions' user data with each consumer's reported generation in mind
622           // higher generations overwrite lower generations in case of a conflict
623           // note that a conflict could exists only if user data is for different generations
624   
625           for (Map.Entry<String, Subscription> subscriptionEntry: subscriptions.entrySet()) {
626               String consumer = subscriptionEntry.getKey();
627               Subscription subscription = subscriptionEntry.getValue();
628               if (subscription.userData() != null) {
629                   // since this is our 2nd time to deserialize memberData, rewind userData is necessary
630                   subscription.userData().rewind();
631               }
632               MemberData memberData = memberData(subscriptionEntry.getValue());
633   
634               // we already have the maxGeneration info, so just compare the current generation of memberData, and put into prevAssignment
635               if (memberData.generation.isPresent() && memberData.generation.get() < maxGeneration) {
636                   // if the current member's generation is lower than maxGeneration, put into prevAssignment if needed
637                   updatePrevAssignment(prevAssignment, memberData.partitions, consumer, memberData.generation.get());
638               } else if (!memberData.generation.isPresent() && maxGeneration > DEFAULT_GENERATION) {
639                   // if maxGeneration is larger then DEFAULT_GENERATION
640                   // put all (no generation) partitions as DEFAULT_GENERATION into prevAssignment if needed
641                   updatePrevAssignment(prevAssignment, memberData.partitions, consumer, DEFAULT_GENERATION);
642               }
643           }
644       }

646       /**
647        * determine if the current assignment is a balanced one
648        *
649        * @param currentAssignment:            the assignment whose balance needs to be checked
650        * @param sortedCurrentSubscriptions:   an ascending sorted set of consumers based on how many topic partitions are already assigned to them
651        * @param allSubscriptions:             a mapping of all consumers to all potential topics that can be assigned to them
652        * @param partitionsPerTopic:           The number of partitions for each subscribed topic
653        * @param totalPartitionCount           total partition count to be assigned
654        * @return                              true if the given assignment is balanced; false otherwise
655        */
656       private boolean isBalanced(Map<String, List<TopicPartition>> currentAssignment,
657                                  TreeSet<String> sortedCurrentSubscriptions,
658                                  Map<String, List<String>> allSubscriptions,
659                                  Map<String, Integer> partitionsPerTopic,
660                                  int totalPartitionCount) {
661           int min = currentAssignment.get(sortedCurrentSubscriptions.first()).size();
662           int max = currentAssignment.get(sortedCurrentSubscriptions.last()).size();
663           if (min >= max - 1)
664               // if minimum and maximum numbers of partitions assigned to consumers differ by at most one return true
665               return true;
666   
667           // create a mapping from partitions to the consumer assigned to them
668           final Map<TopicPartition, String> allPartitions = new HashMap<>();
669           Set<Entry<String, List<TopicPartition>>> assignments = currentAssignment.entrySet();
670           for (Map.Entry<String, List<TopicPartition>> entry: assignments) {
671               List<TopicPartition> topicPartitions = entry.getValue();
672               for (TopicPartition topicPartition: topicPartitions) {
673                   if (allPartitions.containsKey(topicPartition))
674                       log.error("{} is assigned to more than one consumer.", topicPartition);
675                   allPartitions.put(topicPartition, entry.getKey());
676               }
677           }
678   
679           // for each consumer that does not have all the topic partitions it can get make sure none of the topic partitions it
680           // could but did not get cannot be moved to it (because that would break the balance)
681           for (String consumer: sortedCurrentSubscriptions) {
682               List<TopicPartition> consumerPartitions = currentAssignment.get(consumer);
683               int consumerPartitionCount = consumerPartitions.size();
684   
685               // skip if this consumer already has all the topic partitions it can get
686               List<String> allSubscribedTopics = allSubscriptions.get(consumer);
687               int maxAssignmentSize = getMaxAssignmentSize(totalPartitionCount, allSubscribedTopics, partitionsPerTopic);
688   
689               if (consumerPartitionCount == maxAssignmentSize)
690                   continue;
691   
692               // otherwise make sure it cannot get any more
693               for (String topic: allSubscribedTopics) {
694                   int partitionCount = partitionsPerTopic.get(topic);
695                   for (int i = 0; i < partitionCount; i++) {
696                       TopicPartition topicPartition = new TopicPartition(topic, i);
697                       if (!currentAssignment.get(consumer).contains(topicPartition)) {
698                           String otherConsumer = allPartitions.get(topicPartition);
699                           int otherConsumerPartitionCount = currentAssignment.get(otherConsumer).size();
700                           if (consumerPartitionCount < otherConsumerPartitionCount) {
701                               log.debug("{} can be moved from consumer {} to consumer {} for a more balanced assignment.",
702                                   topicPartition, otherConsumer, consumer);
703                               return false;
704                           }
705                       }
706                   }
707               }
708           }
709           return true;
710       }
711   
712       /**
713        * get the maximum assigned partition size of the {@code allSubscribedTopics}
714        *
715        * @param totalPartitionCount           total partition count to be assigned
716        * @param allSubscribedTopics           the subscribed topics of a consumer
717        * @param partitionsPerTopic            The number of partitions for each subscribed topic
718        * @return                              maximum assigned partition size
719        */
720       private int getMaxAssignmentSize(int totalPartitionCount,
721                                        List<String> allSubscribedTopics,
722                                        Map<String, Integer> partitionsPerTopic) {
723           int maxAssignmentSize;
724           if (allSubscribedTopics.size() == partitionsPerTopic.size()) {
725               maxAssignmentSize = totalPartitionCount;
726           } else {
727               maxAssignmentSize = allSubscribedTopics.stream().map(topic -> partitionsPerTopic.get(topic)).reduce(0, Integer::sum);
728           }
729           return maxAssignmentSize;

732       /**
733        * @return the balance score of the given assignment, as the sum of assigned partitions size difference of all consumer pairs.
734        * A perfectly balanced assignment (with all consumers getting the same number of partitions) has a balance score of 0.
735        * Lower balance score indicates a more balanced assignment.
736        */
737       private int getBalanceScore(Map<String, List<TopicPartition>> assignment) {
738           int score = 0;
739   
740           Map<String, Integer> consumer2AssignmentSize = new HashMap<>();
741           for (Entry<String, List<TopicPartition>> entry: assignment.entrySet())
742               consumer2AssignmentSize.put(entry.getKey(), entry.getValue().size());
743   
744           Iterator<Entry<String, Integer>> it = consumer2AssignmentSize.entrySet().iterator();
745           while (it.hasNext()) {
746               Entry<String, Integer> entry = it.next();
747               int consumerAssignmentSize = entry.getValue();
748               it.remove();
749               for (Entry<String, Integer> otherEntry: consumer2AssignmentSize.entrySet())
750                   score += Math.abs(consumerAssignmentSize - otherEntry.getValue());
751           }
752   
753           return score;
754       }

815                            TreeSet<String> sortedCurrentSubscriptions,

837               if (!canParticipateInReassignment(topic, topic2AllPotentialConsumers)) {

839                       fixedPartitions.add(new TopicPartition(topic, i));
840                   }
841               }

843           unassignedPartitions.removeAll(fixedPartitions);
844   
845           // narrow down the reassignment scope to only those consumers that are subject to reassignment
846           Map<String, List<TopicPartition>> fixedAssignments = new HashMap<>();
847           for (String consumer: consumer2AllPotentialTopics.keySet())
848               if (!canParticipateInReassignment(consumer, currentAssignment,
849                   consumer2AllPotentialTopics, topic2AllPotentialConsumers, partitionsPerTopic, totalPartitionCount)) {
850                   sortedCurrentSubscriptions.remove(consumer);
851                   fixedAssignments.put(consumer, currentAssignment.remove(consumer));
852               }

859           if (!revocationRequired) {

862           }

865               consumer2AllPotentialTopics, topic2AllPotentialConsumers, currentPartitionConsumer, partitionsPerTopic, totalPartitionCount);
866   
867           // if we are not preserving existing assignments and we have made changes to the current assignment
868           // make sure we are getting a more balanced assignment; otherwise, revert to previous assignment
869           if (!initializing && reassignmentPerformed && getBalanceScore(currentAssignment) >= getBalanceScore(preBalanceAssignment)) {
870               deepCopy(preBalanceAssignment, currentAssignment);
871               currentPartitionConsumer.clear();
872               currentPartitionConsumer.putAll(preBalancePartitionConsumers);
873           }
874   
875           // add the fixed assignments (those that could not change) back
876           for (Entry<String, List<TopicPartition>> entry: fixedAssignments.entrySet()) {
877               String consumer = entry.getKey();
878               currentAssignment.put(consumer, entry.getValue());
879               sortedCurrentSubscriptions.add(consumer);
880           }
881   
882           fixedAssignments.clear();
883       }

885       private boolean performReassignments(List<TopicPartition> reassignablePartitions,
886                                            Map<String, List<TopicPartition>> currentAssignment,
887                                            Map<TopicPartition, ConsumerGenerationPair> prevAssignment,
888                                            TreeSet<String> sortedCurrentSubscriptions,
889                                            Map<String, List<String>> consumer2AllPotentialTopics,
890                                            Map<String, List<String>> topic2AllPotentialConsumers,
891                                            Map<TopicPartition, String> currentPartitionConsumer,

893                                            int totalPartitionCount) {
894           boolean reassignmentPerformed = false;
895           boolean modified;
896   
897           // repeat reassignment until no partition can be moved to improve the balance
898           do {
899               modified = false;
900               // reassign all reassignable partitions (starting from the partition with least potential consumers and if needed)
901               // until the full list is processed or a balance is achieved
902               Iterator<TopicPartition> partitionIterator = reassignablePartitions.iterator();

905                   TopicPartition partition = partitionIterator.next();
906   
907                   // the partition must have at least two consumers
908                   if (topic2AllPotentialConsumers.get(partition.topic()).size() <= 1)
909                       log.error("Expected more than one potential consumer for partition '{}'", partition);
910   
911                   // the partition must have a current consumer
912                   String consumer = currentPartitionConsumer.get(partition);
913                   if (consumer == null)
914                       log.error("Expected partition '{}' to be assigned to a consumer", partition);
915   
916                   if (prevAssignment.containsKey(partition) &&
917                       currentAssignment.get(consumer).size() > currentAssignment.get(prevAssignment.get(partition).consumer).size() + 1) {
918                       reassignPartition(partition, currentAssignment, sortedCurrentSubscriptions, currentPartitionConsumer, prevAssignment.get(partition).consumer);
919                       reassignmentPerformed = true;
920                       modified = true;
921                       continue;
922                   }
923   
924                   // check if a better-suited consumer exist for the partition; if so, reassign it
925                   for (String otherConsumer: topic2AllPotentialConsumers.get(partition.topic())) {
926                       if (currentAssignment.get(consumer).size() > currentAssignment.get(otherConsumer).size() + 1) {
927                           reassignPartition(partition, currentAssignment, sortedCurrentSubscriptions, currentPartitionConsumer, consumer2AllPotentialTopics);
928                           reassignmentPerformed = true;
929                           modified = true;
930                           break;
931                       }
932                   }
933               }
934           } while (modified);
935   
936           return reassignmentPerformed;
937       }
