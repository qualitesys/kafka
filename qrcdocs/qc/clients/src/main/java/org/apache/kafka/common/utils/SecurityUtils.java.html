39       private static final Map<String, ResourceType> NAME_TO_RESOURCE_TYPES;
40       private static final Map<String, AclOperation> NAME_TO_OPERATIONS;
41       private static final Map<String, AclPermissionType> NAME_TO_PERMISSION_TYPES;
42   
43       static {
44           NAME_TO_RESOURCE_TYPES = new HashMap<>(ResourceType.values().length);
45           NAME_TO_OPERATIONS = new HashMap<>(AclOperation.values().length);
46           NAME_TO_PERMISSION_TYPES = new HashMap<>(AclPermissionType.values().length);
47   
48           for (ResourceType resourceType : ResourceType.values()) {
49               String resourceTypeName = toPascalCase(resourceType.name());
50               NAME_TO_RESOURCE_TYPES.put(resourceTypeName, resourceType);
51               NAME_TO_RESOURCE_TYPES.put(resourceTypeName.toUpperCase(Locale.ROOT), resourceType);
52           }
53           for (AclOperation operation : AclOperation.values()) {
54               String operationName = toPascalCase(operation.name());
55               NAME_TO_OPERATIONS.put(operationName, operation);
56               NAME_TO_OPERATIONS.put(operationName.toUpperCase(Locale.ROOT), operation);
57           }
58           for (AclPermissionType permissionType : AclPermissionType.values()) {
59               String permissionName  = toPascalCase(permissionType.name());
60               NAME_TO_PERMISSION_TYPES.put(permissionName, permissionType);
61               NAME_TO_PERMISSION_TYPES.put(permissionName.toUpperCase(Locale.ROOT), permissionType);
62           }
63       }

65       public static KafkaPrincipal parseKafkaPrincipal(String str) {
66           if (str == null || str.isEmpty()) {
67               throw new IllegalArgumentException("expected a string in format principalType:principalName but got " + str);
68           }
69   
70           String[] split = str.split(":", 2);
71   
72           if (split.length != 2) {
73               throw new IllegalArgumentException("expected a string in format principalType:principalName but got " + str);
74           }
75   
76           return new KafkaPrincipal(split[0], split[1]);
77       }

79       public static void addConfiguredSecurityProviders(Map<String, ?> configs) {
80           String securityProviderClassesStr = (String) configs.get(SecurityConfig.SECURITY_PROVIDERS_CONFIG);
81           if (securityProviderClassesStr == null || securityProviderClassesStr.equals("")) {
82               return;
83           }
84           try {
85               String[] securityProviderClasses = securityProviderClassesStr.replaceAll("\\s+", "").split(",");
86               for (int index = 0; index < securityProviderClasses.length; index++) {
87                   SecurityProviderCreator securityProviderCreator =
88                       (SecurityProviderCreator) Class.forName(securityProviderClasses[index]).getConstructor().newInstance();
89                   securityProviderCreator.configure(configs);
90                   Security.insertProviderAt(securityProviderCreator.getProvider(), index + 1);
91               }
92           } catch (ClassCastException e) {
93               LOGGER.error("Creators provided through " + SecurityConfig.SECURITY_PROVIDERS_CONFIG +
94                       " are expected to be sub-classes of SecurityProviderCreator");
95           } catch (ClassNotFoundException cnfe) {
96               LOGGER.error("Unrecognized security provider creator class", cnfe);
97           } catch (ReflectiveOperationException e) {
98               LOGGER.error("Unexpected implementation of security provider creator class", e);
99           }
100       }

114       // We use Pascal-case to store these values, so lookup using provided key first to avoid
115       // case conversion for the common case. For backward compatibility, also perform
116       // case-insensitive look up (without underscores) by converting the key to upper-case.
117       private static <T> T valueFromMap(Map<String, T> map, String key, T unknown) {
118           T value = map.get(key);
119           if (value == null) {
120               value = map.get(key.toUpperCase(Locale.ROOT));
121           }
122           return value == null ? unknown : value;
123       }

137       private static String toPascalCase(String name) {
138           StringBuilder builder = new StringBuilder();
139           boolean capitalizeNext = true;
140           for (char c : name.toCharArray()) {
141               if (c == '_')
142                   capitalizeNext = true;
143               else if (capitalizeNext) {
144                   builder.append(Character.toUpperCase(c));
145                   capitalizeNext = false;
146               } else
147                   builder.append(Character.toLowerCase(c));
148           }
149           return builder.toString();
150       }

152       public static void authorizeByResourceTypeCheckArgs(AclOperation op,
153                                                           ResourceType type) {
154           if (type == ResourceType.ANY) {
155               throw new IllegalArgumentException(
156                   "Must specify a non-filter resource type for authorizeByResourceType");
157           }
158   
159           if (type == ResourceType.UNKNOWN) {
160               throw new IllegalArgumentException(
161                   "Unknown resource type");
162           }
163   
164           if (op == AclOperation.ANY) {
165               throw new IllegalArgumentException(
166                   "Must specify a non-filter operation type for authorizeByResourceType");
167           }
168   
169           if (op == AclOperation.UNKNOWN) {
170               throw new IllegalArgumentException(
171                   "Unknown operation type");
172           }
173       }
