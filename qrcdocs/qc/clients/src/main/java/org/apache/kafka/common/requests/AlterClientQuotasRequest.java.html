43           public Builder(Collection<ClientQuotaAlteration> entries, boolean validateOnly) {
44               super(ApiKeys.ALTER_CLIENT_QUOTAS);
45   
46               List<EntryData> entryData = new ArrayList<>(entries.size());
47               for (ClientQuotaAlteration entry : entries) {
48                   List<EntityData> entityData = new ArrayList<>(entry.entity().entries().size());
49                   for (Map.Entry<String, String> entityEntries : entry.entity().entries().entrySet()) {
50                       entityData.add(new EntityData()
51                               .setEntityType(entityEntries.getKey())
52                               .setEntityName(entityEntries.getValue()));
53                   }
54   
55                   List<OpData> opData = new ArrayList<>(entry.ops().size());
56                   for (ClientQuotaAlteration.Op op : entry.ops()) {
57                       opData.add(new OpData()
58                               .setKey(op.key())
59                               .setValue(op.value() == null ? 0.0 : op.value())
60                               .setRemove(op.value() == null));
61                   }
62   
63                   entryData.add(new EntryData()
64                           .setEntity(entityData)
65                           .setOps(opData));
66               }
67   
68               this.data = new AlterClientQuotasRequestData()
69                       .setEntries(entryData)
70                       .setValidateOnly(validateOnly);
71           }

91       public List<ClientQuotaAlteration> entries() {
92           List<ClientQuotaAlteration> entries = new ArrayList<>(data.entries().size());
93           for (EntryData entryData : data.entries()) {
94               Map<String, String> entity = new HashMap<>(entryData.entity().size());
95               for (EntityData entityData : entryData.entity()) {
96                   entity.put(entityData.entityType(), entityData.entityName());
97               }
98   
99               List<ClientQuotaAlteration.Op> ops = new ArrayList<>(entryData.ops().size());
100               for (OpData opData : entryData.ops()) {
101                   Double value = opData.remove() ? null : opData.value();
102                   ops.add(new ClientQuotaAlteration.Op(opData.key(), value));
103               }
104   
105               entries.add(new ClientQuotaAlteration(new ClientQuotaEntity(entity), ops));
106           }
107           return entries;
108       }
