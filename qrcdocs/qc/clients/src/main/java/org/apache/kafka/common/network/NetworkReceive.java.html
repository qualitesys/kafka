17   package org.apache.kafka.common.network;
18   
19   import org.apache.kafka.common.memory.MemoryPool;
20   import org.slf4j.Logger;
21   import org.slf4j.LoggerFactory;
22   
23   import java.io.EOFException;
24   import java.io.IOException;
25   import java.nio.ByteBuffer;
26   import java.nio.channels.ScatteringByteChannel;
27   
28   /**
29    * A size delimited Receive that consists of a 4 byte network-ordered size N followed by N bytes of content
30    */
31   public class NetworkReceive implements Receive {
32   
33       public final static String UNKNOWN_SOURCE = "";
34       public final static int UNLIMITED = -1;
35       private static final Logger log = LoggerFactory.getLogger(NetworkReceive.class);
36       private static final ByteBuffer EMPTY_BUFFER = ByteBuffer.allocate(0);
37   
38       private final String source;
39       private final ByteBuffer size;
40       private final int maxSize;
41       private final MemoryPool memoryPool;
42       private int requestedBufferSize = -1;
43       private ByteBuffer buffer;
44   
45   
46       public NetworkReceive(String source, ByteBuffer buffer) {
47           this.source = source;
48           this.buffer = buffer;
49           this.size = null;
50           this.maxSize = UNLIMITED;
51           this.memoryPool = MemoryPool.NONE;
52       }
53   
54       public NetworkReceive(String source) {
55           this.source = source;
56           this.size = ByteBuffer.allocate(4);
57           this.buffer = null;
58           this.maxSize = UNLIMITED;
59           this.memoryPool = MemoryPool.NONE;
60       }
61   
62       public NetworkReceive(int maxSize, String source) {
63           this.source = source;
64           this.size = ByteBuffer.allocate(4);
65           this.buffer = null;
66           this.maxSize = maxSize;
67           this.memoryPool = MemoryPool.NONE;
68       }
69   
70       public NetworkReceive(int maxSize, String source, MemoryPool memoryPool) {
71           this.source = source;
72           this.size = ByteBuffer.allocate(4);
73           this.buffer = null;
74           this.maxSize = maxSize;
75           this.memoryPool = memoryPool;
76       }
77   
78       public NetworkReceive() {
79           this(UNKNOWN_SOURCE);
80       }
81   
82       @Override
83       public String source() {
84           return source;
85       }
86   
87       @Override
88       public boolean complete() {
89           return !size.hasRemaining() && buffer != null && !buffer.hasRemaining();
90       }
91   
92       public long readFrom(ScatteringByteChannel channel) throws IOException {
93           int read = 0;
94           if (size.hasRemaining()) {
95               int bytesRead = channel.read(size);
96               if (bytesRead < 0)
97                   throw new EOFException();
98               read += bytesRead;
99               if (!size.hasRemaining()) {
100                   size.rewind();
101                   int receiveSize = size.getInt();
102                   if (receiveSize < 0)
103                       throw new InvalidReceiveException("Invalid receive (size = " + receiveSize + ")");
104                   if (maxSize != UNLIMITED && receiveSize > maxSize)
105                       throw new InvalidReceiveException("Invalid receive (size = " + receiveSize + " larger than " + maxSize + ")");
106                   requestedBufferSize = receiveSize; //may be 0 for some payloads (SASL)
107                   if (receiveSize == 0) {
108                       buffer = EMPTY_BUFFER;
109                   }
110               }
111           }
112           if (buffer == null && requestedBufferSize != -1) { //we know the size we want but havent been able to allocate it yet
113               buffer = memoryPool.tryAllocate(requestedBufferSize);
114               if (buffer == null)
115                   log.trace("Broker low on memory - could not allocate buffer of size {} for source {}", requestedBufferSize, source);
116           }
117           if (buffer != null) {
118               int bytesRead = channel.read(buffer);
119               if (bytesRead < 0)
120                   throw new EOFException();
121               read += bytesRead;
122           }
123   
124           return read;
125       }
126   
127       @Override
128       public boolean requiredMemoryAmountKnown() {
129           return requestedBufferSize != -1;
130       }
131   
132       @Override
133       public boolean memoryAllocated() {
134           return buffer != null;
135       }
136   
137   
138       @Override
139       public void close() throws IOException {
140           if (buffer != null && buffer != EMPTY_BUFFER) {
141               memoryPool.release(buffer);
142               buffer = null;
143           }
144       }
145   
146       public ByteBuffer payload() {
147           return this.buffer;
148       }
149   
150       public int bytesRead() {
151           if (buffer == null)
152               return size.position();
153           return buffer.position() + size.position();
154       }
155   
156       /**
157        * Returns the total size of the receive including payload and size buffer
158        * for use in metrics. This is consistent with {@link NetworkSend#size()}
159        */
160       public int size() {
161           return payload().limit() + size.limit();
162       }
163   
164   }
