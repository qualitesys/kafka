17   package org.apache.kafka.clients.admin;
18   
19   import java.util.Objects;
20   
21   /**
22    * Represents a range of versions that a particular broker supports for some feature.
23    */
24   public class SupportedVersionRange {
25       private final short minVersion;
26   
27       private final short maxVersion;
28   
29       /**
30        * Raises an exception unless the following conditions are met:
31        *  1 <= minVersion <= maxVersion.
32        *
33        * @param minVersion           The minimum version value.
34        * @param maxVersion           The maximum version value.
35        *
36        * @throws IllegalArgumentException   Raised when the condition described above is not met.
37        */
38       SupportedVersionRange(final short minVersion, final short maxVersion) {
39           if (minVersion < 1 || maxVersion < 1 || maxVersion < minVersion) {
40               throw new IllegalArgumentException(
41                   String.format(
42                       "Expected 1 <= minVersion <= maxVersion but received minVersion:%d, maxVersion:%d.",
43                       minVersion,
44                       maxVersion));
45           }
46           this.minVersion = minVersion;
47           this.maxVersion = maxVersion;
48       }
49   
50       public short minVersion() {
51           return minVersion;
52       }
53   
54       public short maxVersion() {
55           return maxVersion;
56       }
57   
58       @Override
59       public boolean equals(Object other) {
60           if (this == other) {
61               return true;
62           }
63   
64           if (other == null || getClass() != other.getClass()) {
65               return false;
66           }
67   
68           final SupportedVersionRange that = (SupportedVersionRange) other;
69           return this.minVersion == that.minVersion && this.maxVersion == that.maxVersion;
70       }
71   
72       @Override
73       public int hashCode() {
74           return Objects.hash(minVersion, maxVersion);
75       }
76   
77       @Override
78       public String toString() {
79           return String.format("SupportedVersionRange[min_version:%d, max_version:%d]", minVersion, maxVersion);
80       }
81   }
