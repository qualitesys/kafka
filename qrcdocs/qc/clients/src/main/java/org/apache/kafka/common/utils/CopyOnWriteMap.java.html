17   package org.apache.kafka.common.utils;
18   
19   import java.util.Collection;
20   import java.util.Collections;
21   import java.util.HashMap;
22   import java.util.Map;
23   import java.util.Set;
24   import java.util.concurrent.ConcurrentMap;
25   
26   /**
27    * A simple read-optimized map implementation that synchronizes only writes and does a full copy on each modification
28    */
29   public class CopyOnWriteMap<K, V> implements ConcurrentMap<K, V> {
30   
31       private volatile Map<K, V> map;
32   
33       public CopyOnWriteMap() {
34           this.map = Collections.emptyMap();
35       }
36   
37       public CopyOnWriteMap(Map<K, V> map) {
38           this.map = Collections.unmodifiableMap(map);
39       }
40   
41       @Override
42       public boolean containsKey(Object k) {
43           return map.containsKey(k);
44       }
45   
46       @Override
47       public boolean containsValue(Object v) {
48           return map.containsValue(v);
49       }
50   
51       @Override
52       public Set<java.util.Map.Entry<K, V>> entrySet() {
53           return map.entrySet();
54       }
55   
56       @Override
57       public V get(Object k) {
58           return map.get(k);
59       }
60   
61       @Override
62       public boolean isEmpty() {
63           return map.isEmpty();
64       }
65   
66       @Override
67       public Set<K> keySet() {
68           return map.keySet();
69       }
70   
71       @Override
72       public int size() {
73           return map.size();
74       }
75   
76       @Override
77       public Collection<V> values() {
78           return map.values();
79       }
80   
81       @Override
82       public synchronized void clear() {
83           this.map = Collections.emptyMap();
84       }
85   
86       @Override
87       public synchronized V put(K k, V v) {
88           Map<K, V> copy = new HashMap<K, V>(this.map);
89           V prev = copy.put(k, v);
90           this.map = Collections.unmodifiableMap(copy);
91           return prev;
92       }
93   
94       @Override
95       public synchronized void putAll(Map<? extends K, ? extends V> entries) {
96           Map<K, V> copy = new HashMap<K, V>(this.map);
97           copy.putAll(entries);
98           this.map = Collections.unmodifiableMap(copy);
99       }
100   
101       @Override
102       public synchronized V remove(Object key) {
103           Map<K, V> copy = new HashMap<K, V>(this.map);
104           V prev = copy.remove(key);
105           this.map = Collections.unmodifiableMap(copy);
106           return prev;
107       }
108   
109       @Override
110       public synchronized V putIfAbsent(K k, V v) {
111           if (!containsKey(k))
112               return put(k, v);
113           else
114               return get(k);
115       }
116   
117       @Override
118       public synchronized boolean remove(Object k, Object v) {
119           if (containsKey(k) && get(k).equals(v)) {
120               remove(k);
121               return true;
122           } else {
123               return false;
124           }
125       }
126   
127       @Override
128       public synchronized boolean replace(K k, V original, V replacement) {
129           if (containsKey(k) && get(k).equals(original)) {
130               put(k, replacement);
131               return true;
132           } else {
133               return false;
134           }
135       }
136   
137       @Override
138       public synchronized V replace(K k, V v) {
139           if (containsKey(k)) {
140               return put(k, v);
141           } else {
142               return null;
143           }
144       }
145   
146   }
