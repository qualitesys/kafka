17   package org.apache.kafka.common.security.kerberos;
18   
19   import java.io.IOException;
20   import java.util.Locale;
21   import java.util.regex.Matcher;
22   import java.util.regex.Pattern;
23   
24   /**
25    * An encoding of a rule for translating kerberos names.
26    */
27   class KerberosRule {
28   
29       /**
30        * A pattern that matches a string without '$' and then a single
31        * parameter with $n.
32        */
33       private static final Pattern PARAMETER_PATTERN = Pattern.compile("([^$]*)(\\$(\\d*))?");
34   
35       /**
36        * A pattern that recognizes simple/non-simple names.
37        */
38       private static final Pattern NON_SIMPLE_PATTERN = Pattern.compile("[/@]");
39   
40       private final String defaultRealm;
41       private final boolean isDefault;
42       private final int numOfComponents;
43       private final String format;
44       private final Pattern match;
45       private final Pattern fromPattern;
46       private final String toPattern;
47       private final boolean repeat;
48       private final boolean toLowerCase;
49       private final boolean toUpperCase;
50   
51       KerberosRule(String defaultRealm) {
52           this.defaultRealm = defaultRealm;
53           isDefault = true;
54           numOfComponents = 0;
55           format = null;
56           match = null;
57           fromPattern = null;
58           toPattern = null;
59           repeat = false;
60           toLowerCase = false;
61           toUpperCase = false;
62       }
63   
64       KerberosRule(String defaultRealm, int numOfComponents, String format, String match, String fromPattern,
65                    String toPattern, boolean repeat, boolean toLowerCase, boolean toUpperCase) {
66           this.defaultRealm = defaultRealm;
67           isDefault = false;
68           this.numOfComponents = numOfComponents;
69           this.format = format;
70           this.match = match == null ? null : Pattern.compile(match);
71           this.fromPattern =
72                   fromPattern == null ? null : Pattern.compile(fromPattern);
73           this.toPattern = toPattern;
74           this.repeat = repeat;
75           this.toLowerCase = toLowerCase;
76           this.toUpperCase = toUpperCase;
77       }
78   
79       @Override
80       public String toString() {
81           StringBuilder buf = new StringBuilder();
82           if (isDefault) {
83               buf.append("DEFAULT");
84           } else {
85               buf.append("RULE:[");
86               buf.append(numOfComponents);
87               buf.append(':');
88               buf.append(format);
89               buf.append(']');
90               if (match != null) {
91                   buf.append('(');
92                   buf.append(match);
93                   buf.append(')');
94               }
95               if (fromPattern != null) {
96                   buf.append("s/");
97                   buf.append(fromPattern);
98                   buf.append('/');
99                   buf.append(toPattern);
100                   buf.append('/');
101                   if (repeat) {
102                       buf.append('g');
103                   }
104               }
105               if (toLowerCase) {
106                   buf.append("/L");
107               }
108               if (toUpperCase) {
109                   buf.append("/U");
110               }
111           }
112           return buf.toString();
113       }
114   
115       /**
116        * Replace the numbered parameters of the form $n where n is from 0 to
117        * the length of params - 1. Normal text is copied directly and $n is replaced
118        * by the corresponding parameter.
119        * @param format the string to replace parameters again
120        * @param params the list of parameters
121        * @return the generated string with the parameter references replaced.
122        * @throws BadFormatString
123        */
124       static String replaceParameters(String format,
125                                       String[] params) throws BadFormatString {
126           Matcher match = PARAMETER_PATTERN.matcher(format);
127           int start = 0;
128           StringBuilder result = new StringBuilder();
129           while (start < format.length() && match.find(start)) {
130               result.append(match.group(1));
131               String paramNum = match.group(3);
132               if (paramNum != null) {
133                   try {
134                       int num = Integer.parseInt(paramNum);
135                       if (num < 0 || num >= params.length) {
136                           throw new BadFormatString("index " + num + " from " + format +
137                                   " is outside of the valid range 0 to " +
138                                   (params.length - 1));
139                       }
140                       result.append(params[num]);
141                   } catch (NumberFormatException nfe) {
142                       throw new BadFormatString("bad format in username mapping in " +
143                               paramNum, nfe);
144                   }
145   
146               }
147               start = match.end();
148           }
149           return result.toString();
150       }
151   
152       /**
153        * Replace the matches of the from pattern in the base string with the value
154        * of the to string.
155        * @param base the string to transform
156        * @param from the pattern to look for in the base string
157        * @param to the string to replace matches of the pattern with
158        * @param repeat whether the substitution should be repeated
159        * @return
160        */
161       static String replaceSubstitution(String base, Pattern from, String to,
162                                         boolean repeat) {
163           Matcher match = from.matcher(base);
164           if (repeat) {
165               return match.replaceAll(to);
166           } else {
167               return match.replaceFirst(to);
168           }
169       }
170   
171       /**
172        * Try to apply this rule to the given name represented as a parameter
173        * array.
174        * @param params first element is the realm, second and later elements are
175        *        are the components of the name "a/b@FOO" -> {"FOO", "a", "b"}
176        * @return the short name if this rule applies or null
177        * @throws IOException throws if something is wrong with the rules
178        */
179       String apply(String[] params) throws IOException {
180           String result = null;
181           if (isDefault) {
182               if (defaultRealm.equals(params[0])) {
183                   result = params[1];
184               }
185           } else if (params.length - 1 == numOfComponents) {
186               String base = replaceParameters(format, params);
187               if (match == null || match.matcher(base).matches()) {
188                   if (fromPattern == null) {
189                       result = base;
190                   } else {
191                       result = replaceSubstitution(base, fromPattern, toPattern,  repeat);
192                   }
193               }
194           }
195           if (result != null && NON_SIMPLE_PATTERN.matcher(result).find()) {
196               throw new NoMatchingRule("Non-simple name " + result + " after auth_to_local rule " + this);
197           }
198           if (toLowerCase && result != null) {
199               result = result.toLowerCase(Locale.ENGLISH);
200           } else if (toUpperCase && result != null) {
201               result = result.toUpperCase(Locale.ENGLISH);
202           }
203   
204           return result;
205       }
206   }
