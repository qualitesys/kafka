18   package org.apache.kafka.common.security.oauthbearer.secured;
19   
20   import org.apache.kafka.common.security.oauthbearer.OAuthBearerToken;
21   
22   /**
23    * An instance of <code>AccessTokenValidator</code> acts as a function object that, given an access
24    * token in base-64 encoded JWT format, can parse the data, perform validation, and construct an
25    * {@link OAuthBearerToken} for use by the caller.
26    *
27    * The primary reason for this abstraction is that client and broker may have different libraries
28    * available to them to perform these operations. Additionally, the exact steps for validation may
29    * differ between implementations. To put this more concretely: the implementation in the Kafka
30    * client does not have bundled a robust library to perform this logic, and it is not the
31    * responsibility of the client to perform vigorous validation. However, the Kafka broker ships with
32    * a richer set of library dependencies that can perform more substantial validation and is also
33    * expected to perform a trust-but-verify test of the access token's signature.
34    *
35    * See:
36    *
37    * <ul>
38    *     <li><a href="https://datatracker.ietf.org/doc/html/rfc6749#section-1.4">RFC 6749, Section 1.4</a></li>
39    *     <li><a href="https://datatracker.ietf.org/doc/html/rfc6750#section-2.1">RFC 6750, Section 2.1</a></li>
40    *     <li><a href="https://datatracker.ietf.org/doc/html/draft-ietf-oauth-access-token-jwt">RFC 6750, Section 2.1</a></li>
41    * </ul>
42    *
43    * @see LoginAccessTokenValidator A basic AccessTokenValidator used by client-side login
44    *                                authentication
45    * @see ValidatorAccessTokenValidator A more robust AccessTokenValidator that is used on the broker
46    *                                    to validate the token's contents and verify the signature
47    */
48   
49   public interface AccessTokenValidator {
50   
51       /**
52        * Accepts an OAuth JWT access token in base-64 encoded format, validates, and returns an
53        * OAuthBearerToken.
54        *
55        * @param accessToken Non-<code>null</code> JWT access token
56        *
57        * @return {@link OAuthBearerToken}
58        *
59        * @throws ValidateException Thrown on errors performing validation of given token
60        */
61   
62       OAuthBearerToken validate(String accessToken) throws ValidateException;
63   
64   }
