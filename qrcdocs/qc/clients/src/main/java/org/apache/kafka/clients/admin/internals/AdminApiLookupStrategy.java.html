19   import org.apache.kafka.common.requests.AbstractRequest;
20   import org.apache.kafka.common.requests.AbstractResponse;
21   
22   import java.util.Collections;
23   import java.util.List;
24   import java.util.Map;
25   import java.util.Set;
26   
27   import static java.util.Collections.emptyMap;
28   import static java.util.Collections.singletonMap;

30   public interface AdminApiLookupStrategy<T> {
31   
32       /**
33        * Define the scope of a given key for lookup. Key lookups are complicated
34        * by the need to accommodate different batching mechanics. For example,
35        * a `Metadata` request supports arbitrary batching of topic partitions in
36        * order to discover partitions leaders. This can be supported by returning
37        * a single scope object for all keys.
38        *
39        * On the other hand, `FindCoordinator` requests only support lookup of a
40        * single key. This can be supported by returning a different scope object
41        * for each lookup key.
42        *
43        * Note that if the {@link ApiRequestScope#destinationBrokerId()} maps to
44        * a specific brokerId, then lookup will be skipped. See the use of
45        * {@link StaticBrokerStrategy} in {@link DescribeProducersHandler} for
46        * an example of this usage.
47        *
48        * @param key the lookup key
49        *
50        * @return request scope indicating how lookup requests can be batched together
51        */
52       ApiRequestScope lookupScope(T key);
53   
54       /**
55        * Build the lookup request for a set of keys. The grouping of the keys is controlled
56        * through {@link #lookupScope(Object)}. In other words, each set of keys that map
57        * to the same request scope object will be sent to this method.
58        *
59        * @param keys the set of keys that require lookup
60        *
61        * @return a builder for the lookup request
62        */
63       AbstractRequest.Builder<?> buildRequest(Set<T> keys);
64   
65       /**
66        * Callback that is invoked when a lookup request returns successfully. The handler
67        * should parse the response, check for errors, and return a result indicating
68        * which keys were mapped to a brokerId successfully and which keys received
69        * a fatal error (e.g. a topic authorization failure).
70        *
71        * Note that keys which receive a retriable error should be left out of the
72        * result. They will be retried automatically. For example, if the response of
73        * `FindCoordinator` request indicates an unavailable coordinator, then the key
74        * should be left out of the result so that the request will be retried.
75        *
76        * @param keys the set of keys from the associated request
77        * @param response the response received from the broker
78        *
79        * @return a result indicating which keys mapped successfully to a brokerId and
80        *         which encountered a fatal error
81        */
82       LookupResult<T> handleResponse(Set<T> keys, AbstractResponse response);
83   
84       class LookupResult<K> {
85           // This is the set of keys that have been completed by the lookup phase itself.
86           // The driver will not attempt lookup or fulfillment for completed keys.
87           public final List<K> completedKeys;
88   
89           // This is the set of keys that have been mapped to a specific broker for
90           // fulfillment of the API request.
91           public final Map<K, Integer> mappedKeys;
92   
93           // This is the set of keys that have encountered a fatal error during the lookup
94           // phase. The driver will not attempt lookup or fulfillment for failed keys.
95           public final Map<K, Throwable> failedKeys;
96   
97           public LookupResult(
98               Map<K, Throwable> failedKeys,
99               Map<K, Integer> mappedKeys
100           ) {
101               this(Collections.emptyList(), failedKeys, mappedKeys);
102           }
103   
104           public LookupResult(
105               List<K> completedKeys,
106               Map<K, Throwable> failedKeys,
107               Map<K, Integer> mappedKeys
108           ) {
109               this.completedKeys = Collections.unmodifiableList(completedKeys);
110               this.failedKeys = Collections.unmodifiableMap(failedKeys);
111               this.mappedKeys = Collections.unmodifiableMap(mappedKeys);
112           }
113   
114           static <K> LookupResult<K> empty() {
115               return new LookupResult<>(emptyMap(), emptyMap());
116           }
117   
118           static <K> LookupResult<K> failed(K key, Throwable exception) {
119               return new LookupResult<>(singletonMap(key, exception), emptyMap());
120           }
121   
122           static <K> LookupResult<K> mapped(K key, Integer brokerId) {
123               return new LookupResult<>(emptyMap(), singletonMap(key, brokerId));
124           }
125   
126       }
127   
128   }
