18   package org.apache.kafka.common.utils;
19   
20   import java.util.Collection;
21   import java.util.ConcurrentModificationException;
22   import java.util.Iterator;
23   import java.util.Objects;
24   
25   /**
26    * An iterator that cycles through the {@code Iterator} of a {@code Collection}
27    * indefinitely. Useful for tasks such as round-robin load balancing. This class
28    * does not provide thread-safe access. This {@code Iterator} supports
29    * {@code null} elements in the underlying {@code Collection}. This
30    * {@code Iterator} does not support any modification to the underlying
31    * {@code Collection} after it has been wrapped by this class. Changing the
32    * underlying {@code Collection} may cause a
33    * {@link ConcurrentModificationException} or some other undefined behavior.
34    */
35   public class CircularIterator<T> implements Iterator<T> {
36   
37       private final Iterable<T> iterable;
38       private Iterator<T> iterator;
39       private T nextValue;
40   
41       /**
42        * Create a new instance of a CircularIterator. The ordering of this
43        * Iterator will be dictated by the Iterator returned by Collection itself.
44        *
45        * @param col The collection to iterate indefinitely
46        *
47        * @throws NullPointerException if col is {@code null}
48        * @throws IllegalArgumentException if col is empty.
49        */
50       public CircularIterator(final Collection<T> col) {
51           this.iterable = Objects.requireNonNull(col);
52           this.iterator = col.iterator();
53           if (col.isEmpty()) {
54               throw new IllegalArgumentException("CircularIterator can only be used on non-empty lists");
55           }
56           this.nextValue = advance();
57       }
58   
59       /**
60        * Returns true since the iteration will forever cycle through the provided
61        * {@code Collection}.
62        *
63        * @return Always true
64        */
65       @Override
66       public boolean hasNext() {
67           return true;
68       }
69   
70       @Override
71       public T next() {
72           final T next = nextValue;
73           nextValue = advance();
74           return next;
75       }
76   
77       /**
78        * Return the next value in the {@code Iterator}, restarting the
79        * {@code Iterator} if necessary.
80        *
81        * @return The next value in the iterator
82        */
83       private T advance() {
84           if (!iterator.hasNext()) {
85               iterator = iterable.iterator();
86           }
87           return iterator.next();
88       }
89   
90       /**
91        * Peek at the next value in the Iterator. Calling this method multiple
92        * times will return the same element without advancing this Iterator. The
93        * value returned by this method will be the next item returned by
94        * {@code next()}.
95        *
96        * @return The next value in this {@code Iterator}
97        */
98       public T peek() {
99           return nextValue;
100       }
101   
102       @Override
103       public void remove() {
104           throw new UnsupportedOperationException();
105       }
106   }
