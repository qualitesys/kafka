37   /**
38    * A {@code CallbackHandler} that recognizes
39    * {@link OAuthBearerValidatorCallback} and validates an unsecured OAuth 2
40    * bearer token. It requires there to be an <code>"exp" (Expiration Time)</code>
41    * claim of type Number. If <code>"iat" (Issued At)</code> or
42    * <code>"nbf" (Not Before)</code> claims are present each must be a number that
43    * precedes the Expiration Time claim, and if both are present the Not Before
44    * claim must not precede the Issued At claim. It also accepts the following
45    * options, none of which are required:
46    * <ul>
47    * <li>{@code unsecuredValidatorPrincipalClaimName} set to a non-empty value if
48    * you wish a particular String claim holding a principal name to be checked for
49    * existence; the default is to check for the existence of the '{@code sub}'
50    * claim</li>
51    * <li>{@code unsecuredValidatorScopeClaimName} set to a custom claim name if
52    * you wish the name of the String or String List claim holding any token scope
53    * to be something other than '{@code scope}'</li>
54    * <li>{@code unsecuredValidatorRequiredScope} set to a space-delimited list of
55    * scope values if you wish the String/String List claim holding the token scope
56    * to be checked to make sure it contains certain values</li>
57    * <li>{@code unsecuredValidatorAllowableClockSkewMs} set to a positive integer
58    * value if you wish to allow up to some number of positive milliseconds of
59    * clock skew (the default is 0)</li>
60    * <ul>
61    * For example:
62    *
63    * <pre>
64    * KafkaServer {
65    *      org.apache.kafka.common.security.oauthbearer.OAuthBearerLoginModule Required
66    *      unsecuredLoginStringClaim_sub="thePrincipalName"
67    *      unsecuredLoginListClaim_scope=",KAFKA_BROKER,LOGIN_TO_KAFKA"
68    *      unsecuredValidatorRequiredScope="LOGIN_TO_KAFKA"
69    *      unsecuredValidatorAllowableClockSkewMs="3000";
70    * };
71    * </pre>
72    * It also recognizes {@link OAuthBearerExtensionsValidatorCallback} and validates every extension passed to it.
73    *
74    * This class is the default when the SASL mechanism is OAUTHBEARER and no value
75    * is explicitly set via the
76    * {@code listener.name.sasl_[plaintext|ssl].oauthbearer.sasl.server.callback.handler.class}
77    * broker configuration property.
78    * It is worth noting that this class is not suitable for production use due to the use of unsecured JWT tokens and
79    * validation of every given extension.
80    */
81   public class OAuthBearerUnsecuredValidatorCallbackHandler implements AuthenticateCallbackHandler {
82       private static final Logger log = LoggerFactory.getLogger(OAuthBearerUnsecuredValidatorCallbackHandler.class);
83       private static final String OPTION_PREFIX = "unsecuredValidator";
84       private static final String PRINCIPAL_CLAIM_NAME_OPTION = OPTION_PREFIX + "PrincipalClaimName";
85       private static final String SCOPE_CLAIM_NAME_OPTION = OPTION_PREFIX + "ScopeClaimName";
86       private static final String REQUIRED_SCOPE_OPTION = OPTION_PREFIX + "RequiredScope";
87       private static final String ALLOWABLE_CLOCK_SKEW_MILLIS_OPTION = OPTION_PREFIX + "AllowableClockSkewMs";
88       private Time time = Time.SYSTEM;
89       private Map<String, String> moduleOptions = null;
90       private boolean configured = false;
91   
92       /**
93        * For testing
94        *
95        * @param time
96        *            the mandatory time to set
97        */
98       void time(Time time) {
99           this.time = Objects.requireNonNull(time);
100       }
101   
102       /**
103        * Return true if this instance has been configured, otherwise false
104        *
105        * @return true if this instance has been configured, otherwise false
106        */
107       public boolean configured() {
108           return configured;
109       }
110   
111       @SuppressWarnings("unchecked")
112       @Override
113       public void configure(Map<String, ?> configs, String saslMechanism, List<AppConfigurationEntry> jaasConfigEntries) {
114           if (!OAuthBearerLoginModule.OAUTHBEARER_MECHANISM.equals(saslMechanism))
115               throw new IllegalArgumentException(String.format("Unexpected SASL mechanism: %s", saslMechanism));
116           if (Objects.requireNonNull(jaasConfigEntries).size() != 1 || jaasConfigEntries.get(0) == null)
117               throw new IllegalArgumentException(
118                       String.format("Must supply exactly 1 non-null JAAS mechanism configuration (size was %d)",
119                               jaasConfigEntries.size()));
120           final Map<String, String> unmodifiableModuleOptions = Collections
121                   .unmodifiableMap((Map<String, String>) jaasConfigEntries.get(0).getOptions());
122           this.moduleOptions = unmodifiableModuleOptions;
123           configured = true;
124       }
125   
126       @Override
127       public void handle(Callback[] callbacks) throws UnsupportedCallbackException {
128           if (!configured())
129               throw new IllegalStateException("Callback handler not configured");
130           for (Callback callback : callbacks) {
131               if (callback instanceof OAuthBearerValidatorCallback) {
132                   OAuthBearerValidatorCallback validationCallback = (OAuthBearerValidatorCallback) callback;
133                   try {
134                       handleCallback(validationCallback);
135                   } catch (OAuthBearerIllegalTokenException e) {
136                       OAuthBearerValidationResult failureReason = e.reason();
137                       String failureScope = failureReason.failureScope();
138                       validationCallback.error(failureScope != null ? "insufficient_scope" : "invalid_token",
139                               failureScope, failureReason.failureOpenIdConfig());
140                   }
141               } else if (callback instanceof OAuthBearerExtensionsValidatorCallback) {
142                   OAuthBearerExtensionsValidatorCallback extensionsCallback = (OAuthBearerExtensionsValidatorCallback) callback;
143                   extensionsCallback.inputExtensions().map().forEach((extensionName, v) -> extensionsCallback.valid(extensionName));
144               } else
145                   throw new UnsupportedCallbackException(callback);
146           }
147       }
148   
149       @Override
150       public void close() {
151           // empty
152       }
153   
154       private void handleCallback(OAuthBearerValidatorCallback callback) {
155           String tokenValue = callback.tokenValue();
156           if (tokenValue == null)
157               throw new IllegalArgumentException("Callback missing required token value");
158           String principalClaimName = principalClaimName();
159           String scopeClaimName = scopeClaimName();
160           List<String> requiredScope = requiredScope();
161           int allowableClockSkewMs = allowableClockSkewMs();
162           OAuthBearerUnsecuredJws unsecuredJwt = new OAuthBearerUnsecuredJws(tokenValue, principalClaimName,
163                   scopeClaimName);
164           long now = time.milliseconds();
165           OAuthBearerValidationUtils
166                   .validateClaimForExistenceAndType(unsecuredJwt, true, principalClaimName, String.class)
167                   .throwExceptionIfFailed();
168           OAuthBearerValidationUtils.validateIssuedAt(unsecuredJwt, false, now, allowableClockSkewMs)
169                   .throwExceptionIfFailed();
170           OAuthBearerValidationUtils.validateExpirationTime(unsecuredJwt, now, allowableClockSkewMs)
171                   .throwExceptionIfFailed();
172           OAuthBearerValidationUtils.validateTimeConsistency(unsecuredJwt).throwExceptionIfFailed();
173           OAuthBearerValidationUtils.validateScope(unsecuredJwt, requiredScope).throwExceptionIfFailed();
174           log.info("Successfully validated token with principal {}: {}", unsecuredJwt.principalName(),
175                   unsecuredJwt.claims());
176           callback.token(unsecuredJwt);
177       }
178   
179       private String principalClaimName() {
180           String principalClaimNameValue = option(PRINCIPAL_CLAIM_NAME_OPTION);
181           return Utils.isBlank(principalClaimNameValue) ? "sub" : principalClaimNameValue.trim();
182       }
183   
184       private String scopeClaimName() {
185           String scopeClaimNameValue = option(SCOPE_CLAIM_NAME_OPTION);
186           return Utils.isBlank(scopeClaimNameValue) ? "scope" : scopeClaimNameValue.trim();
187       }
188   
189       private List<String> requiredScope() {
190           String requiredSpaceDelimitedScope = option(REQUIRED_SCOPE_OPTION);
191           return Utils.isBlank(requiredSpaceDelimitedScope) ? Collections.emptyList() : OAuthBearerScopeUtils.parseScope(requiredSpaceDelimitedScope.trim());
192       }
193   
194       private int allowableClockSkewMs() {
195           String allowableClockSkewMsValue = option(ALLOWABLE_CLOCK_SKEW_MILLIS_OPTION);
196           int allowableClockSkewMs = 0;
197           try {
198               allowableClockSkewMs = Utils.isBlank(allowableClockSkewMsValue) ? 0 : Integer.parseInt(allowableClockSkewMsValue.trim());
199           } catch (NumberFormatException e) {
200               throw new OAuthBearerConfigException(e.getMessage(), e);
201           }
202           if (allowableClockSkewMs < 0) {
203               throw new OAuthBearerConfigException(
204                       String.format("Allowable clock skew millis must not be negative: %s", allowableClockSkewMsValue));
205           }
206           return allowableClockSkewMs;
207       }
208   
209       private String option(String key) {
210           if (!configured)
211               throw new IllegalStateException("Callback handler not configured");
212           return moduleOptions.get(Objects.requireNonNull(key));
213       }
