17   package org.apache.kafka.common.config.provider;
18   
19   import org.apache.kafka.common.config.ConfigData;
20   import org.apache.kafka.common.config.ConfigException;
21   import org.slf4j.Logger;
22   import org.slf4j.LoggerFactory;
23   
24   import java.io.File;
25   import java.io.IOException;
26   import java.nio.charset.StandardCharsets;
27   import java.nio.file.Files;
28   import java.nio.file.Path;
29   import java.util.Map;
30   import java.util.Set;
31   import java.util.function.Predicate;
32   import java.util.stream.Collectors;
33   import java.util.stream.Stream;
34   
35   import static java.util.Collections.emptyMap;
36   
37   /**
38    * An implementation of {@link ConfigProvider} based on a directory of files.
39    * Property keys correspond to the names of the regular (i.e. non-directory)
40    * files in a directory given by the path parameter.
41    * Property values are taken from the file contents corresponding to each key.
42    */
43   public class DirectoryConfigProvider implements ConfigProvider {
44   
45       private static final Logger log = LoggerFactory.getLogger(DirectoryConfigProvider.class);
46   
47       @Override
48       public void configure(Map<String, ?> configs) { }
49   
50       @Override
51       public void close() throws IOException { }
52   
53       /**
54        * Retrieves the data contained in regular files in the directory given by {@code path}.
55        * Non-regular files (such as directories) in the given directory are silently ignored.
56        * @param path the directory where data files reside.
57        * @return the configuration data.
58        */
59       @Override
60       public ConfigData get(String path) {
61           return get(path, Files::isRegularFile);
62       }
63   
64       /**
65        * Retrieves the data contained in the regular files named by {@code keys} in the directory given by {@code path}.
66        * Non-regular files (such as directories) in the given directory are silently ignored.
67        * @param path the directory where data files reside.
68        * @param keys the keys whose values will be retrieved.
69        * @return the configuration data.
70        */
71       @Override
72       public ConfigData get(String path, Set<String> keys) {
73           return get(path, pathname ->
74                   Files.isRegularFile(pathname)
75                           && keys.contains(pathname.getFileName().toString()));
76       }
77   
78       private static ConfigData get(String path, Predicate<Path> fileFilter) {
79           Map<String, String> map = emptyMap();
80           if (path != null && !path.isEmpty()) {
81               Path dir = new File(path).toPath();
82               if (!Files.isDirectory(dir)) {
83                   log.warn("The path {} is not a directory", path);
84               } else {
85                   try (Stream<Path> stream = Files.list(dir)) {
86                       map = stream
87                           .filter(fileFilter)
88                           .collect(Collectors.toMap(
89                               p -> p.getFileName().toString(),
90                               p -> read(p)));
91                   } catch (IOException e) {
92                       log.error("Could not list directory {}", dir, e);
93                       throw new ConfigException("Could not list directory " + dir);
94                   }
95               }
96           }
97           return new ConfigData(map);
98       }
99   
100       private static String read(Path path) {
101           try {
102               return new String(Files.readAllBytes(path), StandardCharsets.UTF_8);
103           } catch (IOException e) {
104               log.error("Could not read file {} for property {}", path, path.getFileName(), e);
105               throw new ConfigException("Could not read file " + path + " for property " + path.getFileName());
106           }
107       }
108   
109   }
