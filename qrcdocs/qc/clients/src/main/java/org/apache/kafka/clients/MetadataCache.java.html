41   /**
42    * An internal mutable cache of nodes, topics, and partitions in the Kafka cluster. This keeps an up-to-date Cluster
43    * instance which is optimized for read access.
44    */

68       private MetadataCache(String clusterId,
69                             Map<Integer, Node> nodes,
70                             Collection<PartitionMetadata> partitions,
71                             Set<String> unauthorizedTopics,
72                             Set<String> invalidTopics,
73                             Set<String> internalTopics,
74                             Node controller,
75                             Map<String, Uuid> topicIds,
76                             Cluster clusterInstance) {
77           this.clusterId = clusterId;
78           this.nodes = nodes;
79           this.unauthorizedTopics = unauthorizedTopics;
80           this.invalidTopics = invalidTopics;

92           } else {
93               this.clusterInstance = clusterInstance;
94           }

121       /**
122        * Merges the metadata cache's contents with the provided metadata, returning a new metadata cache. The provided
123        * metadata is presumed to be more recent than the cache's metadata, and therefore all overlapping metadata will
124        * be overridden.
125        *
126        * @param newClusterId the new cluster Id
127        * @param newNodes the new set of nodes
128        * @param addPartitions partitions to add
129        * @param addUnauthorizedTopics unauthorized topics to add
130        * @param addInternalTopics internal topics to add
131        * @param newController the new controller node
132        * @param topicIds the mapping from topic name to topic ID from the MetadataResponse
133        * @param retainTopic returns whether a topic's metadata should be retained
134        * @return the merged metadata cache
135        */
136       MetadataCache mergeWith(String newClusterId,
137                               Map<Integer, Node> newNodes,
138                               Collection<PartitionMetadata> addPartitions,
139                               Set<String> addUnauthorizedTopics,
140                               Set<String> addInvalidTopics,
141                               Set<String> addInternalTopics,
142                               Node newController,
143                               Map<String, Uuid> topicIds,
144                               BiPredicate<String, Boolean> retainTopic) {
145   
146           Predicate<String> shouldRetainTopic = topic -> retainTopic.test(topic, internalTopics.contains(topic));
147   
148           Map<TopicPartition, PartitionMetadata> newMetadataByPartition = new HashMap<>(addPartitions.size());
149   
150           // We want the most recent topic ID. We start with the previous ID stored for retained topics and then
151           // update with newest information from the MetadataResponse. We always take the latest state, removing existing
152           // topic IDs if the latest state contains the topic name but not a topic ID.
153           Map<String, Uuid> newTopicIds = topicIds.entrySet().stream()
154                   .filter(entry -> shouldRetainTopic.test(entry.getKey()))
155                   .collect(Collectors.toMap(Map.Entry::getKey, Map.Entry::getValue));
156   
157           for (PartitionMetadata partition : addPartitions) {
158               newMetadataByPartition.put(partition.topicPartition, partition);
159               Uuid id = topicIds.get(partition.topic());
160               if (id != null)
161                   newTopicIds.put(partition.topic(), id);
162               else
163                   // Remove if the latest metadata does not have a topic ID
164                   newTopicIds.remove(partition.topic());
165           }
166           for (Map.Entry<TopicPartition, PartitionMetadata> entry : metadataByPartition.entrySet()) {
167               if (shouldRetainTopic.test(entry.getKey().topic())) {
168                   newMetadataByPartition.putIfAbsent(entry.getKey(), entry.getValue());
169               }
170           }
171   
172           Set<String> newUnauthorizedTopics = fillSet(addUnauthorizedTopics, unauthorizedTopics, shouldRetainTopic);
173           Set<String> newInvalidTopics = fillSet(addInvalidTopics, invalidTopics, shouldRetainTopic);
174           Set<String> newInternalTopics = fillSet(addInternalTopics, internalTopics, shouldRetainTopic);
175   
176           return new MetadataCache(newClusterId, newNodes, newMetadataByPartition.values(), newUnauthorizedTopics,
177                   newInvalidTopics, newInternalTopics, newController, newTopicIds);
178       }
179   
180       /**
181        * Copies {@code baseSet} and adds all non-existent elements in {@code fillSet} such that {@code predicate} is true.
182        * In other words, all elements of {@code baseSet} will be contained in the result, with additional non-overlapping
183        * elements in {@code fillSet} where the predicate is true.
184        *
185        * @param baseSet the base elements for the resulting set
186        * @param fillSet elements to be filled into the resulting set
187        * @param predicate tested against the fill set to determine whether elements should be added to the base set
188        */
189       private static <T> Set<T> fillSet(Set<T> baseSet, Set<T> fillSet, Predicate<T> predicate) {
190           Set<T> result = new HashSet<>(baseSet);
191           for (T element : fillSet) {
192               if (predicate.test(element)) {
193                   result.add(element);
194               }
195           }
196           return result;
197       }

199       private void computeClusterView() {
200           List<PartitionInfo> partitionInfos = metadataByPartition.values()
201                   .stream()
202                   .map(metadata -> MetadataResponse.toPartitionInfo(metadata, nodes))
203                   .collect(Collectors.toList());
204           this.clusterInstance = new Cluster(clusterId, nodes.values(), partitionInfos, unauthorizedTopics,
205                   invalidTopics, internalTopics, controller, topicIds);
206       }

208       static MetadataCache bootstrap(List<InetSocketAddress> addresses) {
209           Map<Integer, Node> nodes = new HashMap<>();
210           int nodeId = -1;
211           for (InetSocketAddress address : addresses) {
212               nodes.put(nodeId, new Node(nodeId, address.getHostString(), address.getPort()));
213               nodeId--;
214           }
215           return new MetadataCache(null, nodes, Collections.emptyList(),
216                   Collections.emptySet(), Collections.emptySet(), Collections.emptySet(),
217                   null, Collections.emptyMap(), Cluster.bootstrap(addresses));
218       }

220       static MetadataCache empty() {
221           return new MetadataCache(null, Collections.emptyMap(), Collections.emptyList(),
222                   Collections.emptySet(), Collections.emptySet(), Collections.emptySet(), null, Collections.emptyMap(), Cluster.empty());
223       }
