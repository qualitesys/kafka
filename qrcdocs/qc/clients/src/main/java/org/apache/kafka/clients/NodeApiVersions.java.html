38   /**
39    * An internal class which represents the API versions supported by a particular node.
40    */
41   public class NodeApiVersions {
42   
43       // A map of the usable versions of each API, keyed by the ApiKeys instance
44       private final Map<ApiKeys, ApiVersion> supportedVersions = new EnumMap<>(ApiKeys.class);
45   
46       // List of APIs which the broker supports, but which are unknown to the client
47       private final List<ApiVersion> unknownApis = new ArrayList<>();
48   
49       private final Map<String, SupportedVersionRange> supportedFeatures;
50   
51       /**
52        * Create a NodeApiVersions object with the current ApiVersions.
53        *
54        * @return A new NodeApiVersions object.
55        */
56       public static NodeApiVersions create() {
57           return create(Collections.emptyList());
58       }
59   
60       /**
61        * Create a NodeApiVersions object.
62        *
63        * @param overrides API versions to override. Any ApiVersion not specified here will be set to the current client
64        *                  value.
65        * @return A new NodeApiVersions object.
66        */
67       public static NodeApiVersions create(Collection<ApiVersion> overrides) {
68           List<ApiVersion> apiVersions = new LinkedList<>(overrides);
69           for (ApiKeys apiKey : ApiKeys.zkBrokerApis()) {
70               boolean exists = false;
71               for (ApiVersion apiVersion : apiVersions) {
72                   if (apiVersion.apiKey() == apiKey.id) {
73                       exists = true;
74                       break;
75                   }
76               }
77               if (!exists) apiVersions.add(ApiVersionsResponse.toApiVersion(apiKey));
78           }
79           return new NodeApiVersions(apiVersions, Collections.emptyList());
80       }
81   
82   
83       /**
84        * Create a NodeApiVersions object with a single ApiKey. It is mainly used in tests.
85        *
86        * @param apiKey ApiKey's id.
87        * @param minVersion ApiKey's minimum version.
88        * @param maxVersion ApiKey's maximum version.
89        * @return A new NodeApiVersions object.
90        */
91       public static NodeApiVersions create(short apiKey, short minVersion, short maxVersion) {
92           return create(Collections.singleton(new ApiVersion()
93                   .setApiKey(apiKey)
94                   .setMinVersion(minVersion)
95                   .setMaxVersion(maxVersion)));
96       }
97   
98       public NodeApiVersions(Collection<ApiVersion> nodeApiVersions, Collection<SupportedFeatureKey> nodeSupportedFeatures) {
99           for (ApiVersion nodeApiVersion : nodeApiVersions) {
100               if (ApiKeys.hasId(nodeApiVersion.apiKey())) {
101                   ApiKeys nodeApiKey = ApiKeys.forId(nodeApiVersion.apiKey());
102                   supportedVersions.put(nodeApiKey, nodeApiVersion);
103               } else {
104                   // Newer brokers may support ApiKeys we don't know about
105                   unknownApis.add(nodeApiVersion);
106               }
107           }
108   
109           Map<String, SupportedVersionRange> supportedFeaturesBuilder = new HashMap<>();
110           for (SupportedFeatureKey supportedFeature : nodeSupportedFeatures) {
111               supportedFeaturesBuilder.put(supportedFeature.name(),
112                       new SupportedVersionRange(supportedFeature.minVersion(), supportedFeature.maxVersion()));
113           }
114           this.supportedFeatures = Collections.unmodifiableMap(supportedFeaturesBuilder);
115       }
116   
117       /**
118        * Return the most recent version supported by both the node and the local software.
119        */
120       public short latestUsableVersion(ApiKeys apiKey) {
121           return latestUsableVersion(apiKey, apiKey.oldestVersion(), apiKey.latestVersion());
122       }
123   
124       /**
125        * Get the latest version supported by the broker within an allowed range of versions
126        */
127       public short latestUsableVersion(ApiKeys apiKey, short oldestAllowedVersion, short latestAllowedVersion) {
128           if (!supportedVersions.containsKey(apiKey))
129               throw new UnsupportedVersionException("The broker does not support " + apiKey);
130           ApiVersion supportedVersion = supportedVersions.get(apiKey);
131           Optional<ApiVersion> intersectVersion = ApiVersionsResponse.intersect(supportedVersion,
132               new ApiVersion()
133                   .setApiKey(apiKey.id)
134                   .setMinVersion(oldestAllowedVersion)
135                   .setMaxVersion(latestAllowedVersion));
136   
137           if (intersectVersion.isPresent())
138               return intersectVersion.get().maxVersion();
139           else
140               throw new UnsupportedVersionException("The broker does not support " + apiKey +
141                   " with version in range [" + oldestAllowedVersion + "," + latestAllowedVersion + "]. The supported" +
142                   " range is [" + supportedVersion.minVersion() + "," + supportedVersion.maxVersion() + "].");
143       }
144   
145       /**
146        * Convert the object to a string with no linebreaks.<p/>
147        * <p>
148        * This toString method is relatively expensive, so avoid calling it unless debug logging is turned on.
149        */
150       @Override
151       public String toString() {
152           return toString(false);
153       }
154   
155       /**
156        * Convert the object to a string.
157        *
158        * @param lineBreaks True if we should add a linebreak after each api.
159        */
160       public String toString(boolean lineBreaks) {
161           // The apiVersion collection may not be in sorted order.  We put it into
162           // a TreeMap before printing it out to ensure that we always print in
163           // ascending order.
164           TreeMap<Short, String> apiKeysText = new TreeMap<>();
165           for (ApiVersion supportedVersion : this.supportedVersions.values())
166               apiKeysText.put(supportedVersion.apiKey(), apiVersionToText(supportedVersion));
167           for (ApiVersion apiVersion : unknownApis)
168               apiKeysText.put(apiVersion.apiKey(), apiVersionToText(apiVersion));
169   
170           // Also handle the case where some apiKey types are not specified at all in the given ApiVersions,
171           // which may happen when the remote is too old.
172           for (ApiKeys apiKey : ApiKeys.zkBrokerApis()) {
173               if (!apiKeysText.containsKey(apiKey.id)) {
174                   StringBuilder bld = new StringBuilder();
175                   bld.append(apiKey.name).append("(").
176                           append(apiKey.id).append("): ").append("UNSUPPORTED");
177                   apiKeysText.put(apiKey.id, bld.toString());
178               }
179           }
180           String separator = lineBreaks ? ",\n\t" : ", ";
181           StringBuilder bld = new StringBuilder();
182           bld.append("(");
183           if (lineBreaks)
184               bld.append("\n\t");
185           bld.append(Utils.join(apiKeysText.values(), separator));
186           if (lineBreaks)
187               bld.append("\n");
188           bld.append(")");
189           return bld.toString();
190       }
191   
192       private String apiVersionToText(ApiVersion apiVersion) {
193           StringBuilder bld = new StringBuilder();
194           ApiKeys apiKey = null;
195           if (ApiKeys.hasId(apiVersion.apiKey())) {
196               apiKey = ApiKeys.forId(apiVersion.apiKey());
197               bld.append(apiKey.name).append("(").append(apiKey.id).append("): ");
198           } else {
199               bld.append("UNKNOWN(").append(apiVersion.apiKey()).append("): ");
200           }
201   
202           if (apiVersion.minVersion() == apiVersion.maxVersion()) {
203               bld.append(apiVersion.minVersion());
204           } else {
205               bld.append(apiVersion.minVersion()).append(" to ").append(apiVersion.maxVersion());
206           }
207   
208           if (apiKey != null) {
209               ApiVersion supportedVersion = supportedVersions.get(apiKey);
210               if (apiKey.latestVersion() < supportedVersion.minVersion()) {
211                   bld.append(" [unusable: node too new]");
212               } else if (supportedVersion.maxVersion() < apiKey.oldestVersion()) {
213                   bld.append(" [unusable: node too old]");
214               } else {
215                   short latestUsableVersion = Utils.min(apiKey.latestVersion(), supportedVersion.maxVersion());
216                   bld.append(" [usable: ").append(latestUsableVersion).append("]");
217               }
218           }
219           return bld.toString();
220       }
221   
222       /**
223        * Get the version information for a given API.
224        *
225        * @param apiKey The api key to lookup
226        * @return The api version information from the broker or null if it is unsupported
227        */
228       public ApiVersion apiVersion(ApiKeys apiKey) {
229           return supportedVersions.get(apiKey);
230       }

232       public Map<ApiKeys, ApiVersion> allSupportedApiVersions() {
233           return supportedVersions;
234       }

236       public Map<String, SupportedVersionRange> supportedFeatures() {
237           return supportedFeatures;
238       }
