33   import static org.apache.kafka.common.record.Records.LOG_OVERHEAD;
34   import static org.apache.kafka.common.record.Records.HEADER_SIZE_UP_TO_MAGIC;
35   import static org.apache.kafka.common.record.Records.MAGIC_OFFSET;
36   import static org.apache.kafka.common.record.Records.OFFSET_OFFSET;
37   import static org.apache.kafka.common.record.Records.SIZE_OFFSET;

39   /**
40    * A log input stream which is backed by a {@link FileChannel}.
41    */
42   public class FileLogInputStream implements LogInputStream<FileLogInputStream.FileChannelRecordBatch> {
43       private int position;
44       private final int end;
45       private final FileRecords fileRecords;
46       private final ByteBuffer logHeaderBuffer = ByteBuffer.allocate(HEADER_SIZE_UP_TO_MAGIC);
47   
48       /**
49        * Create a new log input stream over the FileChannel
50        * @param records Underlying FileRecords instance
51        * @param start Position in the file channel to start from
52        * @param end Position in the file channel not to read past
53        */
54       FileLogInputStream(FileRecords records,
55                          int start,
56                          int end) {
57           this.fileRecords = records;
58           this.position = start;
59           this.end = end;
60       }
61   
62       @Override
63       public FileChannelRecordBatch nextBatch() throws IOException {
64           FileChannel channel = fileRecords.channel();
65           if (position >= end - HEADER_SIZE_UP_TO_MAGIC)
66               return null;
67   
68           logHeaderBuffer.rewind();
69           Utils.readFullyOrFail(channel, logHeaderBuffer, position, "log header");
70   
71           logHeaderBuffer.rewind();
72           long offset = logHeaderBuffer.getLong(OFFSET_OFFSET);
73           int size = logHeaderBuffer.getInt(SIZE_OFFSET);
74   
75           // V0 has the smallest overhead, stricter checking is done later
76           if (size < LegacyRecord.RECORD_OVERHEAD_V0)
77               throw new CorruptRecordException(String.format("Found record size %d smaller than minimum record " +
78                               "overhead (%d) in file %s.", size, LegacyRecord.RECORD_OVERHEAD_V0, fileRecords.file()));
79   
80           if (position > end - LOG_OVERHEAD - size)
81               return null;
82   
83           byte magic = logHeaderBuffer.get(MAGIC_OFFSET);
84           final FileChannelRecordBatch batch;
85   
86           if (magic < RecordBatch.MAGIC_VALUE_V2)
87               batch = new LegacyFileChannelRecordBatch(offset, magic, fileRecords, position, size);
88           else
89               batch = new DefaultFileChannelRecordBatch(offset, magic, fileRecords, position, size);
90   
91           position += batch.sizeInBytes();
92           return batch;
93       }
94   
95       /**
96        * Log entry backed by an underlying FileChannel. This allows iteration over the record batches
97        * without needing to read the record data into memory until it is needed. The downside
98        * is that entries will generally no longer be readable when the underlying channel is closed.
99        */
100       public abstract static class FileChannelRecordBatch extends AbstractRecordBatch {
101           protected final long offset;
102           protected final byte magic;
103           protected final FileRecords fileRecords;
104           protected final int position;
105           protected final int batchSize;
106   
107           private RecordBatch fullBatch;
108           private RecordBatch batchHeader;
109   
110           FileChannelRecordBatch(long offset,
111                                  byte magic,
112                                  FileRecords fileRecords,
113                                  int position,
114                                  int batchSize) {
115               this.offset = offset;
116               this.magic = magic;
117               this.fileRecords = fileRecords;
118               this.position = position;
119               this.batchSize = batchSize;
120           }
121   
122           @Override
123           public CompressionType compressionType() {
124               return loadBatchHeader().compressionType();
125           }
126   
127           @Override
128           public TimestampType timestampType() {
129               return loadBatchHeader().timestampType();
130           }
131   
132           @Override
133           public long checksum() {
134               return loadBatchHeader().checksum();
135           }
136   
137           @Override
138           public long maxTimestamp() {
139               return loadBatchHeader().maxTimestamp();
140           }
141   
142           public int position() {
143               return position;
144           }
145   
146           @Override
147           public byte magic() {
148               return magic;
149           }
150   
151           @Override
152           public Iterator<Record> iterator() {
153               return loadFullBatch().iterator();
154           }
155   
156           @Override
157           public CloseableIterator<Record> streamingIterator(BufferSupplier bufferSupplier) {
158               return loadFullBatch().streamingIterator(bufferSupplier);
159           }
160   
161           @Override
162           public boolean isValid() {
163               return loadFullBatch().isValid();
164           }
165   
166           @Override
167           public void ensureValid() {
168               loadFullBatch().ensureValid();
169           }
170   
171           @Override
172           public int sizeInBytes() {
173               return LOG_OVERHEAD + batchSize;
174           }
175   
176           @Override
177           public void writeTo(ByteBuffer buffer) {
178               FileChannel channel = fileRecords.channel();
179               try {
180                   int limit = buffer.limit();
181                   buffer.limit(buffer.position() + sizeInBytes());
182                   Utils.readFully(channel, buffer, position);
183                   buffer.limit(limit);
184               } catch (IOException e) {
185                   throw new KafkaException("Failed to read record batch at position " + position + " from " + fileRecords, e);
186               }
187           }
188   
189           protected abstract RecordBatch toMemoryRecordBatch(ByteBuffer buffer);
190   
191           protected abstract int headerSize();
192   
193           protected RecordBatch loadFullBatch() {
194               if (fullBatch == null) {
195                   batchHeader = null;
196                   fullBatch = loadBatchWithSize(sizeInBytes(), "full record batch");
197               }
198               return fullBatch;
199           }
200   
201           protected RecordBatch loadBatchHeader() {
202               if (fullBatch != null)
203                   return fullBatch;
204   
205               if (batchHeader == null)
206                   batchHeader = loadBatchWithSize(headerSize(), "record batch header");
207   
208               return batchHeader;
209           }
210   
211           private RecordBatch loadBatchWithSize(int size, String description) {
212               FileChannel channel = fileRecords.channel();
213               try {
214                   ByteBuffer buffer = ByteBuffer.allocate(size);
215                   Utils.readFullyOrFail(channel, buffer, position, description);
216                   buffer.rewind();
217                   return toMemoryRecordBatch(buffer);
218               } catch (IOException e) {
219                   throw new KafkaException("Failed to load record batch at position " + position + " from " + fileRecords, e);
220               }
221           }
222   
223           @Override
224           public boolean equals(Object o) {
225               if (this == o)
226                   return true;
227               if (o == null || getClass() != o.getClass())
228                   return false;
229   
230               FileChannelRecordBatch that = (FileChannelRecordBatch) o;
231   
232               FileChannel channel = fileRecords == null ? null : fileRecords.channel();
233               FileChannel thatChannel = that.fileRecords == null ? null : that.fileRecords.channel();
234   
235               return offset == that.offset &&
236                       position == that.position &&
237                       batchSize == that.batchSize &&
238                       Objects.equals(channel, thatChannel);
239           }
240   
241           @Override
242           public int hashCode() {
243               FileChannel channel = fileRecords == null ? null : fileRecords.channel();
244   
245               int result = Long.hashCode(offset);
246               result = 31 * result + (channel != null ? channel.hashCode() : 0);
247               result = 31 * result + position;
248               result = 31 * result + batchSize;
249               return result;
250           }
251   
252           @Override
253           public String toString() {
254               return "FileChannelRecordBatch(magic: " + magic +
255                       ", offset: " + offset +
256                       ", size: " + batchSize + ")";
257           }
258       }
