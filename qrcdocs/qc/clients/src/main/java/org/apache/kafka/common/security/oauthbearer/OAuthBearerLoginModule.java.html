42   /**
43    * The {@code LoginModule} for the SASL/OAUTHBEARER mechanism. When a client
44    * (whether a non-broker client or a broker when SASL/OAUTHBEARER is the
45    * inter-broker protocol) connects to Kafka the {@code OAuthBearerLoginModule}
46    * instance asks its configured {@link AuthenticateCallbackHandler}
47    * implementation to handle an instance of {@link OAuthBearerTokenCallback} and
48    * return an instance of {@link OAuthBearerToken}. A default, builtin
49    * {@link AuthenticateCallbackHandler} implementation creates an unsecured token
50    * as defined by these JAAS module options:
51    * <p>
52    * <table>
53    * <tr>
54    * <th>JAAS Module Option for Unsecured Token Retrieval</th>
55    * <th>Documentation</th>
56    * </tr>
57    * <tr>
58    * <td>{@code unsecuredLoginStringClaim_<claimname>="value"}</td>
59    * <td>Creates a {@code String} claim with the given name and value. Any valid
60    * claim name can be specified except '{@code iat}' and '{@code exp}' (these are
61    * automatically generated).</td>
62    * </tr>
63    * <tr>
64    * <td>{@code unsecuredLoginNumberClaim_<claimname>="value"}</td>
65    * <td>Creates a {@code Number} claim with the given name and value. Any valid
66    * claim name can be specified except '{@code iat}' and '{@code exp}' (these are
67    * automatically generated).</td>
68    * </tr>
69    * <tr>
70    * <td>{@code unsecuredLoginListClaim_<claimname>="value"}</td>
71    * <td>Creates a {@code String List} claim with the given name and values parsed
72    * from the given value where the first character is taken as the delimiter. For
73    * example: {@code unsecuredLoginListClaim_fubar="|value1|value2"}. Any valid
74    * claim name can be specified except '{@code iat}' and '{@code exp}' (these are
75    * automatically generated).</td>
76    * </tr>
77    * <tr>
78    * <td>{@code unsecuredLoginPrincipalClaimName}</td>
79    * <td>Set to a custom claim name if you wish the name of the {@code String}
80    * claim holding the principal name to be something other than
81    * '{@code sub}'.</td>
82    * </tr>
83    * <tr>
84    * <td>{@code unsecuredLoginLifetimeSeconds}</td>
85    * <td>Set to an integer value if the token expiration is to be set to something
86    * other than the default value of 3600 seconds (which is 1 hour). The
87    * '{@code exp}' claim will be set to reflect the expiration time.</td>
88    * </tr>
89    * <tr>
90    * <td>{@code unsecuredLoginScopeClaimName}</td>
91    * <td>Set to a custom claim name if you wish the name of the {@code String} or
92    * {@code String List} claim holding any token scope to be something other than
93    * '{@code scope}'.</td>
94    * </tr>
95    * </table>
96    * <p>
97    * <p>
98    * You can also add custom unsecured SASL extensions when using the default, builtin {@link AuthenticateCallbackHandler}
99    * implementation through using the configurable option {@code unsecuredLoginExtension_<extensionname>}. Note that there
100    * are validations for the key/values in order to conform to the SASL/OAUTHBEARER standard
101    * (https://tools.ietf.org/html/rfc7628#section-3.1), including the reserved key at
102    * {@link org.apache.kafka.common.security.oauthbearer.internals.OAuthBearerClientInitialResponse#AUTH_KEY}.
103    * The {@code OAuthBearerLoginModule} instance also asks its configured {@link AuthenticateCallbackHandler}
104    * implementation to handle an instance of {@link SaslExtensionsCallback} and return an instance of {@link SaslExtensions}.
105    * The configured callback handler does not need to handle this callback, though -- any {@code UnsupportedCallbackException}
106    * that is thrown is ignored, and no SASL extensions will be associated with the login.
107    * <p>
108    * Production use cases will require writing an implementation of
109    * {@link AuthenticateCallbackHandler} that can handle an instance of
110    * {@link OAuthBearerTokenCallback} and declaring it via either the
111    * {@code sasl.login.callback.handler.class} configuration option for a
112    * non-broker client or via the
113    * {@code listener.name.sasl_ssl.oauthbearer.sasl.login.callback.handler.class}
114    * configuration option for brokers (when SASL/OAUTHBEARER is the inter-broker
115    * protocol).
116    * <p>
117    * This class stores the retrieved {@link OAuthBearerToken} in the
118    * {@code Subject}'s private credentials where the {@code SaslClient} can
119    * retrieve it. An appropriate, builtin {@code SaslClient} implementation is
120    * automatically used and configured such that it can perform that retrieval.
121    * <p>
122    * Here is a typical, basic JAAS configuration for a client leveraging unsecured
123    * SASL/OAUTHBEARER authentication:
124    *
125    * <pre>
126    * KafkaClient {
127    *      org.apache.kafka.common.security.oauthbearer.OAuthBearerLoginModule Required
128    *      unsecuredLoginStringClaim_sub="thePrincipalName";
129    * };
130    * </pre>
131    *
132    * An implementation of the {@link Login} interface specific to the
133    * {@code OAUTHBEARER} mechanism is automatically applied; it periodically
134    * refreshes any token before it expires so that the client can continue to make
135    * connections to brokers. The parameters that impact how the refresh algorithm
136    * operates are specified as part of the producer/consumer/broker configuration
137    * and are as follows. See the documentation for these properties elsewhere for
138    * details.
139    * <p>
140    * <table>
141    * <tr>
142    * <th>Producer/Consumer/Broker Configuration Property</th>
143    * </tr>
144    * <tr>
145    * <td>{@code sasl.login.refresh.window.factor}</td>
146    * </tr>
147    * <tr>
148    * <td>{@code sasl.login.refresh.window.jitter}</td>
149    * </tr>
150    * <tr>
151    * <td>{@code sasl.login.refresh.min.period.seconds}</td>
152    * </tr>
153    * <tr>
154    * <td>{@code sasl.login.refresh.min.buffer.seconds}</td>
155    * </tr>
156    * </table>
157    * <p>
158    * When a broker accepts a SASL/OAUTHBEARER connection the instance of the
159    * builtin {@code SaslServer} implementation asks its configured
160    * {@link AuthenticateCallbackHandler} implementation to handle an instance of
161    * {@link OAuthBearerValidatorCallback} constructed with the OAuth 2 Bearer
162    * Token's compact serialization and return an instance of
163    * {@link OAuthBearerToken} if the value validates. A default, builtin
164    * {@link AuthenticateCallbackHandler} implementation validates an unsecured
165    * token as defined by these JAAS module options:
166    * <p>
167    * <table>
168    * <tr>
169    * <th>JAAS Module Option for Unsecured Token Validation</th>
170    * <th>Documentation</th>
171    * </tr>
172    * <tr>
173    * <td>{@code unsecuredValidatorPrincipalClaimName="value"}</td>
174    * <td>Set to a non-empty value if you wish a particular {@code String} claim
175    * holding a principal name to be checked for existence; the default is to check
176    * for the existence of the '{@code sub}' claim.</td>
177    * </tr>
178    * <tr>
179    * <td>{@code unsecuredValidatorScopeClaimName="value"}</td>
180    * <td>Set to a custom claim name if you wish the name of the {@code String} or
181    * {@code String List} claim holding any token scope to be something other than
182    * '{@code scope}'.</td>
183    * </tr>
184    * <tr>
185    * <td>{@code unsecuredValidatorRequiredScope="value"}</td>
186    * <td>Set to a space-delimited list of scope values if you wish the
187    * {@code String/String List} claim holding the token scope to be checked to
188    * make sure it contains certain values.</td>
189    * </tr>
190    * <tr>
191    * <td>{@code unsecuredValidatorAllowableClockSkewMs="value"}</td>
192    * <td>Set to a positive integer value if you wish to allow up to some number of
193    * positive milliseconds of clock skew (the default is 0).</td>
194    * </tr>
195    * </table>
196    * <p>
197    * Here is a typical, basic JAAS configuration for a broker leveraging unsecured
198    * SASL/OAUTHBEARER validation:
199    *
200    * <pre>
201    * KafkaServer {
202    *      org.apache.kafka.common.security.oauthbearer.OAuthBearerLoginModule Required
203    *      unsecuredLoginStringClaim_sub="thePrincipalName";
204    * };
205    * </pre>
206    *
207    * Production use cases will require writing an implementation of
208    * {@link AuthenticateCallbackHandler} that can handle an instance of
209    * {@link OAuthBearerValidatorCallback} and declaring it via the
210    * {@code listener.name.sasl_ssl.oauthbearer.sasl.server.callback.handler.class}
211    * broker configuration option.
212    * <p>
213    * The builtin {@code SaslServer} implementation for SASL/OAUTHBEARER in Kafka
214    * makes the instance of {@link OAuthBearerToken} available upon successful
215    * authentication via the negotiated property "{@code OAUTHBEARER.token}"; the
216    * token could be used in a custom authorizer (to authorize based on JWT claims
217    * rather than ACLs, for example).
218    * <p>
219    * This implementation's {@code logout()} method will logout the specific token
220    * that this instance logged in if it's {@code Subject} instance is shared
221    * across multiple {@code LoginContext}s and there happen to be multiple tokens
222    * on the {@code Subject}. This functionality is useful because it means a new
223    * token with a longer lifetime can be created before a soon-to-expire token is
224    * actually logged out. Otherwise, if multiple simultaneous tokens were not
225    * supported like this, the soon-to-be expired token would have to be logged out
226    * first, and then if the new token could not be retrieved (maybe the
227    * authorization server is temporarily unavailable, for example) the client
228    * would be left without a token and would be unable to create new connections.
229    * Better to mitigate this possibility by leaving the existing token (which
230    * still has some lifetime left) in place until a new replacement token is
231    * actually retrieved. This implementation supports this.
232    *
233    * @see SaslConfigs#SASL_LOGIN_REFRESH_WINDOW_FACTOR_DOC
234    * @see SaslConfigs#SASL_LOGIN_REFRESH_WINDOW_JITTER_DOC
235    * @see SaslConfigs#SASL_LOGIN_REFRESH_MIN_PERIOD_SECONDS_DOC
236    * @see SaslConfigs#SASL_LOGIN_REFRESH_BUFFER_SECONDS_DOC
237    */
238   public class OAuthBearerLoginModule implements LoginModule {
239   
240       /**
241        * Login state transitions:
242        *   Initial state: NOT_LOGGED_IN
243        *   login()      : NOT_LOGGED_IN => LOGGED_IN_NOT_COMMITTED
244        *   commit()     : LOGGED_IN_NOT_COMMITTED => COMMITTED
245        *   abort()      : LOGGED_IN_NOT_COMMITTED => NOT_LOGGED_IN
246        *   logout()     : Any state => NOT_LOGGED_IN
247        */
248       private enum LoginState {
249           NOT_LOGGED_IN,
250           LOGGED_IN_NOT_COMMITTED,
251           COMMITTED
252       }
253   
254       /**
255        * The SASL Mechanism name for OAuth 2: {@code OAUTHBEARER}
256        */
257       public static final String OAUTHBEARER_MECHANISM = "OAUTHBEARER";
258       private static final Logger log = LoggerFactory.getLogger(OAuthBearerLoginModule.class);
259       private static final SaslExtensions EMPTY_EXTENSIONS = new SaslExtensions(Collections.emptyMap());
260       private Subject subject = null;
261       private AuthenticateCallbackHandler callbackHandler = null;
262       private OAuthBearerToken tokenRequiringCommit = null;
263       private OAuthBearerToken myCommittedToken = null;
264       private SaslExtensions extensionsRequiringCommit = null;
265       private SaslExtensions myCommittedExtensions = null;
266       private LoginState loginState;
267   
268       static {
269           OAuthBearerSaslClientProvider.initialize(); // not part of public API
270           OAuthBearerSaslServerProvider.initialize(); // not part of public API
271       }
272   
273       @Override
274       public void initialize(Subject subject, CallbackHandler callbackHandler, Map<String, ?> sharedState,
275               Map<String, ?> options) {
276           this.subject = Objects.requireNonNull(subject);
277           if (!(Objects.requireNonNull(callbackHandler) instanceof AuthenticateCallbackHandler))
278               throw new IllegalArgumentException(String.format("Callback handler must be castable to %s: %s",
279                       AuthenticateCallbackHandler.class.getName(), callbackHandler.getClass().getName()));
280           this.callbackHandler = (AuthenticateCallbackHandler) callbackHandler;
281       }
282   
283       @Override
284       public boolean login() throws LoginException {
285           if (loginState == LoginState.LOGGED_IN_NOT_COMMITTED) {
286               if (tokenRequiringCommit != null)
287                   throw new IllegalStateException(String.format(
288                       "Already have an uncommitted token with private credential token count=%d", committedTokenCount()));
289               else
290                   throw new IllegalStateException("Already logged in without a token");
291           }
292           if (loginState == LoginState.COMMITTED) {
293               if (myCommittedToken != null)
294                   throw new IllegalStateException(String.format(
295                       "Already have a committed token with private credential token count=%d; must login on another login context or logout here first before reusing the same login context",
296                       committedTokenCount()));
297               else
298                   throw new IllegalStateException("Login has already been committed without a token");
299           }
300   
301           identifyToken();
302           if (tokenRequiringCommit != null)
303               identifyExtensions();
304           else
305               log.debug("Logged in without a token, this login cannot be used to establish client connections");
306   
307           loginState = LoginState.LOGGED_IN_NOT_COMMITTED;
308           log.debug("Login succeeded; invoke commit() to commit it; current committed token count={}",
309                   committedTokenCount());
310           return true;
311       }
312   
313       private void identifyToken() throws LoginException {
314           OAuthBearerTokenCallback tokenCallback = new OAuthBearerTokenCallback();
315           try {
316               callbackHandler.handle(new Callback[] {tokenCallback});
317           } catch (IOException | UnsupportedCallbackException e) {
318               log.error(e.getMessage(), e);
319               throw new LoginException("An internal error occurred while retrieving token from callback handler");
320           }
321   
322           tokenRequiringCommit = tokenCallback.token();
323           if (tokenCallback.errorCode() != null) {
324               log.info("Login failed: {} : {} (URI={})", tokenCallback.errorCode(), tokenCallback.errorDescription(),
325                       tokenCallback.errorUri());
326               throw new LoginException(tokenCallback.errorDescription());
327           }
328       }
329   
330       /**
331        * Attaches SASL extensions to the Subject
332        */
333       private void identifyExtensions() throws LoginException {
334           SaslExtensionsCallback extensionsCallback = new SaslExtensionsCallback();
335           try {
336               callbackHandler.handle(new Callback[] {extensionsCallback});
337               extensionsRequiringCommit = extensionsCallback.extensions();
338           } catch (IOException e) {
339               log.error(e.getMessage(), e);
340               throw new LoginException("An internal error occurred while retrieving SASL extensions from callback handler");
341           } catch (UnsupportedCallbackException e) {
342               extensionsRequiringCommit = EMPTY_EXTENSIONS;
343               log.debug("CallbackHandler {} does not support SASL extensions. No extensions will be added", callbackHandler.getClass().getName());
344           }
345           if (extensionsRequiringCommit ==  null) {
346               log.error("SASL Extensions cannot be null. Check whether your callback handler is explicitly setting them as null.");
347               throw new LoginException("Extensions cannot be null.");
348           }
349       }
350   
351       @Override
352       public boolean logout() {
353           if (loginState == LoginState.LOGGED_IN_NOT_COMMITTED)
354               throw new IllegalStateException(
355                       "Cannot call logout() immediately after login(); need to first invoke commit() or abort()");
356           if (loginState != LoginState.COMMITTED) {
357               log.debug("Nothing here to log out");
358               return false;
359           }
360           if (myCommittedToken != null) {
361               log.trace("Logging out my token; current committed token count = {}", committedTokenCount());
362               for (Iterator<Object> iterator = subject.getPrivateCredentials().iterator(); iterator.hasNext(); ) {
363                   Object privateCredential = iterator.next();
364                   if (privateCredential == myCommittedToken) {
365                       iterator.remove();
366                       myCommittedToken = null;
367                       break;
368                   }
369               }
370               log.debug("Done logging out my token; committed token count is now {}", committedTokenCount());
371           } else
372               log.debug("No tokens to logout for this login");
373   
374           if (myCommittedExtensions != null) {
375               log.trace("Logging out my extensions");
376               if (subject.getPublicCredentials().removeIf(e -> myCommittedExtensions == e))
377                   myCommittedExtensions = null;
378               log.debug("Done logging out my extensions");
379           } else
380               log.debug("No extensions to logout for this login");
381   
382           loginState = LoginState.NOT_LOGGED_IN;
383           return true;
384       }
385   
386       @Override
387       public boolean commit() {
388           if (loginState != LoginState.LOGGED_IN_NOT_COMMITTED) {
389               log.debug("Nothing here to commit");
390               return false;
391           }
392   
393           if (tokenRequiringCommit != null) {
394               log.trace("Committing my token; current committed token count = {}", committedTokenCount());
395               subject.getPrivateCredentials().add(tokenRequiringCommit);
396               myCommittedToken = tokenRequiringCommit;
397               tokenRequiringCommit = null;
398               log.debug("Done committing my token; committed token count is now {}", committedTokenCount());
399           } else
400               log.debug("No tokens to commit, this login cannot be used to establish client connections");
401   
402           if (extensionsRequiringCommit != null) {
403               subject.getPublicCredentials().add(extensionsRequiringCommit);
404               myCommittedExtensions = extensionsRequiringCommit;
405               extensionsRequiringCommit = null;
406           }
407   
408           loginState = LoginState.COMMITTED;
409           return true;
410       }
411   
412       @Override
413       public boolean abort() {
414           if (loginState == LoginState.LOGGED_IN_NOT_COMMITTED) {
415               log.debug("Login aborted");
416               tokenRequiringCommit = null;
417               extensionsRequiringCommit = null;
418               loginState = LoginState.NOT_LOGGED_IN;
419               return true;
420           }
421           log.debug("Nothing here to abort");
422           return false;
423       }
424   
425       private int committedTokenCount() {
426           return subject.getPrivateCredentials(OAuthBearerToken.class).size();
427       }
