139       @SuppressWarnings("unchecked")
140       @Override
141       public void configure(Map<String, ?> configs) throws KafkaException {
142           try {
143               this.configs = configs;
144               if (mode == Mode.SERVER) {
145                   createServerCallbackHandlers(configs);
146                   createConnectionsMaxReauthMsMap(configs);
147               } else
148                   createClientCallbackHandler(configs);
149               for (Map.Entry<String, AuthenticateCallbackHandler> entry : saslCallbackHandlers.entrySet()) {
150                   String mechanism = entry.getKey();
151                   entry.getValue().configure(configs, mechanism, jaasContexts.get(mechanism).configurationEntries());
152               }
153   
154               Class<? extends Login> defaultLoginClass = defaultLoginClass();
155               if (mode == Mode.SERVER && jaasContexts.containsKey(SaslConfigs.GSSAPI_MECHANISM)) {
156                   String defaultRealm;
157                   try {
158                       defaultRealm = defaultKerberosRealm();
159                   } catch (Exception ke) {
160                       defaultRealm = "";
161                   }
162                   List<String> principalToLocalRules = (List<String>) configs.get(BrokerSecurityConfigs.SASL_KERBEROS_PRINCIPAL_TO_LOCAL_RULES_CONFIG);
163                   if (principalToLocalRules != null)
164                       kerberosShortNamer = KerberosShortNamer.fromUnparsedRules(defaultRealm, principalToLocalRules);
165               }
166               for (Map.Entry<String, JaasContext> entry : jaasContexts.entrySet()) {
167                   String mechanism = entry.getKey();
168                   // With static JAAS configuration, use KerberosLogin if Kerberos is enabled. With dynamic JAAS configuration,
169                   // use KerberosLogin only for the LoginContext corresponding to GSSAPI
170                   LoginManager loginManager = LoginManager.acquireLoginManager(entry.getValue(), mechanism, defaultLoginClass, configs);
171                   loginManagers.put(mechanism, loginManager);
172                   Subject subject = loginManager.subject();
173                   subjects.put(mechanism, subject);
174                   if (mode == Mode.SERVER && mechanism.equals(SaslConfigs.GSSAPI_MECHANISM))
175                       maybeAddNativeGssapiCredentials(subject);
176               }
177               if (this.securityProtocol == SecurityProtocol.SASL_SSL) {
178                   // Disable SSL client authentication as we are using SASL authentication
179                   this.sslFactory = new SslFactory(mode, sslClientAuthOverride, isInterBrokerListener);
180                   this.sslFactory.configure(configs);
181               }
182           } catch (Throwable e) {
183               close();
184               throw new KafkaException(e);
185           }
186       }

210       @Override
211       public KafkaChannel buildChannel(String id, SelectionKey key, int maxReceiveSize,
212                                        MemoryPool memoryPool, ChannelMetadataRegistry metadataRegistry) throws KafkaException {
213           try {
214               SocketChannel socketChannel = (SocketChannel) key.channel();
215               Socket socket = socketChannel.socket();
216               TransportLayer transportLayer = buildTransportLayer(id, key, socketChannel, metadataRegistry);
217               Supplier<Authenticator> authenticatorCreator;
218               if (mode == Mode.SERVER) {
219                   authenticatorCreator = () -> buildServerAuthenticator(configs,
220                           Collections.unmodifiableMap(saslCallbackHandlers),
221                           id,
222                           transportLayer,
223                           Collections.unmodifiableMap(subjects),
224                           Collections.unmodifiableMap(connectionsMaxReauthMsByMechanism),
225                           metadataRegistry);
226               } else {
227                   LoginManager loginManager = loginManagers.get(clientSaslMechanism);
228                   authenticatorCreator = () -> buildClientAuthenticator(configs,
229                           saslCallbackHandlers.get(clientSaslMechanism),
230                           id,
231                           socket.getInetAddress().getHostName(),
232                           loginManager.serviceName(),
233                           transportLayer,
234                           subjects.get(clientSaslMechanism));
235               }
236               return new KafkaChannel(id, transportLayer, authenticatorCreator, maxReceiveSize,
237                   memoryPool != null ? memoryPool : MemoryPool.NONE, metadataRegistry);
238           } catch (Exception e) {
239               log.info("Failed to create channel due to ", e);
240               throw new KafkaException(e);
241           }
242       }

300       private void createClientCallbackHandler(Map<String, ?> configs) {
301           @SuppressWarnings("unchecked")
302           Class<? extends AuthenticateCallbackHandler> clazz = (Class<? extends AuthenticateCallbackHandler>) configs.get(SaslConfigs.SASL_CLIENT_CALLBACK_HANDLER_CLASS);
303           if (clazz == null)
304               clazz = clientCallbackHandlerClass();
305           AuthenticateCallbackHandler callbackHandler = Utils.newInstance(clazz);
306           saslCallbackHandlers.put(clientSaslMechanism, callbackHandler);
307       }

309       private void createServerCallbackHandlers(Map<String, ?> configs) {
310           for (String mechanism : jaasContexts.keySet()) {
311               AuthenticateCallbackHandler callbackHandler;
312               String prefix = ListenerName.saslMechanismPrefix(mechanism);
313               @SuppressWarnings("unchecked")
314               Class<? extends AuthenticateCallbackHandler> clazz =
315                       (Class<? extends AuthenticateCallbackHandler>) configs.get(prefix + BrokerSecurityConfigs.SASL_SERVER_CALLBACK_HANDLER_CLASS);
316               if (clazz != null)
317                   callbackHandler = Utils.newInstance(clazz);
318               else if (mechanism.equals(PlainSaslServer.PLAIN_MECHANISM))
319                   callbackHandler = new PlainServerCallbackHandler();
320               else if (ScramMechanism.isScram(mechanism))
321                   callbackHandler = new ScramServerCallbackHandler(credentialCache.cache(mechanism, ScramCredential.class), tokenCache);
322               else if (mechanism.equals(OAuthBearerLoginModule.OAUTHBEARER_MECHANISM))
323                   callbackHandler = new OAuthBearerUnsecuredValidatorCallbackHandler();
324               else
325                   callbackHandler = new SaslServerCallbackHandler();
326               saslCallbackHandlers.put(mechanism, callbackHandler);
327           }
328       }

349       private Class<? extends AuthenticateCallbackHandler> clientCallbackHandlerClass() {
350           switch (clientSaslMechanism) {
351               case SaslConfigs.GSSAPI_MECHANISM:
352                   return KerberosClientCallbackHandler.class;
353               case OAuthBearerLoginModule.OAUTHBEARER_MECHANISM:

355               default:
356                   return SaslClientCallbackHandler.class;
357           }

360       // As described in http://docs.oracle.com/javase/8/docs/technotes/guides/security/jgss/jgss-features.html:
361       // "To enable Java GSS to delegate to the native GSS library and its list of native mechanisms,
362       // set the system property "sun.security.jgss.native" to true"
363       // "In addition, when performing operations as a particular Subject, for example, Subject.doAs(...)
364       // or Subject.doAsPrivileged(...), the to-be-used GSSCredential should be added to Subject's
365       // private credential set. Otherwise, the GSS operations will fail since no credential is found."
366       private void maybeAddNativeGssapiCredentials(Subject subject) {
367           boolean usingNativeJgss = Boolean.getBoolean(GSS_NATIVE_PROP);
368           if (usingNativeJgss && subject.getPrivateCredentials(GSSCredential.class).isEmpty()) {
369   
370               final String servicePrincipal = SaslClientAuthenticator.firstPrincipal(subject);
371               KerberosName kerberosName;
372               try {
373                   kerberosName = KerberosName.parse(servicePrincipal);
374               } catch (IllegalArgumentException e) {
375                   throw new KafkaException("Principal has name with unexpected format " + servicePrincipal);
376               }
377               final String servicePrincipalName = kerberosName.serviceName();
378               final String serviceHostname = kerberosName.hostName();
379   
380               try {
381                   GSSManager manager = gssManager();
382                   // This Oid is used to represent the Kerberos version 5 GSS-API mechanism. It is defined in
383                   // RFC 1964.
384                   Oid krb5Mechanism = new Oid("1.2.840.113554.1.2.2");
385                   GSSName gssName = manager.createName(servicePrincipalName + "@" + serviceHostname, GSSName.NT_HOSTBASED_SERVICE);
386                   GSSCredential cred = manager.createCredential(gssName,
387                           GSSContext.INDEFINITE_LIFETIME, krb5Mechanism, GSSCredential.ACCEPT_ONLY);
388                   subject.getPrivateCredentials().add(cred);
389                   log.info("Configured native GSSAPI private credentials for {}@{}", serviceHostname, serviceHostname);
390               } catch (GSSException ex) {
391                   log.warn("Cannot add private credential to subject; clients authentication may fail", ex);
392               }
393           }
394       }
