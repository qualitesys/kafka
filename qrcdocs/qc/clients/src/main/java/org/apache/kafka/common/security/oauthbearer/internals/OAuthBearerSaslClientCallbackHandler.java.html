44   /**
45    * An implementation of {@code AuthenticateCallbackHandler} that recognizes
46    * {@link OAuthBearerTokenCallback} and retrieves OAuth 2 Bearer Token that was
47    * created when the {@code OAuthBearerLoginModule} logged in by looking for an
48    * instance of {@link OAuthBearerToken} in the {@code Subject}'s private
49    * credentials. This class also recognizes {@link SaslExtensionsCallback} and retrieves any SASL extensions that were
50    * created when the {@code OAuthBearerLoginModule} logged in by looking for an instance of {@link SaslExtensions}
51    * in the {@code Subject}'s public credentials
52    * <p>
53    * Use of this class is configured automatically and does not need to be
54    * explicitly set via the {@code sasl.client.callback.handler.class}
55    * configuration property.
56    */
57   public class OAuthBearerSaslClientCallbackHandler implements AuthenticateCallbackHandler {
58       private static final Logger log = LoggerFactory.getLogger(OAuthBearerSaslClientCallbackHandler.class);
59       private boolean configured = false;
60   
61       /**
62        * Return true if this instance has been configured, otherwise false
63        *
64        * @return true if this instance has been configured, otherwise false
65        */
66       public boolean configured() {
67           return configured;
68       }
69   
70       @Override
71       public void configure(Map<String, ?> configs, String saslMechanism, List<AppConfigurationEntry> jaasConfigEntries) {
72           if (!OAuthBearerLoginModule.OAUTHBEARER_MECHANISM.equals(saslMechanism))
73               throw new IllegalArgumentException(String.format("Unexpected SASL mechanism: %s", saslMechanism));
74           configured = true;
75       }
76   
77       @Override
78       public void handle(Callback[] callbacks) throws IOException, UnsupportedCallbackException {
79           if (!configured())
80               throw new IllegalStateException("Callback handler not configured");
81           for (Callback callback : callbacks) {
82               if (callback instanceof OAuthBearerTokenCallback)
83                   handleCallback((OAuthBearerTokenCallback) callback);
84               else if (callback instanceof SaslExtensionsCallback)
85                   handleCallback((SaslExtensionsCallback) callback, Subject.getSubject(AccessController.getContext()));
86               else
87                   throw new UnsupportedCallbackException(callback);
88           }
89       }
90   
91       @Override
92       public void close() {
93           // empty
94       }
95   
96       private void handleCallback(OAuthBearerTokenCallback callback) throws IOException {
97           if (callback.token() != null)
98               throw new IllegalArgumentException("Callback had a token already");
99           Subject subject = Subject.getSubject(AccessController.getContext());
100           Set<OAuthBearerToken> privateCredentials = subject != null
101               ? subject.getPrivateCredentials(OAuthBearerToken.class)
102               : Collections.emptySet();
103           if (privateCredentials.size() == 0)
104               throw new IOException("No OAuth Bearer tokens in Subject's private credentials");
105           if (privateCredentials.size() == 1)
106               callback.token(privateCredentials.iterator().next());
107           else {
108               /*
109                * There a very small window of time upon token refresh (on the order of milliseconds)
110                * where both an old and a new token appear on the Subject's private credentials.
111                * Rather than implement a lock to eliminate this window, we will deal with it by
112                * checking for the existence of multiple tokens and choosing the one that has the
113                * longest lifetime.  It is also possible that a bug could cause multiple tokens to
114                * exist (e.g. KAFKA-7902), so dealing with the unlikely possibility that occurs
115                * during normal operation also allows us to deal more robustly with potential bugs.
116                */
117               SortedSet<OAuthBearerToken> sortedByLifetime =
118                   new TreeSet<>(
119                       new Comparator<OAuthBearerToken>() {
120                           @Override
121                           public int compare(OAuthBearerToken o1, OAuthBearerToken o2) {
122                               return Long.compare(o1.lifetimeMs(), o2.lifetimeMs());
123                           }
124                       });
125               sortedByLifetime.addAll(privateCredentials);
126               log.warn("Found {} OAuth Bearer tokens in Subject's private credentials; the oldest expires at {}, will use the newest, which expires at {}",
127                   sortedByLifetime.size(),
128                   new Date(sortedByLifetime.first().lifetimeMs()),
129                   new Date(sortedByLifetime.last().lifetimeMs()));
130               callback.token(sortedByLifetime.last());
131           }
132       }
133   
134       /**
135        * Attaches the first {@link SaslExtensions} found in the public credentials of the Subject
136        */
137       private static void handleCallback(SaslExtensionsCallback extensionsCallback, Subject subject) {
138           if (subject != null && !subject.getPublicCredentials(SaslExtensions.class).isEmpty()) {
139               SaslExtensions extensions = subject.getPublicCredentials(SaslExtensions.class).iterator().next();
140               extensionsCallback.extensions(extensions);
141           }
142       }
143   }
