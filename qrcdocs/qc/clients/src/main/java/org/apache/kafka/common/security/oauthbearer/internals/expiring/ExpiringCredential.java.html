19   /**
20    * A credential that expires and that can potentially be refreshed
21    * 
22    * @see ExpiringCredentialRefreshingLogin
23    */
24   public interface ExpiringCredential {
25       /**
26        * The name of the principal to which this credential applies (used only for
27        * logging)
28        * 
29        * @return the always non-null/non-empty principal name
30        */
31       String principalName();
32   
33       /**
34        * When the credential became valid, in terms of the number of milliseconds
35        * since the epoch, if known, otherwise null. An expiring credential may not
36        * necessarily indicate when it was created -- just when it expires -- so we
37        * need to support a null return value here.
38        * 
39        * @return the time when the credential became valid, in terms of the number of
40        *         milliseconds since the epoch, if known, otherwise null
41        */
42       Long startTimeMs();
43   
44       /**
45        * When the credential expires, in terms of the number of milliseconds since the
46        * epoch. All expiring credentials by definition must indicate their expiration
47        * time -- thus, unlike other methods, we do not support a null return value
48        * here.
49        * 
50        * @return the time when the credential expires, in terms of the number of
51        *         milliseconds since the epoch
52        */
53       long expireTimeMs();
54   
55       /**
56        * The point after which the credential can no longer be refreshed, in terms of
57        * the number of milliseconds since the epoch, if any, otherwise null. Some
58        * expiring credentials can be refreshed over and over again without limit, so
59        * we support a null return value here.
60        * 
61        * @return the point after which the credential can no longer be refreshed, in
62        *         terms of the number of milliseconds since the epoch, if any,
63        *         otherwise null
64        */
65       Long absoluteLastRefreshTimeMs();
66   }
