26   import java.util.HashMap;
27   import java.util.HashSet;
28   import java.util.Iterator;
29   import java.util.Map;
30   import java.util.Set;
31   import java.util.concurrent.ConcurrentLinkedDeque;
32   import java.util.concurrent.locks.Lock;
33   import java.util.concurrent.locks.ReentrantLock;
34   import java.util.function.Function;

36   /**
37    * Manages a suite of integer Gauges.
38    */
39   public final class IntGaugeSuite<K> implements AutoCloseable {
40       /**
41        * The log4j logger.
42        */
43       private final Logger log;
44   
45       /**
46        * The name of this suite.
47        */
48       private final String suiteName;
49   
50       /**
51        * The metrics object to use.
52        */
53       private final Metrics metrics;
54   
55       /**
56        * A user-supplied callback which translates keys into unique metric names.
57        */
58       private final Function<K, MetricName> metricNameCalculator;
59   
60       /**
61        * The maximum number of gauges that we will ever create at once.
62        */
63       private final int maxEntries;
64   
65       /**
66        * A map from keys to gauges.  Protected by the object monitor.
67        */
68       private final Map<K, StoredIntGauge> gauges;
69   
70       /**
71        * The keys of gauges that can be removed, since their value is zero.
72        * Protected by the object monitor.
73        */
74       private final Set<K> removable;
75   
76       /**
77        * A lockless list of pending metrics additions and removals.
78        */
79       private final ConcurrentLinkedDeque<PendingMetricsChange> pending;
80   
81       /**
82        * A lock which serializes modifications to metrics.  This lock is not
83        * required to create a new pending operation.
84        */
85       private final Lock modifyMetricsLock;
86   
87       /**
88        * True if this suite is closed.  Protected by the object monitor.
89        */
90       private boolean closed;
91   
92       /**
93        * A pending metrics addition or removal.
94        */
95       private static class PendingMetricsChange {
96           /**
97            * The name of the metric to add or remove.
98            */
99           private final MetricName metricName;
100   
101           /**
102            * In an addition, this field is the MetricValueProvider to add.
103            * In a removal, this field is null.
104            */
105           private final MetricValueProvider<?> provider;
106   
107           PendingMetricsChange(MetricName metricName, MetricValueProvider<?> provider) {
108               this.metricName = metricName;
109               this.provider = provider;
110           }
111       }
112   
113       /**
114        * The gauge object which we register with the metrics system.
115        */
116       private static class StoredIntGauge implements Gauge<Integer> {
117           private final MetricName metricName;
118           private int value;
119   
120           StoredIntGauge(MetricName metricName) {
121               this.metricName = metricName;
122               this.value = 1;
123           }
124   
125           /**
126            * This callback is invoked when the metrics system retrieves the value of this gauge.
127            */
128           @Override
129           public synchronized Integer value(MetricConfig config, long now) {
130               return value;
131           }
132   
133           synchronized int increment() {
134               return ++value;
135           }
136   
137           synchronized int decrement() {
138               return --value;
139           }
140   
141           synchronized int value() {
142               return value;
143           }
144       }
145   
146       public IntGaugeSuite(Logger log,
147                            String suiteName,
148                            Metrics metrics,
149                            Function<K, MetricName> metricNameCalculator,
150                            int maxEntries) {
151           this.log = log;
152           this.suiteName = suiteName;
153           this.metrics = metrics;
154           this.metricNameCalculator = metricNameCalculator;
155           this.maxEntries = maxEntries;
156           this.gauges = new HashMap<>(1);
157           this.removable = new HashSet<>();
158           this.pending = new ConcurrentLinkedDeque<>();
159           this.modifyMetricsLock = new ReentrantLock();
160           this.closed = false;
161           log.trace("{}: created new gauge suite with maxEntries = {}.",
162               suiteName, maxEntries);
163       }
164   
165       public void increment(K key) {
166           synchronized (this) {
167               if (closed) {
168                   log.warn("{}: Attempted to increment {}, but the GaugeSuite was closed.",
169                       suiteName, key.toString());
170                   return;
171               }
172               StoredIntGauge gauge = gauges.get(key);
173               if (gauge != null) {
174                   // Fast path: increment the existing counter.
175                   if (gauge.increment() > 0) {
176                       removable.remove(key);
177                   }
178                   return;
179               }
180               if (gauges.size() == maxEntries) {
181                   if (removable.isEmpty()) {
182                       log.debug("{}: Attempted to increment {}, but there are already {} entries.",
183                           suiteName, key.toString(), maxEntries);
184                       return;
185                   }
186                   Iterator<K> iter = removable.iterator();
187                   K keyToRemove = iter.next();
188                   iter.remove();
189                   MetricName metricNameToRemove = gauges.get(keyToRemove).metricName;
190                   gauges.remove(keyToRemove);
191                   pending.push(new PendingMetricsChange(metricNameToRemove, null));
192                   log.trace("{}: Removing the metric {}, which has a value of 0.",
193                       suiteName, keyToRemove.toString());
194               }
195               MetricName metricNameToAdd = metricNameCalculator.apply(key);
196               gauge = new StoredIntGauge(metricNameToAdd);
197               gauges.put(key, gauge);
198               pending.push(new PendingMetricsChange(metricNameToAdd, gauge));
199               log.trace("{}: Adding a new metric {}.", suiteName, key.toString());
200           }
201           // Drop the object monitor and perform any pending metrics additions or removals.
202           performPendingMetricsOperations();
203       }
204   
205       /**
206        * Perform pending metrics additions or removals.
207        * It is important to perform them in order.  For example, we don't want to try
208        * to remove a metric that we haven't finished adding yet.
209        */
210       private void performPendingMetricsOperations() {
211           modifyMetricsLock.lock();
212           try {
213               log.trace("{}: entering performPendingMetricsOperations", suiteName);
214               for (PendingMetricsChange change = pending.pollLast();
215                    change != null;
216                    change = pending.pollLast()) {
217                   if (change.provider == null) {
218                       if (log.isTraceEnabled()) {
219                           log.trace("{}: removing metric {}", suiteName, change.metricName);
220                       }
221                       metrics.removeMetric(change.metricName);
222                   } else {
223                       if (log.isTraceEnabled()) {
224                           log.trace("{}: adding metric {}", suiteName, change.metricName);
225                       }
226                       metrics.addMetric(change.metricName, change.provider);
227                   }
228               }
229               log.trace("{}: leaving performPendingMetricsOperations", suiteName);
230           } finally {
231               modifyMetricsLock.unlock();
232           }
233       }
234   
235       public synchronized void decrement(K key) {
236           if (closed) {
237               log.warn("{}: Attempted to decrement {}, but the gauge suite was closed.",
238                   suiteName, key.toString());
239               return;
240           }
241           StoredIntGauge gauge = gauges.get(key);
242           if (gauge == null) {
243               log.debug("{}: Attempted to decrement {}, but no such metric was registered.",
244                   suiteName, key.toString());
245           } else {
246               int cur = gauge.decrement();
247               log.trace("{}: Removed a reference to {}.  {} reference(s) remaining.",
248                   suiteName, key.toString(), cur);
249               if (cur <= 0) {
250                   removable.add(key);
251               }
252           }
253       }
254   
255       @Override
256       public synchronized void close() {
257           if (closed) {
258               log.trace("{}: gauge suite is already closed.", suiteName);
259               return;
260           }
261           closed = true;
262           int prevSize = 0;
263           for (Iterator<StoredIntGauge> iter = gauges.values().iterator(); iter.hasNext(); ) {
264               pending.push(new PendingMetricsChange(iter.next().metricName, null));
265               prevSize++;
266               iter.remove();
267           }
268           performPendingMetricsOperations();
269           log.trace("{}: closed {} metric(s).", suiteName, prevSize);
270       }
271   
272       /**
273        * Get the maximum number of metrics this suite can create.
274        */
275       public int maxEntries() {
276           return maxEntries;
277       }
278   
279       // Visible for testing only.
280       Metrics metrics() {
281           return metrics;
282       }
283   
284       /**
285        * Return a map from keys to current reference counts.
286        * Visible for testing only.
287        */
288       synchronized Map<K, Integer> values() {
289           HashMap<K, Integer> values = new HashMap<>();
290           for (Map.Entry<K, StoredIntGauge> entry : gauges.entrySet()) {
291               values.put(entry.getKey(), entry.getValue().value());
292           }
293           return values;
294       }
295   }
