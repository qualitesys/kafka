17   package org.apache.kafka.server.quota;
18   
19   import org.apache.kafka.common.Cluster;
20   import org.apache.kafka.common.Configurable;
21   import org.apache.kafka.common.security.auth.KafkaPrincipal;
22   
23   import java.util.Map;
24   
25   /**
26    * Quota callback interface for brokers that enables customization of client quota computation.
27    */
28   public interface ClientQuotaCallback extends Configurable {
29   
30       /**
31        * Quota callback invoked to determine the quota metric tags to be applied for a request.
32        * Quota limits are associated with quota metrics and all clients which use the same
33        * metric tags share the quota limit.
34        *
35        * @param quotaType Type of quota requested
36        * @param principal The user principal of the connection for which quota is requested
37        * @param clientId  The client id associated with the request
38        * @return quota metric tags that indicate which other clients share this quota
39        */
40       Map<String, String> quotaMetricTags(ClientQuotaType quotaType, KafkaPrincipal principal, String clientId);
41   
42       /**
43        * Returns the quota limit associated with the provided metric tags. These tags were returned from
44        * a previous call to {@link #quotaMetricTags(ClientQuotaType, KafkaPrincipal, String)}. This method is
45        * invoked by quota managers to obtain the current quota limit applied to a metric when the first request
46        * using these tags is processed. It is also invoked after a quota update or cluster metadata change.
47        * If the tags are no longer in use after the update, (e.g. this is a {user, client-id} quota metric
48        * and the quota now in use is a {user} quota), null is returned.
49        *
50        * @param quotaType  Type of quota requested
51        * @param metricTags Metric tags for a quota metric of type `quotaType`
52        * @return the quota limit for the provided metric tags or null if the metric tags are no longer in use
53        */
54       Double quotaLimit(ClientQuotaType quotaType, Map<String, String> metricTags);
55   
56       /**
57        * Quota configuration update callback that is invoked when quota configuration for an entity is
58        * updated in ZooKeeper. This is useful to track configured quotas if built-in quota configuration
59        * tools are used for quota management.
60        *
61        * @param quotaType   Type of quota being updated
62        * @param quotaEntity The quota entity for which quota is being updated
63        * @param newValue    The new quota value
64        */
65       void updateQuota(ClientQuotaType quotaType, ClientQuotaEntity quotaEntity, double newValue);
66   
67       /**
68        * Quota configuration removal callback that is invoked when quota configuration for an entity is
69        * removed in ZooKeeper. This is useful to track configured quotas if built-in quota configuration
70        * tools are used for quota management.
71        *
72        * @param quotaType   Type of quota being updated
73        * @param quotaEntity The quota entity for which quota is being updated
74        */
75       void removeQuota(ClientQuotaType quotaType, ClientQuotaEntity quotaEntity);
76   
77       /**
78        * Returns true if any of the existing quota configs may have been updated since the last call
79        * to this method for the provided quota type. Quota updates as a result of calls to
80        * {@link #updateClusterMetadata(Cluster)}, {@link #updateQuota(ClientQuotaType, ClientQuotaEntity, double)}
81        * and {@link #removeQuota(ClientQuotaType, ClientQuotaEntity)} are automatically processed.
82        * So callbacks that rely only on built-in quota configuration tools always return false. Quota callbacks
83        * with external quota configuration or custom reconfigurable quota configs that affect quota limits must
84        * return true if existing metric configs may need to be updated. This method is invoked on every request
85        * and hence is expected to be handled by callbacks as a simple flag that is updated when quotas change.
86        *
87        * @param quotaType Type of quota
88        */
89       boolean quotaResetRequired(ClientQuotaType quotaType);
90   
91       /**
92        * Metadata update callback that is invoked whenever UpdateMetadata request is received from
93        * the controller. This is useful if quota computation takes partitions into account.
94        * Topics that are being deleted will not be included in `cluster`.
95        *
96        * @param cluster Cluster metadata including partitions and their leaders if known
97        * @return true if quotas have changed and metric configs may need to be updated
98        */
99       boolean updateClusterMetadata(Cluster cluster);
100   
101       /**
102        * Closes this instance.
103        */
104       void close();
105   }
