17   package org.apache.kafka.server.policy;
18   
19   import org.apache.kafka.common.Configurable;
20   import org.apache.kafka.common.errors.PolicyViolationException;
21   
22   import java.util.Collections;
23   import java.util.List;
24   import java.util.Map;
25   import java.util.Objects;
26   
27   /**
28    * <p>An interface for enforcing a policy on create topics requests.
29    *
30    * <p>Common use cases are requiring that the replication factor, <code>min.insync.replicas</code> and/or retention settings for a
31    * topic are within an allowable range.
32    *
33    * <p>If <code>create.topic.policy.class.name</code> is defined, Kafka will create an instance of the specified class
34    * using the default constructor and will then pass the broker configs to its <code>configure()</code> method. During
35    * broker shutdown, the <code>close()</code> method will be invoked so that resources can be released (if necessary).
36    */
37   public interface CreateTopicPolicy extends Configurable, AutoCloseable {
38   
39       /**
40        * Class containing the create request parameters.
41        */
42       class RequestMetadata {
43           private final String topic;
44           private final Integer numPartitions;
45           private final Short replicationFactor;
46           private final Map<Integer, List<Integer>> replicasAssignments;
47           private final Map<String, String> configs;
48   
49           /**
50            * Create an instance of this class with the provided parameters.
51            *
52            * This constructor is public to make testing of <code>CreateTopicPolicy</code> implementations easier.
53            *
54            * @param topic the name of the topic to created.
55            * @param numPartitions the number of partitions to create or null if replicasAssignments is set.
56            * @param replicationFactor the replication factor for the topic or null if replicaAssignments is set.
57            * @param replicasAssignments replica assignments or null if numPartitions and replicationFactor is set. The
58            *                            assignment is a map from partition id to replica (broker) ids.
59            * @param configs topic configs for the topic to be created, not including broker defaults. Broker configs are
60            *                passed via the {@code configure()} method of the policy implementation.
61            */
62           public RequestMetadata(String topic, Integer numPartitions, Short replicationFactor,
63                           Map<Integer, List<Integer>> replicasAssignments, Map<String, String> configs) {
64               this.topic = topic;
65               this.numPartitions = numPartitions;
66               this.replicationFactor = replicationFactor;
67               this.replicasAssignments = replicasAssignments == null ? null : Collections.unmodifiableMap(replicasAssignments);
68               this.configs = Collections.unmodifiableMap(configs);
69           }
70   
71           /**
72            * Return the name of the topic to create.
73            */
74           public String topic() {
75               return topic;
76           }
77   
78           /**
79            * Return the number of partitions to create or null if replicaAssignments is not null.
80            */
81           public Integer numPartitions() {
82               return numPartitions;
83           }
84   
85           /**
86            * Return the number of replicas to create or null if replicaAssignments is not null.
87            */
88           public Short replicationFactor() {
89               return replicationFactor;
90           }
91   
92           /**
93            * Return a map from partition id to replica (broker) ids or null if numPartitions and replicationFactor are
94            * set instead.
95            */
96           public Map<Integer, List<Integer>> replicasAssignments() {
97               return replicasAssignments;
98           }
99   
100           /**
101            * Return topic configs in the request, not including broker defaults. Broker configs are passed via
102            * the {@code configure()} method of the policy implementation.
103            */
104           public Map<String, String> configs() {
105               return configs;
106           }
107   
108           @Override
109           public int hashCode() {
110               return Objects.hash(topic, numPartitions, replicationFactor,
111                   replicasAssignments, configs);
112           }
113   
114           @Override
115           public boolean equals(Object o) {
116               if (this == o) return true;
117               if (o == null || getClass() != o.getClass()) return false;
118               RequestMetadata other = (RequestMetadata) o;
119               return topic.equals(other.topic) &&
120                   Objects.equals(numPartitions, other.numPartitions) &&
121                   Objects.equals(replicationFactor, other.replicationFactor) &&
122                   Objects.equals(replicasAssignments, other.replicasAssignments) &&
123                   configs.equals(other.configs);
124           }
125   
126           @Override
127           public String toString() {
128               return "CreateTopicPolicy.RequestMetadata(topic=" + topic +
129                       ", numPartitions=" + numPartitions +
130                       ", replicationFactor=" + replicationFactor +
131                       ", replicasAssignments=" + replicasAssignments +
132                       ", configs=" + configs + ")";
133           }
134       }
135   
136       /**
137        * Validate the request parameters and throw a <code>PolicyViolationException</code> with a suitable error
138        * message if the create topics request parameters for the provided topic do not satisfy this policy.
139        *
140        * Clients will receive the POLICY_VIOLATION error code along with the exception's message. Note that validation
141        * failure only affects the relevant topic, other topics in the request will still be processed.
142        *
143        * @param requestMetadata the create topics request parameters for the provided topic.
144        * @throws PolicyViolationException if the request parameters do not satisfy this policy.
145        */
146       void validate(RequestMetadata requestMetadata) throws PolicyViolationException;
147   }
