49   /**
50    * A class encapsulating some of the logic around metadata.
51    * <p>
52    * This class is shared by the client thread (for partitioning) and the background sender thread.
53    *
54    * Metadata is maintained for only a subset of topics, which can be added to over time. When we request metadata for a
55    * topic we don't have any metadata for it will trigger a metadata update.
56    * <p>
57    * If topic expiry is enabled for the metadata, any topic that has not been used within the expiry interval
58    * is removed from the metadata refresh set after an update. Consumers disable topic expiry since they explicitly
59    * manage topics while producers rely on topic expiry to limit the refresh set.
60    */
61   public class Metadata implements Closeable {
62       private final Logger log;
63       private final long refreshBackoffMs;
64       private final long metadataExpireMs;
65       private int updateVersion;  // bumped on every metadata response
66       private int requestVersion; // bumped on every new topic addition
67       private long lastRefreshMs;
68       private long lastSuccessfulRefreshMs;
69       private KafkaException fatalException;

75       private final ClusterResourceListeners clusterResourceListeners;
76       private boolean isClosed;
77       private final Map<TopicPartition, Integer> lastSeenLeaderEpochs;

115       /**
116        * Return the next time when the current cluster info can be updated (i.e., backoff time has elapsed).
117        *
118        * @param nowMs current time in ms
119        * @return remaining time in ms till the cluster info can be updated again
120        */
121       public synchronized long timeToAllowUpdate(long nowMs) {
122           return Math.max(this.lastRefreshMs + this.refreshBackoffMs - nowMs, 0);
123       }

125       /**
126        * The next time to update the cluster info is the maximum of the time the current info will expire and the time the
127        * current info can be updated (i.e. backoff time has elapsed); If an update has been request then the expiry time
128        * is now
129        *
130        * @param nowMs current time in ms
131        * @return remaining time in ms till updating the cluster info
132        */
133       public synchronized long timeToNextUpdate(long nowMs) {
134           long timeToExpire = updateRequested() ? 0 : Math.max(this.lastSuccessfulRefreshMs + this.metadataExpireMs - nowMs, 0);
135           return Math.max(timeToExpire, timeToAllowUpdate(nowMs));
136       }

158       /**
159        * Request an update for the partition metadata iff we have seen a newer leader epoch. This is called by the client
160        * any time it handles a response from the broker that includes leader epoch, except for UpdateMetadata which
161        * follows a different code path ({@link #update}).
162        *
163        * @param topicPartition
164        * @param leaderEpoch
165        * @return true if we updated the last seen epoch, false otherwise
166        */
167       public synchronized boolean updateLastSeenEpochIfNewer(TopicPartition topicPartition, int leaderEpoch) {
168           Objects.requireNonNull(topicPartition, "TopicPartition cannot be null");
169           if (leaderEpoch < 0)
170               throw new IllegalArgumentException("Invalid leader epoch " + leaderEpoch + " (must be non-negative)");
171   
172           Integer oldEpoch = lastSeenLeaderEpochs.get(topicPartition);
173           log.trace("Determining if we should replace existing epoch {} with new epoch {} for partition {}", oldEpoch, leaderEpoch, topicPartition);
174   
175           final boolean updated;
176           if (oldEpoch == null) {
177               log.debug("Not replacing null epoch with new epoch {} for partition {}", leaderEpoch, topicPartition);
178               updated = false;
179           } else if (leaderEpoch > oldEpoch) {
180               log.debug("Updating last seen epoch from {} to {} for partition {}", oldEpoch, leaderEpoch, topicPartition);
181               lastSeenLeaderEpochs.put(topicPartition, leaderEpoch);
182               updated = true;
183           } else {
184               log.debug("Not replacing existing epoch {} with new epoch {} for partition {}", oldEpoch, leaderEpoch, topicPartition);
185               updated = false;
186           }
187   
188           this.needFullUpdate = this.needFullUpdate || updated;
189           return updated;
190       }

206        * Return the cached partition info if it exists and a newer leader epoch isn't known about.
207        */
208       synchronized Optional<MetadataResponse.PartitionMetadata> partitionMetadataIfCurrent(TopicPartition topicPartition) {
209           Integer epoch = lastSeenLeaderEpochs.get(topicPartition);
210           Optional<MetadataResponse.PartitionMetadata> partitionMetadata = cache.partitionMetadata(topicPartition);

216                       metadata.leaderEpoch.orElse(NO_PARTITION_LEADER_EPOCH).equals(epoch));
217           }

227       public synchronized LeaderAndEpoch currentLeader(TopicPartition topicPartition) {
228           Optional<MetadataResponse.PartitionMetadata> maybeMetadata = partitionMetadataIfCurrent(topicPartition);
229           if (!maybeMetadata.isPresent())
230               return new LeaderAndEpoch(Optional.empty(), Optional.ofNullable(lastSeenLeaderEpochs.get(topicPartition)));
231   
232           MetadataResponse.PartitionMetadata partitionMetadata = maybeMetadata.get();
233           Optional<Integer> leaderEpochOpt = partitionMetadata.leaderEpoch;
234           Optional<Node> leaderNodeOpt = partitionMetadata.leaderId.flatMap(cache::nodeById);
235           return new LeaderAndEpoch(leaderNodeOpt, leaderEpochOpt);

253       /**
254        * Updates the cluster metadata. If topic expiry is enabled, expiry time
255        * is set for topics if required and expired topics are removed from the metadata.
256        *
257        * @param requestVersion The request version corresponding to the update response, as provided by
258        *     {@link #newMetadataRequestAndVersion(long)}.
259        * @param response metadata response received from the broker
260        * @param isPartialUpdate whether the metadata request was for a subset of the active topics
261        * @param nowMs current time in milliseconds
262        */
263       public synchronized void update(int requestVersion, MetadataResponse response, boolean isPartialUpdate, long nowMs) {
264           Objects.requireNonNull(response, "Metadata response cannot be null");
265           if (isClosed())
266               throw new IllegalStateException("Update requested after metadata close");
267   
268           this.needPartialUpdate = requestVersion < this.requestVersion;
269           this.lastRefreshMs = nowMs;
270           this.updateVersion += 1;
271           if (!isPartialUpdate) {
272               this.needFullUpdate = false;
273               this.lastSuccessfulRefreshMs = nowMs;
274           }
275   
276           String previousClusterId = cache.clusterResource().clusterId();
277   
278           this.cache = handleMetadataResponse(response, isPartialUpdate, nowMs);
279   
280           Cluster cluster = cache.cluster();
281           maybeSetMetadataError(cluster);
282   
283           this.lastSeenLeaderEpochs.keySet().removeIf(tp -> !retainTopic(tp.topic(), false, nowMs));
284   
285           String newClusterId = cache.clusterResource().clusterId();
286           if (!Objects.equals(previousClusterId, newClusterId)) {
287               log.info("Cluster ID: {}", newClusterId);
288           }
289           clusterResourceListeners.onUpdate(cache.clusterResource());
290   
291           log.debug("Updated cluster metadata updateVersion {} to {}", this.updateVersion, this.cache);
292       }

314       /**
315        * Transform a MetadataResponse into a new MetadataCache instance.
316        */
317       private MetadataCache handleMetadataResponse(MetadataResponse metadataResponse, boolean isPartialUpdate, long nowMs) {
318           // All encountered topics.
319           Set<String> topics = new HashSet<>();
320   
321           // Retained topics to be passed to the metadata cache.
322           Set<String> internalTopics = new HashSet<>();
323           Set<String> unauthorizedTopics = new HashSet<>();
324           Set<String> invalidTopics = new HashSet<>();
325   
326           List<MetadataResponse.PartitionMetadata> partitions = new ArrayList<>();
327           Map<String, Uuid> topicIds = new HashMap<>();
328           Map<String, Uuid> oldTopicIds = cache.topicIds();
329           for (MetadataResponse.TopicMetadata metadata : metadataResponse.topicMetadata()) {
330               String topicName = metadata.topic();
331               Uuid topicId = metadata.topicId();
332               topics.add(topicName);
333               // We can only reason about topic ID changes when both IDs are valid, so keep oldId null unless the new metadata contains a topic ID
334               Uuid oldTopicId = null;
335               if (!Uuid.ZERO_UUID.equals(topicId)) {
336                   topicIds.put(topicName, topicId);
337                   oldTopicId = oldTopicIds.get(topicName);
338               } else {
339                   topicId = null;
340               }
341   
342               if (!retainTopic(topicName, metadata.isInternal(), nowMs))
343                   continue;
344   
345               if (metadata.isInternal())
346                   internalTopics.add(topicName);
347   
348               if (metadata.error() == Errors.NONE) {
349                   for (MetadataResponse.PartitionMetadata partitionMetadata : metadata.partitionMetadata()) {
350                       // Even if the partition's metadata includes an error, we need to handle
351                       // the update to catch new epochs
352                       updateLatestMetadata(partitionMetadata, metadataResponse.hasReliableLeaderEpochs(), topicId, oldTopicId)
353                           .ifPresent(partitions::add);
354   
355                       if (partitionMetadata.error.exception() instanceof InvalidMetadataException) {
356                           log.debug("Requesting metadata update for partition {} due to error {}",
357                                   partitionMetadata.topicPartition, partitionMetadata.error);
358                           requestUpdate();
359                       }
360                   }
361               } else {
362                   if (metadata.error().exception() instanceof InvalidMetadataException) {
363                       log.debug("Requesting metadata update for topic {} due to error {}", topicName, metadata.error());
364                       requestUpdate();
365                   }
366   
367                   if (metadata.error() == Errors.INVALID_TOPIC_EXCEPTION)
368                       invalidTopics.add(topicName);
369                   else if (metadata.error() == Errors.TOPIC_AUTHORIZATION_FAILED)
370                       unauthorizedTopics.add(topicName);
371               }
372           }
373   
374           Map<Integer, Node> nodes = metadataResponse.brokersById();
375           if (isPartialUpdate)
376               return this.cache.mergeWith(metadataResponse.clusterId(), nodes, partitions,
377                   unauthorizedTopics, invalidTopics, internalTopics, metadataResponse.controller(), topicIds,
378                   (topic, isInternal) -> !topics.contains(topic) && retainTopic(topic, isInternal, nowMs));
379           else
380               return new MetadataCache(metadataResponse.clusterId(), nodes, partitions,
381                   unauthorizedTopics, invalidTopics, internalTopics, metadataResponse.controller(), topicIds);
382       }
383   
384       /**
385        * Compute the latest partition metadata to cache given ordering by leader epochs (if both
386        * available and reliable) and whether the topic ID changed.
387        */
388       private Optional<MetadataResponse.PartitionMetadata> updateLatestMetadata(
389               MetadataResponse.PartitionMetadata partitionMetadata,
390               boolean hasReliableLeaderEpoch,
391               Uuid topicId,
392               Uuid oldTopicId) {
393           TopicPartition tp = partitionMetadata.topicPartition;
394           if (hasReliableLeaderEpoch && partitionMetadata.leaderEpoch.isPresent()) {
395               int newEpoch = partitionMetadata.leaderEpoch.get();
396               Integer currentEpoch = lastSeenLeaderEpochs.get(tp);
397               if (topicId != null && !topicId.equals(oldTopicId)) {
398                   // If the new topic ID is valid and different from the last seen topic ID, update the metadata.
399                   // Between the time that a topic is deleted and re-created, the client may lose track of the
400                   // corresponding topicId (i.e. `oldTopicId` will be null). In this case, when we discover the new
401                   // topicId, we allow the corresponding leader epoch to override the last seen value.
402                   log.info("Resetting the last seen epoch of partition {} to {} since the associated topicId changed from {} to {}",
403                            tp, newEpoch, oldTopicId, topicId);
404                   lastSeenLeaderEpochs.put(tp, newEpoch);
405                   return Optional.of(partitionMetadata);
406               } else if (currentEpoch == null || newEpoch >= currentEpoch) {
407                   // If the received leader epoch is at least the same as the previous one, update the metadata
408                   log.debug("Updating last seen epoch for partition {} from {} to epoch {} from new metadata", tp, currentEpoch, newEpoch);
409                   lastSeenLeaderEpochs.put(tp, newEpoch);
410                   return Optional.of(partitionMetadata);
411               } else {
412                   // Otherwise ignore the new metadata and use the previously cached info
413                   log.debug("Got metadata for an older epoch {} (current is {}) for partition {}, not updating", newEpoch, currentEpoch, tp);
414                   return cache.partitionMetadata(tp);
415               }
416           } else {
417               // Handle old cluster formats as well as error responses where leader and epoch are missing
418               lastSeenLeaderEpochs.remove(tp);
419               return Optional.of(partitionMetadata.withoutLeaderEpoch());
420           }
421       }

423       /**
424        * If any non-retriable exceptions were encountered during metadata update, clear and throw the exception.
425        * This is used by the consumer to propagate any fatal exceptions or topic exceptions for any of the topics

456           fatalException = null;

462       // We may be able to recover from this exception if metadata for this topic is no longer needed
463       private KafkaException recoverableException() {
464           if (!unauthorizedTopics.isEmpty())
465               return new TopicAuthorizationException(unauthorizedTopics);
466           else if (!invalidTopics.isEmpty())
467               return new InvalidTopicException(invalidTopics);
468           else
469               return null;
470       }

474               return new TopicAuthorizationException(Collections.singleton(topic));
475           else if (invalidTopics.contains(topic))
476               return new InvalidTopicException(Collections.singleton(topic));
477           else
478               return null;
479       }
