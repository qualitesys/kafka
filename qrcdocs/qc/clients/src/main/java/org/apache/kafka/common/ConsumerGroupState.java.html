18   package org.apache.kafka.common;
19   
20   import java.util.Arrays;
21   import java.util.Map;
22   import java.util.function.Function;
23   import java.util.stream.Collectors;
24   
25   /**
26    * The consumer group state.
27    */
28   public enum ConsumerGroupState {
29       UNKNOWN("Unknown"),
30       PREPARING_REBALANCE("PreparingRebalance"),
31       COMPLETING_REBALANCE("CompletingRebalance"),
32       STABLE("Stable"),
33       DEAD("Dead"),
34       EMPTY("Empty");
35   
36       private final static Map<String, ConsumerGroupState> NAME_TO_ENUM = Arrays.stream(values())
37           .collect(Collectors.toMap(state -> state.name, Function.identity()));;
38   
39       private final String name;
40   
41       ConsumerGroupState(String name) {
42           this.name = name;
43       }
44   
45       /**
46        * Parse a string into a consumer group state.
47        */
48       public static ConsumerGroupState parse(String name) {
49           ConsumerGroupState state = NAME_TO_ENUM.get(name);
50           return state == null ? UNKNOWN : state;
51       }
52   
53       @Override
54       public String toString() {
55           return name;
56       }
57   }
