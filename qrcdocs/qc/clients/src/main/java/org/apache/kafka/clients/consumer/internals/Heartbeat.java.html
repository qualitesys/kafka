17   package org.apache.kafka.clients.consumer.internals;
18   
19   import org.apache.kafka.clients.GroupRebalanceConfig;
20   import org.apache.kafka.common.utils.LogContext;
21   import org.apache.kafka.common.utils.Time;
22   import org.apache.kafka.common.utils.Timer;
23   
24   import org.slf4j.Logger;
25   
26   /**
27    * A helper class for managing the heartbeat to the coordinator
28    */
29   public final class Heartbeat {
30       private final int maxPollIntervalMs;
31       private final GroupRebalanceConfig rebalanceConfig;
32       private final Time time;
33       private final Timer heartbeatTimer;
34       private final Timer sessionTimer;
35       private final Timer pollTimer;
36       private final Logger log;
37   
38       private volatile long lastHeartbeatSend = 0L;
39       private volatile boolean heartbeatInFlight = false;
40   
41       public Heartbeat(GroupRebalanceConfig config,
42                        Time time) {
43           if (config.heartbeatIntervalMs >= config.sessionTimeoutMs)
44               throw new IllegalArgumentException("Heartbeat must be set lower than the session timeout");
45           this.rebalanceConfig = config;
46           this.time = time;
47           this.heartbeatTimer = time.timer(config.heartbeatIntervalMs);
48           this.sessionTimer = time.timer(config.sessionTimeoutMs);
49           this.maxPollIntervalMs = config.rebalanceTimeoutMs;
50           this.pollTimer = time.timer(maxPollIntervalMs);
51   
52           final LogContext logContext = new LogContext("[Heartbeat groupID=" + config.groupId + "] ");
53           this.log = logContext.logger(getClass());
54       }
55   
56       private void update(long now) {
57           heartbeatTimer.update(now);
58           sessionTimer.update(now);
59           pollTimer.update(now);
60       }
61   
62       public void poll(long now) {
63           update(now);
64           pollTimer.reset(maxPollIntervalMs);
65       }
66   
67       boolean hasInflight() {
68           return heartbeatInFlight;
69       }
70   
71       void sentHeartbeat(long now) {
72           lastHeartbeatSend = now;
73           heartbeatInFlight = true;
74           update(now);
75           heartbeatTimer.reset(rebalanceConfig.heartbeatIntervalMs);
76   
77           if (log.isTraceEnabled()) {
78               log.trace("Sending heartbeat request with {}ms remaining on timer", heartbeatTimer.remainingMs());
79           }
80       }
81   
82       void failHeartbeat() {
83           update(time.milliseconds());
84           heartbeatInFlight = false;
85           heartbeatTimer.reset(rebalanceConfig.retryBackoffMs);
86   
87           log.trace("Heartbeat failed, reset the timer to {}ms remaining", heartbeatTimer.remainingMs());
88       }
89   
90       void receiveHeartbeat() {
91           update(time.milliseconds());
92           heartbeatInFlight = false;
93           sessionTimer.reset(rebalanceConfig.sessionTimeoutMs);
94       }
95   
96       boolean shouldHeartbeat(long now) {
97           update(now);
98           return heartbeatTimer.isExpired();
99       }
100       
101       long lastHeartbeatSend() {
102           return this.lastHeartbeatSend;
103       }
104   
105       long timeToNextHeartbeat(long now) {
106           update(now);
107           return heartbeatTimer.remainingMs();
108       }
109   
110       boolean sessionTimeoutExpired(long now) {
111           update(now);
112           return sessionTimer.isExpired();
113       }
114   
115       void resetTimeouts() {
116           update(time.milliseconds());
117           sessionTimer.reset(rebalanceConfig.sessionTimeoutMs);
118           pollTimer.reset(maxPollIntervalMs);
119           heartbeatTimer.reset(rebalanceConfig.heartbeatIntervalMs);
120       }
121   
122       void resetSessionTimeout() {
123           update(time.milliseconds());
124           sessionTimer.reset(rebalanceConfig.sessionTimeoutMs);
125       }
126   
127       boolean pollTimeoutExpired(long now) {
128           update(now);
129           return pollTimer.isExpired();
130       }
131   
132       long lastPollTime() {
133           return pollTimer.currentTimeMs();
134       }
135   }
