18   
19   import org.apache.kafka.common.utils.LogContext;
20   import org.apache.kafka.common.utils.Time;
21   import org.apache.kafka.common.utils.Timer;
22   import org.slf4j.Logger;
23   
24   import java.util.HashMap;
25   import java.util.Map;
26   import java.util.Optional;
27   import java.util.Set;
28   import java.util.stream.Collectors;
29   
30   public class CandidateState implements EpochState {
31       private final int localId;
32       private final int epoch;
33       private final int retries;
34       private final Map<Integer, State> voteStates = new HashMap<>();
35       private final Optional<LogOffsetMetadata> highWatermark;
36       private final int electionTimeoutMs;
37       private final Timer electionTimer;
38       private final Timer backoffTimer;
39       private final Logger log;
40   
41       /**
42        * The life time of a candidate state is the following:
43        *
44        *  1. Once started, it would keep record of the received votes.
45        *  2. If majority votes granted, it can then end its life and will be replaced by a leader state;
46        *  3. If majority votes rejected or election timed out, it would transit into a backing off phase;
47        *     after the backoff phase completes, it would end its left and be replaced by a new candidate state with bumped retry.
48        */
49       private boolean isBackingOff;
50   
51       protected CandidateState(
52           Time time,
53           int localId,
54           int epoch,
55           Set<Integer> voters,
56           Optional<LogOffsetMetadata> highWatermark,
57           int retries,
58           int electionTimeoutMs,
59           LogContext logContext
60       ) {
61           this.localId = localId;
62           this.epoch = epoch;
63           this.highWatermark = highWatermark;
64           this.retries = retries;
65           this.isBackingOff = false;
66           this.electionTimeoutMs = electionTimeoutMs;
67           this.electionTimer = time.timer(electionTimeoutMs);
68           this.backoffTimer = time.timer(0);
69           this.log = logContext.logger(CandidateState.class);
70   
71           for (Integer voterId : voters) {
72               voteStates.put(voterId, State.UNRECORDED);
73           }
74           voteStates.put(localId, State.GRANTED);
75       }
76   
77       public int localId() {
78           return localId;
79       }
80   
81       public int majoritySize() {
82           return voteStates.size() / 2 + 1;
83       }
84   
85       private long numGranted() {
86           return voteStates.values().stream().filter(state -> state == State.GRANTED).count();
87       }
88   
89       private long numUnrecorded() {
90           return voteStates.values().stream().filter(state -> state == State.UNRECORDED).count();
91       }
92   
93       /**
94        * Check if the candidate is backing off for the next election
95        */
96       public boolean isBackingOff() {
97           return isBackingOff;
98       }
99   
100       public int retries() {
101           return retries;
102       }
103   
104       /**
105        * Check whether we have received enough votes to conclude the election and become leader.
106        *
107        * @return true if at least a majority of nodes have granted the vote
108        */
109       public boolean isVoteGranted() {
110           return numGranted() >= majoritySize();
111       }
112   
113       /**
114        * Check if we have received enough rejections that it is no longer possible to reach a
115        * majority of grants.
116        *
117        * @return true if the vote is rejected, false if the vote is already or can still be granted
118        */
119       public boolean isVoteRejected() {
120           return numGranted() + numUnrecorded() < majoritySize();
121       }
122   
123       /**
124        * Record a granted vote from one of the voters.
125        *
126        * @param remoteNodeId The id of the voter
127        * @return true if the voter had not been previously recorded
128        * @throws IllegalArgumentException if the remote node is not a voter or if the vote had already been
129        *         rejected by this node
130        */
131       public boolean recordGrantedVote(int remoteNodeId) {
132           State state = voteStates.get(remoteNodeId);
133           if (state == null) {
134               throw new IllegalArgumentException("Attempt to grant vote to non-voter " + remoteNodeId);
135           } else if (state == State.REJECTED) {
136               throw new IllegalArgumentException("Attempt to grant vote from node " + remoteNodeId +
137                   " which previously rejected our request");
138           }
139           return voteStates.put(remoteNodeId, State.GRANTED) == State.UNRECORDED;
140       }
141   
142       /**
143        * Record a rejected vote from one of the voters.
144        *
145        * @param remoteNodeId The id of the voter
146        * @return true if the rejected vote had not been previously recorded
147        * @throws IllegalArgumentException if the remote node is not a voter or if the vote had already been
148        *         granted by this node
149        */
150       public boolean recordRejectedVote(int remoteNodeId) {
151           State state = voteStates.get(remoteNodeId);
152           if (state == null) {
153               throw new IllegalArgumentException("Attempt to reject vote to non-voter " + remoteNodeId);
154           } else if (state == State.GRANTED) {
155               throw new IllegalArgumentException("Attempt to reject vote from node " + remoteNodeId +
156                   " which previously granted our request");
157           }
158   
159           return voteStates.put(remoteNodeId, State.REJECTED) == State.UNRECORDED;
160       }
161   
162       /**
163        * Record the current election has failed since we've either received sufficient rejecting voters or election timed out
164        */
165       public void startBackingOff(long currentTimeMs, long backoffDurationMs) {
166           this.backoffTimer.update(currentTimeMs);
167           this.backoffTimer.reset(backoffDurationMs);
168           this.isBackingOff = true;
169       }
170   
171       /**
172        * Get the set of voters which have not been counted as granted or rejected yet.
173        *
174        * @return The set of unrecorded voters
175        */
176       public Set<Integer> unrecordedVoters() {
177           return votersInState(State.UNRECORDED);
178       }
179   
180       /**
181        * Get the set of voters that have granted our vote requests.
182        *
183        * @return The set of granting voters, which should always contain the ID of the candidate
184        */
185       public Set<Integer> grantingVoters() {
186           return votersInState(State.GRANTED);
187       }
188   
189       /**
190        * Get the set of voters that have rejected our candidacy.
191        *
192        * @return The set of rejecting voters
193        */
194       public Set<Integer> rejectingVoters() {
195           return votersInState(State.REJECTED);
196       }
197   
198       private Set<Integer> votersInState(State state) {
199           return voteStates.entrySet().stream()
200               .filter(entry -> entry.getValue() == state)
201               .map(Map.Entry::getKey)
202               .collect(Collectors.toSet());
203       }
204   
205       public boolean hasElectionTimeoutExpired(long currentTimeMs) {
206           electionTimer.update(currentTimeMs);
207           return electionTimer.isExpired();
208       }
209   
210       public boolean isBackoffComplete(long currentTimeMs) {
211           backoffTimer.update(currentTimeMs);
212           return backoffTimer.isExpired();
213       }
214   
215       public long remainingBackoffMs(long currentTimeMs) {
216           if (!isBackingOff) {
217               throw new IllegalStateException("Candidate is not currently backing off");
218           }
219           backoffTimer.update(currentTimeMs);
220           return backoffTimer.remainingMs();
221       }
222   
223       public long remainingElectionTimeMs(long currentTimeMs) {
224           electionTimer.update(currentTimeMs);
225           return electionTimer.remainingMs();
226       }

228       @Override
229       public ElectionState election() {
230           return ElectionState.withVotedCandidate(epoch, localId, voteStates.keySet());
231       }
232   
233       @Override
234       public int epoch() {
235           return epoch;
236       }
237   
238       @Override
239       public Optional<LogOffsetMetadata> highWatermark() {
240           return highWatermark;
241       }
242   
243       @Override
244       public boolean canGrantVote(int candidateId, boolean isLogUpToDate) {
245           // Still reject vote request even candidateId = localId, Although the candidate votes for
246           // itself, this vote is implicit and not "granted".
247           log.debug("Rejecting vote request from candidate {} since we are already candidate in epoch {}",
248               candidateId, epoch);
249           return false;
250       }
251   
252       @Override
253       public String toString() {
254           return "CandidateState(" +
255               "localId=" + localId +
256               ", epoch=" + epoch +
257               ", retries=" + retries +
258               ", electionTimeoutMs=" + electionTimeoutMs +
259               ')';
260       }
261   
262       @Override
263       public String name() {
264           return "Candidate";
265       }
266   
267       @Override
268       public void close() {}
269   
270       private enum State {
271           UNRECORDED,
272           GRANTED,
273           REJECTED
274       }
275   }
