18   
19   import org.apache.kafka.common.TopicPartition;
20   import org.apache.kafka.common.Uuid;
21   import org.apache.kafka.common.record.Records;
22   import org.apache.kafka.snapshot.RawSnapshotReader;
23   import org.apache.kafka.snapshot.RawSnapshotWriter;
24   
25   import java.util.Optional;
26   
27   public interface ReplicatedLog extends AutoCloseable {
28   
29       /**
30        * Write a set of records to the local leader log. These messages will either
31        * be written atomically in a single batch or the call will fail and raise an
32        * exception.
33        *
34        * @return the metadata information of the appended batch
35        * @throws IllegalArgumentException if the record set is empty
36        * @throws RuntimeException if the batch base offset doesn't match the log end offset
37        */
38       LogAppendInfo appendAsLeader(Records records, int epoch);
39   
40       /**
41        * Append a set of records that were replicated from the leader. The main
42        * difference from appendAsLeader is that we do not need to assign the epoch
43        * or do additional validation.
44        *
45        * @return the metadata information of the appended batch
46        * @throws IllegalArgumentException if the record set is empty
47        * @throws RuntimeException if the batch base offset doesn't match the log end offset
48        */
49       LogAppendInfo appendAsFollower(Records records);
50   
51       /**
52        * Read a set of records within a range of offsets.
53        */
54       LogFetchInfo read(long startOffsetInclusive, Isolation isolation);
55   
56       /**
57        * Return the latest epoch. For an empty log, the latest epoch is defined
58        * as 0. We refer to this as the "primordial epoch" and it is never allowed
59        * to have a leader or any records associated with it (leader epochs always start
60        * from 1). Basically this just saves us the trouble of having to use `Option`
61        * all over the place.
62        */
63       int lastFetchedEpoch();
64   
65       /**
66        * Validate the given offset and epoch against the log and oldest snapshot.
67        *
68        * Returns the largest valid offset and epoch given `offset` and `epoch` as the upper bound.
69        * This can result in three possible values returned:
70        *
71        * 1. ValidOffsetAndEpoch.valid if the given offset and epoch is valid in the log.
72        *
73        * 2. ValidOffsetAndEpoch.diverging if the given offset and epoch is not valid; and the
74        * largest valid offset and epoch is in the log.
75        *
76        * 3. ValidOffsetAndEpoch.snapshot if the given offset and epoch is not valid; and the largest
77        * valid offset and epoch is less than the oldest snapshot.
78        *
79        * @param offset the offset to validate
80        * @param epoch the epoch of the record at offset - 1
81        * @return the largest valid offset and epoch
82        */
83       default ValidOffsetAndEpoch validateOffsetAndEpoch(long offset, int epoch) {
84           if (startOffset() == 0 && offset == 0) {
85               return ValidOffsetAndEpoch.valid(new OffsetAndEpoch(0, 0));
86           }
87   
88           Optional<OffsetAndEpoch> earliestSnapshotId = earliestSnapshotId();
89           if (earliestSnapshotId.isPresent() &&
90               ((offset < startOffset()) ||
91                (offset == startOffset() && epoch != earliestSnapshotId.get().epoch) ||
92                (epoch < earliestSnapshotId.get().epoch))
93           ) {
94               /* Send a snapshot if the leader has a snapshot at the log start offset and
95                * 1. the fetch offset is less than the log start offset or
96                * 2. the fetch offset is equal to the log start offset and last fetch epoch doesn't match
97                *    the oldest snapshot or
98                * 3. last fetch epoch is less than the oldest snapshot's epoch
99                */
100               OffsetAndEpoch latestSnapshotId = latestSnapshotId().orElseThrow(() -> new IllegalStateException(
101                   String.format(
102                       "Log start offset (%s) is greater than zero but latest snapshot was not found",
103                       startOffset()
104                   )
105               ));
106   
107               return ValidOffsetAndEpoch.snapshot(latestSnapshotId);
108           } else {
109               OffsetAndEpoch endOffsetAndEpoch = endOffsetForEpoch(epoch);
110   
111               if (endOffsetAndEpoch.epoch != epoch || endOffsetAndEpoch.offset < offset) {
112                   return ValidOffsetAndEpoch.diverging(endOffsetAndEpoch);
113               } else {
114                   return ValidOffsetAndEpoch.valid(new OffsetAndEpoch(offset, epoch));
115               }
116           }
117       }
118   
119       /**
120        * Find the first epoch less than or equal to the given epoch and its end offset.
121        */
122       OffsetAndEpoch endOffsetForEpoch(int epoch);
123   
124       /**
125        * Get the current log end offset metadata. This is always one plus the offset of the last
126        * written record. When the log is empty, the end offset is equal to the start offset.
127        */
128       LogOffsetMetadata endOffset();
129   
130       /**
131        * Get the high watermark.
132        */
133       LogOffsetMetadata highWatermark();
134   
135       /**
136        * Get the current log start offset. This is the offset of the first written
137        * entry, if one exists, or the end offset otherwise.
138        */
139       long startOffset();
140   
141       /**
142        * Initialize a new leader epoch beginning at the current log end offset. This API is invoked
143        * after becoming a leader and ensures that we can always determine the end offset and epoch
144        * with {@link #endOffsetForEpoch(int)} for any previous epoch.
145        *
146        * @param epoch Epoch of the newly elected leader
147        */
148       void initializeLeaderEpoch(int epoch);
149   
150       /**
151        * Truncate the log to the given offset. All records with offsets greater than or equal to
152        * the given offset will be removed.
153        *
154        * @param offset The offset to truncate to
155        */
156       void truncateTo(long offset);
157   
158       /**
159        * Fully truncate the log if the latest snapshot is later than the log end offset.
160        *
161        * In general this operation empties the log and sets the log start offset, high watermark and
162        * log end offset to the latest snapshot's end offset.
163        *
164        * @return true when the log is fully truncated, otherwise returns false
165        */
166       boolean truncateToLatestSnapshot();
167   
168       /**
169        * Update the high watermark and associated metadata (which is used to avoid
170        * index lookups when handling reads with {@link #read(long, Isolation)} with
171        * the {@link Isolation#COMMITTED} isolation level.
172        *
173        * @param offsetMetadata The offset and optional metadata
174        */
175       void updateHighWatermark(LogOffsetMetadata offsetMetadata);
176   
177       /**
178        * Delete all snapshots prior to the given snapshot
179        *
180        * The replicated log's start offset can be increased and older segments can be deleted when
181        * there is a snapshot greater than the current log start offset.
182        */
183       boolean deleteBeforeSnapshot(OffsetAndEpoch snapshotId);
184   
185       /**
186        * Flush the current log to disk.
187        *
188        * @param forceFlushActiveSegment Whether to force flush the active segment. Should be `true` during close; otherwise false.
189        */
190       void flush(boolean forceFlushActiveSegment);
191   
192       /**
193        * Possibly perform cleaning of snapshots and logs
194        */
195       boolean maybeClean();
196   
197       /**
198        * Get the last offset which has been flushed to disk.
199        */
200       long lastFlushedOffset();
201   
202       /**
203        * Return the topic partition associated with the log.
204        */
205       TopicPartition topicPartition();
206   
207       /**
208        * Return the topic ID associated with the log.
209        */
210       Uuid topicId();
211   
212       /**
213        * Truncate to an offset and epoch.
214        *
215        * @param endOffset offset and epoch to truncate to
216        * @return the truncation offset
217        */
218       default long truncateToEndOffset(OffsetAndEpoch endOffset) {
219           final long truncationOffset;
220           int leaderEpoch = endOffset.epoch;
221           if (leaderEpoch == 0) {
222               truncationOffset = Math.min(endOffset.offset, endOffset().offset);
223           } else {
224               OffsetAndEpoch localEndOffset = endOffsetForEpoch(leaderEpoch);
225               if (localEndOffset.epoch == leaderEpoch) {
226                   truncationOffset = Math.min(localEndOffset.offset, endOffset.offset);
227               } else {
228                   truncationOffset = localEndOffset.offset;
229               }
230           }
231   
232           truncateTo(truncationOffset);
233           return truncationOffset;
234       }
235   
236       /**
237        * Create a writable snapshot for the given snapshot id.
238        *
239        * See {@link RawSnapshotWriter} for details on how to use this object. The caller of
240        * this method is responsible for invoking {@link RawSnapshotWriter#close()}. If a
241        * snapshot already exists or it is less than log start offset then return an
242        * {@link Optional#empty()}.
243        *
244        * Snapshots created using this method will be validated against the existing snapshots
245        * and the replicated log.
246        *
247        * @param snapshotId the end offset and epoch that identifies the snapshot
248        * @return a writable snapshot if it doesn't already exists and greater than the log start
249        *         offset
250        * @throws IllegalArgumentException if validate is true and end offset is greater than the
251        *         high-watermark
252        */
253       Optional<RawSnapshotWriter> createNewSnapshot(OffsetAndEpoch snapshotId);
254   
255       /**
256        * Create a writable snapshot for the given snapshot id.
257        *
258        * See {@link RawSnapshotWriter} for details on how to use this object. The caller of
259        * this method is responsible for invoking {@link RawSnapshotWriter#close()}. If a
260        * snapshot already exists then return an {@link Optional#empty()}.
261        *
262        * Snapshots created using this method will not be validated against the existing snapshots
263        * and the replicated log. This is useful when creating snapshot from a trusted source like
264        * the quorum leader.
265        *
266        * @param snapshotId the end offset and epoch that identifies the snapshot
267        * @return a writable snapshot if it doesn't already exists
268        */
269       Optional<RawSnapshotWriter> storeSnapshot(OffsetAndEpoch snapshotId);
270   
271       /**
272        * Opens a readable snapshot for the given snapshot id.
273        *
274        * Returns an Optional with a readable snapshot, if the snapshot exists, otherwise
275        * returns an empty Optional. See {@link RawSnapshotReader} for details on how to
276        * use this object.
277        *
278        * @param snapshotId the end offset and epoch that identifies the snapshot
279        * @return an Optional with a readable snapshot, if the snapshot exists, otherwise
280        *         returns an empty Optional
281        */
282       Optional<RawSnapshotReader> readSnapshot(OffsetAndEpoch snapshotId);
283   
284       /**
285        * Returns the latest readable snapshot if one exists.
286        *
287        * @return an Optional with the latest readable snapshot, if one exists, otherwise
288        *         returns an empty Optional
289        */
290       Optional<RawSnapshotReader> latestSnapshot();
291   
292        /**
293         * Returns the latest snapshot id if one exists.
294         *
295         * @return an Optional snapshot id of the latest snashot if one exists, otherwise returns an
296         *         empty Optional
297         */
298       Optional<OffsetAndEpoch> latestSnapshotId();
299   
300       /**
301        * Returns the snapshot id at the log start offset.
302        *
303        * If the log start offset is nonzero then it is expected that there is a snapshot with an end
304        * offset equal to the start offset.
305        *
306        * @return an Optional snapshot id at the log start offset if nonzero, otherwise returns an empty
307        *         Optional
308        */
309       Optional<OffsetAndEpoch> earliestSnapshotId();
310   
311       /**
312        * Notifies the replicated log when a new snapshot is available.
313        */
314       void onSnapshotFrozen(OffsetAndEpoch snapshotId);
315   
316       default void close() {}
317   }
