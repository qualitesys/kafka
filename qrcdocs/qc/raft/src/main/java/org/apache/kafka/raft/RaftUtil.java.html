41       public static ApiMessage errorResponse(ApiKeys apiKey, Errors error) {
42           switch (apiKey) {
43               case VOTE:
44                   return new VoteResponseData().setErrorCode(error.code());
45               case BEGIN_QUORUM_EPOCH:
46                   return new BeginQuorumEpochResponseData().setErrorCode(error.code());
47               case END_QUORUM_EPOCH:
48                   return new EndQuorumEpochResponseData().setErrorCode(error.code());
49               case FETCH:
50                   return new FetchResponseData().setErrorCode(error.code());
51               default:
52                   throw new IllegalArgumentException("Received response for unexpected request type: " + apiKey);
53           }
54       }

56       public static FetchRequestData singletonFetchRequest(
57           TopicPartition topicPartition,
58           Uuid topicId,
59           Consumer<FetchRequestData.FetchPartition> partitionConsumer
60       ) {
61           FetchRequestData.FetchPartition fetchPartition =
62               new FetchRequestData.FetchPartition()
63                   .setPartition(topicPartition.partition());
64           partitionConsumer.accept(fetchPartition);
65   
66           FetchRequestData.FetchTopic fetchTopic =
67               new FetchRequestData.FetchTopic()
68                   .setTopic(topicPartition.topic())
69                   .setTopicId(topicId)
70                   .setPartitions(singletonList(fetchPartition));
71   
72           return new FetchRequestData()
73               .setTopics(singletonList(fetchTopic));
74       }

76       public static FetchResponseData singletonFetchResponse(
77           TopicPartition topicPartition,
78           Uuid topicId,
79           Errors topLevelError,
80           Consumer<FetchResponseData.PartitionData> partitionConsumer
81       ) {
82           FetchResponseData.PartitionData fetchablePartition =
83               new FetchResponseData.PartitionData();
84   
85           fetchablePartition.setPartitionIndex(topicPartition.partition());
86   
87           partitionConsumer.accept(fetchablePartition);
88   
89           FetchResponseData.FetchableTopicResponse fetchableTopic =
90               new FetchResponseData.FetchableTopicResponse()
91                   .setTopic(topicPartition.topic())
92                   .setTopicId(topicId)
93                   .setPartitions(Collections.singletonList(fetchablePartition));
94   
95           return new FetchResponseData()
96               .setErrorCode(topLevelError.code())
97               .setResponses(Collections.singletonList(fetchableTopic));
98       }

100       static boolean hasValidTopicPartition(FetchRequestData data, TopicPartition topicPartition, Uuid topicId) {
101           return data.topics().size() == 1 &&
102               data.topics().get(0).topicId().equals(topicId) &&
103               data.topics().get(0).partitions().size() == 1 &&
104               data.topics().get(0).partitions().get(0).partition() == topicPartition.partition();
105       }
