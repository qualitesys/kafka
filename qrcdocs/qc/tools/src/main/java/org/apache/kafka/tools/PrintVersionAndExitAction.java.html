18   
19   import net.sourceforge.argparse4j.inf.Argument;
20   import net.sourceforge.argparse4j.inf.ArgumentAction;
21   import net.sourceforge.argparse4j.inf.ArgumentParser;
22   import org.apache.kafka.common.utils.AppInfoParser;
23   import org.apache.kafka.common.utils.Exit;
24   
25   import java.util.Map;
26   
27   class PrintVersionAndExitAction implements ArgumentAction {
28   
29       @Override
30       public void run(
31           ArgumentParser parser,
32           Argument arg,
33           Map<String, Object> attrs,
34           String flag,
35           Object value
36       ) {
37           String version = AppInfoParser.getVersion();
38           String commitId = AppInfoParser.getCommitId();
39           System.out.println(version + " (Commit:" + commitId + ")");
40           Exit.exit(0);
41       }
42   
43       @Override
44       public void onAttach(Argument arg) {
45   
46       }
47   
48       @Override
49       public boolean consumeArgument() {
50           return false;
51       }
52   }
