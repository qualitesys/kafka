66   /**
67    * Command line consumer designed for system testing. It outputs consumer events to STDOUT as JSON
68    * formatted objects. The "name" field in each JSON event identifies the event type. The following
69    * events are currently supported:
70    *
71    * <ul>
72    * <li>partitions_revoked: outputs the partitions revoked through {@link ConsumerRebalanceListener#onPartitionsRevoked(Collection)}.
73    *     See {@link org.apache.kafka.tools.VerifiableConsumer.PartitionsRevoked}</li>
74    * <li>partitions_assigned: outputs the partitions assigned through {@link ConsumerRebalanceListener#onPartitionsAssigned(Collection)}
75    *     See {@link org.apache.kafka.tools.VerifiableConsumer.PartitionsAssigned}.</li>
76    * <li>records_consumed: contains a summary of records consumed in a single call to {@link KafkaConsumer#poll(Duration)}.
77    *     See {@link org.apache.kafka.tools.VerifiableConsumer.RecordsConsumed}.</li>
78    * <li>record_data: contains the key, value, and offset of an individual consumed record (only included if verbose
79    *     output is enabled). See {@link org.apache.kafka.tools.VerifiableConsumer.RecordData}.</li>
80    * <li>offsets_committed: The result of every offset commit (only included if auto-commit is not enabled).
81    *     See {@link org.apache.kafka.tools.VerifiableConsumer.OffsetsCommitted}</li>
82    * <li>shutdown_complete: emitted after the consumer returns from {@link KafkaConsumer#close()}.
83    *     See {@link org.apache.kafka.tools.VerifiableConsumer.ShutdownComplete}.</li>
84    * </ul>
85    */
86   public class VerifiableConsumer implements Closeable, OffsetCommitCallback, ConsumerRebalanceListener {

133       private boolean hasMessageLimit() {
134           return maxMessages >= 0;
135       }
136   
137       private boolean isFinished() {
138           return hasMessageLimit() && consumedMessages >= maxMessages;
139       }

141       private Map<TopicPartition, OffsetAndMetadata> onRecordsReceived(ConsumerRecords<String, String> records) {
142           Map<TopicPartition, OffsetAndMetadata> offsets = new HashMap<>();
143   
144           List<RecordSetSummary> summaries = new ArrayList<>();
145           for (TopicPartition tp : records.partitions()) {
146               List<ConsumerRecord<String, String>> partitionRecords = records.records(tp);
147   
148               if (hasMessageLimit() && consumedMessages + partitionRecords.size() > maxMessages)
149                   partitionRecords = partitionRecords.subList(0, maxMessages - consumedMessages);
150   
151               if (partitionRecords.isEmpty())
152                   continue;
153   
154               long minOffset = partitionRecords.get(0).offset();
155               long maxOffset = partitionRecords.get(partitionRecords.size() - 1).offset();
156   
157               offsets.put(tp, new OffsetAndMetadata(maxOffset + 1));
158               summaries.add(new RecordSetSummary(tp.topic(), tp.partition(),
159                       partitionRecords.size(), minOffset, maxOffset));
160   
161               if (verbose) {
162                   for (ConsumerRecord<String, String> record : partitionRecords) {
163                       printJson(new RecordData(record));
164                   }
165               }
166   
167               consumedMessages += partitionRecords.size();
168               if (isFinished())
169                   break;
170           }
171   
172           printJson(new RecordsConsumed(records.count(), summaries));
173           return offsets;
174       }

176       @Override
177       public void onComplete(Map<TopicPartition, OffsetAndMetadata> offsets, Exception exception) {
178           List<CommitData> committedOffsets = new ArrayList<>();
179           for (Map.Entry<TopicPartition, OffsetAndMetadata> offsetEntry : offsets.entrySet()) {
180               TopicPartition tp = offsetEntry.getKey();
181               committedOffsets.add(new CommitData(tp.topic(), tp.partition(), offsetEntry.getValue().offset()));
182           }
183   
184           boolean success = true;
185           String error = null;
186           if (exception != null) {
187               success = false;
188               error = exception.getMessage();
189           }
190           printJson(new OffsetsCommitted(committedOffsets, error, success));
191       }

211       public void commitSync(Map<TopicPartition, OffsetAndMetadata> offsets) {
212           try {
213               consumer.commitSync(offsets);
214               onComplete(offsets, null);
215           } catch (WakeupException e) {
216               // we only call wakeup() once to close the consumer, so this recursion should be safe
217               commitSync(offsets);
218               throw e;
219           } catch (FencedInstanceIdException e) {
220               throw e;
221           } catch (Exception e) {
222               onComplete(offsets, e);
223           }
224       }

226       public void run() {
227           try {
228               printJson(new StartupComplete());
229               consumer.subscribe(Collections.singletonList(topic), this);
230   
231               while (!isFinished()) {
232                   ConsumerRecords<String, String> records = consumer.poll(Duration.ofMillis(Long.MAX_VALUE));
233                   Map<TopicPartition, OffsetAndMetadata> offsets = onRecordsReceived(records);
234   
235                   if (!useAutoCommit) {
236                       if (useAsyncCommit)
237                           consumer.commitAsync(offsets, this);
238                       else
239                           commitSync(offsets);
240                   }
241               }
242           } catch (WakeupException e) {
243               // ignore, we are closing
244               log.trace("Caught WakeupException because consumer is shutdown, ignore and terminate.", e);
245           } catch (Throwable t) {
246               // Log the error so it goes to the service log and not stdout
247               log.error("Error during processing, terminating consumer process: ", t);
248           } finally {
249               consumer.close();
250               printJson(new ShutdownComplete());
251               shutdownLatch.countDown();
252           }
253       }
254   
255       public void close() {
256           boolean interrupted = false;
257           try {
258               consumer.wakeup();
259               while (true) {
260                   try {
261                       shutdownLatch.await();
262                       return;
263                   } catch (InterruptedException e) {
264                       interrupted = true;
265                   }
266               }
267           } finally {
268               if (interrupted)
269                   Thread.currentThread().interrupt();
270           }
271       }

605       public static VerifiableConsumer createFromArgs(ArgumentParser parser, String[] args) throws ArgumentParserException {
606           Namespace res = parser.parseArgs(args);
607   
608           boolean useAutoCommit = res.getBoolean("useAutoCommit");
609           String configFile = res.getString("consumer.config");
610           String brokerHostandPort = null;
611   
612           Properties consumerProps = new Properties();
613           if (configFile != null) {
614               try {
615                   consumerProps.putAll(Utils.loadProps(configFile));
616               } catch (IOException e) {
617                   throw new ArgumentParserException(e.getMessage(), parser);
618               }
619           }
620   
621           consumerProps.put(ConsumerConfig.GROUP_ID_CONFIG, res.getString("groupId"));
622   
623           String groupInstanceId = res.getString("groupInstanceId");
624           if (groupInstanceId != null) {
625               consumerProps.put(ConsumerConfig.GROUP_INSTANCE_ID_CONFIG, groupInstanceId);
626           }
627   
628   
629           if (res.get("bootstrapServer") != null) {
630               brokerHostandPort = res.getString("bootstrapServer");
631           } else if (res.getString("brokerList") != null) {
632               brokerHostandPort = res.getString("brokerList");
633           } else {
634               parser.printHelp();
635               // Can't use `Exit.exit` here because it didn't exist until 0.11.0.0.
636               System.exit(0);
637           }
638           consumerProps.put(ConsumerConfig.BOOTSTRAP_SERVERS_CONFIG, brokerHostandPort);
639   
640           consumerProps.put(ConsumerConfig.ENABLE_AUTO_COMMIT_CONFIG, useAutoCommit);
641           consumerProps.put(ConsumerConfig.AUTO_OFFSET_RESET_CONFIG, res.getString("resetPolicy"));
642           consumerProps.put(ConsumerConfig.SESSION_TIMEOUT_MS_CONFIG, Integer.toString(res.getInt("sessionTimeout")));
643           consumerProps.put(ConsumerConfig.PARTITION_ASSIGNMENT_STRATEGY_CONFIG, res.getString("assignmentStrategy"));
644   
645           StringDeserializer deserializer = new StringDeserializer();
646           KafkaConsumer<String, String> consumer = new KafkaConsumer<>(consumerProps, deserializer, deserializer);
647   
648           String topic = res.getString("topic");
649           int maxMessages = res.getInt("maxMessages");
650           boolean verbose = res.getBoolean("verbose");
651   
652           return new VerifiableConsumer(
653                   consumer,
654                   System.out,
655                   topic,
656                   maxMessages,
657                   useAutoCommit,
658                   false,
659                   verbose);
660       }

662       public static void main(String[] args) {
663           ArgumentParser parser = argParser();
664           if (args.length == 0) {
665               parser.printHelp();
666               // Can't use `Exit.exit` here because it didn't exist until 0.11.0.0.
667               System.exit(0);
668           }
669           try {
670               final VerifiableConsumer consumer = createFromArgs(parser, args);
671               // Can't use `Exit.addShutdownHook` here because it didn't exist until 2.5.0.
672               Runtime.getRuntime().addShutdownHook(new Thread(consumer::close, "verifiable-consumer-shutdown-hook"));
673   
674               consumer.run();
675           } catch (ArgumentParserException e) {
676               parser.handleError(e);
677               // Can't use `Exit.exit` here because it didn't exist until 0.11.0.0.
678               System.exit(1);
679           }
680       }
